97 RingBufferImpl() noexcept
98 : m_size(0)
99 , m_read_idx(0)
100 , m_write_idx(0)

107 // Check for lock-free types. atomic_occupied_t can easily be changed to an integer type that supports lock-free
108 // atomic operations.
109#if defined(__cpp_lib_atomic_is_always_lock_free)
110 static_assert(atomic_index_t::is_always_lock_free, "Expecting lock-free types");
111 static_assert(atomic_occupied_t::is_always_lock_free, "Expecting lock-free types");

119 ~RingBufferImpl()
120 {
121 // Instead of just calling pop() until we're empty, we do it manually to save some atomic operations.
122 index_t remaining = size();
123 index_t idx = m_read_idx.load(std::memory_order_relaxed);
124 while (remaining > 0) {
125 const auto occupied_status = m_nodes[idx].m_occupied.load(std::memory_order_relaxed);
126 MNRY_ASSERT(occupied_status != occupied_t::IN_TRANSITION);
127 if (occupied_status == occupied_t::OCCUPIED) {
128 get_pointer(idx)->~T();
129 --remaining;
130 }
131 idx = increment(idx);
132 }
133 for (index_t i = 0; i < k_capacity; ++i) {
134 (m_nodes + i)->~Node();
135 }
136 scene_rdl2::util::alignedFree(m_nodes);
137 }

199 template <typename... Args>
200 void emplace(Args&&... args)
201 {
202 constexpr bool no_except_construct = noexcept(T(std::forward<Args>(args)...));
203 // cppcheck-suppress syntaxError
204 IF_CONSTEXPR (k_lock_traits == LockTraits::SPIN_LOCK && !no_except_construct) {
205 // We don't have a way to check which constructor is called by emplace for our class static exception
206 // checks used to determine our lock traits. We only check exception safety in move construction and move
207 // assignment when we check our top-level traits. Check here, and if it can throw, create our type before we
208 // call push in an effort to use the move operations.
209 push_impl(T(std::forward<Args>(args)...));
210 } else {
211 push_impl(std::forward<Args>(args)...);
212 }
213 }
214
215 // This method does not meet the strong exception guarantee.
216 // If T's move constructor on return throws, we've lost that data forever.
217 NO_DISCARD bool try_pop(T& ret)
218 {
219 IF_CONSTEXPR (Traits::consumer_traits == ConsumerTraits::SINGLE_CONSUMER &&
220 k_lock_traits == LockTraits::SPIN_LOCK) {
221 return try_pop_single_impl<PopImplementation::MOVE>(ret);
222 } else IF_CONSTEXPR (Traits::consumer_traits == ConsumerTraits::SINGLE_CONSUMER &&
223 k_lock_traits == LockTraits::LOCK_FREE) {
224 return try_pop_single_lock_free_impl<PopImplementation::MOVE>(ret);
225 } else IF_CONSTEXPR (Traits::consumer_traits == ConsumerTraits::MULTIPLE_CONSUMERS &&
226 k_lock_traits == LockTraits::SPIN_LOCK) {
227 return try_pop_impl<PopImplementation::MOVE>(ret);
228 } else {
229 return try_pop_lock_free_impl<PopImplementation::MOVE>(ret);
230 }
231 }
232
233 // This method does not meet the strong exception guarantee.
234 // If T's move constructor on return throws, we've lost that data forever.
235 T pop()
236 {
237 // We don't want to make a default constructor a requirement of T, so we just allocate stack space for it, and
238 // tell our pop function to do placement new for a copy constructor.
239 storage_t storage;
240 T* const p = reinterpret_cast<T*>(std::addressof(storage));
241 do_pop_dispatch<PopImplementation::PLACEMENT>(*p);
242 T ret{ std::move(*p) };
243 p->~T();
244 return ret;
245 }
246
247 // This method does not meet the strong exception guarantee.
248 // If T's move constructor on return throws, we've lost that data forever.
249 void pop(T& ret)
250 {
251 do_pop_dispatch<PopImplementation::MOVE>(ret);
252 }

268 template <PopImplementation pop_implementation>
269 void do_pop_dispatch(T& ret)
270 {
271 bool popped;
272 do {
273 IF_CONSTEXPR (Traits::consumer_traits == ConsumerTraits::SINGLE_CONSUMER &&
274 k_lock_traits == LockTraits::SPIN_LOCK) {
275 popped = try_pop_single_impl<pop_implementation>(ret);
276 } else IF_CONSTEXPR (Traits::consumer_traits == ConsumerTraits::SINGLE_CONSUMER &&
277 k_lock_traits == LockTraits::LOCK_FREE) {
278 popped = try_pop_single_lock_free_impl<pop_implementation>(ret);
279 } else IF_CONSTEXPR (Traits::consumer_traits == ConsumerTraits::MULTIPLE_CONSUMERS &&
280 k_lock_traits == LockTraits::SPIN_LOCK) {
281 popped = try_pop_impl<pop_implementation>(ret);
282 } else {
283 popped = try_pop_lock_free_impl<pop_implementation>(ret);
284 }
285 if (!popped) {
286 wait(m_size, 0u);
287 }
288 } while (!popped);
289 }

291 template <typename... Args>
292 void push_impl(Args&&... args)
293 {
294 bool pushed;
295 do {
296 // Does it make you nervous to see std::forward in a loop? I understand. The contents, however, will not be
297 // consumed unless the push succeeds (i.e. there will be no moves until success).
298 IF_CONSTEXPR (Traits::producer_traits == ProducerTraits::SINGLE_PRODUCER &&
299 k_lock_traits == LockTraits::SPIN_LOCK) {
300 pushed = try_push_single_impl(std::forward<Args>(args)...);
301 } else IF_CONSTEXPR (Traits::producer_traits == ProducerTraits::SINGLE_PRODUCER &&
302 k_lock_traits == LockTraits::LOCK_FREE) {
303 pushed = try_push_single_lock_free_impl(std::forward<Args>(args)...);
304 } else IF_CONSTEXPR (Traits::producer_traits == ProducerTraits::MULTIPLE_PRODUCERS &&
305 k_lock_traits == LockTraits::SPIN_LOCK) {
306 pushed = try_push_impl(std::forward<Args>(args)...);
307 } else {
308 pushed = try_push_lock_free_impl(std::forward<Args>(args)...);
309 }
310 if (!pushed) {
311 wait(m_size, k_capacity);
312 }
313 } while (!pushed);
314 }

316 // This version of push assumes that the operations on the stored class will not throw exceptions, and that nothing
317 // will interfere with the thread executing the function to completion.
318 // This version uses a(n implicit) spinlock.
319 template <typename... Args>
320 NO_DISCARD bool try_push_impl(Args&&... t)
321 {
322 index_t size = m_size;
323 if (size == k_capacity) {
324 return false;
325 }
326 // Increment size counter. This allows us to reserve our space before any changes take place.
327 // A failed CAS may set size to capacity, so check for a full buffer again in loop.
328 while (!m_size.compare_exchange_weak(size, size + 1u)) {
329 if (size == k_capacity) {
330 return false;
331 }
332 }
333
334 // We know we have space (and have reserved it) at this point. We just have to grab our write index.
335
336 // Get and increment write pos.
337 index_t write_idx = m_write_idx;
338 while (!m_write_idx.compare_exchange_weak(write_idx, increment(write_idx)))
339 ;
340
341 MNRY_ASSERT(write_idx < k_capacity);
342
343 // Wait until our slot isn't occupied, and then mark it as in transition so that:
344 // A) other writers know they can't use this index.
345 // B) readers know they can't read from this index yet.
346 //
347 // What guarantees that our slot doesn't get stolen by another writer between obtaining our index and marking it
348 // as in transition? Nothing. It's completely possible that we have wrapped around the buffer and that a later
349 // thread is writing to our slot, but they will have to mark the slot as in transition or occupied, stopping us
350 // from writing to it at the same point, and it will become available once a reader clears it up.
351 //
352 // What guarantees us that a reader will read from our slot if another writer has stolen it? The read index will
353 // have to be incremented. If we have looped so that there are two writes, we will eventually loop so that there
354 // are two reads.
355 occupied_t occupied_value = occupied_t::UNOCCUPIED;
356 while (!m_nodes[write_idx].m_occupied.compare_exchange_weak(occupied_value, occupied_t::IN_TRANSITION)) {
357 // We only want to update this when it's unoccupied, so reset our expected value.
358 wait(m_nodes[write_idx].m_occupied, occupied_value);
359 occupied_value = occupied_t::UNOCCUPIED;
360 }
361
362 // Construct
363 do_construct(write_idx, std::forward<Args>(t)...);
364
365 // Inform readers that this slot if valid and can be read from.
366 m_nodes[write_idx].m_occupied = occupied_t::OCCUPIED;
367 // We may notify another writer after we've filled it up again, which does us no good.
368 notify_all(m_size);
369 notify_all(m_nodes[write_idx].m_occupied);
370 return true;
371 }

373 // This version of push handles the class throwing exceptions while still maintaining lock-free semantics.
374 template <typename... Args>
375 NO_DISCARD bool try_push_lock_free_impl(Args&&... t)
376 {
377 index_t size = m_size;
378 if (size == k_capacity) {
379 return false;
380 }
381 // Increment size counter. This allows us to reserve our space before any changes take place.
382 // A failed CAS may set size to capacity, so check for a full buffer again in loop.
383 while (!m_size.compare_exchange_weak(size, size + 1u)) {
384 if (size == k_capacity) {
385 return false;
386 }
387 }
388
389 // We know we have space (and have reserved it) at this point. We just have to grab our write index.
390
391 while (true) {
392 // Get and increment write pos.
393 index_t write_idx = m_write_idx;
394 while (!m_write_idx.compare_exchange_weak(write_idx, increment(write_idx)))
395 ;
396
397 MNRY_ASSERT(write_idx < k_capacity);
398
399 // Wait until our slot isn't occupied, and then mark it as in transition so that:
400 // A) other writers know they can't use this index.
401 // B) readers know they can't read from this index yet.
402 //
403 // What guarantees that our slot doesn't get stolen by another writer between obtaining our index and
404 // marking it as in transition? Nothing. It's completely possible that we have wrapped around the buffer and
405 // that a later thread is writing to our slot, but they will have to mark the slot as in transition or
406 // occupied, stopping us from writing to it at the same point, and it will become available once a reader
407 // clears it up.
408 //
409 // What guarantees us that a reader will read from our slot if another writer has stolen it? The read index
410 // will have to be incremented. If we have looped so that there are two writes, we will eventually loop so
411 // that there are two reads.
412 occupied_t occupied_value = occupied_t::UNOCCUPIED;
413 if (m_nodes[write_idx].m_occupied.compare_exchange_strong(occupied_value, occupied_t::IN_TRANSITION)) {
414 auto cleanup = finally([this, write_idx] {
415 // We may notify another writer after we've filled it up again, which does us no good, so we notify
416 // all threads.
417 notify_all(m_size);
418 notify_all(m_nodes[write_idx].m_occupied);
419 });
420
421 try {
422 // Construct
423 do_construct(write_idx, std::forward<Args>(t)...);
424 } catch (...) {
425 m_nodes[write_idx].m_occupied = occupied_t::EXCEPTION;
426 throw;
427 }
428
429 // Inform readers that this slot if valid and can be read from.
430 m_nodes[write_idx].m_occupied = occupied_t::OCCUPIED;
431 return true;
432 }
433 }
434 }

436 // Special case for a single producer.
437 // This version of push assumes that the operations on the stored class will not throw exceptions, and that nothing
438 // will interfere with the thread executing the function to completion.
439 // This version uses a(n implicit) spinlock.
440 template <typename... Args>
441 NO_DISCARD bool try_push_single_impl(Args&&... t)
442 {
443 // We're the only writer. m_size will only decrease.
444 if (m_size == k_capacity) {
445 return false;
446 }
447
448 // We only have one thread inserting. We don't have to increment this right away, but either way, asynchronous
449 // readers will be busy waiting either for size or for the data to be ready.
450 ++m_size;
451
452 // Get and increment write pos.
453 const index_t write_idx =
454 m_write_idx.exchange(increment(m_write_idx.load(std::memory_order_relaxed)), std::memory_order_relaxed);
455 MNRY_ASSERT(write_idx < k_capacity);
456
457 // Wait until it's something other than occupied.
458 // Can't atomic::wait here because we need to wait if we're either occupied or in transition
459 while (m_nodes[write_idx].m_occupied != occupied_t::UNOCCUPIED) {
460 do_pause();
461 }
462
463 // Construct
464 do_construct(write_idx, std::forward<Args>(t)...);
465
466 // Inform readers that this slot if valid and can be read from.
467 m_nodes[write_idx].m_occupied = occupied_t::OCCUPIED;
468
469 // We only need to wake up one thread, but we have two wait variables, and we can't control which thread gets
470 // the notification. Inform one thread that the size has changed, and inform all threads (including the one that
471 // knows the size has changed) that the status has changed
472 notify_one(m_size);
473 notify_all(m_nodes[write_idx].m_occupied);
474 return true;
475 }

477 // Special case for a single producer.
478 // This version of push handles the class throwing exceptions while still maintaining lock-free semantics.
479 template <typename... Args>
480 NO_DISCARD bool try_push_single_lock_free_impl(Args&&... t)
481 {
482 // We're the only writer. m_size will only decrease.
483 if (m_size == k_capacity) {
484 return false;
485 }
486
487 // We only have one thread inserting. We don't have to increment this right away, but either way, asynchronous
488 // readers will be busy waiting either for size or for the data to be ready.
489 ++m_size;
490
491 while (true) {
492 // Get and increment write pos.
493 const index_t write_idx =
494 m_write_idx.exchange(increment(m_write_idx.load(std::memory_order_relaxed)), std::memory_order_relaxed);
495 MNRY_ASSERT(write_idx < k_capacity);
496
497 occupied_t occupied_value = occupied_t::UNOCCUPIED;
498 if (m_nodes[write_idx].m_occupied.compare_exchange_strong(occupied_value, occupied_t::IN_TRANSITION)) {
499 auto cleanup = finally([this, write_idx] {
500 // There are only other readers, so we don't need to worry about waking another writer.
501 // We only need to wake up one thread, but we have two wait variables, and we can't control which
502 // thread gets the notification. Inform one thread that the size has changed, and inform all threads
503 // (including the one that knows the size has changed) that the status has changed
504 notify_one(m_size);
505 notify_all(m_nodes[write_idx].m_occupied);
506 });
507
508 try {
509 // Construct
510 do_construct(write_idx, std::forward<Args>(t)...);
511 } catch (...) {
512 m_nodes[write_idx].m_occupied = occupied_t::EXCEPTION;
513 throw;
514 }
515
516 // Inform readers that this slot if valid and can be read from.
517 m_nodes[write_idx].m_occupied = occupied_t::OCCUPIED;
518 return true;
519 }
520 }

523 template <typename Iterator>
524 void push_batch(Iterator first, Iterator last, std::input_iterator_tag)
525 {
526 static_assert(Traits::producer_traits == ProducerTraits::SINGLE_PRODUCER,
527 "Batch mode supported for single producer only");
528
529 // With input iterators, we can't count, and we can't go over the values again. The best we can do is add them
530 // one at a time.
531 for (; first != last; ++first) {
532 push_impl(*first);
533 }
534 }
535
536 // This function works for forward iterators, but is more efficient with random access iterators.
537 template <typename Iterator>
538 void push_batch(Iterator first, Iterator last, std::forward_iterator_tag)
539 {
540 static_assert(Traits::producer_traits == ProducerTraits::SINGLE_PRODUCER,
541 "Batch mode supported for single producer only");
542 index_t num_elements_to_process = std::distance(first, last);
543
544 while (first != last) {
545 // We're the only writer. m_size will only decrease, meaning this is pessimistic (which is perfect).
546 const auto container_size = m_size.load(std::memory_order_acquire);
547 if (container_size == capacity()) {
548 wait(m_size, container_size);
549 }
550
551 const index_t mini_batch_size = std::min(num_elements_to_process, capacity() - container_size);
552
553 MNRY_ASSERT(mini_batch_size <= static_cast<index_t>(std::distance(first, last)));
554 const auto next = std::next(first, mini_batch_size);
555 [[gnu::unused]] bool success = try_push_batch_impl(first, next, mini_batch_size);
556 MNRY_ASSERT(success);
557 first = next;
558 num_elements_to_process -= mini_batch_size;
559 }
560 }
561
562 // This version uses a(n implicit) spinlock.
563 // We pass in num_elements even though it should be the difference between first and last so that we don't have to
564 // iterate over [first, last] when using forward_iterators.
565 // Precondition: This function should only be called when there is enough room to add num_elements.
566 template <typename Iterator>
567 NO_DISCARD bool try_push_batch_impl(Iterator first, Iterator last, index_t num_elements)
568 {
569 // TODO: C++20: This is a good case for concepts.
570 static_assert(std::is_base_of<std::forward_iterator_tag,
571 typename std::iterator_traits<Iterator>::iterator_category>::value,
572 "This only works for forward iterators or better");
573
574 // We pass in num_elements even though it can be derived through the iterators for efficiency in the case of
575 // forward iterators.
576 MNRY_ASSERT(std::distance(first, last) == num_elements);
577
578 // Since we're the one producer, this should succeed if this function is called properly.
579 MNRY_ASSERT(capacity() - m_size.load() >= num_elements);
580
581 // We only have one thread inserting. We don't have to increment this right away, but either way, asynchronous
582 // readers will be busy waiting either for container_size or for the data to be ready. Doing it here means:
583 // * We only pay for this atomic operation once.
584 // * Consumers can get data as soon as we update the occupied state.
585 // The drawback is that we block consumers (even if they do a try_pop) because they think there is data to be
586 // read but the data is inaccessible.
587 m_size += num_elements;
588 notify_all(m_size);
589
590 index_t write_idx = m_write_idx.load(std::memory_order_relaxed);
591 for (; first != last; ++first, write_idx = increment(write_idx)) {
592 MNRY_ASSERT(write_idx < k_capacity);
593
594 // Wait until it's something other than occupied.
595 // Can't atomic::wait here because we need to wait if we're either unoccupied or in transition
596 while (m_nodes[write_idx].m_occupied != occupied_t::UNOCCUPIED) {
597 do_pause();
598 }
599
600 // Construct
601 new (get_pointer(write_idx)) T(std::move_if_noexcept(*first));
602
603 // Inform readers that this slot is valid and can be read from.
604 m_nodes[write_idx].m_occupied = occupied_t::OCCUPIED;
605
606 // We have already informed all threads about the size change. We won't wake another writer (we're the only
607 // one), so just wake one thread on the status change.
608 notify_one(m_nodes[write_idx].m_occupied);
609 }
610 m_write_idx.store(write_idx, std::memory_order_relaxed);
611
612 // If uploading memory, do memcpy and do a m_size.notify_all() here instead of the notify one
613
614 return true;
615 }

617 // This version of pop assumes that the operations on the stored class will not throw exceptions, and that nothing
618 // will interfere with the thread executing the function to completion.
619 // This version uses a(n implicit) spinlock.
620 template <PopImplementation pop_implementation>
621 NO_DISCARD bool try_pop_impl(T& ret)
622 {
623 // Check for empty buffer
624 // Decrement counter
625 index_t size = m_size;
626 if (size == 0) {
627 return false;
628 }
629 while (!m_size.compare_exchange_weak(size, size - 1u)) {
630 if (size == 0) {
631 return false;
632 }
633 }
634
635 MNRY_ASSERT(size <= k_capacity);
636
637 // Get and increment read pos
638 index_t read_idx = m_read_idx;
639 while (!m_read_idx.compare_exchange_weak(read_idx, increment(read_idx)))
640 ;
641
642 MNRY_ASSERT(read_idx < k_capacity);
643
644 // Wait until we have valid data
645 occupied_t occupied_value = occupied_t::OCCUPIED;
646 while (!m_nodes[read_idx].m_occupied.compare_exchange_weak(occupied_value, occupied_t::IN_TRANSITION)) {
647 wait(m_nodes[read_idx].m_occupied, occupied_value);
648 occupied_value = occupied_t::OCCUPIED;
649 }

652 IF_CONSTEXPR (pop_implementation == PopImplementation::PLACEMENT) {
653 new (std::addressof(ret)) T{ std::move_if_noexcept(*get_pointer(read_idx)) };
654 } else IF_CONSTEXPR (pop_implementation == PopImplementation::MOVE) {
655 ret = std::move_if_noexcept(*get_pointer(read_idx));
656 }
657
658 // Destroy
659 get_pointer(read_idx)->~T();
660
661 // Mark as not occupied
662 m_nodes[read_idx].m_occupied = occupied_t::UNOCCUPIED;
663 // We may notify another reader after we've emptied it up again, which does us no good.
664 notify_all(m_size);
665 notify_all(m_nodes[read_idx].m_occupied);
666 } catch (...) {
667 get_pointer(read_idx)->~T();
668 m_nodes[read_idx].m_occupied = occupied_t::OCCUPIED;

677 // This version of pop handles the class throwing exceptions while still maintaining lock-free semantics.
678 template <PopImplementation pop_implementation>
679 NO_DISCARD bool try_pop_lock_free_impl(T& ret)
680 {
681 // Check for empty buffer
682 // Decrement counter
683 index_t size = m_size;
684 if (size == 0) {
685 return false;
686 }
687 while (!m_size.compare_exchange_weak(size, size - 1u)) {
688 if (size == 0) {
689 return false;
690 }
691 }
692
693 MNRY_ASSERT(size <= k_capacity);
694
695 while (true) {
696 // Get and increment read pos
697 index_t read_idx = m_read_idx;
698 while (!m_read_idx.compare_exchange_weak(read_idx, increment(read_idx)))
699 ;
700
701 MNRY_ASSERT(read_idx < k_capacity);
702
703 auto cleanup_function = [this, read_idx] {
704 // Mark as not occupied
705 m_nodes[read_idx].m_occupied = occupied_t::UNOCCUPIED;
706
707 // We may notify another reader after we've emptied it up again, which does us no good.
708 notify_all(m_size);
709 notify_all(m_nodes[read_idx].m_occupied);
710 };
711
712 // Wait until we have valid data
713 occupied_t occupied_value = occupied_t::OCCUPIED;
714 if (m_nodes[read_idx].m_occupied.compare_exchange_strong(occupied_value, occupied_t::IN_TRANSITION)) {
715 auto cleanup = finally(cleanup_function);
716
717 try {
718 IF_CONSTEXPR (pop_implementation == PopImplementation::PLACEMENT) {
719 new (std::addressof(ret)) T{ std::move_if_noexcept(*get_pointer(read_idx)) };
720 } else IF_CONSTEXPR (pop_implementation == PopImplementation::MOVE) {
721 ret = std::move_if_noexcept(*get_pointer(read_idx));
722 }
723
724 // Destroy
725 get_pointer(read_idx)->~T();
726 return true;
727 } catch (...) {
728 get_pointer(read_idx)->~T();
729 throw;
730 }
731 } else if (occupied_value == occupied_t::EXCEPTION) {
732 auto cleanup = finally(cleanup_function);
733 return false;
734 }
735 }
736 }

738 // Special case for a single consumer.
739 // This version of pop assumes that the operations on the stored class will not throw exceptions, and that nothing
740 // will interfere with the thread executing the function to completion.
741 // This version uses a(n implicit) spinlock.
742 template <PopImplementation pop_implementation>
743 NO_DISCARD bool try_pop_single_impl(T& ret)
744 {
745 // Check for empty buffer
746 // We're the only reader. m_size will only increase.
747 if (m_size == 0) {
748 return false;
749 }
750
751 // We only have one thread reading. We don't have to decrement this right away, but either way, asynchronous
752 // writers will be busy waiting either for size or for the data to be ready.
753 --m_size;
754
755 // Get and increment read pos
756 const index_t read_idx =
757 m_read_idx.exchange(increment(m_read_idx.load(std::memory_order_relaxed)), std::memory_order_relaxed);
758 MNRY_ASSERT(read_idx < k_capacity);
759
760 // Wait until we have valid data
761 // Can't atomic::wait here because we need to wait if we're either unoccupied or in transition
762 while (m_nodes[read_idx].m_occupied != occupied_t::OCCUPIED) {
763 do_pause();
764 }
765
766 try {
767 IF_CONSTEXPR (pop_implementation == PopImplementation::PLACEMENT) {
768 new (std::addressof(ret)) T{ std::move_if_noexcept(*get_pointer(read_idx)) };
769 } else IF_CONSTEXPR (pop_implementation == PopImplementation::MOVE) {
770 ret = std::move_if_noexcept(*get_pointer(read_idx));
771 }
772
773 // Destroy
774 get_pointer(read_idx)->~T();
775
776 // Mark as not occupied
777 m_nodes[read_idx].m_occupied = occupied_t::UNOCCUPIED;
778
779 // We only need to wake up one thread, but we have two wait variables, and we can't control which
780 // thread gets the notification. Inform one thread that the size has changed, and inform all threads
781 // (including the one that knows the size has changed) that the status has changed
782 notify_one(m_size);
783 notify_all(m_nodes[read_idx].m_occupied);
784 } catch (...) {
785 get_pointer(read_idx)->~T();
786 m_nodes[read_idx].m_occupied = occupied_t::UNOCCUPIED;
787
788 // We only need to wake up one thread, but we have two wait variables, and we can't control which
789 // thread gets the notification. Inform one thread that the size has changed, and inform all threads

791 notify_one(m_size);
792 notify_all(m_nodes[read_idx].m_occupied);

794 }
795 return true;
