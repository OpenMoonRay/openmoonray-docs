50void
51convertSampleIdRangeToPasses(const unsigned startSampleId, const unsigned endSampleId, std::vector<Pass> &passes)
52{
53 MNRY_ASSERT(3 <= MAX_RENDER_PASSES);
54
55 unsigned sId = startSampleId;
56 unsigned eId = endSampleId;
57 if (sId == eId) return; // early exit
58
59 Pass pass;
60
61 if ((sId & (unsigned)63) != 0) {
62 unsigned baseId = sId & ~(unsigned)63;
63 pass.mStartPixelIdx = (sId - baseId) & (unsigned)63;
64 pass.mEndPixelIdx = std::min((int)(eId - baseId), 64);
65 pass.mStartSampleIdx = sId / 64;
66 pass.mEndSampleIdx = pass.mStartSampleIdx + 1;
67 passes.push_back(pass);
68 sId = baseId + 64;
69 if (eId <= sId) return;
70 }
71
72 if (sId + 64 <= eId) {
73 unsigned loop = (eId - sId) / 64;
74 pass.mStartPixelIdx = 0;
75 pass.mEndPixelIdx = 64;
76 pass.mStartSampleIdx = sId / 64;
77 pass.mEndSampleIdx = pass.mStartSampleIdx + loop;
78 passes.push_back(pass);
79 sId += loop * 64;
80 if (eId <= sId) return;
81 }
82
83 pass.mStartPixelIdx = 0;
84 pass.mEndPixelIdx = eId & (unsigned)63;
85 pass.mStartSampleIdx = sId / 64;
86 pass.mEndSampleIdx = pass.mStartSampleIdx + 1;
87 passes.push_back(pass);
88}

90std::vector<unsigned>
91createAdaptiveIterationPixSampleIdTable(const unsigned maxSampleId)
92//
93// Return pixel samples value table. Adaptive sampling logic would like to evaluate variance at
94// this tables pixel sample count.
95//
96{
97 std::vector<unsigned> adaptiveIterationPixSampleIdTable;
98
99 unsigned currSampleId = 0;
100 unsigned sampleStep = 4;
101 for (unsigned i = 0; i < MAX_RENDER_PASSES; ++i) {
102 if (maxSampleId <= currSampleId + 1) break;
103
104 // we have to store endSampleId value here (i.e. sampleId + 1 value)
105 adaptiveIterationPixSampleIdTable.push_back(currSampleId + 1);
106
107 currSampleId += sampleStep;
108 sampleStep += 2;
109 }
110 adaptiveIterationPixSampleIdTable.push_back(maxSampleId);
111
112 /* useful debug dump
113 auto dumpTbl = [&](const std::vector<unsigned> &tbl) -> std::string {
114 int w0 = std::to_string(tbl.size()).size();
115 int w1 = std::to_string(tbl.back()).size();
116 int w2 = std::to_string(tbl.back() * 64).size();
117 std::ostringstream ostr;
118 ostr << "tbl (size:" << tbl.size() << ") {\n";
119 for (size_t i = 0; i < tbl.size(); ++i) {
120 ostr << " i:" << std::setw(w0) << i
121 << " " << std::setw(w1) << tbl[i]
122 << " (tileSample:" << std::setw(w2) << tbl[i] * 64 << ")" << std::endl;
123 }
124 ostr << "}";
125 return ostr.str();
126 };
127 std::cerr << dumpTbl(adaptiveIterationPixSampleIdTable) << std::endl;
128 */
129 /* useful debug dump 2
130 auto dumpTblSimple = [&](const std::vector<unsigned> &tbl) -> std::string {
131 std::ostringstream ostr;
132 for (size_t i = 0; i < tbl.size(); ++i) {
133 ostr << std::setw(5) << tbl[i];
134 if (i != tbl.size() - 1) ostr << ',';
135 if ((i + 1) % 10 == 0) ostr << '\n';
136 }
137 return ostr.str();
138 };
139 std::cerr << dumpTblSimple(adaptiveIterationPixSampleIdTable) << std::endl;
140 */
141
142 return adaptiveIterationPixSampleIdTable;
143}

145bool
146isAdaptiveIterationBoundary(const std::vector<unsigned> &adaptiveIterationPixSampleIdTbl,
147 const unsigned pixSampleId)
148//
149// Return that the given pixSampleId is on the adaptive sample iteration boundary or not.
150//
151{
152 return (pixSampleId == *std::lower_bound(adaptiveIterationPixSampleIdTbl.begin(),
153 adaptiveIterationPixSampleIdTbl.end(),
154 pixSampleId));
155}

157unsigned
158findAdaptiveStartTableId(const std::vector<unsigned> &adaptiveIterationPixSampleIdTbl,
159 const unsigned startPixSampleId)
160//
161// Find table id of proper adaptive iteration sample boundary for start
162//
163{
164 auto itr = std::lower_bound(adaptiveIterationPixSampleIdTbl.begin(), adaptiveIterationPixSampleIdTbl.end(),
165 startPixSampleId);
166 if (startPixSampleId < *itr) --itr;
167 return std::distance(adaptiveIterationPixSampleIdTbl.begin(), itr);
168}

182template <bool debugMsg>
183void
184convertSampleIdRangeToPassesAdaptive(const unsigned startSampleId, const unsigned endSampleId,
185 const std::vector<unsigned> &adaptiveIterationPixSampleIdTable,
186 std::vector<Pass> &passes,
187 std::ostringstream *debugOstr)
188{
189 auto showPass = [](const Pass &pass) -> std::string {
190 std::ostringstream ostr;
191 ostr
192 << "pix(" << pass.mStartPixelIdx << " ~ " << pass.mEndPixelIdx << ") "
193 << "smp(" << pass.mStartSampleIdx << " ~ " << pass.mEndSampleIdx << ")";
194 return ostr.str();
195 };
196
197 MNRY_ASSERT(3 <= MAX_RENDER_PASSES);
198
199 unsigned sId = startSampleId; // tilebase sampleId (i.e. pixelbase sampleId * 64)
200 unsigned eId = endSampleId; // tilebase sampleId (i.e. pixelbase sampleId * 64)
201 if (sId == eId) return; // early exit
202
203 Pass pass;
204
205 if ((sId & (unsigned)63) != 0) {
206 //
207 // pass-stage-A
208 // Create first small pass to fill the gap to the appropriate pixel Id boundary of the tile.
209 // (i.e. this pass should end by pixId = 63)
210 //
211 unsigned baseId = sId & ~(unsigned)63;
212 pass.mStartPixelIdx = (sId - baseId) & (unsigned)63;
213 pass.mEndPixelIdx = std::min((int)(eId - baseId), 64);
214 pass.mStartSampleIdx = sId / 64;
215 pass.mEndSampleIdx = pass.mStartSampleIdx + 1;
216 if (debugMsg) *debugOstr << " pass-stage-A:" << showPass(pass) << '\n';
217 passes.push_back(pass);
218 sId = baseId + 64;
219
220 if (eId <= sId) {
221 return; // Special case for single small pass
222 }
223 }
224
225 // sId is tile pixel boundary position. (i.e. sId is pixId=0 position)
226
227 if (sId + 64 <= eId) {
228 //
229 // We have samples request which cover at least one sample per pixel for all the pixels of the tile.
230 //
231 unsigned currPixSampleId = sId / 64;
232 if (!isAdaptiveIterationBoundary(adaptiveIterationPixSampleIdTable, currPixSampleId)) {
233 unsigned endTblId = findAdaptiveEndTableId(adaptiveIterationPixSampleIdTable, currPixSampleId);
234 unsigned currEndPixSampleId = adaptiveIterationPixSampleIdTable[endTblId];
235 if ((currEndPixSampleId << 6) <= eId) {
236 //
237 // pass-stage-B
238 // We have to fill the gap to the appropriate adaptive iteration boundary
239 //
240 pass.mStartPixelIdx = 0;
241 pass.mEndPixelIdx = 64;
242 pass.mStartSampleIdx = currPixSampleId;
243 pass.mEndSampleIdx = currEndPixSampleId;
244 if (debugMsg) *debugOstr << " pass-stage-B:" << showPass(pass) << '\n';
245 passes.push_back(pass);
246 sId = currEndPixSampleId * 64;
247 } else {
248 //
249 // This is a case that we don't have enough sample request to reach next adaptive iteration
250 // boundary. So just skip in this case.
251 //
252 }
253 }
254
255 currPixSampleId = sId / 64;
256 unsigned endPixSampleId = eId / 64;
257 unsigned startTblId = findAdaptiveStartTableId(adaptiveIterationPixSampleIdTable, currPixSampleId);
258 // endTblId search is also using findAdaptiveStartTableId() function here. This is intended.
259 // We need endTblId value by same logic of computing startTblId at this point.
260 unsigned endTblId = findAdaptiveStartTableId(adaptiveIterationPixSampleIdTable, endPixSampleId);
261
262 if (startTblId < endTblId) {
263 //
264 // pass-stage-C
265 // Create main middle passes which contain main adaptive sampling passes.
266 // All the middle passes have full tile pixels (pixId=0~63) with appropriate adaptive iteration
267 // sampling range.
268 //
269 for (unsigned tblId = startTblId; tblId < endTblId; ++tblId) {
270 unsigned currStartPixSampleId = adaptiveIterationPixSampleIdTable[tblId];
271 unsigned currEndPixSampleId = adaptiveIterationPixSampleIdTable[tblId + 1];
272
273 pass.mStartPixelIdx = 0;
274 pass.mEndPixelIdx = 64;
275 pass.mStartSampleIdx = currStartPixSampleId;
276 pass.mEndSampleIdx = currEndPixSampleId;
277 if (debugMsg) *debugOstr << " pass-stage-C:" << showPass(pass) << '\n';
278 passes.push_back(pass);
279
280 sId = currEndPixSampleId << 6; // convert from pix sampleId to tile sampleId
281 }
282 }
283
284 if (sId + 64 <= eId) {
285 //
286 // pass-stage-D
287 // We still have some samples left and this samples covers entire pixels of tile.
288 // (i.e. end sample is not match with adaptive iteration range boundary.
289 //
290 unsigned loop = (eId - sId) / 64;
291 pass.mStartPixelIdx = 0;
292 pass.mEndPixelIdx = 64;
293 pass.mStartSampleIdx = sId / 64;
294 pass.mEndSampleIdx = pass.mStartSampleIdx + loop;
295 if (debugMsg) *debugOstr << " pass-stage-D:" << showPass(pass) << '\n';
296 passes.push_back(pass);
297 sId += loop * 64;
298 }
299
300 if (eId <= sId) {
301 return; // Special case. All the samples are converted into passes. we are done.
302 }
303 }
304
305 //
306 // pass-stage-E
307 // Create last small pass to fill the remaining samples.
308 //
309 pass.mStartPixelIdx = 0;
310 pass.mEndPixelIdx = eId & (unsigned)63;
311 pass.mStartSampleIdx = sId / 64;
312 pass.mEndSampleIdx = pass.mStartSampleIdx + 1;
313 if (debugMsg) *debugOstr << " pass-stage-E:" << showPass(pass) << '\n';
314 passes.push_back(pass);
315}

317unsigned
318findAdaptiveEndPixSampleId(const std::vector<unsigned> &adaptiveIterationPixSampleIdTbl,
319 const unsigned startPixSampleId,
320 const unsigned adaptiveIterationSteps)
321//
322// find adaptiveEndSampleId based on the adaptiveIterationSampleIdTbl data
323//
324{
325 size_t tblId = findAdaptiveStartTableId(adaptiveIterationPixSampleIdTbl, startPixSampleId);
326 if (!isAdaptiveIterationBoundary(adaptiveIterationPixSampleIdTbl, startPixSampleId)) {
327 // startPixSampleId is not boundary of adaptiveIterationPixSampleId table.
328 // In this case, we just return end pix sampleId as next boundary of
329 // adaptiveIterationPixSampleId and just skip adaptiveIterationSteps.
330 tblId += 1;
331 } else {
332 tblId += adaptiveIterationSteps;
333 }
334 return adaptiveIterationPixSampleIdTbl[std::min(tblId, (size_t)adaptiveIterationPixSampleIdTbl.size() - 1)];
335}

337unsigned
338calcStartResumeUniformSampleTotal(const unsigned unAlignedW,
339 const unsigned unAlignedH,
340 const unsigned checkpointStartTileSampleId)
341{
342 //
343 // We have to convert tileSampleId to pixel samples total of entire image.
344 //
345 // This function is only supported single machine environment so far.
346 // Not support multi-machine execution yet.
347 // Internally single machine version of Film::getPixelFillOrder() is used.
348 // If we need to support multi-machine, we have to think bit more detail because each machine
349 // has different pixel fill order (See Film.h)
350 //
351 if ((unAlignedW % 8) == 0 && (unAlignedH % 8) == 0) {
352 // W, H are tile aligned.
353 unsigned totalTiles = (unAlignedW / 8) * (unAlignedH / 8);
354 return totalTiles * checkpointStartTileSampleId;
355 }
356
357 int totalTileX = (unAlignedW + 7) / 8;
358 int totalTileY = (unAlignedH + 7) / 8;
359 int totalFullTiles = (totalTileX - 1) * (totalTileY - 1);
360 int totalXBoundTiles = totalTileY - 1;
361 int totalYBoundTiles = totalTileX - 1;
362 int totalXYBoundTile = 1;
363
364 int xSize = unAlignedW % 8;
365 int ySize = unAlignedH % 8;
366 unsigned xBoundTileSamples = 0;
367 unsigned yBoundTileSamples = 0;
368 unsigned xyBoundTileSamples = 0;
369 for (unsigned pixId = 0; pixId < checkpointStartTileSampleId; ++pixId) {
370 unsigned pixelPerm = Film::getPixelFillOrder(pixId & 63);
371 unsigned px = pixelPerm % 8;
372 unsigned py = pixelPerm / 8;
373 if (px < xSize) xBoundTileSamples++;
374 if (py < ySize) yBoundTileSamples++;
375 if (px < xSize && py < ySize) xyBoundTileSamples++;
376 }
377
378 int totalSamples =
379 totalFullTiles * checkpointStartTileSampleId +
380 totalXBoundTiles * xBoundTileSamples +
381 totalYBoundTiles * yBoundTileSamples +
382 totalXYBoundTile * xyBoundTileSamples;
383 return totalSamples;
384}

440// static function
441unsigned
442RenderDriver::revertFilmObjectAndResetWorkQueue(RenderDriver *driver, const FrameState &fs)
443//
444// Revert film object from file for resume render and
445// reset workQueue for first renderPass based on reverted information when if checkpointMode is TIME_BASED
446// return progressCheckpointStartTileSampleId number (this is a tile sample and not pixel sample).
447//
448{
449 if (!fs.mRenderContext->getSceneContext().getResumeRender()) {
450 return 0; // this is not a resume render case.
451 }
452
453 //
454 // Resume render mode
455 //
456 // So far only process filmId = 0. If we support multi films, need to consider bit more
457 //
458 int filmId = 0;
459 Film *films = driver->mFilms;
460 Film &film = films[filmId];
461 unsigned progressCheckpointStartTileSampleId = 0;
462 if (!film.getResumedFromFileCondition()) {
463 driver->mProgressEstimation.revertFilm(true);
464
465 // This is a very first time to render. So we try to revert film object from file.
466 RenderOutputDriver *renderOutputDriver = fs.mRenderContext->getRenderOutputDriver();
467 renderOutputDriver->resetErrors(); // clean up error buffer
468 if (!driver->revertFilmData(renderOutputDriver, fs, progressCheckpointStartTileSampleId)) {
469 // revert from file failed. -> fall back to standard operation.
470 scene_rdl2::logging::Logger::error("Fall back on standard rendering");
471 films[0].clearAllBuffers();
472 driver->mProgressEstimation.revertFilm(false);
473 return 0; // fall back to normal render
474 }
475
476 // set tried condition and skip all additional revertFilmData() regardless of
477 // the result of revertFilmData()
478 film.setResumedFromFileCondition();
479
480 if (fs.mSamplingMode == SamplingMode::ADAPTIVE) {
481 unsigned pixelSampleId =
482 film.getAdaptiveRenderTilesTable()->reset(film,
483 *driver->getTiles(),
484 fs.mMinSamplesPerPixel,
485 fs.mMaxSamplesPerPixel,
486 fs.mTargetAdaptiveError,
487 fs.mViewport);
488
489 // We need to enable adjust adaptiveTree update timing logic and execute adaptiveTreeUpdate operation by current film
490 // condition before start rendering under resume adaptive sampling rendering mode.
491 UIntTable adaptiveIterationPixSampleIdTable =
492 createAdaptiveIterationPixSampleIdTable(fs.mMaxSamplesPerPixel);
493 film.enableAdjustAdaptiveTreeUpdateTiming(adaptiveIterationPixSampleIdTable);
494 film.updateAdaptiveErrorAll(film.getRenderBuffer(), *film.getRenderBufferOdd(), fs.mTargetAdaptiveError);
495
496 // Adaptive sampling case, progressCheckpointStartTileSampleId is always boundary of
497 // same pixel samples in the tile.
498 progressCheckpointStartTileSampleId = pixelSampleId * 64;
499 if (fs.mCheckpointMode == CheckpointMode::TIME_BASED) {
500 //
501 // We only setup workQueue for TIME_BASED checkpoint mode.
502 // If user select QUALITY_BASED checkpoint mode, all workQueue will be constructed
503 // inside RenderDriver::progressCheckpointRenderFrame()
504 //
505
506 unsigned startSampleId = pixelSampleId * 64;
507 // Added 64 samples to tile (i.e. = add 1 sample per pixel).
508 // Why adding 64 samples to the tile?
509 // This first render schedule is used by very first checkpoint render stint as estimation.
510 // If we set 1 sample per tile like non resumable render estimation, this 1 sample per tile
511 // is not enough in some condition and we can not get reasonable quality of estimation result.
512 // If 1 sample per tile is skipped somehow depending on sampling parameter change and/or
513 // other variance condition of the pixels, estimated averaged 1 sample cost is extreamply
514 // light (i.e. short time). As a result, estimated total sampling count becomes huge number.
515 // This situation makes totally break of checkpoint-logic.
516 // In order to reduce risk, Using 64 samples per tile for initial estimation stage.
517 unsigned endSampleId = startSampleId + 64;
518 std::vector<Pass> newPasses;
519 convertSampleIdRangeToPassesAdaptive<false>(startSampleId, endSampleId,
520 adaptiveIterationPixSampleIdTable, newPasses,
521 nullptr);
522# ifdef PRINT_DEBUG_MESSAGE_PROGRESS_CHECKPOINT
523 std::cerr << ">> RenderFrameCheckpointResume.cc revertFilmObjectAndResetWorkQueue()"
524 << std::endl;
525 std::cerr << showAllPasses("", newPasses, startSampleId, endSampleId,
526 64, fs.mMaxSamplesPerPixel * 64)
527 << std::endl;
528# endif
529 TileWorkQueue *workQueue = &driver->mTileWorkQueue;
530 const unsigned numTiles = unsigned(driver->getTiles()->size());
531 workQueue->init(RenderMode::PROGRESS_CHECKPOINT, numTiles,
532 newPasses.size(), mcrt_common::getNumTBBThreads(), &newPasses[0], nullptr);
533 }
534
535 } else {
536 unsigned startSampleId = progressCheckpointStartTileSampleId;
537 unsigned endSampleId = startSampleId + 1;
538
539 // Setup start condition for uniform resume sampling. This is important for progress info
540 unsigned startSamples =
541 calcStartResumeUniformSampleTotal(driver->mUnalignedW,
542 driver->mUnalignedH,
543 progressCheckpointStartTileSampleId);
544 driver->mProgressEstimation.setStartUniformSamples(startSamples);
545
546 if (fs.mCheckpointMode == CheckpointMode::TIME_BASED) {
547 //
548 // We only setup workQueue for TIME_BASED checkpoint mode. Otherwise workQueue will be
549 // setup inside RenderDriver::progressCheckpointRenderFrame()
550 //
551 // re-create workQueue based on resume start sampleId for first renderPasses()
552 std::vector<Pass> newPasses;
553 convertSampleIdRangeToPasses(startSampleId, endSampleId, newPasses);
554# ifdef PRINT_DEBUG_MESSAGE_PROGRESS_CHECKPOINT
555 std::cerr << ">> RenderFrameCheckpointResume.cc 1st estimate renderPasses "
556 << showAllPasses("", newPasses) << std::endl;
557# endif // end PRINT_DEBUG_MESSAGE_PROGRESS_CHECKPOINT
558 TileWorkQueue *workQueue = &driver->mTileWorkQueue;
559 const unsigned numTiles = unsigned(driver->getTiles()->size());
560 workQueue->init(RenderMode::PROGRESS_CHECKPOINT, numTiles,
561 newPasses.size(), mcrt_common::getNumTBBThreads(), &newPasses[0], nullptr);
562 }
563 }
564
565 driver->mProgressEstimation.revertFilm(false);
566 }
567
568 fs.mRenderContext->getResumeHistoryMetaData()->setStartTileSampleId(progressCheckpointStartTileSampleId);
569
570 return progressCheckpointStartTileSampleId;
571}

575// static function
576bool
577RenderDriver::progressCheckpointRenderFrame(RenderDriver *driver, const FrameState &fs,
578 const unsigned progressCheckpointStartTileSampleId,
579 const TileSampleSpecialEvent *tileSampleSpecialEvent)
580//
581// Rendering single frame by checkpoint mode. Create multiple checkpoint files inside this function.
582// Support 2 different checkpoint mode so far. it's TIME_BASED and QUALITY_BASED checkpoint mode.
583//
584// Returns true if we ran to completion or false if we were canceled.
585{
586#ifdef RUNTIME_VERIFY0
587 PixSampleCountRuntimeVerify::init(driver->getWidth(), driver->getHeight());
588#endif // end RUNTIME_VERIFY0
589
590 //------------------------------
591 //
592 // checkpoint related setup
593 //
594 RenderFrameTimingRecord &timingRec = driver->getRenderFrameTimingRecord();
595 {
596 scene_rdl2::fb_util::Tiler tiler(driver->getWidth(), driver->getHeight());
597 unsigned resumeStartWholeSamples = progressCheckpointStartTileSampleId * tiler.mNumTiles;
598 timingRec.reset(resumeStartWholeSamples); // reset condition for new frame. set FrameStartTime as well
599 }
600
601 // There should only be a single coarse pass at this point. That pass is
602 // assumed to only render a single sample per tile.
603
604 // Compute end time for estimation stage
605 double frameBudget = 0.0;
606 if (fs.mCheckpointMode == CheckpointMode::TIME_BASED) {
607 // We don't account renderPrep time for estimation. (This is a one of major difference between
608 // REALTIME rendermode).
609 if (fs.mCheckpointInterval * 60.0f > 5.0f) {
610 // The 1st checkpoint interval is adjusted to 5 sec in order to create a pretty rough
611 // immediate checkpoint data.
612 frameBudget = 5.0; // sec : user defined but not include frame update duration for 1st stint.
613 }
614 double predictedEnd = timingRec.setRenderFrameTimeBudget(frameBudget);
615
616 driver->mFrameEndTime = predictedEnd;
617 } else {
618 // QualityBased checkpoint render : basically we don't need to measure timing because we don't
619 // do any cost estimation for quality based checkpoint render
620 driver->mFrameEndTime = 0.0f;
621 }
622 unsigned firstTileSamples = driver->mTileWorkQueue.getTotalTileSamples();
623
624 // The TileWorkQueue is already configured with a single pass which renders
625 // a single sample per tile. We use this as our baseline sample cost estimate.
626# ifdef PRINT_DEBUG_MESSAGE_PROGRESS_CHECKPOINT
627 rec_time::RecTime recTime;
628 recTime.start();
629 if (fs.mCheckpointMode == CheckpointMode::TIME_BASED) {
630 std::cerr << ">> RenderFrameCheckpointResume.cc progressCheckpointRenderFrame() TIME_BASED"
631 << " 1st stint start (frameBudget:" << frameBudget << " sec) {" << std::endl;
632 } else {
633 std::cerr << ">> RenderFrameCheckpointResume.cc progressCheckpointRenderFrame() QUALITY_BASED"
634 << " start (qualitySteps:" << fs.mCheckpointQualitySteps << ")" << std::endl;
635 }
636# endif // end PRINT_DEBUG_MESSAGE_PROGRESS_CHECKPOINT
637
638 // store timing info for resume history
639 fs.mRenderContext->getResumeHistoryMetaData()->setMCRTStintStartTime();
640
641 //------------------------------
642 //
643 // initial estimation renderPasses()
644 //
645 unsigned startSampleId = 0;
646 unsigned endSampleId = 0;
647 if (fs.mCheckpointMode == CheckpointMode::TIME_BASED) {
648 //
649 // TIME_BASED checkpoint render
650 // We do initial estimation phase before start main checkpoint stint loop
651 //
652 startSampleId = progressCheckpointStartTileSampleId;
653 endSampleId = startSampleId + firstTileSamples;
654
655 driver->mParallelInitFrameUpdateTime.start();
656 driver->mCheckpointEstimationStage = true;
657 renderPasses(driver, fs, false); // no cancelation requested
658 driver->mCheckpointEstimationStage = false;
659 if (!driver->mParallelInitFrameUpdate) {
660 float time = driver->mParallelInitFrameUpdateTime.end();
661 driver->mCheckpointEstimationTime.set(time);
662 /* useful debug message
663 std::cerr << ">> RenderFrameCheckpointResume.cc READY-FOR-DISPLAY nonParallel"
664 << " time:" << time << " sec\n";
665 */
666 }
667 } else {
668 //
669 // QUALITY_BASED checkpoint render
670 // We skip all estimation conputation for quality based checkpoint render
671 //
672 startSampleId = progressCheckpointStartTileSampleId;
673 endSampleId = startSampleId;
674 }
675
676#ifdef RUNTIME_VERIFY0
677 std::cerr << ">> RenderFrameCheckpointResume.cc after initial estimation pass "
678 << PixSampleCountRuntimeVerify::get()->show() << '\n';
679#endif // end RUNTIME_VERIFY0
680
681 driver->setReadyForDisplay();
682 if (driver->mStopAtFrameReadyForDisplay) {
683 return false; // This is a equivalent with cancel condition
684 }
685
686 double renderStart = scene_rdl2::util::getSeconds(); // for time cap
687
688 // initial adaptive tile sampleCap is same number of current sampling total
689 if (fs.mSamplingMode != SamplingMode::UNIFORM) {
690 driver->mAdaptiveTileSampleCap = firstTileSamples;
691 }
692
693 //------------------------------
694 //
695 // checkpoint stint loop
696 //
697 driver->mCheckpointController.reset();
698 for (size_t checkpointStintId = 0; ; ++checkpointStintId) {
699 double durationSec = scene_rdl2::util::getSeconds() - renderStart;
700 if (fs.mCheckpointTimeCap > 0.0f && durationSec / 60.0 > (double)fs.mCheckpointTimeCap) {
701 std::ostringstream ostr;
702 ostr << "===>>> Rendering time was exceeded to the time cap (" << fs.mCheckpointTimeCap
703 << " min) and finish render <<<===";
704 scene_rdl2::logging::Logger::info(ostr.str());
705# ifdef PRINT_DEBUG_MESSAGE_PROGRESS_CHECKPOINT
706 std::cerr << ostr.str() << std::endl;
707# endif // end PRINT_DEBUG_MESSAGE_PROGRESS_CHECKPOINT
708 break; // exceeded time cap -> end
709 }
710 if (fs.mCheckpointSampleCap > 0 && endSampleId == fs.mCheckpointSampleCap) {
711 std::ostringstream ostr;
712 ostr << "===>>> Rendering sample was exceeded to the sample cap (tileSample:"
713 << fs.mCheckpointSampleCap
714 << " pixSample:" << fs.mCheckpointSampleCap / 64
715 << ") and finish render <<<===";
716 scene_rdl2::logging::Logger::info(ostr.str());
717# ifdef PRINT_DEBUG_MESSAGE_PROGRESS_CHECKPOINT
718 std::cerr << ostr.str() << std::endl;
719# endif // end PRINT_DEBUG_MESSAGE_PROGRESS_CHECKPOINT
720 break; // exceeded sample cap -> end
721 }
722
723 //
724 // One checkpoint render stint starting
725 //
726 RenderPassesResult stintResult =
727 progressCheckpointRenderStint(driver, fs, progressCheckpointStartTileSampleId,
728 tileSampleSpecialEvent, startSampleId, endSampleId);
729
730# ifdef PRINT_DEBUG_MESSAGE_PROGRESS_CHECKPOINT
731 std::cerr << ">> RenderFrameCheckpointResume.cc progressCheckpointRenderFrame() } checkpoint stint end."
732 << " interval:" << recTime.end() << " sec" << std::endl;
733# endif // end PRINT_DEBUG_MESSAGE_PROGRESS_CHECKPOINT
734
735 if (stintResult == RenderPassesResult::ERROR_OR_CANCEL) {
736 return false;
737 }
738 if (stintResult == RenderPassesResult::STOP_AT_PASS_BOUNDARY) {
739 driver->mFrameCompleteAtPassBoundary = true;
740 break; // We've completed
741 }
742
743#ifdef PRINT_DEBUG_MESSAGE_PROGRESS_CHECKPOINT
744 std::cerr << ">> RenderFrameCheckpointResume.cc checkpoint stint end. interval:" << recTime.end()
745 << " sec" << std::endl;
746#endif // end PRINT_DEBUG_MESSAGE_PROGRESS_CHECKPOINT
747
748 //
749 // output checkpoint file
750 //
751#ifdef PRINT_DEBUG_MESSAGE_PROGRESS_CHECKPOINT
752 std::cerr << ">> RenderFrameCheckpointResume.cc checkpoint file output."
753 << " endSampleId:" << endSampleId << '\n'
754 << "=========================================================" << std::endl;
755#endif // end PRINT_DEBUG_MESSAGE_PROGRESS_CHECKPOINT
756
757 // record MCRT end timing for resumeHistory
758 fs.mRenderContext->getResumeHistoryMetaData()->setMCRTStintEndTime(endSampleId - 1);
759
760 checkpointFileOutput(driver, fs, endSampleId);
761
762 if (stintResult == RenderPassesResult::COMPLETED_ALL_SAMPLES) {
763 break; // We've completed whole samples and done.
764 }
765
766 // not completed. Let's record start time for next checkpoint stint
767 fs.mRenderContext->getResumeHistoryMetaData()->setMCRTStintStartTime();
768
769 frameBudget = fs.mCheckpointInterval * 60.0f; // convert to secound from minute
770
771# ifdef PRINT_DEBUG_MESSAGE_PROGRESS_CHECKPOINT
772 recTime.start();
773 std::cerr << ">> RenderFrameCheckpointResume.cc progressCheckpointRenderFrame() checkpoint 2nd/later"
774 << " stint start frameBudget:" << frameBudget << " sec) {" << std::endl;
775# endif // end PRINT_DEBUG_MESSAGE_PROGRESS_CHECKPOINT
776
777 { // compute end time for next checkpoint render stint
778 timingRec.newStint();
779
780 double predictedEnd = timingRec.setRenderFrameTimeBudget(frameBudget);
781 driver->mFrameEndTime = predictedEnd;
782 }
783 } // checkpoint stint loop
784 fs.mRenderContext->getRenderOutputDriver()->setLastCheckpointRenderTileSamples(endSampleId);
785
786 // finalize sync (wait all ImageWriteDriver task) here
787 fs.mRenderContext->getResumeHistoryMetaData()->setFinalizeSyncStartTime();
788 ImageWriteDriver::get()->conditionWaitUntilAllCompleted(); // condition wait
789 fs.mRenderContext->getResumeHistoryMetaData()->setFinalizeSyncEndTime();
790
791 // mark frame as fully complete.
792 driver->setFrameComplete();
793 driver->mProgressEstimation.setFrameComplete();
794
795#ifdef RUNTIME_VERIFY0
796 std::cerr << PixSampleCountRuntimeVerify::get()->show() << '\n';
797#endif // end RUNTIME_VERIFY0
798
799 return true;
800}

802// static function
803RenderDriver::RenderPassesResult
804RenderDriver::progressCheckpointRenderStint(RenderDriver *driver, const FrameState &fs,
805 const unsigned processStartTileSampleId,
806 const TileSampleSpecialEvent *tileSampleSpecialEvent,
807 unsigned &startSampleId, unsigned &endSampleId)
808//
809// This function try to render some checkpoint stint computation. Not include checkpoint file output.
810// Under TIME_BASED mode, try to occupied entire time budget based on the average cost of single
811// sample. Under QUALITY_BASED mode, just used pre-computed quality tile sample id table then determines
812// sample volumes.
813//
814// startSampleId, endSampleId
815// These 2 argument values are reference and updated inside this function.
816// These 2 values are not indicate start and end sampleId of this checkpoint render stint.
817// Start/end sampleId is computed based on estimation logic and not defined by argument.
818//
819{
820 RenderFrameTimingRecord &timingRec = driver->getRenderFrameTimingRecord();
821
822 UIntTable adaptiveIterationPixSampleIdTable;
823 const UIntTable *adaptiveIterationPixSampleIdTablePtr = nullptr;
824 if (fs.mSamplingMode != SamplingMode::UNIFORM) {
825 // Adaptive sampling case, we setup adaptiveIterationPixSampleIdTable
826 adaptiveIterationPixSampleIdTable = createAdaptiveIterationPixSampleIdTable(fs.mMaxSamplesPerPixel);
827 adaptiveIterationPixSampleIdTablePtr = &adaptiveIterationPixSampleIdTable;
828
829 // We do enable adjust adaptiveTree update timing logic for checkpoint + adaptive sampling.
830 for (unsigned ifilm = 0; ifilm < driver->mLastNumActiveFilms; ++ifilm) {
831 driver->mFilms[ifilm].enableAdjustAdaptiveTreeUpdateTiming(adaptiveIterationPixSampleIdTable);
832 }
833 }
834
835 if (startSampleId == processStartTileSampleId) {
836 // This is a very first checkpoint stint just after estimation render has been completed.
837 // We should initialize progress estimation logic here by some initial condition.
838 unsigned finalTileSamples = fs.mMaxSamplesPerPixel * 64;
839 driver->mProgressEstimation.startEstimation(timingRec, processStartTileSampleId, finalTileSamples);
840 }
841
842 if (fs.mCheckpointMode == CheckpointMode::TIME_BASED) {
843 //
844 // TIME_BASED checkpoint rendering
845 //
846 while (1) { // Retry loop if still time-budget is not run out.
847 double remainingTime = 0.0;
848
849 if (isProgressCheckpointComplete(driver, fs)) {
850 return RenderPassesResult::COMPLETED_ALL_SAMPLES; // We've completed -> done!
851 }
852
853 double now = scene_rdl2::util::getSeconds();
854 remainingTime = driver->mFrameEndTime - now;
855 if (timingRec.isComplete(remainingTime, now)) { // Do we have time to render more samples?
856 break; // -> Time budget for this stint expired and end rendering loop
857 }
858
859 {
860 unsigned numNewSamplesPerTile =
861 calcCheckpointStintStartEndSampleId(driver, fs,
862 remainingTime,
863 adaptiveIterationPixSampleIdTablePtr,
864 startSampleId, endSampleId);
865 if (numNewSamplesPerTile == 0) {
866 return RenderPassesResult::COMPLETED_ALL_SAMPLES; // sample budget expired -> done!
867 }
868 }
869
870 RenderPassesResult result =
871 checkpointRenderMiniStintLoop(driver, fs,
872 adaptiveIterationPixSampleIdTablePtr,
873 tileSampleSpecialEvent,
874 startSampleId, endSampleId);
875 if (result != RenderPassesResult::OK_AND_TRY_NEXT) {
876 return result;
877 }
878
879 if (endSampleId == 64) { // Check if we still need to run extrapolation before displaying.
880 driver->setCoarsePassesComplete();
881 break; // endSampleId == 64 should be last pass
882 }
883 if (endSampleId == fs.mCheckpointSampleCap) { // checkpoint sample cap
884 break;
885 }
886 } // while (1) : Retry loop if still time-budget is not run out
887
888 } else if (fs.mCheckpointMode == CheckpointMode::QUALITY_BASED) {
889 //
890 // QUALITY_BASED checkpoint rendering : logic is bit simpler than time based mode
891 //
892
893 // Check finish condition first.
894 if (isProgressCheckpointComplete(driver, fs)) {
895 return RenderPassesResult::COMPLETED_ALL_SAMPLES; // We've completed -> done!
896 }
897
898 {
899 unsigned numNewSamplesPerTile =
900 calcCheckpointStintStartEndSampleId(driver, fs,
901 0.0, // This argument is not used by QUALITY_BASED
902 adaptiveIterationPixSampleIdTablePtr,
903 startSampleId, endSampleId);
904 if (numNewSamplesPerTile == 0) {
905 return RenderPassesResult::COMPLETED_ALL_SAMPLES; // no additional samples -> done!
906 }
907 }
908
909 RenderPassesResult result =
910 checkpointRenderMiniStintLoop(driver, fs,
911 adaptiveIterationPixSampleIdTablePtr, tileSampleSpecialEvent,
912 startSampleId, endSampleId);
913 if (result != RenderPassesResult::OK_AND_TRY_NEXT) {
914 return result;
915 }
916
917 // update pass info just in case even QUALITY_BASED checkpoint mode
918 driver->mProgressEstimation.updatePassInfo(timingRec);
919
920 if (endSampleId == 64) { // Check if we still need to run extrapolation before displaying.
921 driver->setCoarsePassesComplete();
922 }
923
924 // We need to check all possible samples are completed here.
925 // We have to return render finish condition from this function due to avoid
926 // unnecessary empty checkpoint stint loop. If we do empty extra checkpoint stint loop,
927 // exactly same checkpoint file is created twice and have to pay extra checkpoint write cost.
928 // Even more, empty extra checkpoint stint loop executes exactly same post checkpoint script
929 // twice. This is pretty bad idea and might creates critical situation.
930 if (isProgressCheckpointComplete(driver, fs)) {
931 return RenderPassesResult::COMPLETED_ALL_SAMPLES; // We've completed -> done!
932 }
933 }
934
935 return RenderPassesResult::OK_AND_TRY_NEXT; // need to continue more checkpoint stint
936}

939bool
940RenderDriver::isProgressCheckpointComplete(RenderDriver *driver, const FrameState &fs)
941{
942 if (fs.mSamplingMode == SamplingMode::UNIFORM) {
943 //
944 // UNIFORM sampling mode
945 //
946 const unsigned spp = fs.mMaxSamplesPerPixel;
947 const unsigned finalWholeSamples = driver->getWidth() * driver->getHeight() * spp;
948
949 RenderFrameTimingRecord &timingRec = driver->getRenderFrameTimingRecord();
950
951# ifdef PRINT_DEBUG_MESSAGE_PROGRESS_CHECKPOINT
952 std::cerr << ">> RenderFrameCheckpointResume.cc isProgressCheckpointComplete() } endTest: numSamples:"
953 << timingRec.getNumSamplesAll()
954 << " finalWholeSamples:" << finalWholeSamples << std::endl;
955# endif // end PRINT_DEBUG_MESSAGE_PROGRESS_CHECKPOINT
956
957 if (timingRec.getNumSamplesAll() >= finalWholeSamples) {
958# ifdef PRINT_DEBUG_MESSAGE_PROGRESS_CHECKPOINT
959 std::cerr << ">> RenderFrameCheckpointResume.cc isProgressCheckpointComplete() done" << std::endl;
960# endif // end PRINT_DEBUG_MESSAGE_PROGRESS_CHECKPOINT
961 return true; // We've completed whole samples -> end loop
962 }
963
964 } else {
965 //
966 // ADAPTIVE sampling mode : so far only test filmId = 0
967 //
968 if (driver->mFilms[0].getAdaptiveDone()) {
969 return true; // all tiles have been converged -> end loop
970 }
971 }
972
973 return false;
974}

976// static function
977unsigned
978RenderDriver::calcCheckpointStintStartEndSampleId(RenderDriver *driver, const FrameState &fs,
979 const double remainingTime, // for time based mode
980 const UIntTable *adaptiveIterationPixSampleIdTable,
981 unsigned &startSampleId,
982 unsigned &endSampleId)
983//
984// Compute startSampleId and endSampleId for checkpoint stint based on the checkpoint mode (TIME_BASED /
985// QUALITY_BASED) with related information.
986//
987// adaptiveIterationPixSampleIdTable is only set properly when adaptive sampling mode otherwise might be nullptr
988//
989{
990 //
991 // 1st phase : compute estimated samples per tile based on the checkpoint mode
992 //
993 double estimatedSamplesPerTile =
994 estimateSamplesPerTile(driver, fs,
995 remainingTime, adaptiveIterationPixSampleIdTable, endSampleId);
996
997 //
998 // 2nd phase : apply some sample number special rules and check end condition.
999 //
1000 unsigned numNewSamplesPerTile = scene_rdl2::math::max(unsigned(estimatedSamplesPerTile), 1u); // for next render
1001
1002 // Compute startSampleId and endSampleId for this checkpoint stint.
1003 // (note: pixels wrap around after we exceed 64).
1004 const unsigned finalTileSamples = fs.mMaxSamplesPerPixel * 64;
1005 startSampleId = endSampleId;
1006 endSampleId = std::min(startSampleId + numNewSamplesPerTile, finalTileSamples);
1007
1008 if (startSampleId == endSampleId) {
1009 return 0; // sample budget expired -> done!
1010 }
1011
1012 if (startSampleId < 64 && 64 <= endSampleId) {
1013 endSampleId = 64; // We do not render coarse pass and fine pass in same checkpoint stint
1014 numNewSamplesPerTile = endSampleId - startSampleId; // re-compute new samples per tile
1015 }
1016 if (startSampleId < fs.mCheckpointSampleCap && fs.mCheckpointSampleCap <= endSampleId) {
1017 // checkpoint sample cap operation.
1018 endSampleId = fs.mCheckpointSampleCap;
1019 numNewSamplesPerTile = endSampleId - startSampleId; // re-compute new samples per tile
1020 }
1021
1022 return numNewSamplesPerTile;
1023}

1025// static function
1026double
1027RenderDriver::estimateSamplesPerTile(RenderDriver *driver, const FrameState &fs,
1028 const double remainingTime, // for time based mode
1029 const UIntTable *adaptiveIterationPixSampleIdTable,
1030 const unsigned startTileSampleId) // for quality based mode
1031//
1032// This function calculates samples per tile for this checkpoint stint based on the current checkpoint mode.
1033// Just return samplesPerTile number. Later stage convert this samplesPerTile number to the start/end sampleId.
1034// Return value uses double precision due to all timing related information has already double precision
1035// (and there is no special reason to cast float then reduce precision inside this function. Performance of
1036// this function is also not so important because this function is executed once or very few counts at every
1037// checkpoint stint).
1038//
1039// adaptiveIterationPixSampleIdTable is only used when non SamplingMode::UNIFORM w/ QUALITY_BASED mode.
1040//
1041{
1042 double estimatedSamplesPerTile = 0.0;
1043 if (fs.mCheckpointMode == CheckpointMode::TIME_BASED) {
1044 //
1045 // TIME_BASED checkpoint : estimation of samplePerTile based on timing history
1046 //
1047 RenderFrameTimingRecord &timingRec = driver->getRenderFrameTimingRecord();
1048
1049 estimatedSamplesPerTile =
1050 (fs.mSamplingMode == SamplingMode::UNIFORM)?
1051 timingRec.estimateSamplesAllStint(remainingTime): // using whole result
1052 timingRec.estimateSamples(remainingTime); // only using previous phase
1053
1054 // Sepcial adjustment for time based mode
1055 if (driver->mAdaptiveTileSampleCap > 0 &&
1056 estimatedSamplesPerTile > driver->mAdaptiveTileSampleCap) {
1057 // Only for adaptive sampling under TIME_BASED checkpoint render mode.
1058 // We can only increase 2x bigger CAP than current value at each checkpoint stint as max.
1059 // This is a safety logic to ignore wrong passes which includes massive tile samples
1060 // (sometimes massive passes are created if sample cost estimation is pretty pure case).
1061 driver->mAdaptiveTileSampleCap =
1062 std::min((unsigned)estimatedSamplesPerTile, driver->mAdaptiveTileSampleCap * 2);
1063 estimatedSamplesPerTile =
1064 std::min(estimatedSamplesPerTile, (double)driver->mAdaptiveTileSampleCap);
1065 }
1066
1067 } else {
1068 //
1069 // QUALITY_BASED checkpoint : estimation of samplePerTile based on quality steps of iterations
1070 //
1071 unsigned overRunSamples = startTileSampleId % 64;
1072 unsigned fillUpSamples = 0;
1073 if (overRunSamples) {
1074 // startTileSampleId is not boundary of equal pixel samples inside one tile.
1075 // In this case, we should add special sampling in order to fill up
1076 // to next boundary of equal pixel samples inside one tile.
1077 fillUpSamples = 64 - overRunSamples;
1078 }
1079
1080 switch (fs.mSamplingMode) {
1081 case SamplingMode::UNIFORM : // uniform sampling
1082 // Uniform sampling case, we use qualitySteps as increased pixel samples number.
1083 estimatedSamplesPerTile =
1084 fillUpSamples +
1085 fs.mCheckpointQualitySteps * 64.0; // converted tile samples
1086 break;
1087
1088 default : { // adaptive sampling
1089 unsigned pixStartSampleId = (startTileSampleId + fillUpSamples) / 64;
1090 unsigned pixEndSampleId = findAdaptiveEndPixSampleId(*adaptiveIterationPixSampleIdTable,
1091 pixStartSampleId,
1092 fs.mCheckpointQualitySteps);
1093 estimatedSamplesPerTile =
1094 fillUpSamples +
1095 (pixEndSampleId - pixStartSampleId) * 64.0;
1096 } break;
1097 }
1098 }
1099
1100# ifdef PRINT_DEBUG_MESSAGE_PROGRESS_CHECKPOINT
1101 if (fs.mCheckpointMode == CheckpointMode::TIME_BASED) {
1102 std::cerr << ">> RenderFrameCheckpointResume.cc TIME_BASED remainingTime:" << remainingTime << " sec"
1103 << " estimatedSamplesPerTile:" << estimatedSamplesPerTile << std::endl;
1104 } else {
1105 std::cerr << ">> RenderFrameCheckpointResume.cc QUALITY_BASED"
1106 << " qualitySteps:" << fs.mCheckpointQualitySteps
1107 << " estimatedSamplesPerTile:" << estimatedSamplesPerTile << std::endl;
1108 }
1109# endif // end PRINT_DEBUG_MESSAGE_PROGRESS_CHECKPOINT
1110
1111 return estimatedSamplesPerTile;
1112}

1114// static function
1115RenderDriver::RenderPassesResult
1116RenderDriver::checkpointRenderMiniStintLoop(RenderDriver *driver,
1117 const FrameState &fs,
1118 const UIntTable *adaptiveIterationPixSampleIdTable,
1119 const TileSampleSpecialEvent *tileSampleSpecialEvent,
1120 unsigned &startSampleId, unsigned &endSampleId) // tile sampleId
1121//
1122// In order to support tileSampleSpecialEvent, we have to break down start/end sampleId into mini-stint.
1123// This function creates mini-stint and render them sequentially. After finish every mini-stint,
1124// we do call specialEvent callBack function is needed.
1125// If no tileSampleSepcialEvent is specified, just process start/end sample id span as single stint
1126// without callBack.
1127//
1128// startSampleId, endSampleId
1129// These 2 argument values are reference and updated inside this function.
1130// These 2 values are not indicate start and end sampleId of this checkpoint render stint.
1131// Start/end sampleId is computed based on estimation logic and not defined by argument.
1132//
1133{
1134 // First of all, create miniStint sampleId table based on start/end sampleId with
1135 // tileSampleSpecialEvent information.
1136 UIntTable miniStintSampleIdTable =
1137 calcCheckpointMiniStintStartEndId(startSampleId, endSampleId, tileSampleSpecialEvent);
1138
1139 // Execute checkpointRenderMiniStint() function based on the table which created above.
1140 for (size_t miniStintId = 0; miniStintId < miniStintSampleIdTable.size() - 1; ++miniStintId) {
1141 startSampleId = miniStintSampleIdTable[miniStintId];
1142 endSampleId = miniStintSampleIdTable[miniStintId + 1];
1143 RenderPassesResult result =
1144 checkpointRenderMicroStintLoop(driver, fs, adaptiveIterationPixSampleIdTable,
1145 startSampleId, endSampleId);
1146 if (result != RenderPassesResult::OK_AND_TRY_NEXT) {
1147 return result;
1148 }
1149
1150 if (tileSampleSpecialEvent && tileSampleSpecialEvent->table().size()) {
1151 // Only test is specialEventTileSampleIdTable size is not 0 case.
1152 if (isRequiredSpecialEvent(tileSampleSpecialEvent->table(), endSampleId)) {
1153 // Execute specialEvent call back procedure. This call is done by single thread
1154 // and always run thread safe way.
1155 if (!tileSampleSpecialEvent->callBack()(endSampleId - 1)) {
1156 return RenderPassesResult::ERROR_OR_CANCEL;
1157 }
1158 }
1159 }
1160 }
1161
1162 return RenderPassesResult::OK_AND_TRY_NEXT;
1163}

1165// static function
1166RenderDriver::UIntTable
1167RenderDriver::calcCheckpointMiniStintStartEndId(const unsigned startSampleId,
1168 const unsigned endSampleId,
1169 const TileSampleSpecialEvent *tileSampleSpecialEvent)
1170//
1171// Create mini-stint tile sample id table based on tileSampleSpecialEvent information
1172//
1173{
1174 UIntTable miniStintSampleIdArray;
1175 if (!tileSampleSpecialEvent || !tileSampleSpecialEvent->table().size()) {
1176 //
1177 // no tileSampleSpecialEvent information or it's table is empty
1178 // -> no need to split start/end sampleId span.
1179 //
1180 miniStintSampleIdArray.push_back(startSampleId);
1181 miniStintSampleIdArray.push_back(endSampleId);
1182 return miniStintSampleIdArray;
1183 }
1184
1185 // Find end sampleId based on current start/end sampleId span with specialEventTileSampleIdArray
1186 auto findEndId = [&](const UIntTable &tbl,
1187 const unsigned startSampleId, const unsigned endSampleId) -> unsigned {
1188 auto itrStart = std::lower_bound(tbl.begin(), tbl.end(), startSampleId);
1189 auto itrEnd = std::lower_bound(itrStart, tbl.end(), endSampleId - 1);
1190 if (itrStart == itrEnd) {
1191 return endSampleId;
1192 }
1193 return (*itrStart) + 1;
1194 };
1195
1196 miniStintSampleIdArray.push_back(startSampleId);
1197 unsigned currStartSampleId = startSampleId;
1198 while (1) {
1199 if (currStartSampleId == endSampleId) break;
1200 unsigned currEndSampleId = findEndId(tileSampleSpecialEvent->table(), currStartSampleId, endSampleId);
1201 miniStintSampleIdArray.push_back(currEndSampleId);
1202 currStartSampleId = currEndSampleId;
1203 }
1204
1205 return miniStintSampleIdArray;
1206}

1208// static function
1209bool
1210RenderDriver::isRequiredSpecialEvent(const UIntTable &specialEventTileSampleIdArray,
1211 const unsigned endSampleId)
1212//
1213// Determine this endSampleId needs to do special event (=true) or not (=false).
1214//
1215{
1216 auto itrEnd =
1217 std::lower_bound(specialEventTileSampleIdArray.begin(), specialEventTileSampleIdArray.end(),
1218 endSampleId - 1);
1219 if ((*itrEnd) + 1 == endSampleId) return true;
1220 return false;
1221}

1223RenderDriver::RenderPassesResult
1224RenderDriver::checkpointRenderMicroStintLoop(RenderDriver *driver,
