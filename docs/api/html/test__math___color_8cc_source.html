168

170 feenableexcept(FE_DIVBYZERO | FE_INVALID | FE_OVERFLOW);
171
172 const Color a(0, 1, 2);
173 CPPUNIT_ASSERT(a == Color(0, 1, 2));
174 CPPUNIT_ASSERT(!(a == Color(1, 1, 2)));
175 CPPUNIT_ASSERT(!(a == Color(0, 2, 2)));
176 CPPUNIT_ASSERT(!(a == Color(0, 1, 1)));
177 CPPUNIT_ASSERT(!(a == Color(0, -1, -2)));
178
179 CPPUNIT_ASSERT(!(a != Color(0, 1, 2)));
180 CPPUNIT_ASSERT(a != Color(1, 1, 2));
181 CPPUNIT_ASSERT(a != Color(0, 2, 2));
182 CPPUNIT_ASSERT(a != Color(0, 1, 1));
183 CPPUNIT_ASSERT(a != Color(0, -1, -2));
184
185 CPPUNIT_ASSERT(a < (2 * a));
186 CPPUNIT_ASSERT(!(a < (a / 2)));
187
188 CPPUNIT_ASSERT(isEqual(a, a));
189 CPPUNIT_ASSERT(!isEqual(a, -a));
190 CPPUNIT_ASSERT(isEqualFixedEps(a, a));
191 CPPUNIT_ASSERT(!isEqualFixedEps(a, 0.99f * a));
192
193 CPPUNIT_ASSERT(!isBlack(a));
194 CPPUNIT_ASSERT(isBlack(Color(0, 0, 0)));
195 CPPUNIT_ASSERT(!isExactlyZero(a));
196 CPPUNIT_ASSERT(isExactlyZero(Color(0, 0, 0)));
197
198 CPPUNIT_ASSERT(isBlack(sBlack));
199 CPPUNIT_ASSERT(!isBlack(sWhite));
200
201 const Color b(-1, -2, -3);
202 CPPUNIT_ASSERT(a == select(true, a, b));
203 CPPUNIT_ASSERT(b == select(false, a, b));
204

206 fedisableexcept(FE_ALL_EXCEPT);
