34int main(int argc, char* argv[])
35{
36 logging::Logger::init();
37
38 // Register recognized command line options.
39 size_t elemsPerLine;
40 po::options_description optionsDesc("Options");
41 optionsDesc.add_options()
42 ("help,h", "Print help message")
43 ("in,i", po::value<std::string>()->required(), "Input file (.rdla | .rdlb)")
44 ("out,o", po::value<std::string>()->required(), "Output file (.rdla | .rdlb)")
45 ("elements,e", po::value<size_t>(&elemsPerLine)->default_value(0), "Number of ascii array elements per-line, 0=unlimited")
46 ("dso_path,d", po::value<std::string>(),
47 "The path to the dsos"); // dummy to please boost, will parse below;
48
49 // The "in" and "out" options can also be specified positionally as the
50 // first and second arguments.
51 po::positional_options_description positionalDesc;
52 positionalDesc.add("in", 1);
53 positionalDesc.add("out", 1);
54
55 // Parse the command line.
56 std::string dsoPath;
57 po::variables_map varsMap;
58 try {
59 po::store(po::command_line_parser(argc, argv).options(optionsDesc)
60 .positional(positionalDesc)
61 .run(), varsMap);
62 if (varsMap.count("help")) {
63 // Dump a usage friendly message rather than an error message.
64 printUsage(std::cout, argv[0], optionsDesc);
65 return EXIT_SUCCESS;
66 }
67 po::notify(varsMap);
68
69 // Parse the dso path here, instead of using boost. This paves the way
70 // for when we drop dependency on boost.
71 dsoPath = rdl2::DsoFinder::parseDsoPath(argc, argv);
72 } catch (po::error& e) {
73 // Something went wrong while parsing the options. Print the error
74 // message and a usage message.
75 logging::Logger::error(e.what());
76 printUsage(std::cerr, argv[0], optionsDesc);
77 return EXIT_FAILURE;
78 }
79
80 try {
81 // Create a SceneContext, read from the input file, and write to the
82 // output file. Simple as that!
83 rdl2::SceneContext context;
84 context.setProxyModeEnabled(true);
85 if (!dsoPath.empty()) {
86 context.setDsoPath(dsoPath);
87 }
88 rdl2::readSceneFromFile(varsMap["in"].as<std::string>(), context);
89 rdl2::writeSceneToFile(context, varsMap["out"].as<std::string>(),
90 true, // deltaEncoding
91 true, // skipDefaults
92 elemsPerLine);
93 } catch (std::exception& e) {
94 logging::Logger::error(e.what());
95 return EXIT_FAILURE;
96 }
97
98 return EXIT_SUCCESS;
99}
