18// A super-simple memory pool that only cares about one type for a fixed upper-bound.
19// This memory pool neither constructs nor destroys the type: it is assumed that whatever mechanism is using the
20// pool will construct/destroy as needed. The reason this is typed is so that we don't have to do alignment and
21// size math nor worry about heterogeneous allocation sizes at runtime.
22template <typename T>
23class TypedStaticallySizedMemoryPool
24{
25public:
26 explicit TypedStaticallySizedMemoryPool(std::size_t n)
27 : mNumAllocated(0)
28#if DEBUG_MEMORY_POOL_LIMITS
29 , mCountDebug(n)
30#endif
31 , mStorage(new AlignedStorage[n])
32 {
33 }
34
35 TypedStaticallySizedMemoryPool(const TypedStaticallySizedMemoryPool&) = delete;
36 TypedStaticallySizedMemoryPool(TypedStaticallySizedMemoryPool&&) noexcept = default;
37 TypedStaticallySizedMemoryPool& operator=(const TypedStaticallySizedMemoryPool&) = delete;
38 TypedStaticallySizedMemoryPool& operator=(TypedStaticallySizedMemoryPool&&) noexcept = default;
39
40 T* allocate(std::size_t n) noexcept
41 {
42#if DEBUG_MEMORY_POOL_LIMITS
43 MNRY_ASSERT_REQUIRE(mNumAllocated + n <= mCountDebug);
44#endif
45 // Deliberately do not call constructor.
46 T* const p = getPointer(mNumAllocated);
47 mNumAllocated += n;
48 return p;
49 }
50
51 void clear() noexcept
52 {
53 // Deliberately do not call destructors.
54 mNumAllocated = 0;
55 }
56
57private:
58 T* getPointer(std::size_t n) noexcept
59 {
60 return reinterpret_cast<T*>(std::addressof(mStorage[n]));
61 }
62
63 using AlignedStorage = typename std::aligned_storage<sizeof(T), std::alignment_of<T>::value>::type;
64 std::size_t mNumAllocated;
65#if DEBUG_MEMORY_POOL_LIMITS
66 std::size_t mCountDebug;
67#endif
68 std::unique_ptr<AlignedStorage[]> mStorage;
69};

123template <typename T1, typename T2, std::size_t alignment>
124bool operator==(const TypedStaticallySizedPoolAllocator<T1>& lhs,
125 const TypedStaticallySizedPoolAllocator<T2>& rhs) noexcept
126{
127 // Pool allocators are not equal unless the underlying pool is the same. We don't want to allocate from one and
128 // deallocate from another.
129 return std::addressof(lhs.getPool()) == std::addressof(rhs.getPool());
130}

135{
136 return !(lhs == rhs);

151template <typename T>
152void* operator new(std::size_t bytes, scene_rdl2::alloc::TypedStaticallySizedMemoryPool<T>& pool) noexcept
153{
154 // operator new passes in the number of bytes to allocate, but our pool works in relation to T.
155 MNRY_ASSERT(sizeof(T) == bytes);
156 return pool.allocate(1);
157}

159template <typename T>
160void* operator new[](std::size_t bytes, scene_rdl2::alloc::TypedStaticallySizedMemoryPool<T>& pool) noexcept
161{
162 // operator new passes in the number of bytes to allocate, but our pool works in relation to T.
163 MNRY_ASSERT(bytes % sizeof(T) == 0);
164 return pool.allocate(bytes / sizeof(T));
165}
