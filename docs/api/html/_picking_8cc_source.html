28float
29computeOpenGLDepth(mcrt_common::ThreadLocalState *tls, const Scene* scene,
30 int cameraId, int pixelX, int pixelY)
31{
32 // Initialize intersection
33 shading::Intersection isect;
34
35 // Initialize ray
36 mcrt_common::RayDifferential ray;
37 scene->getCamera(cameraId)->createRay(&ray, pixelX + 0.5f, pixelY + 0.5f,
38 0.5f, 0.5f, 0.5f, false);
39
40 // Initialize depth to the far plane of the camera
41 float depth = scene->getCamera(cameraId)->getRdlCamera()->get(scene_rdl2::rdl2::Camera::sFarKey);
42 int lobeType = 0;
43 // Intersect the ray with the scene. This does the heavy lifting.
44 bool hitGeom = scene->intersectRay(tls, ray, isect, lobeType);
45 // Transform intersection point into camera space
46 scene_rdl2::math::Vec3f p = transformPoint(scene->getCamera(cameraId)->getRender2Camera(), isect.getP());
47
48 // Only executed for primary rays
49 if (hitGeom) {
50 depth = std::max(0.f, -p.z);
51 }
52
53 return depth;
54}

96void
97computeLightContributions(mcrt_common::ThreadLocalState *tls, const Scene* scene,
98 const int x, const int y,
99 shading::LightContribArray& lightContributions,
100 int numSamples, float textureFilterScale)
101{
102 // Initialize intersection
103 shading::Intersection isect;
104
105 mcrt_common::RayDifferential ray;
106 // pick with primary camera (0)
107 scene->getCamera(0)->createRay(&ray, x + 0.5f, y + 0.5f,
108 0.5f, 0.5f, 0.5f, false);
109
110 // Test whether we hit geometry
111 int lobeType = 0;
112 bool hitGeom = scene->intersectRay(tls, ray, isect, lobeType);
113
114 // Test whether we hit visible lights in camera, if closer than geometry
115 std::vector<const Light*> hitLights;
116 std::vector<pbr::LightIntersection> hitLightIsects;
117 scene->pickVisibleLights(ray, hitGeom ? ray.getEnd() : pbr::sInfiniteLightDistance, hitLights,
118 hitLightIsects);
119
120 // Evaluate the radiance of all visible lights in camera
121 for (size_t i = 0; i < hitLights.size(); ++i) {
122 const Light *hitLight = hitLights[i];
123
124 LightFilterRandomValues lightFilterR = {
125 scene_rdl2::math::Vec2f(0.f, 0.f),
126 scene_rdl2::math::Vec3f(0.f, 0.f, 0.f)}; // light filters don't apply to camera rays
127 scene_rdl2::math::Color contribution = hitLight->eval(tls, ray.getDirection(), ray.getOrigin(),
128 lightFilterR, ray.getTime(), hitLightIsects[i], true, nullptr, ray.getDirFootprint());
129 shading::LightContrib lightContrib(hitLight->getRdlLight(), scene_rdl2::math::luminance(contribution));
130 lightContributions.push_back(lightContrib);
131 }
132 if (!hitLights.empty() || !hitGeom) {
133 return;
134 }
135
136 // Early return if isect doesn't provide all the
137 // required attributes shader request
138 if (!isect.hasAllRequiredAttributes()) {
139 return;
140 }
141
142 // At this point we have hit a geometry that is not blocked by any visible lights.
143 // Now we get the light contribution on that geometry.
144
145 geom::initIntersectionPhase2(isect, tls, 0, 0, 0, true, scene_rdl2::math::Vec2f(0.0f), -ray.getDirection());
146
147 isect.setCameraId(0); // primary
148
149 float rayEpsilon = isect.getEpsilonHint();
150 if (rayEpsilon <= 0.0f) {
151 // Compute automatic ray-tracing bias
152 rayEpsilon = sHitEpsilonStart * scene_rdl2::math::max(ray.getEnd(), 1.0f);
153 }
154
155 // Transfer the ray to its intersection before we run shaders. This is
156 // needed for texture filtering based on ray differentials.
157 // Also scale the final differentials by a user factor. This is left until
158 // the very end and not baked into the ray differentials since the factor
159 // will typically be > 1, and would cause the ray differentials to be larger
160 // than necessary.
161 isect.transferAndComputeDerivatives(tls, &ray, textureFilterScale);
162
163 isect.setMediumIor(1.f);
164
165 const scene_rdl2::rdl2::Material* material = isect.getMaterial()->asA<scene_rdl2::rdl2::Material>();
166 MNRY_ASSERT(material != NULL);
167 shading::Bsdf bsdf;
168 shading::State state(&isect);
169 auto shadingTls = tls->mShadingTls.get();
170 shading::BsdfBuilder builder(bsdf, shadingTls, state);
171 shading::shade(material, shadingTls, state, builder);
172
173 // For bssrdf or bsdfs which contain both reflection and transmission lobes
174 // or is spherical, a single normal can't be used for culling so skip normal
175 // culling.
176 scene_rdl2::math::Vec3f normal(scene_rdl2::math::zero);
177 scene_rdl2::math::Vec3f *normalPtr = nullptr;
178 if (!bsdf.hasSubsurface() && !bsdf.getIsSpherical() &&
179 ((bsdf.getType() & shading::BsdfLobe::ALL_SURFACE_SIDES) != shading::BsdfLobe::ALL_SURFACE_SIDES)) {
180 normal = (bsdf.getType() & shading::BsdfLobe::REFLECTION) ? isect.getNg() : -isect.getNg();
181 normalPtr = &normal;
182 }
183
184 // Get the lights contributing to this pixel
185 LightSet lightSet;
186 bool hasRayTerminatorLights;
187 computeActiveLights(&tls->mArena, scene, isect, normalPtr, bsdf, /* rayTime = */ 0.f,
188 lightSet, hasRayTerminatorLights);
189
190 // Populate the array with the light and contribution data to be returned.
191 int count = lightSet.getLightCount();
192
193 shading::BsdfSlice slice(isect.getNg(), -ray.getDirection(), true,
194 isect.isEntering(), ispc::SHADOW_TERMINATOR_FIX_OFF, shading::BsdfLobe::ALL);
195 for (int i = 0; i < count; ++i) {
196 const Light* light = lightSet.getLight(i);
197 const LightFilterList* lightFilterList = lightSet.getLightFilterList(i);
198 scene_rdl2::math::Color contribution(scene_rdl2::math::sBlack);
199 for (int j = 0; j < numSamples; ++j) {
200
201 scene_rdl2::math::Vec3f r;
202 r[0] = float(rand())/float(RAND_MAX);
203 r[1] = float(rand())/float(RAND_MAX);
204 r[2] = float(rand())/float(RAND_MAX);
205
206 scene_rdl2::math::Vec3f wi;
207 LightIntersection sampleIsect;
208 if (!light->sample(isect.getP(), normalPtr, ray.getTime(), r, wi, sampleIsect, ray.getDirFootprint())) {
209 continue;
210 }
211
212 // Is this light sample blocked by geometry ?
213 mcrt_common::Ray shadowRay(isect.getP(), wi, rayEpsilon,
214 sampleIsect.distance * sHitEpsilonEnd,
215 ray.getTime(), ray.getDepth() + 1);
216 if (scene->isRayOccluded(tls, shadowRay)) {
217 continue;
218 }
219
220 // Evaluate and sum up contribution
221 float pdfLight;
222 LightFilterRandomValues lightFilterR = {
223 scene_rdl2::math::Vec2f(0.f, 0.f),
224 scene_rdl2::math::Vec3f(0.f, 0.f, 0.f)}; // light filters don't apply to camera rays
225 scene_rdl2::math::Color Li = light->eval(tls, wi, isect.getP(), lightFilterR, ray.getTime(), sampleIsect,
226 false, lightFilterList, ray.getDirFootprint(), &pdfLight);
227 if (pdfLight == 0.f) {
228 continue;
229 }
230 float pdfBsdf;
231 scene_rdl2::math::Color f = bsdf.eval(slice, wi, pdfBsdf);
232
233 scene_rdl2::math::Color color = f * Li / pdfLight;
234 contribution += color;
235 }
236
237 // Average the colors and keep track of light's contribution luminance
238 contribution /= numSamples;
239 shading::LightContrib lightContrib(light->getRdlLight(), luminance(contribution));
240 lightContributions.push_back(lightContrib);
241 }
242}
