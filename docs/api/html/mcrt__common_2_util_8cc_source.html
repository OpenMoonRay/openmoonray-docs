39void
40debugPrintCallstack(const char *contextString)
41{
42 static tbb::mutex mutex;
43
44 mutex.lock();
45
46 if (!contextString) contextString = "-- Callstack:\n";
47 std::printf("%s\n", contextString);
48
49 const int MAX_BACKTRACE_SYMBOLS = 64;
50 void *pointers[MAX_BACKTRACE_SYMBOLS];
51
52 size_t size = backtrace(pointers, MAX_BACKTRACE_SYMBOLS);
53 char **strings = backtrace_symbols(pointers, size);
54
55 char indent[MAX_BACKTRACE_SYMBOLS + 1];
56 memset(indent, ' ', sizeof(indent));
57
58 int wrapperIndex = -1;
59
60 // Iterate to size - 1 since final may not be null terminated.
61 for(size_t i = 0; i < size - 1; ++i) {
62 // Indent backtrace.
63 indent[i - wrapperIndex - 1] = 0;
64 char *p = strrchr(strings[i], '/');
65 std::printf("%s%s\n", indent, p ? (p + 1) : strings[i]);
66 indent[i - wrapperIndex - 1] = ' ';
67 }
68
69 std::printf("\n");
70 std::fflush(stdout);
71
72 free(strings);
73
74 mutex.unlock();
75}
