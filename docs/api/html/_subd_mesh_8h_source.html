60 bool initTextureSt()
61 {
62 mTextureRate = shading::RATE_UNKNOWN;
63 const shading::TypedAttributeKey<scene_rdl2::math::Vec2f>& stKey = shading::StandardAttributes::sSurfaceST;
64 if (!mPrimitiveAttributeTable.hasAttribute(stKey)) {
65 return false;
66 }
67 const shading::PrimitiveAttribute<scene_rdl2::math::Vec2f>& textureSt =
68 mPrimitiveAttributeTable.getAttribute(stKey);
69 mTextureVertices.resize(mVertices.size());
70 shading::AttributeRate stRate = textureSt.getRate();
71 switch (stRate) {
72 case shading::RATE_UNIFORM:
73 //case shading::RATE_PART: // Unsupported as faceToPart(i) not accessible here
74 case shading::RATE_FACE_VARYING: {
75 mTextureRate = shading::RATE_FACE_VARYING;
76 // This is exactly the same operation we did for other face varying
77 // attributes during FaceVaryingAttributes::addAttributeBuffer
78 std::vector<std::vector<FVarVertex<scene_rdl2::math::Vec2f>>> textureVertexGroups(
79 mVertices.size());
80 size_t indexOffset = 0;
81 size_t faceCount = mFaceVertexCount.size();
82 mTextureIndices.resize(mIndices.size());
83 for (size_t f = 0; f < faceCount; ++f) {
84 size_t nFv = mFaceVertexCount[f];
85 for (size_t v = 0; v < nFv; ++v) {
86 int vid = mIndices[indexOffset + v];
87 size_t i;
88 switch (stRate) {
89 case shading::RATE_UNIFORM: i = f; break;
90 //case shading::RATE_PART: i = faceToPart(f); break;
91 default: i = indexOffset + v; break;
92 }
93 const scene_rdl2::math::Vec2f& st = textureSt[i];
94 if (textureVertexGroups[vid].empty()) {
95 int stVid = vid;
96 textureVertexGroups[vid].emplace_back(st, stVid);
97 mTextureVertices[stVid] = st;
98 mTextureIndices[indexOffset + v] = stVid;
99 } else {
100 bool foundDuplicatedSt = false;
101 for (const auto& textureVertex :
102 textureVertexGroups[vid]) {
103 if (scene_rdl2::math::isEqual(textureVertex.mData, st)) {
104 mTextureIndices[indexOffset + v] =
105 textureVertex.mVertexId;
106 foundDuplicatedSt = true;
107 break;
108 }
109 }
110 if (!foundDuplicatedSt) {
111 int stVid = mTextureVertices.size();
112 mTextureVertices.push_back(st);
113 textureVertexGroups[vid].emplace_back(st, stVid);
114 mTextureIndices[indexOffset + v] = stVid;
115 }
116 }
117 }
118 indexOffset += nFv;
119 }
120 break;}
121 case shading::RATE_VARYING:
122 case shading::RATE_VERTEX:
123 mTextureRate = shading::RATE_VERTEX;
124 mTextureIndices = mIndices;
125 MNRY_ASSERT_REQUIRE(textureSt.size() >= mVertices.size());
126 for (size_t v = 0; v < mVertices.size(); ++v) {
127 mTextureVertices[v] = textureSt[v];
128 }
129 break;
130 case shading::RATE_CONSTANT:
131 mTextureRate = shading::RATE_VERTEX;
132 mTextureIndices = mIndices;
133 for (size_t v = 0; v < mVertices.size(); ++v) {
134 mTextureVertices[v] = textureSt[0];
135 }
136 break;
137 default:
138 mTextureRate = shading::RATE_UNKNOWN;
139 scene_rdl2::Logger::error("unsupported rate for surface st");
140 return false;
141 }
142 return true;
143 }

145 // this method is called when we do the reverse normal operation. Since the
146 // main control mesh index buffer got reversed, the corresponding
147 // texture index buffers will also need to be reversed
148 void reverseTextureIndices()
149 {
150 size_t indexOffset = 0;
151 for (size_t i = 0; i < mFaceVertexCount.size(); ++i) {
152 int nFv = mFaceVertexCount[i];
153 std::reverse(mTextureIndices.begin() + indexOffset,
154 mTextureIndices.begin() + indexOffset + nFv);
155 indexOffset += nFv;
156 }
157 }
