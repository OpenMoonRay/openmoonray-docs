38// convenience / clarity typedef's. Why both 8 & 64 bit pointers?
39// We have byte-wise offsets for some things (offsets in attrib keys)
40// and using an 8-bit pointer saves math in the pointer arithmetic
41typedef uniform int8* uniform PTR8;
42typedef uniform int64* uniform PTR64;
43
44// when ISPC de-references the attribute key pointers it needs to
45// get to the data contained therein. The current AttributeKey
46// type (class) cannot be digested by ISPC, so this proxy struct is used
47// instead
48struct AttributeKey {
49 uniform unsigned int mIndex;
50 uniform unsigned int mOffset;
51 uniform unsigned int mFlags;
52 uniform unsigned int mObjectType;
53};
54
55struct rdl2Ctx {
56 // an attribute doesn't have to be 64 bits in size,
57 // so we want to use an int8
58 PTR8 attributes;
59 PTR64 bindings;
60};

62inline uniform rdl2Ctx getRdl2Ctx(PTR8 sceneObj)
63{
64 uniform rdl2Ctx r;
65
66 // within the scene object that we are, find mAttributeStorage
67 PTR64 attribStoragePtr =
68 (PTR64)(sceneObj + SCENEOBJ_ATTRIB_OFFSET);
69
70 // deref pointer to get mAttributeStorage
71 r.attributes = (int8*)(*attribStoragePtr);
72
73 // find where (map) mBindings live
74 PTR64 uniform bindingsPtr =
75 (PTR64)(sceneObj + SCENEOBJ_BINDINGS_OFFSET);
76
77 // deref to get mBindings
78 r.bindings = (PTR64)(*bindingsPtr);
79
80 return r;
81}

84inline PTR64 getBinding(uniform rdl2Ctx& ctx,
85 uniform AttributeKey* uniform key)
86{
87
88 // within the bindings, get the pointer to the thing I am bound to
89 PTR64 ptrToMapObjPtr =
90 (PTR64)(ctx.bindings + key->mIndex);
91
92 // now de-ref to get my map object pointer
93 PTR64 mapObjPtr = (int64*)(*ptrToMapObjPtr);
94
95 // is there a mapping? There isn't if the mapping object pointer
96 // is null
97 if (mapObjPtr != NULL) {
98
99 // withing my map scene object, find my sample function pointer
100 PTR64 mapPtr =
101 (PTR64)*(mapObjPtr + (MAP_SAMPLEFUNC_OFFSET/8));
102
103 return mapPtr;
104 }
105 return NULL;
106}
