25/// @class IsotropicBsdfTable BsdfTable.h <pbr/BsdfTable.h>
26/// @brief An 3D isotropic bsdf lookup table that uses the (H, D)
27/// parameterization of the hemisphere. See the following paper for details:
28///
29/// A New Change of Variables for Efficient BRDF Representation
30/// Szymon Rusinkiewicz, Eurographics Workshop on Rendering 1998
31/// http://www.cs.princeton.edu/~smr/papers/brdf_change_of_variables/
32///

85
86 finline int getIndex(int indexThetaH, int indexThetaD, int indexPhiD) const {
87 MNRY_ASSERT(indexThetaH >= 0 && indexThetaH < mSizeThetaH);
88 MNRY_ASSERT(indexThetaD >= 0 && indexThetaD < mSizeThetaD);
89 MNRY_ASSERT(indexPhiD >= 0 && indexPhiD < mSizePhiD);
90 return 3 * (indexPhiD +
91 indexThetaD * mSizePhiD +
92 indexThetaH * mSizePhiD * mSizeThetaD);
93 }
94
95 finline scene_rdl2::math::Color getColor(int indexThetaH, int indexThetaD, int indexPhiD) const {
96 int index = getIndex(indexThetaH, indexThetaD, indexPhiD);
97 return scene_rdl2::math::Color(mData[index], mData[index + 1], mData[index + 2]);
98 }
