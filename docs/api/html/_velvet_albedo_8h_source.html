41 static void getAlbedoTableIndices(float cosTheta,
42 float roughness,
43 int& index1,
44 int& index2)
45 {
46 int rIndex1, rIndex2;
47 getAvgTableIndices(roughness,
48 rIndex1, rIndex2);
49
50 cosTheta = scene_rdl2::math::clamp(cosTheta, 0.0f, 1.0f);
51 const int cosIndex = cosTheta * ispc::VELVET_ALBEDO_COS_THETA_STEPS - 1;
52 const int albedoTableSize = ispc::VELVET_ALBEDO_TABLE_SIZE - 1;
53 index1 = scene_rdl2::math::clamp(rIndex1*ispc::VELVET_ALBEDO_COS_THETA_STEPS + cosIndex, 0, albedoTableSize);
54 index2 = scene_rdl2::math::clamp(rIndex2*ispc::VELVET_ALBEDO_COS_THETA_STEPS + cosIndex, 0, albedoTableSize);
55 }
