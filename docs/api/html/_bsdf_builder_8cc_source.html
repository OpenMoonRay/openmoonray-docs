133 finline void
134 stageAttenuator(const LobeAttenuator * attenuator)
135 {
136 MNRY_ASSERT(attenuator);
137 MNRY_ASSERT(mStagedAttenuatorCount < (BSDF_BUILDER_MAX_ATTENUATORS - 1));
138
139 mStagedAttenuators[mStagedAttenuatorCount] = attenuator;
140 ++mStagedAttenuatorCount;
141 }
142
143 // This function is called after adding an individual component,
144 // or a set of adjacent components. This allows any component that
145 // will somehow affect subsequent components to avoid affecting the
146 // adjacent components. This is achieved by 'staging' these effects
147 // as the adjacent components are added, and then finally moving these
148 // effects to an 'active' state, after which subsequent components will
149 // be affected.
150 finline void
151 accumulateAttenuation()
152 {
153 MNRY_ASSERT((mActiveAttenuatorCount + mStagedAttenuatorCount) <= BSDF_BUILDER_MAX_ATTENUATORS);
154
155 // Append any staged attenuators to the active list for subsequent BsdfLobes
156 for (size_t i = 0; i < mStagedAttenuatorCount; ++i) {
157 mActiveAttenuators[mActiveAttenuatorCount] = mStagedAttenuators[i];
158 ++mActiveAttenuatorCount;
159 }
160
161 // "Reset" the staging area (so to speak)
162 mStagedAttenuatorCount = 0;
163
164 // apply the recently accumulated weight to transmittance for
165 // attenuating subsequent lobes
166 mCurrentTransmittance *= (1.0f - mWeightAccum);
167 mWeightAccum = 0.f;
168 }

170 finline Fresnel*
171 createFresnel(const scene_rdl2::math::Color& eta,
172 const scene_rdl2::math::Color& k,
173 const bool isConductor)
174 {
175 if (isConductor) {
176 // Currently, Moonray does not respect the "media IOR" for conductors.
177 // Eventually we should compute *relative* IOR, using mCurrentMediumIor
178 return mTls->mArena->allocWithArgs<ConductorFresnel>(eta, k);
179 } else {
180 shading::ShaderIor ior(mState,
181 eta.r,
182 mIsThinGeo);
183 return mTls->mArena->allocWithArgs<DielectricFresnel>(ior.getIncident(),
184 ior.getTransmitted());
185 }
186 }

209 finline BsdfLobe *
210 placeUnderPreviousLobes(BsdfLobe * lobe)
211 {
212 MNRY_ASSERT(lobe);
213
214 if (mActiveAttenuatorCount > 0) {
215 int idx = mActiveAttenuatorCount - 1;
216 while (idx >= 0) {
217 const LobeAttenuator& attenuate = *(mActiveAttenuators[idx--]);
218 lobe = attenuate(mTls->mArena, lobe);
219 }
220 }
221
222 return lobe;
223 }

280 finline void
281 addComponent(const MicrofacetAnisotropicClearcoat& component,
282 float weight,
283 ispc::BsdfBuilderBehavior combineBehavior,
284 const int label)
285 {
286 const float roughnessU = component.getRoughnessU();
287 const float roughnessV = component.getRoughnessV();
288 const float averageRoughness = (roughnessU + roughnessV) * 0.5f;
289
290 // Adapt normal to prevent reflection ray from self-intersecting this geometry
291 const scene_rdl2::math::Vec3f adaptedNormal = mState.adaptNormal(component.getN());
292
293 // We currently only support anisotropy with Beckmann
294 BsdfLobe * lobe =
295 mTls->mArena->allocWithArgs<AnisoCookTorranceBsdfLobe>(adaptedNormal,
296 component.getShadingTangent(),
297 roughnessU,
298 roughnessV);
299
300 Fresnel * fresnel = applyFresnelBehavior(scene_rdl2::math::Color(component.getEta()),
301 scene_rdl2::math::sBlack, // k
302 false, // isConductor
303 weight,
304 lobe);
305
306 lobe = handleIridescence(lobe, component.getIridescence());
307
308 scene_rdl2::math::Color scale = scene_rdl2::math::sWhite;
309
310 if (isUnder(combineBehavior)) {
311 // account for the non-dielectric lobes above
312 scale *= mCurrentTransmittance;
313 lobe->setScale(scale);
314 // account for dielectric/clearcoat lobes above
315 lobe = placeUnderPreviousLobes(lobe);
316 } else {
317 lobe->setScale(scale);
318 }
319
320 if (isOver(combineBehavior)) {
321 // The Attenuation Effect Diminishes with Weight
322 const scene_rdl2::math::Color
323 attenuationColor = lerp(scene_rdl2::math::sWhite,
324 component.getAttenuationColor(),
325 weight);
326 ClearcoatAttenuator * attenuator =
327 mTls->mArena->allocWithArgs<ClearcoatAttenuator>(
328 mTls->mArena,
329 adaptedNormal,
330 averageRoughness,
331 mCurrentMediumIor,
332 component.getEta(),
333 component.getRefracts(),
334 component.getThickness() * weight,
335 attenuationColor,
336 fresnel);
337
338 stageAttenuator(attenuator);
339 }
340
341 lobe->setLabel(label);
342 mBsdf.addLobe(lobe);
343 }

671 {
672 // Switch to Lambertian diffuse if max_subsurface_per_path has been reached
673 if (!mState.isSubsurfaceAllowed()) {
674 LambertianBRDF diffuse(component.getN(),
675 component.getAlbedo());
676 addComponent(diffuse,
677 weight,
678 combineBehavior,
679 label);
680 return;

682
683 auto bssrdf = createBSSRDF(ispc::SUBSURFACE_DIPOLE_DIFFUSION,
684 mTls->mArena,
685 component.getN(),
686 component.getAlbedo(),
687 component.getRadius(),
688 component.getMaterial(),
689 component.getEvalNormalFn());
690
691 scene_rdl2::math::Color scale(weight);
692
693 if (isUnder(combineBehavior)) {
694 // account for the non-dielectric lobes above
695 scale *= mCurrentTransmittance;
696 bssrdf->setScale(scale);
697
698 // setup a MultipleTransmissionFresnel so this bssrdf can
699 // be attenuated by previous lobes
700 MultipleTransmissionFresnel * fresnel =
701 mTls->mArena->allocWithCtor<MultipleTransmissionFresnel>();
702 bssrdf->setTransmissionFresnel(fresnel);
703
704 // account for dielectric lobes above (clearcoat not supported for bssrdfs)
705 placeUnderPreviousLobes(bssrdf);
706 } else {
707 bssrdf->setScale(scale);
708 }
709
710 if (isOver(combineBehavior)) {
711 // account for this lobe's energy allocation
712 mWeightAccum += weight;
713 }
714
715 bssrdf->setTraceSet(component.getTraceSet());
716 bssrdf->setLabel(label);
717 mBsdf.setBssrdf(bssrdf);
718 }
