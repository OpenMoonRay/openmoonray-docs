32finline void
33PathIntegrator::addDirectVisibleBsdfLobeSampleContribution(pbr::TLState *pbrTls,
34 const Subpixel &sp, int cameraId, const PathVertex &pv,
35 const BsdfSampler &bSampler, int lobeIndex, bool doIndirect, const BsdfSample &bsmp,
36 const mcrt_common::RayDifferential &parentRay, float rayEpsilon, float shadowRayEpsilon,
37 scene_rdl2::math::Color &radiance, unsigned& sequenceID, float *aovs) const
38{
39 MNRY_ASSERT(pbrTls->isIntegratorAccumulatorRunning());
40 // This matches the behavior in the vectorized codepath.
41 if (isBlack(bsmp.tDirect)) {
42 return;
43 }
44
45 const Light *light = bsmp.lp.light;
46 const shading::BsdfLobe &lobe = *bSampler.getLobe(lobeIndex);
47
48 // Create a ray using the origin of the parent ray.
49 // Note the origin is actually the point on the surface that
50 // the parent ray hit, not the original origin of the parent ray.
51 // It has been moved there when the intersection was processed.
52 // The direction vector to the light is bsmp[s].wi, so use that
53 // for the direction and the differential directions.
54 // bsmp[s].distance is the distance from the surface point to
55 // the light, so we use that for the ray length minus an
56 // epsilon factor.
57 const scene_rdl2::math::Vec3f& P = parentRay.getOrigin();
58 float tfar = bsmp.distance * sHitEpsilonEnd;
59 float time = parentRay.getTime();
60 int rayDepth = parentRay.getDepth() + 1;
61 float presence = 0.0f;
62 scene_rdl2::math::Color contrib(0.0f);
63
64 mcrt_common::RayDifferential shadowRay(P, bsmp.wi,
65 parentRay.getOriginX(), bsmp.wi,
66 parentRay.getOriginY(), bsmp.wi,
67 rayEpsilon, tfar, time, rayDepth);
68
69 // Ray termination lights are used in an attempt to cheaply fill in the zeros which result from
70 // terminating ray paths too early, which is done by forcing no occlusion.
71 // A path can only be terminated when doIndirect is false, but don't apply a ray termination light
72 // unless we have a (non-hair) transmission lobe.
73 bool isOccluded;
74 if (!doIndirect && light->getIsRayTerminator()) {
75 if (!lobe.matchesFlags(shading::BsdfLobe::ALL_TRANSMISSION) || lobe.getIsHair()) {
76 return;
77 } else {
78 isOccluded = false;
79 }
80 } else {
81 isOccluded = isRayOccluded(pbrTls, light, shadowRay, rayEpsilon, shadowRayEpsilon, presence);
82 }
83 if (!isOccluded) {
84 // We can't reuse shadowRay because it can be modified in occlusion query
85 mcrt_common::Ray trRay(P, bsmp.wi, scene_rdl2::math::max(rayEpsilon, shadowRayEpsilon), tfar, time, rayDepth);
86 scene_rdl2::math::Color tr = transmittance(pbrTls, trRay, sp.mPixel, sp.mSubpixelIndex, sequenceID, light);
87 contrib = tr * bsmp.tDirect * (1.0f - presence);
88 radiance += contrib;
89 }
90
91 // LPE
92 if (aovs) {
93 EXCL_ACCUMULATOR_PROFILE(pbrTls, EXCL_ACCUM_AOVS);
94 const FrameState &fs = *pbrTls->mFs;
95 const AovSchema &aovSchema = *fs.mAovSchema;
96 const LightAovs &lightAovs = *fs.mLightAovs;
97 // transition
98 int lpeStateId = pv.lpeStateId;
99 lpeStateId = lightAovs.scatterEventTransition(pbrTls,
100 lpeStateId, bSampler.getBsdf(), lobe);
101 lpeStateId = lightAovs.lightEventTransition(pbrTls, lpeStateId, light);
102 // accumulate matching aovs
103 if (fs.mAovSchema->hasLpePrefixFlags(AovSchema::sLpePrefixUnoccluded)) {
104 aovAccumLightAovs(pbrTls, aovSchema, cameraId, lightAovs, bsmp.tDirect,
105 &contrib, AovSchema::sLpePrefixUnoccluded, lpeStateId, aovs);
106 } else {
107 aovAccumLightAovs(pbrTls, aovSchema, cameraId, lightAovs, contrib,
108 nullptr, AovSchema::sLpePrefixNone, lpeStateId, aovs);
109 }
110 }
111
112 // TODO: we don't store light intersection normal when a bsdf sample
113 // ends up hitting a light. We don't store Li either.
114 // If the intersection distance is closer than the distant light, then
115 // assume the hit wasn't due to a distant or env light.
116 if (DebugRayRecorder::isRecordingEnabled()) {
117 if (bsmp.distance < sDistantLightDistance) {
118 mcrt_common::Ray debugRay(parentRay.getOrigin(), bsmp.wi, 0.0f);
119 RAYDB_EXTEND_RAY_NO_HIT(pbrTls, debugRay, bsmp.distance);
120 RAYDB_SET_CONTRIBUTION(pbrTls, scene_rdl2::math::sWhite);
121 RAYDB_ADD_TAGS(pbrTls, TAG_AREALIGHT);
122 } else {
123 mcrt_common::Ray debugRay(parentRay.getOrigin(), bsmp.wi, 0.0f);
124 RAYDB_EXTEND_RAY_NO_HIT(pbrTls, debugRay, 40.0f);
125 RAYDB_SET_CONTRIBUTION(pbrTls, scene_rdl2::math::sWhite);
126 RAYDB_ADD_TAGS(pbrTls, TAG_ENVLIGHT);
127 }
128 }
129}

132void
133PathIntegrator::addDirectVisibleBsdfSampleContributions(pbr::TLState *pbrTls,
134 const Subpixel &sp, int cameraId, const PathVertex &pv,
135 const BsdfSampler &bSampler, bool doIndirect, const BsdfSample *bsmp,
136 const mcrt_common::RayDifferential &parentRay, float rayEpsilon, float shadowRayEpsilon,
137 scene_rdl2::math::Color &radiance, unsigned& sequenceID, float *aovs) const
138{
139 // Trace bsdf sample shadow rays
140 int s = 0;
141 const int lobeCount = bSampler.getLobeCount();
142 for (int lobeIndex = 0; lobeIndex < lobeCount; ++lobeIndex) {
143 const int lobeSampleCount = bSampler.getLobeSampleCount(lobeIndex);
144 for (int i = 0; i < lobeSampleCount; ++i, ++s) {
145 if (bsmp[s].isValid() && bsmp[s].didHitLight()) {
146 addDirectVisibleBsdfLobeSampleContribution(pbrTls, sp, cameraId, pv, bSampler,
147 lobeIndex, doIndirect, bsmp[s], parentRay, rayEpsilon, shadowRayEpsilon,
148 radiance, sequenceID, aovs);
149 }
150 }
151 }

154void
155PathIntegrator::addDirectVisibleLightSampleContributions(pbr::TLState *pbrTls,
156 Subpixel const& sp, int cameraId, const PathVertex &pv,
157 const LightSetSampler &lSampler, const LightSample *lsmp,
158 const mcrt_common::RayDifferential &parentRay, float rayEpsilon, float shadowRayEpsilon,
159 scene_rdl2::math::Color &radiance, unsigned& sequenceID, float *aovs) const
160{
161 MNRY_ASSERT(pbrTls->isIntegratorAccumulatorRunning());
162 // Trace light sample shadow rays
163 int s = 0;
164 const int lightCount = lSampler.getLightCount();
165 const int lightSampleCount = lSampler.getLightSampleCount();
166 for (int lightIndex = 0; lightIndex < lightCount; ++lightIndex) {
167 const Light *light = lSampler.getLight(lightIndex);
168
169 // Ray termination lights are used in an attempt to cheaply fill in the zeros which result from
170 // terminating ray paths too early. We exclude them from light samples because these samples represent
171 // natural ends to light paths.
172 if (light->getIsRayTerminator()) {
173 s += lightSampleCount;
174 continue;
175 }
176
177 for (int i = 0; i < lightSampleCount; ++i, ++s) {
178 if (lsmp[s].isInvalid()) {
179 continue;
180 }
181 scene_rdl2::math::Color lightT = lsmp[s].t;
182 // This matches the behavior in the vectorized codepath.
183 if (isBlack(lightT)) {
184 continue;
185 }
186 const scene_rdl2::math::Vec3f &P = parentRay.getOrigin();
187 float tfar = lsmp[s].distance * sHitEpsilonEnd;
188 float time = parentRay.getTime();
189 int rayDepth = parentRay.getDepth() + 1;
190 mcrt_common::RayDifferential shadowRay(P, lsmp[s].wi,
191 parentRay.getOriginX(), lsmp[s].wi,
192 parentRay.getOriginY(), lsmp[s].wi,
193 rayEpsilon, tfar, time, rayDepth);
194 float presence = 0.0f;
195 scene_rdl2::math::Color tr;
196
197 const FrameState &fs = *pbrTls->mFs;
198 const bool hasUnoccludedFlag = fs.mAovSchema->hasLpePrefixFlags(AovSchema::sLpePrefixUnoccluded);
199 if (isRayOccluded(pbrTls, light, shadowRay, rayEpsilon, shadowRayEpsilon, presence)) {
200 // Calculate clear radius falloff
201 // only do extra calculations if clear radius falloff enabled
202 if (light->getClearRadiusFalloffDistance() != 0.f &&
203 tfar < light->getClearRadius() + light->getClearRadiusFalloffDistance()) {
204 // compute unoccluded pixel value
205 lightT *= (1.0f - presence);
206 mcrt_common::Ray trRay(P, lsmp[s].wi, scene_rdl2::math::max(rayEpsilon, shadowRayEpsilon), tfar, time, rayDepth);
207 tr = transmittance(pbrTls, trRay, sp.mPixel, sp.mSubpixelIndex, sequenceID, light);
208
209 radiance += calculateShadowFalloff(light, tfar, tr * lightT);
210 }
211
212 // Visibility LPE
213 // A ray is occluded, we must record that for the shadow aov.
214 if (aovs) {
215 EXCL_ACCUMULATOR_PROFILE(pbrTls, EXCL_ACCUM_AOVS);
216 const LightAovs &lightAovs = *fs.mLightAovs;
217
218 // If there is no visibility AOV and if we don't have unoccluded flags, then we don't need to bother
219 // with accumulating these values here.
220 if (lightAovs.hasVisibilityEntries() || hasUnoccludedFlag) {
221
222 const AovSchema &aovSchema = *fs.mAovSchema;
223 const Light *light = lSampler.getLight(lightIndex);
224 bool addVisibility = true;
225 for (unsigned l = 0; l < shading::Bsdf::maxLobes; ++l) {
226 if (!lsmp[s].lp.lobe[l]) continue;
227
228 const shading::BsdfLobe &lobe = *lsmp[s].lp.lobe[l];
229 const scene_rdl2::math::Color &unoccludedLobeVal = lsmp[s].lp.t[l];
230 int lpeStateId = pv.lpeStateId;
231 // transition
232 lpeStateId = lightAovs.scatterEventTransition(pbrTls,
233 lpeStateId, lSampler.getBsdf(), lobe);
234 lpeStateId = lightAovs.lightEventTransition(pbrTls,
235 lpeStateId, light);
236
237 // visibility aov
238 if (addVisibility) {
239 if (aovAccumVisibilityAovs(pbrTls, aovSchema, cameraId, lightAovs,
240 scene_rdl2::math::Vec2f(0.0f, 1.0f), lpeStateId, aovs)) {
241 // add visibility aov at most once per shadow ray
242 addVisibility = false;
243 }
244 }
245
246 // unoccluded prefix LPEs
247 if (hasUnoccludedFlag) {
248 // If it's occluded but we have the unoccluded flag set, only contribute this to any
249 // pre-occlusion aovs.
250 aovAccumLightAovs(pbrTls, aovSchema, cameraId, lightAovs, unoccludedLobeVal, nullptr,
251 AovSchema::sLpePrefixUnoccluded, lpeStateId, aovs);
252 }
253 }
254 }
255 }
256 } else {
257 // Take into account presence and transmittance
258 lightT *= (1.0f - presence);
259 mcrt_common::Ray trRay(P, lsmp[s].wi, scene_rdl2::math::max(rayEpsilon, shadowRayEpsilon), tfar, time, rayDepth);
260 tr = transmittance(pbrTls, trRay, sp.mPixel, sp.mSubpixelIndex, sequenceID, light);
261 radiance += tr * lightT;
262
263 // LPE
264 if (aovs) {
265 EXCL_ACCUMULATOR_PROFILE(pbrTls, EXCL_ACCUM_AOVS);
266 const AovSchema &aovSchema = *fs.mAovSchema;
267 const LightAovs &lightAovs = *fs.mLightAovs;
268 const Light *light = lSampler.getLight(lightIndex);
269 bool addVisibility = true;
270 for (unsigned l = 0; l < shading::Bsdf::maxLobes; ++l) {
271 if (!lsmp[s].lp.lobe[l]) continue;
272 const shading::BsdfLobe &lobe = *lsmp[s].lp.lobe[l];
273 const scene_rdl2::math::Color &unoccludedLobeVal = lsmp[s].lp.t[l];
274 const scene_rdl2::math::Color &lobeVal = tr * lsmp[s].lp.t[l] * (1.0f - presence);
275 const shading::Bsdf &bsdf = lSampler.getBsdf();
276 // transition
277 int lpeStateId = pv.lpeStateId;
278 lpeStateId = lightAovs.scatterEventTransition(pbrTls,
279 lpeStateId, bsdf, lobe);
280 lpeStateId = lightAovs.lightEventTransition(pbrTls,
281 lpeStateId, light);
282
283 // Accumulate aovs depending on whether or not the unoccluded flag is set.
284 if (hasUnoccludedFlag) {
285 // If the unoccluded flag is set we have to add occluded and unoccluded
286 // (without presence and volume transmittance) separately.
287 aovAccumLightAovs(pbrTls, aovSchema, cameraId, lightAovs, unoccludedLobeVal, &lobeVal,
288 AovSchema::sLpePrefixUnoccluded, lpeStateId, aovs);
289 } else {
290 // Otherwise, just add the contribution to all non-pre-occlusion aovs.
291 aovAccumLightAovs(pbrTls, aovSchema, cameraId, lightAovs, lobeVal, nullptr,
292 AovSchema::sLpePrefixNone, lpeStateId, aovs);
293 }
294
295 // visibility aov
296 if (addVisibility) {
297 if (aovAccumVisibilityAovs(pbrTls, aovSchema, cameraId, lightAovs,
298 scene_rdl2::math::Vec2f(reduceTransparency(tr) * (1 - presence), 1.0f),
299 lpeStateId, aovs)) {
300 // add visibility aov at most once per shadow ray
301 addVisibility = false;
302 }
303 }
304 }
305 }
306
307 // TODO: we don't store light sample normal
308 // If the intersection distance is closer than the distant light, then
309 // assume the hit wasn't due to a distant or env light.
310 if (DebugRayRecorder::isRecordingEnabled()) {
311 if (lsmp[s].distance < sDistantLightDistance) {
312 mcrt_common::Ray debugRay(P, lsmp[s].wi, 0.0f);
313 RAYDB_EXTEND_RAY_NO_HIT(pbrTls, debugRay, lsmp[s].distance);
314 RAYDB_SET_CONTRIBUTION(pbrTls, lsmp[s].Li);
315 RAYDB_ADD_TAGS(pbrTls, TAG_AREALIGHT);
316 } else {
317 mcrt_common::Ray debugRay(P, lsmp[s].wi, 0.0f);
318 RAYDB_EXTEND_RAY_NO_HIT(pbrTls, debugRay, 40.0f);
319 RAYDB_SET_CONTRIBUTION(pbrTls, lsmp[s].Li);
320 RAYDB_ADD_TAGS(pbrTls, TAG_ENVLIGHT);
321 }
322 }
323 }
324 }
325 }
326}

329void
330PathIntegrator::addIndirectOrDirectVisibleContributions(
331 pbr::TLState *pbrTls,
332 const Subpixel &sp, int cameraId, const PathVertex &parentPv, const BsdfSampler &bSampler,
333 const BsdfSample *bsmp, const mcrt_common::RayDifferential &parentRay,
334 float rayEpsilon, float shadowRayEpsilon,
335 const shading::Intersection &isect, shading::BsdfLobe::Type indirectFlags,
336 const scene_rdl2::rdl2::Material* newPriorityList[4], int newPriorityListCount[4],
337 scene_rdl2::math::Color &radiance, unsigned& sequenceID,
338 float *aovs) const
339{
340 MNRY_ASSERT(pbrTls->isIntegratorAccumulatorRunning());
341
342 scene_rdl2::math::Vec3f wo = -parentRay.getDirection();
343
344 // Trace bsdf sample continuation rays. We accumulate either direct or
345 // indirect lighting contributions accordingly
346 int s = 0;
347 const int lobeCount = bSampler.getLobeCount();
348 for (int lobeIndex = 0; lobeIndex < lobeCount; ++lobeIndex) {
349 const shading::BsdfLobe *lobe = bSampler.getLobe(lobeIndex);
350
351 // hair lobe type is a special case that often requires more bounces.
352 const bool hairLobe = lobe->getIsHair();
353 const bool doIndirect = lobe->matchesFlags(indirectFlags)
354 || (hairLobe && (parentPv.hairDepth < mMaxHairDepth));
355 const bool diffuseLobe = lobe->matchesFlags(shading::BsdfLobe::ALL_DIFFUSE);
356 const bool glossyLobe = lobe->matchesFlags(shading::BsdfLobe::ALL_GLOSSY);
357 const bool mirrorLobe = lobe->matchesFlags(shading::BsdfLobe::ALL_MIRROR);
358 const bool transmissionLobe = lobe->matchesFlags(shading::BsdfLobe::ALL_TRANSMISSION);
359
360 const scene_rdl2::math::Vec2f minRoughness = (doIndirect ? computeMinRoughness(*lobe,
361 mRoughnessClampingFactor, parentPv.minRoughness) : scene_rdl2::math::Vec2f(0.0f));
362
363 const int lobeSampleCount = bSampler.getLobeSampleCount(lobeIndex);
364 for (int i = 0; i < lobeSampleCount; ++i, ++s) {
365 if (bsmp[s].isInvalid()) {
366 continue;
367 }
368
369 // If this sample hit a light, compute direct lighting contribution only
370 if (bsmp[s].didHitLight()) {
371 addDirectVisibleBsdfLobeSampleContribution(pbrTls, sp, cameraId, parentPv, bSampler,
372 lobeIndex, doIndirect, bsmp[s], parentRay, rayEpsilon, shadowRayEpsilon,
373 radiance, sequenceID, aovs);
374 }
375
376 if (!doIndirect) {
377 continue;
378 }
379
380 // We have some self-intersections when rays leave at grazing
381 // angle, so we adjust the rayEpsilon accordingly.
382 // We only trace up to the bsmp[s].distance. It is set to the distance
383 // to the intersected light in this direction, if any.
384 const float denom = scene_rdl2::math::abs(dot(isect.getNg(), bsmp[s].wi));
385 // isect.getNg() itself or the dot product above can be zero.
386 const float start = scene_rdl2::math::isZero(denom) ? rayEpsilon : rayEpsilon / denom;
387 float end = (bsmp[s].distance == scene_rdl2::math::sMaxValue ? scene_rdl2::math::sMaxValue :
388 bsmp[s].distance * sHitEpsilonEnd);
389 if (end <= start) {
390 continue;
391 }
392
393 // Check transparency threshold
394 float newAccumOpacity;
395 if (mirrorLobe && lobe->matchesFlags(shading::BsdfLobe::ALL_TRANSMISSION)) {
396 float lobeTransparency = reduceTransparency(bsmp[s].f);
397 newAccumOpacity = parentPv.accumOpacity + (1 - lobeTransparency) * (1 - parentPv.accumOpacity);
398 if (newAccumOpacity > mTransparencyThreshold) {
399 continue;
400 }
401 } else {
402 newAccumOpacity = parentPv.accumOpacity;
403 }
404
405 // Prepare a path vertex
406 PathVertex pv;
407 pv.pathThroughput = bsmp[s].tIndirect;
408 pv.pathPixelWeight = 0.0f;
409 // Use previous path pixel weight for aovPathPixelWeight as there's existing logic
410 // in vector mode that sometimes assumes that pv.pathPixelWeight = 0. Thus, we must seperately
411 // keep track of the pathPixelWeight for aovs. See comment in PathIntegratorMultiSampler.ispc::
412 // addIndirectOrDirectVisibleContributionsBundled().
413 pv.aovPathPixelWeight = parentPv.pathPixelWeight;
414 pv.pathDistance = parentPv.pathDistance + parentRay.getEnd();
415 pv.minRoughness = minRoughness;
416 pv.diffuseDepth = parentPv.diffuseDepth + (diffuseLobe ? 1 : 0);
417 pv.glossyDepth = parentPv.glossyDepth + (glossyLobe ? 1 : 0);
418 pv.mirrorDepth = parentPv.mirrorDepth + (mirrorLobe ? 1 : 0);
419 pv.nonMirrorDepth = parentPv.nonMirrorDepth + (mirrorLobe ? 0 : 1);
420 pv.hairDepth = parentPv.hairDepth + (hairLobe ? 1 : 0);
421 pv.volumeDepth = parentPv.volumeDepth + 1;
422 pv.presenceDepth = parentPv.presenceDepth;
423 pv.totalPresence = parentPv.totalPresence;
424 pv.subsurfaceDepth = parentPv.subsurfaceDepth;
425 pv.accumOpacity = newAccumOpacity;
426 pv.lobeType = lobe->getType();
427
428 // LPE
429 if (aovs) {
430 const FrameState &fs =*pbrTls->mFs;
431 const LightAovs &lightAovs = *fs.mLightAovs;
432 const shading::Bsdf &bsdf = bSampler.getBsdf();
433
434 // transition
435 pv.lpeStateId = lightAovs.scatterEventTransition(pbrTls, parentPv.lpeStateId, bsdf, *lobe);
436
437 // Accumulate post scatter extra aovs
438 aovAccumPostScatterExtraAovs(pbrTls, fs, pv, cameraId, bsdf, aovs);
439 }
440 // Prepare a mcrt_common::RayDifferential
441 mcrt_common::RayDifferential ray(parentRay, start, end);
442
443 if (transmissionLobe) {
444 // copy in the new priority list into the ray
445 setPriorityList(ray, newPriorityList, newPriorityListCount);
446 }
447
448 // Scatter and scale our next ray differential
449 // We scatter the ray based on the sampled lobe
450 scatterAndScale(isect, *lobe, wo, bsmp[s].wi,
451 ((lobe->getDifferentialFlags() & shading::BsdfLobe::IGNORES_INCOMING_DIFFERENTIALS) ?
452 sp.mPrimaryRayDiffScale : 1.0f),
453 bsmp[s].sample[0], bsmp[s].sample[1], ray);
454
455 CHECK_CANCELLATION(pbrTls, return);
456
457 // Recurse
458 scene_rdl2::math::Color radianceIndirect;
459 float transparencyIndirect;
460 // the volume attenuation along this ray to the first hit (or infinity)
461 VolumeTransmittance vtIndirect;
462 ++sequenceID;
463 bool hitVolume;
464
465 IndirectRadianceType indirectRadianceType = computeRadianceRecurse(
466 pbrTls, ray, sp, cameraId,
467 pv, lobe, radianceIndirect, transparencyIndirect,
468 vtIndirect, sequenceID, aovs, nullptr, nullptr, nullptr, false, hitVolume);
469
470 if (indirectRadianceType != NONE) {
471 // Accumulate indirect lighting contribution
472 // In the bundled version, this contribution gets accounted for by
473 // queueing a BundledRadiance for the direct radiance and/or emission,
474 // and spawning one or more new rays for the indirect radiance.
475 radiance += radianceIndirect;
476
477 // path guiding is trained on indirect radiance contributions.
478 // it turns out that including any other radiance contributions in the
479 // training causes the path guide to overwhelmingly favor direct lighting
480 // directions. this is the exact opposite of what we want. we are trying
481 // to build a distribution that favors important indirect lighting.
482 mPathGuide.recordRadiance(parentRay.getOrigin(), bsmp[s].wi, radianceIndirect);
483 }
484 if (!bsmp[s].didHitLight()) {
485 const FrameState &fs = *pbrTls->mFs;
486 const AovSchema &aovSchema = *fs.mAovSchema;
487 const LightAovs &lightAovs = *fs.mLightAovs;
488 if (fs.mAovSchema->hasLpePrefixFlags(AovSchema::sLpePrefixUnoccluded)) {
489 // transition
490 int lpeStateId = pv.lpeStateId;
491 lpeStateId = lightAovs.lightEventTransition(pbrTls,
492 lpeStateId, bsmp[s].lp.light);
493 // accumulate matching aovs.
494 aovAccumLightAovs(pbrTls, aovSchema, cameraId, lightAovs,
495 bsmp[s].tDirect, nullptr, AovSchema::sLpePrefixNone, lpeStateId, aovs);
496 }
497 }
498 }
499 }
500}

502scene_rdl2::math::Color
503PathIntegrator::computeRadianceBsdfMultiSampler(pbr::TLState *pbrTls,
504 const Subpixel &sp, int cameraId, const PathVertex &pv, const mcrt_common::RayDifferential &ray,
505 const shading::Intersection &isect, const shading::Bsdf &bsdf, const shading::BsdfSlice &slice,
506 bool doIndirect, const shading::BsdfLobe::Type indirectFlags, const scene_rdl2::rdl2::Material *newPriorityList[4],
507 int newPriorityListCount[4], const LightSet &activeLightSet, const scene_rdl2::math::Vec3f *cullingNormal,
508 float rayEpsilon, float shadowRayEpsilon,
509 const scene_rdl2::math::Color &ssAov, unsigned &sequenceID, float *aovs) const
510{
511 // TODO:
512 // A couple things seem out of place with the parameters to function.
513 //
514 // First, why do we require both a doIndirect boolean and an indirectFlags?
515 // Ideally we could determine doIndirect from the indirecFlags. But there is
516 // some complication with hair depth that prevents this.
517 //
518 // Second, the ssAov parameter is unfortunate. It is just passed along to the
519 // material aov packing function. I think it would be best to pack the material
520 // aovs before calling this function. But that would require the elimination of
521 // the "albedo" material aov - which relies on the bsdf samples. That might actually
522 // be a good thing.
523
524 // In case it isn't clear, the outputs of this function are the returned radiance
525 // and accumulated aovs. The sequenceID is both and input
526 // and an output.
527 scene_rdl2::math::Color radiance = scene_rdl2::math::sBlack;
528
529 // We use a BsdfSampler object to keep track of sampling strategies and
530 // sample budget per lobe.
531 // We only want to split on the first scattering event seen from the
532 // camera (either directly, either through one or many mirror bounces)
533 const int maxSamplesPerLobe = (pv.nonMirrorDepth == 0 ? mBsdfSamples :
534 scene_rdl2::math::min(mBsdfSamples, 1));
535
536 scene_rdl2::alloc::Arena *arena = pbrTls->mArena;
537
538 // TODO: I feel that the bsdf is conceptually const at this point
539 // I don't know why or for what reason the bsdf sampler needs a non-const bsdf
540 shading::Bsdf &constCastBsdf = const_cast<shading::Bsdf &>(bsdf);
541
542 BsdfSampler bSampler(arena, constCastBsdf, slice, maxSamplesPerLobe, doIndirect, mPathGuide);
543
544 const int bsdfSampleCount = bSampler.getSampleCount();
545 BsdfSample *bsmp = arena->allocArray<BsdfSample>(bsdfSampleCount);
546
547 // We use a LightSetSampler object to keep track of sampling strategies and
548 // sample budget per light.
549 // We use the same splitting strategy as for lobes above.
550 const int maxSamplesPerLight = (pv.nonMirrorDepth == 0 ? mLightSamples :
551 scene_rdl2::math::min(mLightSamples, 1));
552 LightSetSampler lSampler(arena, activeLightSet, bsdf, isect.getP(), maxSamplesPerLight);
553
554 const int lightSetSampleCount = lSampler.getSampleCount();
555 LightSample *lsmp = arena->allocArray<LightSample>(lightSetSampleCount);
556
557 // Draw Bsdf and LightSet samples and compute tentative contributions.
558 drawBsdfSamples(pbrTls, bSampler, lSampler, sp, pv, isect.getP(), cullingNormal,
559 ray.getTime(), sequenceID, bsmp, mSampleClampingDepth,
560 sp.mSampleClampingValue, indirectFlags, ray.getDirFootprint());
561
562 // now that we have drawn our bsdf samples, we can fully
563 // pack our material aovs
564 if (aovs) {
565 const FrameState &fs = *pbrTls->mFs;
566 const AovSchema &aovSchema = *fs.mAovSchema;
567 const Scene *scene = MNRY_VERIFY(pbrTls->mFs->mScene);
568 aovSetMaterialAovs(pbrTls, aovSchema, cameraId, *fs.mLightAovs, *fs.mMaterialAovs,
569 isect, ray, *scene, bsdf, ssAov,
570 &bSampler, bsmp, pv.aovPathPixelWeight, pv.lpeStateId, aovs);
571 }
572
573 drawLightSetSamples(pbrTls, lSampler, bSampler, sp, pv, isect.getP(), cullingNormal, ray.getTime(),
574 sequenceID, lsmp, mSampleClampingDepth, sp.mSampleClampingValue, ray.getDirFootprint());
575
576 CHECK_CANCELLATION(pbrTls, return scene_rdl2::math::sBlack );
577
578 //---------------------------------------------------------------------
579 // Apply Russian Roulette (RR). Note we only do RR past a non-mirror
580 // bounce, to avoid breaking the nice stratification of samples on the
581 // first non-mirror hit.
582 // TODO: We should find a way to use efficiency optimized RR. For this we
583 // need an estimate of the final pixel color so our threshold can be
584 // computed accordingly.
585 if (pv.nonMirrorDepth > 0 && mRussianRouletteThreshold > 0.0f) {
586 applyRussianRoulette(bSampler, bsmp, sp, pv, sequenceID,
587 mRussianRouletteThreshold, mInvRussianRouletteThreshold);
588 applyRussianRoulette(lSampler, lsmp, sp, pv, sequenceID,
589 mRussianRouletteThreshold, mInvRussianRouletteThreshold);
590 }
591
592 CHECK_CANCELLATION(pbrTls, return scene_rdl2::math::sBlack );
593
594 //---------------------------------------------------------------------
595 // Let's trace some rays to resolve visibility / shadowing and sum up
596 // contributions. We trace one ray per valid sample and affect all
597 // contributions for that sample accordingly.
598
599 if (doIndirect) {
600
601 // Note: This will recurse
602 addIndirectOrDirectVisibleContributions(pbrTls, sp, cameraId, pv, bSampler, bsmp,
603 ray, rayEpsilon, shadowRayEpsilon, isect, indirectFlags, newPriorityList, newPriorityListCount,
604 radiance, sequenceID, aovs);
605 checkForNan(radiance, "Direct or indirect contributions", sp, pv, ray,
606 isect);
607 } else {
608 // TODO: Incorrect transparency if there is no indirect
609 addDirectVisibleBsdfSampleContributions(pbrTls, sp, cameraId, pv, bSampler, false, bsmp, ray,
610 rayEpsilon, shadowRayEpsilon, radiance, sequenceID, aovs);
611 checkForNan(radiance, "Direct contributions", sp, pv, ray, isect);
612 }
613 addDirectVisibleLightSampleContributions(pbrTls, sp, cameraId, pv, lSampler, lsmp, ray,
614 rayEpsilon, shadowRayEpsilon, radiance, sequenceID, aovs);
615 checkForNan(radiance, "Direct contributions", sp, pv, ray, isect);
616
617 return radiance;
618}
