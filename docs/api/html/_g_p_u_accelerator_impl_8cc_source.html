181 virtual void visitInstance(geom::Instance& i) override
182 {
183 if (mFailed) {
184 return;
185 }
186
187 const auto& ref = i.getReference();
188 // visit the referenced Primitive if it's not visited yet
189 if (mSharedGroups.insert(std::make_pair(ref, nullptr)).second) {
190 GPUPrimitiveGroup *group = new GPUPrimitiveGroup();
191 GPUBVHBuilder builder(mLayer, mGeometry, group, mSharedGroups);
192 ref->getPrimitive()->accept(builder);
193 // mark the BVH representation of referenced primitive (group)
194 // has been correctly constructed so that all the instances
195 // reference it can start accessing it
196 mSharedGroups[ref] = group;
197 }
198 // wait for the first visited instance to construct the shared scene
199 SharedGroupMap::const_iterator it = mSharedGroups.find(ref);
200 while (it == mSharedGroups.end() || !it->second) {
201 it = mSharedGroups.find(ref);
202 }
203 auto pImpl = geom::internal::PrimitivePrivateAccess::getPrimitiveImpl(&i);
204 MNRY_ASSERT_REQUIRE(pImpl != nullptr);
205 MNRY_ASSERT_REQUIRE(pImpl->getType() == geom::internal::Primitive::INSTANCE);
206 auto pInstance = static_cast<geom::internal::Instance*>(pImpl);
207
208 createInstance(*pInstance, mSharedGroups[ref]);
209 }

267void
268GPUBVHBuilder::createBox(const geom::internal::Box& geomBox)
269{
270 GPUBox* gpuBox = new GPUBox();
271 mParentGroup->mCustomPrimitives.push_back(gpuBox);
272
273 // Boxes only have one motion sample
274
275 gpuBox->mInputFlags = 0;
276 gpuBox->mIsSingleSided = geomBox.getIsSingleSided();
277 gpuBox->mIsNormalReversed = geomBox.getIsNormalReversed();
278 gpuBox->mVisibleShadow = resolveVisibilityMask(geomBox) & scene_rdl2::rdl2::SHADOW;
279
280 if (!getShadowLinking(geomBox,
281 gpuBox->mShadowLinkAssignmentIds,
282 gpuBox->mShadowLinkLightIds)) {
283 mFailed = true;
284 mWhyFailed = "There was a problem uploading the shadow linking IDs to the GPU";
285 return;
286 }
287
288 std::vector<int> assignmentIds(1);
289 assignmentIds[0] = geomBox.getIntersectionAssignmentId(0);
290 if (gpuBox->mAssignmentIds.allocAndUpload(assignmentIds) != cudaSuccess) {
291 mFailed = true;
292 mWhyFailed = "There was a problem uploading the assignment IDs to the GPU";
293 return;
294 }
295
296 gpuBox->mL2P = mat43ToGPUXform(geomBox.getL2P());
297 gpuBox->mP2L = mat43ToGPUXform(geomBox.getP2L());
298 gpuBox->mLength = geomBox.getLength();
299 gpuBox->mWidth = geomBox.getWidth();
300 gpuBox->mHeight = geomBox.getHeight();
301}

303void
304GPUBVHBuilder::createCurves(const geom::internal::Curves& geomCurves,
305 const geom::Curves::Type curvesType,
306 const int tessellationRate)
307{
308 geom::internal::Curves::Spans spans;
309 geomCurves.getTessellatedSpans(spans);
310
311 GPUCurve* gpuCurve = new GPUCurve();
312 mParentGroup->mCustomPrimitives.push_back(gpuCurve);
313
314 gpuCurve->mInputFlags = 0;
315 gpuCurve->mIsSingleSided = false;
316 gpuCurve->mIsNormalReversed = false;
317 gpuCurve->mVisibleShadow = resolveVisibilityMask(geomCurves) & scene_rdl2::rdl2::SHADOW;
318
319 if (!getShadowLinking(geomCurves,
320 gpuCurve->mShadowLinkAssignmentIds,
321 gpuCurve->mShadowLinkLightIds)) {
322 mFailed = true;
323 mWhyFailed = "There was a problem uploading the shadow linking IDs to the GPU";
324 return;
325 }
326
327 gpuCurve->mMotionSamplesCount = geomCurves.getMotionSamplesCount();
328 gpuCurve->mHostIndices.resize(spans.mSpanCount);
329 gpuCurve->mNumControlPoints = spans.mVertexCount;
330 gpuCurve->mHostControlPoints.resize(spans.mVertexCount * gpuCurve->mMotionSamplesCount);
331
332 switch (curvesType) {
333 case geom::Curves::Type::LINEAR:
334 gpuCurve->mBasis = LINEAR;
335 break;
336 case geom::Curves::Type::BEZIER:
337 gpuCurve->mBasis = BEZIER; // the rdlcurves default
338 break;
339 case geom::Curves::Type::BSPLINE:
340 gpuCurve->mBasis = BSPLINE; // the furdeform/willow default
341 break;
342 default:
343 MNRY_ASSERT_REQUIRE(false);
344 }
345
346 const unsigned int* indices = reinterpret_cast<const unsigned int*>(spans.mIndexBufferDesc.mData);
347 std::vector<int> assignmentIds(spans.mSpanCount);
348
349 for (size_t i = 0; i < spans.mSpanCount; i++) {
350 // We only want every third element of the index buffer to get the vertex index.
351 // See: geom/prim/Curves.h struct IndexData
352 gpuCurve->mHostIndices[i] = indices[i * 3];
353 assignmentIds[i] = geomCurves.getIntersectionAssignmentId(i);
354 }
355
356 const geom::Curves::VertexBuffer& verts = geomCurves.getVertexBuffer();
357
358 for (int ms = 0; ms < gpuCurve->mMotionSamplesCount; ms++) {
359 for (size_t i = 0; i < spans.mVertexCount; i++) {
360 const scene_rdl2::math::Vec3fa& cp = verts(i, ms);
361 gpuCurve->mHostControlPoints[ms * spans.mVertexCount + i] =
362 {cp.x, cp.y, cp.z, cp.w};
363 }
364 }
365
366 if (gpuCurve->mAssignmentIds.allocAndUpload(assignmentIds) != cudaSuccess) {
367 mFailed = true;
368 mWhyFailed = "There was a problem uploading the assignment IDs to the GPU";
369 return;
370 }
371
372 if (gpuCurve->mIndices.allocAndUpload(gpuCurve->mHostIndices) != cudaSuccess) {
373 mFailed = true;
374 mWhyFailed = "There was a problem uploading the curve indices to the GPU";
375 return;
376 }
377
378 if (gpuCurve->mControlPoints.allocAndUpload(gpuCurve->mHostControlPoints)) {
379 mFailed = true;
380 mWhyFailed = "There was a problem uploading the curve control points to the GPU";
381 return;
382 }
383
384 gpuCurve->mSegmentsPerCurve = tessellationRate; // for bezier and bspline, embree default
385}

387void
388GPUBVHBuilder::createRoundCurves(const geom::internal::Curves& geomCurves,
389 const geom::Curves::Type curvesType)
390{
391 if (curvesType == geom::Curves::Type::BEZIER) {
392 mFailed = true;
393 mWhyFailed = "Round bezier curves are not supported in XPU mode";
394 return;
395 }
396
397 // This code assumes that there is a max of 2 motion samples.
398 // TODO: add support for round curves with more motion samples.
399 bool motionBlur = geomCurves.getMotionSamplesCount() > 1;
400 if (geomCurves.getMotionSamplesCount() > 2) {
401 mFailed = true;
402 mWhyFailed = "Round curves with more than 2 motion samples are currently unsupported in XPU mode";
403 return;
404 }
405
406 geom::internal::Curves::Spans spans;
407 geomCurves.getTessellatedSpans(spans);
408
409 GPURoundCurves* gpuCurve = new GPURoundCurves();
410 if (!motionBlur) {
411 mParentGroup->mRoundCurves.push_back(gpuCurve);
412 } else {
413 mParentGroup->mRoundCurvesMB.push_back(gpuCurve);
414 }
415
416 gpuCurve->mInputFlags = 0;
417 gpuCurve->mIsSingleSided = false;
418 gpuCurve->mIsNormalReversed = false;
419 gpuCurve->mVisibleShadow = resolveVisibilityMask(geomCurves) & scene_rdl2::rdl2::SHADOW;
420
421 if (!getShadowLinking(geomCurves,
422 gpuCurve->mShadowLinkAssignmentIds,
423 gpuCurve->mShadowLinkLightIds)) {
424 mFailed = true;
425 mWhyFailed = "There was a problem uploading the shadow linking IDs to the GPU";
426 return;
427 }
428
429 gpuCurve->mMotionSamplesCount = geomCurves.getMotionSamplesCount();
430
431 switch (curvesType) {
432 case geom::Curves::Type::LINEAR:
433 gpuCurve->mType = OPTIX_PRIMITIVE_TYPE_ROUND_LINEAR;
434 break;
435 case geom::Curves::Type::BSPLINE:
436 gpuCurve->mType = OPTIX_PRIMITIVE_TYPE_ROUND_CUBIC_BSPLINE;
437 break;
438 default:
439 MNRY_ASSERT_REQUIRE(false);
440 }
441
442 gpuCurve->mNumControlPoints = spans.mVertexCount;
443 const unsigned int* indices = reinterpret_cast<const unsigned int*>(spans.mIndexBufferDesc.mData);
444
445 std::vector<int> assignmentIds(spans.mSpanCount);
446 std::vector<unsigned int> hostIndices(spans.mSpanCount);
447 std::vector<float3> hostVertices(spans.mVertexCount * gpuCurve->mMotionSamplesCount);
448 std::vector<float> hostWidths(spans.mVertexCount * gpuCurve->mMotionSamplesCount);
449
450 for (size_t i = 0; i < spans.mSpanCount; i++) {
451 // We only want every third element of the index buffer to get the vertex index.
452 // See: geom/prim/Curves.h struct IndexData
453 hostIndices[i] = indices[i * 3];
454 assignmentIds[i] = geomCurves.getIntersectionAssignmentId(i);
455 }
456
457 const geom::Curves::VertexBuffer& verts = geomCurves.getVertexBuffer();
458
459 for (int ms = 0; ms < gpuCurve->mMotionSamplesCount; ms++) {
460 for (size_t i = 0; i < spans.mVertexCount; i++) {
461 const scene_rdl2::math::Vec3fa& cp = verts(i, ms);
462 hostVertices[ms * spans.mVertexCount + i] = {cp.x, cp.y, cp.z};
463 hostWidths[ms * spans.mVertexCount + i] = cp.w;
464 }
465 }
466
467 if (gpuCurve->mAssignmentIds.allocAndUpload(assignmentIds) != cudaSuccess) {
468 mFailed = true;
469 mWhyFailed = "There was a problem uploading the assignment IDs to the GPU";
470 return;
471 }
472
473 if (gpuCurve->mIndices.allocAndUpload(hostIndices) != cudaSuccess) {
474 mFailed = true;
475 mWhyFailed = "There was a problem uploading the curve indices to the GPU";
476 return;
477 }
478
479 if (gpuCurve->mVertices.allocAndUpload(hostVertices) != cudaSuccess) {
480 mFailed = true;
481 mWhyFailed = "There was a problem uploading the curve vertices to the GPU";
482 return;
483 }
484
485 if (gpuCurve->mWidths.allocAndUpload(hostWidths) != cudaSuccess) {
486 mFailed = true;
487 mWhyFailed = "There was a problem uploading the curve widths to the GPU";
488 return;
489 }
490
491 // Get the vertex/width buffer pointers. For curves without motion blur, the
492 // second pointer is null.
493 gpuCurve->mVerticesPtrs[0] = gpuCurve->mVertices.deviceptr();
494 gpuCurve->mWidthsPtrs[0] = gpuCurve->mWidths.deviceptr();
495 if (gpuCurve->mMotionSamplesCount == 2) {
496 gpuCurve->mVerticesPtrs[1] = gpuCurve->mVertices.deviceptr() +
497 gpuCurve->mNumControlPoints * sizeof(float3);
498 gpuCurve->mWidthsPtrs[1] = gpuCurve->mWidths.deviceptr() +
499 gpuCurve->mNumControlPoints * sizeof(float);
500 } else {
501 gpuCurve->mVerticesPtrs[1] = 0;
502 gpuCurve->mWidthsPtrs[1] = 0;
503 }
504}

507GPUBVHBuilder::createPoints(const geom::internal::Points& geomPoints)
508{
509 GPUPoints* gpuPoints = new GPUPoints();
510 mParentGroup->mCustomPrimitives.push_back(gpuPoints);
511
512 gpuPoints->mInputFlags = 0;
513 gpuPoints->mIsSingleSided = false;
514 gpuPoints->mIsNormalReversed = false;
515 gpuPoints->mVisibleShadow = resolveVisibilityMask(geomPoints) & scene_rdl2::rdl2::SHADOW;
516
517 if (!getShadowLinking(geomPoints,
518 gpuPoints->mShadowLinkAssignmentIds,
519 gpuPoints->mShadowLinkLightIds)) {
520 mFailed = true;
521 mWhyFailed = "There was a problem uploading the shadow linking IDs to the GPU";
522 return;
523 }
524
525 gpuPoints->mMotionSamplesCount = geomPoints.getMotionSamplesCount();
526
527 int numPoints = geomPoints.getSubPrimitiveCount();
528 std::vector<int> assignmentIds(numPoints);
529 gpuPoints->mHostPoints.resize(numPoints * gpuPoints->mMotionSamplesCount);
530
531 const geom::Points::VertexBuffer& verts = geomPoints.getVertexBuffer();
532 const geom::Points::RadiusBuffer& radii = geomPoints.getRadiusBuffer();
533
534 for (int i = 0; i < numPoints; i++) {
535 const float radius = radii[i]; // radius doesn't have motion samples
536 for (int ms = 0; ms < gpuPoints->mMotionSamplesCount; ms++) {
537 const scene_rdl2::math::Vec3f& vert = verts(i, ms);
538 gpuPoints->mHostPoints[i * gpuPoints->mMotionSamplesCount + ms] =
539 {vert.x, vert.y, vert.z, radius};
540 }
541 assignmentIds[i] = geomPoints.getIntersectionAssignmentId(i);
542 }
543
544 if (gpuPoints->mAssignmentIds.allocAndUpload(assignmentIds) != cudaSuccess) {
545 mFailed = true;
546 mWhyFailed = "There was a problem uploading the assignment IDs to the GPU";
547 return;
548 }
549
550 if (gpuPoints->mPoints.allocAndUpload(gpuPoints->mHostPoints) != cudaSuccess) {
551 mFailed = true;
552 mWhyFailed = "There was a problem uploading the points to the GPU";
553 return;
554 }

557void
558GPUBVHBuilder::createPolyMesh(const geom::internal::Mesh& geomMesh)
559{
560 geom::internal::Mesh::TessellatedMesh mesh;
561 geomMesh.getTessellatedMesh(mesh);
562
563 // This code assumes that there is a max of 2 motion samples
564 // TODO: add support for mesh with more motion samples.
565 const size_t mbSamples = mesh.mVertexBufferDesc.size();
566 const bool enableMotionBlur = mbSamples > 1;
567 if (mbSamples > 2) {
568 mFailed = true;
569 mWhyFailed = "Meshes with more than 2 motion samples are currently unsupported in XPU mode";
570 return;
571 }
572
573 GPUTriMesh* gpuMesh = new GPUTriMesh();
574 gpuMesh->mName = geomMesh.getName();
575 if (enableMotionBlur) {
576 mParentGroup->mTriMeshesMB.push_back(gpuMesh);
577 } else {
578 mParentGroup->mTriMeshes.push_back(gpuMesh);
579 }
580
581 bool hasVolumeAssignment = geomMesh.hasVolumeAssignment(mLayer);
582
583 gpuMesh->mInputFlags = 0;
584 gpuMesh->mIsSingleSided = geomMesh.getIsSingleSided() && !hasVolumeAssignment;
585 gpuMesh->mIsNormalReversed = geomMesh.getIsNormalReversed();
586 gpuMesh->mVisibleShadow = resolveVisibilityMask(geomMesh) & scene_rdl2::rdl2::SHADOW;
587 gpuMesh->mEnableMotionBlur = enableMotionBlur;
588
589 if (!getShadowLinking(geomMesh,
590 gpuMesh->mShadowLinkAssignmentIds,
591 gpuMesh->mShadowLinkLightIds)) {
592 mFailed = true;
593 mWhyFailed = "There was a problem uploading the shadow linking IDs to the GPU";
594 return;
595 }
596
597 unsigned int vertsPerFace;
598 switch (mesh.mIndexBufferType) {
599 case geom::internal::MeshIndexType::TRIANGLE:
600 {
601 vertsPerFace = 3;
602 break;
603 }
604 case geom::internal::MeshIndexType::QUAD:
605 {
606 vertsPerFace = 4;
607 break;
608 }
609 default:
610 MNRY_ASSERT_REQUIRE(false);
611 break;
612 }
613
614 if (vertsPerFace == 3) {
615 gpuMesh->mNumVertices = mesh.mVertexCount;
616 gpuMesh->mNumFaces = mesh.mFaceCount;
617
618 // When a mesh uses motion blur, the vertices of each motion key are stored consecutively in a single buffer.
619 // According to Nvidia, GAS construction is most efficient when each motion buffer is 16-byte aligned. This
620 // isn't a problem for the first motion key buffer, but the second motion key buffer may not be correctly aligned.
621 // We handle that case here. Note that motionKeyOffsets is {0, 0} if there is no motion. This way, if some mesh
622 // don't support motion blur, their mVerticesPtrs will both point to the same buffer. Optix needs both pointers
623 // to point to a valid buffer.
624 const std::array<size_t, 2> motionKeyOffsets = {0, enableMotionBlur ? scene_rdl2::util::alignUp<size_t>(mesh.mVertexCount, 4) : 0};
625 const size_t gpuVerticesSize = motionKeyOffsets[1] + mesh.mVertexCount;
626 std::vector<float3> gpuVertices(gpuVerticesSize);
627
628 std::vector<unsigned int> gpuIndices(mesh.mFaceCount * 3);
629 const unsigned int* indices = reinterpret_cast<const unsigned int*>(mesh.mIndexBufferDesc.mData);
630 std::vector<int> assignmentIds(mesh.mFaceCount);
631
632 // First set all assignmentIds and indices
633 for (size_t i = 0; i < mesh.mFaceCount; i++) {
634 assignmentIds[i] = geomMesh.getIntersectionAssignmentId(i);
635 gpuIndices[i * 3 + 0] = indices[i * 3 + 0];
636 gpuIndices[i * 3 + 1] = indices[i * 3 + 1];
637 gpuIndices[i * 3 + 2] = indices[i * 3 + 2];
638 }
639
640 for (size_t ms = 0; ms < mbSamples; ms++) {
641 // Because the offset and stride members of mVertexBufferDesc are in bytes, we want to perform pointer
642 // arithmetic on pointers to char (or anything byte sized).
643 const char* vertices = reinterpret_cast<const char*>(mesh.mVertexBufferDesc[ms].mData);
644
645 const unsigned int offset = mesh.mVertexBufferDesc[ms].mOffset;
646 const unsigned int stride = mesh.mVertexBufferDesc[ms].mStride;
647 const size_t motionKeyOffset = motionKeyOffsets[ms];
648
649 for (size_t i = 0; i < mesh.mVertexCount; i++) {
650 scene_rdl2::math::Vec3fa vtx = *reinterpret_cast<const scene_rdl2::math::Vec3fa*>(&vertices[offset + stride * i]);
651 gpuVertices[motionKeyOffset + i] = {vtx.x, vtx.y, vtx.z};
652 }
653 }
654
655 if (gpuMesh->mVertices.allocAndUpload(gpuVertices) != cudaSuccess) {
656 mFailed = true;
657 mWhyFailed = "There was a problem uploading the vertices to the GPU";
658 return;
659 }
660 if (gpuMesh->mIndices.allocAndUpload(gpuIndices) != cudaSuccess) {
661 mFailed = true;
662 mWhyFailed = "There was a problem uploading the indices to the GPU";
663 return;
664 }
665 if (gpuMesh->mAssignmentIds.allocAndUpload(assignmentIds) != cudaSuccess) {
666 mFailed = true;
667 mWhyFailed = "There was a problem uploading the assignment IDs to the GPU";
668 return;
669 }
670
671 gpuMesh->mVerticesPtrs[0] = gpuMesh->mVertices.deviceptr();
672 gpuMesh->mVerticesPtrs[1] = gpuMesh->mVertices.deviceptr() + motionKeyOffsets[1] * sizeof(float3);
673
674 } else if (vertsPerFace == 4) {
675 // convert quads to tris
676
677 gpuMesh->mNumVertices = mesh.mVertexCount;
678 gpuMesh->mNumFaces = mesh.mFaceCount * 2;
679
680 // See above for reason this code exists:
681 const std::array<size_t, 2> motionKeyOffsets = {0, enableMotionBlur ? scene_rdl2::util::alignUp<size_t>(mesh.mVertexCount, 4) : 0};
682 const size_t gpuVerticesSize = motionKeyOffsets[1] + mesh.mVertexCount;
683 std::vector<float3> gpuVertices(gpuVerticesSize);
684
685 std::vector<unsigned int> gpuIndices(mesh.mFaceCount * 6);
686 const unsigned int* indices = reinterpret_cast<const unsigned int*>(mesh.mIndexBufferDesc.mData);
687 std::vector<int> assignmentIds(mesh.mFaceCount * 2);
688
689 // First set all assignmentIds:
690 for (size_t i = 0; i < mesh.mFaceCount; i++) {
691 assignmentIds[i * 2 + 0] = geomMesh.getIntersectionAssignmentId(i);
692 assignmentIds[i * 2 + 1] = assignmentIds[i * 2];
693 // quad -> two tris
694 unsigned int idx0 = indices[i * 4 + 0];
695 unsigned int idx1 = indices[i * 4 + 1];
696 unsigned int idx2 = indices[i * 4 + 2];
697 unsigned int idx3 = indices[i * 4 + 3];
698 // first tri 0-1-3
699 gpuIndices[i * 6 + 0] = idx0;
700 gpuIndices[i * 6 + 1] = idx1;
701 gpuIndices[i * 6 + 2] = idx3;
702 // second tri 1-2-3
703 gpuIndices[i * 6 + 3] = idx1;
704 gpuIndices[i * 6 + 4] = idx2;
705 gpuIndices[i * 6 + 5] = idx3;
706 }
707
708 for (size_t ms = 0; ms < mbSamples; ms++) {
709 // Because the offset and stride members of mVertexBufferDesc are in bytes, we want to perform pointer
710 // arithmetic on pointers to char (or anything byte sized).
711 const char* vertices = reinterpret_cast<const char*>(mesh.mVertexBufferDesc[ms].mData);
712
713 const unsigned int offset = mesh.mVertexBufferDesc[ms].mOffset;
714 const unsigned int stride = mesh.mVertexBufferDesc[ms].mStride;
715 const size_t motionKeyOffset = motionKeyOffsets[ms];
716
717 for (size_t i = 0; i < mesh.mVertexCount; i++) {
718 scene_rdl2::math::Vec3fa vtx = *reinterpret_cast<const scene_rdl2::math::Vec3fa*>(&vertices[offset + stride * i]);
719 gpuVertices[motionKeyOffset + i] = {vtx.x, vtx.y, vtx.z};
720 }
721 }
722
723 if (gpuMesh->mVertices.allocAndUpload(gpuVertices) != cudaSuccess) {
724 mFailed = true;
725 mWhyFailed = "There was a problem uploading the vertices to the GPU";
726 return;
727 }
728 if (gpuMesh->mIndices.allocAndUpload(gpuIndices) != cudaSuccess) {
729 mFailed = true;
730 mWhyFailed = "There was a problem uploading the indices to the GPU";
731 return;
732 }
733 if (gpuMesh->mAssignmentIds.allocAndUpload(assignmentIds) != cudaSuccess) {
734 mFailed = true;
735 mWhyFailed = "There was a problem uploading the assignment IDs to the GPU";
736 return;
737 }
738
739 gpuMesh->mVerticesPtrs[0] = gpuMesh->mVertices.deviceptr();
740 gpuMesh->mVerticesPtrs[1] = gpuMesh->mVertices.deviceptr() + motionKeyOffsets[1] * sizeof(float3);
741 }
742}

781void
782GPUBVHBuilder::createInstance(const geom::internal::Instance& instance,
783 GPUPrimitiveGroup* group)
784{
785 const geom::internal::MotionTransform& l2pXform = instance.getLocal2Parent();
786
787 scene_rdl2::math::Xform3f xforms[GPUInstance::sNumMotionKeys];
788 bool hasMotionBlur;
789 if (!l2pXform.isMotion()) {
790 xforms[0] = l2pXform.getStaticXform();
791 hasMotionBlur = false;
792 } else {
793 // eval() the l2pXform at sNumMotionKeys discrete timesteps between time [0, 1].
794 // Optix can't slerp() between matrices so we need to generate smaller
795 // timesteps it can lerp() between.
796 for (int i = 0; i < GPUInstance::sNumMotionKeys; i++) {
797 float t = i / static_cast<float>(GPUInstance::sNumMotionKeys - 1);
798 xforms[i] = l2pXform.eval(t);
799 }
800 hasMotionBlur = true;
801 }
802
803 // The visibility mask needs to take volume assignments into account.
804 // We use the visibility flags of the instance geometry, but do the
805 // volume/surface shift based on the reference's assignments.
806 const std::shared_ptr<geom::SharedPrimitive> &ref = instance.getReference();
807 bool hasSurfaceAssignment = ref->getHasSurfaceAssignment();
808 bool hasVolumeAssignment = ref->getHasVolumeAssignment();
809 if (!hasSurfaceAssignment && !hasVolumeAssignment) {
810 // In the case of multi-level instancing we'll need to
811 // recurse through the primitives to check for volume assignments
812 GetAssignments ga(hasVolumeAssignment, hasSurfaceAssignment);
813 ref->getPrimitive()->accept(ga);
814 }
815 unsigned int mask = 0;
816 if (hasSurfaceAssignment) {
817 mask |= mGeometry->getVisibilityMask();
818 }
819 if (hasVolumeAssignment) {
820 mask |= mGeometry->getVisibilityMask() << scene_rdl2::rdl2::sNumVisibilityTypes;
821 }
822
823 bool visibleShadow = mask & scene_rdl2::rdl2::SHADOW;
824 if (!visibleShadow) {
825 // Instance doesn't cast a shadow so it doesn't exist for occlusion
826 // queries. Just skip it. We will need to add some more logic when
827 // regular rays are supported by XPU.
828 return;
829 }
830
831 GPUInstance* gpuInstance = new GPUInstance();
832 mParentGroup->mInstances.push_back(gpuInstance);
833 gpuInstance->mGroup = group;
834
835 if (hasMotionBlur) {
836 for (int i = 0; i < GPUInstance::sNumMotionKeys; i++) {
837 gpuInstance->mXforms[i] = mat43ToGPUXform(xforms[i]);
838 }
839 } else {
840 gpuInstance->mXforms[0] = mat43ToGPUXform(xforms[0]);
841 }
842
843 gpuInstance->mHasMotionBlur = hasMotionBlur;
844}

846unsigned int
847GPUBVHBuilder::resolveVisibilityMask(const geom::internal::NamedPrimitive& np) const
848{
849 unsigned int mask = 0;
850 if (np.hasSurfaceAssignment(mLayer)) {
851 // visibility for surface ray query
852 mask |= mGeometry->getVisibilityMask();
853 }
854 if (np.hasVolumeAssignment(mLayer)) {
855 // visibility for volume ray query
856 mask |= mGeometry->getVisibilityMask() << scene_rdl2::rdl2::sNumVisibilityTypes;
857 }
858 return mask;
859}

861bool
862GPUBVHBuilder::getShadowLinking(const geom::internal::NamedPrimitive& np,
863 GPUBuffer<int>& assignmentIdsBuf,
864 GPUBuffer<unsigned long long>& lightIdsBuf) const
865{
866 std::vector<int> assignmentIds;
867 std::vector<unsigned long long> lightIds;
868
869 for (const auto& element : np.getShadowLinkings()) {
870 int assignmentId = element.first;
871 const geom::internal::ShadowLinking *linking = element.second;
872
873 if (linking == nullptr) {
874 // skip any empty
875 continue;
876 }
877
878 for (const auto& light : linking->getLights()) {
879 assignmentIds.push_back(assignmentId);
880 lightIds.push_back(reinterpret_cast<intptr_t>(light));
881 }
882 }
883 if (assignmentIdsBuf.allocAndUpload(assignmentIds) != cudaSuccess) {
884 return false;
885 }
886 if (lightIdsBuf.allocAndUpload(lightIds) != cudaSuccess) {
887 return false;
888 }
889 return true;
890}

1113bool
1114buildGPUBVHBottomUp(const scene_rdl2::rdl2::Layer* layer,
1115 scene_rdl2::rdl2::Geometry* geometry,
1116 GPUPrimitiveGroup* rootGroup,
1117 SharedGroupMap& groups,
1118 std::unordered_set<scene_rdl2::rdl2::Geometry*>& visitedGeometry,
1119 std::string* errorMsg)
1120{
1121 // Extremely similar to EmbreeAccelerator.cc buildBVHBottomUp(). This is intentional so
1122 // it behaves the same and is easy to maintain and debug.
1123
1124 geom::Procedural* procedural = geometry->getProcedural();
1125 // All parts in a procedural are unassigned in the layer
1126 if (!procedural) {
1127 return true;
1128 }
1129 // This BVH part for this particular rdl geometry has been constructed
1130 if (visitedGeometry.find(geometry) != visitedGeometry.end()) {
1131 return true;
1132 }
1133 // Do a bottom up traversal so that referenced BVH is built first
1134 const scene_rdl2::rdl2::SceneObjectVector& references =
1135 geometry->get(scene_rdl2::rdl2::Geometry::sReferenceGeometries);
1136 for (const auto& ref : references) {
1137 if (!ref->isA<scene_rdl2::rdl2::Geometry>()) {
1138 continue;
1139 }
1140 scene_rdl2::rdl2::Geometry* referencedGeometry = ref->asA<scene_rdl2::rdl2::Geometry>();
1141 buildGPUBVHBottomUp(layer,
1142 referencedGeometry,
1143 rootGroup,
1144 groups,
1145 visitedGeometry,
1146 errorMsg);
1147 }
1148 // We disable parallelism here to solve the non-deterministic
1149 // issue for some hair/fur related scenes.
1150 // Also, testing shows no speed gain is achieved when this for() loop is parallelized.
1151 bool doParallel = false;
1152 if (procedural->isReference()) {
1153 const std::shared_ptr<geom::SharedPrimitive>& ref =
1154 procedural->getReference();
1155 if (groups.insert(std::make_pair(ref, nullptr)).second) {
1156 GPUPrimitiveGroup *group = new GPUPrimitiveGroup();
1157 GPUBVHBuilder builder(layer, geometry, group, groups);
1158 ref->getPrimitive()->accept(builder);
1159 // mark the BVH representation of referenced primitive (group)
1160 // has been correctly constructed so that all the instances
1161 // reference it can start accessing it
1162 groups[ref] = group;
1163 }
1164 } else {
1165 GPUBVHBuilder geomBuilder(layer,
1166 geometry,
1167 rootGroup,
1168 groups);
1169 procedural->forEachPrimitive(geomBuilder, doParallel);
1170 if (geomBuilder.hasFailed()) {
1171 *errorMsg = geomBuilder.whyFailed();
1172 return false;
1173 }
1174 }
1175 visitedGeometry.insert(geometry);
1176
1177 return true;
1178}

1180bool
1181GPUAcceleratorImpl::build(CUstream cudaStream,
1182 OptixDeviceContext context,
1183 const scene_rdl2::rdl2::Layer *layer,
1184 const scene_rdl2::rdl2::SceneContext::GeometrySetVector& geometrySets,
1185 const scene_rdl2::rdl2::Layer::GeometryToRootShadersMap* g2s,
1186 std::string* errorMsg)
1187{
1188 // See embree EmbreeAccelerator::build()
1189
1190 mRootGroup = new GPUPrimitiveGroup();
1191
1192 std::unordered_set<scene_rdl2::rdl2::Geometry*> visitedGeometry;
1193 for (const auto& geometrySet : geometrySets) {
1194 const scene_rdl2::rdl2::SceneObjectIndexable& geometries = geometrySet->getGeometries();
1195 for (auto& sceneObject : geometries) {
1196 scene_rdl2::rdl2::Geometry* geometry = sceneObject->asA<scene_rdl2::rdl2::Geometry>();
1197 if (g2s != nullptr && g2s->find(geometry) == g2s->end()) {
1198 continue;
1199 }
1200 if (!buildGPUBVHBottomUp(layer,
1201 geometry,
1202 mRootGroup,
1203 mSharedGroups,
1204 visitedGeometry,
1205 errorMsg)) {
1206 return false;
1207 }
1208 }
1209 }
1210
1211 unsigned int sbtOffset = 0;
1212 mRootGroup->setSBTOffset(sbtOffset);
1213 for (auto& groupEntry : mSharedGroups) {
1214 GPUPrimitiveGroup *group = groupEntry.second;
1215 group->setSBTOffset(sbtOffset);
1216 }
1217
1218 // now build the root group and referenced/instanced groups
1219 if (!mRootGroup->build(cudaStream, context, errorMsg)) {
1220 return false;
1221 }
1222
1223 return true;
1224}

1425 mRootGroup->getSBTRecords(mProgramGroups, hitgroupRecs);
1426 for (auto& groupEntry : mSharedGroups) {

1428 group->getSBTRecords(mProgramGroups, hitgroupRecs);

1438 mSBT.hitgroupRecordBase = mHitGroupRecordBuf.deviceptr();

1446void

1448{
1449 // std::cout << "occluded(): " << numRays << std::endl;
1450
1451 MNRY_ASSERT_REQUIRE(numRays <= mRaysBufSize);
1452
1453 // Setup the global GPU parameters
1454 GPUParams params;
1455 params.mAccel = mRootGroup->mTopLevelIAS;
1456 params.mNumRays = numRays;
1457 params.mRaysBuf = mRaysBuf.ptr();
1458 params.mIsOccludedBuf = mIsOccludedBuf.ptr();
1459 mParamsBuf.upload(&params);
1460
1461 // Upload the rays to the GPU
1462 mRaysBuf.upload(rays, numRays);
1463
1464 if (optixLaunch(mPipeline,
1465 mCudaStream,
1466 mParamsBuf.deviceptr(),
1467 mParamsBuf.sizeInBytes(),
1468 &mSBT,
1469 numRays, 1, 1) != OPTIX_SUCCESS) {
1470 // There isn't any feasible way to recover from this, but it shouldn't
1471 // happen unless the code is broken. Log the error and try to keep going.
1472 scene_rdl2::logging::Logger::error("GPU: optixLaunch() failure");
1473 }
1474
1475 // Wait for the GPU to finish processing the rays
1476 cudaDeviceSynchronize();
1477 cudaError_t error = cudaGetLastError();
1478 if (error != cudaSuccess) {
1479 // There isn't any feasible way to recover from this, but it shouldn't
1480 // happen unless the code is broken. Log the error and try to keep going.
1481 scene_rdl2::logging::Logger::error("GPU: optixLaunch() cudaDeviceSynchronize() error: ",
1482 cudaGetErrorString(error));
1483 }
1484
1485 // Download the intersection results from the GPU
1486 mIsOccludedBuf.download(mOutputOcclusionBuf, numRays);
1487}
