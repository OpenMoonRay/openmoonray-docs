232void
233SubdivisionMesh::transformPrimitive(
234 const MotionBlurParams& motionBlurParams,
235 const shading::XformSamples& prim2render)
236{
237 size_t motionSamplesCount = getMotionSamplesCount();
238 shading::XformSamples p2r = prim2render;
239 if (motionSamplesCount > 1 && prim2render.size() == 1) {
240 p2r.resize(motionSamplesCount, prim2render[0]);
241 }
242 const shading::PrimitiveAttributeTable* primAttrTab = &mImpl->mSubdMesh->getPrimitiveAttributeTable();
243 transformVertexBuffer(mImpl->mSubdMesh->getControlVertexBuffer(), p2r, motionBlurParams,
244 mImpl->mSubdMesh->getMotionBlurType(), mImpl->mSubdMesh->getCurvedMotionBlurSampleCount(),
245 primAttrTab);
246 float shutterOpenDelta, shutterCloseDelta;
247 motionBlurParams.getMotionBlurDelta(shutterOpenDelta, shutterCloseDelta);
248 mImpl->mSubdMesh->setTransform(p2r, shutterOpenDelta, shutterCloseDelta);
249}
