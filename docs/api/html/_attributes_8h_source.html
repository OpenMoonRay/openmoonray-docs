407 size_t mNumKeys;
408 // TODO: This table is somewhat memory-wasteful in the service
409 // of performance (it allows a direct lookup).
410 // The table memory use is proportional to the total number of
411 // AttributeKeys and will become a problem if there's a large
412 // number of primitives and the number of AttributeKeys grows too large.
413 // Two solutions:
414 // 1) Do an indirect lookup, so that only offsets for the keys actually
415 // used (as opposed to all keys) are stored here.

750template <typename T>
751void
752Attributes::motionBlurInterpolate(TypedAttributeKey<T> key,
753 float shutterOpenDelta, float shutterCloseDelta)
754{
755 AttributeRate keyRate = getRate(key);
756 MNRY_ASSERT_REQUIRE(getTimeSampleCount(key) == 2,
757 "only support two time samples for motionblur at this moment");
758
759 switch (keyRate) {
760 case RATE_CONSTANT:
761 {
762 T& t0 = getConstantInternal(key, 0);
763 T& t1 = getConstantInternal(key, 1);
764 T newT0 = scene_rdl2::math::lerp(t0, t1, shutterOpenDelta);
765 T newT1 = scene_rdl2::math::lerp(t0, t1, shutterCloseDelta);
766 t0 = newT0;
767 t1 = newT1;
768 }
769 break;
770 case RATE_UNIFORM:
771 {
772 for (size_t f = 0; f < mNumFaces; ++f) {
773 T& t0 = getUniformInternal(key, f, 0);
774 T& t1 = getUniformInternal(key, f, 1);
775 T newT0 = scene_rdl2::math::lerp(t0, t1, shutterOpenDelta);
776 T newT1 = scene_rdl2::math::lerp(t0, t1, shutterCloseDelta);
777 t0 = newT0;
778 t1 = newT1;
779 }
780 }
781 break;
782 case RATE_VARYING:
783 {
784 for (size_t v = 0; v < mNumVaryings; ++v) {
785 T& t0 = getVaryingInternal(key, v, 0);
786 T& t1 = getVaryingInternal(key, v, 1);
787 T newT0 = scene_rdl2::math::lerp(t0, t1, shutterOpenDelta);
788 T newT1 = scene_rdl2::math::lerp(t0, t1, shutterCloseDelta);
789 t0 = newT0;
790 t1 = newT1;
791 }
792 }
793 break;
794 case RATE_FACE_VARYING:
795 {
796 for (size_t v = 0; v < mNumFaceVaryings; ++v) {
797 T& t0 = getFaceVaryingInternal(key, v, 0);
798 T& t1 = getFaceVaryingInternal(key, v, 1);
799 T newT0 = scene_rdl2::math::lerp(t0, t1, shutterOpenDelta);
800 T newT1 = scene_rdl2::math::lerp(t0, t1, shutterCloseDelta);
801 t0 = newT0;
802 t1 = newT1;
803 }
804 }
805 break;
806 case RATE_VERTEX:
807 {
808 for (size_t v = 0; v < mNumVertices; ++v) {
809 T& t0 = getVertexInternal(key, v, 0);
810 T& t1 = getVertexInternal(key, v, 1);
