30/// @class ProceduralContext

34class ProceduralContext

37public:
38 virtual ~ProceduralContext() {}

40 /// query scene_rdl2::rdl2::Layer the procedural belong

42 virtual const scene_rdl2::rdl2::Layer *getRdlLayer() const = 0;

44 /// query scene_rdl2::rdl2::Geometry the procedural belong
45 /// @return scene_rdl2::rdl2::Geometry the pointer to scene_rdl2::rdl2::Geometry
46 virtual const scene_rdl2::rdl2::Geometry *getRdlGeometry() const = 0;
47
48 /// Get the current frame number
49 /// @return the current frame no.
50 virtual int getCurrentFrame() const = 0;
51
52 /// Get the number of worker threads in use.
53 /// @return the number of worker threads in use
54 virtual int getThreads() const = 0;
55
56 /// Get the parameters related to motion blur
57 /// @return the MotionBlurParams that wraps motion blur related parameters
58 virtual const MotionBlurParams& getMotionBlurParams() const = 0;
59
60 /// Get the motion blur steps in frame relative value
61 /// (For example: -1 stands for previous frame, 0 stands for current frame)
62 /// @return the vector contains motion steps
63 virtual const std::vector<float>& getMotionSteps() const = 0;
64
65 /// Get the shutter open time in frame relative value
66 /// (For example: -0.5 stands for the middle of previous frame and current frame)
67 /// @return the shutter open time in frame relative value
68 virtual float getShutterOpen() const = 0;
69
70 /// Get the shutter close time in frame relative value
71 /// (For example: 0.5 stands for the middle of current frame and next frame)
72 /// @return the shutter close time in frame relative value
73 virtual float getShutterClose() const = 0;
74
75 /// query whether camera motion blur is turned on
76 /// @return whether camera motino blur is turned on
77 virtual bool isMotionBlurOn() const = 0;
78
79 /// Return the delta fraction of shutter open/close time in
80 /// motionSteps duration
81 virtual void getMotionBlurDelta(
82 float& shutterOpenDelta, float& shutterCloseDelta) const = 0;
83
84};
