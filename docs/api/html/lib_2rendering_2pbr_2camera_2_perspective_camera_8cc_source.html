28finline Vec3f
29getFocusPlaneHitPoint(const Vec3f& focalPoint, const Vec3f& raster, const Mat4f& rt2c,
30 float focusDistance)
31{
32 const Vec3f dir = normalize(transformH(rt2c, raster) - focalPoint);
33
34 // We want to take our initial ray (origin at mFocalPoint) and find its
35 // intersection with the focal plane. All work is done in camera space.
36#if 0
37 const Vec3f focalPlaneNormal(0, 0, +1);
38 const Vec3f focalPlanePoint(0, 0, -focusDistance);
39
40 // Ray/plane intersection
41 const float t = dot(focalPlanePoint - focalPoint, focalPlaneNormal) / dot(dir, focalPlaneNormal);
42#else
43 // Since we have a bunch of one-component vectors, the above code simplifies
44 // to some one-dimensional math. Yay!
45 // Since 'focalPoint' is in camera space, it only has 'x' set, so we don't
46 // have to subtract the z value when simplifying the above vector
47 // subtraction.
48 const float n = -focusDistance;
49 const float d = dir.z;
50 const float t = n/d;
51#endif
52 const Vec3f pFocus = focalPoint + dir * t;
53 return pFocus;
54}

74// Compute signed interocular offset due to stereo (translation in camera
75// space along x axis).
76// Warning: This function doesn't work in CENTER mode.
77finline float
78computeInterocularOffset(StereoView stereoView,
79 float interocularDistance)
80{
81 MNRY_ASSERT(stereoView != StereoView::CENTER);
82
83 float interocularOffset =
84 (stereoView == StereoView::LEFT ? 0.5f : -0.5f) *
85 interocularDistance;
86
87 return interocularOffset;
88}

110void
111PerspectiveCamera::createDOFRay(mcrt_common::RayDifferential *dstRay,
112 const Vec3f &raster,
113 float lensX,
114 float lensY,
115 float time) const
116{
117 // TODO: adjust ray differentials for non-zero origin due to dof
118
119 const Mat4f ct2render = computeCamera2Render(time);
120 const Mat4f rt2ct = computeRaster2Camera(time);
121
122 // Transform camera origin on the lens, from camera-space at time t to
123 // camera-space at t=0
124
125 // mFocalPoint is in camera space
126 const Vec3f nx(1.0f, 0.0f, 0.0f);
127 const Vec3f ny(0.0f, 1.0f, 0.0f);
128 const Vec3f ctHitPoint = getFocusPlaneHitPoint(mFocalPoint,
129 raster,
130 rt2ct,
131 getDofFocusDistance());
132 const Vec3f ctHitPointAuxX = getFocusPlaneHitPoint(mFocalPoint,
133 raster + nx,
134 rt2ct,
135 getDofFocusDistance());
136 const Vec3f ctHitPointAuxY = getFocusPlaneHitPoint(mFocalPoint,
137 raster + ny,
138 rt2ct,
139 getDofFocusDistance());
140
141 // Change the origin to that of the lens sample + mFocalPoint
142 const Vec3f ctLensPoint = mFocalPoint + Vec3f(lensX, lensY, 0.0f);
143 const Vec3f renderLensPoint = transformPoint(ct2render, ctLensPoint);
144
145 const Vec3f ctDirPrimary = ctHitPoint - ctLensPoint;
146 const Vec3f dirPrimaryNormalized = normalize(ctDirPrimary);
147 const float tNear = getClipPlaneT(getNear(), mFocalPoint,
148 dirPrimaryNormalized);
149 const float tFar = getClipPlaneT(getFar(), mFocalPoint,
150 dirPrimaryNormalized);
151
152 // Change the direction to the point intersected at the focal plane - the new
153 // origin.
154 *dstRay = mcrt_common::RayDifferential(
155 renderLensPoint, normalize(transformVector(ct2render, ctDirPrimary)),
156 renderLensPoint, normalize(transformVector(ct2render, ctHitPointAuxX - ctLensPoint)),
157 renderLensPoint, normalize(transformVector(ct2render, ctHitPointAuxY - ctLensPoint)),
158 tNear, tFar, time, 0);
159}

161void
162PerspectiveCamera::createSimpleRay(mcrt_common::RayDifferential *dstRay,
163 const Vec3f &raster,
164 float time) const
165{
166 const Mat4f ct2render = computeCamera2Render(time);
167 const Mat4f rt2ct = computeRaster2Camera(time);
168
169 const Vec3f renderStart = transformPoint(ct2render, mFocalPoint);
170 const Vec3f ctDirPrimary = transformH(rt2ct, raster) - mFocalPoint;
171 const Vec3f ctDirAuxX = transformH(rt2ct, raster + Vec3f(1.0f, 0.0f, 0.0f)) - mFocalPoint;
172 const Vec3f ctDirAuxy = transformH(rt2ct, raster + Vec3f(0.0f, 1.0f, 0.0f)) - mFocalPoint;
173
174 const Vec3f dirPrimaryNormalized = normalize(ctDirPrimary);
175 const float tNear = getClipPlaneT(getNear(), mFocalPoint,
176 dirPrimaryNormalized);
177 const float tFar = getClipPlaneT(getFar(), mFocalPoint,
178 dirPrimaryNormalized);
179
180 *dstRay = mcrt_common::RayDifferential(
181 renderStart, normalize(transformVector(ct2render, ctDirPrimary)),
182 renderStart, normalize(transformVector(ct2render, ctDirAuxX)),
183 renderStart, normalize(transformVector(ct2render, ctDirAuxy)),
184 tNear, tFar, time, 0);
185}

209// Compute projection matrix
210Mat4f
211PerspectiveCamera::computeC2S(float t) const
212{
213 const auto cam = getRdlCamera();
214
215 // Compute additional horizontal film offset and translation due to stereo
216 // See diagram pbr/doc/stereo_projection_top_view.jpg for details.
217 float interocularOffset = 0.0f;
218 const StereoView stereoView = getStereoViewImpl();
219 if (stereoView != StereoView::CENTER) {
220 // Get stereo properties
221 interocularOffset = computeInterocularOffset(stereoView,
222 cam->get(sStereoInterocularDistance));
223 }
224
225 return cam->computeProjectionMatrix(t, getWindow(), interocularOffset);
226}

253void
254PerspectiveCamera::computeFrustumImpl(mcrt_common::Frustum *f, float t,
255 bool useRenderRegion) const
256{
257 // C2S is an OpenGL-style projection matrix so we can just extract the
258 // clip plane coefficients (Ax+By+Cz+D=0) by adding/subtracting matrix columns.
259 //
260 // Derivation: Let v = [x y z w=1] a point in camera space,
261 // C2S = 4x4 column-major projection matrix with 4 columns col0..3
262 // Then v' = v * C2S = [v dot col0, v dot col1, v dot col2, v dot col3]
263 // = [x', y', z', w']
264 // For v' to be in the frustum:
265 // -w' < x' < w'
266 // -w' < y' < w'
267 // -w' < z' < w'
268 // To test x' against left clipping plane, use the inequality:
269 // -w' < x'
270 // -(v dot col3) < (v dot col0)
271 // 0 < (v dot col0) + (v dot col3)
272 // 0 < v dot (col0 + col3)
273 // 0 < x * (col0.x + col3.x) + y * (col0.y + col3.y) + z * (col0.z + col3.z) + w * (col0.w + col3.w)
274 // ... but w=1:
275 // 0 < x * (col0.x + col3.x) + y * (col0.y + col3.y) + z * (col0.z + col3.z) + (col0.w + col3.w)
276 // ... this is a plane equation of the form:
277 // 0 < x * A + y * B + c * Z + D
278 // ... just add col0 and col3 to get the ABCD coefficients
279 // ... similar derivation for the other planes
280
281 if (useRenderRegion) {
282 f->mC2S = computeRegionC2S(t);
283 f->mViewport[0] = (int)getRegionToApertureOffsetX();
284 f->mViewport[1] = (int)getRegionToApertureOffsetY();
285 f->mViewport[2] = f->mViewport[0] + (int)getRegionWindowWidth();
286 f->mViewport[3] = f->mViewport[1] + (int)getRegionWindowHeight();
287 } else {
288 f->mC2S = computeC2S(t);
289 f->mViewport[0] = 0;
290 f->mViewport[1] = 0;
291 f->mViewport[2] = (int)getApertureWindowWidth();
292 f->mViewport[3] = (int)getApertureWindowHeight();
293 }
294
295 // right: col 3 - col 0
296 f->mClipPlanes[0][0] = f->mC2S[0][3] - f->mC2S[0][0];
297 f->mClipPlanes[0][1] = f->mC2S[1][3] - f->mC2S[1][0];
298 f->mClipPlanes[0][2] = f->mC2S[2][3] - f->mC2S[2][0];
299 f->mClipPlanes[0][3] = f->mC2S[3][3] - f->mC2S[3][0];
300 // left: col 3 + col 0
301 f->mClipPlanes[1][0] = f->mC2S[0][3] + f->mC2S[0][0];
302 f->mClipPlanes[1][1] = f->mC2S[1][3] + f->mC2S[1][0];
303 f->mClipPlanes[1][2] = f->mC2S[2][3] + f->mC2S[2][0];
304 f->mClipPlanes[1][3] = f->mC2S[3][3] + f->mC2S[3][0];
305 // bottom: col 3 + col 1
306 f->mClipPlanes[2][0] = f->mC2S[0][3] + f->mC2S[0][1];
307 f->mClipPlanes[2][1] = f->mC2S[1][3] + f->mC2S[1][1];
308 f->mClipPlanes[2][2] = f->mC2S[2][3] + f->mC2S[2][1];
309 f->mClipPlanes[2][3] = f->mC2S[3][3] + f->mC2S[3][1];
310 // top: col 3 - col 1
311 f->mClipPlanes[3][0] = f->mC2S[0][3] - f->mC2S[0][1];
312 f->mClipPlanes[3][1] = f->mC2S[1][3] - f->mC2S[1][1];
313 f->mClipPlanes[3][2] = f->mC2S[2][3] - f->mC2S[2][1];
314 f->mClipPlanes[3][3] = f->mC2S[3][3] - f->mC2S[3][1];
315 // far: col3 - col 2
316 f->mClipPlanes[4][0] = f->mC2S[0][3] - f->mC2S[0][2];
317 f->mClipPlanes[4][1] = f->mC2S[1][3] - f->mC2S[1][2];
318 f->mClipPlanes[4][2] = f->mC2S[2][3] - f->mC2S[2][2];
319 f->mClipPlanes[4][3] = f->mC2S[3][3] - f->mC2S[3][2];
320 // near: col3 + col2
321 f->mClipPlanes[5][0] = f->mC2S[0][3] + f->mC2S[0][2];
322 f->mClipPlanes[5][1] = f->mC2S[1][3] + f->mC2S[1][2];
323 f->mClipPlanes[5][2] = f->mC2S[2][3] + f->mC2S[2][2];
324 f->mClipPlanes[5][3] = f->mC2S[3][3] + f->mC2S[3][2];
325}
