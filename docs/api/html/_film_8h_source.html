184 // Add the value(s) from aovs the statistics buffer at index idx.
185 // px, py is the pixel
186 // idx is the index into the mAovBuf array for the statistics buffer
187 // aovs is the value in one, two, or three floats
188 void addSampleStatistics(unsigned px, unsigned py, std::size_t idx, const float* aovs)
189 {
190 mTiler.linearToTiledCoords(px, py, &px, &py);
191 scene_rdl2::fb_util::VariablePixelBuffer& buf = mAovBuf[idx];
192
193 switch (buf.getFormat()) {
194 case scene_rdl2::fb_util::VariablePixelBuffer::FLOAT_VARIANCE:
195 buf.getFloatVarianceBuffer().getPixel(px, py).push(aovs[0]);
196 break;
197 case scene_rdl2::fb_util::VariablePixelBuffer::FLOAT2_VARIANCE:
198 buf.getFloat2VarianceBuffer().getPixel(px, py).push(scene_rdl2::math::Vec2f(aovs[0], aovs[1]));
199 break;
200 case scene_rdl2::fb_util::VariablePixelBuffer::FLOAT3_VARIANCE:
201 buf.getFloat3VarianceBuffer().getPixel(px, py).push(scene_rdl2::math::Vec3f(aovs[0], aovs[1], aovs[2]));
202 break;
203 case scene_rdl2::fb_util::VariablePixelBuffer::RGB_VARIANCE:
204 buf.getRgbVarianceBuffer().getPixel(px, py).push(luminance(scene_rdl2::math::Color(aovs[0], aovs[1], aovs[2])));
205 break;
206 default:
207 MNRY_ASSERT(0 && "unexpected aov buffer format");
208 }
209 }

463inline unsigned
464Film::getNumRenderBufferPixelSamples(unsigned px, unsigned py) const
465{
466 mTiler.linearToTiledCoords(px, py, &px, &py);
467
468 const float weightValue = util::atomicLoad(&mWeightBuf.getPixel(px, py), std::memory_order_relaxed);
469
470 // Round to nearest as an approximation of the number of samples.
471 return static_cast<unsigned>(weightValue + 0.5f);
472}

474inline void
475Film::fillPixelSampleCountBuffer(scene_rdl2::fb_util::PixelBuffer<unsigned>& buf) const
476{
477 const unsigned w = getWidth();
478 const unsigned h = getHeight();
479 buf.init(w, h);
480
481 for (unsigned x = 0; x < w; ++x) {
482 for (unsigned y = 0; y < h; ++y) {
483 buf.setPixel(x, y, getNumRenderBufferPixelSamples(x, y));
484 }
485 }

551
552// General purpose extrapolation functions. Can be used to extrapolate any
553// type of buffers.
554// The HAS_DATA_FUNC functor should return 0xffffffff if valid data is
555// contained in the pixel or 0 if not. The 0xffffffff is important!
556// The HAS_DATA_FUNC functor should expect two parameters. The first
557// is the value of the buffer (or a const reference to the value). The
558// second is the offset to the pixel data in the buffer.
559
560//
561// Fast path when we are extrapolating without a viewport and we aren't doing
562// distributed rendering.
563//
564// Warning: If the buffer isn't aligned to tile boundaries, pixels along the
565// viewport edges may not get extrapolated during early progressive passes
566// since the pixels they should have been extrapolated from may have been
567// clipped out.
568//
569
570template<typename PIXEL_TYPE, typename HAS_DATA_FUNC>
571inline void
572extrapolateBufferFastPath(scene_rdl2::fb_util::PixelBuffer<PIXEL_TYPE> *dstTiledBuffer,
573 const scene_rdl2::fb_util::PixelBuffer<PIXEL_TYPE> &srcTiledBuffer,
574 const scene_rdl2::fb_util::Tiler &tiler,
575 HAS_DATA_FUNC hasData,
576 bool parallel)
577{
578 MNRY_ASSERT(dstTiledBuffer->getWidth() == srcTiledBuffer.getWidth());
579 MNRY_ASSERT(dstTiledBuffer->getHeight() == srcTiledBuffer.getHeight());
580 MNRY_ASSERT((dstTiledBuffer->getWidth() & 7) == 0);
581 MNRY_ASSERT((dstTiledBuffer->getHeight() & 7) == 0);
582
583 const unsigned numTiles = tiler.mNumTiles;
584
585 simpleLoop(parallel, 0u, numTiles, [&](unsigned i)
586 {
587 const unsigned tileOfs = i << 6;
588 PIXEL_TYPE *__restrict dst = dstTiledBuffer->getData() + tileOfs;
589 const PIXEL_TYPE *__restrict src = srcTiledBuffer.getData() + tileOfs;
590
591 for (unsigned y = 0; y < 8; ++y) {
592
593 for (unsigned x = 0; x < 8; ++x) {
594
595 unsigned ofs0 = ((y & ~0) << 3) + (x & ~0) + 0;
596 unsigned ofs1 = ((y & ~1) << 3) + (x & ~1) + 0;
597 unsigned ofs2 = ((y & ~3) << 3) + (x & ~3) + 18;
598 unsigned ofs3 = ((y & ~7) << 3) + (x & ~7) + 18;
599
600 unsigned resultOfs = 0xffffffff;
601
602 uint32_t mask3 = hasData(src[ofs3], tileOfs + ofs3);
603 uint32_t mask2 = hasData(src[ofs2], tileOfs + ofs2);
604 uint32_t mask1 = hasData(src[ofs1], tileOfs + ofs1);
605 uint32_t mask0 = hasData(src[ofs0], tileOfs + ofs0);
606
607 resultOfs = (mask3 & ofs3) | (~mask3 & resultOfs);
608 resultOfs = (mask2 & ofs2) | (~mask2 & resultOfs);
609 resultOfs = (mask1 & ofs1) | (~mask1 & resultOfs);
610 resultOfs = (mask0 & ofs0) | (~mask0 & resultOfs);
611
612 // Reasons for not filling in a pixel are:
613 // a) This function was called without first rendering at least
614 // a single sample per tile.
615 // b) The pixel landed on a portion of the tile which was clipped
616 // out either by viewport clipping or because the buffer
617 // wasn't aligned to tile boundaries.
618 if (resultOfs != 0xffffffff) {
619 *dst = src[resultOfs];
620 }
621
622 ++dst;
623 }
624 }
625 });
626}

655
656 unsigned baseX = baseTileX + (tileX << 3);
657 unsigned minX = std::max<unsigned>(baseX, viewport.mMinX) - baseX;
658 unsigned maxX = std::min<unsigned>(baseX + 8, viewport.mMaxX + 1) - baseX;
659 MNRY_ASSERT(minX < maxX && minX < 8 && maxX <= 8);
660
661 unsigned tileOfs = tiler.linearCoordsToCoarseTileOffset(baseX, baseY);
662 MNRY_ASSERT((tileOfs & 63) == 0);
663
664 PIXEL_TYPE *__restrict dst = dstTiledBuffer->getData() + tileOfs;
665 const PIXEL_TYPE *__restrict src = srcTiledBuffer.getData() + tileOfs;

667 for (unsigned y = minY; y < maxY; ++y) {
668
669 for (unsigned x = minX; x < maxX; ++x) {
670
671 unsigned ofs0 = ((y & ~0) << 3) + (x & ~0) + 0;
672 unsigned ofs1 = ((y & ~1) << 3) + (x & ~1) + 0;
673 unsigned ofs2 = ((y & ~3) << 3) + (x & ~3) + 18;
674 unsigned ofs3 = ((y & ~7) << 3) + (x & ~7) + 18;
