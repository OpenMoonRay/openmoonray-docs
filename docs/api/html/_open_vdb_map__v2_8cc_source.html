67void
68OpenVdbMap_v2::update()
69{
70 if (hasChanged(attrTextureCoordSource)) {
71 mRequiredAttributes.clear();
72 mOptionalAttributes.clear();
73 if (get(attrTextureCoordSource) == ispc::OPENVDBMAP_COORD_REFP) {
74 mRequiredAttributes.push_back(mIspc.mRefPKey);
75 }
76 }
77
78 mXform = fauxstd::make_unique<moonray::shading::Xform>(this, nullptr, nullptr, nullptr);
79 mIspc.mXform = mXform->getIspcXform();
80
81 if (hasChanged(attrVdbSource) ||
82 hasChanged(attrTexture) ||
83 hasChanged(attrOpenVdbGeometry)) {
84
85 mSampler.reset();
86
87 // If using OpenVdbGeometry, we'll append its transform, otherwise not.
88 const Mat4d* grid2world = nullptr;
89
90 // Resolve the vdb filename and potentially a grid2world transform
91 std::string vdbFilename;
92 if (get(attrVdbSource) == VDB_SOURCE_OPENVDB_GEOMETRY) {
93 for (SceneObject* sceneObject : get(attrOpenVdbGeometry)) {
94 if (!sceneObject) {
95 error("You must specify an OpenVdbGeometry object");
96 return;
97 }
98
99 Geometry* geom = sceneObject->asA<Geometry>();
100 if (!moonray::shading::isOpenVdbGeometry(geom)) {
101 error("The object specified for 'open_vdb_geometry' is not an OpenVdbGeometry object");
102 return;
103 }
104
105 AttributeKey<String> modelAttrKey = moonray::shading::getModelAttributeKey(geom);
106 vdbFilename = geom->get<std::string>(modelAttrKey);
107
108 // make sure we successfully resolved a filename
109 if (vdbFilename.empty()) {
110 if (get(attrShowWarnings)) {
111 warn("No .vdb filename specified for OpenVdbGeometry ", geom->getName());
112 }
113 mSamplers[geom] = nullptr; // add a null entry for this geometry in the map
114 continue; // skip to next OpenVdbGeometry in the list
115 }
116
117 // We only need the render2world transform if using P
118 const Mat4d* render2world = nullptr;
119 if (get(attrTextureCoordSource) == ispc::OPENVDBMAP_COORD_P) {
120 const SceneContext* ctx = getSceneClass().getSceneContext();
121 render2world = ctx->getRender2World();
122 }
123
124 // Construct an uninitialized OpenVdbSampler, then try to initialize it.
125 // If initialization fails for any reason, destroy this useless object
126 // immediately. Two-phase construct-and-init design pattern
127 mSamplers[geom] = fauxstd::make_unique<moonray::shading::OpenVdbSampler>();
128
129 std::string errorMsg;
130 grid2world = &geom->get(Node::sNodeXformKey);
131 if (!mSamplers[geom]->initialize(vdbFilename,
132 get(attrGridName),
133 render2world,
134 grid2world,
135 errorMsg)) {
136 mSamplers[geom].reset();
137 if (get(attrShowWarnings)) {
138 warn("Unable to load grid from file ", vdbFilename);
139 }
140 }
141 }
142 } else { // VDB_SOURCE_TEXTURE
143 vdbFilename = get(attrTexture);
144 if (vdbFilename.empty()) {
145 error("You must specify a valid .vdb filename");
146 return;
147 }
148
149 // We only need the render2world transform if using P
150 const Mat4d* render2world = nullptr;
151 if (get(attrTextureCoordSource) == ispc::OPENVDBMAP_COORD_P) {
152 const SceneContext* ctx = getSceneClass().getSceneContext();
153 render2world = ctx->getRender2World();
154 }
155
156 // Construct an uninitialized OpenVdbSampler, then try to initialize it.
157 // If initialization fails for any reason, destroy this useless object
158 // immediately. Two-phase construct-and-init design pattern
159 mSampler = fauxstd::make_unique<moonray::shading::OpenVdbSampler>();
160
161 std::string errorMsg;
162 if (!mSampler->initialize(vdbFilename,
163 get(attrGridName),
164 render2world,
165 grid2world,
166 errorMsg)) {
167 mSampler.reset();
168 if (get(attrShowWarnings)) {
169 warn("Unable to load grid from file ", vdbFilename);
170 }
171 }
172 }
173 }

176// called by scalar and vector sample funcs
177void
178OpenVdbMap_v2::sampleVdb(const Map* map,
179 moonray::shading::TLState *tls,
180 const float px, const float py, const float pz,
181 const Geometry* geom,
182 scene_rdl2::math::Color *out)
183{
184 const OpenVdbMap_v2* me = static_cast<const OpenVdbMap_v2*>(map);
185
186 const Vec3f P(px, py, pz);
187 const moonray::shading::OpenVdbSampler::Interpolation interp =
188 static_cast<moonray::shading::OpenVdbSampler::Interpolation>(me->get(attrInterpolation));
189
190 if (me->get(attrVdbSource) == VDB_SOURCE_OPENVDB_GEOMETRY) {
191 if (me->mSamplers.find(geom) == me->mSamplers.end() || !me->mSamplers.at(geom)) {
192 *out = me->get(attrDefaultValue);
193 return;
194 }
195
196 if (me->get(attrShowActiveField)) {
197 *out = me->mSamplers.at(geom)->getIsActive(tls, P) ?
198 scene_rdl2::math::Color(1.f, 1.f, 1.f) :
199 scene_rdl2::math::Color(0.f, 0.f, 0.f);
200 } else {
201 *out = me->mSamplers.at(geom)->sample(tls, P, interp);
202 }
203
204 } else {
205 if (!me->mSampler) {
206 *out = me->get(attrDefaultValue);
207 return;
208 }
209
210 if (me->get(attrShowActiveField)) {
211 *out = me->mSampler->getIsActive(tls, P) ?
212 scene_rdl2::math::Color(1.f, 1.f, 1.f) :
213 scene_rdl2::math::Color(0.f, 0.f, 0.f);
214 } else {
215 *out = me->mSampler->sample(tls, P, interp);
216 }
217 }

220void
221OpenVdbMap_v2::sample(const Map* map,
222 moonray::shading::TLState *tls,
223 const moonray::shading::State& state,
224 scene_rdl2::math::Color* sample)
225{
226 const OpenVdbMap_v2* me = static_cast<const OpenVdbMap_v2*>(map);
227
228 Vec3f P = state.getP();
229
230 if (me->get(attrTextureCoordSource) == ispc::OPENVDBMAP_COORD_REFP) {
231 state.getRefP(P);
232 } else if (me->get(attrTextureCoordSource) == ispc::OPENVDBMAP_COORD_INPUT) {
233 P = evalVec3f(me, attrInputTextureCoordinate, tls, state);
234 }
235
236 const Geometry* geom = state.getGeometryObject();
237
238 sampleVdb(map, tls, P.x, P.y, P.z, geom, sample);
239}
