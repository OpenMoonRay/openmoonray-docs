20//
21// A PixelBuffer represents a 2D buffer of elements of pixel type T.
22// It's a low level container which places no semantic constraints of any type
23// on the layout of the data contained. The layout semantics are dictated by
24// higher level code.
25//
26template<typename T>
27class PixelBuffer
28{
29 static_assert(std::is_trivially_destructible<T>::value, "We don't call destructors");
30 struct AlignedDeleter
31 {
32 void operator()(T* t) const { util::alignedFreeArray(t); }
33 };
34
35public:
36 using PixelType = T;
37 using value_type = T;
38 using pointer = T*;
39 using const_pointer = const T*;
40 using iterator = pointer;
41 using const_iterator = const_pointer;
42
43 PixelBuffer() noexcept : mData(), mRawData(nullptr), mWidth(0), mHeight(0), mBytesAllocated(0) {}
44
45 static uint32_t hudValidation(bool verbose)
46 {
47 PIXELBUFFER_VALIDATION;
48 }
49
50 bool init(unsigned width, unsigned height)
51 {
52 unsigned area = width * height;
53 unsigned bytesToAllocate = area * static_cast<unsigned>(sizeof(T));
54 MNRY_ASSERT(bytesToAllocate);
55
56 if (mBytesAllocated < bytesToAllocate) {
57 mBytesAllocated = bytesToAllocate;
58 // mRawData is a member of the Hybrid Uniform Data struct so that ISPC can access it.
59 // This (deliberately) doesn't call the constructor on objects!
60 mRawData = (uint8_t *)util::alignedMallocArray<T>(area, CACHE_LINE_SIZE), AlignedDeleter();
61 mData.reset((T*)mRawData);
62 }
63
64 MNRY_ASSERT(mBytesAllocated >= bytesToAllocate);
65
66 mWidth = width;
67 mHeight = height;
68
69 return mData.get() != nullptr;
70 }
71
72 explicit operator bool() const noexcept { return static_cast<bool>(mData); }
73
74 // Explicitly frees up any allocated memory.
75 void cleanUp() noexcept
76 {
77 mWidth = 0;
78 mHeight = 0;
79 mBytesAllocated = 0;
80 mData.reset();
81 mRawData = nullptr;
82 }
83
84 // Clear the buffer to zeros.
85 void clear()
86 {
87 static_assert(std::is_trivially_copyable<T>::value, "Calling memset");
88 if (mData) {
89 std::memset(mData.get(), 0, mBytesAllocated);
90 }
91 }
92
93 // Clear the buffer with a specific value
94 void clear(const T &val)
95 {
96 if (mData) {
97 unsigned area = mWidth * mHeight;
98 T *data = mData.get();
99
100 for (unsigned i = 0; i < area; ++i) {
101 data[i] = val;
102 }
103 }
104 }
105
106 unsigned getWidth() const noexcept { return mWidth; }
107 unsigned getHeight() const noexcept { return mHeight; }
108 unsigned getArea() const noexcept { return mWidth * mHeight; }
109
110 // x and y are straight indices here, the valid range is 0 to width - 1, 0 to height - 1 respectively
111 T &getPixel(unsigned x, unsigned y)
112 {
113 MNRY_ASSERT(mData && x < mWidth && y < mHeight);
114 return mData.get()[y * mWidth + x];
115 }
116
117 const T &getPixel(unsigned x, unsigned y) const
118 {
119 return const_cast<PixelBuffer<T> *>(this)->getPixel(x, y);
120 }
121
125 T *getRow(unsigned row)
126 {
127 MNRY_ASSERT(mData && row < mHeight);
128 return &mData.get()[row * mWidth];
129 }
130
134 const T *getRow(unsigned row) const
135 {
136 return const_cast<PixelBuffer<T> *>(this)->getRow(row);
137 }

139 template <typename U>
140 std::shared_ptr<U> getDataSharedAs()
141 {
142 // C++17 adds std::reinterpret_pointer_cast. So, have to provide it manually for now
143 // http://en.cppreference.com/w/cpp/memory/shared_ptr/pointer_cast
144 return std::shared_ptr<U>(mData, reinterpret_cast<typename std::shared_ptr<U>::element_type *>(mData.get()));
145 }
146
147 std::shared_ptr<T> getDataShared()
148 {
149 return mData;
150 }
151
155 T *getData() noexcept
156 {
157 return mData.get();
158 }
159
163 const T *getData() const noexcept
164 {
165 return mData.get();
166 }
167
168 void setPixel(unsigned x, unsigned y, const T &val)
169 {
170 MNRY_ASSERT(mData && x < mWidth && y < mHeight);
171 mData.get()[y * mWidth + x] = val;
172 }
173
174 void addPixel(unsigned x, unsigned y, const T &val)
175 {
176 MNRY_ASSERT(mData && x < mWidth && y < mHeight);
177 mData.get()[y * mWidth + x] += val;
178 }
179
180 void clone(const PixelBuffer<T> &src)
181 {
182 static_assert(std::is_trivially_copyable<T>::value, "Calling memcpy");
183 init(src.getWidth(), src.getHeight());
184 memcpy(mData.get(), src.mData.get(), getArea() * sizeof(T));
185 }
186
187private:
188 // C++ uses a shared_ptr mData, so that is broken out of the Hybrid Uniform Data struct.
189 std::shared_ptr<T> mData; // Pixel[0] is the bottom left of the image.
190 PIXELBUFFER_MEMBERS;
191};
