20/// The phase function is based on Henyey Greenstein model that is quite
21/// standard for production rendering. We can extend to support other models
22/// if there is actual need.
23class VolumePhase
24{
25public:
26 explicit VolumePhase(float g): mG(g) {}
27
28 float eval(const scene_rdl2::math::Vec3f& wo, const scene_rdl2::math::Vec3f& wi) const {
29 if (scene_rdl2::math::abs(mG) <= 1e-3f) {
30 return scene_rdl2::math::sOneOverFourPi;
31 } else {
32 float cosTheta = scene_rdl2::math::dot(wo, wi);
33 float denom = 1 + mG * mG + 2 * mG * cosTheta;
34 return scene_rdl2::math::isZero(denom) ? 0.0f :
35 scene_rdl2::math::sOneOverFourPi * (1 - mG * mG) / (denom * scene_rdl2::math::sqrt(denom));
36 }
37 }
38
39 scene_rdl2::math::Vec3f sample(const scene_rdl2::math::Vec3f& wo, float u1, float u2) const {
40 if (scene_rdl2::math::abs(mG) <= 1e-3f) {
41 // isotropic case
42 return shading::sampleSphereUniform(u1, u2);
43 } else {
44 // anisotropic case
45 // Henyey Greenstein model can be analytically inversed
46 u1 = scene_rdl2::math::clamp(u1, scene_rdl2::math::sEpsilon);
47 float sqrTerm = (1.0f - mG * mG) / (1.0f - mG + 2.0f * mG * u1);
48 float cosTheta = (1.0f + mG * mG - sqrTerm * sqrTerm) / (2.0f * mG);
49 float sinTheta = scene_rdl2::math::sqrt(scene_rdl2::math::max(0.0f,
50 1.0f - cosTheta * cosTheta));
51 float phi = scene_rdl2::math::sTwoPi * u2;
52 float cosPhi, sinPhi;
53 scene_rdl2::math::sincos(phi, &sinPhi, &cosPhi);
54 scene_rdl2::math::Vec3f localDir(sinTheta * cosPhi, sinTheta * sinPhi, cosTheta);
55 scene_rdl2::math::ReferenceFrame frame(-wo);
56 return frame.localToGlobal(localDir);
57 }
58 }
59
60 float pdf(const scene_rdl2::math::Vec3f& wo, const scene_rdl2::math::Vec3f& wi) const {
61 return eval(wo, wi);
62 }
63
64private:
65 float mG;
66};
