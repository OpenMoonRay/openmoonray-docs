34///
35/// @class Bssrdf Bssrdf.h <shading/bssrdf/Bssrdf.h>
36/// @brief Base class to implement a Bssrdf with importance sampling.
37///
38/// The integrator uses a local multiple scattering sampling as the
39/// method described in:
40/// [2] "Efficient Rendering of Local Subsurface Scattering", Mertens et al.,
41/// Pacific Conference on Computer Graphics and Applications, 2003
42/// The integrator uses a global multiple scattering sampling as the
43/// approximation described in:
44/// [4] "Efficient rendering of human skin", D'Eon et al., EGSR 2007
45///
46
47class Bssrdf
48{
49public:
50 enum Property {
51 PROPERTY_NONE = 0,

53 PROPERTY_RADIUS = 1 << 1,
54 PROPERTY_PBR_VALIDITY = 1 << 2

85 /// Sampling API:
86
87 /// This function returns Rd(r), the diffuse reflectance
88 /// The radius r should be in render space
89 /// The global flag tells the bssrdf whether this is for local scattering
90 /// or global scattering.
91 /// Important note: it is the responsibility of the integrator to query
92 /// getScale() and getTransmissionFresnel() and to apply it accordingly.
93 /// Unlike for Bsdf, the result of eval() does NOT contain any scale nor
94 /// fresnel term.
95 virtual scene_rdl2::math::Color eval(float r, bool global = false) const = 0;
96
97 /// Sample a position around the origin, in the tangent plane as defined by N.
98 /// We also return the distance r from the origin for the sampled position.
99 /// It's up to the integrator to re-project the sampled points onto the surface.
100 /// We return the pdf of sampling dPi with respect to surface area
101 /// measure (this is sTwoPi * r away from the radial pdf of sampling r).
102 virtual float sampleLocal(float r1, float r2, scene_rdl2::math::Vec3f &dPi, float &r) const = 0;
103
104 /// This is only needed for unit-testing and verify that the pdf integrates to 1.
105 virtual float pdfLocal(float r) const = 0;
106
107 /// For Diffusion-Based BSSRDFs, we use the analytically computed
108 /// diffuse reflectance to compute the area-compensation factor
109 virtual scene_rdl2::math::Color diffuseReflectance() const = 0;
110
111 /// Sample a direction going into the sub-surface, to go search for global
112 /// scattering, through a thin layer. It's up to the caller to trace through
113 /// the volume to find the back-side of the thin layer (Pi, r) and apply the
114 /// Bssrdf accordingly.
115 /// We return the pdf of sampling the direction ws with respect to the
116 /// solid angle measure.
117 // TODO: Move cosThetaMax to ctor ?
118 float sampleGlobal(float r1, float r2, float cosThetaMax, scene_rdl2::math::Vec3f &ws) const;
119
120
121 /// This is the maximum radius the integrator should use to integrate
122 /// subsurface scattering. Beyond this radius, the function Rd(d) is
123 /// negligibly small.
124 float getMaxRadius() const { return mMaxRadius; }
125
126 const scene_rdl2::math::ReferenceFrame &getFrame() const { return mFrame; }
127
128 /// A label can be set on a bssrdf
129 void setLabel(int label) { mLabel = label; }
130 int getLabel() const { return mLabel; }
131
132 /// bssrdfs can have certain properties (e.g. Color)
133 /// this API is used to query if a bssrdf has a particular property and to

135 bool hasProperty(Property property) const { return mPropertyFlags & property; }
136 int32_t getPropertyFlags() const { return mPropertyFlags; }
137 virtual bool getProperty(Property property, float *dest) const;

139 // prints out a description of this bssrdf with the provided indentation
