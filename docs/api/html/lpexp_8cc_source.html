45lpexp::FirstLast
46lpexp::Cat::genAuto(NdfAutomata &automata)const
47{
48 NdfAutomata::State * first = NULL;
49 NdfAutomata::State * last = NULL;
50 // Sequentially create the states for the expressions and link them all by
51 // lambda transitions. Making the begin state of the first one our begin, and the
52 // end state of the last one our end
53 for (std::list<LPexp *>::const_iterator i = m_children.begin(); i != m_children.end(); ++i) {
54 FirstLast fl = (*i)->genAuto(automata);
55 if (!first)
56 first = fl.first;
57 else
58 // This is not the first of the list, so link it from the previous
59 // one end state
60 last->addTransition(lambda, fl.first);
61 last = fl.second;
62 }
63 return FirstLast(first, last);
64}

120lpexp::FirstLast
121lpexp::Orlist::genAuto(NdfAutomata &automata)const
122{
123 // Cat was like a serial circuit and this is a parallel one. We need
124 // two new states begin and end
125 NdfAutomata::State *begin = automata.newState();
126 NdfAutomata::State *end = automata.newState();
127 for (std::list<LPexp *>::const_iterator i = m_children.begin(); i != m_children.end(); ++i) {
128 // And then for every child we create its part of automata and link our begin to its
129 // begin and its end to our end with lambda transitions
130 FirstLast fl = (*i)->genAuto(automata);
131 begin->addTransition(lambda, fl.first);
132 fl.second->addTransition(lambda, end);
133 }
134 return FirstLast(begin, end);
135}

166lpexp::FirstLast
167lpexp::Repeat::genAuto(NdfAutomata &automata)const
168{
169 NdfAutomata::State *begin = automata.newState();
170 NdfAutomata::State *end = automata.newState();
171 FirstLast fl = m_child->genAuto(automata);
172 begin->addTransition(lambda, fl.first);
173 fl.second->addTransition(lambda, end);
174 // Easy, make its begin and end states almost the same with
175 // lambda transitions so it can repeat for ever
176 begin->addTransition(lambda, end);
177 end->addTransition(lambda, begin);
178 return FirstLast(begin, end);
179}

183lpexp::FirstLast
184lpexp::NRepeat::genAuto(NdfAutomata &automata)const
185{
186 NdfAutomata::State *first = NULL;
187 NdfAutomata::State *last = NULL;
188 int i;
189 // This is a bit trickier. For {m.n} we first make a concatenation of
190 // the child expression m times
191 for (i = 0; i < m_min; ++i) {
192 FirstLast fl = m_child->genAuto(automata);
193 if (!first)
194 first = fl.first;
195 else
196 last->addTransition(lambda, fl.first);
197 last = fl.second;
198 }
199 // And then n - m aditional movements. But we make them optional using
200 // lambda transitions
201 if (!last && i < m_max)
202 first = last = automata.newState();
203 for (; i < m_max; ++i) {
204 FirstLast fl = m_child->genAuto(automata);
205 last->addTransition(lambda, fl.first);
206 // Since this repetitions are optional, put a bypass with lambda
207 last->addTransition(lambda, fl.second);
208 last = fl.second;
209 }
210 return FirstLast(first, last);
211}

215void
216lpexp::Rule::genAuto(NdfAutomata &automata)const
217{
218 // First generate the actual automata
219 FirstLast fl = m_child->genAuto(automata);
220 // now, put the rule in the last state (making it a final state)
221 fl.second->setRule(m_rule);
222 // And then make its begin state accessible from the master initial state
223 // of the automata so it becomes initial too
224 automata.getInitial()->addTransition(lambda, fl.first);
225}
