22//
23// Frame buffer which containes entire image. This is one frame of rendered image.
24// Under distributed MCRT situation, entire image is split into multiple PartialFrames.
25// This RenderFb is used for re-comstruct perticular frame from multiple PartialFrame messages which
26// are sent from upstream MCRT computations.
27// Internally keeps track of PartialFrame receive condition from different upstream MCRT computations.
28//
29class RenderFb
30{
31public:
32 RenderFb() : mNumMachines(0), mReceived(), mProgress(), mCompleteTotal(0), mProgressTotal(0.0f) {}
33 ~RenderFb() {}
34
35 void setup(int numMachines);
36
37 void reset();
38
39 bool storePartialFrame(const PartialFrame &partialFrame); // return complete condition
40 bool isActive() const { return (mCompleteTotal > 0)? true: false; }
41 bool isComplete() const { return (mCompleteTotal == mNumMachines)? true: false; }
42
43 // return complete percentage (0.0~1.0)
44 float unpackSparseTiles(const std::vector<std::vector<fb_util::Tile>> &tiles, fb_util::VariablePixelBuffer &tiledFrame);
45
46 void show() const;
47
48protected:
49 int mNumMachines; // number of upstream MCRT computation
50
51 std::vector<std::vector<uint8_t>> mUpstreamBuffers;
52 std::unique_ptr<bool []> mReceived;
53 std::unique_ptr<float []> mProgress;
54
55 int mCompleteTotal;
56 float mProgressTotal;
57};

59//
60// Multiple frames of RenderFb.
61//
62// We need some number (like ~ 10frames) of multiple frames RenderFb array in order to manage
63// re-construct render frame more robustly.
64// In general, under high number of MCRT computation case, It's pretty difficult to sync all
65// MCRT computations are working on same frame. Sometimes Merger might have possibility to received
66// different frame from different upstream MCRT computations. This means Merger need to be smart enough
67// to understand frameId difference to re-construct frame. This is the main purpose of RenderFbArray.
68//
69// RenderFbArray keeps multiple frames which defined by startFrameId and endFrameId.
70// Capacity of array is totalCacheFrame = endFrameId - startFrameId + 1.
71// startFrameId/endFrameId information is updated by shiftFbTbl() API. one shiftFbTbl() call increments
72// both of mStartFrameId and mEndFrameId. In this case oldest frameFb data is destructed and recycled
73// by new frame.
74//
75class RenderFbArray
76{
77public:
78 RenderFbArray(int totalCacheFrame, int numMachines);
79
80 // re-construct image from PartialFrame Message.
81 // return true when all frames are received and frame is completed.
82 // return false when still some partialFrame is not ready yet.
83 bool storePartialFrame(const PartialFrame &partial, const math::Viewport &vp); // return complete frame condition
84
85 // Shift internal frame buffer table and dispose oldest fb data and create empty new fb
86 // Both mStartFrameId and mEndFrameId are incremented.
87 void shiftFbTbl();
88
89 // Test fbArray is already accessed or not.
90 bool isActive() { return (mStartFrameId >= 0)? true: false; }
91
92 // Try to find complete frame and set first complete frame as startFrame and return true
93 // If can not find complete frame, return false
94 bool completeTestAndShift();
95 bool isCompleteLocal(int localFrameId) { return mFbTbl[localFrameId]->isComplete(); }
96 bool isComplete(int globalFrameId) { return getRenderFb(globalFrameId)->isComplete(); }
97
98 RenderFb *getLocal(int localFrameId) { return mFbTbl[localFrameId]; }
99
100 int getStartFrameId() const { return mStartFrameId; }
101
102 void show() const;
103
104protected:
105 int getLocalFrameId(int globalFrameId) { return globalFrameId - mStartFrameId; }
106 RenderFb *getRenderFb(int globalFrameId) { return mFbTbl[getLocalFrameId(globalFrameId)]; }
107
108 //------------------------------
109
110 int mTotalCacheFrame;
111 int mNumMachines;
112
113 std::unique_ptr<RenderFb []> mFbArray;
114
115 int mStartFrameId; // global frameId
116 int mEndFrameId; // global frameId
117 std::vector<RenderFb *> mFbTbl;
118};
