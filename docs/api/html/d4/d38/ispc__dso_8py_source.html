8'''
9Methods for creating rdl2 dsos from ispc code
10
11The shader writer is expected to author:
12 dsoName.json
13 dsoName.ispc
14 dsoName.cc
15
16From dsoName.json, we produce the intermediate targets
17 attributes.cc - expected to be included by dsoName.cc
18 attributesISPC.cc - built as dso source
19 attributes.isph - expected to be included by dsoName.ispc
20
21and then pass this source to build the rdl dso and proxy dso
22'''

32def IspcKeyType(rdl2Type):
33 '''
34 From the rdl2::Type, determine the corresponding ISPC key type.
35 Not all rdl2 attributes types are supported in ISPC code.
36 '''
37 dataType = ''
38 if rdl2Type == 'Bool':
39 dataType += 'BoolAttrKeyISPC'
40 elif rdl2Type == 'Int':
41 dataType += 'IntAttrKeyISPC'
42 elif rdl2Type == 'Float':
43 dataType += 'FloatAttrKeyISPC'
44 elif rdl2Type == 'Vec2f':
45 dataType += 'Float2AttrKeyISPC'
46 elif rdl2Type == 'Vec3f' or rdl2Type == 'Rgb':
47 dataType += 'Float3AttrKeyISPC'
48 elif rdl2Type == 'Vec4f' or rdl2Type == 'Rgba':
49 dataType += 'Float4AttrKeyISPC'
50 else:
51 dataType += 'UNKNOWN_ATTR_TYPE'
52 return dataType

54def IspcType(rdl2Type):
55 '''
56 From the rdl2::Type, determine the corresponding ISPC shader type.
57 Not all rdl2 attributes types are supported in ISPC shader code.
58 '''
59 shaderType = ''
60 if rdl2Type == 'Bool':
61 shaderType += 'bool'
62 elif rdl2Type == 'Int':
63 shaderType += 'int'
64 elif rdl2Type == 'Float':
65 shaderType += 'float'
66 elif rdl2Type == 'Vec2f':
67 shaderType += 'Vec2f'
68 elif rdl2Type == 'Vec3f':
69 shaderType += 'Vec3f'
70 elif rdl2Type == 'Rgb':
71 shaderType = 'Color'
72 return shaderType

101def defineGetAttrFn(shaderType, retType, attrName):
102 text = ('inline uniform ' + retType +
103 '\n' + getAttrFn(attrName) +
104 '(const uniform ' + shaderType + ' * uniform obj) {\n' +
105 ' return ' + getAttrFn(retType) + '(obj, ' + attrName + ');\n}\n')
106 return text

108def defineEvalAttrFn(shaderType, retType, attrName):
109 # evalAttrName(obj, tls, state);
110 text = ('inline varying ' + retType +
111 '\n' + evalAttrFn(attrName) +
112 '(const uniform ' + shaderType + ' * uniform obj, ' +
113 'uniform ShadingTLState *uniform tls, ' +
114 'const varying State &state) {\n' +
115 ' return ' + evalAttrFn(retType) +
116 '(obj, tls, state, ' + attrName + ');\n}\n')
117
118 return text

120def defineEvalCompFn(shaderType, compName, attrColor, attrFactor, attrShow):
121 # evalCompName
122 text = 'inline varying Color\n'
123 text +=(evalCompFn(compName) +
124 '(const uniform ' + shaderType + ' * uniform obj, ' +
125 'uniform ShadingTLState *uniform tls, ' +
126 'const varying State &state)\n')
127 text += '{\n'
128 text += ' Color result = Color_ctor(0.f);\n'
129 text += ' if (' + getAttrFn(attrShow) + '(obj)) {\n'
130 text += ' const uniform float factor = ' + getAttrFn(attrFactor) + '(obj);\n'
131 text += ' if (!isZero(factor)) {\n'
132 text += ' result = ' + evalAttrFn(attrColor) + '(obj, tls, state) * factor;\n'
133 text += ' }\n'
134 text += ' }\n'
135 text += ' return result;\n'
136 text += '}\n'
137
138 return text

140def defineEvalNormalFn(shaderType, normal, attrNameNormal, attrNameBlend, attrNameSpace):
141 # evalNormalAttrName
142 text = 'inline Vec3f\n'
143 text +=(evalNormalFn(normal) + '(const uniform ' + shaderType +
144 ' * uniform obj, uniform ShadingTLState *uniform tls, ' +
145 'const varying State &state)\n')
146 text += '{\n'
147 text +=(' return evalNormal(obj, tls, state, ' + attrNameNormal +
148 ', ' + attrNameBlend + ', ' + attrNameSpace + ');\n')
149 text += '}\n'

151 return text

153def mergeJson(source, destination):
154 for key, value in source.items():
155 if isinstance(value, dict):
156 node = destination.setdefault(key, OrderedDict())
157 mergeJson(value, node)
158 else:
159 destination[key] = value

161def declareAttribute(data, text, attr, ns, dataKeywords, aliases, name=''):
162 if not name == '':
163 text += ('%s::%s = sceneClass.declareAttribute<%s>("%s"' %
164 (ns, attr, data['type'], name))
165 else:
166 text += ('%s::%s = sceneClass.declareAttribute<%s>("%s"' %
167 (ns, attr, data['type'], data['name']))
168 if 'default' in data:
169 text += ', %s' % data['default']
170
171 flags = 'FLAGS_NONE'
172 if 'flags' in data:
173 flags = data['flags']
174 interface = 'INTERFACE_GENERIC'
175 if 'interface' in data:
176 interface = data['interface']
177 aliasStr = '{' + ', '.join('"' + x + '"' for x in aliases) + '}'
178 text += ', %s, %s, %s' % (flags, interface, aliasStr)
179 text += ');\n'
180
181 if 'group' in data:
182 text += ('sceneClass.setGroup("%s", %s::%s);\n' %
183 (data['group'], ns, attr))
184 if 'enum' in data:
185 for enum, value in data['enum'].iteritems():
186 text += ('sceneClass.setEnumValue(%s::%s, %s, "%s");\n' %
187 (ns, attr, value, enum))
188 if 'metadata' in data:
189 for metaKey, metaStr in data['metadata'].iteritems():
190 text += ('sceneClass.setMetadata(%s::%s, "%s", "%s");\n' %
191 (ns, attr, metaKey, metaStr))
192
193 # Add rest as metadata, including "comment"
194 for metaKey, metaStr in data.iteritems():
195 if not metaKey in dataKeywords :
196 text += ('sceneClass.setMetadata(%s::%s, "%s", "%s");\n' %
197 (ns, attr, metaKey, metaStr))
198 return text

200def declareISPCAttributeKey(data, text, attr, ns):
201 # if the keytype is unknown, then very likely this attribute is
202 # not supported in ispc and exists only for c++ code
203 keyType = IspcKeyType(data['type'])
204 if keyType != 'UNKNOWN_ATTR_TYPE':
205 text += ('namespace ' + ns + ' { extern AttributeKey<%s> %s;' %
206 (data['type'], attr) + ' }\n')
207 text += ('%s *%s = (%s *) &%s::%s;\n' %
208 (keyType, attr, keyType, ns, attr))
209 return text

211def declareISPCAttributeFunctions(data, text, attr, shader):
212 # if the keytype is unknown, then very likely this attribute is
213 # not supported in ispc and exists only for c++ code
214 keyType = IspcKeyType(data['type'])
215 if keyType != 'UNKNOWN_ATTR_TYPE':
216 text += '//-------------------------------------------------\n'
217 text += '// ' + attr + '\n'
218 text += '//-------------------------------------------------\n'
219 text += 'extern uniform ' + keyType + ' * uniform ' + attr + ';\n'
220 ispcType = IspcType(data['type'])
221 shaderType = shader['type']
222 needsEval = False
223 if ispcType == 'Color' or ispcType == 'float' or ispcType == 'Vec2f' or ispcType == 'Vec3f':
224 if 'flags' in data and 'FLAGS_BINDABLE' in data['flags']:
225 needsEval = True
226 text += defineGetAttrFn(shaderType, ispcType, attr)
227 if needsEval:
228 text += defineEvalAttrFn(shaderType, ispcType, attr)
229 return text

231def addIncludeDependencies(target, source, env):
232 '''
233 Add directive includes as target dependencies.
234 '''
235 shader = json.loads(env.File(source[0]).get_contents(), object_pairs_hook=OrderedDict)
236 if 'directives' in shader.keys():
237 for attr, data in shader['directives'].iteritems():
238 if attr == 'include' and type(data) == list:
239 for includeFile in data:
240 includeFile = env.File('#%s' % includeFile)
241 env.Depends(target, includeFile)

243def BuildIspcDsoSource(target, source, env):
244 '''
245 Create attributes.cc (target[0]), attributesISPC.cc (target[1]),
246 attributes.isph (target[2]), labels.h (target[3]),
247 and labels.isph (target[4]) from dsoName.json (source[0])
248 '''
249
250 # load the shader attribute definition from the json file
251 shader = json.loads(env.File(source[0]).get_contents(), object_pairs_hook=OrderedDict)
252
253 # Include other json files from a directives
254 # block which looks like the following:
255 #{
256 # "directives": {
257 # "include": [
258 # "lib/shaders/dwabase/json/specular.json",
259 # "lib/shaders/dwabase/json/metallic.json"
260 # ]
261 # }
262 #}
263 if 'directives' in shader.keys():
264 for attr, data in shader['directives'].iteritems():
265 if attr == 'include' and type(data) == list:
266 for includeFile in data:
267 includeFile = env.File('#%s' % includeFile)
268 try:
269 includeJson = json.loads(includeFile.get_contents(), object_pairs_hook=OrderedDict)
270 mergeJson(includeJson, shader)
271 except ValueError as err:
272 print 'Error with file: %s' % includeFile
273 print 'Error: %s' % err
274
275 shader.pop('directives')
276
277 # Process group ordering block which looks like this:
278 #{
279 # "groups": {
280 # "order": [
281 # "Specular",
282 # "Diffuse",
283 # "Clearcoat",
284 # "Iridescence",
285 # "Transmission",
286 # "Emission",
287 # "Normal",
288 # "Common"
289 # ]
290 # }
291 #}
292 if 'groups' in shader.keys():
293 for groupsAttr, groupsData in shader['groups'].iteritems():
294 if groupsAttr == 'order' and type(groupsData) == list:
295 attributesSorted = OrderedDict()
296 for group in groupsData:
297 for attr, attrData in shader['attributes'].iteritems():
298 if attrData['group'] == group:
299 attributesSorted[attr] = shader['attributes'].pop(attr)
300 for attr, attrData in shader['attributes'].iteritems():
301 attributesSorted[attr] = shader['attributes'].pop(attr)
302 shader['attributes'] = attributesSorted
303 shader.pop('groups')
304
305 # unlike standard rdl2 dsos, we place the shader attributes into
306 # a non-anonymous namespace so we can set the (global) ispc
307 # attribute keys to the same names. at the end of attributes.cc
308 # we set 'using namespace ns' which should have a similar effect
309 # to using an anonymous namespace.
310 ns = shader['name'] + '_attr'
311
312 # create attributes.cc (target[0])
313 # included by dsoName.cc
314 text = '#include <scene_rdl2/scene/rdl2/rdl2.h>\n'
315 text += 'using namespace scene_rdl2::rdl2;\n'
316 text += 'RDL2_DSO_ATTR_DECLARE_NS(' + ns + ')\n'
317 for attr, data in shader['attributes'].iteritems():
318 if 'multi' in data:
319 for i in range(int(data['multi'])):
320 multiAttr = attr + str(i)
321 text += 'AttributeKey<%s> %s;\n' % (data['type'], multiAttr)
322 else:
323 text += 'AttributeKey<%s> %s;\n' % (data['type'], attr)
324 text += 'RDL2_DSO_ATTR_DEFINE(%s)\n' % (shader['type'])
325
326 # Establish set of attribute keywords that have specific meaning.
327 # Any data who's key is not in this list will be added as metadata
328 # Obviously, if 'metadata' exists it will also be added as metadata
329 dataKeywords = ['type', 'name', 'default', 'flags', 'interface', 'aliases',
330 'group', 'enum', 'multi', 'metadata']
331 for attr, data in shader['attributes'].iteritems():
332
333 # attributes can have aliases
334 aliases = list()
335 if 'aliases' in data:
336 aliases = list(data['aliases'])
337
338 if 'multi' in data:
339 for i in range(int(data['multi'])):
340 name = data['name'] + str(i)
341 attrName = attr + str(i)
342 attrAliases = aliases
343 for j in range(len(attrAliases)):
344 attrAliases[j] = attrAliases[j] + str(i)
345 text = declareAttribute(data, text, attrName, ns, dataKeywords, attrAliases, name)
346 continue;
347 else:
348 text = declareAttribute(data, text, attr, ns, dataKeywords, aliases)
349
350 if 'labels' in shader:
351 text += 'static const char *labels[] = {\n'
352 for variable, label in shader['labels'].iteritems():
353 text += ' "%s",\n' % label
354 text += ' nullptr\n};\n'
355 text += 'sceneClass.declareDataPtr("labels", labels);\n'
356
357 if 'interface_flags' in shader:
358 text += 'rdl2_dso_interface |= %s;\n' % shader['interface_flags']
359
360 text += 'RDL2_DSO_ATTR_END\n'
361 text += 'using namespace ' + ns + ';\n'
362 f = open(str(target[0]), "wb")
363 f.write(text)
364 f.close()
365
366 # create attributesISPC.cc (target[1])
367 # this file creates the global ispc attribute keys
368 # namespace ns { extern AttributeKey<Type> attrName; }
369 # TypeAttrKeyISPC *attrName = (TypeAttrKeyISPC *) &ns::attrName;
370 text = '#include <scene_rdl2/scene/rdl2/rdl2.h>\n'
371 text += '#include <scene_rdl2/scene/rdl2/ISPCSupport.h>\n'
372 text += 'using namespace scene_rdl2::rdl2;\n'
373 for attr, data in shader['attributes'].iteritems():
374 if 'multi' in data:
375 for i in range(int(data['multi'])):
376 attrName = attr + str(i)
377 text = declareISPCAttributeKey(data, text, attrName, ns)
378 continue;
379 else:
380 text = declareISPCAttributeKey(data, text, attr, ns)
381 f = open(str(target[1]), "wb")
382 f.write(text)
383 f.close()
384
385 # create attributes.isph (target[2])
386 # included by dsoName.ispc
387 # extern uniform <Type>AttrKeyISPC * uniform attrKey;
388 assert str(target[2]).endswith('.isph')
389 text = '#pragma once\n'
390 text += '#include <rendering/shading/ispc/Shading.isph>\n'
391 text += '#include <scene_rdl2/scene/rdl2/rdl2.isph>\n'
392 text += '#include <scene_rdl2/scene/rdl2/ISPCSupport.h>\n'
393 for attr, data in shader['attributes'].iteritems():
394 if 'multi' in data:
395 for i in range(int(data['multi'])):
396 attrName = attr + str(i)
397 text = declareISPCAttributeFunctions(data, text, attrName, shader)
398 continue;
399 else:
400 text = declareISPCAttributeFunctions(data, text, attr, shader)
401
402 if 'components' in shader:
403 for comp, data in shader['components'].iteritems():
404 shaderType = shader['type']
405 text += '//-------------------------------------------------\n'
406 text += '// Component ' + comp + '\n'
407 text += '//-------------------------------------------------\n'
408 text += defineEvalCompFn(shaderType, comp, data['color'],
409 data['factor'], data['show'])
410 if 'normals' in shader:
411 for normal, data in shader['normals'].iteritems():
412 text += '//-----------------------------------------------------\n'
413 text += '// Normal ' + normal + '\n'
414 text += '//-----------------------------------------------------\n'
415 shaderType = shader['type']
416 text += defineEvalNormalFn(shaderType, normal, data['value'], data['dial'], data['space'])
417 f = open(str(target[2]), "wb")
418 f.write(text)
419 f.close()
420
421 # create labels.h (target[3])
422 # optionally included by shaders that assign labels to lobes
423 assert str(target[3]).endswith('.h')
424 text = '#pragma once\n'
425 if 'labels' in shader:
426 val = 1
427 for variable, label in shader['labels'].iteritems():
428 text += 'static const int %s = %d;\n' % (variable, val)
429 val = val + 1
430 f = open(str(target[3]), 'wb')
431 f.write(text)
432 f.close()
433
434 # create labels.isph
435 # optionally included by shaders that assign labels to lobes
436 assert str(target[4]).endswith('.isph')
437 text = '#pragma once\n'
438 if 'labels' in shader:
439 val = 1
440 for variable, label in shader['labels'].iteritems():
441 text += 'static const uniform int %s = %d;\n' % (variable, val)
442 val = val + 1
443 f = open(str(target[4]), 'wb')
444 f.write(text)
445 f.close()

447def DWAIspcDso(parentEnv, target, ccSource, ispcSource, jsonSource, **kwargs):
448 '''
449 return a dso based on source (.cc, .ispc files) and attribute_source (.json)
450 several source files are generated from the .json
451 '''
452
453 # autogenerate attributes.cc, attributesISPC.cc, attributes.isph,
454 # labels.h, and labels.isph
455 # from the json source in the directory of the target
456 targetDir = os.path.dirname(target.abspath)
457 autogen = [targetDir + '/attributes.cc',
458 targetDir + '/attributesISPC.cc',
459 targetDir + '/attributes.isph',
460 targetDir + '/labels.h',
461 targetDir + '/labels.isph']
462 builder = parentEnv.Command(autogen, jsonSource, BuildIspcDsoSource)
463 # If I add the include dependencies to the autogen file they will get
464 # re-generated each build. This way they only re-regenerate when an
465 # include json file is changed.
466 addIncludeDependencies(builder, jsonSource, parentEnv)
467 # Scons is smart enough to know that changes to the program
468 # code of the BuildIspcDsoSource() function should trigger a rebuild of
469 # the builder's targets, but it is not smart enough to know that
470 # changes to the program text of functions that BuildIspcDsoSource()
471 # calls should also trigger a change. For example, adding a
472 # print 'foo' to BuildIspcDsoSource() will trigger a rebuild of
473 # the targets as expected, but adding a print 'foo' to
474 # IspcKeyType() (a function called by BuildIspcDsoSource()) will not.
475 # The solution to this problem that I've come up with is to depend the builder
476 # on this file itself. I use BuildIspcDsoSource.func_code.co_filename
477 # instead of __file__ because __file__ can be either the .py or compiled
478 # .pyc file while the former is always and unambiguously the .py file.
479 # The other, less desirable solution would be to remove all the function
480 # calls from BuildIspcDsoSource(), but that would substantially reduce
481 # the readability of an already fairly complicated function.
482 parentEnv.Depends(builder, BuildIspcDsoSource.func_code.co_filename)
483
484 # now build the dso with the correct paths
485 env = parentEnv.Clone();
486
487 env.AppendUnique(CPPPATH=[env.Dir('.')])
488
489 sources = [ccSource]
490 sources += [autogen[1]]
491 ispc_output, ispc_header = env.IspcShared([ispcSource], **kwargs)
492 sources += ispc_output
493
494 dso = env.DWARdl2Dso(target, sources, RDL2_ATTRIBUTES_SOURCE = autogen[0],
495 RPATH_AUTO_ORIGIN_RT_DIR='$INSTALL_DIR/rdl2dso',
496 **kwargs)
497 bc = env.SharedIspc2Bc(ispcSource, **kwargs)
498 dso.update({'bc' : bc})
499 return dso
