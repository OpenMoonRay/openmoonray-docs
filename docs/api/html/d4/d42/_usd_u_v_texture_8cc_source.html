72void
73UsdUVTexture::update()
74{
75 const std::string filename = get(attrFile);
76 const std::size_t udimPos = filename.find("<UDIM>");
77 const bool areWeAUdim = udimPos != std::string::npos;
78
79 const scene_rdl2::rdl2::SceneVariables &sv = getSceneClass().getSceneContext()->getSceneVariables();
80 mIspc.mFatalColor = asIspc(sv.get(scene_rdl2::rdl2::SceneVariables::sFatalColor));
81
82 const bool applyGamma = (get(attrSourceColorSpace) == ispc::GAMMA_ON ||
83 get(attrSourceColorSpace) == ispc::GAMMA_AUTO);
84
85
86 // If wrap types are set to the default which uses
87 // the metadata from the file then fall back to "black"
88 moonray::shading::WrapType wrapS = static_cast<moonray::shading::WrapType>(get(attrWrapS));
89 if (wrapS == moonray::shading::WrapType::Default) {
90 wrapS = moonray::shading::WrapType::Periodic;
91 }
92
93 moonray::shading::WrapType wrapT = static_cast<moonray::shading::WrapType>(get(attrWrapT));
94 if (wrapT == moonray::shading::WrapType::Default) {
95 wrapT = moonray::shading::WrapType::Periodic;
96 }
97
98 if (areWeAUdim) {
99 // Udim update, if needed
100 mTexture.reset();
101 mIspc.mTexture = nullptr;
102 bool needsUpdate = false;
103 if (!mUdimTexture) {
104 needsUpdate = true;
105 mUdimTexture = fauxstd::make_unique<moonray::shading::UdimTexture>(this);
106 mIspc.mUdimTexture = &mUdimTexture->getUdimTextureData();
107 }
108
109 if (needsUpdate ||
110 hasChanged(attrFile) ||
111 hasChanged(attrWrapS) ||
112 hasChanged(attrWrapT) ||
113 hasChanged(attrFallback)) {
114
115 std::string errorStr;
116 if (!mUdimTexture->update(this,
117 mLogEventRegistry,
118 filename,
119 applyGamma,
120 wrapS,
121 wrapT,
122 true, // use default/fallback color
123 get(attrFallback),
124 asCpp(mIspc.mFatalColor),
125 get(attrMaxVdim),
126 get(attrUDimValues),
127 get(attrUDimFiles),
128 errorStr)) {
129 fatal(errorStr);
130 return;
131 }
132 }
133 } else {
134 // Non-udim update, if needed
135 mUdimTexture.reset();
136 mIspc.mUdimTexture = nullptr;
137 bool needsUpdate = false;
138 if (!mTexture) {
139 needsUpdate = true;
140 mTexture = fauxstd::make_unique<moonray::shading::BasicTexture>(this, mLogEventRegistry);
141 mIspc.mTexture = &mTexture->getBasicTextureData();
142 }
143 if (needsUpdate ||
144 hasChanged(attrFile) ||
145 hasChanged(attrWrapS) ||
146 hasChanged(attrWrapT) ||
147 hasChanged(attrFallback)) {
148
149 std::string errorStr;
150 if (!mTexture->update(filename,
151 applyGamma,
152 wrapS,
153 wrapT,
154 true, // use default/fallback color
155 get(attrFallback),
156 asCpp(mIspc.mFatalColor),
157 errorStr)) {
158 fatal(errorStr);
159 return;
160 }
161 }
162 }
163
164 if ((mTexture && !mTexture->isValid()) ||
165 (mUdimTexture && !mUdimTexture->isValid())) {
166 fatal("texture is not valid");
167 }
168}

170void
171UsdUVTexture::sample(const scene_rdl2::rdl2::Map *self,
172 moonray::shading::TLState *tls,
173 const moonray::shading::State &state,
174 Color *sample)
175{
176 UsdUVTexture const *me = static_cast<UsdUVTexture const *>(self);
177
178 float dsdx, dsdy, dtdx, dtdy;
179
180 Vec2f st = evalVec2f(me, attrSt, tls, state);
181
182 // TODO: How do we get input texture coordinates derivatives ?
183 dsdx = dsdy = dtdx = dtdy = 0.0;
184
185 Color4 tx;
186 if (me->mTexture) {
187 // Invert t coord.
188 st.y = 1.0 - st.y;
189
190 // Set and scale derivatives.
191 float derivatives[4] = { dsdx, -dtdx, dsdy, -dtdy };
192
193 // sample the texture
194 tx = me->mTexture->sample(tls,
195 state,
196 st,
197 derivatives);
198 } else if (me->mUdimTexture) {
199 // compute udim index
200 int udim = me->mUdimTexture->computeUdim(tls, st.x, st.y);
201 if (udim == -1) {
202 logEvent(me, tls, me->mIspc.mUsdUVTextureDataPtr->sErrorInvalidUdimCoord);
203 *sample = asCpp(me->mIspc.mFatalColor);
204 return;
205 }
206
207 // take fractional parts of st
208 st.x = st.x - int(st.x);
209 st.y = st.y - int(st.y);
210
211 // Invert t coord.
212 st.y = 1.0 - st.y;
213
214 // Set derivatives.
215 float derivatives[4] = { dsdx, -dtdx, dsdy, -dtdy };
216
217 // sample the texture
218 tx = me->mUdimTexture->sample(tls,
219 state,
220 udim,
221 st,
222 derivatives);
223 }
224
225 Color rgb(tx);
226 rgb = rgb * me->get(attrScale) + me->get(attrBias);
227
228 *sample = rgb;
229}
