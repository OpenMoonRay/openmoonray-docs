29 /*! 8-wide AVX bool type. */
30 struct avxb
31 {
32 typedef avxb Mask; // mask type for us
33 enum { size = 8 }; // number of SIMD elements
34 union { // data
35 __m256 m256;
36 struct { __m128 l,h; };
37 int32 v[8];
38 };
39
40 ////////////////////////////////////////////////////////////////////////////////
41 /// Constructors, Assignment & Cast Operators
42 ////////////////////////////////////////////////////////////////////////////////
43
44 __forceinline avxb () {}
45 __forceinline avxb ( const avxb& a ) { m256 = a.m256; }
46 __forceinline avxb& operator=( const avxb& a ) { m256 = a.m256; return *this; }
47
48 __forceinline avxb( const __m256 a ) : m256(a) {}
49 __forceinline operator const __m256&( void ) const { return m256; }
50 __forceinline operator const __m256i( void ) const { return _mm256_castps_si256(m256); }
51 __forceinline operator const __m256d( void ) const { return _mm256_castps_pd(m256); }
52
53 __forceinline avxb ( const int a )
54 {
55#if defined (__AVX2__)
56 assert(a >= 0 && a <= 255);
57 const __m256i mask = _mm256_set_epi32(0x80, 0x40, 0x20, 0x10, 0x8, 0x4, 0x2, 0x1);
58 const __m256i b = _mm256_set1_epi32(a);
59 const __m256i c = _mm256_and_si256(b,mask);
60 m256 = _mm256_castsi256_ps(_mm256_cmpeq_epi32(c,mask));
61#else
62 l = _mm_lookupmask_ps[a & 0xF];
63 h = _mm_lookupmask_ps[a >> 4];
64#endif
65 }
66
67 __forceinline avxb ( const sseb& a ) : m256(_mm256_insertf128_ps(_mm256_castps128_ps256(a),a,1)) {}
68 __forceinline avxb ( const sseb& a, const sseb& b) : m256(_mm256_insertf128_ps(_mm256_castps128_ps256(a),b,1)) {}
69 __forceinline avxb ( const __m128 a, const __m128 b) : l(a), h(b) {}
70
71 __forceinline avxb ( bool a ) : m256(avxb(sseb(a), sseb(a))) {}
72 __forceinline avxb ( bool a, bool b) : m256(avxb(sseb(a), sseb(b))) {}
73 __forceinline avxb ( bool a, bool b, bool c, bool d) : m256(avxb(sseb(a,b), sseb(c,d))) {}
74 // MoonRay: added underscores to fix compile warning
75 __forceinline avxb ( bool a, bool b, bool c, bool d, bool e, bool f, bool g, bool _h ) : m256(avxb(sseb(a,b,c,d), sseb(e,f,g,_h))) {}
