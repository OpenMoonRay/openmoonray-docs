60 template <typename F>
61 bool
62 crawlAllPix(F pixFunc) const
63 {
64 for (unsigned y = 0; y < mHeight; ++y) {
65 for (unsigned x = 0; x < mWidth; ++x) {
66 if (!pixFunc(x, y)) return false;
67 }
68 }
69 return true;
70 }

72 template <typename T>
73 uintptr_t pixAddr(int sx, int sy) const
74 {
75 return (reinterpret_cast<uintptr_t>(mArray.get()) +
76 static_cast<uintptr_t>(mTiler.linearCoordsToTiledOffset(sx, sy) * sizeof(T)));
77 }

79 template <typename T>

82 return static_cast<unsigned>(*reinterpret_cast<T *>(pixAddr<T>(sx, sy)));
83 }

85 template <typename T>

87 {

91 template <typename T>
92 unsigned atomicReservePix(int sx, int sy, unsigned sampleDelta)
93 {
94 T castDelta = static_cast<T>(sampleDelta);
95 T *addr = reinterpret_cast<T*>(pixAddr<T>(sx, sy));
96 T expected = util::atomicLoad(addr, std::memory_order_relaxed);
97 while (true) {
98 T desired = expected + castDelta;
99 if (util::atomicCompareAndSwapWeak(addr, expected, desired, std::memory_order_relaxed)) break;
100 }
101 return static_cast<unsigned>(expected);
102 }
