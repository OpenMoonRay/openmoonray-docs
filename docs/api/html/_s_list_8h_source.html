8// Low level, non-thread safe and thread safe singly linked list implementations.
9//
10// The next pointer is overlaid with the memory itself so adding a structure to
11// a list will corrupt the contents. Because of this, this list is mainly useful
12// for keeping track of unused structures.
13//
14// Each structure needs to be at least large enough to contain a pointer since
15// it holds the mNext pointer.
16//

27// Not thread safe.
28class SList
29{
30public:
31 struct Entry
32 {
33 Entry * mNext;
34 };
35
36 SList() : mHead(nullptr)
37 {
38 }
39
40 // Do not call unless you are sure we're sync'ed.
41 finline void init()
42 {
43 mHead = nullptr;
44 }
45
46 finline bool isEmpty() const
47 {
48 return mHead == nullptr;
49 }
50
51 // Only push "unused" entries into this list since it corrupts contents.
52 finline void push(Entry *entry)
53 {
54 ((Entry *)entry)->mNext = mHead;
55 mHead = ((Entry *)entry);
56 }
57
58 finline Entry *pop()
59 {
60 if (!mHead) {
61 return nullptr;
62 }
63
64 Entry *popped = mHead;
65 mHead = mHead->mNext;
66 return popped;
67 }
68
69 // Returns what was at the head of the list, or nullptr if the list was empty.
70 finline Entry *clear()
71 {
72 Entry *oldHead = mHead;
73 mHead = nullptr;
74 return oldHead;
75 }
76
77 // Never thread safe.
78 finline unsigned size() const
79 {
80 unsigned size = 0;
81 Entry *curr = mHead;
82 while (curr) {
83 curr = curr->mNext;
84 ++size;
85 }
86 return size;
87 }
88
89protected:
90 Entry * mHead;
91};
