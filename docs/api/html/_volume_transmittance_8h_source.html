35 scene_rdl2::math::Color transmittance() const
36 {
37 // total transmittance is the combination of the regular extinction

39 // block light

43 void update(const scene_rdl2::math::Color& tr, const scene_rdl2::math::Color& trh)
44 {
45 // update transmittance and holdouts transmittance
46 mTransmittanceE *= tr;
47 mTransmittanceH *= trh;
48 // See Wrenninge pg. 201. Update co-dependent minimum transmittance and
49 // alpha transmittance. Note the Wrenninge book has an error, we must use
50 // the previous transmittanceM value when computing transmittanceA.
51 scene_rdl2::math::Color oldTm = mTransmittanceMin;
52 mTransmittanceMin = trh * mTransmittanceMin + (scene_rdl2::math::Color(1.f) - trh) * mTransmittanceAlpha;
53 mTransmittanceAlpha = tr * mTransmittanceAlpha + (scene_rdl2::math::Color(1.f) - tr) * oldTm;
54 }
