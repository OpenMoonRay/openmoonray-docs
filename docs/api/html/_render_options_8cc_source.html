46void
47addOverride(std::vector<RenderOptions::AttributeOverride>& overrides,
48 const std::string& object, const std::string& attribute,
49 const std::string& value, const std::string& binding,
50 bool isSceneVar = false)
51{
52 // Look for an existing override of the same object/attribute.
53 auto iter = std::find_if(overrides.begin(), overrides.end(),
54 [&object, &attribute](const RenderOptions::AttributeOverride& attrOverride) {
55 return attrOverride.mObject == object &&
56 attrOverride.mAttribute == attribute;
57 });
58
59 // If we didn't find one, create it.
60 if (iter == overrides.end()) {
61 overrides.push_back(RenderOptions::AttributeOverride(object, attribute, isSceneVar));
62 iter = --overrides.end();
63 }
64
65 // Set the value, if provided.
66 if (!value.empty()) {
67 iter->mValue = value;
68 }
69
70 // Set the binding, if provided.
71 if (!binding.empty()) {
72 iter->mBinding = binding;
73 }
74}

122void

124{
125 Args args(argc, argv);
126 Args::StringArray values;
127 std::vector<std::string> validFlags;
128
129 // instead of changing Args class to access its data we will just
130 // consume the command line arguments and rebuild the string here
131 for( int i = 0; i < argc; i++) {
132 mCommandLine += argv[i];
133 mCommandLine += " ";
134 }
135
136 {
137 std::vector<std::string> sceneFiles;
138
139 int foundAtIndex = args.getFlagValues("-in", -1 /*get all filenames*/, values);
140 validFlags.push_back("-in");

143 sceneFiles.push_back(v);

146 }

151 }

158 for (const std::string& v : values) {
159 deltasFiles.push_back(v);
160 }
161 foundAtIndex =
162 args.getFlagValues("-deltas", -1 /*get all filenames*/, values, foundAtIndex + 1);
163 }
164 if (!deltasFiles.empty()) setDeltasFiles(std::move(deltasFiles));
165 }

203 {
204 validFlags.push_back("-snap_path");

207 setSnapshotPath(values[0]);

211 validFlags.push_back("-override_lut");

213 setColorRenderTransformOverrideLut(values[0]);
214 }
215
216 {
217 std::vector<RdlaGlobal> rdlaGlobals;
218
219 validFlags.push_back("-rdla_set");
220 int foundAtIndex = args.getFlagValues("-rdla_set", 2, values);
221 while (foundAtIndex >= 0) {
222 rdlaGlobals.emplace_back(values[0], values[1]);
223 foundAtIndex = args.getFlagValues("-rdla_set", 2, values, foundAtIndex + 1);
224 }
225
226 if (!rdlaGlobals.empty()) {
227 setRdlaGlobals(std::move(rdlaGlobals));
228 }
229 }
230
231 {
232 std::vector<AttributeOverride> overrides;
233
234
235 int foundAtIndex = args.getFlagValues("-scene_var", 2, values);
236 validFlags.push_back("-scene_var");
237 while (foundAtIndex >= 0) {

239 foundAtIndex = args.getFlagValues("-scene_var", 2, values, foundAtIndex + 1);

241
242 foundAtIndex = args.getFlagValues("-attr_set", 3, values);
243 validFlags.push_back("-attr_set");
244 while (foundAtIndex >= 0) {
245 addOverride(overrides, values[0], values[1], values[2], "", true);
246 foundAtIndex = args.getFlagValues("-attr_set", 3, values, foundAtIndex + 1);

249 foundAtIndex = args.getFlagValues("-attr_bind", 3, values);

253 foundAtIndex = args.getFlagValues("-attr_bind", 3, values, foundAtIndex + 1);

256 foundAtIndex = args.getFlagValues("-out", 1, values);
257 validFlags.push_back("-out");

263 foundAtIndex = args.getFlagValues("-info", 0, values);
264 validFlags.push_back("-info");

266 addOverride(overrides, "__SceneVariables__", "info", "true", "", true);
267 foundAtIndex = args.getFlagValues("-info", 0, values, foundAtIndex + 1);
268 }
269
270 foundAtIndex = args.getFlagValues("-debug", 0, values);
271 validFlags.push_back("-debug");
272 while (foundAtIndex >= 0) {
273 addOverride(overrides, "__SceneVariables__", "debug", "true", "", true);
274 foundAtIndex = args.getFlagValues("-debug", 0, values, foundAtIndex + 1);
275 }
276
277 foundAtIndex = args.getFlagValues("-stats", 1, values);
278 validFlags.push_back("-stats");
279 while (foundAtIndex >= 0) {
280 addOverride(overrides, "__SceneVariables__", "stats file", values[0], "", true);
281 foundAtIndex = args.getFlagValues("-stats", 1, values, foundAtIndex + 1);
282 }
283
284 foundAtIndex = args.getFlagValues("-size", 2, values);
285 validFlags.push_back("-size");
286 while (foundAtIndex >= 0) {
287 addOverride(overrides, "__SceneVariables__", "frame width", values[0], "", true);

292 foundAtIndex = args.getFlagValues("-res", 1, values);

346
347 foundAtIndex = args.getFlagValues("-depth_range", -1, values);

349 while (foundAtIndex >= 0) {

351 addOverride(overrides, "__SceneVariables__", "debug rays depth range", values[0], "", true);
352 } else if (values.size() == 2) {
353 addOverride(overrides, "__SceneVariables__", "debug rays depth range", values[0] + ", " + values[1], "", true);

383 foundAtIndex = args.getFlagValues("-resumable_output", 0, values, foundAtIndex + 1);

387 validFlags.push_back("-resume_render");
388 while (foundAtIndex >= 0) {
389 addOverride(overrides, "__SceneVariables__", "resume_render", "true", "", true);
390 foundAtIndex = args.getFlagValues("-resume_render", 0, values, foundAtIndex + 1);

394 validFlags.push_back("-debug_console");
395 while (foundAtIndex >= 0) {
396 addOverride(overrides, "__SceneVariables__", "debug_console", values[0], "", true);

399
400 if (!overrides.empty()) {

402 }

404
405 validFlags.push_back("-athena_tags");
406 if (args.getFlagValues("-athena_tags", 1, values) >= 0) {
407 setAthenaTags(values[0]);
408 }
409
410 validFlags.push_back("-guid");
411 if (args.getFlagValues("-guid", 1, values) >= 0) {
412 setGUID(scene_rdl2::util::GUID(values[0]));
413 }
414
415 //
416 // For developer profiling only, not documented or exposed to user.
417 // Use like so:
418 //
419 // -set_vectorized_overrides n1 n2 n3 n4 n5 n6 n7 n8 n9
420 //
421 // where:
422 // n1 = a float containing the perThreadRayStatePoolSize / 1024
423 // n2 = a float containing the primaryRayQueueSize / 1024
424 // n3 = a float containing the incoherentRayQueueSize / 1024
425 // n4 = a float containing the occlusionQueueSize / 1024
426 // n5 = a float containing the shadeQueueSize / 1024
427 // n6 = a float containing the radianceQueueSize / 1024
428 // n7 = a float containing the shadingWorkloadChunkSize / 1024
429 // n8 = a float containing the presenceShadowsQueueSize / 1024
430 //
431 // For example, to mimic the defaults you'd use:
432 //
433 // -set_vectorized_overrides 64 0.5 1 1 0.125 0.5 0.03125 1
434 //
435 mHasVectorizedOverrides = false;
436 validFlags.push_back("-set_vectorized_overrides");
437 if (args.getFlagValues("-set_vectorized_overrides", 8, values) >= 0) {
438 mHasVectorizedOverrides = true;
439 float perThreadRayStatePoolSize = std::stof(values[0]);
440 float primaryRayQueueSize = std::stof(values[1]);
441 float incoherentRayQueueSize = std::stof(values[2]);
442 float occlusionQueueSize = std::stof(values[3]);
443 float shadeQueueSize = std::stof(values[4]);
444 float radianceQueueSize = std::stof(values[5]);
445 float shadingWorkloadChunkSize = std::stof(values[6]);
446 float presenceShadowsQueueSize = std::stof(values[7]);
447
448 mPerThreadRayStatePoolSize = unsigned(perThreadRayStatePoolSize * 1024.f);
449 mPrimaryRayQueueSize = unsigned(primaryRayQueueSize * 1024.f);
450 mIncoherentRayQueueSize = unsigned(incoherentRayQueueSize * 1024.f);
451 mOcclusionQueueSize = unsigned(occlusionQueueSize * 1024.f);
452 mShadeQueueSize = unsigned(shadeQueueSize * 1024.f);
453 mRadianceQueueSize = unsigned(radianceQueueSize * 1024.f);
454 mShadingWorkloadChunkSize = unsigned(shadingWorkloadChunkSize * 1024.f);
455 mPresenceShadowsQueueSize = unsigned(presenceShadowsQueueSize * 1024.f);
456 }
457
458 if ( ! args.allFlagsValid( validFlags ) ) {
459 fprintf(stderr, "Invalid Input Flag Found! Exiting %s...\n", argv[0]);
460 exit(-1);
461 }
462}

626

723void
724RenderOptions::setDesiredExecutionMode(const std::string &execMode)
725{
726 mcrt_common::ExecutionMode mode = mExecutionMode;
727
728 if (execMode == "auto") {
729 mode = mcrt_common::ExecutionMode::AUTO;
730 } else if (execMode == "vector" || execMode == "vectorized") {
731 mode = mcrt_common::ExecutionMode::VECTORIZED;
732 } else if (execMode == "scalar") {
733 mode = mcrt_common::ExecutionMode::SCALAR;
734 } else if (execMode == "xpu") {
735 mode = mcrt_common::ExecutionMode::XPU;
736 } else {
737 std::stringstream errMsg;
738 errMsg << "Unexpected string passed to setDesiredExecutionMode(): '" << execMode << "'!";
739 throw scene_rdl2::except::ValueError(errMsg.str());
740 }
741
742 // Call out to non-string overload.
743 setDesiredExecutionMode(mode);
744}
