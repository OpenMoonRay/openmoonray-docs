31/*
32 * Timer that is scope based.
33 * timing starts in the constructor and stops in the destructor.
34 * It updates a given stat object via its operator+=() method.
35 */
36template <typename T>
37class RAIITimer
38{
39public:
40 __forceinline RAIITimer(T &stat) {
41 mStat = &stat;
42 mStart = getTime();
43 }
44
45 // timer stops and updates upon destruction
46 __forceinline ~RAIITimer() {
47 (*mStat) += (getTime() - mStart);
48 }
49
50private:
51 T* mStat;
52 double mStart;
53};

58/*
59 * Manual timer that has a classic start / stop / lap interface
60 * the constructor does NOT start the timer
61 * the destructor does NOT stop the timer
62 * starting or stopping a timer that is already started / stopped (respectively)
63 * will trip a MNRY_ASSERT
64 */
65template <typename T>
66class Timer
67{
68public:
69 typedef T type;
70
71 __forceinline Timer(T &stat) {
72 MNRY_DURING_ASSERTS(mStarted = false);
73 mStat = &stat;
74 mStart = getTime();
75 }
76 __forceinline ~Timer() {}
77
78 __forceinline void start() {
79 MNRY_ASSERT(mStarted == false);
80 MNRY_DURING_ASSERTS(mStarted = true);
81 mStart = getTime();
82 }
83
84 __forceinline void stop() {
85 MNRY_ASSERT(mStarted == true);
86 MNRY_DURING_ASSERTS(mStarted = false);
87 (*mStat) += (getTime() - mStart);
88 }
89
90 __forceinline void lap() {
91 MNRY_ASSERT(mStarted == true);
92 double tick = getTime();
93 (*mStat) += (tick - mStart);
94 mStart = tick;
95 }
96
97private:
98 T* mStat;
99 double mStart;
100 MNRY_DURING_ASSERTS(bool mStarted);
101};
