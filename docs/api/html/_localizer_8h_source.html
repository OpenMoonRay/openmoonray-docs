18/**
19 * By "localize" we mean "copy all the external assets that a scene references
20 * to a single location". Essentially we are isolating a scene such that all
21 * its asset data and the RDL data lives in the same directory. (Unfortunately
22 * this will not help your scene speak new languages. Bummer!)
23 *
24 * A localizer can localize RDL2 input files, writing the localized result to
25 * the given output file. Multiple files can be localized with the same
26 * localizer, though configurable options are fixed at construction time.
27 */
28class Localizer
29{
30public:
31 /**
32 * Create a new localizer.
33 */
34 Localizer(bool forceOverwrite, bool relativePaths, std::string& dsoPath);
35
36 /**
37 * Localize the given RDL2 input file, copying all its dependent assets
38 * into the same directory as the output file. The output file is written
39 * with any localized attribute data updated to reflect the new file path.
40 */
41 void localize(const std::string& inFile, const std::string& outFile);
42
43private:
44 // If true, overwrite destination files if they already exist.
45 bool mForceOverwrite;
46
47 // If true, the new paths in the output RDL file will use paths relative to
48 // the RDL file instead of absolute paths.
49 bool mRelativePaths;
50
51 // The dso search path. If set, we passed in -dso_path on the command line.
52 std::string& mDsoPath;
53};
54
55} // namespace rdl2_localize
