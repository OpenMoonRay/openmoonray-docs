25///
26/// @class NormalizedDiffusionBssrdf NormalizedDiffusion.h <shading/bssrdf/NormalizedDiffusion.h>
27/// @brief Derived class to implement the Normalized Diffusion Bssrdf with importance sampling.
28///
29/// This Bssrdf implements "Normalized Diffusion" approximation as described in:
30/// [1] "Approximate Reflectance Profiles for Efficient Subsurface Scattering",
31/// Burley Christensen, Pixar Technical Memo #15-04 - July, 2014
32/// http://graphics.pixar.com/library/ApproxBSSRDF/paper.pdf
33///
34/// [2] http://www-history.mcs.st-andrews.ac.uk/history/HistTopics/Quadratic_etc_equations.html
35///
36class NormalizedDiffusionBssrdf : public Bssrdf
37{
38public:
39 // Artist-friendly constructor
40 NormalizedDiffusionBssrdf(const scene_rdl2::math::Vec3f &N, const scene_rdl2::math::Color &albedo,
41 const scene_rdl2::math::Color &radius, const scene_rdl2::rdl2::Material* material,
42 const scene_rdl2::rdl2::EvalNormalFunc evalNormalFn);
43 NormalizedDiffusionBssrdf(scene_rdl2::alloc::Arena *arena, const NormalizedDiffusionBssrdfv &bssrdfv, int lane);
44
45 ~NormalizedDiffusionBssrdf() {}
46
47 /// Sampling API:
48 /// This function returns R(r), equation (3) in [1]
49 scene_rdl2::math::Color eval(float r, bool global = false) const override;
50
51 /// We use exact importance sampling by inverting the CDF in eq (11) in [1]
52 float sampleLocal(float r1, float r2, scene_rdl2::math::Vec3f &dPi, float &r) const override;
53
54 /// This is only needed for unit-testing and verify that the pdf integrates to 1.
55 float pdfLocal(float r) const override;
56
57 /// For NormalizedDiffusion, diffuseReflectance is same as the Albedo [1]
58 scene_rdl2::math::Color diffuseReflectance() const override { return mAlbedo; }
59
60 bool getProperty(Property property, float *dest) const override;
61
62 void show(std::ostream& os, const std::string& indent) const override;
63
64private:
65 // Copy is disabled
66 NormalizedDiffusionBssrdf(const NormalizedDiffusionBssrdf &other) =delete;
67 const NormalizedDiffusionBssrdf &operator=(const NormalizedDiffusionBssrdf &other) =delete;
68
69 scene_rdl2::math::Color mAlbedo; // 'A' surface albedo
70 scene_rdl2::math::Color mDmfp; // 'ld' diffuse mean free path length, aka 'translucency radius'
71 scene_rdl2::math::Color mD; // 'd' shaping parameter
72
73 // Used for local importance sampling
74 scene_rdl2::math::Color mChannelCdf;
75 scene_rdl2::math::Color mMaxR; // per-channel distance at which R(r) < 0.001
76};
