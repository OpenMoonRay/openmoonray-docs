35// Utility interpolation function for attribute baking
36template <typename T>
37T
38bilinearInterpolate(const Vec2f& uv,
39 const T& a0, const T& a1, const T& a2, const T& a3)
40{
41 return (1.0f - uv[0]) * (1.0f - uv[1]) * a0 +
42 ( uv[0]) * (1.0f - uv[1]) * a1 +
43 ( uv[0]) * ( uv[1]) * a2 +
44 (1.0f - uv[0]) * ( uv[1]) * a3;
45}

47template <typename T>
48void*
49Curves::getBakedAttributeData(const TypedAttributeKey<T>& key,
50 size_t& numElements, AttributeRate &newRate) const
51{
52 size_t curvesCount = mCurvesVertexCount.size();
53 size_t vertexCount = 0;
54 for (size_t i = 0; i < curvesCount; ++i) {
55 vertexCount += mCurvesVertexCount[i];
56 }
57
58 Attributes *attributes = getAttributes();
59 size_t timeSamples = attributes->getTimeSampleCount(key);
60
61 void* data;
62 newRate = attributes->getRate(key);
63
64 switch (newRate) {
65 case RATE_CONSTANT:
66 {
67 numElements = timeSamples;
68 T* tdata = new T[numElements];
69 data = tdata;
70 for (size_t t = 0; t < timeSamples; t++) {
71 tdata[t] = attributes->getConstant(key, t);
72 }
73 break;
74 }
75 case RATE_UNIFORM:
76 {
77 numElements = curvesCount * timeSamples;
78 T* tdata = new T[numElements];
79 data = tdata;
80 for (size_t c = 0, dstIdx = 0; c < curvesCount; c++) {
81 for (size_t t = 0; t < timeSamples; t++) {
82 tdata[dstIdx++] = attributes->getUniform(key, c, t);
83 }
84 }
85 break;
86 }
87 case RATE_VARYING:
88 case RATE_VERTEX:
89 {
90 if (key == StandardAttributes::sNormal) {
91 // reformat to FACE_VARYING if the normals are vertex rate
92 numElements = vertexCount * timeSamples;
93 T* tdata = new T[numElements];
94 data = tdata;
95 for (size_t v = 0, dstIdx = 0; v < vertexCount; v++) {
96 for (size_t t = 0; t < timeSamples; t++) {
97 tdata[dstIdx++] = attributes->getVertex(key, v, t);
98 }
99 }
100 } else {
101 numElements = getVertexCount() * timeSamples;
102 T* tdata = new T[numElements];
103 data = tdata;
104 for (size_t v = 0, dstIdx = 0; v < vertexCount; v++) {
105 for (size_t t = 0; t < timeSamples; t++) {
106 tdata[dstIdx++] = attributes->getVertex(key, v, t);
107 }
108 }
109 }
110 break;
111 }
112 default:
113 MNRY_ASSERT(false, "unknown attribute rate");
114 break;
115 }
116
117 return data;
118}

182std::unique_ptr<BakedAttribute>
183Curves::getBakedAttribute(const AttributeKey& key) const
184{
185 Attributes *attributes = getAttributes();
186 size_t timeSamples = attributes->getTimeSampleCount(key);
187
188 std::unique_ptr<BakedAttribute> battr = fauxstd::make_unique<BakedAttribute>();
189
190 battr->mName = key.getName();
191 battr->mTimeSampleCount = timeSamples;
192 battr->mType = key.getType();
193 battr->mData = nullptr;
194
195 switch (battr->mType) {
196 case AttributeType::TYPE_BOOL:
197 battr->mData = getBakedAttributeData(TypedAttributeKey<bool>(key),
198 battr->mNumElements, battr->mRate);
199 break;
200 case AttributeType::TYPE_INT:
201 battr->mData = getBakedAttributeData(TypedAttributeKey<int>(key),
202 battr->mNumElements, battr->mRate);
203 break;
204 case AttributeType::TYPE_LONG:
205 battr->mData = getBakedAttributeData(TypedAttributeKey<long>(key),
206 battr->mNumElements, battr->mRate);
207 break;
208 case AttributeType::TYPE_FLOAT:
209 battr->mData = getBakedAttributeData(TypedAttributeKey<float>(key),
210 battr->mNumElements, battr->mRate);
211 break;
212 case AttributeType::TYPE_DOUBLE:
213 battr->mData = getBakedAttributeData(TypedAttributeKey<double>(key),
214 battr->mNumElements, battr->mRate);
215 break;
216 case AttributeType::TYPE_STRING:
217 battr->mData = getBakedAttributeData(TypedAttributeKey<std::string>(key),
218 battr->mNumElements, battr->mRate);
219 break;
220 case AttributeType::TYPE_RGB:
221 battr->mData = getBakedAttributeData(TypedAttributeKey<scene_rdl2::math::Color>(key),
222 battr->mNumElements, battr->mRate);
223 break;
224 case AttributeType::TYPE_RGBA:
225 battr->mData = getBakedAttributeData(TypedAttributeKey<scene_rdl2::math::Color4>(key),
226 battr->mNumElements, battr->mRate);
227 break;
228 case AttributeType::TYPE_VEC2F:
229 battr->mData = getBakedAttributeData(TypedAttributeKey<Vec2f>(key),
230 battr->mNumElements, battr->mRate);
231 break;
232 case AttributeType::TYPE_VEC3F:
233 battr->mData = getBakedAttributeData(TypedAttributeKey<Vec3f>(key),
234 battr->mNumElements, battr->mRate);
235 break;
236 case AttributeType::TYPE_VEC4F:
237 battr->mData = getBakedAttributeData(TypedAttributeKey<Vec4f>(key),
238 battr->mNumElements, battr->mRate);
239 break;
240 case AttributeType::TYPE_MAT4F:
241 battr->mData = getBakedAttributeData(TypedAttributeKey<scene_rdl2::math::Mat4f>(key),
242 battr->mNumElements, battr->mRate);
243 break;
244 default:
245 MNRY_ASSERT(false, (std::string("unsupported attribute type ") +
246 std::string(attributeTypeName(key.getType())) +
247 std::string(" for attribute ") + std::string(key.getName())).c_str());
248 break;
249 }
250
251 return battr;
252}

254void
255Curves::getBakedCurves(BakedCurves& bakedCurves) const
256{
257 bakedCurves.mName = mName;
258 bakedCurves.mType = static_cast<BakedCurves::Type>(mType);
259 bakedCurves.mMotionSampleCount = getMotionSamplesCount();
260 bakedCurves.mCurvesVertexCount.resize(getCurvesVertexCount().size());
261 bakedCurves.mVertexCount = 0;
262 for (size_t i = 0; i < getCurvesVertexCount().size(); ++i) {
263 bakedCurves.mCurvesVertexCount[i] = getCurvesVertexCount()[i];
264 bakedCurves.mVertexCount += bakedCurves.mCurvesVertexCount[i];
265 }
266 bakedCurves.mVertexBuffer.resize(bakedCurves.mVertexCount * bakedCurves.mMotionSampleCount);
267 bakedCurves.mRadii.resize(bakedCurves.mVertexCount * bakedCurves.mMotionSampleCount);
268 for (size_t v = 0; v < bakedCurves.mVertexCount; v++) {
269 for (size_t t = 0; t < bakedCurves.mMotionSampleCount; t++) {
270 const scene_rdl2::math::Vec3fa& vtx = mVertexBuffer(v, t);
271 scene_rdl2::math::Vec3f p(vtx.x, vtx.y, vtx.z);
272 bakedCurves.mVertexBuffer[v * bakedCurves.mMotionSampleCount + t] = p;
273 bakedCurves.mRadii[v * bakedCurves.mMotionSampleCount + t] = vtx.w;
274 }
275 }
276
277 if (mLayerAssignmentId.getType() == LayerAssignmentId::Type::CONSTANT) {
278 bakedCurves.mLayerAssignmentIds = std::vector<int>(bakedCurves.mCurvesVertexCount.size(),
279 mLayerAssignmentId.getConstId());
280 } else {
281 bakedCurves.mLayerAssignmentIds = mLayerAssignmentId.getVaryingId();
282 }
283
284 Attributes *attributes = getAttributes();
285 const PrimitiveAttributeTable* patable = getPrimitiveAttributeTable();
286 for (const auto& entry : *patable) {
287 const AttributeKey& key = entry.first;
288 if (attributes->hasAttribute(key)) {
289 bakedCurves.mAttrs.push_back(getBakedAttribute(key));
290 }
291 }
292}
