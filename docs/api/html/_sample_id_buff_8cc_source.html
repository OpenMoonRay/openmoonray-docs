29bool
30SampleIdBuff::verify(const FloatBuffer &weightBuff) const
31//
32// Run verification as compared with final weight buffer value
33//
34{
35 unsigned totalError = 0;
36 crawlAllPix([&](unsigned sx, unsigned sy) -> bool {
37 if (getSampleId(sx, sy) != getSampleId(weightBuff, sx, sy)) {
38 if (totalError < 100) { // We only dump the first 100 errors
39 std::cerr << "sx:" << sx << " sy:" << sy
40 << " " << getSampleId(sx, sy)
41 << " != " << getSampleId(weightBuff, sx, sy) << '\n';
42 }
43 totalError++;
44 }
45 return true;
46 });
47 if (totalError > 0) {
48 std::cerr << ">> SampleIdBuffer.cc verify failed totalError:" << totalError << " pixels\n";
49 return false;
50 }
51 return true;
52}

69void
70SampleIdBuff::initSub(unsigned width, unsigned height, unsigned maxSampleId)
71{
72 mWidth = width;
73 mHeight = height;
74 mTiler.~Tiler();
75 new(&mTiler) Tiler(mWidth, mHeight);
76 mPixDepth = (maxSampleId < 256) ? 1 : (maxSampleId < 65536) ? 2 : 4; // byte
77
78 unsigned alignedWidth = (mWidth + 7) & ~7;
79 unsigned alignedHeight = (mHeight + 7) & ~7;
80 unsigned memSizeByte = alignedWidth * alignedHeight * mPixDepth;
81 mArray.reset(new char[memSizeByte]);
82
83 switch (mPixDepth) {
84 case 1 :
85 mGetFunc = [&](int sx, int sy) -> unsigned { return getPix<unsigned char>(sx, sy); };
86 mSetFunc = [&](int sx, int sy, unsigned v) { setPix<unsigned char>(sx, sy, v); };
87 mAtomicReserveFunc = [&](int sx, int sy, unsigned sampleDelta) -> unsigned {
88 return atomicReservePix<unsigned char>(sx, sy, sampleDelta);
89 };
90 break;
91 case 2 :
92 mGetFunc = [&](int sx, int sy) -> unsigned { return getPix<unsigned short>(sx, sy); };
93 mSetFunc = [&](int sx, int sy, unsigned v) { setPix<unsigned short>(sx, sy, v); };
94 mAtomicReserveFunc = [&](int sx, int sy, unsigned sampleDelta) -> unsigned {
95 return atomicReservePix<unsigned short>(sx, sy, sampleDelta);
96 };
97 break;
98 default :
99 mGetFunc = [&](int sx, int sy) -> unsigned { return getPix<unsigned int>(sx, sy); };
100 mSetFunc = [&](int sx, int sy, unsigned v) { setPix<unsigned int>(sx, sy, v); };
101 mAtomicReserveFunc = [&](int sx, int sy, unsigned sampleDelta) -> unsigned {
102 return atomicReservePix<unsigned int>(sx, sy, sampleDelta);
103 };
104 break;
105 }
106}

108void
109SampleIdBuff::setupSrcWeight(unsigned maxSampleId, const GetSrcWeightFunc &getSrcWeightFunc)
110{
111#ifdef RUNTIME_VERIFY
112 unsigned long long srcTotal = 0;
113 unsigned long long dstTotal = 0;
114#endif // end RUNTIME_VERIFY
115
116 crawlAllPix([&](unsigned sx, unsigned sy) -> bool {
117 setSampleId(sx, sy, std::min(getSrcWeightFunc(sx, sy), maxSampleId));
118
119# ifdef RUNTIME_VERIFY
120 srcTotal += std::min(getSrcWeightFunc(sx, sy), maxSampleId);
121 dstTotal += getSampleId(sx, sy);
122# endif // end RUNTIME_VERIFY
123
124 return true;
125 });
126
127# ifdef RUNTIME_VERIFY
128 std::cerr << ">> SampleIdBuff.cc runtime verify for SampleIdBuff::setupSrcWeight() "
129 << ((srcTotal == dstTotal) ? "OK" : "NG") << '\n';
130# endif // end RUNTIME_VERIFY
131}
