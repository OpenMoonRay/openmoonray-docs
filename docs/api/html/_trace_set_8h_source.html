24/// @class ContainerWrapper
25/// @brief This class is similar to std::reference_wrapper. It exists because
26/// concrete references to the RDL containers in the Layer aren't stored, but
27/// are accessed through get() calls. The iterators provided on the Layer need
28/// to be able to refer to the containers.
29template <typename Container>
30struct ContainerWrapper
31{
32 using container = Container;
33 using value_type = typename container::value_type;
34 using const_reference = typename container::const_reference;
35 using size_type = typename container::size_type;
36
37 explicit ContainerWrapper(const Container& c) : mC(std::addressof(c))
38 {
39 }
40
41 ContainerWrapper(const ContainerWrapper&) = default;
42 ContainerWrapper& operator=(const ContainerWrapper&) = default;
43 ContainerWrapper(ContainerWrapper&&) = default;
44 ContainerWrapper& operator=(ContainerWrapper&&) = default;
45
46 ~ContainerWrapper()
47 {
48 mC = nullptr; // For debugging...
49 }
50
51 const_reference operator[](size_type i) const
52 {
53 return (*mC)[i];
54 }
55
56 const Container* mC;
57};
58
59template <typename Container>
60ContainerWrapper<Container> makeContainerWrapper(const Container& c)
61{
62 return ContainerWrapper<Container>(c);
63}

66/**
67 * The TraceSet is a set of objects that can be ray traced. It stores a list of
68 * unique Geometry / Part pairs.
69 * Each assignment is made up of the following tuple:
70 * (Geometry*, String)
71 *
72 * When the assign() method is called, it returns a 32-bit unsigned integer.
73 * This is the assignment ID. It is unique for a particular Geometry/Part pair.
74 * It can be used to quickly and efficiently look up which object has been
75 * intersected.
76 *
77 * You can also get the assignment ID from the Geometry / Part pair, but this
78 * is a slow operation.
79 *
80 * Calling the assign() method again with an existing Geometry/Part pair will
81 * return the same assignment ID that was there before.
82 */
83
84class TraceSet : public SceneObject

86public:
87 typedef SceneObject Parent;

92 typename SceneObjectIndexable::index_iterator>
93 GeometryIterator;

96 static SceneObjectInterface declare(SceneClass& sceneClass);
97
98 /**
99 * Returns the number of assignments made in this trace set so far.
100 */
101 int32_t getAssignmentCount() const;
102
103 /**
104 * Adds a new assignment in the trace set. The Geometry and part name
105 * form a unique entry.
106 *
107 * @param geometry The Geometry on which the part lives.
108 * @param partName The name of the part.
109 * @return The assignment ID that can be used for fast lookups.
110 */
111 int32_t assign(Geometry* geometry, const String& partName);
112
113 /**
114 * Given a valid assignment ID, this will return a std::pair containing the
115 * Geometry and Part assignments which are set in the TraceSet. If the
116 * assignmentId is invalid, except::IndexError is thrown.
117 *
118 * @param assignmentId The assignment ID to look up assignments for.
119 * @return A std::pair of the Geometry* and std::string assignments.
120 * @throw except::IndexError If the assignmentId is invalid.
121 */
122 GeometryPartPair lookupGeomAndPart(int32_t assignmentId) const;
123
124 /**
125 * Given a Geometry and part name on that Geometry, this will return the
126 * assignment ID for that assignment, which can be used for fast assignment
127 * lookups. For efficiency, you should save this value to use for multiple
128 * lookups. If no assignment is found, -1 is returned.
129 *
130 * @param geometry The Geometry on which the part lives.
131 * @param partName The name of the part with the assignment.
132 * @return The assignment ID that can be used for fast lookups.
133 */
134 int32_t getAssignmentId(const Geometry* geometry, const String& partName) const;
135
136 /**
137 * Given a Geometry, this will return whether or not the trace set contains
138 * said geometry.
139 *
140 * @param geometry The Geometry to check to see if it exists in the layer.
141 * @return Whether the geometry exists in they layer or not.
142 */
143 bool contains(const Geometry* geometry) const;
144
145 /// The iterators returned by this function are a little different from
146 /// standard iterators: when dereferenced, they don't return an object, they
147 /// return an index. This index can then be used in the TraceSet to lookup
148 /// whatever information is needed.
149 ///
150 /// Calls to begin() and end() must reference the same object. The object
151 /// passed into these calls is then used to iterate over entries in the
152 /// trace set that match the passed in object.
153 GeometryIterator begin(const Geometry* geometry) const;
154 GeometryIterator end(const Geometry* geometry) const;
155
156protected:
157 friend AsciiWriter;
