101/**
102 * Copies a larger buffer into a smaller buffer as defined by two viewports
103 * @param roiViewport Viewport for the smaller Region-of-Interest buffer
104 * @param vp Viewport of the full buffer
105 * @param numChannels Number of channels per pixel
106 * @param targetPtr Destination ROI raw buffer of type PixelComponentType
107 * @param srcPtr Source full buffer of type PixelComponentType
108 * @param bufferSize Output variable that is set to the number of bytes in the destination buffer
109 */
110template<typename PixelComponentType>
111finline uint8_t *
112copyRoiBuffer(const math::Viewport& roiViewPort, const math::Viewport& vp,
113 const int numChannels, PixelComponentType* targetPtr,
114 const PixelComponentType* srcPtr, size_t& bufferSize)
115{
116 // Viewport based
117 const int yRange = roiViewPort.max().y - roiViewPort.min().y + 1;
118 const int xRange = roiViewPort.max().x - roiViewPort.min().x + 1;
119 #ifdef DEBUG
120 const int pixels = xRange * yRange;
121 #endif
122 const int yStart = roiViewPort.min().y;
123 const int xStart = roiViewPort.min().x;
124 // Window based
125 const int windowWidth = vp.width();
126 const size_t endPoint = xRange * numChannels;
127
128 // Perform the copy from the src to the target buffer
129 size_t targetIndex = 0;
130 for (int y = 0; y < yRange; ++y) {
131 const int startPoint = (y + yStart) * windowWidth * numChannels + (xStart * numChannels);
132
133 for(size_t src = 0; src < endPoint; ++src) {
134 targetPtr[targetIndex++] = srcPtr[startPoint + src];
135 }
136 }
137
138 bufferSize = xRange * yRange * numChannels * sizeof(PixelComponentType);
139
140 MNRY_ASSERT(targetIndex == pixels * numChannels);
141
142 return reinterpret_cast<uint8_t *>(targetPtr);
143}
