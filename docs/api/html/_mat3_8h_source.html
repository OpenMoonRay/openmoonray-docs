43 /** \brief 3x3 Column Major Matrix
44 *
45 * Matrix formats can be confusing. When we say <b>column major</b> it's only meaningful when we
46 * are discussing it in certain context, such as transformations. <b>We are not referring to the underlying
47 * storage of the matrix data</b>, which could be in any form (arrays, vectors, etc.). Regardless of row major
48 * or column major, multiplications between matrices do not change (multiply a row by a column).
49 * However, when we define transformations of point, vectors, etc., we need to be clear about the meanings
50 * of the rows and columns of a matrix and their corresponding multiplication with vectors.
51 *
52 * DWA specifies the transformation matrix in column major, so vector transformation is applied using
53 * post-multiply. Normal transformation is done using pre-multiply. These are defined as follows.
54 *
55 * <b>Row vector</b> means vectors are defined as v = [x, y, z], also can be considered as a 1 by n matrix, where n = 3.
56 *
57 * Column vector is the row vector transposed, which is a n by 1 matrix.
58 *
59 * Matrix multiply for a row vector can be done by post-multiply, where the matrix is after the row vector:
60 *
61 * u = v * M (equivalent to transpose(M) * transpose(v))
62 *
63 * Or it's possible to do pre-multiply, which implicitly transposes the vector v into a column vector:
64 *
65 * w = M * v (where v is assumed to be a column vector)
66 *
67 * In general, u != w (note the missing transpose for M), because matrix multiplication is not commutative.
68 *
69 * The set of <b>transform*()</b> interface should be preferable over multiplication since their intentions are unambiguous.
70 */

269 /// Slerp between two rotation matrices
270 /// WARNING: This only works if the matrix only contains rotation
271 /// and produces undefined results otherwise. In the general case you should
272 /// use slerp()
273 template<typename VectorT, typename U> __forceinline Mat3<VectorT> slerpR(const Mat3<VectorT>& a, const Mat3<VectorT>& b, const U t) {
274 typename Mat3<VectorT>::Quat qa = a.quat();
275 typename Mat3<VectorT>::Quat qb = b.quat();
276 if (dot(qa, qb) < 0) { qb *= U(-1); }
277 return Mat3<VectorT>(slerp(qa, qb, t));
278 }

280 /// Slerp between two 3x3 rotation/scale matrices
281 /// This produces undefined results on matrix contains reflection operation
282 template<typename VectorT, typename U> __forceinline Mat3<VectorT> slerp(const Mat3<VectorT>& a, const Mat3<VectorT>& b, const U t) {
283 typename Mat3<VectorT>::Quat qa;
284 Mat3<VectorT> sa;
285 decompose(a, sa, qa);
286 typename Mat3<VectorT>::Quat qb;
287 Mat3<VectorT> sb;
288 decompose(b, sb, qb);
289
290 return lerp(sa, sb, t) * (Mat3<VectorT>(slerp(qa, qb, t)));
291 }

302 /// Perform a polar decomposition into a scale matrix and a rotation matrix,
303 /// M = S * R
304 /// where R is returned in the form of a quaternion q.
305 /// Note that in general S will not be a diagonal matrix, since the transformation it represents is not tied to
306 /// any particular coordinate frame, i.e. the 3 mutually orthogonal scaling directions can be arbitrary.
307 /// (In fact the directions will be the eigenvectors of S.)
308 /// For further details see:
309 /// [1] "Polar Matrix Decomposition" - Ken Shoemake. Graphics Gems IV
310 /// [2] "Matrix Animation and Polar Decomposition" - Ken Shoemake, Tom Duff
311 /// Note that the order of matrix multiplication differs from these papers, because we express points as row vectors.
312 template<typename VectorT>
313 __forceinline DecomposeErrorCode decompose(const Mat3<VectorT>& M, Mat3<VectorT>& S, typename Mat3<VectorT>::Quat& q)
314 {
315 const float detM = M.det();
316
317 // Non-positive determinants not supported
318 if (detM <= 0.0f) {
319 q = one;
320 S = M;
321 return (detM == 0.0f) ? DecomposeErrorCode::SINGULAR : DecomposeErrorCode::FLIPPED;
322 }
323
324 typedef Mat3<VectorT> Mat3T;
325 constexpr int maxCount = 100;
326 Mat3T R = M;
327
328 int count = 0;
329 for (count = 0; count < maxCount; count++) {
330 const Mat3T Rnext = 0.5f * (R + R.transposed().inverse());
331 const Mat3T Rdiff = Rnext - R;
332
333 typename VectorT::Scalar norm = 0.0f;
334 for (size_t i = 0; i < 3; ++i) {
335 norm = max(norm, abs(Rdiff[i].x)+abs(Rdiff[i].y)+abs(Rdiff[i].z));
336 }
337
338 // Note that if norm is below the threshold, we break here before updating R. This allows an input matrix that's
339 // a pure rotation to pass straight through without modification.
340 if (norm <= sEpsilon) break;
341
342 R = Rnext;
343 }
344
345 q = R.quat();
346 S = M * R.inverse();
347
348 MNRY_ASSERT(count < maxCount, "Reached max iterations in matrix decompose function; result may be unexpected.");
349 if (count >= maxCount) {
350 return DecomposeErrorCode::MAX_ITER;
351 }
352
353 return DecomposeErrorCode::SUCCESS;
354 }
