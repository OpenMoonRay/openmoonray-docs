241 // -- High speed float value to gamma 2.2 8bit conversion by lookup table --

243 // Regarding to the computation of gamma correction and quantized to 8bit range by lookup table,
244 // I directory generate lookup table index from input float value's bit pattern.
245 //
246 // First of all, we consider following 4 values (f1 ~ f4).
247 //
248 // -- IEEE singleFloat bit pattern --
249 // s| exp | mantissa expVal | gamma255
250 // f1 = 0.000005077049536 : 0 01101101 01010100101101110000110 => 109 0
251 // f2 = 0.000005077049991 : 0 01101101 01010100101101110000111 => 109 1
252 // f3 = 0.999999940395355 : 0 01111110 11111111111111111111111 => 126 254
253 // f4 = 1.000000000000000 : 0 01111111 00000000000000000000000 => 127 255
254 // ^|<---+-->|<----------+---------->|
255 // | | |
256 // | | +-- mantissaBit
257 // | +- exponentBit
258 // +-- signeBit
259 //
260 // f1 is maxinum single float value which converted to 0 after gamma/8bit conversion.
261 // f2 is minimum single float value which converted to 1.
262 // f3 is maxinum single float value which converted to 254.
263 // f4 is minimum single float value which converted to 255.
264 //
265 // This means all the float value equal or smaller than f1 should be converted to 0. And
266 // all the float value equal or bigger than f4 should be converted to 255. We only need to consider between f2 to f3.
267 // (There is no float value between f1 and f2 also f3 and f4).
268 //
269 // Exponent value range for f2 ~ f3 is 109(=0x6d,=01101101) ~ 126(=0x7e,=01111110).
270 // In order to keep all range from 109 to 126, we need at least 5bit. And 5bit can store value range from 109 to 140.
271 // This can save entire range from f1 to f4 (We only use 109 to 126. not using 127 to 140 actually).
272 //
273 // For mantissa, basically we don't need all 23bit of precision. After several tests, I decided to use
274 // top left 7bit for mantissa and final tableId bit size is 12bit.
275 //
276 // expPart manPart
277 // tableId (12bit) = 1 0 9 8 7 6 5 4 3 2 1 0
278 // |<---+--->|<-----+----->|
279 // | |
280 // | +-- from mantissaBit : 7bit 0~127 : top left 7bit of single float mantissa
281 // |
282 // +-- from exponentBit : 5bit 0~31 : exponent range from 109(=0x60) to 140(=0x8c)
283 // But only use from 109 to 126
284 //
285 // Following is a table index and it's input float value and output gamma255 table value (0~255)
286 //
287 // tableId | expPart | manPart | expMask | manMask | inputF | g255
288 // 0 0 0 109 0x000000 0.0000038147 0
289 // 1 0 1 109 0x010000 0.0000038445 0

291 // 3 0 3 109 0x030000 0.0000039041 0

293 // 5 0 5 109 0x050000 0.0000039637 0
294 // 6 0 6 109 0x060000 0.0000039935 0

300 // 2302 17 126 126 0x7e0000 0.9921875000 254
301 // 2303 17 127 126 0x7f0000 0.9960937500 254
302 // 2304 18 0 127 0x000000 1.0000000000 255
303 //
304 // expPart is a value of exponent part of tableId. expPart = (tableId >> 7) & 0x1f;
305 // manPart is a value of mantissa part of tableId. manPart = tableId & 0x7f;
306 // expMask is input value's exponent bitMask based on expPart. expMask = expPart + 109
307 // manMask is input value's mantissa bitMask based on manPart. manMask = manPart << 16
308 // inputF is reconstruct float value based on tableId. inputF = (expMask << 23) | manMask
309 // g255 is gamma corrected 8bit value of inputF g255 = pow(inputF, 1.0/2.2) * 255
310 //
311 // tableId is 12bit and can support 4096 table size. However, we only use 0 ~ 2304.
312 // (max exponent is 126 and exp(126) w/ man(0x7f0000 = full mantissa bitMask) goes to tblId 2303.
313 // But we need index 2304 as to get max value (i.e. 255). This is why max tbl id is 2304 instead of 2303)
314 //
315 // Following logic is a IEEE single float to tablId direct conversion logic
316 //
317 // int convFloatToTblId(const float f)
318 // {
319 // union Uni {
320 // float f;
321 // unsigned u;
322 // };
323 // const Uni *uni = (const Uni *)&f;
324 // unsigned expMask = (uni->u >> 23) & 0xff; // ... (a)
325 // unsigned manMaskShifted = (uni->u >> 16) & 0x7f; // ... (b)
326 //
327 // unsigned tblId = 0;
328 // if (expMask < 109) {
329 // tblId = 0; // ... (c)
330 // } else if (expMask < 127) {
331 // tblId = (((expMask - 109) & 0x1f) << 7) | manMaskShifted; // ... (d)
332 // } else {
333 // tblId = 2304; // ... (e)
334 // }
335 // return tblId;
336 // }

339 // (b) : manMaskShifted is shifted bitmask pattern of mantissa (only consider top left 7bit of mantissa bit)
340 // (c) : if exponent value is less than 109, pick tableId as 0 (i.e. final out value is 0)

342 // (e) : if exponent value is equal or bigger than 127, tableId is 2304.
343 //
344 // Let's remove branch flow by bitMask operation
345 //
346 /* naive version
347 unsigned tblId = 0;
348 if (expMask < 109) {
349 tblId = 0;

360 int expMaxMask = (static_cast<int>(expMask) - 127) >> 31; // ... (B)

372 // (D) : mask for exponent min (109) : expMinMask = (expMask < 109)? 0x0: 0xffffffff
373 // (E) : tblId3 (all exponent) is now clipped to 0 ~ 2304. This is result tblId
