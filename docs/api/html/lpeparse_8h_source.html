49/// Light path expression parser
50///
51/// As most hand written descendant parsers, it is a bit messy in the implementation.
52/// But the interface more or less resembles the grammar of the language:
53///
54/// symbol := <char> | 'string'
55/// listsym := <lambda> | listsym sym
56/// listexp := <lambda> | listexp exp
57/// group := '<' exp exp exp listexp '>'
58/// catexp := ( listexp )
59/// sorexp := exp '|' exp
60/// borexp := [ listsym ]
61/// wildexp := . | [^ listsym ]
62/// repeat := exp *
63/// repeat+ := exp +
64/// mnrepeat := exp {<int>} | exp {<int>,} | exp {<int>,<int>}
65/// exp := wildexp | borexp | sorexp | catexp | group | repeat | repeat+ | mnrepeat
66/// topexp := exp | topexp exp
67///
68class Parser
69{
70 public:
71
72 Parser(const std::vector<ustring> *user_events = NULL,
73 const std::vector<ustring> *user_scatterings = NULL);
74
75 /// Parse a string and return the resulting light path expression tree or NULL if failed
76 LPexp *parse(const char *text);
77
78 /// Check for error in the last parsed string
79 bool error()const { return m_error.size() > 0; };
80 /// Get the error string
81 const char *getErrorMsg()const { return m_error.c_str(); };
82 /// Get the position of the string where the error appeared
83 int getErrorPos()const { return m_pos; };
84
85 private:
86
87 /// Current char being parsed
88 char head()const { return m_text[m_pos]; };
89 /// Any input (including head) left?
90 bool hasInput()const { return m_text.size() > m_pos; };
91 /// Go to the next char in the string
92 void next() { m_pos++; };
93
94 /// build the complete pattern for a pathtracing stop
95 ///
96 /// That means the three basic label match expressions, plus those for the custom labels (if any),
97 /// plus an additional wildcard to eat extra custom labels at the end, and finally the stop mark.
98 /// So you provide the regexps for the basic three labels:
99 ///
100 /// \param etype Event type (CFVL)
101 /// \param dir Direction (RD)
102 /// \param scatter Scattering type (DGSs)
103 ///
104 /// That of course can be wildcards or any other expression willing to match them. And then you can
105 /// provide extra expressions to match the custom labels:
106 ///
107 /// \param custom Custom label regexps
108 ///
109 /// And the function will automatically add a "[^stop_mark]*stop_mark" at the end.
110 ///
111 LPexp *buildStop(LPexp *etype, LPexp *scatter, const std::list<LPexp*> &custom);
112 /// Gicen that a symbol is ready in head() to parse, parse it
113 LPexp *parseSymbol();
114 /// Gicen that a symbol is ready in head() to parse, parse it as a ustring
115 /// and report it was a custom symbol in the iscustom flag
116 ustring parseRawSymbol(bool &iscustom);
117 /// Given that the begining of a concatenation of regexps is ready to parse, parse it
118 /// and it can be optionally be enclosed in parentheis ()
119 LPexp *parseCat();
120 /// Given that a fully qualified group like <.RD'custom'> is ready to parse, parse it
121 LPexp *parseGroup();
122 /// Given that a ^abcde] (note missing [) is ready to parse, parse it
123 LPexp *parseNegor();
124 /// Given that [abcde] is ready to parse, parse it, but if it finds that it was actually
125 /// [^abcd], fall back to parseNegor
126 LPexp *parseOrlist();
127 /// Given that a range like {5,7} or {2,} or {3} is ready to parse, parse it and return
128 /// the range. Second number being -1 if if was {2,} and equals the first if it was {3}
129 std::pair<int, int> parseRange();
130 /// Take an already parsed lpexp and parse its possible modifier (*={}) if present and
131 /// return the new lpexp
132 LPexp *parseModifier(LPexp *e);
133 /// Generic parse whatever comes next (just one item)
134 LPexp *_parse();
135
136 // error string
137 std::string m_error;
138 // True if we are actually parsing a group <>, since otherwise everything gets
139 // automatically converted to a group, this prevents that happening
140 bool m_ingroup;
141 // maps each basic label to its expected possition in the appearance order, for instance
142 // the direction label can't be in the fisrt pos of a group. This way we know where to put the
143 // expression when the user writes just S, which translates to <..S>
144 SymbolToInt m_label_position;
145 // The set of the basic labels
146 SymbolSet m_basic_labels;
147 // The black list for our wildcards, that have to exclude always the stop mark
148 SymbolSet m_minus_stop;
149
150 // Current text being parsed
151 std::string m_text;
152 // Current position in the text for head()
153 size_t m_pos;
154};
