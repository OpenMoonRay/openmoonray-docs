72void
73RaySwitchMaterial::shade(const scene_rdl2::rdl2::Material* self,
74 moonray::shading::TLState *tls,
75 const State& state,
76 BsdfBuilder& bsdfBuilder)
77{
78 // If this material is cutting out, terminate the ray
79 if (!state.isIndirect() && self->get(attrCutoutCameraRays)) {
80 bsdfBuilder.setEarlyTermination();
81 return;
82 }
83
84 // We have already performed substitutions in the integrator so we don't need to switch here.
85 // If there was no substitution, use the default material.
86
87 const SceneObject* defaultMaterialObj = self->get(attrDefaultMaterial);
88 if (defaultMaterialObj) {
89 moonray::shading::shade(self, defaultMaterialObj->asA<scene_rdl2::rdl2::Material>(), tls, state, bsdfBuilder);
90 }
91}

119const scene_rdl2::rdl2::Material*
120RaySwitchMaterial::raySwitch(const scene_rdl2::rdl2::RaySwitchContext& ctx) const
121{
122 switch (ctx.mRayType) {
123 case scene_rdl2::rdl2::RaySwitchContext::RayType::CameraRay:
124 {
125 if(get(attrCutoutCameraRays)) {
126 // If cutting out camera rays, return this material.
127 // Calling shade on this material will trigger early
128 // termination of the rays.
129 return this;
130 }
131 const SceneObject* materialObj = get(attrCameraRayMaterial);
132 if (materialObj) {
133 return materialObj->asA<scene_rdl2::rdl2::Material>();
134 }
135 }
136 break;
137 case scene_rdl2::rdl2::RaySwitchContext::RayType::IndirectMirrorRay:
138 {
139 const SceneObject* materialObj = get(attrIndirectMirrorRayMaterial);
140 if (materialObj) {
141 return materialObj->asA<scene_rdl2::rdl2::Material>();
142 }
143 }
144 break;
145 case scene_rdl2::rdl2::RaySwitchContext::RayType::IndirectGlossyRay:
146 {
147 const SceneObject* materialObj = get(attrIndirectGlossyRayMaterial);
148 if (materialObj) {
149 return materialObj->asA<scene_rdl2::rdl2::Material>();
150 }
151 }
152 break;
153 case scene_rdl2::rdl2::RaySwitchContext::RayType::IndirectDiffuseRay:
154 {
155 const SceneObject* materialObj = get(attrIndirectDiffuseRayMaterial);
156 if (materialObj) {
157 return materialObj->asA<scene_rdl2::rdl2::Material>();
158 }
159 }
160 break;
161 }
162
163 return this;
164}
