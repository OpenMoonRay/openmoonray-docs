19void
20TestStateMachine::testLpe()
21{
22 StateMachine m;
23 CPPUNIT_ASSERT(m.addExpression("CD*L", 1) == 0);
24 CPPUNIT_ASSERT(m.addExpression("CSL", 2) == 0);
25 CPPUNIT_ASSERT(m.addExpression("NOT AN LPE", 3) != 0);
26 CPPUNIT_ASSERT(m.addExpression("C<.D'diffuse'>L", 4) == 0);
27 CPPUNIT_ASSERT(m.addExpression("C[<.D'diffuse'><.D'base'>]L", 5) == 0);
28
29 int diffuseLabel = m.getLabelId("diffuse");
30 int baseLabel = m.getLabelId("base");
31 CPPUNIT_ASSERT(diffuseLabel >= 0);
32 CPPUNIT_ASSERT(baseLabel >= 0);
33
34 m.build();
35
36 // test some paths
37 int stateId = 0; // initial state
38 {
39 // C_ RD L_
40 stateId = m.transition(stateId, EVENT_TYPE_CAMERA, EVENT_SCATTERING_TYPE_NONE, sNoLabel);
41 stateId = m.transition(stateId, EVENT_TYPE_REFLECTION, EVENT_SCATTERING_TYPE_DIFFUSE, sNoLabel);
42 stateId = m.transition(stateId, EVENT_TYPE_LIGHT, EVENT_SCATTERING_TYPE_NONE, sNoLabel);
43 CPPUNIT_ASSERT(m.isValid(stateId, 1));
44 CPPUNIT_ASSERT(!m.isValid(stateId, 2));
45 CPPUNIT_ASSERT(!m.isValid(stateId, 3));
46 CPPUNIT_ASSERT(!m.isValid(stateId, 4));
47 CPPUNIT_ASSERT(!m.isValid(stateId, 5));
48 }
49
50 stateId = 0; // initial state
51 {
52 // C_ Ts L_
53 stateId = m.transition(stateId, EVENT_TYPE_CAMERA, EVENT_SCATTERING_TYPE_NONE, sNoLabel);
54 stateId = m.transition(stateId, EVENT_TYPE_TRANSMISSION, EVENT_SCATTERING_TYPE_STRAIGHT, sNoLabel);
55 stateId = m.transition(stateId, EVENT_TYPE_LIGHT, EVENT_SCATTERING_TYPE_NONE, sNoLabel);
56 CPPUNIT_ASSERT(!m.isValid(stateId, 1));
57 CPPUNIT_ASSERT(!m.isValid(stateId, 2));
58 CPPUNIT_ASSERT(!m.isValid(stateId, 3));
59 CPPUNIT_ASSERT(!m.isValid(stateId, 4));
60 CPPUNIT_ASSERT(!m.isValid(stateId, 5));
61 }
62
63 stateId = 0; // initial state
64 {
65 // C_ RD'diffuse' L_
66 stateId = m.transition(stateId, EVENT_TYPE_CAMERA, EVENT_SCATTERING_TYPE_NONE, sNoLabel);
67 stateId = m.transition(stateId, EVENT_TYPE_REFLECTION, EVENT_SCATTERING_TYPE_DIFFUSE, diffuseLabel);
68 stateId = m.transition(stateId, EVENT_TYPE_LIGHT, EVENT_SCATTERING_TYPE_NONE, sNoLabel);
69 CPPUNIT_ASSERT(m.isValid(stateId, 1));
70 CPPUNIT_ASSERT(!m.isValid(stateId, 2));
71 CPPUNIT_ASSERT(!m.isValid(stateId, 3));
72 CPPUNIT_ASSERT(m.isValid(stateId, 4));
73 CPPUNIT_ASSERT(m.isValid(stateId, 5));
74 }
75
76 stateId = 0; // initial state
77 {
78 // C_ RD'base' L_
79 stateId = m.transition(stateId, EVENT_TYPE_CAMERA, EVENT_SCATTERING_TYPE_NONE, sNoLabel);
80 stateId = m.transition(stateId, EVENT_TYPE_REFLECTION, EVENT_SCATTERING_TYPE_DIFFUSE, baseLabel);
81 stateId = m.transition(stateId, EVENT_TYPE_LIGHT, EVENT_SCATTERING_TYPE_NONE, sNoLabel);
82 CPPUNIT_ASSERT(m.isValid(stateId, 1));
83 CPPUNIT_ASSERT(!m.isValid(stateId, 2));
84 CPPUNIT_ASSERT(!m.isValid(stateId, 3));
85 CPPUNIT_ASSERT(!m.isValid(stateId, 4));
86 CPPUNIT_ASSERT(m.isValid(stateId, 5));
87 }
88}
