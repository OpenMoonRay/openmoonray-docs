23// Helper function stolen from pbrt3 that emulates the behavior of
24// std::upper_bound(). The implementation here adds some bounds checking
25// for corner cases (e.g., making sure that a valid interval is selected
26// even in the case the predicate evaluates to true or false for all entries),
27// which would normally have to follow a call to std::upper_bound()
28template <typename Predicate>
29int findInterval(int size, const Predicate& pred)
30{
31 int first = 0, len = size;
32 while (len > 0) {
33 int half = len >> 1, middle = first + half;
34 if (pred(middle)) {
35 first = middle + 1;
36 len -= half + 1;
37 } else {
38 len = half;
39 }
40 }
41 const int result = first - 1;
42 const int maxRes = size - 2;
43 return result < 0 ? 0 :
44 result > maxRes ? maxRes : result;
45}

96 struct Distribution1D
97 {
98 Distribution1D() = default;
99
100 Distribution1D(const float* func, int n) :
101 mFunc(func, func + n), mCdf(n + 1)
102 {
103 // Compute integral of step function
104 mCdf[0] = 0;
105 for (int i = 1; i < n + 1; ++i) {
106 mCdf[i] = mCdf[i - 1] + mFunc[i - 1] / n;
107 }
108 // Transform step function integral into CDF
109 mFuncInt = mCdf[n];
110 if (mFuncInt == 0) {
111 for (int i = 1; i < n + 1; ++i) {
112 mCdf[i] = float(i) / float(n);
113 }
114 } else {
115 for (int i = 1; i < n + 1; ++i) {
116 mCdf[i] /= mFuncInt;
117 }
118 }
119 }
120
121 int count() const
122 {
123 return mFunc.size();
124 }
125
126 int sampleDiscrete(float u, float *pdf, float *uRemapped) const
127 {
128 // Find surrounding CDF segments and offset
129 int offset = findInterval(mCdf.size(),
130 [&](int index) {
131 return mCdf[index] <= u;
132 });
133 *pdf = (mFuncInt > 0) ? mFunc[offset] / (mFuncInt * count()) : 0;
134 *uRemapped = (u - mCdf[offset]) / (mCdf[offset + 1] - mCdf[offset]);
135 return offset;
136 }
137
138 float pdfDiscrete(int index) const
139 {
140 return (mFuncInt > 0) ? mFunc[index] / (mFuncInt * count()) : 0;
141 }
142
143 // Distribution1D Public Data
144 std::vector<float> mFunc;
145 std::vector<float> mCdf;
146 float mFuncInt;
147 };

149 Distribution3D(const float* func, int nx, int ny, int nz) :
150 mNx(nx), mNy(ny), mNz(nz)
151 {
152 mConditionalX.reserve(ny * nz);
153 mMarginalY.reserve(nz);
154 std::vector<float> xyIntegrals(nz, 0.0f);
155 for (int z = 0; z < nz; ++z) {
156 std::vector<float> xIntegrals(ny, 0.0f);
157 for (int y = 0; y < ny; ++y) {
158 int offset = z * (ny * nx) + y * nx;
159 mConditionalX.emplace_back(&func[offset], nx);
160 xIntegrals[y] = mConditionalX[z * ny + y].mFuncInt;
161 }
162 mMarginalY.emplace_back(xIntegrals.data(), ny);
163 xyIntegrals[z] = mMarginalY[z].mFuncInt;
164 }
165 mMarginalZ = Distribution1D(xyIntegrals.data(), nz);
166 }
167
168 int count() const
169 {
170 return mMarginalZ.count();
171 }

173 scene_rdl2::math::Vec3i sampleDiscrete(float u1, float u2, float u3, float *pdf,
174 float *uxRemapped, float* uyRemapped, float* uzRemapped) const
175 {
176 float pdfX, pdfY, pdfZ;
177 int z = mMarginalZ.sampleDiscrete(u1, &pdfZ, uzRemapped);
178 int y = mMarginalY[z].sampleDiscrete(u2, &pdfY, uyRemapped);
179 int x = mConditionalX[z * mNy + y].sampleDiscrete(u3, &pdfX, uxRemapped);
180 *pdf = pdfX * pdfY * pdfZ;
181 return scene_rdl2::math::Vec3i(x, y, z);
182 }
183
184 float pdfDiscrete(const scene_rdl2::math::Vec3i& index) const
185 {
186 // out of index boundary
187 if (index[0] < 0 || index[0] >= mNx ||
188 index[1] < 0 || index[1] >= mNy ||
189 index[2] < 0 || index[2] >= mNz) {
190 return 0.0f;
191 }
192 float pdfZ = mMarginalZ.pdfDiscrete(index[2]);
193 float pdfY = mMarginalY[index[2]].pdfDiscrete(index[1]);
194 float pdfX = mConditionalX[index[2] * mNy + index[1]].pdfDiscrete(
195 index[0]);
196 return pdfX * pdfY * pdfZ;
197 }
198
199 int mNx, mNy, mNz;
200 std::vector<Distribution1D> mConditionalX;
201 std::vector<Distribution1D> mMarginalY;
202 Distribution1D mMarginalZ;
203};
