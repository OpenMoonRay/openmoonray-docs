46// Intersection is a bit tricky with instancing.
47// We need to track if there's a new intersection found at each instance level.
48// So we need to store the original ray state before trace new ray,
49// set ray.geomID to invalid, which will be assigned to a valid geom id
50// when a new intersection is found.
51static void
52intersectFunc(const RTCIntersectFunctionNArguments* args)
53{
54 int* valid = (int*)args->valid;
55 unsigned int N = args->N;
56 RTCRayHitN* rayhit = (RTCRayHitN*)args->rayhit;
57 RTCRayN* rays = RTCRayHitN_RayN(rayhit, N);
58 RTCHitN* hits = RTCRayHitN_HitN(rayhit, N);
59 const BVHUserData* userData = (const BVHUserData*)args->geometryUserPtr;
60 const Instance* instance = (const Instance*)userData->mPrimitive;
61 const MotionTransform& xform = instance->getLocal2Parent();
62 const RTCScene referenceScene = instance->getReferenceScene();
63 int geomID = instance->getGeomID();
64 mcrt_common::IntersectContext* context =
65 (mcrt_common::IntersectContext*)args->context;
66 for (unsigned int index = 0; index < N; ++index) {
67 int rayId = RTCRayN_id(rays, N, index);
68 mcrt_common::RayExtension& rayExtension =
69 context->mRayExtension[rayId];
70 if (valid[index] == 0) {
71 continue;
72 }
73 const Mat43 inL2R = rayExtension.l2r;
74 void* inUserData = rayExtension.userData;
75 const void* inInstance0 = rayExtension.instance0OrLight;
76 const void* inInstance1 = rayExtension.instance1;
77 const void* inInstance2 = rayExtension.instance2;
78 const void* inInstance3 = rayExtension.instance3;
79 const int inInstanceAttributesDepth = rayExtension.instanceAttributesDepth;
80 const int inVolumeInstanceState = rayExtension.volumeInstanceState;
81
82 float time = RTCRayN_time(rays, N, index);
83 Mat43 L2P, P2L;
84 if (xform.isStatic()) {
85 L2P = xform.getStaticXform();
86 P2L = xform.getStaticInverse();
87 } else {
88 L2P = xform.eval(time);
89 P2L = L2P.inverse();
90 }
91
92 Vec3f localOrg = scene_rdl2::math::transformPoint(P2L, Vec3f(
93 RTCRayN_org_x(rays, N, index),
94 RTCRayN_org_y(rays, N, index),
95 RTCRayN_org_z(rays, N, index)));
96 Vec3f localDir = scene_rdl2::math::transformVector(P2L, Vec3f(
97 RTCRayN_dir_x(rays, N, index),
98 RTCRayN_dir_y(rays, N, index),
99 RTCRayN_dir_z(rays, N, index)));
100
101 RTCRayHit localRay;
102 localRay.ray.org_x = localOrg.x;
103 localRay.ray.org_y = localOrg.y;
104 localRay.ray.org_z = localOrg.z;
105 localRay.ray.tnear = RTCRayN_tnear(rays, N, index);
106 localRay.ray.dir_x = localDir.x;
107 localRay.ray.dir_y = localDir.y;
108 localRay.ray.dir_z = localDir.z;
109 localRay.ray.time = time;
110 localRay.ray.tfar = RTCRayN_tfar(rays, N, index);
111 localRay.ray.mask = RTCRayN_mask(rays, N, index);
112 localRay.ray.id = rayId;
113 localRay.hit.geomID = RTC_INVALID_GEOMETRY_ID;
114 localRay.hit.instID[0] = RTC_INVALID_GEOMETRY_ID;
115 rayExtension.l2r = Mat43(scene_rdl2::math::one);
116 rayExtension.userData = nullptr;
117 // Clear instance pointers at and beyond the current depth
118 memset(&rayExtension.instance0OrLight + rayExtension.instanceAttributesDepth, 0,
119 (Instance::sMaxInstanceAttributesDepth - rayExtension.instanceAttributesDepth) * sizeof(void*));
120
121 // Increase the instance attribute depth so we know
122 // which instanceN member to set if the ray intersects the instance.
123 // This intersectFunc is called recursively with multi-level instancing.
124 // Note that we only increment the depth if we are at the top level or
125 // the instance has attributes. We always need to set the top level
126 // instance for motion vector support.
127 bool topLevel = (rayExtension.instanceAttributesDepth == 0);
128 if (topLevel || instance->getAttributes()) {
129 rayExtension.instanceAttributesDepth++;
130 }
131
132 // volume instance ids
133 // top level instance primitives are initialized with state = 0
134 // rayExtension.volumeInstanceState will be >= 0 if the volume instance
135 // state is valid, -1 otherwise (invalid transition, or not a volume instance)
136 // currently, only volume rays have a non-null geomTls
137 const int state = topLevel ? 0 : rayExtension.volumeInstanceState;
138 const geom::internal::TLState *geomTls = static_cast<geom::internal::TLState *>(rayExtension.geomTls);
139 if (state >= 0 && geomTls) {
140 MNRY_ASSERT(geomTls->mVolumeRayState.getVolumeAssignmentTable() != nullptr);
141 const geom::internal::VolumeIdFSM &fsm =
142 geomTls->mVolumeRayState.getVolumeAssignmentTable()->getInstanceVolumeIds();
143 rayExtension.volumeInstanceState = fsm.transition(state, instance);
144 }
145
146 rtcIntersect1(referenceScene, &context->mRtcContext, &localRay);
147
148 if (localRay.hit.geomID == RTC_INVALID_GEOMETRY_ID) {
149 // no new intersection found, restore original intersection state
150 rayExtension.userData = inUserData;
151 rayExtension.l2r = inL2R;
152 // we restore the instanceAttributesDepth below
153 rayExtension.instance0OrLight = inInstance0;
154 rayExtension.instance1 = inInstance1;
155 rayExtension.instance2 = inInstance2;
156 rayExtension.instance3 = inInstance3;
157 } else {
158 // intersection is found, update the intersection state
159 if (localRay.hit.instID[0] != localRay.hit.geomID) {
160 // ray intersect a leaf Primitive
161 rayExtension.userData = rtcGetGeometryUserData(
162 rtcGetGeometry(referenceScene, localRay.hit.geomID));
163 }
164 // the way we use instID is different from how embree's official
165 // single level instance geometry. If the ray hits an instance,
166 // we mark the instID = geomID. We use this mechanic to identify
167 // whether the ray hit an instance or not after rtcIntersect call
168 localRay.hit.geomID = geomID;
169 localRay.hit.instID[0] = geomID;
170 // we found a hit, concat scene transform to ray l2r
171 RTCRayN_tfar(rays, N, index) = localRay.ray.tfar;
172 rayExtension.l2r *= L2P;
173 rtcCopyHitToHitN(hits, &localRay.hit, N, index);
174 switch (rayExtension.instanceAttributesDepth) {
175 case 1:
176 if (topLevel) {
177 rayExtension.instance0OrLight = instance;
178 }
179 break;
180 case 2:
181 if (instance->getAttributes()) {
182 rayExtension.instance1 = instance;
183 }
184 break;
185 case 3:
186 if (instance->getAttributes()) {
187 rayExtension.instance2 = instance;
188 }
189 break;
190 case 4:
191 if (instance->getAttributes()) {
192 rayExtension.instance3 = instance;
193 }
194 break;
195 default:
196 // We ignore instances more than sMaxInstanceDepth deep.
197 // This means prim attrs are ignored on those instances.
198 break;
199 }
200 }
201 // restore the previous instance depth before the recursion unwinds
202 rayExtension.instanceAttributesDepth = inInstanceAttributesDepth;
203 // restore volume instance state
204 rayExtension.volumeInstanceState = inVolumeInstanceState;
205 }

214static void
215occludedFunc(const RTCOccludedFunctionNArguments* args)
216{
217 int* valid = (int*)args->valid;
218 unsigned int N = args->N;
219 RTCRayN* rays = args->ray;
220 const BVHUserData* userData = (const BVHUserData*)args->geometryUserPtr;
221 const Instance* instance = (const Instance*)userData->mPrimitive;
222 const MotionTransform& xform = instance->getLocal2Parent();
223 const RTCScene referenceScene = instance->getReferenceScene();
224 mcrt_common::IntersectContext* context =
225 (mcrt_common::IntersectContext*)args->context;
226 for (unsigned int index = 0; index < N; ++index) {
227 if (valid[index] == 0) {
228 continue;
229 }
230 Vec3f localOrg;
231 Vec3f localDir;
232 float time = RTCRayN_time(rays, N, index);
233 const Mat43 P2L = xform.isStatic() ?
234 xform.getStaticInverse() : xform.eval(time).inverse();
235 // transform ray to local space
236 localOrg = scene_rdl2::math::transformPoint(P2L, Vec3f(
237 RTCRayN_org_x(rays, N, index),
238 RTCRayN_org_y(rays, N, index),
239 RTCRayN_org_z(rays, N, index)));
240 localDir = scene_rdl2::math::transformVector(P2L, Vec3f(
241 RTCRayN_dir_x(rays, N, index),
242 RTCRayN_dir_y(rays, N, index),
243 RTCRayN_dir_z(rays, N, index)));
244 RTCRay localRay;
245 localRay.org_x = localOrg.x;
246 localRay.org_y = localOrg.y;
247 localRay.org_z = localOrg.z;
248 localRay.tnear = RTCRayN_tnear(rays, N, index);
249 localRay.dir_x = localDir.x;
250 localRay.dir_y = localDir.y;
251 localRay.dir_z = localDir.z;
252 localRay.time = time;
253 localRay.tfar = RTCRayN_tfar(rays, N, index);
254 localRay.mask = RTCRayN_mask(rays, N, index);
255 localRay.id = RTCRayN_id(rays, N, index);
256 context->mRtcContext.instID[0] = instance->getGeomID();
257 rtcOccluded1(referenceScene, &context->mRtcContext, &localRay);
258 context->mRtcContext.instID[0] = RTC_INVALID_GEOMETRY_ID;
259 RTCRayN_tfar(rays, N, index) = localRay.tfar;
260 }

269BBox3f
270Instance::computeAABB() const
271{
272 const RTCScene& referenceScene = getReferenceScene();
273 const MotionTransform& local2Parent = getLocal2Parent();
274 RTCBounds refBound;
275 rtcGetSceneBounds(referenceScene, &refBound);
276 BBox3f localBound;
277 localBound.lower = Vec3f(refBound.lower_x, refBound.lower_y, refBound.lower_z);
278 localBound.upper = Vec3f(refBound.upper_x, refBound.upper_y, refBound.upper_z);
279 // Instance with empty source scene (usually due to incorrect user setup)
280 // Make it a valid (but meaningless) bounding box so it won't break the
281 // partition process during BVH construction
282 if (!scene_rdl2::math::isFinite(localBound.lower) ||
283 !scene_rdl2::math::isFinite(localBound.upper)) {
284 return BBox3f(0.0f);
285 }
286
287 // MOONRAY-4472
288 // Handle case where the local2Parent MotionTransform has not been
289 // initialized to prevent corrupting the BVH
290 if (!local2Parent.isInitialized()) {
291 return localBound;
292 }
293
294 BBox3f bound;
295 if (local2Parent.isStatic()) {
296 bound = transformBounds(local2Parent.getStaticXform(), localBound);
297 } else {
298 // TODO this sample based bounding box calculation may generate
299 // insufficiently conservative result(though in practice it works well)
300 // pbrt-v3 has an implementation that formulating bbox corner movement
301 // as time space function and evaluate the derivative zero point.
302 // Might worth investigating and port it in the future
303 bound = merge(
304 transformBounds(local2Parent.eval(0), localBound),
305 transformBounds(local2Parent.eval(1), localBound));
306 constexpr size_t sampleCount = 128;
307 const float sampleDelta = 1.0f / sampleCount;
308 for (size_t i = 1; i < sampleCount; ++i) {
309 float t = i * sampleDelta;
310 bound = merge(bound,
311 transformBounds(local2Parent.eval(t), localBound));
312 }
313 }
314 return bound;
315}

317BBox3f
318Instance::computeAABBAtTimeStep(int timeStep) const
319{
320 const RTCScene& referenceScene = getReferenceScene();
321 const MotionTransform& local2Parent = getLocal2Parent();
322 RTCBounds refBound;
323 rtcGetSceneBounds(referenceScene, &refBound);
324 BBox3f localBound;
325 localBound.lower = Vec3f(refBound.lower_x, refBound.lower_y, refBound.lower_z);
326 localBound.upper = Vec3f(refBound.upper_x, refBound.upper_y, refBound.upper_z);
327 // Instance with empty source scene (usually due to incorrect user setup)
328 // Make it a valid (but meaningless) bounding box so it won't break the
329 // partition process during BVH construction
330 if (!scene_rdl2::math::isFinite(localBound.lower) ||
331 !scene_rdl2::math::isFinite(localBound.upper)) {
332 return BBox3f(0.0f);
333 }
334
335 if (local2Parent.isStatic()) {
336 return transformBounds(local2Parent.getStaticXform(), localBound);
337 } else {
338 MNRY_ASSERT(timeStep >= 0 && timeStep < static_cast<int>(getMotionSamplesCount()), "timeStep out of range");
339 return transformBounds(local2Parent.eval(timeStep), localBound);
340 }
