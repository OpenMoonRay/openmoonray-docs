19template <typename T>
20constexpr T
21getZero()
22{
23 // All of the built-in types return 0 with an explicit default constructor
24 // call (e.g. int()). However, we have things like vec3f, which don't behave
25 // like the built-in types; default construction is uninitialized. We have
26 // to construct our types with math::zero.
27 return T(math::zero);
28}

140template <typename T>
141void RunningStats<T>::push(data_type x)
142{
143 const unsigned long n1 = n;
144 ++n;
145 const data_type delta = x - M1;
146 const data_type delta_n = delta / n;
147 const data_type delta_n2 = delta_n * delta_n;
148 const data_type term1 = delta * delta_n * n1;
149 M1 += delta_n;
150 M4 += term1 * delta_n2 * (n*n - 3*n + 3) + 6 * delta_n2 * M2 - 4 * delta_n * M3;
151 M3 += term1 * delta_n * (n - 2) - 3 * delta_n * M2;
152 M2 += term1;
153}

198template <typename T>
199RunningStats<T> operator+(const RunningStats<T>& a, const RunningStats<T>& b)
200{
201 typedef typename RunningStats<T>::data_type data_type;
202
203 RunningStats<T> combined;
204
205 combined.n = a.n + b.n;
206
207 const data_type delta1 = b.M1 - a.M1;
208 const data_type delta2 = delta1*delta1;
209 const data_type delta3 = delta1*delta2;
210 const data_type delta4 = delta2*delta2;
211
212 combined.M1 = (a.n*a.M1 + b.n*b.M1) / combined.n;
213
214 combined.M2 = a.M2 + b.M2 +
215 delta2 * a.n * b.n / combined.n;
216
217 combined.M3 = a.M3 + b.M3 +
218 delta3 * a.n * b.n * (a.n - b.n)/(combined.n*combined.n);
219 combined.M3 += data_type(3)*delta1 * (a.n*b.M2 - b.n*a.M2) / combined.n;
220
221 combined.M4 = a.M4 + b.M4 + delta4*a.n*b.n * (a.n*a.n - a.n*b.n + b.n*b.n) /
222 (combined.n*combined.n*combined.n);
223 combined.M4 += data_type(6)*delta2 * (a.n*a.n*b.M2 + b.n*b.n*a.M2)/(combined.n*combined.n) +
224 data_type(4)*delta1*(a.n*b.M3 - b.n*a.M3) / combined.n;
225
226 return combined;
227}

245template <typename T>
246void RunningStatsLightWeight<T>::push(T x)
247{
248 ++n;
249
250 // See Knuth TAOCP vol 2, 3rd edition, page 232
251 if (unlikely(n == 1)) {
252 mOldM = mNewM = x;
253 mOldS = getZero<T>();
254 } else {
255 mNewM = mOldM + (x - mOldM)/T(n);
256 mNewS = mOldS + (x - mOldM)*(x - mNewM);
257
258 // set up for next iteration
259 mOldM = mNewM;
260 mOldS = mNewS;
261 }
262}

315template <typename T>
316int
317RunningStatsLightWeightFulldump<T>::fillPixBuffer(float *dstPixBufferAddr) const
318//
319// Store all this pixel data into float buffer
320//
321{
322 //
323 // This is a tricky part.
324 // Very first 4 byte of *this data is sample count n (unsigned int).
325 // And final pixel buffer is compressed by HALF_FLOAT or FLOAT compression data type.
326 // So we need to cast unsigned int n value to float here. (This is not a bitimage copy.)
327 //
328 // MNRY_ASSERT(sizeof(this->n) == 4, "n must be 32-bits"); // somehow this does not work. temporaly commented out.
329
330 float *dstPtr = dstPixBufferAddr;
331 unsigned int ui = *(reinterpret_cast<const unsigned int *>(this));
332 dstPtr[0] = (float)ui;
333
334 // Rest of them are simple float data and ready to do simple copy.
335 size_t total = sizeof(RunningStatsLightWeightFulldump<T>) / sizeof(float);
336 const float *srcPtr = reinterpret_cast<const float *>(this);
337 for (size_t i = 1; i < total; ++i) {
338 dstPtr[i] = srcPtr[i];
339 }
340 return total; // return data size as float count
341}
