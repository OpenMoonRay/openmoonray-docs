46Accumulator *
47initAccumulators(unsigned numThreads, unsigned numTLS)
48{
49 MNRY_ASSERT(gPrivate.mRoot == nullptr && numThreads > 0 && numTLS >= numThreads);
50
51 MOONRAY_THREADSAFE_STATIC_WRITE(gPrivate.mNumThreads = numThreads);
52 MOONRAY_THREADSAFE_STATIC_WRITE(gPrivate.mNumTLS = numTLS);
53
54 // This is freed when we are freeing the contents of the gPrivate.mAccumulators array.
55 MOONRAY_THREADSAFE_STATIC_WRITE(gPrivate.mRoot = scene_rdl2::util::alignedMallocCtorArgs<Accumulator>
56 (CACHE_LINE_SIZE, "Root", 0, numTLS, ACCFLAG_NONE));
57
58 gPrivate.mAccumulators.push_back(gPrivate.mRoot);
59
60 return gPrivate.mRoot;
61}

132unsigned
133snapshotRawAccumulators(std::vector<AccumulatorResult> *dstResults, double rcpTickFrequency)
134{
135 MNRY_ASSERT(gNumAccumulatorsActive == 0);
136 MNRY_ASSERT(gPrivate.mRoot && gPrivate.mNumThreads);
137 MNRY_ASSERT(dstResults);
138 MNRY_ASSERT(rcpTickFrequency > 0.0);
139
140 uint64_t rootTicks = gPrivate.mRoot->getAccumulatedTicks();
141 if (rootTicks == 0) {
142 return 0;
143 }
144
145 const size_t numAccumulators = gPrivate.mAccumulators.size();
146
147 // "results" is an alias for dstResults.
148 std::vector<AccumulatorResult> &results = *dstResults;
149 results.resize(numAccumulators);
150
151 // Root is assumed to be only timed on the main thread outside of any
152 // rendering threads. Because of this, we don't need to divide by the number
153 // of threads to get the actual wall clock time.
154 const double wallClockTime = double(rootTicks) * rcpTickFrequency;
155
156 // Root result is a special case since it's only run on one thread.
157 // The root accumulator is always the first in the array.
158 MNRY_ASSERT(gPrivate.mAccumulators[0] == gPrivate.mRoot);
159
160 AccumulatorResult &rootResult = results[0];
161 rootResult.mName = gPrivate.mRoot->mName.c_str();
162 rootResult.mFlags = gPrivate.mRoot->mFlags;
163 rootResult.mTotalTime = wallClockTime * double(gPrivate.mNumThreads);
164 rootResult.mTimePerThread = wallClockTime;
165 rootResult.mPercentageOfTotal = 100.0;
166
167 // Gather remaining results.
168 const double rcpNumThreads = 1.0 / double(gPrivate.mNumThreads);
169 const double rcpWallClockTime = 1.0 / wallClockTime;
170 unsigned numResults = 1;
171
172 for (unsigned iacc = 1; iacc < gPrivate.mAccumulators.size(); ++iacc) {
173
174 Accumulator *acc = MNRY_VERIFY(gPrivate.mAccumulators[iacc]);
175 AccumulatorResult &result = results[numResults];
176
177 uint64_t ticks = acc->getAccumulatedTicks();
178
179 result.mName = acc->mName.c_str();
180 result.mFlags = acc->mFlags;
181 result.mTotalTime = double(ticks) * rcpTickFrequency;
182 result.mTimePerThread = result.mTotalTime * rcpNumThreads;
183 result.mPercentageOfTotal = result.mTimePerThread * rcpWallClockTime * 100.0;
184
185 ++numResults;
186 }
187
188 // The root accumulator should always be at the start of the list.
189 MNRY_ASSERT(results[0].mName == gPrivate.mRoot->mName.c_str());
190
191 MNRY_ASSERT(gNumAccumulatorsActive == 0);
192
193 return (unsigned)numAccumulators;
194}

196unsigned
197snapshotSortedAccumulators(std::vector<AccumulatorResult> *dstResults,
198 double rcpTickFrequency,
199 double threshold)
200{
201 if (!snapshotRawAccumulators(dstResults, rcpTickFrequency)) {
202 return 0;
203 }
204
205 // "results" is an alias for dstResults.
206 std::vector<AccumulatorResult> &results = *dstResults;
207
208 // Cull results which fall below the threshold.
209 auto newEnd = std::remove_if(results.begin(), results.end(), [&](const AccumulatorResult &acc) {
210 return acc.mTotalTime < threshold || ((acc.mFlags & ACCFLAG_DISPLAYABLE) == 0);
211 });
212
213 results.erase(newEnd, results.end());
214
215 // Greater than.
216 std::stable_sort(results.begin(), results.end(), [](const AccumulatorResult &a, const AccumulatorResult &b) -> bool
217 {
218 MNRY_ASSERT(a.mName && b.mName);
219
220 bool bracketA = (a.mName[0] == '[');
221 bool bracketB = (b.mName[0] == '[');
222
223 if (bracketA) {
224 if (bracketB) {
225 return a.mTimePerThread > b.mTimePerThread;
226 } else {
227 return false;
228 }
229 } else {
230 if (bracketB) {
231 return true;
232 } else {
233 // If an entry is marked with the TOTAL flag, then sort it
234 // at the end of the exclusive times, but before any bracketed
235 // times.
236 if (a.mFlags & ACCFLAG_TOTAL) return false;
237 if (b.mFlags & ACCFLAG_TOTAL) return true;
238 return a.mTimePerThread > b.mTimePerThread;
239 }
240 }
241 });
242
243 return (unsigned)results.size();
244}

269uint64_t Accumulator::getAccumulatedTicks() const
270{
271 uint64_t localTicks = 0;
272
273 for (unsigned i = 0; i < gPrivate.mNumThreads; ++i) {
274
275 const auto &tla = mThreadLocal[i];
276
277 if (tla.mTotalTime > 0) {
278 localTicks += tla.mTotalTime;
279 }
280 }
281
282 return localTicks;
283}
