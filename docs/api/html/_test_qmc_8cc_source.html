157 float
158 haltonLimit(unsigned samples)
159 {
160 const float dimensions = 2.0f;
161 const float ns = static_cast<float>(samples);
162 return pow(log10(ns), dimensions) / ns;

174 // Measure the star discrepancy from one sample point (a single box) in the
175 // space [0, 1)^2.
176 float
177 measureStarDiscrepancy(const SamplePoint2D& measurePoint, const PointContainer& points)
178 {
179 unsigned n = 0;
180 for (auto p = points.begin(); p != points.end(); ++p) {
181 if (p->x < measurePoint.x && p->y < measurePoint.y) {
182 ++n;
183 }
184 }
185
186 return abs(static_cast<float>(n)/static_cast<float>(points.size()) - measurePoint.x*measurePoint.y);
187 }

189 class MeasureStarDiscrepancySplit
190 {
191 public:
192 explicit MeasureStarDiscrepancySplit(const PointContainer& points) :
193 mLUB(0.0f),
194 mRand(sSeed++),
195 mPoints(points)
196 {
197 }
198
199 MeasureStarDiscrepancySplit(MeasureStarDiscrepancySplit& x, tbb::split) :
200 mLUB(0.0f),
201 mRand(sSeed++), // Call with new seed
202 mPoints(x.mPoints)
203 {
204 }
205
206 void operator()(const ::tbb::blocked_range<unsigned>& r)
207 {
208 float lub = mLUB;
209 const auto end = r.end();
210 for (auto i = r.begin(); i != end; ++i) {
211 lub = max(lub, measureStarDiscrepancy(generateRandomPoint(mRand), mPoints));
212 }
213 mLUB = lub;
214 }
215
216 void join(const MeasureStarDiscrepancySplit& y)
217 {
218 mLUB = max(mLUB, y.mLUB);
219 }
220
221 float getLUB() const
222 {
223 return mLUB;
224 }

228 private:
229 float mLUB;
230 util::Random mRand;
231 const PointContainer& mPoints;
232 };

236 // Estimate the star discrepancy for a set of points by repeatedly random
237 // sampling.
238 float
239 measureStarDiscrepancy(const PointContainer& points)
240 {
241 const unsigned numSamples = 100000u;
242 MeasureStarDiscrepancySplit msd(points);
243 tbb::parallel_reduce(tbb::blocked_range<unsigned>(0, numSamples), msd);
244 return msd.getLUB();
245 }

247 template <typename Generator>
248 PointContainer
249 generatePoints(Generator generator, std::size_t numPoints)
250 {
251 PointContainer points;
252 points.reserve(numPoints);
253
254 for (std::size_t s = 0; s < numPoints; ++s) {
255 points.push_back(generator(s));
256 testAssert(points.back().x >= 0.0f && points.back().x < 1.0f, "SamplePoint2D not in range");
257 testAssert(points.back().y >= 0.0f && points.back().y < 1.0f, "SamplePoint2D not in range");
258 }
259
260 return points;
261 }

263 // Aggregate the discrepancy of several random sets of points to get some
264 // statistics.
265 fb_util::RunningStats<>
266 measureRandomDiscrepancy(unsigned nsamples)
267 {
268 util::Random rnd(0x0eadbeef);
269 fb_util::RunningStats<> stats;
270
271 const unsigned numTests = 30;
272 for (unsigned n = 0; n < numTests; ++n) {
273 const auto randomPoints = generatePoints([&rnd](std::size_t) { return generateRandomPoint(rnd); }, nsamples);
274 const float ds = measureStarDiscrepancy(randomPoints);
275 stats.push(ds);
276 }
277
278 return stats;
279 }

281 // Make sure the discrepancy decreases as the number of samples increases.
282 // The discrepancy of a d-dimensional Halton sequence is
283 // D_{N}^{*}(x_i) = O\left ( \frac{(\log N)^{d}}{N} \right )
284 // After a local maximum at N == e^2, this is monotonically and
285 // asymptotically decreasing as the number of samples increases.
286 template <typename Generator>
287 void
288 testDiscrepancyLimit()
289 {
290 const uint32_t scramble[2] = { 0xe2611dd7, 0x373a9312 };
291 float C = 1.0f; // Big-O constant.
292 float lastStarDiscrepancy = 1.0f;
293
294 {
295 // The limit is the Big-O of the Halton sequence as the number of
296 // samples increases. That means that there exists a constant, C, for
297 // which C*O(N) > f(N). We're going to estimate that constant at
298 // the local maximum of the limiting function: e^2 < 8.
299 Generator generator(scramble);
300 const float D = measureStarDiscrepancy(generatePoints(generator, 8));
301 const float O = haltonLimit(8);
302 C = D/O;
303 lastStarDiscrepancy = D;
304 }
305
306 C = min(C, 1.0f);
307
308 const unsigned numTests = 20;
309 for (unsigned testNum = 4; testNum < numTests; ++testNum) {
310 const std::size_t size = 1 << testNum;
311
312 Generator generator(scramble);
313
314 const float starDiscrepancy = measureStarDiscrepancy(generatePoints(generator, size));
315 //std::cout << __func__ << " N(" << size << "): " << starDiscrepancy << " & " << (lastStarDiscrepancy * C) << '\n';
316 testAssert(lastStarDiscrepancy * C > starDiscrepancy, "Discrepancy should decrease as we increase the number of samples");
317 lastStarDiscrepancy = starDiscrepancy;
318 }
319 }

321 // Test that the discrepancy varies little for a number of points using
322 // different shuffle values.
323 template <typename Generator>
324 void
325 testScrambleDiscrepancy()
326 {
327 util::Random rnd(0x0eadbeef);
328
329 // Always use the same size for this test. Different sizes will lead to
330 // different discrepancies.
331 const uint32_t size = 2048;
332
333 // Get the discrepancy values for a set of random points. We will do
334 // our comparisons against these.
335 const auto randomStats = measureRandomDiscrepancy(size);
336
337 // The maximum star discrepancy is 1.0
338 float lastStarDiscrepancy = 1.0f;
339 {
340 // Set up initial comparison.
341 const uint32_t scramble[2] = { rnd.getNextUInt(), rnd.getNextUInt() };
342 Generator generator(scramble);
343 lastStarDiscrepancy = measureStarDiscrepancy(generatePoints(generator, size));
344 }
345
346 const unsigned numTests = 20;
347 for (unsigned testNum = 0; testNum < numTests; ++testNum) {
348 const uint32_t scramble[2] = { rnd.getNextUInt(), rnd.getNextUInt() };
349 Generator generator(scramble);
350
351 const float starDiscrepancy = measureStarDiscrepancy(generatePoints(generator, size));
352 testAssert(starDiscrepancy < randomStats.mean(), "The distribution should (most of the time) do better than purely random sampling. There is a small random chance that this fails.");
353 testAssert(abs(starDiscrepancy - lastStarDiscrepancy) < randomStats.standardDeviation(), "We should be close, no matter how we shuffle.");
354 lastStarDiscrepancy = starDiscrepancy;
355 //std::cout << __func__ << ": " << starDiscrepancy << '\n';
356 }
357 }
