49//----------------------------------------------------------------------------
50
51static void
52runTest(TestBsdfSettings &test, int viewAnglesTheta,
53 int viewAnglesPhy, int sampleCount)
54{
55// printInfo("Input sample count: %d", sampleCount);
56
57 mcrt_common::ThreadLocalState *tls = mcrt_common::getFrameUpdateTLS();
58 pbr::TLState *pbrTls = MNRY_VERIFY(tls->mPbrTls.get());
59 scene_rdl2::alloc::Arena *arena = &tls->mArena;
60
61 {
62 SCOPED_MEM(arena);
63 shading::Bsdf *bsdf = test.bsdfFactory(*arena, test.frame);
64 shading::BsdfSlice slice(test.frame.getN(), test.frame.getN(), true, true,
65 ispc::SHADOW_TERMINATOR_FIX_OFF);
66 const PathGuide pg;
67 BsdfSampler sampler(pbrTls->mArena, *bsdf, slice, test.sMaxSamplesPerLobe,
68 true, pg);
69 sampleCount /= sampler.getSampleCount();
70 }
71
72// printInfo("Sampler sample count: %d", sampler.getSampleCount());
73// printInfo("Adjusted sample count: %d", sampleCount);
74
75 runBsdfTest(test, viewAnglesTheta, viewAnglesPhy, sampleCount);
76}

502void
503TestBsdfSampler::testStochasticFlakes()
504{
505 scene_rdl2::math::ReferenceFrame frame;
506
507 static const size_t flakesDataSize = 100;
508
509 printInfo("##### TestBsdfSampler::testStochasticFlakes() ####################");
510 for (int i = 0; i < sRoughnessCount; i++) {
511 printInfo("===== Roughness: %f ====================", sRoughness[i]);
512
513 Vec3Array normals;
514 ColorArray colors;
515 generateGGXNormals(sRoughness[i] * sRoughness[i], flakesDataSize, normals);
516 generateWeightedFlakeColors(flakesDataSize, colors);
517 for (size_t j = 0; j < flakesDataSize; ++j) {
518 normals[j] = frame.localToGlobal(normals[j]);
519 }
520
521 StochasticFlakesBsdfFactory factory(&normals[0], &colors[0], flakesDataSize, 0.1f, sRoughness[i]);
522 TestBsdfSettings test(factory, frame, true, 0.008, 0.03,
523 (sRoughness[i] > 0.1f && sRoughness[i] < 0.4f),
524 (sRoughness[i] > 0.1f && sRoughness[i] < 0.4f),
525 TestBsdfSettings::BSDF_SAMPLER);
526 runTest(test, sViewAnglesTheta, 1, getSampleCount(sRoughness[i]));
527
528 normals.clear();
529 colors.clear();
530 }
