66bool
67SpotLight::update(const Mat4d &world2render)
68{
69 MNRY_ASSERT(mRdlLight);
70
71 mOn = mRdlLight->get(scene_rdl2::rdl2::Light::sOnKey);
72 if (!mOn) {
73 return false;
74 }
75
76 updateVisibilityFlags();
77 updatePresenceShadows();
78 updateRayTermination();
79 updateTextureFilter();
80 updateMaxShadowDistance();
81 UPDATE_ATTRS_CLEAR_RADIUS
82
83 const Mat4d l2w0 = mRdlLight->get(scene_rdl2::rdl2::Node::sNodeXformKey, /* rayTime = */ 0.f);
84 const Mat4d l2w1 = mRdlLight->get(scene_rdl2::rdl2::Node::sNodeXformKey, /* rayTime = */ 1.f);
85 Mat4f rot = sRotateX180;
86 const Mat4f local2Render0 = rot * toFloat(l2w0 * world2render);
87 const Mat4f local2Render1 = rot * toFloat(l2w1 * world2render);
88
89 // Aspect ratio is local y-dimension divided by local x-dimension; mRcpAspectRatio is its reciprocal
90 mRcpAspectRatio = 1.0f / mRdlLight->get<scene_rdl2::rdl2::Float>(sAspectRatioKey);
91
92 mLensRadius = mRdlLight->get<scene_rdl2::rdl2::Float>(sLensRadiusKey);
93 mRcpLensRadius = 1.0f / mLensRadius;
94 mLensRadiusY = mLensRadius * mRcpAspectRatio;
95 mRcpLensRadiusY = 1.0f / mLensRadiusY;
96
97 // Inner and outer angles are measured from one side to the other
98 float outerConeAngle = mRdlLight->get<scene_rdl2::rdl2::Float>(sOuterConeAngleKey);
99 float innerConeAngle = mRdlLight->get<scene_rdl2::rdl2::Float>(sInnerConeAngleKey);
100
101 // Outer cone is subject to barely any restriction; inner cone angle must not exceed outer cone angle
102 outerConeAngle = clamp(outerConeAngle, -179.999f, 179.999f);
103 innerConeAngle = clamp(innerConeAngle, -179.999f, outerConeAngle);
104
105 float tanHalfAngleOuter = scene_rdl2::math::tan(deg2rad(outerConeAngle * 0.5f));
106 float tanHalfAngleInner = scene_rdl2::math::tan(deg2rad(innerConeAngle * 0.5f));
107 float tanHalfAngleDiff = tanHalfAngleOuter - tanHalfAngleInner;
108
109 mFocalDistance = mRdlLight->get<scene_rdl2::rdl2::Float>(sFocalPlaneDistanceKey);
110 if (mFocalDistance < 0.0001f) {
111 mFocalDistance = 0.0001f;
112 }
113
114 // If cone tapers inwards, we must clamp the focal plane so it's closer than the apex by some arbitrary amount
115 if (tanHalfAngleOuter < 0.0f) {
116 mFocalDistance = std::min(mFocalDistance, -0.999f * mLensRadius / tanHalfAngleOuter);
117 }
118
119 // Radius of pool of light at focal plane
120 mFocalRadius = tanHalfAngleOuter * mFocalDistance + mLensRadius;
121 mRcpFocalRadius = 1.0f / mFocalRadius;
122 mFocalRadiusY = mFocalRadius * mRcpAspectRatio;
123 mRcpFocalRadiusY = 1.0f / mFocalRadiusY;
124
125 // Rate of falloff between inner and outer focal radii
126 mFalloffGradient = mFocalRadius / std::max(tanHalfAngleDiff * mFocalDistance, 1.0e-10f);
127
128 // Distance at which sampling scheme switches from sampling over focal plane to sampling over lens
129 mCrossOverDistance = mFocalDistance * mLensRadius / (mLensRadius + mFocalRadius);
130
131 if (!updateParamAndTransforms(local2Render0, local2Render1, mLensRadius, mLensRadiusY)) {
132 return false;
133 }
134
135 // Compute render space corners of light.
136 // Cached at rayTime = 0
137 computeCorners(mRenderCorners, 0.f);
138
139 // Compute render space quantities.
140 // TODO: mb scale radiance
141 float halfRenderWidth = mLensRadius * mLocal2RenderScale[0];
142 float halfRenderHeight = mLensRadiusY * mLocal2RenderScale[0];
143
144 mArea = sPi * halfRenderWidth * halfRenderHeight;
145 mInvArea = 1.0f / mArea;
146
147 // Constant used in pdf calc
148 float lensRadiusOverFocalRadius = mLensRadius / mFocalRadius;
149 mFocalPlanePdfConst = mInvArea * lensRadiusOverFocalRadius * lensRadiusOverFocalRadius;
150
151 // Compute radiance.
152 mRadiance = computeLightRadiance(mRdlLight, scene_rdl2::rdl2::Light::sColorKey,
153 scene_rdl2::rdl2::Light::sIntensityKey, scene_rdl2::rdl2::Light::sExposureKey, sNormalizedKey,
154 sApplySceneScaleKey, mInvArea);
155 if (isBlack(mRadiance)) {
156 mOn = false;
157 return false;
158 }
159
160 // Falloff curve
161 mFalloffCurve.init(
162 FalloffCurveType(mRdlLight->get<scene_rdl2::rdl2::Int>(sAngleFalloffTypeKey)));
163 mBlackLevel = mRdlLight->get<scene_rdl2::rdl2::Float>(sBlackLevelKey);
164
165 if (!updateImageMap(Distribution2D::Mapping::CIRCULAR)) {
166 return false;
167 }
168
169 return true;
170}

173bool
174SpotLight::canIlluminate(const Vec3f p, const Vec3f *n, float time, float radius,
175 const LightFilterList* lightFilterList) const
176{
177 MNRY_ASSERT(mOn);
178
179 // Cull points which are completely on the backside of the light
180 Vec3f localP = xformPointRender2Local(p, time);
181 if (localP.z <= 0.0f) {
182 return false;
183 }
184
185 // Cull lights which are completely on the backside of the point
186 const float threshold = sEpsilon - radius;
187 if (n) {
188 Plane pl(p, *n);
189 float d0, d1, d2, d3;
190 if (!isMb()) {
191 d0 = pl.getDistance(mRenderCorners[0]);
192 d1 = pl.getDistance(mRenderCorners[1]);
193 d2 = pl.getDistance(mRenderCorners[2]);
194 d3 = pl.getDistance(mRenderCorners[3]);
195 } else {
196 Vec3f corners[4];
197 computeCorners(corners, time);
198 d0 = pl.getDistance(corners[0]);
199 d1 = pl.getDistance(corners[1]);
200 d2 = pl.getDistance(corners[2]);
201 d3 = pl.getDistance(corners[3]);
202 }
203 bool canIllum = d0 > threshold || d1 > threshold ||
204 d2 > threshold || d3 > threshold;
205 if (!canIllum) return false;
206 }
207
208 if (lightFilterList) {
209 return canIlluminateLightFilterList(lightFilterList,
210 { getPosition(time),
211 xformLocal2RenderScale(math::max(mLensRadius, mLensRadiusY), time),
212 p, getXformRender2Local(time, lightFilterList->needsLightXform()),
213 radius, time
214 });
215 }
216
217 return true;
218}

238// Computing a tight bounding box for a spotlight
239// ----------------------------------------------
240//
241// Bounds for a spotlight are computed as bounds for the lens. In the spotlight's local coordinates,
242// the lens is defined by an ellipse in the (x,y)-plane, centred on the origin, with radius mLensRadius
243// in the x-direction and (mLensRadius * mRcpAspectRatio) in the y-direction.
244//
245// The derivation below is based on the somewhat simpler bounds for a disk light; see DiskLight.cc
246//
247// In local space, a point (x,y,z) on the perimeter of the disk can be expressed parametrically as
248//
249// x = r cos t
250// y = r a sin t
251// z = 0
252//
253// with t ranging from 0 to 2pi, r = mLensRadius and a = mRcpAspectRatio.
254// Suppose the local2render matrix is (Mij) with i=0,1,2,3 and j=0,1,2,3.
255// Then the perimeter point (x,y,z) transforms to (x',y',z') in render space, where
256//
257// x' = M00 r cos t + M10 r a sin t + M30
258// y' = M01 r cos t + M11 r a sin t + M31
259// z' = M02 r cos t + M12 r a sin t + M32
260//
261// Consider just the x' coordinate. Its min and max occur at points where dx'/dt=0. Differentiating the above gives
262//
263// dx'/dt = -M00 r sin t + M10 r a cos t
264//
265// If this equals zero then
266//
267// tan t = a M10 / M00
268//
269// Or, using standard trig identities,
270//
271// cos^2 t = M00^2 / (M00^2 + a^2 M10^2)
272// sin^2 t = a^2 M10^2 / (M00^2 + a^2 M10^2)
273//
274// Taking square roots, and ensuring the chosen combination of signs preserves the sign of tan t,
275//
276// (cos t, sin t) = +/- (M00, a M10) / sqrt(M00^2 + a^2 M10^2)
277//
278// Putting these values into the expression for x' gives
279//
280// x' = M30 +/- r (M00^2 + a^2 M10^2) / sqrt(M00^2 + a^2 M10^2)
281// = M30 +/- r sqrt(M00^2 + a^2 M10^2)
282//
283// The expressions for the min & max y' and z' are similar, and the 3 together can be expressed compactly
284// in terms of Vec3f operations.
285
286BBox3f
287SpotLight::getBounds() const
288{
289 const Vec3f pos = getPosition(0.f);
290 const Vec3f xRow = mLocal2Render[0].l.vx;
291 const Vec3f yRow = mRcpAspectRatio * mLocal2Render[0].l.vy;
292 const Vec3f halfDims = max(mLensRadius * sqrt(xRow*xRow + yRow*yRow),
293 Vec3f(sEpsilon)); // ensure non-degeneracy
294 BBox3f bounds(pos - halfDims, pos + halfDims);
295
296 if (isMb()) {
297 const Vec3f pos = getPosition(1.f);
298 const Vec3f xRow = mLocal2Render[1].l.vx;
299 const Vec3f yRow = mRcpAspectRatio * mLocal2Render[1].l.vy;
300 const Vec3f halfDims = max(mLensRadius * sqrt(xRow*xRow + yRow*yRow),
301 Vec3f(sEpsilon)); // ensure non-degeneracy
302 bounds.extend(BBox3f(pos - halfDims, pos + halfDims));
303 }
304
305 return bounds;
306}

308// Helper functions for converting between the spotlight's coordinate systems:
309// Lens coordinates measure a local position on the lens, in world-space distance units.
310// Normalized lens coordinates do the same but range from -1 to 1 over each axis of the lens.
311// Focal coordinates measure a local position on the focal plane, in world-space distance units.
312// Normalized focal coordinates do the same but range from -1 to 1 over each axis of the focal plane.
313// uvs range from 0 to 1 in each axis of the relevant plane.
314// Note that the lens can have a non 1:1 aspect ratio.
315
316Vec2f
317SpotLight::lensToFocal(const Vec2f &lensCoords, const Vec3f &localP) const
318{
319 float f = mFocalDistance;
320 float z = localP.z;
321 const Vec2f &xy = asVec2(localP);
322
323 // Avoid zero denominator
324 if (z == 0.0f) z = 1.0e-20f;
325
326 return ((z-f) * lensCoords + f * xy) / z;
327}
328
329Vec2f
330SpotLight::focalToLens(const Vec2f &focalCoords, const Vec3f &localP) const
331{
332 float f = mFocalDistance;
333 float z = localP.z;
334 const Vec2f &xy = asVec2(localP);
335
336 // Avoid zero denominator
337 float denom = z - f;
338 if (denom == 0.0f) denom = 1.0e-20f;
339
340 return (z * focalCoords - f * xy) / denom;
341}

378bool
379SpotLight::intersect(const Vec3f &p, const Vec3f *n, const Vec3f &wi, float time,
380 float maxDistance, LightIntersection &isect) const
381{
382 // Transform wi into local space.
383 Vec3f localWi = xformVectorRender2LocalRot(wi, time);
384
385 // Check that ray isn't traveling away from the light's surface.
386 if (localWi.z >= 0.0f) {
387 return false;
388 }
389
390 // Transform p into local space.
391 Vec3f localP = xformVectorRender2Local(p - getPosition(time), time);
392
393 // Check that ray origin is in front of light
394 if (localP.z <= 0.0f) {
395 return false;
396 }
397
398 // Get distance along ray to intersection with lens plane
399 float localDistance = -localP.z / localWi.z;
400 float renderDistance = xformLocal2RenderScale(localDistance, time);
401 if (renderDistance > maxDistance) {
402 return false;
403 }
404
405 // Compute and test lens intersection
406 Vec2f lensCoords = asVec2(localP + localDistance * localWi);
407 if (lengthSqr(getNormalizedLensCoords(lensCoords)) > 1.0f) {
408 return false;
409 }
410
411 isect.N = getDirection(time);
412 isect.distance = renderDistance;
413
414 // Map lens pos to focal plane for uvs
415 Vec2f focalCoords = lensToFocal(lensCoords, localP);
416 Vec2f normalizedFocalCoords = getNormalizedFocalCoords(focalCoords);
417 isect.uv = getUvsFromNormalized(normalizedFocalCoords);
418
419 return true;
420}

422// TODO: sample uniformly over the intersection between the 2 disks (lens, focal plane)
423// TODO: put back the texture importance sampling and combine results using MIS
424bool
425SpotLight::sample(const Vec3f &p, const Vec3f *n, float time, const Vec3f& r,
426 Vec3f &wi, LightIntersection &isect, float rayDirFootprint) const
427{
428 MNRY_ASSERT(mOn);
429
430 Vec3f localP = xformPointRender2Local(p, time);
431 if (localP.z <= 0.0f) {
432 return false;
433 }
434
435 // Both sampling strategies need a sample point on a disk
436 Vec2f diskSample;
437 squareSampleToCircle(r[0], r[1], &diskSample.x, &diskSample.y);
438
439 // We'll compute the sample position in the lens' local frame
440 Vec2f lensCoords;
441
442 // Choose a sampling strategy based on which disk subtends a larger solid angle:
443 // the lens, or the focal plane. Each strategy alone suffers from a region of extreme noise,
444 // but when combined, each covers the problem region of the other.
445 if (localP.z > mCrossOverDistance) {
446 // Sample point on lens
447 lensCoords = getLensCoords(diskSample);
448
449 // Map lens pos to focal plane for uvs
450 Vec2f focalCoords = lensToFocal(lensCoords, localP);
451 Vec2f normalizedFocalCoords = getNormalizedFocalCoords(focalCoords);
452 isect.uv = getUvsFromNormalized(normalizedFocalCoords);
453 } else {
454 // Sample point on focal plane
455 Vec2f focalCoords = getFocalCoords(diskSample);
456
457 // Map to lens pos
458 lensCoords = focalToLens(focalCoords, localP);
459
460 // Quit if projected point is outside lens
461 if (lengthSqr(getNormalizedLensCoords(lensCoords)) > 1.0f) {
462 return false;
463 }
464
465 isect.uv = getUvsFromNormalized(diskSample);
466 }
467
468 Vec3f lensPosRender = xformPointLocal2Render(Vec3f(lensCoords.x, lensCoords.y, 0.0f), time);
469 wi = lensPosRender - p;
470
471 if (n && dot(*n, wi) <= 0.0f) {
472 return false;
473 }
474
475 isect.distance = length(wi);
476 isect.N = getDirection(time);
477
478 wi /= isect.distance;
479
480 return true;
481}

484Color
485SpotLight::eval(mcrt_common::ThreadLocalState* tls, const Vec3f &wi, const Vec3f &p, const LightFilterRandomValues& filterR, float time,
486 const LightIntersection &isect, bool fromCamera, const LightFilterList *lightFilterList, float rayDirFootprint,
487 float *pdf) const
488{
489 MNRY_ASSERT(mOn);
490
491 // Apply texture if present
492 Color radiance = mDistribution ? mDistribution->eval(isect.uv.x, isect.uv.y, 0, mTextureFilter) : sWhite;
493
494 // Apply spotlight falloff function at intersection with focal plane
495 Vec2f normalizedFocalCoords = 2.0f * isect.uv - Vec2f(1.0f, 1.0f);
496 float r = length(normalizedFocalCoords);
497 float falloffParam = (1.0f - r) * mFalloffGradient;
498 float falloff = mFalloffCurve.eval(falloffParam);
499 if (fromCamera) {
500 radiance = lerp(Color(mBlackLevel), radiance, falloff);
501 } else {
502 radiance *= falloff;
503 }
504
505 // Apply light color
506 radiance *= mRadiance;
507
508 // Apply light filter if present
509 if (lightFilterList) {
510 evalLightFilterList(lightFilterList,
511 { tls, &isect, getPosition(time),
512 getDirection(time), p,
513 filterR, time,
514 getXformRender2Local(time, lightFilterList->needsLightXform()),
515 wi
516 },
517 radiance);
518 }
519
520 if (pdf) {
521 float scale = areaToSolidAngleScale(wi, isect.N, isect.distance);
522 Vec3f localP = xformPointRender2Local(p, time);
523 if (localP.z > mCrossOverDistance) {
524 // Compute pdf for sampling/intersecting the lens
525 *pdf = mInvArea * scale;
526 } else {
527 // Compute pdf for sampling/intersecting the focal plane
528 float dz = mFocalDistance - localP.z;
529 float zRatio = dz / localP.z;
530 *pdf = mFocalPlanePdfConst * scale * zRatio * zRatio;
531 }
532 }
533
534 return radiance;
535}

537Vec3f
538SpotLight::getEquiAngularPivot(const Vec3f& r, float time) const
539{
540 // Sampling the image distribution has been removed here,
541 // since spotlight textures are no longer applied to the lens.
542 // Instead we generate a uniformly distributed point on the lens.
543 Vec2f diskSample;
544 squareSampleToCircle(r[0], r[1], &diskSample.x, &diskSample.y);
545 Vec2f lensCoords = getLensCoords(diskSample);
546 return xformPointLocal2Render(Vec3f(lensCoords.x, lensCoords.y, 0.0f), time);
547}
