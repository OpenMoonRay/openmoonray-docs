28ListeningSocket::ListeningSocket(int port) :
29 mFD(-1)
30{
31 addrinfo hints;
32 memset(&hints, 0, sizeof(addrinfo));
33 hints.ai_family = AF_UNSPEC; // Allow IPv4 or IPv6
34 hints.ai_socktype = SOCK_STREAM;
35 hints.ai_flags = AI_PASSIVE; // For wildcard IP address
36 hints.ai_protocol = 0; // Any protocol
37 hints.ai_canonname = nullptr;
38 hints.ai_addr = nullptr;
39 hints.ai_next = nullptr;
40
41 constexpr int portdigits = std::numeric_limits<decltype(port)>::digits10;
42 char portstring[portdigits + 1];
43 snprintf(portstring, portdigits + 1, "%i", port);
44
45 addrinfo* result;
46 const int gairesult = getaddrinfo(nullptr, portstring, &hints, &result);
47 if (gairesult != 0) {
48 throw SocketException(std::string("Unable to get server information: ") +
49 gai_strerror(gairesult));
50 }
51
52 addrinfo* rp;
53 int sfd = -1;
54 for (rp = result; rp != nullptr; rp = rp->ai_next) {
55 sfd = socket(rp->ai_family, rp->ai_socktype, rp->ai_protocol);
56 if (sfd < 0) {
57 continue;
58 }
59
60 if (bind(sfd, rp->ai_addr, rp->ai_addrlen) == 0) {
61 // success!
62 break;
63 }
64
65 close(sfd);
66 }
67
68 freeaddrinfo(result);
69
70 if (rp == nullptr) {
71 throw SocketException("Could not bind");
72 }
73
74 mFD = sfd;
75
76 if (listen(mFD, 5) < 0) {
77 throw SocketException("Error on listen: " + std::string(strerror(errno)));
78 }
79}

138Socket Socket::as_client(const std::string& hostname, int port)
139{
140 addrinfo hints;
141 memset(&hints, 0, sizeof(addrinfo));
142 hints.ai_family = AF_UNSPEC; // Allow IPv4 or IPv6
143 hints.ai_socktype = SOCK_STREAM;
144 hints.ai_flags = 0;
145 hints.ai_protocol = 0; // Any protocol
146
147 constexpr int portdigits = std::numeric_limits<decltype(port)>::digits10;
148 char portstring[portdigits + 1];
149 snprintf(portstring, portdigits + 1, "%i", port);
150
151 addrinfo* result;
152 const int gairesult = getaddrinfo(hostname.c_str(), portstring, &hints, &result);
153 if (gairesult != 0) {
154 throw SocketException(std::string("Unable to get host address: ") +
155 gai_strerror(gairesult));
156 }
157
158 addrinfo* rp;
159 int sfd = -1;
160 for (rp = result; rp != nullptr; rp = rp->ai_next) {
161 sfd = socket(rp->ai_family, rp->ai_socktype, rp->ai_protocol);
162 if (sfd < 0) {
163 continue;
164 }
165
166 if (connect(sfd, rp->ai_addr, rp->ai_addrlen) != -1) {
167 // success!
168 break;
169 }
170
171 close(sfd);
172 }
173
174 freeaddrinfo(result);
175
176 if (rp == nullptr) {
177 throw SocketException("Could not connect to host");
178 }
179
180 Socket ret;
181 ret.mFD = sfd;
182 return ret;
183}

214ssize_t Socket::write(const char* buf, ssize_t count)
215{
216 ssize_t n = 0;
217 while (n < count) {
218 const ssize_t ret = ::write(mFD, buf + n, count - n);
219 if (ret < 0) {
220 if (errno == EINTR) {
221 // Signal interruption; try again.
222 continue;
223 } else if (errno == EWOULDBLOCK || errno == EAGAIN) {
224 // Blocking socket. Wait for it to free up.
225 fd_set writefds;
226 FD_ZERO(&writefds);
227 FD_SET(mFD, &writefds);
228
229 // TODO: set timeout? What do we do if we timeout?
230 if (select(mFD + 1, nullptr, &writefds, nullptr, nullptr) < 0) {
231 if (errno == EINTR) {
232 continue;
233 }
234 return n;
235 }
236 assert(FD_ISSET(mFD, &writefds));
237 } else {
238 return n;
239 }
240 } else {
241 n += ret;
242 }
243 }
244 return n;
245}

247ssize_t Socket::read(char* buf, ssize_t count)
248{
249 ssize_t rc = 0;
250 for ( ; ; ) {
251 const ssize_t ret = ::read(mFD, buf + rc, count - rc);
252 if (ret >= 0) {
253 return ret;
254 } else {
255 if (errno == EINTR) {
256 // Signal interruption; try again.
257 continue;
258 } else if (errno == EWOULDBLOCK || errno == EAGAIN) {
259 // Blocking socket. Wait for it to free up.
260 fd_set readfds;
261 FD_ZERO(&readfds);
262 FD_SET(mFD, &readfds);
263
264 // TODO: set timeout? What do we do if we timeout?
265 if (select(mFD + 1, &readfds, nullptr, nullptr, nullptr) < 0) {
266 if (errno == EINTR) {
267 continue;
268 }
269 return rc;
270 }
271 assert(FD_ISSET(mFD, &readfds));
272 } else {
273 return rc;
274 }
275 }
276 }

319bool SocketStreambuf::write_to_socket()
320{
321 const std::ptrdiff_t count = pptr() - pbase();
322 const ssize_t n = mSock.write(pbase(), count);
323 if (n != count) {
324 return false;
325 }
326 pbump(-n);
327 return true;
328}

330SocketStreambuf::int_type SocketStreambuf::overflow(int_type ch)
331{
332 if (mSock && !traits_type::eq_int_type(ch, traits_type::eof())) {
333 if (!write_to_socket()) {
334 return traits_type::eof();
335 }
336 assert(std::less<char*>()(pptr(), epptr()));
337 traits_type::assign(*pptr(), traits_type::to_char_type(ch));
338 pbump(1);
339 return traits_type::not_eof(ch);
340 }
341 return traits_type::eof();
342}

349SocketStreambuf::int_type SocketStreambuf::underflow()
350{
351 if (gptr() < egptr()) { // Buffer not exhausted.
352 return traits_type::not_eof(*gptr());
353 }
354
355 if (!mSock) {
356 return traits_type::eof();
357 }
358
359 char* const base = mInputBuffer.data();
360 char* start = base;
361
362 if (eback() == base) { // Not our first fill
363 // Move the putback characters to the beginning of the buffer.
364 std::memmove(base, egptr() - sPutBack, sPutBack);
365 start += sPutBack;
366 }
367
368 const std::size_t count = mInputBuffer.size() - (start - base);
369 const ssize_t rc = mSock.read(start, count);
370 if (rc <= 0) {
371 return traits_type::eof();
372 }
373 setg(base, start, start + rc);
374 return traits_type::not_eof(*gptr());
375}
