22const std::string
23getErrorDescription()
24{
25 std::array<char, 1024> errbuf;
26#if (_POSIX_C_SOURCE >= 200112L || _XOPEN_SOURCE >= 600) && ! _GNU_SOURCE
27 std::string e("Unknown error");
28 if (strerror_r(errno, errbuf.data(), errbuf.size()) >= 0) {
29 e = errbuf.data();
30 }
31
32 return e;
33#else
34 // Oddly enough, this function may not use the errbuf data at all, and
35 // if you use it instead of the return value, you may get garbage.
36 return std::string(strerror_r(errno, errbuf.data(), errbuf.size()));
37#endif
38}
