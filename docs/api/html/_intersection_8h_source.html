138 void setTable(scene_rdl2::alloc::Arena *arena, const AttributeTable * const table) {
139 MNRY_ASSERT(arena != nullptr);
140 mValidTableOffset = table->getAttributesSize();
141 mNumKeys = table->getNumKeys();
142 mTable = table;
143 // the memory layout for allocated mData:
144 // | | |
145 // |---------------------------------------|----|
146 // |buffer to store primitive attributes | |
147 // table to mark whether a
148 // specific attribute exists
149 int size = mValidTableOffset + mNumKeys;
150 mData = nullptr;
151 if (size) {
152 mData = (char *)arena->alloc(size, sizeof(void*));
153 memset(mData, 0, size);
154 }
155 }

158 return MNRY_VERIFY(mTable);

161 bool isProvided(AttributeKey key) const {
162 return (key < mNumKeys) && (key >= 0) && (mData != nullptr) &&
163 ((*(mData + mValidTableOffset + key) & ATTRIBUTE_INITIALIZED) != 0);
164 }

250 inline void setAttribute(AttributeKey key, Interpolator &interpolator) {
251 switch (key.getType()) {
252 case scene_rdl2::rdl2::TYPE_BOOL:
253 setAttribute(TypedAttributeKey<bool>(key), interpolator);
254 break;
255 case scene_rdl2::rdl2::TYPE_INT:
256 setAttribute(TypedAttributeKey<int>(key), interpolator);
257 break;
258 case scene_rdl2::rdl2::TYPE_LONG:
259 setAttribute(TypedAttributeKey<long>(key), interpolator);
260 break;
261 case scene_rdl2::rdl2::TYPE_FLOAT:
262 setAttribute(TypedAttributeKey<float>(key), interpolator);
263 break;
264 case scene_rdl2::rdl2::TYPE_STRING:
265 setAttribute(TypedAttributeKey<std::string>(key), interpolator);
266 break;
267 case scene_rdl2::rdl2::TYPE_RGB:
268 setAttribute(TypedAttributeKey<scene_rdl2::math::Color>(key), interpolator);
269 break;
270 case scene_rdl2::rdl2::TYPE_RGBA:
271 setAttribute(TypedAttributeKey<scene_rdl2::math::Color4>(key), interpolator);
272 break;
273 case scene_rdl2::rdl2::TYPE_VEC2F:
274 setAttribute(TypedAttributeKey<scene_rdl2::math::Vec2f>(key), interpolator);
275 break;
276 case scene_rdl2::rdl2::TYPE_VEC3F:
277 setAttribute(TypedAttributeKey<scene_rdl2::math::Vec3f>(key), interpolator);
278 break;
279 case scene_rdl2::rdl2::TYPE_MAT4F:
280 setAttribute(TypedAttributeKey<scene_rdl2::math::Mat4f>(key), interpolator);
281 break;
282 default:
283 MNRY_ASSERT(false, (std::string("unsupported attribute type ") +
284 std::string(attributeTypeName(key.getType())) +
285 std::string(" for atttribute ") + std::string(key.getName())).c_str());
286 break;
287 }

299 inline void fillInstanceAttributes(
300 const InstanceAttributes* instanceAttributes,
301 const std::vector<AttributeKey>& keys) {
302 for (auto key : keys) {
303 if (!instanceAttributes->isSupported(key)) {
304 continue;
305 }
306 switch (key.getType()) {
307 case scene_rdl2::rdl2::TYPE_BOOL:
308 setAttribute(TypedAttributeKey<bool>(key), instanceAttributes);
309 break;
310 case scene_rdl2::rdl2::TYPE_INT:
311 setAttribute(TypedAttributeKey<int>(key), instanceAttributes);
312 break;
313 case scene_rdl2::rdl2::TYPE_LONG:
314 setAttribute(TypedAttributeKey<long>(key), instanceAttributes);
315 break;
316 case scene_rdl2::rdl2::TYPE_FLOAT:
317 setAttribute(TypedAttributeKey<float>(key), instanceAttributes);
318 break;
319 case scene_rdl2::rdl2::TYPE_STRING:
320 setAttribute(TypedAttributeKey<std::string>(key),
321 instanceAttributes);
322 break;
323 case scene_rdl2::rdl2::TYPE_RGB:
324 setAttribute(TypedAttributeKey<scene_rdl2::math::Color>(key),
325 instanceAttributes);
326 break;
327 case scene_rdl2::rdl2::TYPE_RGBA:
328 setAttribute(TypedAttributeKey<scene_rdl2::math::Color4>(key),
329 instanceAttributes);
330 break;
331 case scene_rdl2::rdl2::TYPE_VEC2F:
332 setAttribute(TypedAttributeKey<scene_rdl2::math::Vec2f>(key), instanceAttributes);
333 break;
334 case scene_rdl2::rdl2::TYPE_VEC3F:
335 setAttribute(TypedAttributeKey<scene_rdl2::math::Vec3f>(key), instanceAttributes);
336 break;
337 case scene_rdl2::rdl2::TYPE_MAT4F:
338 setAttribute(TypedAttributeKey<scene_rdl2::math::Mat4f>(key),
339 instanceAttributes);
340 break;
341 default:
342 MNRY_ASSERT(false, (std::string("unsupported attribute type ") +
343 std::string(attributeTypeName(key.getType())) +
344 std::string(" for atttribute ") +
345 std::string(key.getName())).c_str());
346 break;
347 }
348 }
349 }

364 inline void validateRequiredAttributes() {
365 // Check whether all the required attributes from shading network is

469 //
470 // transferAndComputeDerivatives must be called before it's valid to call
471 // this function.
472 //
473 // Sort uvs to gain coherency when adding to the shade queue.
474 // Bits 0-13 quantized swizzled uv coordinates, 14 bits = 16,384 mini-tiles.
475 // Bits 14-17 mip level, lower resolution mips sorted earlier, 4 bits = 16 mip levels.
476 // Bits 18-24 udim tile, lower resolution idx sorted earlier, 7 bits = 128 tiles.
477 // Bits 25-31 lightset index, 7 bits = 128 light sets.
478 //
479 // If we exceed any of the limits in the number of bit allocated for each
480 // category, then we clamp the value rather than wrap it.
481 //
482 finline uint32_t computeShadingSortKey() const;

485 finline scene_rdl2::math::Vec3f getdNdx() const
486 {
487 // mN is already flipped at this point, unflip it if necessary to compute
488 // aux normals
489 float sign = isEntering() ? 1.f : -1.f;
490 scene_rdl2::math::Vec3f auxNormalX = normalize(mN * sign + mdNds * mdSdx + mdNdt * mdTdx);
491 return (auxNormalX * sign) - mN;
492 }
493
494 finline scene_rdl2::math::Vec3f getdNdy() const
495 {
496 // mN is already flipped at this point, unflip it if necessary to compute
497 // aux normals
498 float sign = isEntering() ? 1.f : -1.f;
499 scene_rdl2::math::Vec3f auxNormalY = normalize(mN * sign + mdNds * mdSdy + mdNdt * mdTdy);
500 return (auxNormalY * sign) - mN;
501 }

670finline uint32_t
671Intersection::computeShadingSortKey() const
672{
673 MNRY_ASSERT(mFlags.get(GeomInitialized));
674 MNRY_ASSERT(getLayerAssignmentId() >= 0);
675
676 // Divide up uvs into quads such that there are n quads over the 0-1 domain
677 // in each dimension. Quads are themselves stored in Morton order.
678
679 // n = 128 quads per dimension.
680 uint64_t x = ((uint64_t)(mSt.x * 127.9999f)) & 0x7f;
681 uint64_t y = ((uint64_t)(mSt.y * 127.9999f)) & 0x7f;
682
683 // Interleave bits with 64-bit multiplies.
684 // http://graphics.stanford.edu/~seander/bithacks.html#Interleave64bitOps
685 // This interleaved multiply is what converts the coorindates from linear
686 // to morton ordering.
687 uint32_t uv = ((x * 0x0101010101010101ULL & 0x8040201008040201ULL) *
688 0x0102040810204081ULL >> 49) & 0x5555 |
689 ((y * 0x0101010101010101ULL & 0x8040201008040201ULL) *
690 0x0102040810204081ULL >> 48) & 0xAAAA;
691
692 float mipSelector = computeMipSelector(mdSdx, mdTdx, mdSdy, mdTdy);
693
694 uint32_t ms = (uint32_t)minI32(int(mipSelector), 15);
695
696 // Convert uv coordinates into a udim index.
697 int32_t tileU = clampI32(int(mSt.x), 0, 9);
698 int32_t tileV = maxI32(int(mSt.y), 0);
699 uint32_t udim = (uint32_t)minI32(tileV * 10 + tileU, 127);
700
701 // Entries without light set assignments are placed at the end of the
702 // sorted list.
703 uint32_t ls = (uint32_t)minI32(getLayerAssignmentId(), 127);
704
705 MNRY_ASSERT(uv < 16384);
706 MNRY_ASSERT(ms < 16);
707 MNRY_ASSERT(udim < 128);
708 MNRY_ASSERT(ls < 128);
709
710 return (ls << 25) | (udim << 18) | (ms << 14) | uv;
711}
