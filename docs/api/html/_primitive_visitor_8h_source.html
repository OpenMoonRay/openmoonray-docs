28/// @class PrimitiveVisitor
29/// @remark For realtime rendering use. Feature film shader development
30/// does not require this functionality.
31/// @brief Base class for performing operation on Primitives.
32///
33/// To extend the operation performed on Primitives, extend this base class
34/// and supply your own functionality for each type. If you want to apply
35/// operation in base class (Primitive) level, just override visitPrimitive
36class PrimitiveVisitor
37{
38public:
39 virtual ~PrimitiveVisitor() = default;
40 /// override this method to apply an opeartion to the
41 /// primitives Procedural stores
42 virtual void visitPrimitive(Primitive& p);
43 /// override this method to apply an opeartion to the
44 /// Curves primitives Procedural stores
45 virtual void visitCurves(Curves& c);
46 /// override this method to apply an opeartion to the
47 /// Instance primitives Procedural stores
48 virtual void visitInstance(Instance& i);
49 /// override this method to apply an opeartion to the
50 /// Points primitives Procedural stores
51 virtual void visitPoints(Points& p);
52 /// override this method to apply an opeartion to the
53 /// PolyMesh primitives Procedural stores
54 virtual void visitPolygonMesh(PolygonMesh& p);
55 /// override this method to apply an opeartion to the
56 /// PrimitiveGroup primitives Procedural stores
57 virtual void visitPrimitiveGroup(PrimitiveGroup& s);
58 /// override this method to apply an opeartion to the
59 /// Sphere primitives Procedural stores
60 virtual void visitSphere(Sphere& s);
61 /// override this method to apply an opeartion to the
62 /// Box primitives Procedural stores
63 virtual void visitBox(Box& b);
64 /// override this method to apply an opeartion to the
65 /// SubdivisionMesh primitives Procedural stores
66 virtual void visitSubdivisionMesh(SubdivisionMesh& s);
67 /// override this method to apply an opeartion to the
68 /// TransformedPrimitive primitives Procedural stores
69 virtual void visitTransformedPrimitive(TransformedPrimitive& t);
70 /// override this method to apply an operation to the
71 /// VdbVolume primitives Procedural stores
72 virtual void visitVdbVolume(VdbVolume& v);
73};
