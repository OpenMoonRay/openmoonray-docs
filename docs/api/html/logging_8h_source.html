150// Describes a single logging "event" to be saved in the ObjectLogs class.
151// See a detailed description in the ObjectLogs class.
152typedef int LogEvent;
153
154
155// ObjectLogs is a class for very quickly saving logging "events" that may occur
156// in situations where writing to console or to a file would be too expensive.
157// Instead, LogEvents are created in a LogEventRegistry at some previous time.
158// This will associate a string and a logging level with a LogEvent object (which
159// is simply an integer. This LogEvent is saved away and whenever we need to log
160// that this type of event happened, we call ObjectLogs with that LogEvent.
161// ObjectLogs will maintain a counter of how many times each different LogEvent was
162// logged. This count can be used at a later time (e.g. postFrame) to print actual
163// error messages.
164//
165// Sample usage:
166//
167// LogEventRegistry registry;
168// LogEvent mBadAttributeError = registry.createEvent(scene_rdl2::logging::ERROR_LEVEL, "Bad attribute value");
169// ObjectLogs objLog;
170// registry.initLog(objLog);
171// ...
172// objLog.log(mBadAttributeError);
173// ...
174// registry.report(sceneObject, objLog);
175//
176// shading::logEvent is available in Shading.h for convenient logging within Shaders
177class ObjectLogs
178{
179public:
180 ObjectLogs() {
181 mNumEvents = 0;
182 mEventCounts = nullptr;
183 }
184
185 ~ObjectLogs() {
186 delete [] mEventCounts;
187 }
188
189 // Sets the maximum number of event types that need to be supported by this
190 // object.
191 void setNumEvents(int n);
192
193 // Sets all event counts to 0.
194 void clear() {
195 for (int i = 0; i < mNumEvents; i++) {
196 mEventCounts[i] = 0;
197 }
198 }
199
200 // Records an event.
201 void log(LogEvent event) {
202 mEventCounts[(int)event]++;
203 }
204
205 // Gets a count of how many times log(event) was called for event since
206 // the last clear.
207 int getCount(LogEvent event) const {
208 return mEventCounts[(int)event];
209 }
210
211 // Combines two ObjectLogs objects, summing their event counts
212 ObjectLogs &operator+=(const ObjectLogs & other) {
213 if (mNumEvents != other.mNumEvents) {
214 setNumEvents(other.mNumEvents);
215 }
216 for (int i = 0; i < mNumEvents; i++) {
217 mEventCounts[i] += other.mEventCounts[i];
218 }
219 return *this;
220 }
221
222private:
223 int mNumEvents;
224 int *mEventCounts;
225};

228// Maintains a registry of the types of events that could be logged by an object.
229// This class basically provides a mapping from a LogEvent (which is an int)
230// into a string description and logging level.
231// LogEventRegistry and ObjectLogs should be used in pairs, with LogEventRegistrys
232// maintaining string descriptions of events and ObjectLogs maintaining counts.
233// Typically there will be multiple ObjectLogs associated with the same LogEventRegistry
234// since we want to maintain a separate ObjectLog per thread to avoid synchronization.
235class LogEventRegistry
236{
237 public:
238 LogEventRegistry() {}
239
240 LogEvent createEvent(LogLevel level,
241 std::string eventDescription) {
242
243 if (level == FATAL_LEVEL) {
244 Logger::error("Fatal events are not supported while shading, using error instead");
245 level = ERROR_LEVEL;
246 }
247 std::vector<std::string>::iterator itr = std::find(mDescriptions.begin(),
248 mDescriptions.end(),
249 eventDescription);
250 if (itr != mDescriptions.end()) {
251 return LogEvent(std::distance(mDescriptions.begin(), itr));
252 } else {
253 int n = static_cast<int>(mDescriptions.size());
254 mLevels.push_back(level);
255 mDescriptions.push_back(eventDescription);
256 return LogEvent(n);
257 }
258 }
259
260 // Returns a description of a given LogEvent.
261 const std::string &getDescription(LogEvent event) const {
262 return mDescriptions[(int)event];
263 }
264
265 // Initializes an ObjectLog associated with this registry.
266 // This sets the number of different types of LogEvents this ObjectLog
267 // can expect.
268 void initLog(ObjectLogs &logs) const {
269 logs.setNumEvents(static_cast<int>(mLevels.size()));
270 }
271
272 // Logs each event recorded in log for a object SceneObject i.e.
273 // the name and the type of the SceneObject will be associated with
274 // each log message.
275 void report(const std::string& objectName,
276 const std::string& sceneClassName,
277 const ObjectLogs &log) const;
278
279 // Clears the events and descriptions.
280 void clear() {
281 mLevels.clear();
282 mDescriptions.clear();
283 }
284
285 static void setLoggingGlobalSwitch(bool flag) { mLoggingGlobalSwitch = flag; }
286 static bool getLoggingGlobalSwitch() { return mLoggingGlobalSwitch; }
287
288private:
289 std::vector<LogLevel> mLevels;
290 std::vector<std::string> mDescriptions;
291
292 static std::atomic<bool> mLoggingGlobalSwitch;
293};
