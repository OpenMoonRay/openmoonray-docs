25void
26removeOptionalQuoting(std::string& s)
27{
28 std::size_t len = s.size();
29
30 // Only need further processing for strings that are at least two
31 // characters long.
32 if (len < 2) return;
33
34 // Strip double or single quotes if they match and are not escaped.
35 if (s[0] == '"' && s[len - 1] == '"' && s[len - 2] != '\\') {
36 s.erase(s.begin());
37 s.erase(s.end() - 1);
38 } else if (s[0] == '\'' && s[len - 1] == '\'' && s[len - 2] != '\\') {
39 s.erase(s.begin());
40 s.erase(s.end() - 1);
41 }
42}

44void
45removeOptionalParens(std::string& s)
46{
47 std::size_t len = s.size();
48
49 // Only need further processing for strings that are at least two
50 // characters long.
51 if (len < 2) return;
52
53 // Strip parens if they match.
54 if (s[0] == '(' && s[len - 1] == ')') {
55 s.erase(s.begin());
56 s.erase(s.end() - 1);
57 }
58}

60void
61removeOptionalBrackets(std::string& s)
62{
63 std::size_t len = s.size();
64
65 // Only need further processing for strings that are at least two
66 // characters long.
67 if (len < 2) return;
68
69 // Strip brackets if they match.
70 if (s[0] == '[' && s[len - 1] == ']') {
71 s.erase(s.begin());
72 s.erase(s.end() - 1);
73 }

76template <typename T>
77T
78convertNumericFromString(std::string value)
79{
80 T result;
81 try {
82 result = boost::lexical_cast<T>(value);
83 } catch (boost::bad_lexical_cast&) {
84 std::stringstream errMsg;
85 errMsg << "Could not interpret '" << value << "' as " <<
86 attributeTypeName<T>() << '.';
87 throw except::RuntimeError(errMsg.str());
88 }
89
90 return result;

93// If numTokens == 0, we skip the token count check. We just return as many
94// tokens as we find.
95template <typename T>
96std::vector<std::string>
97tokenizeTuple(std::string value, std::size_t numTokens = 0)
98{
99 std::vector<std::string> tokens;
100 boost::algorithm::split(tokens, value, boost::algorithm::is_any_of(","));
101
102 if (numTokens > 0 && tokens.size() != numTokens) {
103 std::stringstream errMsg;
104 errMsg << "Expected " << numTokens << " components for " <<
105 attributeTypeName<T>() << ": '" << value << "'.";
106 throw except::RuntimeError(errMsg.str());
107 }
108
109 // If we only have one token and it's the empty string, remove it. Boost
110 // always generates at least one token.
111 if (tokens.size() == 1 && tokens[0].empty()) {
112 tokens.erase(tokens.begin());
113 }
114
115 return tokens;
116}

118std::vector<std::string>
119tokenizeGroupedList(std::string value, const char* openGroupChars,
120 const char* closeGroupChars)
121{
122 std::vector<std::string> elements;
123 auto isOpenGroupChar = boost::algorithm::is_any_of(openGroupChars);
124 auto isCloseGroupChar = boost::algorithm::is_any_of(closeGroupChars);
125
126 // Split on all commas.
127 std::vector<std::string> tokens;
128 boost::algorithm::split(tokens, value, boost::algorithm::is_any_of(","));
129
130 // Merge groups of tokens between leading and trailing grouping characters.
131 auto first = tokens.end();
132 auto last = tokens.end();
133 for (auto iter = tokens.begin(); iter != tokens.end(); ++iter) {
134 auto trimmedToken = *iter;
135 boost::algorithm::trim(trimmedToken);
136 std::size_t len = trimmedToken.size();
137
138 if (len > 0) {
139 // Look for a leading grouping character.
140 if (isOpenGroupChar(trimmedToken[0])) {
141 first = iter;
142 }
143
144 // Look for a trailing grouping character.
145 if (isCloseGroupChar(trimmedToken[len - 1])) {
146 last = iter;
147 }
148
149 // If we have both, merge the group of tokens into a single element.
150 if (first != tokens.end() && last != tokens.end()) {
151 std::stringstream merged;
152 bool skipFirst = true;
153 for (auto groupIter = first; groupIter != last + 1; ++groupIter) {
154 if (skipFirst) {
155 skipFirst = false;
156 } else {
157 merged << ',';
158 }
159 merged << *groupIter;
160 }
161 elements.push_back(merged.str());
162 first = tokens.end();
163 last = tokens.end();
164 }
165 }
166 }
167
168 // Handle values without grouping characters and single element lists.
169 if (elements.empty() && !value.empty()) {
170 elements.push_back(value);
171 }
172
173 return elements;
174}

176template <typename T>
177T
178tokensToVector(const std::vector<std::string>& tokens)
179{
180 T vec;
181 for (auto iter = tokens.begin(); iter != tokens.end(); ++iter) {
182 vec.push_back(convertFromString<typename T::value_type>(*iter));
183 }
184 return vec;
185}

310template <>
311Bool convertFromString<Bool>(std::string value)
312{
313 boost::algorithm::trim(value);
314 boost::algorithm::to_lower(value);
315
316 // Try to be as accomodating as possible.
317 if (value == "1" || value == "true" || value == "on" || value == "yes") {
318 return true;
319 } else if (value == "0" || value == "false" || value == "off" || value == "no") {
320 return false;
321 }
322
323 std::stringstream errMsg;
324 errMsg << "Could not interpret '" << value << "' as Bool.";
325 throw except::RuntimeError(errMsg.str());
326}

715std::string
716showAttributeFlags(const AttributeFlags &val)
717{
718 const int i = static_cast<int>(val);
719 std::ostringstream ostr;
720 ostr << "AttributeFlags:0x" << std::hex << i << " { ";
721 if (i == 0x0) {
722 ostr << "NONE ";
723 } else {
724 if (i & FLAGS_BINDABLE) ostr << "BINDABLE ";
725 if (i & FLAGS_BLURRABLE) ostr << "BLURRABLE ";
726 if (i & FLAGS_ENUMERABLE) ostr << "ENUMERABLE ";
727 if (i & FLAGS_FILENAME) ostr << "FILENAME ";
728 if (i & FLAGS_CAN_SKIP_GEOM_RELOAD) ostr << "CAN_SKIP_GEOM_RELOAD ";
729 }
730 ostr << "}";
731 return ostr.str();
732}

755 return "TraceSet";

757 return "Layer";

762 } else if (type & INTERFACE_SHADOWSET) {
763 return "ShadowSet";
764 } else if (type & INTERFACE_LIGHTSET) {
765 return "LightSet";
766 } else if (type & INTERFACE_LIGHTFILTERSET) {

769 return "Map";
