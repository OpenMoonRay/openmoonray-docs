160void
161TestSceneContext::testIterateSceneObjects()
162{
163 SceneContext context;
164 context.createSceneObject("ExampleObject", "/seq/shot/pizza");
165 context.createSceneObject("ExampleObject", "/seq/shot/cookie");
166
167 // The map is unordered, so this is a bit trickier.
168 bool sawSceneVars = false;
169 bool sawPizza = false;
170 bool sawCookie = false;
171
172 SceneContext::SceneObjectConstIterator iter = context.beginSceneObject();
173 if (iter->first == "__SceneVariables__") {
174 CPPUNIT_ASSERT(!sawSceneVars);
175 sawSceneVars = true;
176 CPPUNIT_ASSERT(iter->second->getName() == "__SceneVariables__");
177 } else if (iter->first == "/seq/shot/pizza") {
178 CPPUNIT_ASSERT(!sawPizza);
179 sawPizza = true;
180 CPPUNIT_ASSERT(iter->second->getName() == "/seq/shot/pizza");
181 } else if (iter->first == "/seq/shot/cookie") {
182 CPPUNIT_ASSERT(!sawCookie);
183 sawCookie = true;
184 CPPUNIT_ASSERT(iter->second->getName() == "/seq/shot/cookie");
185 }
186
187 ++iter;
188 if (iter->first == "__SceneVariables__") {
189 CPPUNIT_ASSERT(!sawSceneVars);
190 sawSceneVars = true;
191 CPPUNIT_ASSERT(iter->second->getName() == "__SceneVariables__");
192 } else if (iter->first == "/seq/shot/pizza") {
193 CPPUNIT_ASSERT(!sawPizza);
194 sawPizza = true;
195 CPPUNIT_ASSERT(iter->second->getName() == "/seq/shot/pizza");
196 } else if (iter->first == "/seq/shot/cookie") {
197 CPPUNIT_ASSERT(!sawCookie);
198 sawCookie = true;
199 CPPUNIT_ASSERT(iter->second->getName() == "/seq/shot/cookie");
200 }
201
202 ++iter;
203 if (iter->first == "__SceneVariables__") {
204 CPPUNIT_ASSERT(!sawSceneVars);
205 sawSceneVars = true;
206 CPPUNIT_ASSERT(iter->second->getName() == "__SceneVariables__");
207 } else if (iter->first == "/seq/shot/pizza") {
208 CPPUNIT_ASSERT(!sawPizza);
209 sawPizza = true;
210 CPPUNIT_ASSERT(iter->second->getName() == "/seq/shot/pizza");
211 } else if (iter->first == "/seq/shot/cookie") {
212 CPPUNIT_ASSERT(!sawCookie);
213 sawCookie = true;
214 CPPUNIT_ASSERT(iter->second->getName() == "/seq/shot/cookie");
215 }
216
217 ++iter;
218 CPPUNIT_ASSERT(iter == context.endSceneObject());
219
220 CPPUNIT_ASSERT(sawSceneVars);
221 CPPUNIT_ASSERT(sawPizza);
222 CPPUNIT_ASSERT(sawCookie);
223}

331void
332TestSceneContext::testSceneVariables()
333{
334 SceneContext context;
335 const SceneContext& constContext = context;
336
337 // Check some default values.
338 CPPUNIT_ASSERT(context.getSceneVariables().get(SceneVariables::sWarningKey) == true);
339 CPPUNIT_ASSERT(context.getSceneVariables().get(SceneVariables::sInteractiveKey) == false);
340 CPPUNIT_ASSERT(constContext.getSceneVariables().get(SceneVariables::sWarningKey) == true);
341 CPPUNIT_ASSERT(constContext.getSceneVariables().get(SceneVariables::sInteractiveKey) == false);
342 CPPUNIT_ASSERT(constContext.getSceneVariables().get(SceneVariables::sCheckpointActive) == false);
343 CPPUNIT_ASSERT(constContext.getSceneVariables().get(SceneVariables::sCheckpointInterval) == 15.0f);
344 CPPUNIT_ASSERT(constContext.getSceneVariables().get(SceneVariables::sCheckpointTimeCap) == 0.0f);
345 CPPUNIT_ASSERT(constContext.getSceneVariables().get(SceneVariables::sResumableOutput) == false);
346 CPPUNIT_ASSERT(constContext.getSceneVariables().get(SceneVariables::sResumeRender) == false);
347
348 // Try changing them.
349 context.getSceneVariables().beginUpdate();
350 CPPUNIT_ASSERT_NO_THROW(
351 context.getSceneVariables().set(SceneVariables::sWarningKey, false);
352 context.getSceneVariables().set(SceneVariables::sInteractiveKey, true);
353 context.getSceneVariables().set(SceneVariables::sCheckpointActive, true);
354 context.getSceneVariables().set(SceneVariables::sCheckpointInterval, 5.0f);
355 context.getSceneVariables().set(SceneVariables::sCheckpointTimeCap, 1.0f);
356 context.getSceneVariables().set(SceneVariables::sResumableOutput, true);
357 context.getSceneVariables().set(SceneVariables::sResumeRender, true);
358 );
359 context.getSceneVariables().endUpdate();
360
361 // Check that they changed.
362 CPPUNIT_ASSERT(context.getSceneVariables().get(SceneVariables::sWarningKey) == false);
363 CPPUNIT_ASSERT(context.getSceneVariables().get(SceneVariables::sInteractiveKey) == true);
364 CPPUNIT_ASSERT(constContext.getSceneVariables().get(SceneVariables::sWarningKey) == false);
365 CPPUNIT_ASSERT(constContext.getSceneVariables().get(SceneVariables::sInteractiveKey) == true);
366 CPPUNIT_ASSERT(constContext.getSceneVariables().get(SceneVariables::sCheckpointActive) == true);
367 CPPUNIT_ASSERT(constContext.getSceneVariables().get(SceneVariables::sCheckpointInterval) == 5.0f);
368 CPPUNIT_ASSERT(constContext.getSceneVariables().get(SceneVariables::sCheckpointTimeCap) == 1.0f);
369 CPPUNIT_ASSERT(constContext.getSceneVariables().get(SceneVariables::sResumableOutput) == true);
370 CPPUNIT_ASSERT(constContext.getSceneVariables().get(SceneVariables::sResumeRender) == true);
371
372 // The SceneVariables object is a singleton, and we can't create another.
373 SceneVariables* vars = static_cast<SceneVariables*>(
374 context.createSceneObject("SceneVariables", "MoreVariables"));
375 CPPUNIT_ASSERT(vars == &(context.getSceneVariables()));
376}

394
