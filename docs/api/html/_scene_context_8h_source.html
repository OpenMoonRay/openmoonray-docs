376 GeometryVector mGeometries;

378
379 // We also need a mutex to protect them, since the two SceneObjects could
380 // be updated concurrently in different threads. HOWEVER, this does NOT
381 // protect against the value of mTimeRescalingCoeffs being *read* while
382 // it's being updated. In other words, doing an interpolated get() against
383 // ANY object while the camera's shutter interval or SceneVariables motion
384 // steps are being updated is a recipe for threading errors.
385 tbb::mutex mTimeRescalingCoeffsMutex;
386
