31inline NPoint<D> sampleWithinStratum(const NPoint<D>& min, const NPoint<D>& max, RNG& rng)
32{
33 NPoint<D> r;
34 for (unsigned d = 0; d < D; ++d) {
35 do {
36 r[d] = lerp(canonical(rng), min[d], max[d]);
37 } while (r[d] >= max[d]);
38 assert(r[d] >= 0.0f);
39 assert(r[d] < 1.0f);
40 }
41 return r;

89 if (size == 0) {

92 typedef std::uniform_int_distribution<std::size_t> int_distr_t;
93 typedef int_distr_t::param_type int_param_t;
94
95 int_distr_t intDist;
96
97 const float tau = (1.0f + std::sqrt(5.0f)) / 2.0f;
98 const float phi = (2.0f * pi) * (1.0f - 1.0f/tau);
99
100 const std::size_t kNumFibPoints = 30;
101 const std::size_t j = intDist(rng, int_param_t(0, kNumFibPoints - 1u));
102
103 float rj = std::sqrt(static_cast<float>(j)/static_cast<float>(kNumFibPoints));
104 rj += 1.0f;
105 rj /= 4.0f; // Map to [0, 0.5]
106 const float thetaj = j * phi;
107
108 float x;
109 float y;
110 polarToCartesian(rj, thetaj, x, y);
111
112 // Pick a random point
113 NPoint<2> p = mGenPoints[intDist(rng, int_param_t(0, size - 1u))];
114 p += NPoint<2>({x, y});
115
116 // Wrap point
117 while (p[0] < 0.0f) {
118 p[0] += 1.0f;
119 }
120 while (p[1] < 0.0f) {
121 p[1] += 1.0f;
122 }
123 while (p[0] >= 1.0f) {
124 p[0] -= 1.0f;
125 }
126 while (p[1] >= 1.0f) {
127 p[1] -= 1.0f;
128 }

148template <unsigned D, typename RNG, typename CandidateGenerator, typename DistanceFunction = WrappedDistanceSquared<D>>
149NPoint<D> bestCandidatePoint(const DynamicGrid& grid, unsigned nCandidates, RNG& rng, CandidateGenerator candidateGenerator, DistanceFunction distance)
150{
151 typedef NPoint<D> Point;
152
153 static std::vector<Point> otherPoints;
154
155 Point bestCandidate;
156 float maxDist2 = 0.0f;
157 for (unsigned k = 0; k < nCandidates; ++k) {
158 const Point candidate = candidateGenerator(rng);
159
160 otherPoints.clear();
161 auto f = [/*&otherPoints*/](const Point& p) { otherPoints.push_back(p); };
162 grid.visitNeighbors(f, candidate);
163 if (otherPoints.empty()) {
164 grid.visitAll(f);
165 }
166
167 float sampleDist2 = std::numeric_limits<float>::max();
168 for (const auto& otherPoint : otherPoints) {
169 const float d2 = distance(candidate, otherPoint);
170 sampleDist2 = std::min(d2, sampleDist2);
171 }
172
173 if (sampleDist2 > maxDist2) {
174 bestCandidate = candidate;
175 maxDist2 = sampleDist2;
176 }
177 }
178
179 return bestCandidate;
180}
