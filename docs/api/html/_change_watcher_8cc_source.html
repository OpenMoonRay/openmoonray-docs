23/*
24 * You would think that we want to just set up an inotify watcher on the file
25 * in question, watch for IN_MODIFY events, and call it a day. Oh, how simple
26 * and lovely that would be...
27 *
28 * However, many text editors that use some form of swap files (vim, emacs,
29 * gedit, etc) do not just modify the file in place. They may end up doing
30 * other operations (like moves) between the swap file and the actual file.
31 *
32 * Unfortunately, if the originally watched file is moved, the watcher is
33 * automatically removed. This means you'll get a single notification on first
34 * edit, followed by nothing for subsequent edits. You can't just add the
35 * watcher again each time, because you have no simple way of efficiently
36 * synchronizing with the editor to know when that file will reappear. Ugh.
37 *
38 * The solution to this fantastic little problem comes to us from deep in the
39 * Gentoo forums. Rather than watching the file itself, watch the directory
40 * the file is in. You can filter notifications for that file by looking at
41 * the "name" member of the inotify_event struct. This way, you'll get
42 * notifications for that file whether it's modified in place, moved from a
43 * swap file, or teleported to Mars and back. Well, maybe not that last one.
44 */

46ChangeWatcher::ChangeWatcher() :
47 mNotifyDesc(-1)
48{
49 // Initialize inotify in non-blocking mode.
50 mNotifyDesc = inotify_init1(IN_NONBLOCK);
51 if (mNotifyDesc < 0) {
52 throw scene_rdl2::except::RuntimeError(scene_rdl2::util::buildString(
53 "Failed to initialize inotify for file watching: ",
54 std::strerror(errno)));
55 }
56
57 // Allocate some space for the read buffer. We'll allocate enough for one
58 // event struct, plus the name string at the end of it. If we need more
59 // space for the name, we'll reallocate this later.
60 mReadBuffer.resize(sizeof(inotify_event) + 1024, 0);
61}
62
63ChangeWatcher::~ChangeWatcher()
64{
65 // Remove all watchers by iterating over unique keys in the multimap.
66 for (auto iter = mFileFilter.begin(); iter != mFileFilter.end();
67 iter = mFileFilter.upper_bound(iter->first)) {
68 inotify_rm_watch(mNotifyDesc, iter->first);
69 }
70
71 // Close the notification descriptor.
72 if (mNotifyDesc >= 0) {
73 close(mNotifyDesc);
74 }
75}

77void
78ChangeWatcher::watchFile(const std::string& filePath)
79{
80 // Different text editors perform different routines to save a file. These
81 // routines generate multiple inotify events. We watch for IN_CLOSE_WRITE
82 // events because all editors that were tested (vim, gedit, emacs, nano,
83 // gvim, Nedit, kWrite, Houdini) generate a single IN_CLOSE_WRITE event on the
84 // original file near the end of the routine. We watch for IN_MOVED_TO
85 // because sed -i generates that event on the original file.
86 // Gedit performs both and IN_CLOSE_WRITE and a IN_MOVED_TO on the original
87 // file so there is a very slim chance Gedit may detect 2 scene changes from
88 // one edit.
89
90 // Split the file path into the basename and dirname.
91 auto components = scene_rdl2::util::splitPath(filePath);
92 const auto& dirName = components.first;
93 const auto& fileName = components.second;
94
95 // Add the watcher to the directory. If it already exists, we'll get the
96 // same watch descriptor back.
97 int watchDesc = inotify_add_watch(mNotifyDesc, dirName.c_str(), IN_CLOSE_WRITE | IN_MOVED_TO);
98 if (watchDesc < 0) {
99 throw scene_rdl2::except::RuntimeError(scene_rdl2::util::buildString(
100 "Failed to add file watcher for '", filePath, "': ",
101 std::strerror(errno)));
102 }
103
104 // Find the entry in the file filter corresponding to this watcher.
105 auto range = mFileFilter.equal_range(watchDesc);
106 if (range.first == mFileFilter.end() || std::none_of(range.first, range.second,
107 [&fileName](const std::pair<int, std::pair<std::string, std::string> >& item) { return item.second.first == fileName; })) {
108 // Either this is the first entry, or it doesn't exist yet, so add it.
109 mFileFilter.insert(std::make_pair(watchDesc, std::make_pair(fileName, dirName)));
110 }
111}

113bool
114ChangeWatcher::hasChanged(std::set<std::string> * const changedFiles)
115{
116 // We read any notifications out of the buffer, because we want to swallow
117 // them. But we only indicate a change has occurred if the event came from
118 // a watched file.
119 bool changeOccurred = false;
120 ssize_t readLen = 0;
121
122 // We set up inotify in non-blocking mode, so this read should never block.
123tryread:
124 readLen = read(mNotifyDesc, &mReadBuffer[0], mReadBuffer.size());
125 if (readLen < 0) {
126 if (errno == EAGAIN) {
127 // No notification, so nothing has changed since the last read().
128 // Something might have changed on the previous iteration though.
129 return changeOccurred;
130 } else if (errno == EINVAL) {
131 // Buffer too small, double it and try again.
132 mReadBuffer.resize(mReadBuffer.size() * 2, 0);
133 goto tryread;
134 } else {
135 throw scene_rdl2::except::RuntimeError(scene_rdl2::util::buildString(
136 "Failed to read from file watcher descriptor: ",
137 std::strerror(errno)));
138 }
139 } else if (readLen > 0) {
140 int offset = 0;
141 while (offset < readLen) {
142 // Read the next inotify event out of the buffer.
143 inotify_event* event =
144 reinterpret_cast<inotify_event*>(&mReadBuffer[offset]);
145
146 // Was there a filename attached to the event?
147 if (event->len > 0) {
148 // Get the range of files for this watch descriptor (directory).
149 auto range = mFileFilter.equal_range(event->wd);
150 if (range.first == mFileFilter.end()) {
151 // For some reason we got an event for watch descriptor we
152 // didn't keep a record of... that's most certainly a logic
153 // error.
154 throw scene_rdl2::except::RuntimeError("Received an inotify notification"
155 " for an untracked watch descriptor.");
156 }
157
158 // Does this notification match any of our watched files and path?
159 auto it = std::find_if(range.first, range.second,
160 [event](const std::pair<int, std::pair<std::string, std::string> >& item) {
161 return item.second.first == event->name;
162 });
163 // A match was found, now add it to the list
164 if (it != range.second) {
165 if (changedFiles) {
166 changedFiles->insert(it->second.second + '/' + it->second.first);
167 }
168 changeOccurred = true;
169 }
170 }
171
172 // Advance to the next event in the buffer.
173 offset += sizeof(inotify_event) + event->len;
174 }
175
176 // Wait here for 0.1 second for any further notifications to appear,
177 // then try reading again. Keep reading until no more notifications.
178 usleep(100000);
179 goto tryread;
180 }
181
182 return changeOccurred;
183}

185void
186ChangeWatcher::waitForChange()
187{
188 struct pollfd fds[1];
189 fds[0].fd = mNotifyDesc;
190 fds[0].events = POLLIN | POLLPRI;
191
192 do {
193 // Block until file descriptor is ready to read from.
194 if (poll(fds, 1, -1) < 0) {
195 throw scene_rdl2::except::RuntimeError(scene_rdl2::util::buildString(
196 "Polling on file watcher failed: ", std::strerror(errno)));
197 }
198 } while (!hasChanged());
199}
