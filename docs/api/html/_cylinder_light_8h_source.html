85 // Compute local space position from uv cylinder parameterization.
86 finline scene_rdl2::math::Vec3f uv2local(const scene_rdl2::math::Vec2f &uv) const
87 {
88 float phi = uv.y * scene_rdl2::math::sTwoPi;
89 float sinPhi, cosPhi;
90 scene_rdl2::math::sincos(phi, &sinPhi, &cosPhi);
91
92 return scene_rdl2::math::Vec3f(mLocalRadius * cosPhi,
93 (uv.x - mUvOffset.x) * scene_rdl2::math::rcp(mUvScale.x),
94 mLocalRadius * sinPhi);
95 }
96
97 // Compute uv cylinder parameterization from local space position.
98 finline scene_rdl2::math::Vec2f local2uv(const scene_rdl2::math::Vec3f &pos) const
99 {
100 float u = pos.y * mUvScale.x + mUvOffset.x;
101
102 float phi = scene_rdl2::math::atan2(pos.z, pos.x);
103 float v = (phi < 0.0f ? phi + scene_rdl2::math::sTwoPi : phi) * scene_rdl2::math::sOneOverTwoPi;
104
105 MNRY_ASSERT(finite(u));
106 MNRY_ASSERT(finite(v));
107
108 return scene_rdl2::math::Vec2f(u, v);
109 }
