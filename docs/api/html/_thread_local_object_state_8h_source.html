25// threads write to adjacent ThreadLocalObjectStates in RDL objects

27class CACHE_ALIGN ThreadLocalObjectState

29 public:

31 void *memptr;
32 if (auto err = posix_memalign(&memptr, 64, sizeof(ThreadLocalObjectState)*num)) {
33 throw std::bad_alloc();
34 }
35 ThreadLocalObjectState *result = static_cast<ThreadLocalObjectState *>(memptr);
36 {
37 for (int i = 0; i < num; i++) {
