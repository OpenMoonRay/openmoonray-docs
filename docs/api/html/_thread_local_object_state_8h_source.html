16///
17/// State that is local to both the thread and the object. There
18/// is one of these per-thread per-rdl object. Writing object-specific
19/// data to a ThreadLocalObjectState is always thread-safe.
20///
21/// A ThreadLocalState object (see below) contains a vector of these.
22///
23
24// This is cache-line aligned so as to avoid false sharing when multiple
25// threads write to adjacent ThreadLocalObjectStates in RDL objects
26
27class CACHE_ALIGN ThreadLocalObjectState
28{
29 public:
30 static ThreadLocalObjectState *alignedAlloc(int num) {
31 void *memptr;
32 if (auto err = posix_memalign(&memptr, 64, sizeof(ThreadLocalObjectState)*num)) {
33 throw std::bad_alloc();
34 }
35 ThreadLocalObjectState *result = static_cast<ThreadLocalObjectState *>(memptr);
36 {
37 for (int i = 0; i < num; i++) {
38 new (result + i) ThreadLocalObjectState;
39 }
40 }
41 return result;
42 }
43
44 static void deallocate(int num, ThreadLocalObjectState *tlos) {
45 if (tlos == nullptr) return;
46 for (int i = 0; i < num; i++) {
47 tlos[i].~ThreadLocalObjectState();
48 }
49 free(tlos);
50 }
51
52 scene_rdl2::logging::ObjectLogs mLogs;
53 moonray::util::InclusiveExclusiveAverage<int64> mShaderCallStat;
54
55 void clear() {
56 mLogs.clear();
57 mShaderCallStat.reset();
58 }
59};
