32float
33BsdfSlice::computeShadowTerminatorFix(const scene_rdl2::math::Vec3f &N, const scene_rdl2::math::Vec3f &wi) const
34{
35
36 float G;
37
38 switch (mShadowTerminatorFix) {
39 case ispc::SHADOW_TERMINATOR_FIX_OFF:
40 G = 1.0f;
41 break;
42 case ispc::SHADOW_TERMINATOR_FIX_CUSTOM:
43 {
44 // Custom, targeted shadow terminator that only softens the
45 // hard shadow edge and does not affect any other highlights
46 // Thresholding cosine in a smoothstep range [0, sin(0.15)] for cosX->0
47 // and [0, sin(0.1)] for cosY->1. If this is still not soft enough, we can
48 // potentially expose this threshold as a user-option or widen it.
49 // sin(0.15) = 0.14944f;
50 // sin(0.02) = 0.01999f;
51 const float cosX = scene_rdl2::math::clamp(scene_rdl2::math::dot(N, mNg));
52 const float d = scene_rdl2::math::lerp(0.14922f, 0.01999f, cosX);
53 const float t = scene_rdl2::math::clamp(scene_rdl2::math::dot(mNg, wi)/d);
54 // Cubic Hermite Interpolation
55 G = t*t*(3.0f - 2.0f*t);
56 }
57 break;
58 case ispc::SHADOW_TERMINATOR_FIX_SINE_COMPENSATION:
59 // Same as Chiang's method, but replaces cosine(angle(Ng, N)) with sine(angle(Ng, N)).
60 // The basic idea is to penalize small deviations between the geometric
61 // and shading normals less than in Chiang.
62 G = scene_rdl2::math::clamp(scene_rdl2::math::min(1.0f, scene_rdl2::math::dot(mNg, wi) /
63 (std::max(scene_rdl2::math::sEpsilon, scene_rdl2::math::dot(N, wi)) *
64 std::max(scene_rdl2::math::sEpsilon, scene_rdl2::math::length(scene_rdl2::math::cross(mNg, N))))));
65 G = -(G * G * G) + G * G + G;
66 break;
67 case ispc::SHADOW_TERMINATOR_FIX_GGX:
68 // "A Microfacet-Based Shadowing Function to Solve The Bump Terminator Problem"
69 // Conty Estevez, Lecocq, Stein Ray-Tracing Gems 2019
70 {
71 const float cosD = scene_rdl2::math::min(scene_rdl2::math::abs(scene_rdl2::math::dot(mNg, N)), 1.0f);
72 const float tan2D = (1.0f - cosD * cosD) / (cosD * cosD);
73 const float alpha2 = scene_rdl2::math::clamp(0.125f * tan2D);
74
75 const float cosI = scene_rdl2::math::max(scene_rdl2::math::abs(scene_rdl2::math::dot(mNg, wi)), scene_rdl2::math::sEpsilon);
76 const float tan2I = (1.0f - cosI * cosI) / (cosI * cosI);
77 G = 2.0f / (1 + scene_rdl2::math::sqrt(1.0 + alpha2 * tan2I));
78 }
79 break;
80 case ispc::SHADOW_TERMINATOR_FIX_COSINE_COMPENSATION:
81 // "Taming the Shadow Terminator" Chiang, Li, Burley SIGGRAPH 2019
82 G = scene_rdl2::math::clamp(scene_rdl2::math::min(1.0f, scene_rdl2::math::dot(mNg, wi) /
83 (std::max(scene_rdl2::math::sEpsilon, scene_rdl2::math::dot(N, wi)) *
84 std::max(scene_rdl2::math::sEpsilon, scene_rdl2::math::dot(mNg, N)))));
85 G = -(G * G * G) + G * G + G;
86 break;
87 case ispc::SHADOW_TERMINATOR_FIX_NUM_MODES:
88 default:
89 MNRY_ASSERT(0 && "unknown shadow terminator fix type");
90 G = 1.0f;
91 break;
92 }
93
94 return G;
95}
