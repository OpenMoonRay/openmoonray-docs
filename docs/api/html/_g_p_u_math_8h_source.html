170inline __host__ __device__
171void makeCoordSystem(const float3& v1, float3* v2, float3* v3)
172{
173 // v1 must be normalized
174 if (fabs(v1.x) > fabs(v1.y)) {
175 // cheaper than full normalize as we know one component is zero
176 float s = rsqrtf(v1.x * v1.x + v1.z * v1.z);
177 *v2 = {-v1.z * s, 0.f, v1.x * s};
178 } else {
179 float s = rsqrtf(v1.y * v1.y + v1.z * v1.z);
180 *v2 = {0.f, v1.z * s, -v1.y * s};
181 }
182
183 *v3 = cross(v1, *v2);
184}

275// https://en.wikipedia.org/wiki/B%C3%A9zier_curve
276inline __host__ __device__
277float4 getBezierBasis(const float t)
278{
279 const float t1 = 1.f - t;
280 return {t1 * t1 * t1,
281 3.f * t1 * t1 * t,
282 3.f * t1 * t * t,
283 t * t * t};
284}
285
286inline __host__ __device__
287float4 getBezierBasisDerivative(const float t)
288{
289 const float t1 = 1.f - t;
290 return {-3.f * t1 * t1,
291 -6.f * t1 * t + 3.f * t1 * t1,
292 -3.f * t * t + 6.f * t1 * t,
293 3.f * t * t};
294}
295
296inline __host__ __device__
297float4 evalBezier(const float4 cp[4], const float t)
298{
299 float4 basis = getBezierBasis(t);
300 return basis.x * cp[0] + basis.y * cp[1] + basis.z * cp[2] + basis.w * cp[3];
301}
302
303inline __host__ __device__
304float3 evalBezierDerivative(const float4 cp[4], const float t)
305{
306 float4 basis = getBezierBasisDerivative(t);
307 return make_float3(basis.x * cp[0] + basis.y * cp[1] + basis.z * cp[2] + basis.w * cp[3]);
308}

310// From embree bspline_curve.h
311inline __host__ __device__
312float4 getBsplineBasis(const float t)
313{
314 const float t1 = 1.0f - t;
315 return {(1.f / 6.f) * t1 * t1 * t1,
316 (1.f / 6.f) * ((4.0f*(t1*t1*t1) + (t*t*t)) + (12.0f*((t1*t)*t1) + 6.0f*((t*t1)*t))),
317 (1.f / 6.f) * ((4.0f*(t*t*t) + (t1*t1*t1)) + (12.0f*((t*t1)*t) + 6.0f*((t1*t)*t1))),
318 (1.f / 6.f) * t * t * t};
319}
320
321inline __host__ __device__
322float4 getBsplineBasisDerivative(const float t)
323{
324 const float t1 = 1.0f - t;
325 return {0.5f * -t1 * t1,
326 0.5f * (-t * t - 4.0f * (t * t1)),
327 0.5f * (t1 * t1 + 4.0f * (t1 * t)),
328 0.5f * (t * t)};
329}
330
331inline __host__ __device__
332float4 evalBspline(const float4 cp[4], const float t)
333{
334 float4 basis = getBsplineBasis(t);
335 return basis.x * cp[0] + basis.y * cp[1] + basis.z * cp[2] + basis.w * cp[3];
336}
337
338inline __host__ __device__
339float3 evalBsplineDerivative(const float4 cp[4], const float t)
340{
341 float4 basis = getBsplineBasisDerivative(t);
342 return make_float3(basis.x * cp[0] + basis.y * cp[1] + basis.z * cp[2] + basis.w * cp[3]);

354 __host__ __device__
355 static GPUXform identityXform()
356 {
357 return {1.f, 0.f, 0.f, 0.f,

362 __host__ __device__
363 static GPUXform rotateToZAxisXform(const float3& dir)
364 {
365 // transform to space where dir.z is the +Z axis
366 float3 vz = normalize(dir);
367 float3 vx, vy;
368 makeCoordSystem(vz, &vx, &vy);
369 // form the matrix:
370 // [ vx.x, vy.x, vz.x, 0 ]
371 // [ vx.y, vy.y, vz.y, 0 ]
372 // [ vx.z, vy.z, vz.z, 0 ]
373 // take the transpose to invert it
374 return {vx.x, vx.y, vx.z, 0.f,
375 vy.x, vy.y, vy.z, 0.f,
376 vz.x, vz.y, vz.z, 0.f};
377 }
378
379 __host__ __device__
380 float3 transformVector(const float3& v) const
381 {
382 return {m[0][0] * v.x + m[0][1] * v.y + m[0][2] * v.z,
383 m[1][0] * v.x + m[1][1] * v.y + m[1][2] * v.z,
384 m[2][0] * v.x + m[2][1] * v.y + m[2][2] * v.z};
385 }
386
387 __host__ __device__
388 float3 transformPoint(const float3& p) const
389 {
390 return {m[0][0] * p.x + m[0][1] * p.y + m[0][2] * p.z + m[0][3],
391 m[1][0] * p.x + m[1][1] * p.y + m[1][2] * p.z + m[1][3],
392 m[2][0] * p.x + m[2][1] * p.y + m[2][2] * p.z + m[2][3]};
393 }
394
395 __host__ __device__
396 OptixAabb transformAabb(const OptixAabb& bb) const
397 {
398 // TODO: Not the most efficient way to do this, but the easiest
399 // see Graphics Gems pg 548 for a faster way. This slow way is what we do in
400 // CPU-side math::transformBounds()...
401 OptixAabb result = emptyAabb();
402 result = expandAabb(result, transformPoint({bb.minX, bb.minY, bb.minZ}));
403 result = expandAabb(result, transformPoint({bb.minX, bb.minY, bb.maxZ}));
404 result = expandAabb(result, transformPoint({bb.minX, bb.maxY, bb.minZ}));
405 result = expandAabb(result, transformPoint({bb.minX, bb.maxY, bb.maxZ}));
406 result = expandAabb(result, transformPoint({bb.maxX, bb.minY, bb.minZ}));
407 result = expandAabb(result, transformPoint({bb.maxX, bb.minY, bb.maxZ}));
408 result = expandAabb(result, transformPoint({bb.maxX, bb.maxY, bb.minZ}));
409 result = expandAabb(result, transformPoint({bb.maxX, bb.maxY, bb.maxZ}));
410 return result;
411 }

413 __host__ __device__
414 void toOptixTransform(float transform[12]) const
415 {
416 transform[0] = m[0][0]; transform[1] = m[0][1]; transform[2] = m[0][2]; transform[3] = m[0][3];
417 transform[4] = m[1][0]; transform[5] = m[1][1]; transform[6] = m[1][2]; transform[7] = m[1][3];
418 transform[8] = m[2][0]; transform[9] = m[2][1]; transform[10] = m[2][2]; transform[11] = m[2][3];
419 }
420
421 float m[3][4];
422 // This is an affine 4x4 transformation matrix so the last row is always 0 0 0 1
423 // and we don't need to store it. This convention is opposite to what we use
424 // in scene_rdl2 where an affine matrix has the translation in the last row and
425 // 0 0 0 1 in the last column. We use a different convention than scene_rdl2 to match
426 // up with what Optix is expecting.
427};
