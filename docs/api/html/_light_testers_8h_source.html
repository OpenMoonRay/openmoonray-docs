28//
29// The general approach to testing lights is to write a simple version of each
30// light, one per real light, which does simple uniform sampling over the entire
31// surface area. This will give a reference value which the optimized lights
32// will need to match for them to pass the tests. For cases where we are texture
33// mapping the lights, the approach still works. In this case the test version
34// of each light will do the uniform sampling as before but will lookup the
35// texture at the sample point instead of returning a constant color.
36// The associated pdf is still uniform.
37//
38// The various light testers have intimate knowledge of the light types they are
39// testing. In fact, each real light type declares the corresponding light
40// tester as a friend. This is done to avoid reimplementing the entire
41// functionality from scratch, e.g. we can reuse the cached data and various
42// transform in each light. The main things tested here are that the importance
43// sampling functions are unbiased, and the sampling and intersection functions
44// are consistent with each other.
45//
46
47class LightTester
48{
49public:
50 struct Intersection
51 {
52 scene_rdl2::math::Vec3f mIllumPoint; // original point we want to illuminate
53 scene_rdl2::math::Vec3f mWi; // direction from point to light
54 scene_rdl2::math::Vec3f mHitPoint; // hit point on light surface
55 scene_rdl2::math::Vec3f mNormal; // normal at hit point on light surface
56 float mDistance; // distance traveled to get to mHitPoint
57
58 void init(const scene_rdl2::math::Vec3f &illumPoint, const scene_rdl2::math::Vec3f &hitPoint, const scene_rdl2::math::Vec3f &normal)
59 {
60 mIllumPoint = illumPoint;
61 mWi = hitPoint - illumPoint;
62 mHitPoint = hitPoint;
63 mNormal = normal;
64 mDistance = mWi.length();
65
66 if (mDistance > 0.f) {
67 mWi /= mDistance;
68 }
69 }
70 };
71
72 LightTester(std::shared_ptr<const Light> light, const char *desc,
73 bool infinite, unsigned isectDataFieldsUsed) :
74 mLight(light),
75 mDesc(desc),
76 mInfinite(infinite),
77 mIsectDataFieldsUsed(isectDataFieldsUsed) {}
78
79 virtual ~LightTester() {}
80
81 const char *getDesc() const { return mDesc;}
82 bool isInfinite() const { return mInfinite; }
83 unsigned getIsectDataFieldsUsed() const { return mIsectDataFieldsUsed;}
84
85 // Returns the total world space area of the light.
86 // Used by TestPDF.
87 float getSurfaceArea() const
88 {
89 return getLight()->mArea;
90 }
91
92 // Returns the world space surface area on the light which is visible to point p.
93 // Used by TestPDF.
94 virtual float getVisibleSurfaceArea(const scene_rdl2::math::Vec3f &p, const scene_rdl2::math::Vec3f &n) const
95 {
96 return getLight()->mArea;
97 }
98
101 virtual bool sample(const scene_rdl2::math::Vec3f &p, float r0, float r1, Intersection *isect) const
102 {
103 return false;
104 }
105
106 scene_rdl2::math::Color eval(const Intersection &isect, float *pdf) const
107 {
108 MNRY_ASSERT(mLight && pdf);
109
110 // Area to solid angle conversion if needed
111 *pdf = mInfinite ? 1.0f :
112 absAreaToSolidAngleScale(isect.mWi, isect.mNormal, isect.mDistance);
113 *pdf /= getSurfaceArea();
114
115 bool backSided = mInfinite ? false :
116 dot(isect.mWi, isect.mNormal) > 0.f;
117 scene_rdl2::math::Color result = scene_rdl2::math::zero;
118
119 if (!backSided && *pdf != 0.f) {
120 // call into the real lights eval function to retrieve the textured
121 // color value at this intersection point, but keep the pdf we've
122 // already computed
123 LightIntersection lightIsect;
124 LightFilterRandomValues filterR = {scene_rdl2::math::Vec2f(0.f, 0.f), scene_rdl2::math::Vec3f(0.f, 0.f, 0.f)};
125 if (getLight()->intersect(isect.mIllumPoint, nullptr, isect.mWi, 0.f, isect.mDistance + 1.f, lightIsect)) {
126 result = getLight()->eval(nullptr, isect.mWi, isect.mIllumPoint, filterR, 0.f, lightIsect, false,
127 nullptr, 0.0f);
128 }
129 }
130
131 return result;
132 }
133
134 virtual const Light *getLight() const { return mLight.get(); }
135 virtual const char *getLightTypeName() const = 0;
136
137protected:
138 std::shared_ptr<const Light> mLight;
139 const char *mDesc;
140 bool mInfinite;
142 unsigned mIsectDataFieldsUsed;
143};

147class RectLightTester : public LightTester
148{
149public:
150 RectLightTester(std::shared_ptr<const RectLight> light, const char *desc) :
151 LightTester(std::static_pointer_cast<const Light>(light), desc, false, 0) {}
152
153 // simple uniform sampling
154 virtual bool sample(const scene_rdl2::math::Vec3f &p, float r0, float r1, Intersection *isect) const override
155 {
156 const RectLight *light = getLight();
157
158 float halfWidth = light->mHalfWidth;
159 float halfHeight = light->mHalfHeight;
160
161 scene_rdl2::math::Vec3f hit;
162 hit.x = scene_rdl2::math::lerp(halfWidth, -halfWidth, r0);
163 hit.y = scene_rdl2::math::lerp(halfHeight, -halfHeight, r1);
164 hit.z = 0.f;
165
166 isect->init(p, light->xformPointLocal2Render(hit, 0.f), light->getDirection(0.f));
167
168 return true;
169 }
170
171 virtual const RectLight *getLight() const override
172 {
173 return static_cast<const RectLight *>(mLight.get());
174 }
175
176 virtual const char *getLightTypeName() const override
177 {
178 return "RectLight";
179 }
180};

184class CylinderLightTester : public LightTester
185{
186public:
187 CylinderLightTester(std::shared_ptr<const CylinderLight> light, const char *desc) :
188 LightTester(std::static_pointer_cast<const Light>(light), desc, false, 0) {}
189
190 // simple uniform sampling
191 virtual bool sample(const scene_rdl2::math::Vec3f &p, float r0, float r1, Intersection *isect) const override
192 {
193 const CylinderLight *light = getLight();
194
195 const scene_rdl2::math::Vec3f localP = light->uv2local(scene_rdl2::math::Vec2f(r0, r1));
196 const scene_rdl2::math::Vec3f localN = scene_rdl2::math::Vec3f(localP.x, 0.0f, localP.z);
197
198 const scene_rdl2::math::Vec3f renderP = light->xformPointLocal2Render(localP, 0.f);
199 const scene_rdl2::math::Vec3f renderN = normalize(light->xformNormalLocal2Render(localN, 0.f));
200
201 // We don't cast light from points not visible from p.
202 scene_rdl2::math::Vec3f wi = normalize(renderP - p);
203 if (dot(renderN, wi) > -scene_rdl2::math::sEpsilon) {
204 return false;
205 }
206
207 isect->init(p, renderP, renderN);
208 return true;
209 }
210
211 virtual const CylinderLight *getLight() const override
212 {
213 return static_cast<const CylinderLight *>(mLight.get());
214 }
215
216 virtual const char *getLightTypeName() const override
217 {
218 return "CylinderLight";
219 }
220};

257class SphereLightTester : public LightTester
258{
259public:
260 SphereLightTester(std::shared_ptr<const SphereLight> light, const char *desc) :
261 LightTester(std::static_pointer_cast<const Light>(light), desc, false, 1) {}
262
263 virtual float getVisibleSurfaceArea(const scene_rdl2::math::Vec3f &p, const scene_rdl2::math::Vec3f &n) const override
264 {
265 const SphereLight *light = getLight();
266
267 // Transform point into unit sphere space. We are taking advantage of sphere
268 // symmetry here.
269 scene_rdl2::math::Vec3f localP = (p - light->getPosition(0.f)) * light->getInvRadius();
270
271 // Compute spherical cap on unit sphere and rescale based on world space
272 // sphere radius.
273 float distToLightSq = localP.lengthSqr();
274 float cosTheta = 1.0f / scene_rdl2::math::sqrt(distToLightSq);
275 return (1.0f - cosTheta) * scene_rdl2::math::sTwoPi * light->mRadiusSqr;
276 }
277
278 // simple uniform sampling
279 virtual bool sample(const scene_rdl2::math::Vec3f &p, float r0, float r1, Intersection *isect) const override
280 {
281 const SphereLight *light = getLight();
282
283 scene_rdl2::math::Vec3f localP = shading::sampleSphereUniform(r0, r1);
284
285 const scene_rdl2::math::Vec3f renderP = light->xformPointLocal2Render(localP, 0.f);
286 const scene_rdl2::math::Vec3f renderN = light->xformVectorLocal2RenderRot(localP, 0.f);
287
288 // We don't cast light from points not visible from p.
289 scene_rdl2::math::Vec3f wi = normalize(renderP - p);
290 if (dot(renderN, wi) > -scene_rdl2::math::sEpsilon) {
291 return false;
292 }
293
294 isect->init(p, renderP, renderN);
295
296 return true;
297 }
298
299 virtual const SphereLight *getLight() const override
300 {
301 return static_cast<const SphereLight *>(mLight.get());
302 }
303
304 virtual const char *getLightTypeName() const override
305 {
306 return "SphereLight";
307 }
308};
