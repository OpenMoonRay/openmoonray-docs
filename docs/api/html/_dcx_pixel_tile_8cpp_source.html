107/*virtual*/
108void
109PixelTile::setDataWindow (const IMATH_NAMESPACE::Box2i& data_window,
110 bool sourceWindowYAxisUp,
111 bool /*force*/)
112{
113 m_dataWindow = data_window;
114 // Flip data window vertically if source window is flipped:
115 if (m_yaxis_up != sourceWindowYAxisUp)
116 {
117 m_dataWindow.min.y = m_top_reference - data_window.max.y;
118 m_dataWindow.max.y = m_top_reference - data_window.min.y;
119 }
120}
