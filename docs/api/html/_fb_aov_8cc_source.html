27void
28FbAov::setup(const PartialMergeTilesTbl *partialMergeTilesTbl,
29 fb_util::VariablePixelBuffer::Format fmt, const unsigned width, const unsigned height,
30 bool storeNumSampleData)
31//
32// setup function for non reference buffer and only do memory allocation and clean if needed.
33// We do not reset mDefaultValue and mClosestFilterStatus here. This function only maintains data buffer
34// memory.
35//
36// If you set storeNumSampleData = false, internal mNumSampleBufferTiled is not accessed.
37// This FbAov data is used in 2 different places which are merge computation and client.
38// Basically merge computation needs numSampleBuffer for merge operation but client does not.
39// (Actually this numSampleBuffer is not the same as a weight buffer. numSampleBuffer is each
40// AOV specific number of sample information. Why we need an independent numSampleBuffer for each
41// AOV is because the snapshot was done at slightly different timing for each AOV.)
42// NumSampleBuffer information is sent by progressiveFrame from merge computation to client in some
43// situations even if client does not require NumSampleBuffer information. In this case, client side
44// implementations want to skip all NumSampleBuffer processing and save memory/CPU resources.
45// This storeNumSampleData is used for that purpose.
46//
47{
48 mReferenceType = FbReferenceType::UNDEF;
49
50 bool needPartialInitA = false; // for mActivePixels, mNumSampleBufferTiled
51 bool needWholeInitA = false;
52 bool needPartialInitB = false; // for mBufferTiled
53 bool needWholeInitB = false;
54 if (!mStatus) {
55 needPartialInitA = true;
56 needPartialInitB = true;
57 }
58
59 if (mActivePixels.getWidth() != width ||
60 mActivePixels.getHeight() != height) {
61 //
62 // mActivePixels and mNumSampleBufferTiled are always changed resolution at same time,
63 // So we only do resolution change test for one of them.
64 //
65 mActivePixels.init(width, height);
66 if (storeNumSampleData) {
67 mNumSampleBufferTiled.init(mActivePixels.getAlignedWidth(), mActivePixels.getAlignedHeight());
68 }
69 needWholeInitA = true;
70 needPartialInitA = false;
71 }
72 if (mBufferTiled.getFormat() != fmt ||
73 mBufferTiled.getWidth() != mActivePixels.getAlignedWidth() ||
74 mBufferTiled.getHeight() != mActivePixels.getAlignedHeight()) {
75 mBufferTiled.init(fmt, mActivePixels.getAlignedWidth(), mActivePixels.getAlignedHeight());
76 needWholeInitB = true;
77 needPartialInitB = false;
78 }
79
80 if (!partialMergeTilesTbl) {
81 if (needPartialInitA) {
82 needPartialInitA = false;
83 needWholeInitA = true;
84 }
85 if (needPartialInitB) {
86 needPartialInitB = false;
87 needWholeInitB = true;
88 }
89 }
90
91 if ((needPartialInitA || needWholeInitA) && (needPartialInitB || needWholeInitB)) {
92# ifdef SINGLE_THREAD
93 resetActivePixels((needPartialInitA)? partialMergeTilesTbl: nullptr);
94 if (storeNumSampleData) {
95 resetNumSampleBufferTiled((needPartialInitA)? partialMergeTilesTbl: nullptr);
96 }
97 resetBufferTiled((needPartialInitB)? partialMergeTilesTbl: nullptr);
98# else // else SINGLE_THREAD
99 tbb::parallel_for(0, 3, [&](unsigned id) {
100 switch (id) {
101 case 0 : resetActivePixels((needPartialInitA)? partialMergeTilesTbl: nullptr); break;
102 case 1 :
103 if (storeNumSampleData) {
104 resetNumSampleBufferTiled((needPartialInitA)? partialMergeTilesTbl: nullptr);
105 }
106 break;
107 case 2 : resetBufferTiled((needPartialInitB)? partialMergeTilesTbl: nullptr); break;
108 }
109 });
110# endif // end !SINGLE_THREDAD
111
112 // runtimeVerifySetup("testA", partialMergeTilesTbl); // runtime verify code
113
114 } else {
115 if (needPartialInitA || needWholeInitA) {
116# ifdef SINGLE_THREAD
117 resetActivePixels((needPartialInitA)? partialMergeTilesTbl: nullptr);
118 if (storeNumSampleData) {
119 resetNumSampleBufferTiled((needPartialInitA)? partialMergeTilesTbl: nullptr);
120 }
121# else // else SINGLE_THREAD
122 tbb::parallel_for(0, 2, [&](unsigned id) {
123 if (id == 0) {
124 resetActivePixels((needPartialInitA)? partialMergeTilesTbl: nullptr);
125 } else {
126 if (storeNumSampleData) {
127 resetNumSampleBufferTiled((needPartialInitA)? partialMergeTilesTbl: nullptr);
128 }
129 }
130 });
131# endif // end !SINGLE_THREAD
132 }
133 if (needPartialInitB || needWholeInitB) {
134 resetBufferTiled((needPartialInitB)? partialMergeTilesTbl: nullptr);
135 }
136
137 /* runtime verify code for debug
138 if ((needPartialInitA || needWholeInitA) || (needPartialInitB || needWholeInitB)) {
139 runtimeVerifySetup("testB", partialMergeTilesTbl); // runtime verify code
140 }
141 */
142 }
143
144 mStatus = true;
145}

184int
185FbAov::getNumChan() const
186{
187 int numChan = 0;
188
189 switch (mReferenceType) {
190 case FbReferenceType::UNDEF :
191 // We only support FLOAT, FLOAT2, FLOAT3 so far.
192 switch (mBufferTiled.getFormat()) {
193 case fb_util::VariablePixelBuffer::FLOAT : numChan = 1; break;
194 case fb_util::VariablePixelBuffer::FLOAT2 : numChan = (mClosestFilterStatus)? 1: 2; break;
195 case fb_util::VariablePixelBuffer::FLOAT3 : numChan = (mClosestFilterStatus)? 2: 3; break;
196 case fb_util::VariablePixelBuffer::FLOAT4 : numChan = (mClosestFilterStatus)? 3: 4; break;
197 default : break;
198 }
199 break;
200 case FbReferenceType::BEAUTY : numChan = 3; break;
201 case FbReferenceType::ALPHA : numChan = 1; break;
202 case FbReferenceType::HEAT_MAP : numChan = 1; break;
203 case FbReferenceType::WEIGHT : numChan = 1; break;
204 case FbReferenceType::BEAUTY_AUX : numChan = 3; break;
205 case FbReferenceType::ALPHA_AUX : numChan = 1; break;
206 }
207 return numChan;
208}

210int
211FbAov::getPix(int sx, int sy, std::vector<float> &out) const
212//
213// Return number of channel (1 ~ 4).
214// Return 0 when data is not constructed yet.
215//
216{
217 unsigned w = mActivePixels.getWidth();
218 unsigned h = mActivePixels.getHeight();
219
220 int numChan = 0;
221 switch (mBufferTiled.getFormat()) {
222 case VariablePixelBuffer::FLOAT : numChan = 1; break;
223 case VariablePixelBuffer::FLOAT2 : numChan = 2; break;
224 case VariablePixelBuffer::FLOAT3 : numChan = 3; break;
225 case VariablePixelBuffer::FLOAT4 : numChan = 4; break;
226 default : break;
227 }
228 if (numChan == 0) return 0;
229
230 fb_util::Tiler tiler(w, h);
231 unsigned tileOfs = tiler.linearCoordsToTiledOffset(sx, sy);
232 const float *srcPix =
233 reinterpret_cast<const float *>(mBufferTiled.getFloatBuffer().getData()) + tileOfs * numChan;
234 out.resize(numChan);
235 for (int i = 0; i < numChan; ++i) {
236 out[i] = srcPix[i];
237 }
238 return numChan;
239}

260int
261FbAov::untile(const bool isSrgb,
262 const bool top2bottom,
263 const math::Viewport *roi,
264 const bool closestFilterDepthOutput,
265 std::vector<unsigned char> &rgbFrame) const
266{
267 auto normalizedDepth = [](float depth, float minDepth, float maxDepth) -> float {
268 return ((minDepth != FLT_MAX)?
269 (1.0f - (depth - minDepth) / (maxDepth - minDepth)): // return normalized depth
270 0.0f); // empty data, return 0.0
271 };
272
273 auto normalizedPos = [](float v, float min, float max) -> float {
274 // Regarding position value
275 // Empty pixels have inf value when using the closestFilter and
276 // we skip display about inf pixels.
277 return (min == FLT_MAX || // empty whole image
278 std::isinf(v)) ? // empty pixels when using closestFilter
279 0.0f: // non active pixel value
280 ((v - min) / (max - min)); // normalized active pixel value
281 };
282
283 std::function<unsigned char(float)> f2ucConversion =
284 (!isSrgb)? fb_util::GammaF2C::g22: fb_util::SrgbF2C::sRGB;
285
286 unsigned w = mActivePixels.getWidth();
287 unsigned h = mActivePixels.getHeight();
288
289 int numChan = 0;
290 switch (mBufferTiled.getFormat()) {
291 case VariablePixelBuffer::FLOAT :
292 numChan = 1;
293 if (isDepthRelatedAov()) {
294 float minDepth, maxDepth;
295 computeDepthMinMax(mBufferTiled.getFloatBuffer().getData(),
296 0, // depthId
297 minDepth, maxDepth);
298 untileSinglePixelMainLoop
299 (w, h, roi,
300 3, // dstNumChan
301 [&](unsigned tileOfs, unsigned pixOfs, unsigned dstOfs) {
302 const float *srcPix = mBufferTiled.getFloatBuffer().getData() + (tileOfs + pixOfs);
303 float depth01 = normalizedDepth(*srcPix, minDepth, maxDepth);
304 unsigned char uc = f2ucConversion(depth01);
305 rgbFrame[dstOfs ] = uc;
306 rgbFrame[dstOfs + 1] = uc;
307 rgbFrame[dstOfs + 2] = uc;
308 },
309 top2bottom);
310 } else {
311 untileSinglePixelMainLoop
312 (w, h, roi,
313 3, // dstNumChan
314 [&](unsigned tileOfs, unsigned pixOfs, unsigned dstOfs) {
315 const float *srcPix = mBufferTiled.getFloatBuffer().getData() + (tileOfs + pixOfs);
316 unsigned char uc = f2ucConversion(*srcPix);
317 rgbFrame[dstOfs ] = uc;
318 rgbFrame[dstOfs + 1] = uc;
319 rgbFrame[dstOfs + 2] = uc;
320 },
321 top2bottom);
322 }
323 break;
324
325 case VariablePixelBuffer::FLOAT2 :
326 if (mClosestFilterStatus) {
327 // use special logic to generate 8bit display data for closestFilter FLOAT2
328 if (closestFilterDepthOutput) {
329 // output closestFilter depth mode
330 numChan = 1;
331 float minDepth, maxDepth;
332 computeDepthMinMax(mBufferTiled.getFloat2Buffer().getData(),
333 1, // depthId
334 minDepth, maxDepth);
335 untileSinglePixelMainLoop
336 (w, h, roi,
337 3, // dstNumChan
338 [&](unsigned tileOfs, unsigned pixOfs, unsigned dstOfs) {
339 const float *srcPix =
340 reinterpret_cast<const float *>(mBufferTiled.getFloat2Buffer().getData()) +
341 (tileOfs + pixOfs) * 2;
342 float depth01 = normalizedDepth(srcPix[1], minDepth, maxDepth);
343 unsigned char uc = f2ucConversion(depth01);
344 rgbFrame[dstOfs ] = uc;
345 rgbFrame[dstOfs + 1] = uc;
346 rgbFrame[dstOfs + 2] = uc;
347 },
348 top2bottom);
349 } else {
350 // output original data (float) and ignore closestFilter depth
351 numChan = 1;
352 untileSinglePixelMainLoop
353 (w, h, roi,
354 3, // dstNumChan
355 [&](unsigned tileOfs, unsigned pixOfs, unsigned dstOfs) {
356 const float *srcPix =
357 reinterpret_cast<const float *>(mBufferTiled.getFloat2Buffer().getData()) +
358 (tileOfs + pixOfs) * 2;
359 unsigned char uc = f2ucConversion(srcPix[0]);
360 rgbFrame[dstOfs ] = uc;
361 rgbFrame[dstOfs + 1] = uc;
362 rgbFrame[dstOfs + 2] = uc;
363 },
364 top2bottom);
365 }
366 } else {
367 // non closestFilter condition
368 numChan = 2;
369 untileSinglePixelMainLoop
370 (w, h, roi,
371 3, // dstNumChan
372 [&](unsigned tileOfs, unsigned pixOfs, unsigned dstOfs) {
373 const float *srcPix =
374 reinterpret_cast<const float *>(mBufferTiled.getFloat2Buffer().getData()) +
375 (tileOfs + pixOfs) * 2;
376 rgbFrame[dstOfs ] = f2ucConversion(srcPix[0]);
377 rgbFrame[dstOfs + 1] = f2ucConversion(srcPix[1]);
378 rgbFrame[dstOfs + 2] = 0;
379 },
380 top2bottom);
381 }
382 break;
383
384 case VariablePixelBuffer::FLOAT3 :
385 if (mClosestFilterStatus) {
386 // use special logic to generate 8bit display data for closestFilter FLOAT3
387 if (closestFilterDepthOutput) {
388 numChan = 1;
389 // output closestFilter depth mode
390 float minDepth, maxDepth;
391 computeDepthMinMax(mBufferTiled.getFloat2Buffer().getData(),
392 2, // depthId
393 minDepth, maxDepth);
394 untileSinglePixelMainLoop
395 (w, h, roi,
396 3, // dstNumChan
397 [&](unsigned tileOfs, unsigned pixOfs, unsigned dstOfs) {
398 const float *srcPix =
399 reinterpret_cast<const float *>(mBufferTiled.getFloat3Buffer().getData()) +
400 (tileOfs + pixOfs) * 3;
401 float depth01 = normalizedDepth(srcPix[2], minDepth, maxDepth);
402 unsigned char uc = f2ucConversion(depth01);
403 rgbFrame[dstOfs ] = uc;
404 rgbFrame[dstOfs + 1] = uc;
405 rgbFrame[dstOfs + 2] = uc;
406 },
407 top2bottom);
408 } else {
409 numChan = 2;
410 // output original data (float2) and ignore closestFilter depth
411 untileSinglePixelMainLoop
412 (w, h, roi,
413 3, // dstNumChan
414 [&](unsigned tileOfs, unsigned pixOfs, unsigned dstOfs) {
415 const float *srcPix =
416 reinterpret_cast<const float *>(mBufferTiled.getFloat3Buffer().getData()) +
417 (tileOfs + pixOfs) * 3;
418 rgbFrame[dstOfs ] = f2ucConversion(srcPix[0]);
419 rgbFrame[dstOfs + 1] = f2ucConversion(srcPix[1]);
420 rgbFrame[dstOfs + 2] = 0;
421 },
422 top2bottom);
423 }
424 } else {
425 numChan = 3;
426 // non closestFilter condition
427 if (isPositionRelatedAov()) {
428 // use special logic to generate 8bit display data for position related AOV
429 math::Vec3f min, max;
430 computePositionMinMax(mBufferTiled.getFloat3Buffer().getData(), 3, min, max);
431 untileSinglePixelMainLoop
432 (w, h, roi,
433 3, // dstNumChan
434 [&](unsigned tileOfs, unsigned pixOfs, unsigned dstOfs) {
435 const float *srcPix =
436 reinterpret_cast<const float *>(mBufferTiled.getFloat3Buffer().getData()) +
437 (tileOfs + pixOfs) * 3;
438 rgbFrame[dstOfs ] = f2ucConversion(normalizedPos(srcPix[0], min[0], max[0]));
439 rgbFrame[dstOfs + 1] = f2ucConversion(normalizedPos(srcPix[1], min[1], max[1]));
440 rgbFrame[dstOfs + 2] = f2ucConversion(normalizedPos(srcPix[2], min[2], max[2]));
441 },
442 top2bottom);
443 } else {
444 // simple float3 AOV
445 untileSinglePixelMainLoop
446 (w, h, roi,
447 3, // dstNumChan
448 [&](unsigned tileOfs, unsigned pixOfs, unsigned dstOfs) {
449 const float *srcPix =
450 reinterpret_cast<const float *>(mBufferTiled.getFloat3Buffer().getData()) +
451 (tileOfs + pixOfs) * 3;
452 rgbFrame[dstOfs ] = f2ucConversion(srcPix[0]);
453 rgbFrame[dstOfs + 1] = f2ucConversion(srcPix[1]);
454 rgbFrame[dstOfs + 2] = f2ucConversion(srcPix[2]);
455 },
456 top2bottom);
457 }
458 }
459 break;
460
461 case VariablePixelBuffer::FLOAT4 :
462 if (closestFilterDepthOutput) {
463 numChan = 1;
464 // output closestFilter depth mode
465 float minDepth, maxDepth;
466 computeDepthMinMax(mBufferTiled.getFloat2Buffer().getData(),
467 3, // depthId
468 minDepth, maxDepth);
469 untileSinglePixelMainLoop
470 (w, h, roi,
471 3, // dstNumChan
472 [&](unsigned tileOfs, unsigned pixOfs, unsigned dstOfs) {
473 const float *srcPix =
474 reinterpret_cast<const float *>(mBufferTiled.getFloat4Buffer().getData()) +
475 (tileOfs + pixOfs) * 4;
476 float depth01 = normalizedDepth(srcPix[3], minDepth, maxDepth);
477 unsigned char uc = f2ucConversion(depth01);
478 rgbFrame[dstOfs ] = uc;
479 rgbFrame[dstOfs + 1] = uc;
480 rgbFrame[dstOfs + 2] = uc;
481 },
482 top2bottom);
483 } else {
484 numChan = 3;
485 // output original data (float3) and ignore closestFilter depth
486 if (isPositionRelatedAov()) {
487 // use special logic to generate 8bit display data for position related AOV
488 math::Vec4f min, max;
489 computePositionMinMax(mBufferTiled.getFloat4Buffer().getData(), 3, min, max);
490 untileSinglePixelMainLoop
491 (w, h, roi,
492 3, // dstNumChan
493 [&](unsigned tileOfs, unsigned pixOfs, unsigned dstOfs) {
494 const float *srcPix =
495 reinterpret_cast<const float *>(mBufferTiled.getFloat4Buffer().getData()) +
496 (tileOfs + pixOfs) * 4;
497 rgbFrame[dstOfs ] = f2ucConversion(normalizedPos(srcPix[0], min[0], max[0]));
498 rgbFrame[dstOfs + 1] = f2ucConversion(normalizedPos(srcPix[1], min[1], max[1]));
499 rgbFrame[dstOfs + 2] = f2ucConversion(normalizedPos(srcPix[2], min[2], max[2]));
500 },
501 top2bottom);
502 } else {
503 // non position related AOV and ignore closestFilter depth
504 untileSinglePixelMainLoop
505 (w, h, roi,
506 3, // dstNumChan
507 [&](unsigned tileOfs, unsigned pixOfs, unsigned dstOfs) {
508 const float *srcPix =
509 reinterpret_cast<const float *>(mBufferTiled.getFloat4Buffer().getData()) +
510 (tileOfs + pixOfs) * 4;
511 // We only use 1st 3 channels for output and ignore 4th channel
512 rgbFrame[dstOfs ] = f2ucConversion(srcPix[0]);
513 rgbFrame[dstOfs + 1] = f2ucConversion(srcPix[1]);
514 rgbFrame[dstOfs + 2] = f2ucConversion(srcPix[2]);
515 },
516 top2bottom);
517 }
518 }
519 break;
520
521 default :
522 memset(static_cast<void *>(rgbFrame.data()), 0x0, w * h * 3);
523 break;
524 }
525
526 return numChan;
527}

529int
530FbAov::untile(const bool top2bottom,
531 const math::Viewport *roi,
532 const bool closestFilterDepthOutput,
533 std::vector<float> &data) const
534{
535 unsigned w = mActivePixels.getWidth();
536 unsigned h = mActivePixels.getHeight();
537
538 int numChan = 0;
539 switch (mBufferTiled.getFormat()) {
540 case VariablePixelBuffer::FLOAT :
541 numChan = 1;
542 untileSinglePixelMainLoop(w, h, roi,
543 1, // dstNumChan
544 [&](unsigned tileOfs, unsigned pixOfs, unsigned dstOfs) {
545 const float *srcPix =
546 reinterpret_cast<const float *>(mBufferTiled.getFloatBuffer().getData()) +
547 (tileOfs + pixOfs);
548 data[dstOfs] = srcPix[0];
549 }, top2bottom);
550 break;
551
552 case VariablePixelBuffer::FLOAT2 :
553 if (mClosestFilterStatus) {
554 // use special logic to output data for closestFilter FLOAT2
555 if (closestFilterDepthOutput) {
556 // output closestFilter depth mode
557 numChan = 1;
558 untileSinglePixelMainLoop
559 (w, h, roi,
560 1, // dstNumChan
561 [&](unsigned tileOfs, unsigned pixOfs, unsigned dstOfs) {
562 const float *srcPix =
563 reinterpret_cast<const float *>(mBufferTiled.getFloat2Buffer().getData()) +
564 (tileOfs + pixOfs) * 2;
565 data[dstOfs] = srcPix[1]; // last component is depth
566 },
567 top2bottom);
568 } else {
569 // output original data (float) and ignore closestFilter depth
570 numChan = 1;
571 untileSinglePixelMainLoop
572 (w, h, roi,
573 1, // dstNumChan
574 [&](unsigned tileOfs, unsigned pixOfs, unsigned dstOfs) {
575 const float *srcPix =
576 reinterpret_cast<const float *>(mBufferTiled.getFloat2Buffer().getData()) +
577 (tileOfs + pixOfs) * 2;
578 data[dstOfs] = srcPix[0];
579 },
580 top2bottom);
581 }
582 } else {
583 // non closestFilter condition
584 numChan = 2;
585 untileSinglePixelMainLoop
586 (w, h, roi,
587 2, // dstNumChan
588 [&](unsigned tileOfs, unsigned pixOfs, unsigned dstOfs) {
589 const float *srcPix =
590 reinterpret_cast<const float *>(mBufferTiled.getFloat2Buffer().getData()) +
591 (tileOfs + pixOfs) * 2;
592 data[dstOfs ] = srcPix[0];
593 data[dstOfs + 1] = srcPix[1];
594 },
595 top2bottom);
596 }
597 break;
598
599 case VariablePixelBuffer::FLOAT3 :
600 if (mClosestFilterStatus) {
601 // use special logic to output data for closestFilter FLOAT3
602 if (closestFilterDepthOutput) {
603 // output closestFilter depth mode
604 numChan = 1;
605 untileSinglePixelMainLoop
606 (w, h, roi,
607 1, // dstNumChan
608 [&](unsigned tileOfs, unsigned pixOfs, unsigned dstOfs) {
609 const float *srcPix =
610 reinterpret_cast<const float *>(mBufferTiled.getFloat3Buffer().getData()) +
611 (tileOfs + pixOfs) * 3;
612 data[dstOfs] = srcPix[2]; // last component is depth
613 },
614 top2bottom);
615 } else {
616 // output original data (float2) and ignore closestFilter depth
617 numChan = 2;
618 untileSinglePixelMainLoop
619 (w, h, roi,
620 2, // dstNumChan
621 [&](unsigned tileOfs, unsigned pixOfs, unsigned dstOfs) {
622 const float *srcPix =
623 reinterpret_cast<const float *>(mBufferTiled.getFloat3Buffer().getData()) +
624 (tileOfs + pixOfs) * 3;
625 data[dstOfs ] = srcPix[0];
626 data[dstOfs + 1] = srcPix[1];
627 },
628 top2bottom);
629 }
630 } else {
631 // non closestFilter condition
632 numChan = 3;
633 untileSinglePixelMainLoop
634 (w, h, roi,
635 3, // dstNumChan
636 [&](unsigned tileOfs, unsigned pixOfs, unsigned dstOfs) {
637 const float *srcPix =
638 reinterpret_cast<const float *>(mBufferTiled.getFloat3Buffer().getData()) +
639 (tileOfs + pixOfs) * 3;
640 data[dstOfs ] = srcPix[0];
641 data[dstOfs + 1] = srcPix[1];
642 data[dstOfs + 2] = srcPix[2];
643 },
644 top2bottom);
645 }
646 break;
647
648 case VariablePixelBuffer::FLOAT4 :
649 if (mClosestFilterStatus) {
650 if (closestFilterDepthOutput) {
651 // output closestFilter depth mode
652 numChan = 1;
653 untileSinglePixelMainLoop
654 (w, h, roi,
655 1, // dstNumChan
656 [&](unsigned tileOfs, unsigned pixOfs, unsigned dstOfs) {
657 const float *srcPix =
658 reinterpret_cast<const float *>(mBufferTiled.getFloat4Buffer().getData()) +
659 (tileOfs + pixOfs) * 4;

905void
906FbAov::conv888(const FArray &srcData,
907 const bool isSrgb,
908 const bool closestFilterDepthOutput,
909 UCArray &dstRgb888) const
910//
911// This function is designed for debug purpose.
912//
913// srcData should be properly resized and filled up by Fb::untileRenderOutput() with same
914// closestFilterDepthOutput argument setting.
915// This means srcData.size() = pixTotal * numChan (pixTotal supports ROI).
916// (See Fb_untile.cc Fb::untileRenderOutputMain())
917//
918{
919 auto resizeDstBuff = [](const int numChannels, const FArray &srcData, UCArray &dstRgb888) {
920 unsigned pixTotal = srcData.size() / numChannels;
921 unsigned dstSize = pixTotal * 3;
922 if (dstRgb888.size() != dstSize) {
923 dstRgb888.resize(dstSize);
924 }
925 };
926 auto pixLoop = [](const int numChannels, const FArray &srcData, UCArray &dstRgb888,
927 std::function<void(const float *, unsigned char *)> convPixFunc) {
928 unsigned pixTotal = srcData.size() / numChannels;
929 for (unsigned pixOfs = 0; pixOfs < pixTotal; ++pixOfs) {
930 const float *srcPix = &(srcData[pixOfs * numChannels]);
931 unsigned char *dstPix = &(dstRgb888[pixOfs * 3]);
932 convPixFunc(srcPix, dstPix);
933 }
934 };
935 auto normalizedDepth = [](float depth, float minDepth, float maxDepth) -> float {
936 // Regarding depth value
937 return ((minDepth != FLT_MAX)?
938 (1.0f - (depth - minDepth) / (maxDepth - minDepth)): // return normalized depth
939 0.0f); // empty data, return 0.0
940 };
941 auto normalizedPos = [](float v, float min, float max) -> float {
942 // Regarding position value
943 // Empty pixels have inf value when using the closestFilter and
944 // we skip display about inf pixels.
945 return (min == FLT_MAX || // empty whole image
946 std::isinf(v)) ? // empty pixels when using closestFilter
947 0.0f: // non active pixel value
948 ((v - min) / (max - min)); // normalized active pixel value
949 };
950
951 std::function<unsigned char(float)> f2ucConversion =
952 (!isSrgb)? fb_util::GammaF2C::g22: fb_util::SrgbF2C::sRGB;
953
954 switch (mBufferTiled.getFormat()) {
955 case VariablePixelBuffer::FLOAT :
956 resizeDstBuff(1, srcData, dstRgb888);
957 if (isDepthRelatedAov()) {
958 float minDepth, maxDepth;
959 conv888_computeDepthMinMax(srcData,
960 1, // pixFloatCount
961 0, // depthId
962 minDepth, maxDepth);
963 pixLoop(1, srcData, dstRgb888,
964 [&](const float *srcPix, unsigned char *dstPix) {
965 float depth01 = normalizedDepth(*srcPix, minDepth, maxDepth);
966 unsigned char uc = f2ucConversion(depth01);
967 dstPix[0] = uc;
968 dstPix[1] = uc;
969 dstPix[2] = uc;
970 });
971 } else {
972 pixLoop(1, srcData, dstRgb888,
973 [&](const float *srcPix, unsigned char *dstPix) {
974 unsigned char uc = f2ucConversion(*srcPix);
975 dstPix[0] = uc;
976 dstPix[1] = uc;
977 dstPix[2] = uc;
978 });
979 }
980 break;
981
982 case VariablePixelBuffer::FLOAT2 :
983 if (mClosestFilterStatus) {
984 if (closestFilterDepthOutput) {
985 // ouptut closestFilter depth mode
986 // srcData has single channel and it's closestFilter depth
987 resizeDstBuff(1, srcData, dstRgb888);
988 float minDepth, maxDepth;
989 conv888_computeDepthMinMax(srcData,
990 1, // pixFloatCount
991 0, // depthId
992 minDepth, maxDepth);
993 pixLoop(1, srcData, dstRgb888,
994 [&](const float *srcPix, unsigned char *dstPix) {
995 float depth01 = normalizedDepth(*srcPix, minDepth, maxDepth);
996 unsigned char uc = f2ucConversion(depth01);
997 dstPix[0] = uc;
998 dstPix[1] = uc;
999 dstPix[2] = uc;
1000 });
1001 } else {
1002 // output original data (float) and ignore closestFilter depth
1003 // srcData has single channel and it's original data (not closestFilter depth)
1004 resizeDstBuff(1, srcData, dstRgb888);
1005 pixLoop(1, srcData, dstRgb888,
1006 [&](const float *srcPix, unsigned char *dstPix) {
1007 unsigned char uc = f2ucConversion(*srcPix);
1008 dstPix[0] = uc;
1009 dstPix[1] = uc;
1010 dstPix[2] = uc;
1011 });
1012 }
1013 } else {
1014 // non closestFilter condition
1015 // srcData has 2 channels
1016 resizeDstBuff(2, srcData, dstRgb888);
1017 pixLoop(2, srcData, dstRgb888,
1018 [&](const float *srcPix, unsigned char *dstPix) {

1020 dstPix[1] = f2ucConversion(srcPix[1]);

1023 }
1024 break;

1026 case VariablePixelBuffer::FLOAT3 :

1044 });
1045 } else {

1050 pixLoop(2, srcData, dstRgb888,
1051 [&](const float *srcPix, unsigned char *dstPix) {
1052 dstPix[0] = f2ucConversion(srcPix[0]);
1053 dstPix[1] = f2ucConversion(srcPix[1]);
1054 dstPix[2] = 0;
1055 });
1056 }
1057 } else {

1090 minDepth, maxDepth);

1097 dstPix[2] = uc;
1098 });
1099 } else {
1100 // output original data (float3) and ignore closestFilter depth
1101 // srcData has 3 channels and it's original data.
1102 // They are not include closestFilter depth

1310void
1311FbAov::conv888_computeDepthMinMax(const FArray &srcData,
1312 unsigned pixFloatCount,
1313 unsigned depthId,
1314 float &min, float &max) const
1315{
1316 // Basic idea to compute min/max depth is same as FbAov::computeDepthMinMax()
1317
1318 unsigned pixTotal = srcData.size() / pixFloatCount;
1319
1320 //
1321 // First step
1322 //
1323 min = FLT_MAX;
1324 float maxLimit = FLT_MIN;
1325 for (unsigned pixOfs = 0; pixOfs < pixTotal; ++pixOfs) {
1326 unsigned ofs = pixOfs * pixFloatCount + depthId;
