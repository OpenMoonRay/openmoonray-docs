52void
53AthenaCSVStreamBuf::open(bool debug)
54{
55 // From the Athena wiki:
56 // The "athena." identifier followed by your producer or application name,
57 // e.g.: "athena.moonray.stats"
58 char* athena_development = getenv("MNRY_ATHENA_STATS_DEV");
59 if (debug || athena_development != nullptr) {
60 mIdent = std::string("dev-athena.moonray_stats");
61 } else {
62 mIdent = std::string("prod-athena.moonray_stats");
63 }
64
65 const auto serviceEntry = ::getservbyname("syslog", "udp");
66 ::endservent();
67
68 if (serviceEntry == nullptr) {
69 std::cerr << "Failed to lookup syslog from services database" << std::endl;
70 return;
71 }
72
73 mSocket = ::socket(AF_INET, SOCK_DGRAM|SOCK_NONBLOCK, 0);
74 if (mSocket == -1) {
75 std::cerr << "Failed to create socket due to: " << rndr::getErrorDescription() << std::endl;
76 return;
77 }
78
79 struct sockaddr_in addr;
80 addr.sin_addr.s_addr = ::inet_addr("127.0.0.1");
81 addr.sin_family = AF_INET;
82 addr.sin_port = htons(serviceEntry->s_port);
83 if (::connect(mSocket, (struct sockaddr*)&addr, sizeof(addr)) == 0) {
84 mOpened = true;
85 } else {
86 std::cerr << "Failed to connect to socket due to: " << rndr::getErrorDescription() << std::endl;
87 }
88}

90std::streambuf::int_type
91AthenaCSVStreamBuf::overflow(int_type ch)
92{
93 if (!traits_type::eq_int_type(ch, traits_type::eof())) {
94 if (ch != traits_type::to_int_type('\n')) {
95 mStorage.push_back(traits_type::to_char_type(ch));
96 } else {
97 if (!mStorage.empty()) {
98 appendBuffer(mHostname.c_str(), mHostname.length());
99 appendBuffer(',');
100 appendBuffer(mGUID.c_str(), mGUID.length());
101 appendBuffer(',');
102 appendTimestamp();
103 appendBuffer(',');
104 appendBuffer(mStorage.data(), mStorage.size());
105 appendBuffer('\0'); // string must be null terminated
106 flushBuffer();
107 mStorage.clear();
108 }
109 }
110 return traits_type::not_eof(ch);
111 }
112 return traits_type::eof();
113}

122void
123AthenaCSVStreamBuf::appendTimestamp()
124{
125 // digits10 is rounded down, because it represents the
126 // number of digits that can be stored without loss of
127 // information, so add one.
128 constexpr auto digits = std::numeric_limits<MSType>::digits10 + 1;
129
130 // Add one for terminator.
131 char timestampString[digits + 1] = { 0 };
132
133 const auto ms = msSinceEpoch();
134 static_assert(sizeof(MSType) == sizeof(long),
135 "We're using the long conversion for sprintf");
136 std::snprintf(timestampString, digits + 1, "%ld", ms);
137 appendBuffer(timestampString, std::strlen(timestampString));
138}

140void
141AthenaCSVStreamBuf::flushBuffer()
142{
143 if (mSyslogBuffer.size() <= 0) return;
144
145 // TIMESTAMP
146 char timestamp[256];
147 time_t now = time(0);
148 struct tm tmnow;
149 localtime_r(&now, &tmnow);
150 // Note timestamp must be in this old format, ISO standard format isn't accepted
151 //rfc5424: strftime(timestamp, 256, "%FT%T%z", &tmnow);
152 strftime(timestamp, 256, "%b %e %H:%M:%S", &tmnow);
153
154 std::ostringstream msg;
155 msg << "<" << SYSLOG_PRIO << "> "
156 << timestamp << " "
157 << mHostname << " "
158 << mIdent << ": "
159 << mDwaGacId << "," << mSyslogBuffer.data();
160
161 std::string s = msg.str();
162 if (::send(mSocket, s.c_str(), s.size(), 0) == -1) {
163 std::cerr << "Failed to send log message to Athena due to: "
164 << rndr::getErrorDescription()
165 << " : " << s << std::endl;
166 }
167
168 mSyslogBuffer.clear();
169}
