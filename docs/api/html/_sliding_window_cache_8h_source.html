15// This class will return the next zero-based index into an array assumed to be
16// kSIMDSize in length.
17//
18// When necessary, this class will calculate a contiguous subset of values, of
19// length kSIMDSize, through the function passed in.
20//
21// The passed in function must take an unsigned value, n, as the starting
22// value, and compute all values in [n, n+kSIMDSize).
23//
24// E.g. If we ask for the 64th value in the sequence, the class may call f(64)
25// and return 0 as the index into an array. The 65th value may not call the
26// function, but the class will return 1.
27class SlidingWindowCache
28{
29 utype mCachedMin;
30 static const utype sDefaultState = std::numeric_limits<utype>::max();
31
32public:
33 SlidingWindowCache();
34 void reset();
35
36 template <typename CacheCalc>
37 utype getIndex(CacheCalc cacheCalc, utype incrementalValue);
38};

51finline utype SlidingWindowCache::getIndex(CacheCalc cacheCalc, utype incrementalValue)
52{
53 const utype cachedMax = mCachedMin + kSIMDSize;
54 if (incrementalValue < mCachedMin || incrementalValue >= cachedMax) {
55 cacheCalc(incrementalValue);
56 mCachedMin = incrementalValue;
57 }
58
59 return incrementalValue - mCachedMin;
