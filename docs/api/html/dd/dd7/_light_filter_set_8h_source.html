17/**
18 * The LightFilterSet represents a collection of LightFilters with no duplicates. It's
19 * used in Part/Material assignments in the Layer, where you can define which
20 * LightFilters affect a particular Part/Material assignment. That collection of
21 * LightFilters is the LightFilterSet. LightFilterSets can be reused for many Part/Material
22 * assignments.
23 *
24 * It only has one attribute, named "lightfilters", which is a SceneObjectVector.
25 * We provide convenience functions on the LightFilterSet to add and remove LightFilters
26 * from the set, as well as check if a LightFilter is contained in the set. Please use these
27 * functions to maintain the lightfilter's uniqueness invariant. When setting "lightfilters"
28 * directly, make sure the light filters are unique and their order is deterministic.
29 *
30 * You can get the LightFilters as a const SceneObjectVector& with the getLightFilters() method call.
31 */
32class LightFilterSet : public SceneObject
33{
34public:
35 typedef SceneObject Parent;
36
37 LightFilterSet(const SceneClass& sceneClass, const std::string& name);
38 static SceneObjectInterface declare(SceneClass& sceneClass);
39
40 /// Retrieves the set of unique LightFilters in this LightFilterSet.
41 finline const SceneObjectVector& getLightFilters() const;
42
43 /**
44 * Adds the given LightFilter to the LightFilterSet, if it is not already a member of
45 * the set. If it is already a member of the set, this does nothing.
46 *
47 * @param lightfilter The LightFilter to add to the LightFilterSet.
48 */
49 void add(LightFilter* lightfilter);
50
51 /**
52 * Removes the given LightFilter from the LightFilterSet, if it is already a member of
53 * the set. If it is not a member of the set, this does nothing.
54 *
55 * @param lightfilter The LightFilter to remove from the LightFilterSet.
56 */
57 void remove(LightFilter* lightfilter);
58
59 /**
60 * Returns true if the given LightFilter is a member of the LightFilterSet. There's
61 * no need to call this before calling add() or remove(), as they will
62 * gracefully handle those edge cases.
63 *
64 * @param lightfilter The LightFilter to check for membership.
65 * @return True if the lightfilter is a member of the LightFilterSet.
66 */
67 bool contains(const LightFilter* lightfilter) const;
68
69 /**
70 * Check if any of the following that are modified: LightFilters
71 * in the LightFilterSet, SceneObject attributes, and the LightFilterSet itself.
72 * Should only be called after all UpdateGuards.
73 *
74 * @return True if the lightfilter list changed
75 */
76 bool updatePrepLightFilter(UpdateHelper& sceneObjects, size_t depth);
77
78 /**
79 * Alphabetizes light filter list by name
80 */
81 void update() override;
82
83 /// Completely empties the LightFilterSet so that it doesn't contain anything.
84 void clear();
85
86private:
87 static AttributeKey<SceneObjectVector> sLightFiltersKey;
88};
