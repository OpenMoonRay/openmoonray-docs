99///
100/// A RayVertex structure serves a dual responsibilities. During the recording
101/// phase, it's the internal structure we use to build up the collections of
102/// hitpoints along each ray. It also serves as the final representation for
103/// each hitpoint in the final built DebugRayDatabase.
104///
105
106struct DebugRayVertex
107{
108 //
109 // filled in by the app after it calls DebugRayRecorder::extendRay
110 // (76 bytes)
111 //
112
113 scene_rdl2::math::Vec3f mHitPoint; ///< in world space
114 scene_rdl2::math::Vec3f mNormal; ///< from differential geometry
115 scene_rdl2::math::Vec3f mRayDiffX; ///< final world space position of ray differential in x
116 scene_rdl2::math::Vec3f mRayDiffY; ///< final world space position of ray differential in y
117 scene_rdl2::math::Vec3f mFree; ///< TBD
118 scene_rdl2::math::Vec3f mContribution; ///< shader eval with lighting
119 uint32_t mUserTags; ///< bitfield of user specified tags
120
121 //
122 // filled in automatically by the system - do not edit
123 // (20 bytes)
124 //
125
126 uint32_t mId; ///< unique id per ray vertex
127 uint32_t mParentId; ///< 0 means root or no parent
128 uint16_t mThreadId; ///< thread id we were rendered on
129 uint8_t mDepth; ///< which bounce are we on
130 uint8_t mPad;
131
132 uint16_t mScreenX; ///< if this is -1 then we are a dummy ray (record and build time only)
133 uint16_t mScreenY;
134
135 union {
136 uint32_t mNumSubnodes; ///< how many nodes in this entire branch, including the current node
137 uint16_t mParentThreadId; ///< only used by the recorder and builder, not needed when in database form
138 };
139
140 //
141 // these are only valid after rays have been sent though the debug ray builder
142 //
143 bool isRoot() const { return mParentId == uint32_t(-1); }
144 bool isLeaf() const { return mNumSubnodes == 1; }
145};

151///
152/// The steps needed to generate a ray database are:
153///
154/// 1. Record rays using the DebugRayRecorder class. This can be done in
155/// parallel using a separate DebugRayRecorder instance for each separate
156/// thread.
157/// 2. Take all the recorded ray data and use the DebugRayBuilder class to
158/// condition it into a queryable ray database.
159/// 3. Once the data is conditioned, we can store it in a DebugRayDatabase
160/// class which gives us the ability to submit queries against it and also
161/// serialize the data.
162///
163/// Said another way, here is how the data flows:
164/// DebugRayRecorder -> DebugRayBuilder -> DebugRayDatabase
165///
166/// The usage of the DebugRayRecorder is hidden behind the RAYDB_* macros, and
167/// generally you shouldn't need to interact with it directly, although it is
168/// fine to do so if you want.
169///
170/// Each thread will have its own instance of a DebugRayRecorder in thread local
171/// storage. Data from each instance then gets merged inside of the DebugRayRecorder.
172///
173

184 /// which can be recorded. When we run out of entry space, we return the dummy

249///
250/// This class takes a number of RayRecorder instances, merges and sorts all
251/// the data in such a way that we can submit queries against it.
252///
253/// Each recorder will contain a group of ray hitpoints with connectivity
254/// information. There is no limitations on which hitpoints a particular recorder
255/// can record. For example, portions of the same primary ray may be propagated on
256/// different threads in the renderer. Since we record in a thread local fashion,
257/// a single ray may jump around from recorder instance to recorder instance -
258/// this is fine, it's supported behavior.
259///
260/// Internally, data is sorted into a big hierarchical tree. The nodes in this
261/// tree are laid out linearly in memory in the mVertices array. Primary rays
262/// appear at the root, first hits next, second hits another level underneath the
263/// first hits, and so on. All primary rays are sorted amongst themselves based
264/// on screen coordinate in a scanline by scanline fashion. We generate 2
265/// additional "indexes" which accelerate lookups.
266///
267/// 1. We stored a sorted vector of all primary ray indices in the member
268/// mPrimaryRayIndices. Each element in this vector gives the offset of
269/// each successive primary ray (sorted by screen location) in the mVertices
270/// array.
271/// 2. To accelerate finding primary rays for particular image locations,
272/// we have another sorted vector called mScreenPositions. This simply
273/// stores the screen locations for each primary ray in mPrimaryRayIndices
274/// so that we can locate primary rays at any location without having to
275/// touch the mVertices array at all.
276///
277/// After the build step has been executed, you need to call exportDatabase to
278/// initialize an actual DebugRayDatabase object. Once done, the DebugRayBuilder
279/// doesn't need to persist in memory any longer.
280///
281
282class DebugRayBuilder
283{
284public:
285 /// Calling this function mutates the DebugRayRecorders but not in any harmful
286 /// way, i.e. calling build twice in a row will still produce the correct results.
287
288 // @@@ TODO, Pass in a matrix to go from shade space to world space before
289 // saving data out.
290 bool build(unsigned width, unsigned height, std::vector<DebugRayRecorder *> const &recorders);
291 void cleanUp();

301 struct SortChildren
302 {
303 finline bool operator()(Node const *a, Node const *b) const
304 {
305 DebugRayVertex const *vertA = a->mVertex;
306 DebugRayVertex const *vertB = b->mVertex;
307
308 if (vertA->mScreenY != vertB->mScreenY) return vertA->mScreenY < vertB->mScreenY;
309 if (vertA->mScreenX != vertB->mScreenX) return vertA->mScreenX < vertB->mScreenX;
310 if (vertA->mDepth != vertB->mDepth) return vertA->mDepth < vertB->mDepth;
311 if (vertA->mThreadId != vertB->mThreadId) return vertA->mThreadId < vertB->mThreadId;
312
313 MNRY_ASSERT(vertA->mId != vertB->mId);
314 return vertA->mId < vertB->mId;
315 }
316 };

489finline void
490addRayVertex(pbr::TLState *tls,
491 scene_rdl2::math::Vec3f const &hitPoint,
492 scene_rdl2::math::Vec3f const &hitNormal,
493 scene_rdl2::math::Vec3f const &rayDiffX,
494 scene_rdl2::math::Vec3f const &rayDiffY)
495{
496 MNRY_ASSERT(!tls->mRayVertexStack.empty());
497
498 tls->mRayVertexStack.push_back(tls->mRayRecorder->extendRay(tls->mRayVertexStack.back()));
499
500 DebugRayVertex *vert = tls->mRayVertexStack.back();
501 vert->mHitPoint = hitPoint;
502 vert->mNormal = hitNormal;
503 vert->mRayDiffX = rayDiffX;
504 vert->mRayDiffY = rayDiffY;
505
506#ifdef DEBUG
507 // Debug functionality to isolate individual vertices of a ray differential.
508 // You can find the coordinate of a debug ray mesh in in mm_view and set
509 // vertToFind to that value to retrieve the callstack for how we got there.
510 static scene_rdl2::math::Vec3f vertToFind = scene_rdl2::math::Vec3f(FLT_MAX, FLT_MAX, FLT_MAX);
511
512 if (vertToFind != scene_rdl2::math::Vec3f(FLT_MAX, FLT_MAX, FLT_MAX)) {
513
514 scene_rdl2::math::Vec3f halfDx = (rayDiffX - hitPoint) * 0.5f;
515 scene_rdl2::math::Vec3f halfDy = (rayDiffY - hitPoint) * 0.5f;
516 scene_rdl2::math::Vec3f p0 = hitPoint - halfDx - halfDy;
517 scene_rdl2::math::Vec3f p1 = hitPoint + halfDx - halfDy;
518 scene_rdl2::math::Vec3f p2 = hitPoint - halfDx + halfDy;
519 scene_rdl2::math::Vec3f p3 = hitPoint + halfDx + halfDy;
520
521 scene_rdl2::math::Vec3f hit = transformPoint(DebugRayRecorder::getRenderToWorldMatrix(), hitPoint);
522
523 p0 = transformPoint(DebugRayRecorder::getRenderToWorldMatrix(), p0);
524 p1 = transformPoint(DebugRayRecorder::getRenderToWorldMatrix(), p1);
525 p2 = transformPoint(DebugRayRecorder::getRenderToWorldMatrix(), p2);
526 p3 = transformPoint(DebugRayRecorder::getRenderToWorldMatrix(), p3);
527
528 const float eps = 0.01f;
529
530 if (isEqual(vertToFind, hit, eps) ||
531 isEqual(vertToFind, p0, eps) ||
532 isEqual(vertToFind, p1, eps) ||
533 isEqual(vertToFind, p2, eps) ||
534 isEqual(vertToFind, p3, eps)) {
535 scene_rdl2::logging::Logger::info("Hit debug ray vertex at " , vertToFind ,
536 ", hit dx = " , (rayDiffX - hitPoint) ,
537 ", hit dy = " , (rayDiffY - hitPoint));
538 }
539 }
540#endif

544struct ExtendDebugRay
545{
546 // assumes transferred ray
547 finline ExtendDebugRay(pbr::TLState *pbrTls,
548 mcrt_common::RayDifferential const &ray, shading::Intersection const &isect) :
549 mTls(pbrTls)
550 {
551 if (DebugRayRecorder::isRecordingEnabled()) {
552 addRayVertex(mTls,
553 ray.getOrigin(),
554 isect.getN(),
555 ray.hasDifferentials() ? ray.getOriginX() : ray.getOrigin(),
556 ray.hasDifferentials() ? ray.getOriginY() : ray.getOrigin());
557 }
558 }
559
560 // assumes untransferred ray
561 finline ExtendDebugRay(pbr::TLState *pbrTls,
562 mcrt_common::RayDifferential const &ray, float dist) :
563 mTls(pbrTls)
564 {
565 if (DebugRayRecorder::isRecordingEnabled()) {
566 MNRY_ASSERT(dist < sDistantLightDistance);
567 mcrt_common::RayDifferential rd = ray;
568 rd.transfer(ray.getOrigin() + ray.getDirection() * dist, -ray.getDirection());
569 addRayVertex(mTls,
570 rd.getOrigin(),
571 -rd.getDirection(),
572 rd.hasDifferentials() ? rd.getOriginX() : rd.getOrigin(),
573 rd.hasDifferentials() ? rd.getOriginY() : rd.getOrigin());
574 }
575 }
576
577 // overload version of "with hit" constructor which accepts a mcrt_common::Ray
578 // instead of a mcrt_common::RayDifferential
579 finline ExtendDebugRay(pbr::TLState *pbrTls,
580 mcrt_common::Ray const &ray, shading::Intersection const &isect) :
581 mTls(pbrTls)
582 {
583 if (DebugRayRecorder::isRecordingEnabled()) {
584 scene_rdl2::math::ReferenceFrame fr(isect.getN());
585 float dist = ray.getEnd();
586 float scale = dist * 0.1f;
587 addRayVertex(mTls,
588 isect.getP(),
589 isect.getN(),
590 isect.getP() + fr.getX() * scale,
591 isect.getP() + fr.getY() * scale);
592 }
593 }
594
595 // overload version of "NO_HIT" constructor which accepts a mcrt_common::Ray
596 // instead of a mcrt_common::RayDifferential
597 finline ExtendDebugRay(pbr::TLState *pbrTls,
598 mcrt_common::Ray const &ray, float dist) :
599 mTls(pbrTls)
600 {
601 if (DebugRayRecorder::isRecordingEnabled()) {
602 MNRY_ASSERT(dist < sDistantLightDistance);
603 scene_rdl2::math::Vec3f hitPoint = ray.getOrigin() + ray.getDirection() * dist;
604 scene_rdl2::math::Vec3f hitNormal = -ray.getDirection();
605 scene_rdl2::math::ReferenceFrame fr(hitNormal);
606 float scale = dist * 0.1f;
607 addRayVertex(mTls,
608 hitPoint,
609 hitNormal,
610 hitPoint + fr.getX() * scale,
611 hitPoint + fr.getY() * scale);
612 }
613 }
614
615
616 finline ~ExtendDebugRay()
617 {
618 if (DebugRayRecorder::isRecordingEnabled()) {
619 mTls->mRayVertexStack.pop_back();
620 MNRY_ASSERT(!mTls->mRayVertexStack.empty());
621 }
622 }

624 pbr::TLState *mTls;
625};
