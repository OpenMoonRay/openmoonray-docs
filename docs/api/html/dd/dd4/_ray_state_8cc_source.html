23bool
24isRayStateListValid(pbr::TLState *tls, unsigned numEntries, RayState **entries)
25{
26 MNRY_ASSERT(entries);
27
28 // Verify there are no duplicate pointers in the list.
29 scene_rdl2::alloc::Arena *arena = tls->mArena;
30 SCOPED_MEM(arena);
31
32 RayState **p = arena->allocArray<RayState *>(numEntries);
33 memcpy(p, entries, numEntries * sizeof(RayState *));
34 std::sort(p, p + numEntries);
35 MNRY_ASSERT(scene_rdl2::util::isSortedAndUnique(numEntries, p));
36
37 for (unsigned i = 0; i < numEntries; ++i) {
38 MNRY_ASSERT(isValid(p[i]));
39 }
40
41 return true;
42}
