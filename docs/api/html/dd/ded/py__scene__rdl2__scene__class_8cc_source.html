58 PySceneClass_declareAttribute(rdl2::SceneClass& self,
59 const std::string& name,
60 const T& defaultValue,
61 rdl2::AttributeFlags flags,
62 rdl2::SceneObjectInterface objectType,
63 // NOTE: must be a list of strings
64 bp::list& aliasesPyList)

67 // If aliasesPyList is not empty, convert it to a std::vector<std::string>

72 conversions::PyPrimitiveContainerToStdVector<std::string>(aliasesPyList)
73 :
74 std::vector<std::string>{ };
75
76 //-------------------------------------------
77 // Call the actual rdl2::SceneClass::declareAttribute<T>() method
78 //-------------------------------------------
79
80 return self.declareAttribute<T>(name,
81 defaultValue,
82 flags,
83 objectType,
84 aliases);;
