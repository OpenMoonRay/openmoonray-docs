22/**
23 * The RenderOutput defines the "what", "where", and "how" a rendering
24 * result (i.e. AOV) is requested and placed into output. There can
25 * be (and in fact are expected to be) multiple RenderOutput objects
26 * per SceneContext. For example, There might be a RenderOutput object for
27 * the beauty render, a handful of light paths, and a diagnostic heat map.
28 */
29class RenderOutput: public SceneObject
30{
31public:
32 typedef SceneObject Parent;
33
34 /// Type that defines how the result should be encoded. This
35 /// includes Bit depth and Type.
36 enum ChannelFormat {
37 CHANNEL_FORMAT_FLOAT = 0, ///< 32 bit linear floats
38 CHANNEL_FORMAT_HALF ///< 16 bit linear half floats
39 };
40
41 /// Type that defines the image compression scheme. Image
42 /// compression is a per-file/filePart attribute. All RenderOutput
43 /// objects that target the same output image must specify the
44 /// same compression scheme.
45 enum Compression {
46 COMPRESSION_NONE = 0,
47 COMPRESSION_ZIP,
48 COMPRESSION_RLE,
49 COMPRESSION_ZIPS,
50 COMPRESSION_PIZ,
51 COMPRESSION_PXR24,
52 COMPRESSION_B44,
53 COMPRESSION_B44A,
54 COMPRESSION_DWAA,
55 COMPRESSION_DWAB
56 };
57
58 /// The list of result (i.e. AOV) types.
59 enum Result {
60 /// RGB full color render.
61 RESULT_BEAUTY = 0,
62 /// SCALAR full render alpha.
63 RESULT_ALPHA,
64 /// SCALAR full render depth result (camera space)
65 RESULT_DEPTH,
66 /// VEC3, VEC2, or SCALAR depending on Statevariable type
67 RESULT_STATE_VARIABLE,
68 /// RGB, VEC3, VEC2, or SCALAR depending on prim attr type
69 RESULT_PRIMITIVE_ATTRIBUTE,
70 /// SCALAR time per pixel heat map
71 RESULT_HEAT_MAP,
72 /// RGB wireframe render
73 RESULT_WIREFRAME,
74 /// RGB, VEC3, VEC2, or SCALAR
75 RESULT_MATERIAL_AOV,
76 /// RGB
77 RESULT_LIGHT_AOV,
78 /// SCALAR fraction of light samples that hit light
79 RESULT_VISIBILITY_AOV,
80 /// Variance for any AOV result type
81 RESULT_VARIANCE,
82 /// Weight
83 RESULT_WEIGHT,
84 /// RenderBuffer auxiliary sample data (ODD sample) for adaptive sampling
85 RESULT_BEAUTY_AUX,
86 // Cryptomatte
87 RESULT_CRYPTOMATTE,
88 /// Alpha auxiliary sample data (ODD sample) for adaptive sampling
89 RESULT_ALPHA_AUX,
90 /// Display Filter
91 RESULT_DISPLAY_FILTER,
92 };
93
94 /// If the result type is state variable, this enum defines
95 /// the variable. These are all built-in state variables.
96 enum StateVariable {
97 /// VEC3 position
98 STATE_VARIABLE_P = 0,
99 /// VEC3 geometry normal
100 STATE_VARIABLE_NG,
101 /// VEC3 shading normal
102 STATE_VARIABLE_N,
103 /// VEC2 texture st
104 STATE_VARIABLE_ST,
105 /// VEC3 dpds
106 STATE_VARIABLE_DPDS,
107 /// VEC3 dpdt
108 STATE_VARIABLE_DPDT,
109 /// SCALAR dsdx
110 STATE_VARIABLE_DSDX,
111 /// SCALAR dsdy
112 STATE_VARIABLE_DSDY,
113 /// SCALAR dtdx
114 STATE_VARIABLE_DTDX,
115 /// SCALAR dtdy
116 STATE_VARIABLE_DTDY,
117 /// VEC3 world position
118 STATE_VARIABLE_WP,
119 /// Z-DEPTH (-1.f * P.z)
120 STATE_VARIABLE_DEPTH,
121 /// 2D motion vector
122 STATE_VARIABLE_MOTION,
123 };
124
125 /// If the result is primitive attribute, what is the type
126 /// of the primitive attribute? Primitive attributes can
127 /// share the same name and only be disambiguated via type
128 enum PrimitiveAttributeType {
129 PRIMITIVE_ATTRIBUTE_TYPE_FLOAT = 0,
130 PRIMITIVE_ATTRIBUTE_TYPE_VEC2F,
131 PRIMITIVE_ATTRIBUTE_TYPE_VEC3F,
132 PRIMITIVE_ATTRIBUTE_TYPE_RGB
133 };
134
135 enum MathFilter {
136 MATH_FILTER_AVG = 0,
137 MATH_FILTER_SUM,
138 MATH_FILTER_MIN,
139 MATH_FILTER_MAX,
140 MATH_FILTER_FORCE_CONSISTENT_SAMPLING,
141 MATH_FILTER_CLOSEST
142 };
143
144 /// How should channel suffix names be chosen?
145 enum SuffixMode {
146 SUFFIX_MODE_AUTO = 0, /// choose a reaonable suffix based on the output
147 SUFFIX_MODE_RGB, /// always use .RGB
148 SUFFIX_MODE_XYZ, /// always use .XYZ
149 SUFFIX_MODE_UVW, /// always use .UVW
150 SUFFIX_MODE_NUM_MODES
151 };
152
153 /// How should this output be used as an input to the optix denoiser?
154 enum DenoiserInput {
155 DENOISER_INPUT_NONE = 0, /// not a denoiser input
156 DENOISER_INPUT_ALBEDO, /// use this output as the albedo input
157 DENOISER_INPUT_NORMAL, /// use this output as the normal input
158 };

161 RenderOutput(SceneClass const &sceneClass, std::string const &name);

209
210 /// If the result is "Display Filter" what display
211 /// filter object should be used?
212 const DisplayFilter *getDisplayFilter() const;
213
214 /// What file does this AOV go in?
215 finline String getFileName() const;
216 void setFileName(String const &fileName);
217
218 /// Should this AOV go in an exr sub-image? "" means no sub-image
219 finline String getFilePart() const;
220 void setFilePart(String const &filePart);

222 /// What image compression scheme should the file/file part use?
223 /// All RenderOutput objects that target the same file/file part must
224 /// specify the same compression - compression cannot vary per channel.

226 void setCompression(Compression compression);

242 /// What is the channel format, bit depth and type

259 void setDenoise(bool f);

261 /// What file does this AOV go in when checkpoint enable case
262 finline String getCheckpointFileName() const;
263 void setCheckpointFileName(String const &fileName);

267 void setCheckpointMultiVersionFileName(String const &fileName);

269 /// What file is used for resume render input as start condition
270 finline String getResumeFileName() const;
271 void setResumeFileName(String const &fileName);

273 // Cryptomatte depth & number of layers
274 int getCryptomatteDepth() const;
275 int getCryptomatteNumLayers() const;
276
277 /// Returns the camera to use for this output, or nullptr if not specified.
278 const Camera* getCamera() const;
