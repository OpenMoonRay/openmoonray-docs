62// Checks if an unsigned 64 bit number is a power of 4 or not.
63bool isPowerOf4(uint64_t n)
64{
65 // Check that it's a power of 2. This is technically not necessary
66 // as this function is used in context where n is guaranteed to be a power of 2.
67 if (n == 0 || (n & (n - 1)) != 0) { return false; }
68 // Check that the number of trailing zeroes is even:
69 return n & 0x5555555555555555ULL;
70}

117 // Subdivides the strata. This requires updating the tree again, so we need to pass the samples along.
118 void subdivideStrata(const std::vector<Sample>& samples)
119 {
120 // Subdividing the strata doubles the number of samples: ({1x1}, {1x2, 2x1}, {2x2, 1x4, 4x1}, ...)
121 m_numSamples *= 2;
122 // The dimension only changes when we swap from a power of 4 (that is, after the "square" strata).
123 // So, dim is: (1, 2, 2, 4, 4, 8, 8, ...)
124 if (!isPowerOf4(m_numSamples)) {
125 m_dim *= 2;
126 }
127
128 // At each level of the stratum, we have 1 extra level:
129 ++m_level;
130 m_strata.unsetResize(m_numSamples * m_level);
131
132 // Then, update the new stratum with the old values.
133 // Might we worth parallelizing this...
134 for (const Sample& s : samples) {
135 updateStrata(s);
136 }
137 }

139 // Generates a sample in a specific stratum, updating the tree structure.
140 Sample genSample(Grid& grid, uint64_t numCandidates, uint64_t x, uint64_t y, RNG& rng, std::vector<uint64_t>& xoffsets, std::vector<uint64_t>& yoffsets)
141 {
142 // First, we get stratum:
143 xoffsets.clear();
144 yoffsets.clear();
145 getValidOffsets(x, y, xoffsets, yoffsets);
146
147 const double width = 1.0 / m_numSamples;
148
149 std::uniform_int_distribution<size_t> xDist(0, xoffsets.size() - 1);
150 std::uniform_int_distribution<size_t> yDist(0, yoffsets.size() - 1);
151
152 // Generate the candidate samples:
153 Sample bestCandidate;
154 double farthestD2 = 0.0;
155 for (uint64_t i = 0; i < numCandidates; ++i) {
156 // Get a random strata:
157 const uint64_t xoffset = xoffsets[xDist(rng)];
158 const uint64_t yoffset = yoffsets[yDist(rng)];
159
160 // Generate a sample point for this strata:
161 const Sample candidate {
162 std::uniform_real_distribution<double>(width * xoffset, width * (xoffset + 1))(rng),
163 std::uniform_real_distribution<double>(width * yoffset, width * (yoffset + 1))(rng)
164 };
165
166 // Get it's nearest distance:
167 double nearestD2 = std::numeric_limits<double>::max();
168 const auto nearFn = [&candidate, &nearestD2](const Sample& s) {
169 nearestD2 = std::min(nearestD2, wrappedDistanceSquared(candidate, s));
170 };
171 grid.visitNeighbors(nearFn, candidate);
172
173 if (nearestD2 > farthestD2) {
174 farthestD2 = nearestD2;
175 bestCandidate = candidate;
176 }
177 }
178
179 grid.add(bestCandidate);
180
181 // Update the existence of this sample:
182 updateStrata(bestCandidate);
183
184 return bestCandidate;
185 }

187private:
188 void updateStrata(const Sample& sample)
189 {
190 // Here, we essentially loop over all of the elementary intervals for the given number of
191 // samples. For instance, if we have 16 samples, we'd loop over: {(16, 1), (8, 2), (4, 4), (2, 8), (1, 16)}.
192 // At each "level", we update the strata with true if the sample belongs to it.
193 for (uint64_t level = 0, nx = m_numSamples, ny = 1;
194 nx >= 1;
195 nx /= 2, ny *= 2, ++level)
196 {
197 const uint64_t x = sample.x * nx;
198 const uint64_t y = sample.y * ny;
199 const uint64_t index = (level * m_numSamples) + (y * nx + x);
200 m_strata.set(index);
201 }
202 }

204 void getXOffsets(uint64_t x, uint64_t y, uint64_t level, std::vector<uint64_t>& xoffsets) const
205 {
206 // Check, at the specific grid index, the elementary interval is filled. We need the "nx" term in order
207 // to do this, we simply have to do: 2^(m_level - level - 1) or 1 << (m_level - level -1):
208 const uint64_t index = (level * m_numSamples) + (y * (1ULL << (m_level - level - 1ULL)) + x);
209 if (!m_strata.isSet(index)) {
210 // We reached the lowest level, so we are done (e.g. for 16 samples: (16, 1))
211 if (level == 0) {
212 xoffsets.emplace_back(x);
213 } else {
214 getXOffsets(x*2, y/2, level-1, xoffsets);
215 getXOffsets(x*2 + 1, y/2, level-1, xoffsets);
216 }
217 }
218 }
219
220 void getYOffsets(uint64_t x, uint64_t y, uint64_t level, std::vector<uint64_t>& yoffsets) const
221 {
222 // Same as above, basically, only we work backwards. This is because of the ordering of the elementary
223 // intervals in m_strata (see updateStrata for details).
224 // Only, we need to keep track of the number of columns ourselves. That is,
225 // if nx == 1, then we are done (e.g. for 16 samples: (1, 16))
226 const uint64_t nx = 1ULL << (m_level - level - 1ULL);
227 const uint64_t index = (level * m_numSamples) + (y * nx + x);
228 if (!m_strata.isSet(index)) {
229 if (nx == 1) {
230 yoffsets.emplace_back(y);
231 } else {
232 getYOffsets(x/2, y*2, level+1, yoffsets);
233 getYOffsets(x/2, y*2 + 1, level+1, yoffsets);
234 }
235 }
236 }

238 void getValidOffsets(uint64_t x, uint64_t y, std::vector<uint64_t>& xoffsets, std::vector<uint64_t>& yoffsets) const
239 {
240 // If we have an even number of levels, we don't have a square elementary interval. To handle that,
241 // we must half one of the grid sides. For instance, if m_level == 4, then {8x1, 4x2, 2x4, 1x8} are the
242 // elementary intervals. So for x, we wish to start at (4x2), and at (2x4) for y.
243 if (m_level % 2 == 0) {
244 getXOffsets(x, y/2, m_level/2 - 1, xoffsets);
245 getYOffsets(x/2, y, m_level/2, yoffsets);
246 } else {
247 getXOffsets(x, y, m_level/2, xoffsets);
248 getYOffsets(x, y, m_level/2, yoffsets);
249 }
250 }

255 uint64_t m_level; // The current level we are dealing with.

259std::vector<Sample> genSamples(uint64_t numSamples, uint64_t numCandidates, RNG& rng)
260{
261 std::vector<Sample> samples;
262 samples.reserve(numSamples);
263
264 Grid grid;
265
266 ProgressBar progress(numSamples);
267
268 // Keep memory allocated here so we don't keep reallocating memory.
269 std::vector<uint64_t> xoffsets, yoffsets;
270
271 // Generate initial sampling point:
272 Strata strata;
273 samples.emplace_back(strata.genSample(grid, numCandidates, 0, 0, rng, xoffsets, yoffsets));
274
275 std::bernoulli_distribution coinflip(0.5);
276
277 for (uint64_t n = 1; n < numSamples; n *= 4) {
278 strata.subdivideStrata(samples);
279
280 std::uniform_int_distribution<uint64_t> shuffleDist(0, n - 1);
281
282 // Now we go through and calculate the diagonal values at our specific grid level:
283 uint64_t oldShuffle = shuffleDist(rng);
284 for (uint64_t i = 0; (i < n) && (n + i < numSamples); ++i) {
285 const Sample& s = samples[i ^ oldShuffle];
286 const uint64_t x = s.x * strata.getDim();
287 const uint64_t y = s.y * strata.getDim();
288 samples.emplace_back(strata.genSample(grid, numCandidates, x ^ 1, y ^ 1, rng, xoffsets, yoffsets));
289
290 progress.update();
291 progress.draw();
292 }
293
294 strata.subdivideStrata(samples);
295
296 // Now we should have stratum that are square. When picking diagonals here, we use Simon Brown's
297 // rust implementation idea of randomly swapping a value.
298 oldShuffle = shuffleDist(rng);
299 const uint64_t flip = coinflip(rng);
300 for (uint64_t i = 0; (i < n) && (2 * n + i < numSamples); ++i) {
301 const Sample& s = samples[i ^ oldShuffle];
302 const uint64_t x = s.x * strata.getDim();
303 const uint64_t y = s.y * strata.getDim();
304 samples.emplace_back(strata.genSample(grid, numCandidates, x ^ flip, y ^ flip ^ 1, rng, xoffsets, yoffsets));
305
306 progress.update();
307 progress.draw();
308 }
309
310 // Now we find the diagonal of the quads we placed above. To do this, we need to pick a diagonal for
311 // the numbers we just appended. Those happen to be at the 2n position, so we can do [(n << 1) | (i)] to
312 // index these values (it might make more sense to just perform an addition here...)
313 oldShuffle = shuffleDist(rng);
314 for (uint64_t i = 0; (i < n) && (3 * n + i < numSamples); ++i) {
315 const Sample& s = samples[(n << 1) | (i ^ oldShuffle)];
316 const uint64_t x = s.x * strata.getDim();
317 const uint64_t y = s.y * strata.getDim();
318 samples.emplace_back(strata.genSample(grid, numCandidates, x ^ 1, y ^ 1, rng, xoffsets, yoffsets));
319
320 progress.update();
321 progress.draw();
322 }
323 }
324
325 std::cout << std::endl;
326
327 return samples;
328}
