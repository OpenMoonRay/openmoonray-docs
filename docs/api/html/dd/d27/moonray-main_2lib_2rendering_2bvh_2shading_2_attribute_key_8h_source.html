31/// @class AttributeKey
32/// @brief a light weight index with type/name/size info for fast lookup
33/// primitive attributes
34///
35/// An AttributeKey is an index into a table of primitive attributes.
36/// It is very lightweight (just an int) and can be cheaply copied.
37/// The index points into a table of names, data types and data sizes for each key.
38/// The table is a set of static member variables which should only be
39/// modified by StandardAttributes::init() sometime at startup.
40/// It would be desirable to merge this class with scene_rdl2::rdl2::AttributeKey, as they
41/// serve very similar purposes.
42class AttributeKey
43{
44 public:
45 /// Creates an invalid AttributeKey. In general, this should
46 /// only be used when the key will be initialized at a later point.
47 finline AttributeKey();
48 finline AttributeKey(int idx);
49 finline AttributeKey(const AttributeKey &other);
50 finline bool operator==(const AttributeKey &other) const;
51
52 finline int getIndex() const;
53
54 // The following methods look up into the static table via the index
55 // stored in this AttributeKey
56 finline size_t getSize() const;
57 finline const char *getName() const;
58 finline AttributeType getType() const;
59 finline bool hasDerivatives() const;
60
61 finline operator int() const;
62
63 // Invalid AttributeKeys are represented by a -1 index.
64 finline bool isValid() const;
65
66 // Involves locking! only call this on render prepare stage
67 finline bool requestDerivatives() const;
68
69protected:
70 int mIndex;
71
72 // Inserts a key into the table with the given name, type
73 // and size (derived from the type).
74 // if the key already exists, smiply returns index of existing key.
75 template <typename T>
76 static finline int insertKey(const std::string &name, bool requestDerivatives);
77
78 // Looks up key based on name and type and returns index for
79 // that key.
80 // If the key is not found -1 is returned (and asserts in debug mode)
81 template <typename T>
82 static finline int lookup(const std::string &name);
83
84 // Static versions of previous functions
85 static finline bool isValid(int index);
86 static finline const char *getName(AttributeKey key);
87 static finline AttributeType getType(AttributeKey key);
88 static finline size_t getSize(AttributeKey key);
89 static finline bool hasDerivatives(AttributeKey key);
90
91private:
92 static tbb::mutex sRegisterMutex;
93 static std::vector<std::string> sKeyNames;
94 static std::vector<AttributeType> sKeyTypes;
95 static std::vector<size_t> sKeySizes;
96 static std::vector<int8_t> sHasDerivatives;
97 static std::map<std::pair<std::string, AttributeType>, int> sTable;
98 static int sNumKeys;
99};

258bool
259AttributeKey::requestDerivatives() const
260{
261 if (!isValid()) {
262 return false;
263 }
264 // unable to compute differential for non float type attributes
265 AttributeType type = getType();
266 if (type == scene_rdl2::rdl2::TYPE_BOOL || type == scene_rdl2::rdl2::TYPE_INT ||
267 type == scene_rdl2::rdl2::TYPE_LONG || type == scene_rdl2::rdl2::TYPE_STRING) {
268 return false;
269 }
270 {
271 tbb::mutex::scoped_lock lock(sRegisterMutex);
272 sHasDerivatives[mIndex] = 1;
273 }
274 return true;
275}

278template <typename T>
279size_t
280getAttributeTypeSize()
281{
282 AttributeType type = attributeType<T>();
283 // string need special treatment since it's not a plain old data types.
284 // manipulate it in raw memory chunk will cause unwanted destructor call
285 // and seg fault. We use a util pool to manage string object and store the
286 // string pointer instead of string in chunk to avoid this issue
287 // (and save memory usage since we may have repetitive attribute values)
288 if (type == scene_rdl2::rdl2::TYPE_STRING) {
289 return sizeof(std::string*);
290 } else {
291 return sizeof(T);
292 }
293}

295template <typename T>
296int
297AttributeKey::insertKey(const std::string &name, bool requestDerivatives)
298{
299 AttributeType type = attributeType<T>();
300 size_t size = getAttributeTypeSize<T>();
301 std::pair<std::string, AttributeType> lookup(name, type);
302 int index = -1;
303 {
304 tbb::mutex::scoped_lock lock(sRegisterMutex);
305 auto it = sTable.find(lookup);
306 if (it == sTable.end()) {
307 index = static_cast<int>(sKeyNames.size());
308 sTable[lookup] = index;
309 sKeyNames.push_back(name);
310 sKeySizes.push_back(size);
311 sKeyTypes.push_back(type);
312 sHasDerivatives.push_back(0);
313 } else {
314 index = it->second;
315 }
316 if (requestDerivatives &&
317 type != scene_rdl2::rdl2::TYPE_BOOL && type != scene_rdl2::rdl2::TYPE_INT &&
318 type != scene_rdl2::rdl2::TYPE_LONG && type != scene_rdl2::rdl2::TYPE_STRING) {
319 sHasDerivatives[index] = 1;
320 }
321 }
322 return index;
323}

325template <typename T>
326int
327AttributeKey::lookup(const std::string &name)
328{
329 AttributeType type = attributeType<T>();
330 std::pair<std::string, AttributeType> lu(name, type);
331
332 auto it = sTable.find(lu);
333 if (it == sTable.end()) {
334 std::stringstream ss;
335 ss << "Key not found:" << name << " Type:(" << type << "). Add key before using.";
336 MNRY_ASSERT(0, ss.str().c_str());
337 return -1;
338 } else {
339 return it->second;
340 }
341}
