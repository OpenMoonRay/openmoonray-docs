11// Steps for adding new Hybrid Uniform Data (HUD) types which are identical on
12// both the C++ and ISPC side.
13//
14// 1) Create a hybrid header file (*.hh) which will be included by both C++
15// and ISPC headers. This will contain the data members and validation of
16// the hybrid structure.
17// 2) Create the structures on the C++ and ISPC side by including the hybrid
18// file from a C++ and an ISPC header, and defining the structures.
19// 3) To get automatic data layout validation, use the HUD_VALIDATOR() macro
20// defined in <HybridUniformData.h>
21

36// Defining the Hybrid Uniform Data (HUD) members using the following macros:
37//
38// HUD_MEMBER(member_type, member_name)
39// Add a member (POD or structure) to both the C++ and ISPC structures.
40// member_type must be known for both contexts.
41//
42// HUD_CPP_MEMBER(member_type, member_name, size_in_bytes)
43// Add a member to the C++ structure, and insert padding on the ISPC side.
44// member_type must be known on the C++ side.
45// size_in_bytes *must* be at least 2.
46//
47// HUD_PTR(member_type, member_name)
48// Add a member to both the C++ and ISPC structures. member_type
49// must be known or be forward declared for both contexts.
50//
51// HUD_CPP_PTR(member_type, member_name)
52// Add a member to the C++ structure, and insert padding on the ISPC side.
53// member_type must be known or be forward declared on the C++ side.
54//
55// HUD_ISPC_PTR(member_type, member_name) and HUD_ISPC_FNPTR()
56// Add a member to the ISPC structure, and insert padding on the C++ side.
57// member_type must be known or be forward declared on the ISPC side. For
58// function pointers, use HUD_ISPC_FNPTR() instead, to avoid an ISPC bug in
59// the auto-generated C++ header.
60//
61// HUD_ARRAY(member_type, member_name, num_elems)
62// Add a 1-dimensional array to both the C++ and ISPC structures. member_type
63// must be known for both contexts. Note: we don't support arrays of 2 or more
64// dimensions currently, but it's an easy addition when needed.
65//
66// HUD_CPP_ARRAY(member_type, member_name, num_elems, total_size_in_bytes)
67// Add a 1-dimensional array the C++ structure, and insert padding on the ISPC side.
68// member_type must be known on the C++ side. Note: we don't support arrays of
69// 2 or more dimensions currently, but it's an easy addition when needed.
70//
71// HUD_CPP_PAD(member_name, size_in_bytes)
72// Use if padding is needed only on the C++ side.
73//
74// HUD_ISPC_PAD(member_name, size_in_bytes)
75// Use if padding is needed only on the ISPC side.
76//
77// HUD_ALLOW_VALIDATION()
78// This adds a friend declaration internally so that offsets may be verified at
79// startup, even if they are declared potected or private.
80//
81// HUD_VIRTUAL_BASE_CLASS()
82// Use this to tell the ISPC side structure that the corresponding C++ class
83// has a vtable.
84//
85// HUD_PUBLIC()
86// Use this to give the C++ members public access.
87//
88// HUD_PROTECTED()
89// Use this to give the C++ members protected access.
90//
91// HUD_PRIVATE()
92// Use this to give the C++ members private access.
93//
94

149// The following are a set of macros that handle hybrid uniform data layout
150// validation across C++ / ISPC types. It uses a CRC based on the offset of
151// each validated data member within the type as well as the size of the
152// type. See the HUD_VALIDATOR() macro in <HybridUniformData.h> for more details.
153
154#define HUD_UPDATE_CRC() \
155 crc = crc ^ (ofs << 13); \
156 crc = (crc >> 1) | ((crc & 0x1) << 31)
157
158#ifdef ISPC
159
160 #define HUD_BEGIN_VALIDATION(type_name) \
161 uniform uint32_t crc = 0xffaaf0af; \
162 uniform uint32_t ofs = 0; \
163 if (verbose) print(#type_name " (ISPC):\n")
164
165 #define HUD_VALIDATE(type_name, member_name) \
166 ofs = (uniform uint32_t)((uniform intptr_t)(&((((uniform type_name *uniform)(0))->member_name)))); \
167 if (verbose) { \
168 print(" " #member_name ": % / %\n", ofs, sizeof(uniform type_name)); \
169 } \
170 ofs += sizeof(uniform type_name); \
171 HUD_UPDATE_CRC()
172
173 #define HUD_END_VALIDATION \
174 if (verbose) print(" CRC = %\n", crc); \
175 return crc
176
177#else
178
179 #define HUD_BEGIN_VALIDATION(type_name) \
180 uint32_t crc = 0xffaaf0af; \
181 uint32_t ofs = 0; \
182 if (verbose) printf(#type_name " (C++):\n")
183
184 #define HUD_VALIDATE(type_name, member_name) \
185 ofs = (uint32_t)((intptr_t)(&((((type_name *)(0))->member_name)))); \
186 if (verbose) { \
187 printf(" " #member_name ": %d / %d\n", \
188 (int)ofs, (int)sizeof(type_name)); \
189 } \
190 ofs += (uint32_t)sizeof(type_name); \
191 HUD_UPDATE_CRC()
192
193 #define HUD_END_VALIDATION \
194 if (verbose) printf(" CRC = %u\n", crc); \
195 return crc
196
197#endif
