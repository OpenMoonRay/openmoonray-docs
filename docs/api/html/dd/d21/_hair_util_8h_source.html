23// Utility Function For Converting std::pow() into a set of multiplications
24// From PBRTv3: pow() is usually implemented as : exp(log(a)*power)
25// These transcendental functions -exp and log - are, usually, much slower than multiplications.
26// This provides the motivation to convert r^k where k is known at compile time into a set of multiplications.
27// In the hair code, we calculate r^20 and r^22 to convert roughness into variance for linear perception.
28template<int n>
29static float Pow(float v)
30{
31 float n2 = Pow<n/2>(v);
32 return n2 * n2 * Pow<n & 1>(v);
33}
34template<> float Pow<0>(float v) { return 1; }
35template<> float Pow<1>(float v) { return v; }
36

69 // From [2] Section 4.1
70 // Converts roughness (0,1) into longitudinal variance to linearize
71 // the visual response for the *Energy Conserving Longitudinal*.
72 static float
73 longitudinalVar(float r)
74 {
75// MNRY_ASSERT(r > 0.0f && r < 1.0f);
76 float v = scene_rdl2::math::sqr(0.726f * r + 0.812f * r * r + 3.7f * Pow<20>(r));
77 return v;
78 }

80 // From [2] Section 4.1
81 // Convert roughness (0,1) into azimuthal variance to linearize
82 // the visual response for the *Trimmed Logistic Function*.
83 static float
84 azimuthalVar(float r)
85 {
86// MNRY_ASSERT(r > 0.0f && r < 1.0f);
87 float v = sSqrtPiOver8*(0.265f * r + 1.194f * r * r +
88 5.372f * Pow<22>(r));
89 return scene_rdl2::math::max(v, 0.05f);
90 }

92 // From [2], Section 4.2
93 // Convert HairColor to Absorption Coeffs
94 static scene_rdl2::math::Color
95 computeAbsorptionCoefficients(const scene_rdl2::math::Color &hairColor,
96 const float aziRoughness)
97 {
98 scene_rdl2::math::Color sigmaA = scene_rdl2::math::sBlack;
99 const float denom = 5.969f - 0.215f * aziRoughness + 2.532f * scene_rdl2::math::sqr(aziRoughness) -
100 10.73f * Pow<3>(aziRoughness) + 5.574f * Pow<4>(aziRoughness) +
101 0.245f * Pow<5>(aziRoughness);
102 if (!scene_rdl2::math::isZero(denom)) {
103 for (int i = 0; i < 3; ++i) {
104 sigmaA[i] = scene_rdl2::math::sqr(scene_rdl2::math::log(scene_rdl2::math::max(hairColor[i], scene_rdl2::math::sEpsilon)) / denom);
105 }
106 }
107 return sigmaA;
108 }

110 // From [1], Section 6
111 static float
112 logBesselIO(float x)
113 {
114 if (x > 12.0f) {
115 const float oneOverX = 1.0f / x;
116 return x + 0.5f * (-scene_rdl2::math::log(scene_rdl2::math::sTwoPi) + scene_rdl2::math::log(oneOverX + 0.125f*oneOverX));
117 }
118 else {
119 return scene_rdl2::math::log(besselIO(x));
120 }
121 }
122
123 // From [1], Section 6
124 static float
125 besselIO(float x)
126 {
127 float sum = 0.0f;
128 float x2i = 1.0f;
129 int ifact = 1;
130 int i4 = 1;
131 for (int i = 0; i < 10; ++i) {
132 if (i > 1) ifact *= i;
133 sum += x2i / i4 / ifact / ifact;
134 x2i *= x * x;
135 i4 *= 4;
136 }
137 return sum;
138 }

141 // From [1], Section 4.2
142 // Graph Comparison of Gaussian Vs This Function:
143 // https://www.desmos.com/calculator/xuok8nijyn
144 // Proof that this integrates to one:
145 // WolframAlpha Link - https://tinyurl.com/m7bceaj
146 static float
147 deonLongitudinalM(float variance,
148 float sinThetaI, float cosThetaI,
149 float sinThetaO, float cosThetaO)
150 {
151 if (variance < scene_rdl2::math::sEpsilon)
152 return 0.0f;
153
154 float oneOverV = 1.0f / variance;
155 float b = sinThetaI*sinThetaO*oneOverV;
156 float x = cosThetaI*cosThetaO*oneOverV;
157
158 float result = 1.0f;
159 if (variance <= 0.1) {
160 result = scene_rdl2::math::exp(logBesselIO(x) - b - oneOverV + 0.6931f + scene_rdl2::math::log(0.5f*oneOverV));
161 } else {
162 result = 0.5f*oneOverV*scene_rdl2::math::exp(-b)*besselIO(x) / scene_rdl2::math::sinh(oneOverV);
163 }
164 return result;
165 }

167 // From [2], Appendix A
168 // https://en.wikipedia.org/wiki/Logistic_distribution
169 static float
170 logisticFunction(float x,
171 float s)
172 {
173 x = scene_rdl2::math::abs(x);
174 const float r = -x/s;
175 return scene_rdl2::math::exp(r) / (s * scene_rdl2::math::sqr(1 + scene_rdl2::math::exp(r)));
176 }
177
178 // From [2], Appendix A
179 // https://en.wikipedia.org/wiki/Logistic_distribution
180 static float
181 logisticCDF(float x,
182 float s)
183 {
184 return 1.0f / (1.0f + scene_rdl2::math::exp(-x/s));
185 }

187 // From [2], Appendix A
188 // Note - Appendix A has a mistake in the normalization factor
189 // which we've corrected here
190 static float
191 trimmedLogisticFunction(float x,
192 float s,
193 float a,
194 float b)
195 {
196 return logisticFunction(x, s) / (logisticCDF(b,s) - logisticCDF(a,s));
197 }
198
199 // This follows from the CDF derivation, not explained in Appendix A
200 // Comments for the Derivation Inline
201 static float
202 sampleTrimmedLogistic(float r,
203 float s,
204 float a,
205 float b)
206 {
207 // Normalization Factor
208 const float k = logisticCDF(b, s) - logisticCDF(a, s);
209 // logisticCdf(x, s, a, b) = epsilon
210 // Therefore, 1/k * (logisticCdf(x, s) - logisticCdf(a,s)) = epsilon
211 // Therefore, logisticCdf(x,s) = epsilon * k + logisticCdf(a,s)
212 const float logisticCdfX = (r * k + logisticCDF(a,s));
213 float x = 0.0;
214 if (!scene_rdl2::math::isZero(logisticCdfX)) {
215 // Therefore, 1/(1+exp(-x/s)) = epsilon * k + logisticCdf(a,s) = logisticCdfX
216 // Therefore, exp(-x/s) = 1/logisticCdfX - 1
217 // Therefore, x = -s * log(1/logisticCdfX - 1)
218 x = -s * scene_rdl2::math::log(1.0f/logisticCdfX - 1.0f);
219 }
220 return scene_rdl2::math::clamp(x, a, b);
221 }

223 static float
224 unitGaussianForShade(float stddev, float x)
225 {
226 if (stddev < scene_rdl2::math::sEpsilon)
227 return 0.0f;
228
229 // Normalized Gaussian lobe used at shade time
230 // The denominator normalizes the lobe (otherwise would give a peak value
231 // of 1.0)
232 return scene_rdl2::math::exp(-(x * x) / (2.0f * stddev * stddev)) / (stddev * scene_rdl2::math::sSqrtTwoPi);
233 }
234
235 // NOTE : added for future use in case we ever use Cauchy lobe shading.
236 static float
237 unitCauchyForShade(float stddev, float x)
238 {
239 if (stddev < scene_rdl2::math::sEpsilon)
240 return 0.0f;
241
242 // An unnormalized Cauchy lobe with sigma^2 in the
243 // numerator gives you always a peak value of 1.0
244 float retval = stddev * stddev;
245 return retval / (x*x + retval);
246 }
247
248 static float
249 unitCauchyForSample(float stddev, float x)
250 {
251 if (stddev < scene_rdl2::math::sEpsilon)
252 return 0.0f;
253
254 // A normalized Cauchy used for samples with sigma in
255 // the numerator gives a total integral of 1.0.
256 float retval = stddev * stddev;
257 return stddev / (x*x + retval);
258 }

260 // Code for creating two new random numbers using one
261 // From PBRT Section 4.3.3
262 // https://fgiesen.wordpress.com/2009/12/13/decoding-morton-codes/
263 static uint32_t
264 compact1By1(uint32_t x) {
265 // TODO: as of Haswell, the PEXT instruction could do all this in a
266 // single instruction.
267 // x = -f-e -d-c -b-a -9-8 -7-6 -5-4 -3-2 -1-0
268 x &= 0x55555555;
269 // x = --fe --dc --ba --98 --76 --54 --32 --10
270 x = (x ^ (x >> 1)) & 0x33333333;
271 // x = ---- fedc ---- ba98 ---- 7654 ---- 3210
272 x = (x ^ (x >> 2)) & 0x0f0f0f0f;
273 // x = ---- ---- fedc ba98 ---- ---- 7654 3210
274 x = (x ^ (x >> 4)) & 0x00ff00ff;
275 // x = ---- ---- ---- ---- fedc ba98 7654 3210
276 x = (x ^ (x >> 8)) & 0x0000ffff;
277 return x;
278 }

280 // Create Two Unique Random Numbers using a Morton Curve
281 // https://fgiesen.wordpress.com/2009/12/13/decoding-morton-codes/
282 static scene_rdl2::math::Vec2f
283 demuxFloat(float f) {
284 assert(f >= 0 && f < 1);
285 uint64_t v = f * (1ull << 32);
286 uint32_t bits[2] = {compact1By1(v), compact1By1(v >> 1)};
287 return scene_rdl2::math::Vec2f(bits[0] / float(1 << 16), bits[1] / float(1 << 16));
288 }
