173{
174 scene_rdl2::math::ReferenceFrame frame;
175
176 static const int size = 2;
177 const float positions[size] = {0.f, 1.f};
178 const ispc::RampInterpolatorMode interpolators[size] = {
179 ispc::RAMP_INTERPOLATOR_MODE_SMOOTH,
180 ispc::RAMP_INTERPOLATOR_MODE_SMOOTH };
181 const scene_rdl2::math::Color colors[size] = {
182 scene_rdl2::math::Color(1.f, 0.f, 0.f), scene_rdl2::math::Color(1.f, 0.f, 1.f)};
183
184 printInfo("##### TestBsdfOneSampler::testIridescence() ####################");
185 for (int i=0; i < sRoughnessCount; i++) {
186 printInfo("===== Roughness: %f ====================", sRoughness[i]);
187
188 IridescenceBsdfFactory factory(sRoughness[i], 1.0f, ispc::SHADING_IRIDESCENCE_COLOR_USE_HUE_INTERPOLATION,
189 scene_rdl2::math::Color(1.f, 0.f, 0.f), scene_rdl2::math::Color(1.f, 0.f, 1.f), false,
190 ispc::COLOR_RAMP_CONTROL_SPACE_RGB, size, positions, interpolators, colors,
191 2.f, 1.f, 1.f, 1.f);

493void
494TestBsdfOneSampler::testStochasticFlakes()
495{
496 scene_rdl2::math::ReferenceFrame frame;
497
498 static const size_t flakesDataSize = 100;
499
500 printInfo("##### TestBsdfOneSampler::testStochasticFlakes() ####################");
501 for (int i = 0; i < sRoughnessCount; i++) {
502 printInfo("===== Roughness: %f ====================", sRoughness[i]);
503
504 Vec3Array normals;
505 ColorArray colors;
506 generateGGXNormals(sRoughness[i] * sRoughness[i], flakesDataSize, normals);
507 generateWeightedFlakeColors(flakesDataSize, colors);
508 for (size_t j = 0; j < flakesDataSize; ++j) {
509 normals[j] = frame.localToGlobal(normals[j]);
510 }
511
512 StochasticFlakesBsdfFactory factory(&normals[0], &colors[0], flakesDataSize, 0.1f, sRoughness[i]);
513 TestBsdfSettings test(factory, frame, true, 0.008, 0.03,
514 (sRoughness[i] > 0.1f && sRoughness[i] < 0.4f),
515 (sRoughness[i] > 0.1f && sRoughness[i] < 0.4f),
516 TestBsdfSettings::BSDF_ONE_SAMPLER);
517 runTest(test, sViewAnglesTheta, 1, getSampleCount(sRoughness[i]));
518
519 normals.clear();
520 colors.clear();
521 }
522}
