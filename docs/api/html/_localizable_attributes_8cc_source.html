21LocalizableAttributes::LocalizableAttributes(const rdl2::SceneContext& ctx) :
22 mRecords()
23{
24 // Walk all the SceneClasses looking for filename attributes.
25 for (auto classIter = ctx.beginSceneClass(); classIter != ctx.endSceneClass(); ++classIter) {
26 const rdl2::SceneClass* sc = classIter->second;
27 for (auto attrIter = sc->beginAttributes(); attrIter != sc->endAttributes(); ++attrIter) {
28 const rdl2::Attribute* attr = *attrIter;
29 if (attr->isFilename()) {
30 mRecords.emplace(sc, attr);
31 }
32 }
33 }
34}
