25class TileScheduler
26{
27public:
28 enum Type
29 {
30 TOP, // 0
31 BOTTOM, // 1
32 LEFT, // 2
33 RIGHT, // 3
34 MORTON, // 4
35 RANDOM, // 5
36 SPIRAL_SQUARE, // 6
37 SPIRAL_RECT, // 7
38 NUM_TILE_SCHEDULER_TYPES,
39 };
40
41 explicit TileScheduler(Type type);
42 virtual ~TileScheduler() = default;
43
44 Type getType() const { return mType; }
45
46 // The width and height parameters should be the unaligned original render
47 // buffer extents. Viewport is either the full extents of the buffer or a
48 // sub-viewport.
49 // The renderNodeIdx and numRenderNodes parameters only need to be set for
50 // distributed rendering. Returns the number of tiles generated.
51 unsigned generateTiles(scene_rdl2::alloc::Arena *arena,
52 unsigned width,
53 unsigned height,
54 const scene_rdl2::math::Viewport &viewport,
55 unsigned renderNodeIdx = 0,
56 unsigned numRenderNodes = 1,
57 unsigned taskDistribType = 0);
58
59 // Returns true if tiles are split up amongst rendering nodes.
60 bool isDistributed() const { return mNumRenderNodes > 1; }
61 unsigned taskDistribType() const { return mTaskDistribType; } // Film::TaskDistribType
62
63 unsigned getRenderNodeIdx() const { return mRenderNodeIdx; }
64
65 // Returns cached minimal set of tiles for viewport passed in.
66 const std::vector<scene_rdl2::fb_util::Tile> &getTiles() const { return mTiles; }
67
68 // Returns the permuted order of the tile indices
69 const uint32_t* getTileIndices() const { return mTileIndices.get(); }
70
71 // Factory function.
72 static std::unique_ptr<TileScheduler> create(TileScheduler::Type type);
73
74protected:
75 // User must fill out (numTilesX * numTilesY) linear tile indices.
76 // Index zero is assumed to be the bottom left of the grid.
77 // Each cell in the grid should be filled out with the order you want
78 // that tile rendered in. So for example, if you put 15 in the bottom left
79 // cell, you are saying that you want it to be the 15th tile rendered.
80 virtual void generateTileIndices(scene_rdl2::alloc::Arena *arena,
81 unsigned numTilesX,
82 unsigned numTilesY,
83 uint32_t *tileIndices,
84 uint32_t seed) const = 0;
85
86 unsigned mRenderNodeIdx;
87 unsigned mNumRenderNodes;
88 Type mType;
89
90 std::vector<scene_rdl2::fb_util::Tile> mTiles;
91 std::unique_ptr<uint32_t[]> mTileIndices;
92
93 unsigned mTaskDistribType; // Film::TaskDistribType
94};
