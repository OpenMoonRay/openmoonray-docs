43// Returns the number of BundledRadiance entries filled in.
44unsigned
45areSingleRaysOccluded(pbr::TLState *pbrTls, unsigned numEntries, BundledOcclRay **entries,
46 BundledRadiance *results, RayHandlerFlags flags)
47{
48 const FrameState &fs = *pbrTls->mFs;
49 const rt::EmbreeAccelerator *accel = fs.mEmbreeAccel;
50 const bool disableShadowing = !fs.mIntegrator->getEnableShadowing();
51 unsigned numRadiancesFilled = 0;
52
53 for (unsigned i = 0; i < numEntries; ++i) {
54 BundledOcclRay &occlRay = *entries[i];
55
56 MNRY_ASSERT(occlRay.isValid());
57
58 mcrt_common::Ray rtRay;
59
60 rtRay.org[0] = occlRay.mOrigin.x;
61 rtRay.org[1] = occlRay.mOrigin.y;
62 rtRay.org[2] = occlRay.mOrigin.z;
63 rtRay.dir[0] = occlRay.mDir.x;
64 rtRay.dir[1] = occlRay.mDir.y;
65 rtRay.dir[2] = occlRay.mDir.z;
66 rtRay.tnear = occlRay.mMinT;
67 rtRay.tfar = occlRay.mMaxT;
68 rtRay.time = occlRay.mTime;
69 rtRay.mask = scene_rdl2::rdl2::SHADOW;
70 rtRay.geomID = RT_INVALID_RAY_ID;
71 rtRay.ext.instance0OrLight = static_cast<BundledOcclRayData *>(
72 pbrTls->getListItem(occlRay.mDataPtrHandle, 0))->mLight->getRdlLight();
73
74 bool isOccluded;
75 {
76 MNRY_ASSERT(occlRay.mOcclTestType == OcclTestType::STANDARD);
77 EXCL_ACCUMULATOR_PROFILE(pbrTls, EXCL_ACCUM_EMBREE_OCCLUSION);
78 isOccluded = accel->occluded(rtRay);
79 }
80
81 if (!isOccluded || disableShadowing) {
82 // At this point, we know that the ray is not occluded, but we still need to
83 // apply volume transmittance to the final radiance value.
84 scene_rdl2::math::Color tr = getTransmittance(pbrTls, occlRay);
85 occlRay.mRadiance = occlRay.mRadiance * tr;
86 BundledRadiance *result = &results[numRadiancesFilled++];
87 fillBundledRadiance(pbrTls, result, occlRay);
88
89 // LPE
90 if (occlRay.mDataPtrHandle != nullHandle) {
91 EXCL_ACCUMULATOR_PROFILE(pbrTls, EXCL_ACCUM_AOVS);
92
93 const int numItems = pbrTls->getNumListItems(occlRay.mDataPtrHandle);
94 accumLightAovs(pbrTls, occlRay, fs, numItems, scene_rdl2::math::sWhite, &tr,
95 AovSchema::sLpePrefixUnoccluded);
96 accumVisibilityAovs(pbrTls, occlRay, fs, numItems, reduceTransparency(tr));
97 }
98 } else {
99 // LPE: visibility aovs when we don't hit light
100 if (occlRay.mDataPtrHandle != nullHandle) {
101
102 const Light *light = static_cast<BundledOcclRayData *>(
103 pbrTls->getListItem(occlRay.mDataPtrHandle, 0))->mLight;
104
105 // see PathIntegrator::addDirectVisibleLightSampleContributions()
106 if (light->getClearRadiusFalloffDistance() != 0.f &&
107 occlRay.mMaxT < light->getClearRadius() + light->getClearRadiusFalloffDistance()) {
108 scene_rdl2::math::Color tr = getTransmittance(pbrTls, occlRay);
109 occlRay.mRadiance = calculateShadowFalloff(light, occlRay.mMaxT, tr * occlRay.mRadiance);
110 BundledRadiance *result = &results[numRadiancesFilled++];
111 fillBundledRadiance(pbrTls, result, occlRay);
112 }
113
114 EXCL_ACCUMULATOR_PROFILE(pbrTls, EXCL_ACCUM_AOVS);
115
116 const int numItems = pbrTls->getNumListItems(occlRay.mDataPtrHandle);
117 accumVisibilityAovsOccluded(pbrTls, occlRay, fs, numItems);
118
119 // We only accumulate here if we were occluded but we have the flag on. Otherwise
120 // it would already have been filled by the previous call.
121 if (fs.mAovSchema->hasLpePrefixFlags(AovSchema::sLpePrefixUnoccluded)) {
122 accumLightAovs(pbrTls, occlRay, fs, numItems, scene_rdl2::math::sWhite, nullptr,
123 AovSchema::sLpePrefixUnoccluded);
124 }
125 }
126 }
127
128 // LPE
129 // we are responsible for freeing LPE memory
130 if (occlRay.mDataPtrHandle != nullHandle) {
131 pbrTls->freeList(occlRay.mDataPtrHandle);
132 }
133 pbrTls->releaseDeepData(occlRay.mDeepDataHandle);
134 }
135
136 return numRadiancesFilled;
137}

139// Returns the number of BundledRadiance entries filled in.
140unsigned
141forceSingleRaysUnoccluded(pbr::TLState *pbrTls, unsigned numEntries, BundledOcclRay **entries,
142 BundledRadiance *results, RayHandlerFlags flags)
143{
144 const FrameState &fs = *pbrTls->mFs;
145
146 for (unsigned i = 0; i < numEntries; ++i) {
147 BundledOcclRay &occlRay = *entries[i];
148
149 MNRY_ASSERT(occlRay.isValid());
150 MNRY_ASSERT(occlRay.mOcclTestType == OcclTestType::FORCE_NOT_OCCLUDED);
151
152 // At this point, we know that the ray is not occluded, but we still need to
153 // apply volume transmittance to the final radiance value.
154 scene_rdl2::math::Color tr = getTransmittance(pbrTls, occlRay);
155 occlRay.mRadiance = occlRay.mRadiance * tr;
156
157 BundledRadiance *result = &results[i];
158 fillBundledRadiance(pbrTls, result, occlRay);
159
160 // LPE
161 if (occlRay.mDataPtrHandle != nullHandle) {
162 EXCL_ACCUMULATOR_PROFILE(pbrTls, EXCL_ACCUM_AOVS);
163
164 const int numItems = pbrTls->getNumListItems(occlRay.mDataPtrHandle);
165 accumLightAovs(pbrTls, occlRay, fs, numItems, tr, nullptr, AovSchema::sLpePrefixNone);
166 }
167
168 // LPE
169 // we are responsible for freeing LPE memory
170 if (occlRay.mDataPtrHandle != nullHandle) {
171 pbrTls->freeList(occlRay.mDataPtrHandle);
172 }
173 pbrTls->releaseDeepData(occlRay.mDeepDataHandle);
174 }
175
176 return numEntries;
177}

179// Perform all occlusion checks. All the heavy lifting is pretty much done in this loop!
180// Returns the number of BundledRadiance entries filled in.
181unsigned
182computeOcclusionQueriesBundled(pbr::TLState *pbrTls, unsigned numEntries,
183 BundledOcclRay **entries, BundledRadiance *results,
184 RayHandlerFlags flags)
185{
186 // Update ray stats.
187 pbrTls->mStatistics.addToCounter(STATS_OCCLUSION_RAYS, numEntries);
188 pbrTls->mStatistics.addToCounter(STATS_BUNDLED_OCCLUSION_RAYS, numEntries);
189
190 unsigned totalRadiancesFilled = 0;
191
192 // Sort no-op rays to come after standard rays. This allows us to process all standard rays together
193 // and skip the occlusion test on all no-op rays.
194 BundledOcclRay **noOpEntries = std::partition(entries, entries + numEntries,
195 [](BundledOcclRay *r){return r->mOcclTestType == OcclTestType::STANDARD;});
196
197 // Exclude all the no-op entries so we don't run occlusion tests for them
198 unsigned numStandardEntries = noOpEntries - entries;
199 unsigned numNoOpEntries = numEntries - numStandardEntries;
200 numEntries = numStandardEntries;
201
202 if (numEntries) {
203 unsigned numRadiancesFilled = areSingleRaysOccluded(pbrTls, numEntries, entries, results, flags);
204 results += numRadiancesFilled;
205 totalRadiancesFilled += numRadiancesFilled;
206 }
207
208 // Handle no-op rays.
209 if (numNoOpEntries) {
210 totalRadiancesFilled += forceSingleRaysUnoccluded(pbrTls, numNoOpEntries, noOpEntries, results, flags);
211 }
212
213 return totalRadiancesFilled;
214}

216// Perform all presence shadows checks. All the heavy lifting is pretty much done in this loop!
217// Returns the number of BundledRadiance entries filled in.
218unsigned
219computePresenceShadowsQueriesBundled(pbr::TLState *pbrTls, unsigned int numEntries,
220 BundledOcclRay **entries, BundledRadiance *results,
221 RayHandlerFlags flags)
222{
223 // Presence handling code for direct lighting
224 if (numEntries == 0) {
225 return 0;
226 }
227 const FrameState &fs = *pbrTls->mFs;
228 const bool disableShadowing = !fs.mIntegrator->getEnableShadowing();
229 unsigned int numRadiancesFilled = 0;
230 for (unsigned int i = 0; i < numEntries; ++i) {
231 BundledOcclRay &occlRay = *entries[i];
232 MNRY_ASSERT(occlRay.isValid());
233 // we always have the data block here
234 const BundledOcclRayData *b = static_cast<BundledOcclRayData *>(
235 pbrTls->getListItem(occlRay.mDataPtrHandle, 0));
236
237 mcrt_common::Ray shadowRay(occlRay.mOrigin, occlRay.mDir,
238 occlRay.mMinT, occlRay.mMaxT, occlRay.mTime, occlRay.mDepth);
239
240 float presence = 0.0f;
241 accumulateRayPresence(pbrTls,
242 b->mLight,
243 shadowRay,
244 b->mRayEpsilon,
245 fs.mMaxPresenceDepth,
246 presence);
247
248 // At this point, we know that the ray is not occluded, but we still need to
249 // apply volume transmittance to the final radiance value.
250 scene_rdl2::math::Color tr = getTransmittance(pbrTls, occlRay);
251 occlRay.mRadiance = occlRay.mRadiance * tr;
252
253 if (scene_rdl2::math::isEqual(presence, 0.0f) || disableShadowing) {
254 BundledRadiance *result = &results[numRadiancesFilled++];
255 fillBundledRadiance(pbrTls, result, occlRay);
256 } else {
257 // Presence value indicates light is partially blocked.
258 // Scale radiance by (1 - presence)
259 BundledRadiance *result = &results[numRadiancesFilled++];
260 fillBundledRadiance(pbrTls, result, occlRay);
261 result->mRadiance[0] *= (1 - presence);
262 result->mRadiance[1] *= (1 - presence);
263 result->mRadiance[2] *= (1 - presence);
264 }
265
266 // LPE
267 {
268 EXCL_ACCUMULATOR_PROFILE(pbrTls, EXCL_ACCUM_AOVS);
269
270 const int numItems = pbrTls->getNumListItems(occlRay.mDataPtrHandle);
271
272 // If we are rendering with unoccluded flags we want to ignore presence values when accumulating
273 // them to the aov:
274 const scene_rdl2::math::Color occlusionValue = (1.f - presence) * tr;
275 accumLightAovs(pbrTls, occlRay, fs, numItems, scene_rdl2::math::sWhite, &occlusionValue,
276 AovSchema::sLpePrefixUnoccluded);
277
278 accumVisibilityAovs(pbrTls, occlRay, fs, numItems, reduceTransparency(occlusionValue));
279 }
280
281 // we are responsible for freeing data memory
282 pbrTls->freeList(occlRay.mDataPtrHandle);
283 }
284 return numRadiancesFilled;
285}

289void
290rayBundleHandler(mcrt_common::ThreadLocalState *tls, unsigned numEntries,
291 WrappedRayState *wrappedRayStates, void *userData)
292{
293 pbr::TLState *pbrTls = tls->mPbrTls.get();
294
295 EXCL_ACCUMULATOR_PROFILE(pbrTls, EXCL_ACCUM_RAY_HANDLER);
296
297 MNRY_ASSERT(numEntries);
298
299 RayState **rayStates = &wrappedRayStates[0].mRsPtr;
300
301 // By convention, if userData is null then rayState contains an array of raw
302 // RayState pointers, otherwise it contains an array of SortedRayStates (which
303 // need to be converted to raw RayState pointers).
304 RayHandlerFlags handlerFlags = RayHandlerFlags((uint64_t)userData);
305 MNRY_ASSERT(handlerFlags <= RAYHANDLER_FLAGS_MASK);
306
307 if (handlerFlags & RAYHANDLER_SORTED_RAYSTATES) {
308 rayStates = decodeRayStatesInPlace(numEntries, &wrappedRayStates[0].mSortedRs);
309 }
310
311 const FrameState &fs = *pbrTls->mFs;
312
313 scene_rdl2::alloc::Arena *arena = &tls->mArena;
314 SCOPED_MEM(arena);
315
316 // heat map
317 int64_t ticks = 0;
318 MCRT_COMMON_CLOCK_OPEN(fs.mRequiresHeatMap? &ticks : nullptr);
319
320 // Perform all intersection checks.
321 if (numEntries) {
322 pbrTls->mStatistics.addToCounter(STATS_INTERSECTION_RAYS, numEntries);
323 EXCL_ACCUMULATOR_PROFILE(pbrTls, EXCL_ACCUM_EMBREE_INTERSECTION);
324
325 const rt::EmbreeAccelerator *accel = fs.mEmbreeAccel;
326 for (unsigned i = 0; i < numEntries; ++i) {
327 RayState *rs = rayStates[i];
328 MNRY_ASSERT(isValid(rs));
329 accel->intersect(rs->mRay);
330 }
331 }
332
333 // Volumes - compute volume radiance and transmission for each ray
334 for (unsigned i = 0; i < numEntries; ++i) {
335 RayState &rs = *rayStates[i];
336 const mcrt_common::Ray &ray = rs.mRay;
337 const Subpixel &sp = rs.mSubpixel;
338 const int cameraId = rs.mCameraId;
339 PathVertex &pv = rs.mPathVertex;
340 const int lobeType = pv.nonMirrorDepth == 0 ? 0 : pv.lobeType;
341 const unsigned sequenceID = rs.mSequenceID;
342 float *aovs = nullptr;
343 PathIntegrator::DeepParams *deepParams = nullptr; // TODO: MOONRAY-3133 support deep output of volumes
344 rs.mVolRad = scene_rdl2::math::sBlack;
345 VolumeTransmittance vt;
346 vt.reset();
347 float volumeSurfaceT = scene_rdl2::math::sMaxValue;
348 rs.mVolHit = fs.mIntegrator->computeRadianceVolume(pbrTls, ray, sp, cameraId, pv, lobeType,
349 rs.mVolRad, sequenceID, vt, aovs, deepParams, &rs, &volumeSurfaceT);
350 rs.mVolTr = vt.mTransmittanceE;
351 rs.mVolTh = vt.mTransmittanceH;
352 rs.mVolTalpha = vt.mTransmittanceAlpha;
353 rs.mVolTm = vt.mTransmittanceMin;
354 rs.mVolumeSurfaceT = volumeSurfaceT;
355 }
356
357 CHECK_CANCELLATION(pbrTls, return);
358
359 // heat maps
360 MCRT_COMMON_CLOCK_CLOSE();
361 if (fs.mRequiresHeatMap) {
362 heatMapBundledUpdate(pbrTls, ticks, rayStates, numEntries);
363 }
364
365 //
366 // Sort by material to minimize locks when adding to shared shade queues.
367 //
368 struct SortedEntry
369 {
370 uint32_t mSortKey; // Material bundled id is stored in here.
371 uint32_t mRsIdx; // Global ray state index.
372 const shading::Material *mMaterial;
373 };
374 SortedEntry *sortedEntries = arena->allocArray<SortedEntry>(numEntries, CACHE_LINE_SIZE);
375 unsigned numSortedEntries = 0;
376 uint32_t maxSortKey = 0;
377
378 // Allocate memory to gather raystates so we can bulk free them later in the function.
379 unsigned numRayStatesToFree = 0;
380 RayState **rayStatesToFree = arena->allocArray<RayState *>(numEntries);
381
382 RayState *baseRayState = indexToRayState(0);
383 const scene_rdl2::rdl2::Layer *layer = fs.mLayer;
384
385 for (unsigned i = 0; i < numEntries; ++i) {
386
387 SortedEntry &sortedEntry = sortedEntries[numSortedEntries];
388 mcrt_common::Ray &ray = rayStates[i]->mRay;
389 PathVertex &pv = rayStates[i]->mPathVertex;
390
391 if (ray.geomID == -1) {
392 // We didn't hit anything.
393 sortedEntry.mSortKey = 0;
394 sortedEntry.mRsIdx = rayStates[i] - baseRayState;
395 sortedEntry.mMaterial = nullptr;
396 ++numSortedEntries;
397 } else {
398 geom::internal::BVHUserData* userData =
399 static_cast<geom::internal::BVHUserData*>(ray.ext.userData);
400 const geom::internal::Primitive* prim = userData->mPrimitive;
401 const scene_rdl2::rdl2::Material *rdl2Material = MNRY_VERIFY(prim)->getIntersectionMaterial(layer, ray);
402 const shading::Material *material = &rdl2Material->get<const shading::Material>();
403
404 if (material) {
405 // perform material substitution if needed
406 scene_rdl2::rdl2::RaySwitchContext switchCtx;
407 switchCtx.mRayType = lobeTypeToRayType(pv.lobeType);
408 rdl2Material = rdl2Material->raySwitch(switchCtx);
409 material = &rdl2Material->get<const shading::Material>();
410
411 sortedEntry.mSortKey = MNRY_VERIFY(material->getMaterialId());
412 sortedEntry.mRsIdx = rayStates[i] - baseRayState;
413 sortedEntry.mMaterial = material;
414 maxSortKey = std::max(maxSortKey, sortedEntry.mSortKey);
415 ++numSortedEntries;
416 } else {
417 // No material is assigned to this hit point, just skip entry
418 // and free up associated RayState resource.
419 rayStatesToFree[numRayStatesToFree++] = rayStates[i];
420
421 // We may still have volume radiance to consider
422 const RayState &rs = *rayStates[i];
423 if (rs.mVolHit) {
424 // We passed through a volume and then hit a geometry.
425 // But there is no material assigned to the geometry, so
426 // there will be no further processing on this ray. It will
427 // not be passed to the shade queue.
428 // We will add the radiance from the volume to the radiance
429 // queue and set the alpha based on the volume alpha.
430 const float alpha = ray.getDepth() == 0 ?
431 rs.mPathVertex.pathPixelWeight * (1.0f - reduceTransparency(rs.mVolTalpha)) : 0.f;
432 BundledRadiance rad;
433 rad.mRadiance = RenderColor(rs.mVolRad.r, rs.mVolRad.g, rs.mVolRad.b, alpha);
434 rad.mPathPixelWeight = rs.mPathVertex.pathPixelWeight;
435 rad.mPixel = rs.mSubpixel.mPixel;
436 rad.mSubPixelIndex = rs.mSubpixel.mSubpixelIndex;
437 rad.mDeepDataHandle = pbrTls->acquireDeepData(rs.mDeepDataHandle);
438 rad.mCryptomatteDataHandle = pbrTls->acquireCryptomatteData(rs.mCryptomatteDataHandle);
439 rad.mTilePassAndFilm = rs.mTilePassAndFilm;
440 rad.mCameraId = rs.mCameraId;
441 pbrTls->addRadianceQueueEntries(1, &rad);
442 }
443 }
444 }
445 }
446
447 // Do the actual sorting.
448 sortedEntries = scene_rdl2::util::smartSort32<SortedEntry, 0, RAY_HANDLER_STD_SORT_CUTOFF>(numSortedEntries,
449 sortedEntries,
450 maxSortKey, arena);
451
452 //
453 // The SortedEntry array in now sorted by material, with all the entries
454 // which didn't hit anything or have a null material assigned at the start.
455 //
456
457 SortedEntry *endEntry = sortedEntries + numSortedEntries;
458 unsigned numMisses = 0;
459
460 // Aovs.
461 float *aovs = nullptr;
462 unsigned aovNumChannels = 0;
463 if (!fs.mAovSchema->empty()) {
464 // scratch space storage for per-pixel aov packing
465 aovNumChannels = fs.mAovSchema->numChannels();
466 aovs = arena->allocArray<float>(aovNumChannels);
467 fs.mAovSchema->initFloatArray(aovs);
468 }
469
470 // Check if rays which didn't intersect anything hit any lights in the scene.
471 if (sortedEntries->mSortKey == 0) {
472
473 SortedEntry *spanEnd = sortedEntries + 1;
474 while (spanEnd != endEntry && spanEnd->mSortKey == 0) {
475 ++spanEnd;
476 }
477
478 numMisses = unsigned(spanEnd - sortedEntries);
479
480 if (numMisses) {
481
482 EXCL_ACCUMULATOR_PROFILE(pbrTls, EXCL_ACCUM_INTEGRATION);
483
484 BundledRadiance *radiances = arena->allocArray<BundledRadiance>(numMisses, CACHE_LINE_SIZE);
485
486 for (unsigned i = 0; i < numMisses; ++i) {
487
488 RayState *rs = &baseRayState[sortedEntries[i].mRsIdx];
489 // if the ray is not a primary ray and hit a light,
490 // its radiance contribution will be tested in occlusion or
491 // presence shadow ray queue
492 scene_rdl2::math::Color radiance = scene_rdl2::math::sBlack;
493 float alpha = 0.0f;
494 BundledRadiance *rad = &radiances[i];
495
496 // This code path only gets executed for rays which didn't intersect with any geometry
497 // in the scene. We can discard such rays at this point but first need to check if
498 // primary rays intersected any lights, so we can include their contribution if visible.
499
500 const Light *hitLight = nullptr;
501 if (rs->mRay.getDepth() == 0) {
502
503 LightIntersection hitLightIsect;
504 int numHits = 0;
505 const Light *hitLight;
506
507 SequenceIDIntegrator sid(rs->mSubpixel.mPixel,
508 rs->mSubpixel.mSubpixelIndex,
509 fs.mInitialSeed[rs->mCameraId]);
510 IntegratorSample1D lightChoiceSamples(sid);
511 hitLight = fs.mScene->intersectVisibleLight(rs->mRay,
512 sInfiniteLightDistance, lightChoiceSamples, hitLightIsect, numHits);
513
514 if (hitLight) {
515 // Evaluate the radiance on the selected light in camera.
516 // Note: we multiply the radiance contribution by the number of lights hit. This is
517 // because we want to compute the sum of all contributing lights, but we're
518 // stochastically sampling just one.
519
520 LightFilterRandomValues lightFilterR = {
521 scene_rdl2::math::Vec2f(0.f, 0.f),
522 scene_rdl2::math::Vec3f(0.f, 0.f, 0.f)}; // light filters don't apply to camera rays
523 radiance = rs->mPathVertex.pathThroughput *
524 hitLight->eval(tls, rs->mRay.getDirection(), rs->mRay.getOrigin(),
525 lightFilterR, rs->mRay.getTime(), hitLightIsect, true, nullptr,
526 rs->mRay.getDirFootprint()) * numHits;
527 // attenuate based on volume transmittance
528 if (rs->mVolHit) radiance *= (rs->mVolTr * rs->mVolTh);
529
530 // alpha depends on light opacity and volumes
531 if (hitLight->getIsOpaqueInAlpha()) {
532 // We hit a visible light that is opaque in alpha.
533 // Volumes are irrelevant, the alpha contribution is
534 // the full pixel weight.
535 alpha = rs->mPathVertex.pathPixelWeight;
536 } else if (rs->mVolHit) {
537 // We hit a visible light, but the light is not
538 // opaque in alpha (e.g. a distant or env light).
539 // There is a volume along this ray. The volume
540 // alpha transmission determines the alpha contribution.
541 alpha = rs->mPathVertex.pathPixelWeight * (1.f - reduceTransparency(rs->mVolTalpha));
542 } else {
543 // We hit a visible light, but the light is not
544 // opaque in alpha (e.g. a distant or env light).
545 // There is no volume along the ray.
546 // The alpha contribution is 0.
547 alpha = 0.f;
548 }
549 } else if (rs->mVolHit) {
550 // We didn't hit a visible light. We didn't hit geometry.
551 // But we did pass through a volume.
552 // The volume alpha transmission determines the alpha contribution.
553 alpha = rs->mPathVertex.pathPixelWeight * (1.f - reduceTransparency(rs->mVolTalpha));
554 }
555 }
556
557 // add in any volume radiance
558 radiance += rs->mVolRad;
559
560 rad->mRadiance = RenderColor(radiance.r, radiance.g, radiance.b, alpha);
561 rad->mPathPixelWeight = rs->mPathVertex.pathPixelWeight;
562 rad->mPixel = rs->mSubpixel.mPixel;
563 rad->mSubPixelIndex = rs->mSubpixel.mSubpixelIndex;
564 rad->mDeepDataHandle = pbrTls->acquireDeepData(rs->mDeepDataHandle);
565 rad->mCryptomatteDataHandle = pbrTls->acquireCryptomatteData(rs->mCryptomatteDataHandle);
566 rad->mTilePassAndFilm = rs->mTilePassAndFilm;
567 rad->mCameraId = rs->mCameraId;
568
569 // LPE
570 if (!fs.mAovSchema->empty()) {
571 EXCL_ACCUMULATOR_PROFILE(pbrTls, EXCL_ACCUM_AOVS);
572
573 // accumulate background aovs
574 aovAccumBackgroundExtraAovsBundled(pbrTls, fs, rs);
575
576 // Did we hit a volume and do we have volume depth/position AOVs?
577 if (rs->mRay.getDepth() == 0 && rs->mVolHit && rs->mVolumeSurfaceT < scene_rdl2::math::sMaxValue) {
578 aovSetStateVarsVolumeOnly(pbrTls,
579 *fs.mAovSchema,
580 rs->mCameraId,
581 rs->mVolumeSurfaceT,
582 rs->mRay,
583 *fs.mScene,
584 rs->mPathVertex.pathPixelWeight,
585 aovs);
586 aovAddToBundledQueueVolumeOnly(pbrTls,
587 *fs.mAovSchema,
588 rs->mRay,
589 AOV_TYPE_STATE_VAR,
590 aovs,
591 rs->mCameraId,
592 rs->mSubpixel.mPixel,
593 rs->mDeepDataHandle,
594 pbr::getFilm(rs->mTilePassAndFilm));
595 }
596
597 const LightAovs &lightAovs = *fs.mLightAovs;
598 // This is complicated.
599 // Case 1:
600 // rayDepth() == 0. In this case, the ray left
601 // the camera, and hit a light. We use the lpeStateId in
602 // in the ray state.
603 //
604 // Case 2:
605 // We expect that PathIntegratorBundled set lpeStateId to
606 // the scattering event, and lpeStateIdLight to the light
607 // event. In this case we hit no geometry, so we hit the light.
608 // For this reason, we use lpeStateIdLight rather than lpeStateId
609 //
610 int lpeStateId = -1;
611 if (rs->mRay.getDepth() == 0) {
612 if (hitLight) {
613 // case 1
614 int lpeStateId = rs->mPathVertex.lpeStateId;
615 if (lpeStateId >= 0) {
616 // transition to light event
617 lpeStateId = lightAovs.lightEventTransition(pbrTls, lpeStateId, hitLight);
618 }
619 }
620 } else {
621 // case 2
622 // transition already computed in PathIntegratorBundled
623 lpeStateId = rs->mPathVertex.lpeStateIdLight;
624 }
625
626 if (lpeStateId >= 0) {
627 // accumulate results. As this has to do with directly hitting a light, we don't have
628 // to worry about pre-occlusion LPEs here.
629 aovAccumLightAovsBundled(pbrTls, *fs.mAovSchema, rs->mCameraId,
630 lightAovs, radiance, nullptr, AovSchema::sLpePrefixNone, lpeStateId,
631 rad->mPixel, rad->mDeepDataHandle,
632 pbr::getFilm(rad->mTilePassAndFilm));
633 }
634 }
635
636 // It's critical that we don't leak ray states.
637 rayStatesToFree[numRayStatesToFree++] = rs;
638 }
639
640 pbrTls->addRadianceQueueEntries(numMisses, radiances);
641
642 CHECK_CANCELLATION(pbrTls, return);
643 }
644 }
645
646 // Bulk free raystates.
647 MNRY_ASSERT(numRayStatesToFree <= numEntries);
648 pbrTls->freeRayStates(numRayStatesToFree, rayStatesToFree);
649
650 //
651 // Route remaining sortedEntries to their associated materials in batches.
652 // Shade queues are thread safe, multiple threads can add to them simultaneously.
653 //
654
655 uint8_t *memBookmark = arena->getPtr();
656
657 SortedEntry *currEntry = sortedEntries + numMisses;
658 while (currEntry != endEntry) {
659
660 const uint32_t currBundledMatId = MNRY_VERIFY(currEntry->mSortKey);
661
662 SortedEntry *spanEnd = currEntry + 1;
663 while (spanEnd != endEntry && spanEnd->mSortKey == currBundledMatId) {
664 ++spanEnd;
665 }
666
667 // Create entries for shade queue.
668 unsigned numShadeEntries = MNRY_VERIFY(spanEnd - currEntry);
669 shading::SortedRayState *shadeEntries = arena->allocArray<shading::SortedRayState>(numShadeEntries,
670 CACHE_LINE_SIZE);
671
672 for (unsigned i = 0; i < numShadeEntries; ++i) {
673 uint32_t rsIdx = currEntry[i].mRsIdx;
674 RayState *rs = &baseRayState[rsIdx];
675 const mcrt_common::Ray &ray = rs->mRay;
676 shadeEntries[i].mRsIdx = rsIdx;
677
678 // Sort first by geometry and then by primitive within that geometry.
679 // This is to improve locality for postIntersection calls.
680 shadeEntries[i].mSortKey = ((ray.geomID & 0xfff) << 20) | (ray.primID & 0xfffff);
681 }
682
683 // Submit to destination queue.
684 shading::ShadeQueue *shadeQueue = MNRY_VERIFY(currEntry->mMaterial->getShadeQueue());
685 shadeQueue->addEntries(tls, numShadeEntries, shadeEntries, arena);
686
687 CHECK_CANCELLATION(pbrTls, return);
688
689 // Free up entry memory from arena.
690 arena->setPtr(memBookmark);
691
692 currEntry = spanEnd;
693 }
694}
