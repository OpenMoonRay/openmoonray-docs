73template <typename DistanceFunction>
74float findClosest(const Point& p, const Grid& grid, DistanceFunction distance)
75{
76 // Point to grid indices.
77 PointContainer pointsToCheck;
78 static constexpr auto maxNeighbors = Grid::maxNeighboringItems();
79 pointsToCheck.reserve(maxNeighbors);
80 auto f = [&pointsToCheck](const Point& c)
81 {
82 pointsToCheck.push_back(c);
83 };
84
85 // If we don't have a populated neighbor, we want to visit all locations,
86 // to make sure we still get the closest point. Yay, O(N^2)!
87 grid.visitNeighbors(f, p);
88 if (pointsToCheck.empty()) {
89 grid.visitAll(f);
90 }
91
92 float sampleDist2 = std::numeric_limits<float>::max();
93 for (const auto& sample : pointsToCheck) {
94 const float thisDist2 = distance(p, sample);
95 if (thisDist2 < sampleDist2) {
96 sampleDist2 = thisDist2;
97 }
98 }
99
100 return sampleDist2;
101}

132template <typename Container, typename DistanceFunction>
133typename Container::iterator stochasticFindNext(Container& list, const Grid& grid, RNG& rng, DistanceFunction distance)
134{
135 typedef typename Container::iterator iterator;
136 typedef typename Container::size_type size_type;
137 const size_type upperLimit = list.size() / 4u;
138
139 // For each point in list (up to a limit):
140 //
141 // If it is farther than radius to each point in the grid, add to candidate
142 // list.
143 //
144 // Randomly choose one element from candidate list.
145
146 struct ValueType
147 {
148 ValueType(iterator it, float d) : iter(it), distance(d) {}
149 typename Container::iterator iter;
150 float distance;
151 };
152
153 std::vector<ValueType> candidates;
154 candidates.reserve(upperLimit);
155
156 // We want to make a min-heap, so we can readily access the smallest value
157 // we've saved, so swap it out.
158 auto comp = [](const ValueType& a, const ValueType& b) { return a.distance > b.distance; };
159
160 for (auto it = list.begin(); it != list.end(); ++it) {
161 const float c = findClosest(*it, grid, distance);
162 if (candidates.empty() || candidates.size() < upperLimit) {
163 candidates.emplace_back(it, c);
164 if (candidates.size() == upperLimit) {
165 std::make_heap(candidates.begin(), candidates.end(), comp);
166 }
167 } else if (c > candidates.front().distance) {
168 std::pop_heap(candidates.begin(), candidates.end(), comp);
169 candidates.back() = ValueType(it, c);
170 std::push_heap(candidates.begin(), candidates.end(), comp);
171 }
172 }
173
174 std::vector<float> distances;
175 distances.reserve(candidates.size());
176 std::transform(candidates.begin(), candidates.end(), std::back_inserter(distances),
177 [](const ValueType& v) {
178 return v.distance;
179 });
180
181 std::discrete_distribution<size_type> dist(distances.begin(), distances.end());
182 return candidates.at(dist(rng)).iter;
183}

210struct FindFarthest
211{
212 float mMaxDist2;
213 const Grid& mGrid;
214 DistanceFunction mDist;
215 typename Container::iterator mEnd;
216 typename Container::iterator mFar;
217
218 explicit FindFarthest(const Grid& g, typename Container::iterator end, DistanceFunction dist) :
219 mMaxDist2(0.0f),
220 mGrid(g),
221 mDist(dist),
222 mEnd(end),
223 mFar(end)
224 {
225 }
226
227 FindFarthest(FindFarthest& s, tbb::split) :
228 mMaxDist2(0.0f),
229 mGrid(s.mGrid),
230 // We don't allow dynamically changing distance functions. It ruins the cache.
231 mDist(0),
232 mEnd(s.mEnd),
233 mFar(mEnd)
234 {
235 }
236
237 void operator()(const tbb::blocked_range<typename Container::iterator>& r)
238 {
239 float maxDist2 = mMaxDist2;
240 for (auto a = r.begin(); a != r.end(); ++a) {
241 if (maxDist2 <= a->mDist2UpperBound) {
242 const float c = findClosest(*a, mGrid, mDist);
243 a->mDist2UpperBound = c;
244 if (c > maxDist2) {
245 maxDist2 = c;
246 mFar = a;
247 }
248 }
249 }
250 mMaxDist2 = maxDist2;
251 }
252
253 void join(FindFarthest& rhs)
254 {
255 if (mFar == mEnd || mMaxDist2 < rhs.mMaxDist2) {
256 mMaxDist2 = rhs.mMaxDist2;
257 mFar = rhs.mFar;
258 }

286void reorder(const std::string& outFileName, Container unprocessed, const Point& initialPoint, RNG& reng, DistanceFunction distance, unsigned stochasticCount)
287{
288 auto outs = createOutputStream(outFileName);
289
290 Grid grid;
291 // Add initial point to grid.
292 grid.add(initialPoint);
293 (*outs) << initialPoint << '\n';
294
295 ProgressBar progress(unprocessed.size());
296 progress.draw();
297 for (unsigned i = 0; !unprocessed.empty(); ++i) {
298 const bool stochastic = i < stochasticCount && i % 2 == 0;
299 const auto toAdd = (stochastic) ?
300 stochasticFindNext(unprocessed, grid, reng, distance) :
301 findFarthest(unprocessed, grid, distance);
302
303 assert(toAdd != unprocessed.end());
304
305 grid.add(*toAdd);
306 (*outs) << *toAdd << '\n';
307
308 unprocessed.erase(toAdd);
309 progress.update();
310 progress.draw();
311 }
312 std::cout << '\n';
313}
