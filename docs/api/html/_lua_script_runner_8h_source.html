19//
20// This LuaScriptRunner execute Lua script under newly created lua_State.
21// Current version only support data transfer from caller to lua_State by setting Lua global
22// variables. Not support parameter transfer from inside lua_State to outside yet.
23// Typical procedure to execute Lua script is 2 steps.
24//
25// 1) setup Lua global variables
26// 2) execute Lua script by runFile()
27//
28// Lua script is executed as single thread task. If you create 2 LuaScriptRunner object and
29// call each runFile() by different threads, each LuaScriptRunner::runFile() is executed
30// independently by 2 different threads in parallel. Also each LuaScriptRunner creates independent
31// lua_State internally and they are isolated each other.
32//
33
34// >>>>> How to setup Lua global variables? <<<<<
35// There are several APIs to define Lua global variables definitions inside LuaScriptRunner and
36// they are converted into global variables of lua_State just before evaluate Lua script inside
37// runFile().
38//
39// A) Define simple variable
40// We can define 4 different simple variables bool, int, float and string.
41// This is an example.
42//
43// LuaScriptRunner lua;
44// lua.setVarBool("foo", true);
45// lua.setVarInt("bar", 123);
46// lua.setVarFloat("baz", 4.567);
47// lua.setVarString("qux", "hello lua");
48// lua.runFile("test.lua");
49//
50// Lua script test.lua can access following global variables as simple parameters.
51//
52// foo = true
53// bar = 123
54// baz = 4.567
55// qux = "hello lua"
56//
57// B) Define simple array
58// We can define 3 different arrays (simple table) which are int array, float array and
59// string array.
60// This is an example.
61//
62// std::vector<int> iVec = {1, 2, 3, 4};
63// std::vector<float> fVec = {1.23, 2.34};
64// std::vector<std::string> sVec = {"abc", "def", "ghi"};
65// LuaScriptRunner lua;
66// lua.setArrayInt("foovec", iVec);
67// lua.setArrayFloat("barvec", fVec);
68// lua.setArrayString("bazvec", sVec);
69// lua.runFile("test.lua");
70//
71// Lua script test.lua can access following global variables as array.
72//
73// foovec = { 1, 2, 3, 4 } -- int table
74// barvec = { 1.23, 2.34 } -- float table
75// bazvec = { "abc", "def", "ghi" } -- string table
76//
77// C) Define dictionary
78// We can define dictionary (associative table) which has a any member of above APIs.
79// Dictionary can have a other dictionary as a member also.
80// This is an example.
81//
82// std::vector<std::string> strVec = {"abc", "DEF"};
83// std::vector<float> fVec = {1.0, 0.5, 0.25};
84// LuaScriptRunner lua;
85// lua.beginDictionary("footbl"); { // begin dictionary of footbl
86// lua.setVarInt("foo", 123);
87// lua.setArrayString("barvec", strVec);
88// lua.beginDictionary("baztbl"); { // begin dictionary of baztbl
89// lua.setVarFloat("qux", 2.345);
90// lua.setArrayFloat("quuxvec", fVec);
91// }
92// lua.endDictionary(); // end dictionary of baztbl
93// }
94// lua.endDictionary(); // end of dictionary of footbl
95// lua.runFile("test.lua");
96//
97// Lua script test.lua can access following global variables as associative table.
98//
99// footbl = { -- associative table
100// foo = 123, -- int variabl
101// barvec = { "abc", "DEF" }, -- string table
102// baztbl = { -- associative table
103// qux = 2.345, -- float variable
104// quuxvec = { 1.0, 0.5, 0.25 } -- float table
105// }
106// }
107//
108// D) Define complex array
109// We can make more complex array data than example B by using array definition APIs.
110// They are beginArray(), endArray(), setArrayItem*(), beginArrayItemDictionary() and
111// endArrayItemDictionary().
112// Actually, lua array is a special type of associative table and we can store different
113// data type data in different id's of an array Also lua array id starts 1 (not 0) by lua's habit.
114// The following are examples.
115//
116// D-0) Array which combined different simple data
117// This is an example of an array of mixed simple data.
118//
119// LuaScriptRunner lua;
120// lua.beginArray("fooArray"); { // begin of fooArray
121// lua.setArrayItemVarBool(true); // definition of fooArray[1]
122// lua.setArrayItemVarInt(123); // definition of fooArray[2]
123// lua.setArrayItemVarFloat(4.567); // definition of fooArray[3]
124// lua.setArrayItemVarString("hello lua"); // definition of fooArray[4]
125// }
126// lua.endArray(); // end of fooArray
127// lua.runFile("test.lua");
128//
129// Lua script test.lua can access following global variables as array.
130//
131// fooArray = { true, 123, 4.567, "hello lua" } -- mixed simple data type
132//
133// D-1) Array which combined different simple array (array of array data)
134// This is an example of an array of several different array data.
135//
136// std::vector<int> iVec = {1, 2, 3, 4};
137// std::vector<float> fVec = {1.23, 2.34};
138// std::vector<std::string> sVec = {"abc", "def", "ghi"};
139// LuaScriptRunner lua;
140// lua.beginArray("fooArray"); { // begin of fooArray
141// lua.setArrayItemArrayInt(iVec); // definition of fooArray[1] as int array
142// lua.setArrayItemArrayFloat(fVec); // definition of fooArray[2] as float array
143// lua.setArrayItemArrayString(sVec); // definition of fooArray[3] as string array
144// }
145// lua.endArray(); // end of fooArray
146// lua.runFile("test.lua");
147//
148// Lua script test.lua can access following global variables as array.
149//
150// fooArray = {
151// { 1, 2, 3, 4 }, -- fooArray[1] is size 4 of integer array
152// { 1.23, 2.34 }, -- fooArray[2] is size 2 of float array
153// { "abc", "def", "ghi" } -- fooArray[3] is size 3 of string array
154// }
155//
156// D-2) Array which combined dictionary
157// This is an example of an array of dictionaries.
158//
159// LuaScriptRunner lua;
160// lua.beginArray("fooArray"); { // begin of fooArray
161// lua.beginArrayItemDictionary(); { // bgin of array[1]
162// std::vector<std::string> strVec = {"abc", "DEF"};
163// std::vector<float> fVec = {1.0, 0.5, 0.25};
164// lua.setVarInt("foo", 123);
165// lua.setArrayString("barvec", strVec);
166// lua.beginDictionary("baztbl"); {
167// lua.setVarFloat("qux", 2.345);
168// lua.setArrayFloat("quuxvec", fVec);
169// }
170// lua.endDictionary();
171// }
172// lua.endArrayItemDictionary(); // end of array[1]
173// lua.beginArrayItemDictionary(); { // begin of array[2]
174// std::vector<std::string> strVec = {"bcd", "EFG"};
175// std::vector<float> fVec = {11.0, 10.5, 10.25};
176// lua.setVarInt("foo", 234);
177// lua.setArrayString("barvec", strVec);
178// lua.beginDictionary("baztbl"); {
179// lua.setVarFloat("qux", 3.456);
180// lua.setArrayFloat("quuxvec", fVec);
181// }
182// lua.endDictionary();
183// }
184// lua.endArrayItemDictionary(); // end of array[2]
185// }
186// lua.endArray(); // end of array
187// lua.runFile("test.lua");
188//
189// Lua script test.lua can access following global variables as array.
190//
191// fooArray = {
192// { -- fooArray[1]
193// foo = 123,
194// barvec = { "abc", "DEF" },
195// baztbl = {
196// qux = 2.345,
197// quuxvec = { 1.0, 0.5, 0.25 }
198// }
199// },
200// { -- fooArray[2]
201// foo = 234,
202// barvec = { "bcd", "EFG" },
203// baztbl = {
204// qux = 3.456,
205// quuxvec = { 11.0, 10.5, 10.25 }
206// }
207// }
208// }
209//
210// D-3) Array which combined everything
211// This is an example of an array of mixed all of the data.
212//
213// std::vector<int> iVec = {1, 2, 3, 4};
214// LuaScriptRunner lua;
215// lua.beginArray("fooArray"); { // begin of fooArray
216// lua.setArrayItemArrayInt(iVec); // definition of array[1]= size 4 of int array
217// lua.beginArrayItemDictionary(); { // begin of array[2]

249// fooArray = {

264// baztbl = {
265// qux = 3.456,
266// quuxvec = { 11.0, 10.5, 10.25 }

270//
271// E) Convert JSON to lua global parameter
272// We can convert JSON data to lua global parameters.
273// This example reads ./test.json data and converts to lua associative table as fooJSON.
274//
275// LuaScriptRunner lua;
276// lua.setDictionaryByJson("fooJSON", "./test.json");
277// lua.runFile("test.lua");
278//

298 elemSize = function(tbl)
