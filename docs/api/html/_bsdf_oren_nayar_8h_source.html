33static finline float
34getOrenNayarIntensity(const scene_rdl2::math::Vec3f& n,
35 const scene_rdl2::math::Vec3f& v,
36 const scene_rdl2::math::Vec3f& l,
37 const float a,
38 const float b)
39{
40 // We are using the Fujii implementation or Oren-Nayar.
41 // The PBRT implementation was also tested but resulted
42 // in more energy loss.
43 // https://mimosa-pudica.net/improved-oren-nayar.html
44 const float nl = scene_rdl2::math::max(scene_rdl2::math::dot(n, l), 0.0f);
45 const float nv = scene_rdl2::math::max(scene_rdl2::math::dot(n, v), 0.0f);
46 float s = scene_rdl2::math::dot(l, v) - nl * nv;
47
48 float t = 1.0f;
49 if (s > 0.0f) {
50 t = scene_rdl2::math::max(nl, nv) + scene_rdl2::math::sEpsilon;
51 }
52
53 return nl * (a + b * s/t);
54}

56///
57/// @class OrenNayarBsdfLobe BsdfOrenNayar.h <shading/bsdf/BsdfOrenNayar.h>
58/// @brief The OrenNayar diffuse brdf (f = kd / PI)
59///
60/// https://www1.cs.columbia.edu/CAVE/publications/pdfs/Oren_SIGGRAPH94.pdf
61// This class implements the Oren Nayar fast approximation
62class OrenNayarBsdfLobe : public BsdfLobe
63{
64public:
65 // Constructor / Destructor
66 OrenNayarBsdfLobe(const scene_rdl2::math::Vec3f &N, const scene_rdl2::math::Color& albedo, float roughness, bool reflection /* = true */) :
67 BsdfLobe(Type((reflection ? REFLECTION : TRANSMISSION) | DIFFUSE),
68 DifferentialFlags(IGNORES_INCOMING_DIFFERENTIALS), false,
69 PROPERTY_NORMAL | PROPERTY_ROUGHNESS | PROPERTY_PBR_VALIDITY),
70 mFrame(N),
71 mAlbedo(albedo),
72 mRoughness(roughness)
73 {
74 const float s = scene_rdl2::math::sqr(scene_rdl2::math::deg2rad(roughness * 90.0f));
75 mA = 1.0f - (0.5f * (s / (s + 0.33f)));
76 mB = 0.45f * (s / (s + 0.09f));
77 }
78
79 // BsdfLobe API
80 finline scene_rdl2::math::Color eval(const BsdfSlice &slice, const scene_rdl2::math::Vec3f &wi, float *pdf = NULL) const override
81 {
82 const float cosThetaWi = getOrenNayarIntensity(mFrame.getN(),
83 slice.getWo(), wi,
84 mA, mB);
85
86 if (pdf != NULL) {
87 *pdf = cosThetaWi * scene_rdl2::math::sOneOverPi;
88 }
89
90 // Note: we assume this lobe has been setup with a OneMinus*Fresnel
91 // as we want to use 1 - specular_fresnel. Also notice we use
92 // cosThetaWo to evaluate the fresnel term, as an approximation of what
93 // hDotWi would be for the specular lobe.
94 float cosThetaWo = 1.0f;
95 if (getFresnel()) {
96 const scene_rdl2::math::Vec3f N = (matchesFlag(REFLECTION)) ?
97 mFrame.getN() :
98 -mFrame.getN();
99 cosThetaWo = scene_rdl2::math::max(dot(N, slice.getWo()), 0.0f);
100 }
101
102 // Soften hard shadow terminator due to shading normals
103 const float Gs = (matchesFlag(REFLECTION)) ?
104 slice.computeShadowTerminatorFix(mFrame.getN(), wi) :
105 1.0f;
106
107 return mAlbedo * scene_rdl2::math::sOneOverPi * Gs *
108 computeScaleAndFresnel(cosThetaWo) *
109 (slice.getIncludeCosineTerm() ? cosThetaWi : 1.0f);
110 }
111
112
113 finline scene_rdl2::math::Color sample(const BsdfSlice &slice, float r1, float r2,
114 scene_rdl2::math::Vec3f &wi, float &pdf) const override
115 {
116 wi = mFrame.localToGlobal(sampleLocalHemisphereCosine(r1, r2));
117 return eval(slice, wi, &pdf);
118 }
119
120
121 finline scene_rdl2::math::Color albedo(const BsdfSlice &slice) const override
122 {
123 float cosThetaWo = scene_rdl2::math::max(dot(mFrame.getN(), slice.getWo()), 0.0f);
124 return computeScaleAndFresnel(cosThetaWo) * mAlbedo;
125 }
126
127 // no surface curvature required
128 finline void differentials(const scene_rdl2::math::Vec3f &wo, const scene_rdl2::math::Vec3f &wi,
129 float r1, float r2, const scene_rdl2::math::Vec3f &dNdx, const scene_rdl2::math::Vec3f &dNdy,
130 scene_rdl2::math::Vec3f &dDdx, scene_rdl2::math::Vec3f &dDdy) const override
131 {
132 // The hemisphere cosine sampling direction derivative seems like a very
133 // good approximation to the full derivative for diffuse sampling. This
134 // is why we ignore the input differentials.
135
136 // TODO: How can we avoid computing these twice (once here, once in
137 // sample()->sampleLocalHemisphereCosine()) ?
138 localHemisphereCosineDifferentials(r1, r2, dDdx, dDdy);
139
140 // The differentials form rectangles which get long and thin close
141 // to the hemisphere pole and equator. We prefer having ray
142 // differentials that form a square, but preserve the ray footprint.
143 squarifyRectangle(dDdx, dDdy);
144
145 dDdx = mFrame.localToGlobal(dDdx);
146 dDdy = mFrame.localToGlobal(dDdy);
147 }
148
149 bool getProperty(Property property, float *dest) const override
150 {
151 bool handled = true;
152
153 switch (property)
154 {
155 case PROPERTY_ROUGHNESS:
156 *dest = 1.0f;
157 *(dest + 1) = 1.0f;
158 break;
159 case PROPERTY_NORMAL:
160 {
161 const scene_rdl2::math::Vec3f &N = mFrame.getN();
162 *dest = N.x;
163 *(dest + 1) = N.y;
164 *(dest + 2) = N.z;
165 }
166 break;
167 case PROPERTY_PBR_VALIDITY:
168 {
169 scene_rdl2::math::Color res = computeAlbedoPbrValidity(mAlbedo);
170 *dest = res.r;
171 *(dest + 1) = res.g;
172 *(dest + 2) = res.b;
173 }
174 break;
175 case PROPERTY_COLOR:
176 {
177 // Special case handling for oren nayar color property since albedo isn't included
178 // in mScale anymore.
179 scene_rdl2::math::Color result = mAlbedo * computeScaleAndFresnel(1.f);
180 *dest = result.r;
181 *(dest + 1) = result.g;
182 *(dest + 2) = result.b;
183 }
184 break;
185 default:
186 handled = BsdfLobe::getProperty(property, dest);
187 break;
188 }
189
190 return handled;
191 }
192
193 void show(std::ostream& os, const std::string& indent) const override
194 {
195 const scene_rdl2::math::Color& scale = getScale();
196 const scene_rdl2::math::Vec3f& N = mFrame.getN();
197 const Fresnel * const fresnel = getFresnel();
198
199 os << indent << "[OrenNayarBsdfLobe] "
200 << (matchesFlag(REFLECTION) ? "(reflection)" : "(transmission)") << "\n";
201 os << indent << " " << "scale: "
202 << scale.r << " " << scale.g << " " << scale.b << "\n";
203 os << indent << " " << "N: "
204 << N.x << " " << N.y << " " << N.z << "\n";
205 os << indent << " " << "albedo: "
206 << mAlbedo.r << " " << mAlbedo.g << " " << mAlbedo.b << "\n";
207 os << indent << " " << "roughness: " << mRoughness << "\n";
208 if (fresnel) {
209 fresnel->show(os, indent + " ");
210 }
211 }
212
213protected:
214 scene_rdl2::math::ReferenceFrame mFrame;
215 scene_rdl2::math::Color mAlbedo;
216 float mRoughness;
217 float mA;
218 float mB;
219};
