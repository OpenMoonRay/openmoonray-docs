103 "Ray termination color is used for filling in falsely dark areas where ray paths have "

106 "ignoring occlusion by scene geometry. "
107 "Any light can either be a regular light or a ray termination light (but not both). "
108 "Thus they can be freely assigned to light sets, which provides a mechanism for applying "
109 "specific ray termination lights to specific materials, parts or objects. "

232int
233Light::getVisibilityMask() const
234{
235 int visibilityMask = NONE_VISIBLE;
236 visibilityMask |= get(sVisibleDiffuseReflection) ? DIFFUSE_REFLECTION : 0;
237 visibilityMask |= get(sVisibleDiffuseTransmission) ? DIFFUSE_TRANSMISSION : 0;
238 visibilityMask |= get(sVisibleGlossyReflection) ? GLOSSY_REFLECTION : 0;
239 visibilityMask |= get(sVisibleGlossyTransmission) ? GLOSSY_TRANSMISSION : 0;
240 visibilityMask |= get(sVisibleMirrorReflection) ? MIRROR_REFLECTION : 0;
241 visibilityMask |= get(sVisibleMirrorTransmission) ? MIRROR_TRANSMISSION : 0;
242 return visibilityMask;
243}
