132class MirrorTransmissionBsdfLobe : public BsdfLobe
133{
134public:
140 MirrorTransmissionBsdfLobe(const scene_rdl2::math::Vec3f &N,
141 float etaI,
142 float etaT,
143 const scene_rdl2::math::Color &tint,
144 float abbeNumber = 0.0f) :
145 BsdfLobe(Type(TRANSMISSION | MIRROR),
146 DifferentialFlags(0),
147 false,
148 PROPERTY_NORMAL | PROPERTY_ROUGHNESS),
149 mN(N),
150 mEtaI(etaI),
151 mEtaT(etaT),
152 mTint(tint)
153 {
154 // A value of zero is used to turn OFF dispersion.
155 // Values range from below 25 for very dense flint glasses, around 34 for polycarbonate plastics,
156 // up to 65 for common crown glasses, and 75 to 85 for some fluorite and phosphate crown glasses.
157 if (scene_rdl2::math::isZero(abbeNumber)) {
158 mAllowDispersion = false;
159 } else {
160 shading::computeSpectralIOR(etaT,
161 abbeNumber,
162 &mEtaR,
163 &mEtaG,
164 &mEtaB);
165 mAllowDispersion = true;
166 }
167 }
168
169 // BsdfLobe API
170 finline scene_rdl2::math::Color eval(const BsdfSlice &slice, const scene_rdl2::math::Vec3f &wi, float *pdf = NULL) const override
171 {
172 if (pdf != NULL) {
173 *pdf = 0.0f;
174 }
175 return scene_rdl2::math::Color(0.0f, 0.0f, 0.0f);
176 }
177
178 finline scene_rdl2::math::Color sample(const BsdfSlice &slice,
179 float r1,
180 float r2,
181 scene_rdl2::math::Vec3f &wi,
182 float &pdf) const override
183 {
184 MNRY_ASSERT(!scene_rdl2::math::isZero(mEtaT));
185
186 float neta;
187 scene_rdl2::math::Color dispersionColor = scene_rdl2::math::sWhite;
188 if (mAllowDispersion) {
189 float sampledEta;
190 shading::sampleSpectralIOR(r1,
191 mEtaR,
192 mEtaG,
193 mEtaB,
194 &sampledEta,
195 &pdf,
196 dispersionColor);
197 neta = mEtaI / sampledEta;
198 } else {
199 // Probability of sampling the delta function is 1.
200 neta = mEtaI / mEtaT;
201 pdf = 1.0f;
202 }
203
204 // Compute the transmission direction
205 float cosThetaWo, cosThetaWi;
206 if (!computeRefractionDirection(mN,
207 slice.getWo(),
208 neta,
209 wi,
210 cosThetaWo,
211 cosThetaWi)) {
212
213 // Total internal reflection is handled by the reflection lobe
214 // fresnel reflectance
215 pdf = 0.0f;
216 return scene_rdl2::math::Color(0.0f, 0.0f, 0.0f);
217 }
218
219
220 // Note: we assume this lobe has been setup with a OneMinus*Fresnel
221 // as we want to use 1 - specular_fresnel. Also notice we use
222 // cosThetaWo to evaluate the fresnel term, as an approximation of what
223 // hDotWi would be for the specular lobe.
224 return mTint * computeScaleAndFresnel(cosThetaWo) * dispersionColor *
225 (slice.getIncludeCosineTerm() ? 1.0f : 1.0f / cosThetaWi);
226 }
227
228 finline scene_rdl2::math::Color albedo(const BsdfSlice &slice) const override
229 {
230 float cosThetaWo = dot(mN, slice.getWo());
231 return mTint * computeScaleAndFresnel(scene_rdl2::math::abs(cosThetaWo));
232 }
233
234 void differentials(const scene_rdl2::math::Vec3f &wo, const scene_rdl2::math::Vec3f &wi,
235 float r1, float r2, const scene_rdl2::math::Vec3f &dNdx, const scene_rdl2::math::Vec3f &dNdy,
236 scene_rdl2::math::Vec3f &dDdx, scene_rdl2::math::Vec3f &dDdy) const override
237 {
238 computeRefractionDirectionDifferential((mEtaI/mEtaT),
239 wo, wi, mN, dNdx, dNdy,
240 dDdx, dDdy);
241 }
