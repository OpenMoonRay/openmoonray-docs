102void
103HairState::calculateHairFrame(const Vec3f& wo,
104 const Vec3f& hairDir,
105 const float hairRotation,
106 const Vec3f& hairNormal)
107{
108 float UdotO = dot(wo, hairDir);
109 UdotO = clamp(UdotO, -1.0f, 1.0f);
110
111 if (mUsingCustomNormal) {
112 mHairNorm = Vec3f(hairNormal);
113 scene_rdl2::math::Mat4f mat = scene_rdl2::math::Mat4f::rotate(hairDir, hairRotation);
114 mHairNorm = transformVector(mat, mHairNorm);
115 } else {
116 // Set hair normal to align with eyeDir's projection onto normal plane
117 // to ease of computation
118 mHairNorm = wo - (UdotO * hairDir);
119 }
120 mHairBinorm = cross(mHairNorm, hairDir);
121
122 if (lengthSqr(mHairBinorm) > sEpsilon) {
123 mHairNorm = normalize(mHairNorm);
124 mHairBinorm = normalize(mHairBinorm);
125 } else {
126 // rare case that wo is parallel to hairdir, need a safe fallback
127 // to avoid NaN normal/binormal
128 ReferenceFrame frame(wo);
129 mHairNorm = frame.getX();
130 mHairBinorm = frame.getY();
131 }
132
133 mThetaO = sHalfPi - acos(UdotO);
134 sincos(mThetaO, &mSinThetaO, &mCosThetaO);
135
136 if (mUsingCustomNormal) {
137 scene_rdl2::math::Vec3f proj = normalize(wo - hairDir * dot(wo, hairDir));
138 const float cosPhiO = clamp(dot(proj, mHairNorm), -1.0f, 1.0f);
139 const float sinPhiO = clamp(dot(proj, mHairBinorm), -1.0f, 1.0f);
140 mPhiO = atan2(sinPhiO, cosPhiO);
141 } else {
142 // normal is aligned to view direction
143 mPhiO = sPhiO;
144 }
145
146 // "Light Scattering from Human Hair Fibers" - Marschner et al
147 // Appendix B
148 const float tmp = HairUtil::safeSqrt(mEta*mEta - mSinThetaO*mSinThetaO);
149 mEtaP = tmp / mCosThetaO;
150 mEtaPP = mEta*mEta*mCosThetaO / tmp;
151
152}

155void
156HairState::calculateAbsorptionTerm(const float h,
157 const float refractiveIndex,
158 const Color& sigmaA)
159{
160 mAbsorptionTerm = scene_rdl2::math::sWhite;
161
162 float cosGammaT = 0.0f;
163 // Calculate EtaP
164 if (!isZero(cosThetaO())) {
165 const float sinGammaT = h / mEtaP;
166 cosGammaT = HairUtil::safeSqrt(1.0f - sinGammaT*sinGammaT);
167 }
168 const float sinThetaT = sinThetaO() / refractiveIndex;
169 const float cosThetaT = HairUtil::safeSqrt(1.0f - sinThetaT*sinThetaT);
170 if (!isZero(cosThetaT))
171 mAbsorptionTerm = scene_rdl2::math::exp(-sigmaA * 2.0f * cosGammaT / cosThetaT);
172}

174// static function for calculating absorption based
175// on a modified eta and etaP
176Color
177HairState::calculateAbsorptionTerm(const float h,
178 const float eta,
179 const Color& sigmaA,
180 const float etaP,
181 const float cosThetaO,
182 const float sinThetaO)
183{
184 Color absorptionTerm = scene_rdl2::math::sWhite;
185
186 float cosGammaT = 0.0f;
187 // Calculate EtaP
188 if (!isZero(cosThetaO)) {
189 const float sinGammaT = h / etaP;
190 cosGammaT = HairUtil::safeSqrt(1.0f - sinGammaT*sinGammaT);
191 }
192 const float sinThetaT = sinThetaO / eta;
193 const float cosThetaT = HairUtil::safeSqrt(1.0f - sinThetaT*sinThetaT);
194 if (!isZero(cosThetaT))
195 absorptionTerm = scene_rdl2::math::exp(-sigmaA * 2.0f * cosGammaT / cosThetaT);
196
197 return absorptionTerm;
198}

200// Compute the direction in hair local frame (outTheta, outPhi)
201// outTheta is in [-pi/2, pi/2]
202// outPhi is in the range [0, pi] when !mUsingCustomNormal
203// and in the range [-pi, pi] when mUsingCustomNormal
204// WARNING: Don't use these angles to sample directions!!!
205void
206HairState::computeAngles(const Vec3f &direction,
207 const Vec3f &hairDir,
208 float &outTheta,
209 float &outPhi) const
210{
211 const float cosTheta = dot(direction,
212 hairDir);
213 outTheta = sHalfPi - acos(clamp(cosTheta, -1.0f, 1.0f));
214
215 // Project direction into azimuth plane
216 Vec3f proj = direction - hairDir * dot(direction, hairDir);
217 float l = lengthSqr(proj);
218 if (l <= sEpsilon) { outPhi = 0.0f; return; }
219 proj /= sqrt(l);
220
221 const float cosPhi = clamp(dot(proj, mHairNorm), -1.0f, 1.0f);
222 if (mUsingCustomNormal) {
223 const float sinPhi = clamp(dot(proj, mHairBinorm), -1.0f, 1.0f);
224 outPhi = atan2(sinPhi, cosPhi);
225 } else {
226 // normal is aligned to view direction
227 // WARNING: We exploit the symmetry of the bsdf in the azimuth, to use
228 // acos instead of atan2. We get a phi value only in half the possible
229 // range of directions. Don't use these angles to generate directions!!!
230 // Only use these angles to evaluate the pdf or the bsdf.
231 outPhi = acos(cosPhi);
232 }
233}

235void
236HairState::updateAngles(const Vec3f &wi,
237 const float phiI,
238 const float thetaI)
239{
240 mThetaI = thetaI;
241 mPhiI = phiI;
242
243 sincos(mThetaI, &mSinThetaI, &mCosThetaI);
244
245 if (mUsingCustomNormal) {
246 scene_rdl2::math::Vec3f proj = normalize(mWo - mHairDir * dot(mWo, mHairDir));
247 const float cosPhiO = clamp(dot(proj, mHairNorm), -1.0f, 1.0f);
248 const float sinPhiO = clamp(dot(proj, mHairBinorm), -1.0f, 1.0f);
249 mPhiO = atan2(sinPhiO, cosPhiO);
250 } else {
251 // normal is aligned to view direction
252 mPhiO = sPhiO;
253 }
254
255 mPhiD = rangeAngle(mPhiO - mPhiI);
256 mPhiH = (mPhiO + mPhiI) * 0.5f;
257 mThetaD = scene_rdl2::math::abs(mThetaO - mThetaI) * 0.5f;
258
259 mCosPhiDOverTwo = cos(mPhiD*0.5f);
260 sincos(mThetaD, &mSinThetaD, &mCosThetaD);
261
262 // "Light Scattering from Human Hair Fibers" - Marschner et al
263 // Appendix B
264 const float tmp = sqrt(mEta*mEta - mSinThetaD*mSinThetaD);
265 mEtaP = tmp / mCosThetaD;
266 mEtaPP = mEta*mEta*mCosThetaD / tmp;
267
268}
