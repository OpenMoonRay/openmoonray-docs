59//
60// We have a tile structure here to allow us to specify an explicit list of tiles we want
61// rendered and the order we want them rendered in. This allows us to support region of
62// interest, and also gives us the ability to swizzle the tiles for increased cache
63// coherency. A tile is always 8*8 pixels in size unless it's on the border of a clipped
64// viewport. Unlike viewports, the max x and max y coordinates of a tile are non-inclusive.
65//
66struct Tile
67{
68 Tile() :
69 mMinX(0),
70 mMaxX(1),
71 mMinY(0),
72 mMaxY(1) {}
73
74 Tile(unsigned minX, unsigned maxX, unsigned minY, unsigned maxY) :
75 mMinX(minX),
76 mMaxX(maxX),
77 mMinY(minY),
78 mMaxY(maxY) {}
79
80 unsigned getExtentX() const { return mMaxX - mMinX; }
81 unsigned getExtentY() const { return mMaxY - mMinY; }
82 unsigned getArea() const { return getExtentX() * getExtentY(); }
83
84 unsigned mMinX; // Inclusive.
85 unsigned mMaxX; // Non-inclusive.
86 unsigned mMinY; // Inclusive.
87 unsigned mMaxY; // Non-inclusive.
88};
89
90inline bool operator==(const Tile& lhs, const Tile &rhs)
91{
92 return lhs.mMinX == rhs.mMinX &&
93 lhs.mMaxX == rhs.mMaxX &&
94 lhs.mMinY == rhs.mMinY &&
95 lhs.mMaxY == rhs.mMaxY;
96}
97
98inline bool operator!=(const Tile& lhs, const Tile &rhs)
99{
100 return !(lhs == rhs);
101}
102
103inline std::ostream& operator<<(std::ostream& outs, const Tile& t)
104{
105 return outs << '[' << t.mMinX << ", " << t.mMaxX << "] [" <<
106 t.mMinY << ", " << t.mMaxY << ']';
107}
