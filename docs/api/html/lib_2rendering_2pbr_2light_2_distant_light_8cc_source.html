10// mrday - 9/27/17 - DistantLight overhaul
11// ---------------------------------------
12// mslee and I tracked down some bugs and accuracy issues with distant lights. We also wanted to
13// to rework the behaviour of 'normalised' distant lights, so an overhaul was performed. The
14// following points came up during the work.
15//
16// * The new physical interpretation of a distant light is that it's a spherical cap, where the
17// sphere in question is centred on the observer and has infinite radius. This replaces the prior
18// interpretation as a disk light at infinity (whose implementation suffered from subtle bugs).
19//
20// * One accuracy issue was the area computation. The area of the cap is 2pi * (1-cosThetaMax),
21// where ThetaMax represents the angle between the centre of the light and the edge (i.e. half
22// the angular diameter). When cosThetaMax approaches 1, the expression 1-cosThetaMax suffers from
23// catastrophic cancellation, and as a result of this something later in the pipeline was breaking
24// down around thetaMax=0.15 degrees (so 0.3 degrees for the light's angular diameter). This was
25// easy to fix, though, using a trig identity:
26//
27// 1-cos(t) = 2(sin(t))^2.
28//
29// Using sine maintains accuracy all the way down to a zero angle. In fact, an accurate 1-cosThetaMax was
30// useful elsewhere, so we store it in the light's mVersineThetaMax. The versine function is 1-cosine.
31// With this improvement, the light's angle works all the way down to about 0.00000004 degrees - it could
32// perhaps be clamped at this very small value in case it's set to zero (representing a pure directional
33// light).
34//
35// * A somewhat related accuracy issue came up the angle is small. That was in the function
36// sampleLocalSphericalCapUniform() which generates the samples over the light's extent.
37// The newly added version of this funciton now takes as a parameter the more accurately computed
38// versine, rather than the cosine of ThetaMax. It also uses a more numerically robust method to
39// recover the sine from the cosine (or versine). The original calculation was
40//
41// sin(theta) = sqrt(1 - (cos(theta))^2)
42//
43// but this suffers from catastrophic cancellation when theta is near zero. So it was replaced with
44//
45// sin(theta) = sqrt(versine(theta) * (2.0f - versine(theta))
46//
47// which is mathematically equivalent but in floating point is far more numerically accurate for small
48// angles. This fixed a problem with shadow penumbras from small distant lights, in which catastrophic
49// cancellation made it possible to see discrete steps correspoinding to changes of 1 ULP in the cosine
50// value. The numerically robust version eliminates the steps and gives the desired smooth gradient.
51//
52// * There is still one potential accuracy issue. This is the way thresholding is performed in
53// DistantLight::intersect(). By comparing the dot product to cosThetaMax, we lose considerable accuracy
54// when both values are close to 1. But it doesn't appear to cause any trouble so it can be left as-is
55// for now.
56//
57// * Normalisation of distant lights has been reworked. The new definition of 'normalised' for distant
58// lights is that when a distant light of uniform radiance (1,1,1) and any angular extent is placed
59// directly overhead above a Lambertian surface of colour (1,1,1), the resulting outgoing radiance will
60// be (1,1,1). This requires applying an angle-dependent normalisation factor which was computed by
61// integration and derived directly from the rendering equation. The required normalisation factor is
62//
63// 1/(sin(thetaMax))^2
64//
65// after clamping thetaMax to be no greater than a right-angle (since if the distant light covers more
66// than a hemisphere, only the upper hemisphere contributes light to the surface).
67// See DistantLight::update() for the implementation.

135bool
136DistantLight::update(const Mat4d& world2render)
137{
138 MNRY_ASSERT(mRdlLight);
139
140 mOn = mRdlLight->get(scene_rdl2::rdl2::Light::sOnKey);
141 if (!mOn) {
142 return false;
143 }
144
145 updateVisibilityFlags();
146 updatePresenceShadows();
147 updateRayTermination();
148 updateMaxShadowDistance();
149
150 const Mat4d l2w0 = mRdlLight->get(scene_rdl2::rdl2::Node::sNodeXformKey, /* rayTime = */ 0.0f);
151 const Mat4d l2w1 = mRdlLight->get(scene_rdl2::rdl2::Node::sNodeXformKey, /* rayTime = */ 1.0f);
152 const Mat4f local2render0 = Mat4f::orthonormalize(sRotateX180 * toFloat(l2w0 * world2render));
153 const Mat4f local2render1 = Mat4f::orthonormalize(sRotateX180 * toFloat(l2w1 * world2render));
154 ReferenceFrame frame0(local2render0);
155 ReferenceFrame frame1(local2render1);
156 mPosition[0] = mPosition[1] = zero;
157 mOrientation[0] = normalize(math::Quaternion3f(frame0.getX(), frame0.getY(), frame0.getZ()));
158 mOrientation[1] = normalize(math::Quaternion3f(frame1.getX(), frame1.getY(), frame1.getZ()));
159 if (dot(mOrientation[0], mOrientation[1]) < 0) {
160 mOrientation[1] *= -1.f;
161 }
162 mFrame = frame0;
163 mDirection = mFrame.getZ();
164
165 // setup mMb
166 mMb = LIGHT_MB_NONE;
167 const scene_rdl2::rdl2::SceneVariables &vars =
168 getRdlLight()->getSceneClass().getSceneContext()->getSceneVariables();
169 if (vars.get(scene_rdl2::rdl2::SceneVariables::sEnableMotionBlur) &&
170 getRdlLight()->get(scene_rdl2::rdl2::Light::sMbKey) &&
171 (!isEqual(frame0.getX(), frame1.getX()) ||
172 !isEqual(frame0.getY(), frame1.getY()) ||
173 !isEqual(frame0.getZ(), frame1.getZ()))) {
174 mMb = LIGHT_MB_ROTATION;
175 }
176
177 float angularExtent = mRdlLight->get(sAngularExtent);
178 float halfAngle = deg2rad(angularExtent * 0.5f);
179
180 mCullThreshold = scene_rdl2::math::cos(halfAngle + sHalfPi);
181 MNRY_ASSERT(mCullThreshold <= 0.0f);
182
183 mCosThetaMax = scene_rdl2::math::cos(halfAngle);
184 float sinQuarterAngle = scene_rdl2::math::sin(halfAngle * 0.5f);
185 mVersineThetaMax = 2.0f * sinQuarterAngle * sinQuarterAngle;
186
187 // We store solid angle and its inverse in the area members, since this is also useful for the unittest.
188 // We use an accurately computed versine here because 1-cosine suffers from catastrophic cancellation
189 // when the angle is small.
190 mArea = mVersineThetaMax * sTwoPi;
191 mInvArea = 1.0f / mArea;
192
193 // Compute radiance.
194 mRadiance = computeLightRadiance(mRdlLight, scene_rdl2::rdl2::Light::sColorKey,
195 scene_rdl2::rdl2::Light::sIntensityKey, scene_rdl2::rdl2::Light::sExposureKey);
196
197 // Apply normalisation factor if light is normalised. The normalisation factor used here is such that
198 // if a distant light of radiance (1,1,1) is directly overhead a Lambertian surface of colour (1,1,1),
199 // the resulting outgoing radiance at the surface will be (1,1,1) regardless of the light's angular
200 // extent. This factor can be derived from the rendering equation.
201 if (mRdlLight->get<scene_rdl2::rdl2::Bool>(sNormalizedKey)) {
202 float s = scene_rdl2::math::sin(min(halfAngle,sHalfPi));
203 mRadiance *= 1.0f / (s*s);
204 }
205
206 if (isBlack(mRadiance)) {
207 mOn = false;
208 return false;
209 }
210
211 return true;
212}

214bool
215DistantLight::canIlluminate(const Vec3f p, const Vec3f *n, float time, float radius,
216 const LightFilterList* lightFilterList) const
217{
218 MNRY_ASSERT(mOn);
219
220 if (n && dot(-(*n), getDirection(time)) < mCullThreshold) {
221 return false;
222 }
223
224 if (lightFilterList) {
225 return canIlluminateLightFilterList(lightFilterList,
226 { getPosition(time),
227 math::inf, p, globalToLocalXform(time, lightFilterList->needsLightXform()),
228 radius, time
229 });
230 }
231
232 return true;
233}

253bool
254DistantLight::intersect(const Vec3f &p, const Vec3f *n, const Vec3f &wi, float time,
255 float maxDistance, LightIntersection &isect) const
256{
257 if (dot(-wi, getDirection(time)) < mCosThetaMax) {
258 return false;
259 }
260
261 if (sDistantLightDistance > maxDistance) {
262 return false;
263 }
264
265
266 isect.N = -wi;
267 isect.distance = sDistantLightDistance;
268 isect.uv = zero;
269
270 return true;
271}

273bool
274DistantLight::sample(const Vec3f &p, const Vec3f *n, float time, const Vec3f& r,
275 Vec3f &wi, LightIntersection &isect, float rayDirFootprint) const
276{
277 MNRY_ASSERT(mOn);
278
279 Vec3f sample = shading::sampleLocalSphericalCapUniform2(r[0], r[1], mVersineThetaMax);
280 wi = -localToGlobal(sample, time);
281
282 if (n && dot(*n, wi) < sEpsilon) {
283 return false;
284 }
285
286 isect.N = -wi;
287 isect.distance = sDistantLightDistance;
288 isect.uv = zero;
289
290 return true;
291}

293Color
294DistantLight::eval(mcrt_common::ThreadLocalState* tls, const Vec3f &wi, const Vec3f &p, const LightFilterRandomValues& filterR,
295 float time, const LightIntersection &isect, bool fromCamera, const LightFilterList *lightFilterList,
296 float rayDirFootprint, float *pdf) const
297{
298 MNRY_ASSERT(mOn);
299
300 Color radiance = mRadiance;
301
302 if (lightFilterList) {
303 evalLightFilterList(lightFilterList,
304 { tls, &isect, getPosition(time),
305 getDirection(time), p,
306 filterR, time,
307 globalToLocalXform(time, lightFilterList->needsLightXform()),
308 wi
309 },
310 radiance);
311 }
312
313 if (pdf) {
314 *pdf = mInvArea;
315 }
316 return radiance;
317}
