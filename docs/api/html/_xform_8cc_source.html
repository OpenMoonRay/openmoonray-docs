71Xform::Xform(const rdl2::SceneObject *shader,
72 const rdl2::Node *customObject,
73 const rdl2::Camera *customCamera,
74 const std::array<float, 4>* customWindow)
75{
76 MNRY_ASSERT(shader);
77 const rdl2::SceneContext* ctx = shader->getSceneClass().getSceneContext();
78
79 // Render to object
80 scene_rdl2::math::Mat4d objectMatrix(math::one);
81 if (customObject != nullptr) { // use custom geometry for object space
82 objectMatrix = getR2W(ctx) * getW2O(customObject);
83 mIspc.mUseExternalObj = true;
84 } else {
85 // Use the space of shading point's object
86 // Since this is only available during sample, we don't store
87 // a transformation here
88 mIspc.mGetR2OFn = (intptr_t) getR2O;
89 mIspc.mUseExternalObj = false;
90 }
91
92 precalculateMatrices(shader,
93 objectMatrix,
94 customCamera,
95 customWindow);
96}

112void
113Xform::precalculateMatrices(const rdl2::SceneObject *shader,
114 const scene_rdl2::math::Mat4d& r2oMatrix,
115 const rdl2::Camera *customCamera,
116 const std::array<float, 4>* customWindow)
117{
118 MNRY_ASSERT(shader);
119 const rdl2::SceneContext* ctx = shader->getSceneClass().getSceneContext();
120
121 std::array<float, 4> window;
122 // Build window out of screen's aspect ratio, if custom window is not provided
123 if (customWindow == nullptr) {
124 const rdl2::SceneVariables &vars = ctx->getSceneVariables();
125 const math::HalfOpenViewport& aperture = vars.getRezedApertureWindow();
126 const float invAspectRatio = float(aperture.height()) / float(aperture.width());
127 window = {-1.0f, -invAspectRatio, 1.0f, invAspectRatio};
128 } else {
129 window = *customWindow;
130 }
131
132 // Pre-calculate various xforms and their inverse xforms
133 std::vector<const rdl2::Camera*> cameras = ctx->getActiveCameras();
134 for (size_t i = 0; i < cameras.size(); i++) {
135 // Use current active camera, if custom camera is not provided
136 // Render to Camera
137 const rdl2::Camera* cam = (customCamera == nullptr) ? cameras[i] : customCamera;
138 math::asCpp(mIspc.mR2C[i]) = math::xform<math::Xform3f>(getR2W(ctx) * getW2C(cam));
139 math::asCpp(mIspc.mR2CInv[i]) = math::asCpp(mIspc.mR2C[0]).inverse();
140 // Render to screen
141 math::asCpp(mIspc.mR2S[i]) = toFloat(getR2W(ctx) * getW2C(cam)) * getC2S(cam, window);
142 math::asCpp(mIspc.mR2SInv[i]) = math::asCpp(mIspc.mR2S[i]).inverse();
143 }
144
145 // Render to world
146 math::asCpp(mIspc.mR2W) = math::xform<math::Xform3f>(getR2W(ctx));
147 math::asCpp(mIspc.mR2WInv) = math::asCpp(mIspc.mR2W).inverse();
148
149 // Render to object
150 math::asCpp(mIspc.mR2O) = math::xform<math::Xform3f>(r2oMatrix);
151 math::asCpp(mIspc.mR2OInv) = math::asCpp(mIspc.mR2O).inverse();
152}

160math::Vec3f
161Xform::transformPoint(const int srcSpace,
162 const int dstSpace,
163 const State &state,
164 const math::Vec3f inPoint) const
165{
166 math::Vec3f rPoint;
167 math::Vec3f result;
168 int cameraId = state.getCameraId();
169
170 // Transform to render space if necessary
171 switch(srcSpace) {
172 case ispc::SHADING_SPACE_RENDER:
173 rPoint = inPoint;
174 break;
175 case ispc::SHADING_SPACE_CAMERA:
176 rPoint = math::transformPoint(math::asCpp(mIspc.mR2CInv[cameraId]), inPoint);
177 break;
178 case ispc::SHADING_SPACE_WORLD:
179 rPoint = math::transformPoint(math::asCpp(mIspc.mR2WInv), inPoint);
180 break;
181 case ispc::SHADING_SPACE_SCREEN:
182 rPoint = math::transformH(math::asCpp(mIspc.mR2SInv[cameraId]), inPoint);
183 break;
184 case ispc::SHADING_SPACE_OBJECT:
185 {
186 if (mIspc.mUseExternalObj) {
187 rPoint = math::transformPoint(math::asCpp(mIspc.mR2OInv), inPoint);
188 } else {
189 math::Xform3f r2o;
190 getR2O(state.getGeometryObject(), &r2o);
191 rPoint = math::transformPoint(r2o.inverse(), inPoint);
192 }
193 }
194 break;
195 default:
196 MNRY_ASSERT(0 && "unknown src space");
197 break;
198 }
199
200 // Transform from render space
201 switch(dstSpace) {
202 case ispc::SHADING_SPACE_RENDER:
203 result = rPoint;
204 break;
205 case ispc::SHADING_SPACE_CAMERA:
206 result = math::transformPoint(math::asCpp(mIspc.mR2C[cameraId]), rPoint);
207 break;
208 case ispc::SHADING_SPACE_WORLD:
209 result = math::transformPoint(math::asCpp(mIspc.mR2W), rPoint);
210 break;
211 case ispc::SHADING_SPACE_SCREEN:
212 result = math::transformH(math::asCpp(mIspc.mR2S[cameraId]), rPoint);
213 break;
214 case ispc::SHADING_SPACE_OBJECT:
215 {
216 if (mIspc.mUseExternalObj) {
217 result = math::transformPoint(math::asCpp(mIspc.mR2O), rPoint);
218 } else {
219 math::Xform3f r2o;
220 getR2O(state.getGeometryObject(), &r2o);
221 result = math::transformPoint(r2o, rPoint);
222 }
223 }
224 break;
225 default:
226 MNRY_ASSERT(0 && "unknown dst space");
227 break;
228 }
229
230 return result;
231}
