18// Description:
19//
20// Writes to static variables are a major cause of race conditions in
21// threaded code. The Intel compiler has warnings that can trigger for
22// the following accesses to static variables:
23//
24// static int x=0;
25// if(x>1) {}; // warning #1710: reference to statically allocated variable
26// x = 1; // warning #1711: assignment to statically allocated variable
27// int* p=&x; // warning #1712: address taken of statically allocated variable
28//
29//
30// 1712 currently triggers in a lot of system-level code that cannot
31// be avoided. There is an open issue with Intel about this.
32//
33// To catch this condition the warning can be enabled by default in
34// Intel compiler builds and treated as an error. The macro in this
35// file can then be used to disable this warning under two conditions:
36//
37// * the code is 100% guaranteed to be threadsafe
38// * the code is not 100% guaranteed to be threadsafe
39//
40// There should be no code in the second category long term, it is
41// only provided as a temporary convenience for code that will soon
42// be cleaned up. Code in the first category can be tagged as
43// threadsafe for one of the following reasons:
44//
45// * the variable is guaranteed never to be modified and accessed
46// from more than one thread simultaneously
47// * it is protected by a lock or other threading primitive
48// * it can be proven to be safe some other way
49//
50// Needless to say it is important to be sure that code tagged as safe
51// really is and remains safe. The tagged code should be commented to
52// indicate why it is threadsafe, eg:
53//
54// static atomic<int> x = 0;
55// MOONRAY_THREADSAFE_STATIC_WRITE(x = 1; ) // safe since x is atomic variable
56//
57// or:
58//
59// static atomic<int> x = 0;
60// MOONRAY_START_THREADSAFE_STATIC_WRITE
61// x = 1; // safe since x is atomic variable
62// [...] // more threadsafe code
63// MOONRAY_FINISH_THREADSAFE_STATIC_WRITE
64//
65// Since this option only exists with the Intel compiler, it can be
66// enabled only for that compiler, and is a noop on gcc.
67
68
69#if defined(__ICC)
70
71// use these defines to bracket a region of code that has safe static
72// accesses. Keep the region as small as possible
73#define MOONRAY_START_THREADSAFE_STATIC_REFERENCE __pragma(warning(disable:1710))
74#define MOONRAY_FINISH_THREADSAFE_STATIC_REFERENCE __pragma(warning(default:1710))
75#define MOONRAY_START_THREADSAFE_STATIC_WRITE __pragma(warning(disable:1711))
76#define MOONRAY_FINISH_THREADSAFE_STATIC_WRITE __pragma(warning(default:1711))
77#define MOONRAY_START_THREADSAFE_STATIC_ADDRESS __pragma(warning(disable:1712))
78#define MOONRAY_FINISH_THREADSAFE_STATIC_ADDRESS __pragma(warning(default:1712))
79
80// use these defines to bracket a region of code that has unsafe
81// static accesses. Keep the region as small as possible
82#define MOONRAY_START_NON_THREADSAFE_STATIC_REFERENCE __pragma(warning(disable:1710))
83#define MOONRAY_FINISH_NON_THREADSAFE_STATIC_REFERENCE __pragma(warning(default:1710))
84#define MOONRAY_START_NON_THREADSAFE_STATIC_WRITE __pragma(warning(disable:1711))
85#define MOONRAY_FINISH_NON_THREADSAFE_STATIC_WRITE __pragma(warning(default:1711))
86#define MOONRAY_START_NON_THREADSAFE_STATIC_ADDRESS __pragma(warning(disable:1712))
87#define MOONRAY_FINISH_NON_THREADSAFE_STATIC_ADDRESS __pragma(warning(default:1712))
88
89// simpler version for one-line cases
90#define MOONRAY_THREADSAFE_STATIC_REFERENCE(CODE) __pragma(warning(disable:1710)); CODE; __pragma(warning(default:1710))
91#define MOONRAY_THREADSAFE_STATIC_WRITE(CODE) __pragma(warning(disable:1711)); CODE; __pragma(warning(default:1711))
92#define MOONRAY_THREADSAFE_STATIC_ADDRESS(CODE) __pragma(warning(disable:1712)); CODE; __pragma(warning(default:1712))
93
94#else // gcc does not support these compiler warnings
