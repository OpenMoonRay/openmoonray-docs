40Bssrdf::Bssrdf(scene_rdl2::alloc::Arena *arena, const Bssrdfv &bssrdfv, int lane)
41 : mFresnel(nullptr)
42{
43 if (bssrdfv.mFresnel && ((1 << lane) & bssrdfv.mFresnelMask)) {
44 setTransmissionFresnel(createFresnel(arena, bssrdfv.mFresnel, lane));
45 }
46
47 mFrame.mX = Vec3f(bssrdfv.mFrame.mX.x[lane], bssrdfv.mFrame.mX.y[lane], bssrdfv.mFrame.mX.z[lane]);
48 mFrame.mY = Vec3f(bssrdfv.mFrame.mY.x[lane], bssrdfv.mFrame.mY.y[lane], bssrdfv.mFrame.mY.z[lane]);
49 mFrame.mZ = Vec3f(bssrdfv.mFrame.mZ.x[lane], bssrdfv.mFrame.mZ.y[lane], bssrdfv.mFrame.mZ.z[lane]);
50 mScale = Color(bssrdfv.mScale.r[lane], bssrdfv.mScale.g[lane], bssrdfv.mScale.b[lane]);
51 mMaxRadius = bssrdfv.mMaxRadius[lane];
52 mLabel = bssrdfv.mLabel;
53 mPropertyFlags = bssrdfv.mPropertyFlags;
54 mBssrdfTraceSet = (const scene_rdl2::rdl2::TraceSet *)bssrdfv.mBssrdfTraceSet;
55 mMaterial = (const scene_rdl2::rdl2::Material*)bssrdfv.mMaterial;
56 mEvalNormalFunc = (scene_rdl2::rdl2::EvalNormalFunc)bssrdfv.mEvalNormalFn;
57}

62Bssrdf::sampleGlobal(float r1, float r2, float cosThetaMax, Vec3f &ws) const
63{
64 // Sample direction within a spherical cap below the surface
65 ws = sampleLocalSphericalCapUniform(r1, r2, cosThetaMax);
66
67 float pdf = 1.0f / (sTwoPi * (1.0f - cosThetaMax));

81Bssrdf *
82createBSSRDF(ispc::SubsurfaceType type,
83 scene_rdl2::alloc::Arena* arena,
84 const Vec3f &N,
85 const Color& trlColor,
86 const Color& trlRadius,
87 const scene_rdl2::rdl2::Material* material,
88 const scene_rdl2::rdl2::EvalNormalFunc evalNormalFn)
89{
90 Bssrdf *bssrdf = nullptr;
91
92 switch(type) {
93 case ispc::SubsurfaceType::SUBSURFACE_DIPOLE_DIFFUSION:
94 bssrdf = arena->allocWithArgs<DipoleBssrdf>(N, /* eta = */ 1.0f, trlColor, trlRadius, material, evalNormalFn);
95 break;
96
97 case ispc::SubsurfaceType::SUBSURFACE_NORMALIZED_DIFFUSION:
98 default :
99 bssrdf = arena->allocWithArgs<NormalizedDiffusionBssrdf>(N, trlColor, trlRadius, material, evalNormalFn);
100 break;
101 }
102 return bssrdf;
103}

105Bssrdf *
106createBSSRDF(scene_rdl2::alloc::Arena* arena,
107 const Bssrdfv *bssrdfv,
108 int lane)
109{
110 Bssrdf *bssrdf = nullptr;
111 if (bssrdfv && ((1 << lane) & bssrdfv->mMask)) {
112
113 switch (bssrdfv->mType) {
114 case ispc::SUBSURFACE_DIPOLE_DIFFUSION:
115 bssrdf = arena->allocWithArgs<DipoleBssrdf>(arena, *((const DipoleBssrdfv *)bssrdfv), lane);
116 break;
117
118 case ispc::SUBSURFACE_NORMALIZED_DIFFUSION:
119 bssrdf = arena->allocWithArgs<NormalizedDiffusionBssrdf>(arena, *((const NormalizedDiffusionBssrdfv *)bssrdfv), lane);
120 break;
121
122 default:
123 MNRY_ASSERT(0);
124 }
125 }
126 return bssrdf;
127}
