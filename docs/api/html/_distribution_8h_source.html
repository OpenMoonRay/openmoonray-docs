55/// @class GuideDistribution1D Distribution.h <pbr/core/Distribution.h>
56/// @brief A utility object that can sample according to a a piecewise constant
57/// 1D distribution. This class uses the method of guide table from Chen and
58/// Asau (1974):
59/// Chen, H.-C. and Asau, Y. (1974). On generating random variates from an
60/// empirical distribution. AIEE transactions, 6:153–166.
61/// We store a cdf table and a guide table of the same size.
62///
63class GuideDistribution1D
64{
65public:
66 typedef uint32_t size_type;
67
68 /// Constructor / Destructor
69 /// Weights and cdf are left un-initialized
70 explicit GuideDistribution1D(const size_type size);
71 explicit GuideDistribution1D(const size_type size, float *const cdf, uint32_t *const guide);
72 GuideDistribution1D(GuideDistribution1D&& other);
73 GuideDistribution1D& operator=(GuideDistribution1D&& other);
74
75 ~GuideDistribution1D();
76
77

79 static uint32_t hudValidation(const bool verbose)
80 {
81 GUIDE_DISTRIBUTION_1D_VALIDATION;
82 }
83 HUD_AS_ISPC_METHODS(GuideDistribution1D);
84
85
86 /// Accessors
87 finline size_type getSize() const
88 {
89 return mSizeCdf;
90 }
91
92 finline float getInvSize() const
93 {
94 return mInvSizeCdf;
95 }

97 finline float getTotalWeight() const
98 {
99 return mTotalWeight;
100 }

114 return mLinearCoeffLow;
115 }
116
117 finline float getLinearCoeffHigh() const
118 {
119 return mLinearCoeffHigh;
120 }
121
122 finline size_type getGuideIndex(const float r) const
123 {
124 MNRY_ASSERT(r >= 0.0f && r <= 1.0f);
125 const float rj = r * mSizeGuide;
126 size_type j = static_cast<size_type>(rj);
127 j = scene_rdl2::math::min(j, mSizeGuide - 1);
128 return mGuide[j];
129 }
130
131 finline const float *getCdf() const
132 {
133 return mCdf;
134 }
135
136
137 /// Setup the distribution weight function. Make sure to initialize each
138 /// weight with a value strictly > 0.0. The weights are normalized into a
139 /// proper pdf during tabulateCdf().
140 finline void setWeight(const size_type index, const float weight)
141 {
142 mCdf[index] = weight;
143 }
144
145 /// Helper to scale an existing weight.
146 finline void scaleWeight(const size_type index, const float scale)
147 {
148 mCdf[index] *= scale;
149 }
150
151 /// Make sure to compute the cdf when you're done setting ALL weights.
152 /// Returns the integral of the weight function.
153 float tabulateCdf();

155 float pdfDiscrete(const size_type index) const;
156
157 float pdfContinuous(const float u) const;
158
159 float pdfLinear(const float u) const;
160
161 /// Discrete sampling: returns the index of the segment in the distribution
162 /// corresponding to the uniform parameter r in [0,1)
163 /// Optionally returns the pdf of sampling this segment of the distribution
164 /// Returns the index of the interval in the discrete distribution where the value r falls.
165 /// rRemapped represents the fractional distance of r along that interval.
166 size_type sampleDiscrete(const float r, float *const pdf = nullptr, float *const rRemapped = nullptr) const;
167
168 /// Continuous sampling: returns the continuous variable u in [0,1), with

172 /// Optionally returns the index of the sampled segment
173 float sampleContinuous(const float r, float *const pdf = nullptr, size_type *const index = nullptr) const;
174
175 /// Sample from a linearly filtered distribution
176 float sampleLinear(const float r, float *const pdf = nullptr) const;
177
178private:
179 /// Copy is disabled

181 const GuideDistribution1D &operator=(const GuideDistribution1D &other);

195///
196/// @class Distribution2D Distribution.h <pbr/core/Distribution.h>
197/// @brief A utility object that can sample according to a piecewise constant
198/// 2D distribution.
199///
200class Distribution2D
201{
202public:
203 enum Mapping {
204 NONE, ///< Undefined / un-initialized
205 PLANAR, ///< Weighted by the image luminance, no-op.
206 CIRCULAR, ///< Used for disk light texture sampling, pixels outside of
207 ///< a maximal inscribed circle are set to black so we don't
208 ///< sample them.
209 HEMISPHERICAL, ///< Assuming a latlong input, the upper half of image is
210 ///< weighted by math::sin(2pi * v), see (see pbrt section 14.6.5).
211 ///< The bottom half of image is set to zero so it's not
212 ///< accessed. (Currently the memory for the bottom half
213 ///< of the image weights goes to waste)
214 SPHERICAL, ///< Weighted by the image luminance * math::sin(2pi * v) over
215 ///< the full image.
216 };
217
218 typedef Distribution1D::size_type size_type;
219
220 /// Constructor / Destructor
221 /// Weights and cdf are left un-initialized
222 Distribution2D(size_type sizeU, size_type sizeV);
223 ~Distribution2D();
224
225 /// HUD validation and type casting
226 static uint32_t hudValidation(const bool verbose)
227 {
228 DISTRIBUTION_2D_VALIDATION;
229 }
230 HUD_AS_ISPC_METHODS(Distribution2D);
231
232 /// Accessors
233 finline size_type getSizeU() const
234 {
235 return mConditional[0]->getSize();
236 }
237
238 finline size_type getSizeV() const
239 {
240 return mSizeV;
241 }
242
243 /// Setup the distribution weight function. Make sure to initialize each
244 /// weight with a value strictly > 0.0. The weights are normalized into a
245 /// proper pdf during tabulateCdf().
246 finline void setWeight(const size_type u, const size_type v, const float weight)
247 {
248 mConditional[v]->setWeight(u, weight);
249 }
250
251 /// Helper to scale an existing weight.
252 finline void scaleWeight(const size_type u, const size_type v, const float scale)
253 {
254 mConditional[v]->scaleWeight(u, scale);
255 }
256
257 /// Make sure to compute the cdf when you're done setting ALL weights.
258 /// At this point a optional mapping can also be applied.
259 void tabulateCdf(const Mapping mapping = PLANAR);
260
261 float pdfNearest(const float u, const float v) const;
262 float pdfBilinear(const float u, const float v) const;
263
264 /// Continuous sampling: returns the continuous variable u in [0,1), with
265 /// density proportional to the distribution, that corresponds to the
266 /// uniform parameter ru in [0,1) (and similarly for v & rv).
267 /// These (u,v) values are passed back in *uv. The pointer uv must be non-null.
268 /// Optionally returns the pdf of sampling this segment of the distribution
269 finline void
270 sample(const float ru, const float rv, scene_rdl2::math::Vec2f *const uv, float *const pdf,
271 const TextureFilterType texFilter) const
272 {
273 MNRY_ASSERT(uv != nullptr);
274 switch (texFilter) {
275 case TEXTURE_FILTER_NEAREST:
276 sampleNearest(ru, rv, uv, pdf);
277 break;
278 case TEXTURE_FILTER_BILINEAR:
279 sampleBilinear(ru, rv, uv, pdf);
280 break;
281 default:
282 sampleNearest(ru, rv, uv, pdf);
283 break;
284 }
285 }
286
287 void sampleNearest(const float ru, const float rv, scene_rdl2::math::Vec2f *const uv, float *const pdf) const;
288 void sampleBilinear(const float ru, const float rv, scene_rdl2::math::Vec2f *const uv, float *const pdf) const;
289
290private:
291 /// Copy is disabled
292 Distribution2D(const Distribution2D &other);
293 const Distribution2D &operator=(const Distribution2D &other);
294
295 // Members
296 DISTRIBUTION_2D_MEMBERS;
297};

317 /// This Constructor takes all Image Based Lighting Color Correction (IBLCC) inputs,
318 /// and performs the color corrections *before* determining the luminence / energy
319 /// distribution.
320 /// Most inputs will be familiar to users but TME (TMI) is explained here:
321 /// TME (TMI) Temperature Color Correction
322 /// One parameter adjusts the color temperature by moving red and blue
323 /// in different directions (you don t have to specify a proper temperature
324 /// like 5300K). The second parameter is a shift along the magenta/green axis
325 /// Color Temperature (T): Red Gain = 1 T/2,
326 /// Blue Gain = 1 + T/2
327 /// Magenta/Green (M): Red Gain = 1 + M/3,
328 /// Green Gain = 1 M*2/3,
329 /// Blue Gain = 1 + M/3
330 /// Exposure (E): RGB Gain *= 2.0^E
