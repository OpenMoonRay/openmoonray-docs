17void
18Fb::garbageCollectUnusedBuffers()
19{
20 if (!mPixelInfoStatus) {
21 mActivePixelsPixelInfo.cleanUp();
22 mPixelInfoBufferTiled.cleanUp();
23 }
24
25 if (!mHeatMapStatus) {
26 mActivePixelsHeatMap.cleanUp();
27 mHeatMapSecBufferTiled.cleanUp();
28 mHeatMapNumSampleBufferTiled.cleanUp();
29 }
30
31 if (!mWeightBufferStatus) {
32 mActivePixelsWeightBuffer.cleanUp();
33 mWeightBufferTiled.cleanUp();
34 }
35
36 if (!mRenderBufferOddStatus) {
37 mActivePixelsRenderBufferOdd.cleanUp();
38 mRenderBufferOddTiled.cleanUp();
39 mRenderBufferOddNumSampleBufferTiled.cleanUp();
40 }
41
42 // try to do garbage collect AOV buffers
43 {
44 unsigned totalActiveAovBuffer = 0;
45 auto itr = mRenderOutput.begin();
46 while (1) {
47 if (itr == mRenderOutput.end()) break;
48
49 if ((itr->second)->garbageCollectUnusedBuffers()) {
50 totalActiveAovBuffer++;
51 itr++;
52 } else {
53 itr = mRenderOutput.erase(itr); // erase this entry
54 }
55 }
56 mRenderOutputStatus = (totalActiveAovBuffer)? true: false; // just in case we update condition
57 }
58}

148std::string
149Fb::showRenderBuffer(const std::string &hd) const
150{
151 int numTilesX = static_cast<int>(getNumTilesX());
152 int numTilesY = static_cast<int>(getNumTilesY());
153 int totalTiles = static_cast<int>(getTotalTiles());
154
155 static const int maxActiveTileToShow = 10;
156
157 std::ostringstream ostr;
158 ostr << hd << "mRenderBufferTiled {\n";
159 ostr << hd << " width:" << mRenderBufferTiled.getWidth() << '\n';
160 ostr << hd << " height:" << mRenderBufferTiled.getHeight() << '\n';
161 ostr << hd << " numTilesX:" << numTilesX << '\n';
162 ostr << hd << " numTilesY:" << numTilesY << '\n';
163 int activeTile = 0;
164 for (int tileId = 0; tileId < totalTiles; ++tileId) {
165 int pixOffset = tileId << 6;
166 uint64_t mask = mActivePixels.getTileMask(tileId);
167 if (mask) {
168 const RenderColor *firstRenderColorOfTile = mRenderBufferTiled.getData() + pixOffset;
169 ostr << " tileId:" << tileId << '\n';
170 if (activeTile < maxActiveTileToShow) {
171 ostr << showRenderBufferTile(hd + " ", mask, firstRenderColorOfTile) << '\n';
172 activeTile++;
173 if (activeTile == maxActiveTileToShow) {
174 ostr << " ... too many active tiles -> skip ...\n";
175 }
176 }
177 }
178 }
179 ostr << hd << "}";
180 return ostr.str();
181}

183std::string
184Fb::showRenderBufferTile(const std::string &hd,
185 const uint64_t mask,
186 const RenderColor *firstRenderColorOfTile) const
187{
188 std::ostringstream ostr;
189 ostr << hd << "RenderBufferTile {\n";
190 if (!mask) {
191 ostr << hd << " empty tile\n";
192 } else {
193 for (int pixY = 7; pixY >= 0; --pixY) {
194 ostr << hd << " ";
195 for (int pixX = 0; pixX < 8; ++pixX) {
196 int pixOffset = pixY * 8 + pixX;
197 if (mask & (0x1 << pixOffset)) {
198 const RenderColor &currPix = firstRenderColorOfTile[pixOffset];
199 int v = static_cast<int>(currPix[0] * 255.0f); // red
200 if (v < 0) v = 0;
201 else if (v > 255) v = 255;
202 ostr << std::setw(2) << std::hex << std::setfill('0') << v << ' ';
203 } else {
204 ostr << " . ";
205 }
206 }
207 ostr << '\n';
208 }
209 }
210 ostr << hd << "}";
211 return ostr.str();
212}
