69std::unique_ptr<Camera> cameraFactory(const rdl2::Camera* rdlCamera)
70{
71 const rdl2::SceneClass& cameraClass = rdlCamera->getSceneClass();
72 const std::string& className = cameraClass.getName();
73 if (className == "PerspectiveCamera") {
74 return std::unique_ptr<Camera>(new PerspectiveCamera(rdlCamera));
75 } else if (className == "OrthographicCamera") {
76 return std::unique_ptr<Camera>(new OrthographicCamera(rdlCamera));
77 } else if (className == "SphericalCamera") {
78 return std::unique_ptr<Camera>(new SphericalCamera(rdlCamera));
79 } else if (className == "DomeMaster3DCamera") {
80 return std::unique_ptr<Camera>(new DomeMaster3DCamera(rdlCamera));
81 } else if (className == "BakeCamera") {
82 return std::unique_ptr<Camera>(new BakeCamera(rdlCamera));
83 } else {
84 MNRY_ASSERT(!"Should not get here");
85 throw std::runtime_error("No valid camera type specified");
86 }
87}

105void
106Scene::updateActiveCameras(const std::vector<const rdl2::Camera *> &cameras)
107{
108 mCameras.clear();
109
110 // Render space == Primary (0) camera space at the first specified motion step.
111 mRender2World = cameras[0]->get(rdl2::Node::sNodeXformKey);
112 mWorld2Render = rtInverse(mRender2World);
113
114 for (auto rdlCamera : cameras) {
115 std::unique_ptr<Camera> camera = cameraFactory(rdlCamera);
116 camera->update(getWorld2Render());
117 mCameras.push_back(std::move(camera));
118 }

186void
187Scene::generateMeshLightGeometry(Light* light)
188{
189 MeshLight* meshLight = (MeshLight*) light;
190 rdl2::Geometry* refGeom = meshLight->getReferenceGeometry();
191
192 // Check that the mesh light has a reference geometry AND
193 // that geometry does not exist in the scene Layer
194 if (refGeom && !mRdlLayer->contains(refGeom)) {
195 // set layer for mesh light
196 meshLight->setLayer(mRdlSceneContext->
197 getSceneObject("MeshLightLayer")->asA<rdl2::Layer>());
198 // set attribute table for mesh light
199 setAttributeTableOnMeshLight(meshLight, refGeom->getName(), mRdlSceneContext);
200
201 geom::Procedural* procedural = refGeom->getProcedural();
202 if (procedural) {
203 // Set geometry on mesh light.
204 MeshLightSetter meshLightSetter(meshLight);
205 procedural->forEachPrimitive(meshLightSetter, false /*parallel*/);
206 }
207 }
208 // This builds the sampling BVH and does any other computation that occurs
209 // after all meshes have been submitted. If no meshes were submitted
210 // then this turns the light off.
211 meshLight->finalize();
212
213 // Get hold of the geometry accelerator's embree device
214 const RTCDevice& rtcDevice = mEmbreeAccel->getRTCDevice();
215 // This builds the embree BVH which is used for light intersection
216 meshLight->setEmbreeAccelerator(rtcDevice);
217
218 // Warn user if light has turned off during the mesh light generation process
219 if (!light->isOn()) {
220 meshLight->getRdlLight()->warn("MeshLight did not load. Check if reference geometry is correct.");
221 }
222}

227void
228Scene::preFrame(const LightAovs &lightAovs, mcrt_common::ExecutionMode executionMode,
229 rt::GeometryManager& geometryManager, bool forceMeshLightGeneration)
230{
231 const rdl2::SceneVariables &vars = mRdlSceneContext->getSceneVariables();
232
233 mPropagateVisibilityBounceType = vars.get(
234 rdl2::SceneVariables::sPropagateVisibilityBounceType);
235
236 // Update camera info
237 for (auto& camera : mCameras) {
238 camera->update(getWorld2Render());
239 }
240
241 // Update light-list
242 // Updating lights needs to happen after the mCamera->update() above
243 // All lights are updated as well
244 // Lights needs to be updated if there is a frame change or motion blur is toggled.
245 bool updateLights = mRdlLayer->lightSetsChanged() || vars.hasChanged(rdl2::SceneVariables::sFrameKey) ||
246 vars.hasChanged(rdl2::SceneVariables::sEnableMotionBlur);
247 if (updateLights) {
248 updateLightList();
249 } else {
250 for (int i = 0; i < getLightCount(); ++i) {
251 Light *light = getLight(i);
252 if (light->isMesh()) {
253 if (forceMeshLightGeneration) {
254 // If all the geometries in the scene have changed, either because of a frame change or other
255 // global setting, then the mesh light's geometry or map shader could have changed without any
256 // rdl attributes changing. We must completely update the mesh light.
257 light->update(getWorld2Render());
258 generateMeshLightGeometry(light);
259 } else {
260 // The attribute table is always updated in the render context. Therefore
261 // we must always set the attribute table to the mesh light even if the
262 // light itself does not need to be updated.
263 MeshLight* meshLight = (MeshLight*)light;
264 rdl2::Geometry* refGeom = meshLight->getReferenceGeometry();
265 setAttributeTableOnMeshLight(meshLight, refGeom->getName(), mRdlSceneContext);
266 }
267 }
268 }
269 }
270
271 // Update light filters. This must be done after we update the lights. Since light filters are mapped
272 // to lights, if we do update the lights, we must also update light filters.
273 if (mRdlLayer->lightFilterSetsChanged() || updateLights) {
274 updateLightFilters();
275 }
276
277 mVisibleLightList.clear();
278 mVisibleLightFilterLists.clear();
279 // Make the visible light list.
280 // Do 3 passes, adding the bounded lights (sphere, rect, disk, spot, cylinder) in the first pass
281 // distant lights in the second pass, env lights in the third pass. This is done
282 // (a) to separate out the unbounded lights, which cannot be put in the light acceleration structure, and
283 // (b) so that all distant lights precede all env lights, since distant lights are closer than env
284 // lights and can be tested for intersection first.
285 for (int pass=0; pass<3; pass++) {
286 for (auto iter = mLightList.begin(); iter != mLightList.end(); ++iter) {
287 Light* light = *iter;
288
289 // Add the bounded lights in pass 0, distant lights in pass 1, env lights in pass 2
290 if (( (pass == 0) && light->isBounded() ) ||
291 ( (pass == 1) && light->isDistant() ) ||
292 ( (pass == 2) && light->isEnv() ) ) {
293
294 if (light->getIsVisibleInCamera()) {
295 mVisibleLightList.push_back(light);
296 // lights visible in camera do not have light filters (for now)
297 mVisibleLightFilterLists.push_back(nullptr);
298 }
299 }
300 }
301 }
302
303 mVisibleLightSet.init(mVisibleLightList.data(), mVisibleLightList.size(), mVisibleLightFilterLists.data());
304
305 // Setup aov labels on pbr lights
306 for (int i = 0; i < getLightCount(); ++i) {
307 Light *light = getLight(i);
308 const rdl2::Light *rdlLight = light->getRdlLight();
309 const auto &label = rdlLight->get(rdl2::Light::sLabel);
310 int32_t labelId = label.empty()? -1: lightAovs.getLabelId(label);
311 light->setLabelId(labelId);
312 }
313
314 // Get hold of the geometry accelerator's embree device
315 const RTCDevice& rtcDevice = mEmbreeAccel->getRTCDevice();
316
317 // Create an acceleration structure for each light set.
318 // This can only be done after update() has been called for each light,
319 // since that's where the lights' bounding boxes are set.
320
321 // First the per-layer light sets
322 LightAccelerator* acc = mLightAccList.data();
323 MNRY_ASSERT(acc);
324 LightPtrList* lightPtrList = mLightSets.data();
325 for (unsigned int i=0; i<mLightSets.size(); i++, acc++, lightPtrList++) {
326 acc->init(lightPtrList->data(), lightPtrList->size(), rtcDevice);
327 }
328
329 // Finally the visible light set
330 size_t visibleLightCount = mVisibleLightSet.getLightCount();
331 acc->init(mVisibleLightSet.getLights(), visibleLightCount, rtcDevice);
332 int * visibleLightAcceleratorIndexMap = new int[visibleLightCount];
333 for (size_t i = 0; i < visibleLightCount; ++i) {
334 visibleLightAcceleratorIndexMap[i] = i;
335 }
336 mVisibleLightAcceleratorIndexMap.reset(visibleLightAcceleratorIndexMap);
337 mVisibleLightSet.setAccelerator(acc, mVisibleLightAcceleratorIndexMap.get());
338
339 // If we need any uv bake maps, now is the time.
340 for (size_t i = 0; i < mCameras.size(); i++) {
341 mCameras[i]->bakeUvMaps();
342 }
343
344 // Build a volumeId to labelId mapping for LPE aov support
345 const geom::internal::VolumeAssignmentTable * volumeAssignmentTable =
346 geometryManager.getVolumeAssignmentTable();
347 int volumeCount = volumeAssignmentTable->getVolumeCount();
348 mVolumeLabelIds.resize(volumeCount, -1);
349 for (int volumeId = 0; volumeId < volumeCount; ++volumeId) {
350 const rdl2::VolumeShader* volumeShader =
351 volumeAssignmentTable->lookupWithVolumeId(volumeId);
352 const auto& label = volumeShader->get(rdl2::VolumeShader::sLabel);
353 int labelId = label.empty()? -1: lightAovs.getLabelId(label);
354 mVolumeLabelIds[volumeId] = labelId;
355 }
356
357 // does this scene have volumes?
358 mHasVolume = volumeAssignmentTable->getVolumeCount() > 0;
359
360 // Store the list of emissive volume primitives
361 geometryManager.getEmissiveRegions(mRdlLayer, mEmissiveRegions);
362
363 // If the ShadowSets or the geometries changed, we must change the shadow linkings.
364 if (mRdlLayer->shadowSetsChanged() || !mRdlLayer->getChangedOrDeformedGeometries().empty()) {
365 geometryManager.updateShadowLinkings(mRdlLayer);
366 }
367}

376void
377Scene::updateLightFilters()
378{
379 clearLightFilters();
380
381 // Gather unique rdl light filters
382 std::unordered_set<const rdl2::LightFilter *> allEnabledRdlLightFilters; // What we are gathering into
383 uint32_t assignmentCount = mRdlLayer->getAssignmentCount();
384 for (uint32_t i = 0; i < assignmentCount; ++i) { // Loop over rdl layers
385 const rdl2::LightSet *lightSet = mRdlLayer->lookupLightSet(i);
386 if (lightSet) {
387 // Gather lightfilters in all the lights.
388 const rdl2::SceneObjectVector &lights = lightSet->getLights();
389 for (auto it = lights.begin(); it != lights.end(); ++it) { // Loop over layer's lights
390 const rdl2::SceneObjectVector & rdlLightFilterObjs =
391 static_cast<const rdl2::Light *>(*it)->get(rdl2::Light::sLightFiltersKey);
392 for (auto it : rdlLightFilterObjs) { // Loop over light's filters
393 const rdl2::LightFilter *rdlLightFilter = it->asA<rdl2::LightFilter>();
394 MNRY_ASSERT(rdlLightFilter);
395 if (rdlLightFilter->isOn()) {
396 allEnabledRdlLightFilters.insert(rdlLightFilter); // Gather filter into set
397 rdlLightFilter->getReferencedLightFilters(allEnabledRdlLightFilters);
398 }
399 }
400 }
401 }
402 }
403
404 // Create list of unique pbr::LightFilters
405 for (const rdl2::LightFilter* rdlLightFilter : allEnabledRdlLightFilters) { // Loop over gathered rdl filters
406
407 const rdl2::SceneClass &lightFilterClass = rdlLightFilter->getSceneClass();
408 const std::string &className = lightFilterClass.getName();
409
410 LightFilter *lightFilter = nullptr;
411
412 if (className == "ColorRampLightFilter") { // Construct filter
413 lightFilter = new ColorRampLightFilter(rdlLightFilter);
414 } else if (className == "CombineLightFilter") {
415 lightFilter = new CombineLightFilter(rdlLightFilter);
416 } else if (className == "CookieLightFilter") {
417 lightFilter = new CookieLightFilter(rdlLightFilter);
418 } else if (className == "BarnDoorLightFilter") {
419 lightFilter = new BarnDoorLightFilter(rdlLightFilter);
420 } else if (className == "DecayLightFilter") {
421 lightFilter = new DecayLightFilter(rdlLightFilter);
422 } else if (className == "IntensityLightFilter") {
423 lightFilter = new IntensityLightFilter(rdlLightFilter);
424 } else if (className == "RodLightFilter") {
425 lightFilter = new RodLightFilter(rdlLightFilter);
426 } else if (className == "VdbLightFilter") {
427 lightFilter = new VdbLightFilter(rdlLightFilter);
428 } else {
429 MNRY_ASSERT(!"Unknown light filter type.");
430 }
431
432 if (lightFilter) {
433 mLightFilters[rdlLightFilter] = std::unique_ptr<LightFilter>(lightFilter); // Build map of rdl -> pbr filter
434 }
435 } // End loop over gathered filters
436
437 // Initialize (or update) all filters
438 mLightFilterNeedsSamples = false;
439 for (auto& lfEntry : mLightFilters) {
440 auto &pbrFilter = *lfEntry.second;
441 pbrFilter.update(mLightFilters, getWorld2Render()); // Init/update pbr filter
442
443 // Record if any light filter needs samples
444 mLightFilterNeedsSamples |= pbrFilter.needsSamples();
445 }
446
447 // There are several light filter related data structures:
448 //
449 // mLightFilters - Map from rdl filter to pbr filter
450 // mIdToLightFilterMap - Per-part LightFilterLists
451 // mLightFilterTable - All LightFilterLists
452 // mLightFilterListMap - Maps a layer to a map of a Light to a LightFilterList
453
454 mIdToLightFilterMap.reserve(assignmentCount);
455 mLightFilterTable.resize(assignmentCount);
456 // Number of unique pbr light filter sets
457 int currentLightFilterSets = 0;
458
459 // unique light filter lists
460 std::vector<std::unordered_set<const rdl2::LightFilter *>> uniqueRdlLightFilterLists;
461
462 for (uint32_t i = 0; i < assignmentCount; ++i) { // Loop over layers
463
464 // get rdl light filters for this layer assignment
465 const rdl2::LightFilterSet *lightFilterSet = mRdlLayer->lookupLightFilterSet(i);
466
467 // get pbr lightset for this assignment
468 const LightPtrList* lightPtrList = mIdToLightSetMap[i];
469 size_t lightCount = 0;
470 if (lightPtrList) {
471 lightCount = lightPtrList->size();
472 }
473
474 // create LightFilterLists for this assignment.
475 // Grab the current light filter set to fill in.
476 LightFilterLists& lightFilterLists = mLightFilterTable[currentLightFilterSets];
477 lightFilterLists.clear();
478
479 for (size_t j = 0; j < lightCount; ++j) { // Loop over lights
480 // get light
481 const Light* light = lightPtrList->at(j);
482 const rdl2::Light* rdlLight = light->getRdlLight();
483
484 // get list of light filters for this light
485 const rdl2::SceneObjectVector &rdlLightFilters = rdlLight->get(
486 rdl2::Light::sLightFiltersKey);
487 size_t lightFilterCount = rdlLightFilters.size();
488 std::unordered_set<const rdl2::LightFilter *> rdlLightFilterList;
489 // If there is a light filter set on this layer assignment, only include light filters in that set
490 for (size_t k = 0; k < lightFilterCount; ++k) { // Loop over light's filters
491 const rdl2::LightFilter* rdlLightFilter = rdlLightFilters[k]->asA<
492 rdl2::LightFilter>();
493 MNRY_ASSERT(rdlLightFilter);
494 if (!lightFilterSet || (lightFilterSet->contains(rdlLightFilter) && // Layer has no filter list OR
495 mLightFilters.find(rdlLightFilter) != mLightFilters.end())) { // pbr filter is in set & exists
496 if (rdlLightFilter->isOn()) {
497 rdlLightFilterList.insert(rdlLightFilter); // Place rdl light filter in set
498 }
499 }
500 }
501
502 // Check if filter list already exists
503 int existingIndex = -1;
504 for (int k = 0; k < uniqueRdlLightFilterLists.size(); ++k) { // Loop over unique filter sets
505 if (uniqueRdlLightFilterLists[k] == rdlLightFilterList) {
506 existingIndex = k; // Note matching set
507 break;
508 }
509 }
510
511 // a null light filter list represents an empty light filter list
512 LightFilterList* lightFilterList = nullptr;
513 if (existingIndex < 0) { // Filter list (set) is new
514 uniqueRdlLightFilterLists.push_back(rdlLightFilterList);
515 // create light filter list from rdl list
516 size_t activeLightFilterCount = rdlLightFilterList.size();
517 if (activeLightFilterCount > 0) { // List is not empty
518 std::unique_ptr<const LightFilter* []> lightFilters( // Create pbr list from rdl list
519 new const LightFilter*[activeLightFilterCount]);
520 size_t k = 0;
521 for (const rdl2::LightFilter* rdlLightFilter : rdlLightFilterList) { // Loop over filters in set
522 lightFilters[k++] = mLightFilters[rdlLightFilter].get(); // Save pointers to pbr filters
523 }
524 lightFilterList = new LightFilterList();
525 lightFilterList->init(std::move(lightFilters), activeLightFilterCount); // Init pbr filter list
526 }
527
528 // add this light filter list to the list of unique pbr light filter lists
529 mLightFilterLists.push_back(std::unique_ptr<LightFilterList>(lightFilterList)); // Save pbr filter list
530 } else {
531 // grab existing light filter list
532 lightFilterList = mLightFilterLists[existingIndex].get();
533 }
534
535 // Add this light filter list to the light filter list map, which is used for linking to volumes.
536 // Grab list of lists for this layer assignment
537 std::unordered_map<const Light*, const LightFilterList *>& lightToLflMap = mLightFilterListMap[i];
538 // set list for this light
539 lightToLflMap[light] = lightFilterList;
540
541 // add light filter list (for this light) to set of all light filter lists for this layer assignment
542 lightFilterLists.push_back(lightFilterList);
543 } // End loop over lights
544
545 // add LightFilterLists for this layer assignment
546 int existingIndex = -1;
547 // loop through all existing light filter sets except this one
548 // The == operator compares each element of the std::vector in order
549 // The lightFilterLists contains raw pointers to light filter list, a pointer can be null
550 // There is a light filter list per light in this layer assignment
551 for (int j = 0; j < currentLightFilterSets; ++j) { // Loop over filter lists
552 if (mLightFilterTable[j] == lightFilterLists) { // Does filter list already exist?
553 existingIndex = j;
554 break;
555 }
556 }
557
558 // Get light filter set for this layer assignment
559 const LightFilterLists* lfl = nullptr;
560 if (existingIndex < 0) {
561 // if we have not generated this light filter set before, it is unique and we can add it to
562 // mLightFilterTable
563 lfl = &mLightFilterTable[currentLightFilterSets++]; // If list is new, add to table
564 } else {
565 // If we have generated it before, grab the original instance of it.
566 lfl = &mLightFilterTable[existingIndex];
567 }
568
569 mIdToLightFilterMap.push_back(lfl); // Add list to map
570 }
571
572 // resize mLightFilterTable to the number of unique light filter sets.
573 mLightFilterTable.resize(currentLightFilterSets);
574}

615//----------------------------------------------------------------------------
616// Scene::updateLightList()
617// The current implementation is a heavy hammer that dumps all pbr lights,
618// and fully re-creates them. This causes light textures to be re-loaded and
619// their ImageDistribution to be re-computed which is not cheap.
620// We are only actually doing this work when a LightSet changes or a Layer
621// LightSet assignment changes.
622// TODO: optimize to handle incremental updates instead
623
624void
625Scene::updateLightList()
626{
627 clearLightList();
628
629 // Gather unique light sets in layer, and unique lights within those light
630 // sets.
631 std::set<const rdl2::LightSet *> rdlLightSets;
632 std::set<const rdl2::Light *> rdlLights;
633 uint32_t assignmentCount = mRdlLayer->getAssignmentCount();
634 for (uint32_t i = 0; i < assignmentCount; ++i) {
635 const rdl2::LightSet *lightSet = mRdlLayer->lookupLightSet(i);
636 if (lightSet) {
637 if (rdlLightSets.insert(lightSet).second) {
638 // Gather lights in this lightset.
639 const rdl2::SceneObjectVector &lights = lightSet->getLights();
640 for (auto it = lights.begin(); it != lights.end(); ++it) {
641 rdlLights.insert(static_cast<const rdl2::Light *>(*it));
642 }
643 }
644 }
645 }
646
647 // Allocate all runtime light sets and create mapping from corresponding
648 // rdl2 light sets.
649 size_t lightSetCount = rdlLightSets.size();
650 mLightSets.resize(lightSetCount);
651 std::map<const rdl2::LightSet *, int> lightSetMap;
652 int lightSetIndex = 0;
653 for (auto it = rdlLightSets.begin(); it != rdlLightSets.end(); ++it) {
654 lightSetMap[*it] = lightSetIndex++;
655 }
656
657 // Create all runtime lights and create mapping from corresponding rdl2
658 // lights.
659 size_t lightCount = rdlLights.size();
660 mLightList.reserve(lightCount);
661
662 mLightSetActiveList.reset(new bool[lightCount * lightSetCount]);
663 memset(mLightSetActiveList.get(), 0, sizeof(bool) * lightCount * lightSetCount);
664
665 for (const rdl2::Light* rdlLight : rdlLights) {
666 std::unique_ptr<Light> light(createLightFromRdlLight(rdlLight));
667 if (light) {
668 // Update the light (transformation, attributes, texture)
669 // update() can return false for a number of reasons, including the
670 // rdlLight being switched off, and the light's isBlack(radiance)
671 // evaluating to true. We should not include these lights in the
672 // scene's light list.
673
674 if (light->update(getWorld2Render())) {
675
676 // Generate the mesh for the mesh light
677 const std::string &className = rdlLight->getSceneClass().getName();
678 if (className == "MeshLight") {
679 generateMeshLightGeometry(light.get());
680 }
681 if (light->isOn()) {
682 std::hash<std::string> name_hash;
683 light->setHash(name_hash(rdlLight->getName()));
684 mLightList.push_back(light.release());
685 mRdlLightToLightMap[rdlLight] = mLightList.size() - 1;
686 }
687 }
688 }
689 }
690
691 // Hookup newly created lights to the relevant light sets.
692 for (const rdl2::LightSet* const rdlLightSet : rdlLightSets) {
693 // null light sets have been culled out above by this stage
694 MNRY_ASSERT(rdlLightSet);
695
696 const auto lightSetIdx = lightSetMap.find(rdlLightSet);
697 MNRY_ASSERT(lightSetIdx != lightSetMap.end());
698
699 LightPtrList &lightPtrList = mLightSets[lightSetIdx->second];
700 const rdl2::SceneObjectVector& lights = rdlLightSet->getLights();
701 lightPtrList.reserve(lights.size());
702
703 // Make 3 passes, adding the bounded lights (sphere, rect, disk, spot, cylinder) in the first pass
704 // distant lights in the second pass, env lights in the third pass. This is done in order to
705 // (a) separate out the unbounded lights, which cannot be put in the light acceleration structure, and
706 // (b) so that all distant lights precede all env lights, since distant lights are closer than env
707 // lights and can be tested for intersection first.
708 for (int pass=0; pass<3; pass++) {
709 for (const auto& lightobj : lights) {
710 const rdl2::Light* const rdlLight = static_cast<const rdl2::Light*>(lightobj);
711 const auto lightIdx = mRdlLightToLightMap.find(rdlLight);
712 if (lightIdx != mRdlLightToLightMap.end()) {
713 Light* light = mLightList[lightIdx->second];
714
715 // Add the bounded lights in pass 0, distant lights in pass 1, env lights in pass 2
716 if (( (pass == 0) && light->isBounded() ) ||
717 ( (pass == 1) && light->isDistant() ) ||
718 ( (pass == 2) && light->isEnv() ) ) {
719
720 lightPtrList.push_back(light);
721 size_t index = lightCount * lightSetIdx->second + lightIdx->second;
722 mLightSetActiveList[index] = true;
723 }
724 }
725 }
726 }
727 }
728
729 // Reserve memory for the light accelerators
730 mLightAccList.reserve(lightSetCount + 1);
731
732 // Create mapping from rdl2 assignment ids to runtime light sets and light accelerators
733 mIdToLightSetMap.reserve(assignmentCount);
734 mIdToLightAcceleratorMap.reserve(assignmentCount);
735 mIdToLightSetActiveList.reserve(assignmentCount);
736 for (uint32_t i = 0; i < assignmentCount; ++i) {
737 const rdl2::LightSet *rdlLightSet = mRdlLayer->lookupLightSet(i);
738 const LightPtrList *lightPtrList = nullptr;
739 const LightAccelerator *lightAccelerator = nullptr;
740
741 bool* lightSetActiveList = nullptr;
742 int remappedIdx = 0;
743 if (rdlLightSet) {
744 auto lightSetIdx = lightSetMap.find(rdlLightSet);
745 MNRY_ASSERT(lightSetIdx != lightSetMap.end());
746 remappedIdx = lightSetIdx->second;
747 lightPtrList = &mLightSets[remappedIdx];
748 lightAccelerator = &mLightAccList[remappedIdx];
749 lightSetActiveList = mLightSetActiveList.get() +
750 remappedIdx * lightCount;
751 }
752
753 // It's ok to add null pointers to this list.
754 mIdToLightSetMap.push_back(lightPtrList);
755 mIdToLightAcceleratorMap.push_back(lightAccelerator);
756 mIdToLightSetActiveList.push_back(lightSetActiveList);
757 }
758}

784Scene::intersectRay(mcrt_common::ThreadLocalState *tls, mcrt_common::Ray &ray, shading::Intersection &isect,
785 const int lobeType) const
786{
787 if (ray.getStart() >= ray.getEnd()) {
788 return false;
789 }
790
791 pbr::TLState *pbrTls = tls->mPbrTls.get();
792
793 pbrTls->mStatistics.incCounter(STATS_INTERSECTION_RAYS);
794
795 // Set the proper ray mask. If lobeType is 0, it's a primary ray from camera.
796 if (lobeType == 0) {
797 ray.mask = rdl2::CAMERA;
798 } else {
799 // Otherwise, we figure out the ray mask based on the lobe type.
800 // for ray mask propagation, we simply merge with existing ray mask.
801 // we should exclude the camera ray mask when propagating.
802 int lobeMask = lobeTypeToRayMask(lobeType);
803 ray.mask = mPropagateVisibilityBounceType
804 ? ((ray.mask | lobeMask) & ~rdl2::CAMERA)
805 : lobeMask;
806 }
807
808 // Intersect with manifold geometry.
809 {
810 EXCL_ACCUMULATOR_PROFILE(pbrTls, EXCL_ACCUM_EMBREE_INTERSECTION);
811 mEmbreeAccel->intersect(ray);
812 }
813
814 if (ray.geomID != -1) {
815 geom::initIntersectionPhase1(isect, tls, ray, mRdlLayer);
816 return true;
817 } else {
818 return false;
819 }

876void
877Scene::intersectVolumes(mcrt_common::ThreadLocalState* tls, const mcrt_common::Ray& ray,
878 int rayMask, const void* light, bool estimateInScatter) const
879{
880 pbr::TLState *pbrTls = tls->mPbrTls.get();
881
882 // need to reset tfar to inf so this probe ray can figure out whether
883 // we are inside a volume enclosed by manifold primitive
884 mcrt_common::Ray volumeRay(ray.org, ray.dir, ray.tnear, math::sMaxValue, ray.time);
885
886 pbrTls->mStatistics.incCounter(STATS_VOLUME_RAYS);
887 if (rayMask != rdl2::CAMERA &&
888 rayMask != rdl2::SHADOW &&
889 mPropagateVisibilityBounceType) {
890 rayMask = ((ray.mask | rayMask) & ~rdl2::CAMERA);
891 }
892 // Shift bits to "see" volume objects
893 volumeRay.mask = (rayMask << rdl2::sNumVisibilityTypes);
894 // Feed in tls for BVH to fill out all volume intersection along the ray
895 auto geomTls = tls->mGeomTls.get();
896 // volume intersections out of tfar range won't be added into
897 // collected intervals during the following intersection test
898 geomTls->resetVolumeRayState(ray.tfar, estimateInScatter);
899 volumeRay.ext.geomTls = (void*)geomTls;
900 volumeRay.ext.instance0OrLight = light; // for ShadowSet evaluation
901 // Intersect with volume geometry.
902 {
903 EXCL_ACCUMULATOR_PROFILE(pbrTls, EXCL_ACCUM_EMBREE_VOLUME);
904 mEmbreeAccel->intersect(volumeRay);
905 }
