58void
59ProjectiveCamera::updateImpl(const Mat4d &world2render)
60{
61 // aperture window: derive from aperture viewport
62 float invAspectRatio = getApertureWindowHeight() / getApertureWindowWidth();
63 mWindow[0] = -1.0f;
64 mWindow[1] = -invAspectRatio;
65 mWindow[2] = 1.0f;
66 mWindow[3] = invAspectRatio;
67
68 // region window: derived from region viewport
69 float offsetX = 2 * getRegionToApertureOffsetX() / getApertureWindowWidth();
70 float offsetY = 2 * getRegionToApertureOffsetY() / getApertureWindowHeight();
71 float widthRatio = 2 * getRegionWindowWidth() / getApertureWindowWidth();
72 float heightRatio = 2 * getRegionWindowHeight() / getApertureWindowHeight();
73 mRenderRegion[0] = -1.0f + offsetX;
74 mRenderRegion[1] = invAspectRatio * (-1.0f + offsetY);
75 mRenderRegion[2] = -1.0f + offsetX + widthRatio;
76 mRenderRegion[3] = invAspectRatio * (-1.0f + offsetY + heightRatio);
77
78 // Compute screen space to raster space (s2r)
79 float zoom[2];
80 zoom[0] = getApertureWindowWidth() * 0.5f;
81 zoom[1] = getApertureWindowHeight() * 0.5f;
82 float offset[2];
83 offset[0] = zoom[0];
84 offset[1] = zoom[1];
85 Mat4f s2r = Mat4f(one);
86 s2r[0][0] = zoom[0];
87 s2r[1][1] = zoom[1];
88 s2r[3][0] = offset[0];
89 s2r[3][1] = offset[1];
90
91 // Compute mC0toS
92 mC0toS = computeC2S(0.0f);
93
94 // Compute mC1toS
95 mC1toS = computeC2S(1.0f);
96
97 // Compute mRtoC0
98 Mat4f c0toR = mC0toS * s2r;
99 mRtoC0 = c0toR.inverse();
100
101 // Compute mRtoC1
102 Mat4f c1toR = mC1toS * s2r;
103 mRtoC1 = c1toR.inverse();
104
105 // Depth-of-field settings
106 // TODO: check focus node attribute
107 const scene_rdl2::rdl2::SceneVariables& vars =
108 getRdlCamera()->getSceneClass().getSceneContext()->getSceneVariables();
109
110 mDof = vars.get(scene_rdl2::rdl2::SceneVariables::sEnableDof) && getRdlCamera()->get(sDofKey);
111
112 if (mDof) {
113 const float focalLength = getFocalDistance();
114 const float fstop = getRdlCamera()->get(sDofApertureKey);
115
116 // One unit in world space == sceneScale meters
117 // Formally:
118 // world space unit * sceneScale = meter
119 // world space unit = meter / sceneScale
120 // world space unit = millimeter / (1000.0f * sceneScale)
121 const float worldToMm = vars.get(scene_rdl2::rdl2::SceneVariables::sSceneScaleKey) * 1000.0f;
122
123 // Convert fstop ratio to entrance pupil diameter in mm (fstop equation):
124 const float entrancePupilDiameter = (focalLength / fstop);
125
126 // Transform entrance pupil diameter to world-space radius:
127 mDofLensRadius = (entrancePupilDiameter / 2.0f) / worldToMm;
128 mDofFocusDistance = getRdlCamera()->get(sDofFocusDistance);
129
130 mLensDistribution.update();
131 } else {
132 mDofLensRadius = 0.0f;
133 mDofFocusDistance = 0.0f;
134 }
135
136}

138Mat4f ProjectiveCamera::computeRaster2Camera(float time) const
139{
140 if (getMotionBlur()) {
141 return lerp(mRtoC0, mRtoC1, time);
142 } else {
143 return mRtoC0;

147Mat4f ProjectiveCamera::computeRaster2Render(float time,
148 const Mat4f& ct2render) const
149{
150 // "raster space at ray time" --> render space
151 if (getMotionBlur()) {
152 // Lerp camera projection.
153 Mat4f rt2ct = lerp(mRtoC0, mRtoC1, time);
154 // Then apply camera to render transform
155 return rt2ct * ct2render;
156 } else {
157 return mRtoC0 * getCamera2Render();
158 }
159}

172void
173ProjectiveCamera::createRayImpl(mcrt_common::RayDifferential *dstRay, float x, float y,
174 float time, float lensU, float lensV) const
175{
176 Vec3f Pr(x, y, -1.0f);
177
178 const bool doDof = getIsDofEnabled();
179
180 if (doDof) {
181 mLensDistribution.sampleLens(lensU, lensV);
182 }
183
184 const float lensX = lensU * mDofLensRadius;
185 const float lensY = lensV * mDofLensRadius;
186
187 if (doDof) {
188 createDOFRay(dstRay, Pr, lensX, lensY, time);
189 } else {
190 createSimpleRay(dstRay, Pr, time);
191 }

193}

195float
196ProjectiveCamera::computeZDistanceImpl(const Vec3f &p, const Vec3f &o, float time) const
197{
198 // This result is typically thought of as the "depth" value used in
199 // depth maps. The ray origin is not needed as the camera is
200 // the origin of all camera spaces. Transform p from render space
201 // to the camera space at time t, and return the negative z component
202 // of that result.
203 const Mat4f render2camera = computeRender2Camera(time);
204 const float cpz = dot(Vec4f(p.x, p.y, p.z, 1.f), render2camera.col2());
205 return -cpz;
206}
