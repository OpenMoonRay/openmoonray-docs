241 template <typename T> T
242 getFaceVaryingAttribute(shading::TypedAttributeKey<T> key, int fid, int vIndex,
243 float time = 0.0f) const
244 {
245 const shading::Attributes* attributes = getAttributes();
246 MNRY_ASSERT(attributes->isSupported(key) &&
247 attributes->getRate(key) == shading::RATE_FACE_VARYING);
248 if (attributes->getTimeSampleCount(key) > 1 && !scene_rdl2::math::isZero(time)) {
249 return attributes->getMotionBlurFaceVarying(key, fid, vIndex, time);
250 } else {
251 return attributes->getFaceVarying(key, fid, vIndex);
252 }
253 }

255 template<typename T> void
256 computeAttributeDerivatives(shading::TypedAttributeKey<T> key,
257 uint32_t fid, uint32_t fvid1, uint32_t fvid2, uint32_t fvid3,
258 uint32_t vid1, uint32_t vid2, uint32_t vid3, float time,
259 const std::array<float, 4>& invA,
260 shading::Intersection& intersection) const
261 {
262 switch (getAttributes()->getRate(key)) {
263 case shading::RATE_VARYING:
264 computeVaryingAttributeDerivatives(key, vid1, vid2, vid3,
265 time, invA, intersection);
266 break;
267 case shading::RATE_FACE_VARYING:
268 computeFaceVaryingAttributeDerivatives(key, fid,
269 fvid1, fvid2, fvid3, time, invA, intersection);
270 break;
271 case shading::RATE_VERTEX:
272 computeVertexAttributeDerivatives(key, vid1, vid2, vid3,
273 time, invA, intersection);
274 break;
275 default:
276 break;
277 }
