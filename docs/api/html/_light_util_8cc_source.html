39void
40computeActiveLights(scene_rdl2::alloc::Arena *arena,
41 const Scene *scene,
42 const shading::Intersection &isect,
43 const scene_rdl2::math::Vec3f *normal,
44 const shading::Bsdf &bsdf,
45 float rayTime,
46 LightSet &lightSet,
47 bool &hasRayTerminatorLights)
48{
49 MNRY_ASSERT(arena);
50
51 int lightSetIdx = isect.getLayerAssignmentId();
52 const LightPtrList *lightList = scene->getLightPtrList(lightSetIdx);
53 const LightFilterLists* lightFilterLists = scene->getLightFilterLists(lightSetIdx);
54
55 // null LightSets and LightFilterLists are supported
56 const Light **activeLights = nullptr;
57 const LightFilterList **activeLightFilterLists = nullptr;
58 // The original light list from "scene" is culled in this function. We must map the
59 // original light list index to the reduced light set index. This is needed for the
60 // LightAccelerator. The LightAccelerator, which is used for intersecting lights,
61 // is constructed with the original light list. The LightSet, which is used for
62 // sampling lights, contains the reduced light list. When we intersect a light in
63 // the LightAccelerator, we want to know which light that corresponds to in the LightSet.
64 // It is possible that we intersect a light in the light accelerator that does not exist
65 // in the LightSet. We map those ids to -1.
66 int* lightIdMap = nullptr;
67
68 const bool hasBssrdf = (bsdf.getBssrdf() != nullptr);
69 const bool hasVolumeSubsurface = (bsdf.getVolumeSubsurface() != nullptr);
70 const float radius = (hasBssrdf ? bsdf.getBssrdf()->getMaxRadius() : 0.0f);
71
72 int activeLightCount = 0;
73 hasRayTerminatorLights = false;
74
75 if (lightList) {
76 size_t upperBound = lightList->size();
77 lightIdMap = arena->allocArray<int>(upperBound);
78 uint8_t * const memBookmark = arena->getPtr();
79 int* activeLightId = arena->allocArray<int>(upperBound);
80
81 const scene_rdl2::math::Vec3f& pos = isect.getP();
82
83 for (size_t i = 0; i < upperBound; ++i) {
84 const Light *light = (*lightList)[i];
85
86 // light culling is done in here
87 // due to the nature of random walk subsurface scattering,
88 // we can not use the position/normal of first hit to cull light.
89 // there may be chance the fist hit can't be lit by the light but
90 // the random walk end point can
91 if (hasVolumeSubsurface ||
92 light->canIlluminate(pos, normal, rayTime, radius, (*lightFilterLists)[i])) {
93 lightIdMap[i] = activeLightCount;
94 activeLightId[activeLightCount++] = i;
95 hasRayTerminatorLights |= light->getIsRayTerminator();
96 } else {
97 lightIdMap[i] = -1;
98 }
99 }
100
101 if (activeLightCount > 0) {
102 activeLights = arena->allocArray<const Light *>(activeLightCount);
103 activeLightFilterLists = arena->allocArray<const LightFilterList *>(activeLightCount);
104
105 for (size_t i = 0; i < activeLightCount; ++i) {
106 activeLights[i] = (*lightList)[activeLightId[i]];
107 activeLightFilterLists[i] = (*lightFilterLists)[activeLightId[i]];
108 }
109 } else {
110 // Reset arena pointer if there are no active lights. If there are active lights, we
111 // must keep the arena pointer where is it because we need to keep activeLights and
112 // activeLightFilterLists in the arena. This means that even though we don't use activeLighId
113 // after this point, it is still hanging around in the arena. This is suboptimal.
114
115 // The pointer is reset to the beginning of activeLightId,
116 // because lightIdMap is always filled and used, even when there are no active lights.
117 arena->setPtr(memBookmark);
118 }
119 }
120
121 // The LightSet does NOT take ownership of the array, however this is not
122 // a memory leak because the array is owned by the memory arena.
123 lightSet.init(activeLights, activeLightCount, activeLightFilterLists);
124
125 // Augment light set with Embree-based acceleration structure
126 const LightAccelerator *acc = scene->getLightAccelerator(lightSetIdx);
127 lightSet.setAccelerator(acc, lightIdMap);
128}

130bool chooseThisLight(const IntegratorSample1D &samples, int depth, unsigned int numHits)
131{
132 // Choose the first light we hit
133 bool chooseThisOne = true;
134
135 // If we're hitting a second or subsequent light,
136 // choose randomly from among those hit
137 if (numHits > 1) {
138 float randVar;
139 samples.getSample(&randVar, depth);
140 chooseThisOne = (randVar * (float)numHits < 1.0f);
141 }
142
143 return chooseThisOne;
144}
