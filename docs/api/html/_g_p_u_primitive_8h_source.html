17// The host-side representation of the geometric primitives in the scene.
18// Typically this is a few parameters and a set of buffers that live on the
19// GPU (as it is up to the host to manage the GPU memory.)
20
21class GPUPrimitive

23public:
24 virtual ~GPUPrimitive() {}

79// Custom primitives for non-trimesh geometry. These have custom intersection
80// programs on the GPU. You cannot mix trimeshes and custom primitives in the
81// same traversable (BVH) so they are treated separately throughout the GPU code.
82
83class GPUCustomPrimitive : public GPUPrimitive
84{
85public:
86 virtual ~GPUCustomPrimitive() {}
87
88 // Optix needs the Aabb(s) to put the custom primitive into the BVH.
89 // If this is something simple like a sphere, there are no sub-primitives and
90 // aabbs.size() == 1. If it is a set of points, there is a separate Aabb
91 // for each point as each sub-primitive may land in a different BVH node.
92 virtual void getPrimitiveAabbs(std::vector<OptixAabb>* aabbs) const = 0;
93
94 // Custom primitives hold onto the host-side geometry for a while so the
95 // getPrimitiveAabbs() function can compute aabbs. After the BVH is built
96 // we can free this memory.
97 virtual void freeHostMemory() {}
98};
