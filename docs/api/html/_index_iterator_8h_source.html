93// PointerToConst will take a pointer and make it point to a const value.
94// E.g. "const T*" will become "const T*" (no-op)
95// "T*" will become "const T*"
96// Any type that's not a pointer will be a no-op.
97
98template <typename T>
99struct PointerToConst
100{
101 typedef T type;
102};
103
104template <typename T>
105struct PointerToConst<T*>
106{
107 // It's not trivial to add const to the value of pointer. "add_const" will
108 // only add a top-level const. E.g. "T*" will become "T* const", not
109 // "const T*". We have to deconstruct the type, and built it back to a
110 // pointer.
111 typedef typename std::add_pointer<
112 typename std::add_const<
113 typename std::remove_pointer<T*>::type
114 >::type>::type type;
115};

118/// @class FilterIndexIterator
119/// @brief Like IndexIterator, it is used to iterate through indices for a
120/// container. However, it will only return indices where the passed in match
121/// equals (operator==) the value in the container. Unlike IndexIterator, it
122/// uses a set of index iterators (e.g. IndexIterator) for its range
123/// deliminators. This is to easily iterate over a sub-range (e.g. that given
124/// by IndexableArray::equal_range).
125
126// We only provide const versions of this iterator, since it just accesses
127// underlying array indices. We don't want the user to be able to change those.
128template <typename ListType, typename EnumerableIterator>
129class FilterIndexIterator
130{
131public:
132 // The PointerToConst allows us to have lists of non-const pointers, but to
133 // use const pointers are our match type.
134 using filter_type = typename detail::PointerToConst<typename ListType::value_type>::type;
135 using enumerable_iterator = EnumerableIterator;
136 using value_type = typename std::iterator_traits<enumerable_iterator>::value_type;
137 using difference_type = typename std::iterator_traits<enumerable_iterator>::difference_type;
138 using pointer = typename std::iterator_traits<enumerable_iterator>::pointer;
139 using reference = typename std::iterator_traits<enumerable_iterator>::reference;
140 using iterator_category = std::bidirectional_iterator_tag;
141
142 // List is copied by value. If you don't want to pay for the copy, pass in
143 // a std::cref.
144 FilterIndexIterator(enumerable_iterator current,
145 enumerable_iterator first,
146 enumerable_iterator last,
147 ListType list,
148 filter_type match) :
149 mIter(current),
150 mFirst(first),
151 mLast(last),
152 mList(list),
153 mMatch(match)
154 {
155 // Find our first matching value. operator++ will do this for us.
156 if (mIter != mLast && mList[*mIter] != mMatch) {
157 ++(*this);
158 }
159 }
160
161 value_type operator*() const
162 {
163 return *mIter;
164 }
165
166 // Prefix decrement operator
167 FilterIndexIterator& operator--()
168 {
169 MNRY_ASSERT(mIter != mFirst);
170 do {
171 --mIter;
172 } while (mIter != mFirst && mList[*mIter] != mMatch);
173 return *this;
174 }
175
176 // Postfix decrement operator
177 FilterIndexIterator operator--(int)
178 {
179 const FilterIndexIterator old(*this);
180 --(*this);
181 return old;
182 }
183
184 // Prefix increment operator
185 FilterIndexIterator& operator++()
186 {
187 MNRY_ASSERT(mIter != mLast);
188 do {
189 ++mIter;
190 } while (mIter != mLast && mList[*mIter] != mMatch);
191 return *this;
192 }
193
194 // Postfix increment operator
195 FilterIndexIterator operator++(int)
196 {
197 const FilterIndexIterator old(*this);
198 ++(*this);
199 return old;
200 }
201
202 friend bool operator==(const FilterIndexIterator& a, const FilterIndexIterator& b)
203 {
204 MNRY_ASSERT(a.mMatch == b.mMatch, "This is probably due to the object"
205 "being passed into to 'begin' and 'end' being different.");
206 return a.mIter == b.mIter;
207 }
208
209 friend bool operator!=(const FilterIndexIterator& a, const FilterIndexIterator& b)
210 {
211 return !(a == b);
212 }
213
214private:
215 enumerable_iterator mIter;
216 enumerable_iterator mFirst;
217 enumerable_iterator mLast;
218 ListType mList;
219 filter_type mMatch;
220};
