14Fresnel *
15createFresnel(scene_rdl2::alloc::Arena* arena,
16 const Fresnelv *fresnelv,
17 int lane)
18{
19 Fresnel *fresnel = nullptr;
20
21 if (fresnelv && ((1 << lane) & fresnelv->mMask)) {
22
23 switch (fresnelv->mType) {
24
25 case ispc::FRESNEL_TYPE_NONE:
26 break;
27
28 case ispc::FRESNEL_TYPE_SCHLICK_FRESNEL:
29 fresnel = arena->allocWithArgs<SchlickFresnel>(arena, (const SchlickFresnelv *)fresnelv, lane);
30 break;
31
32 case ispc::FRESNEL_TYPE_ONE_MINUS_ROUGH_SCHLICK_FRESNEL:
33 fresnel = arena->allocWithArgs<OneMinusRoughSchlickFresnel>(arena, (const OneMinusRoughSchlickFresnelv *)fresnelv, lane);
34 break;
35
36 case ispc::FRESNEL_TYPE_ONE_MINUS_FRESNEL:
37 fresnel = arena->allocWithArgs<OneMinusFresnel>(arena, (const OneMinusFresnelv *)fresnelv, lane);
38 break;
39
40 case ispc::FRESNEL_TYPE_ONE_MINUS_VELVET_FRESNEL:
41 fresnel = arena->allocWithArgs<OneMinusVelvetFresnel>(arena, (const OneMinusVelvetFresnelv *)fresnelv, lane);
42 break;
43
44 case ispc::FRESNEL_TYPE_ONE_MINUS_ROUGH_FRESNEL:
45 fresnel = arena->allocWithArgs<OneMinusRoughFresnel>(arena, (const OneMinusRoughFresnelv *)fresnelv, lane);
46 break;
47
48 case ispc::FRESNEL_TYPE_CONDUCTOR_FRESNEL:
49 fresnel = arena->allocWithArgs<ConductorFresnel>(arena, (const ConductorFresnelv *)fresnelv, lane);
50 break;
51
52 case ispc::FRESNEL_TYPE_DIELECTRIC_FRESNEL:
53 fresnel = arena->allocWithArgs<DielectricFresnel>(arena, (const DielectricFresnelv *)fresnelv, lane);
54 break;
55
56 case ispc::FRESNEL_TYPE_MULTIPLE_TRANSMISSION_FRESNEL:
57 fresnel = arena->allocWithArgs<MultipleTransmissionFresnel>(arena, (const MultipleTransmissionFresnelv *)fresnelv, lane);
58 break;
59
60 default:
61 MNRY_ASSERT(0);
62 }
63 }
64 return fresnel;
65}
