152 template <typename F>
153 void activeTileCrawler(const F tileFunc) const
154 {
155 for (unsigned tileId = 0; tileId < mActivePixels.getNumTiles(); ++tileId) {
156 uint64_t tileMask = mActivePixels.getTileMask(tileId);
157 if (tileMask) {
158 int pixOffset = tileId << 6;
159 tileFunc(tileMask, pixOffset);
160 }
161 }
162 }
163
164 template <typename T, typename F>
165 void activePixelCrawler(uint64_t tileMask, const T *firstDataOfTile, F pixFunc) const
166 {
167 for (unsigned y = 0; y < 8; ++y) {
168 unsigned offset = (y << 3); // y * 8
169
170 uint64_t currTileMask = tileMask >> offset;
171 if (!currTileMask) break; // early exit : rest of them are all empty
172
173 const T *currPix = firstDataOfTile + offset;
174
175 uint64_t currTileScanlineMask =
176 currTileMask & static_cast<uint64_t>(0xff); // get one scanline mask
177 for (unsigned x = 0; x < 8; ++x) {
178 if (!currTileScanlineMask) break; // early exit for scanline
179
180 if (currTileScanlineMask & static_cast<uint64_t>(0x1)) {
181 pixFunc(*currPix);
182 }
183 currPix ++;
184 currTileScanlineMask >>= 1;
185 }
186 }
187 }

189 template <typename F>
190 void
191 partialMergeTilesTblCrawler(const PartialMergeTilesTbl &partialMergeTilesTbl, F resetTileFunc) const
192 {
193 for (size_t tileId = 0; tileId < partialMergeTilesTbl.size(); ++tileId) {
194 if (partialMergeTilesTbl[tileId]) {
195 uint pixOffset = static_cast<uint>(tileId << 6);
196 resetTileFunc(pixOffset);
197 }
198 }
199 }

242finline void
243FbAov::resetNumSampleBufferTiled(const PartialMergeTilesTbl *partialMergeTilesTbl)
244{
245 if (!partialMergeTilesTbl) {
246 mNumSampleBufferTiled.clear();
247 } else {
248 partialMergeTilesTblCrawler
249 (*partialMergeTilesTbl,
250 [&](unsigned pixOffset) {
251 bufferTileClear(mNumSampleBufferTiled.getData() + pixOffset);
252 });
253 }
254}

256finline void
257FbAov::resetBufferTiled(const PartialMergeTilesTbl *partialMergeTilesTbl)
258{
259 if (!partialMergeTilesTbl) {
260 mBufferTiled.clear();
261 } else {
262 switch (mBufferTiled.getFormat()) {
263 case VariablePixelBuffer::FLOAT :
264 partialMergeTilesTblCrawler
265 (*partialMergeTilesTbl,
266 [&](unsigned pixOffset) {
267 bufferTileClear(mBufferTiled.getFloatBuffer().getData() + pixOffset);
268 });
269 break;
270 case VariablePixelBuffer::FLOAT2 :
271 partialMergeTilesTblCrawler
272 (*partialMergeTilesTbl,
273 [&](unsigned pixOffset) {
274 bufferTileClear(mBufferTiled.getFloat2Buffer().getData() + pixOffset);
275 });
276 break;
277 case VariablePixelBuffer::FLOAT3 :
278 partialMergeTilesTblCrawler
279 (*partialMergeTilesTbl,
280 [&](unsigned pixOffset) {
281 bufferTileClear(mBufferTiled.getFloat3Buffer().getData() + pixOffset);
282 });
283 break;
284 case VariablePixelBuffer::FLOAT4 :
285 partialMergeTilesTblCrawler
286 (*partialMergeTilesTbl,
287 [&](unsigned pixOffset) {
288 bufferTileClear(mBufferTiled.getFloat4Buffer().getData() + pixOffset);
289 });
290 break;
291 default :
292 break;
293 }
294 }
