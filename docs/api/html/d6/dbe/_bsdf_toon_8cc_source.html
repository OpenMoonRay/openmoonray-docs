19// Rodrigues' rotation formula, assume axis is normalized
20scene_rdl2::math::Vec3f
21rotateVector(const scene_rdl2::math::Vec3f& v, const scene_rdl2::math::Vec3f& axis, float theta) {
22 float ct, st;
23 scene_rdl2::math::sincos(theta, &st, &ct);
24 return ct * v + st * scene_rdl2::math::cross(axis, v) + scene_rdl2::math::dot(axis, v) * (1.f - ct) * axis;
25}

217Color
218ToonSpecularBsdfLobe::eval(const BsdfSlice &slice,
219 const Vec3f &wi,
220 float *pdf) const
221{
222
223 Vec3f N = mFrame.getN();
224 const Vec3f& wo = slice.getWo();
225 Vec3f R = wi - 2.0f * scene_rdl2::math::dot(wi, N) * N;
226
227 // Rotate N to "stretch" the specular highlight
228 const float dot_u_l = scene_rdl2::math::dot(R, mdPds);
229 const float dot_u_c = scene_rdl2::math::dot(wo, mdPds);
230 const float dot_u = dot_u_l + dot_u_c;
231 const float rot_u = scene_rdl2::math::clamp(mStretchU * dot_u, -0.5f, 0.5f);
232 N = rotateVector(N, mdPdt, rot_u);
233
234 const float dot_v_l = scene_rdl2::math::dot(R, mdPdt);
235 const float dot_v_c = scene_rdl2::math::dot(wo, mdPdt);
236 const float dot_v = dot_v_l + dot_v_c;
237 const float rot_v = scene_rdl2::math::clamp(-mStretchV * dot_v, -0.5f, 0.5f);
238 N = rotateVector(N, mdPds, rot_v);
239
240 const float cosNO = dot(N, wo);
241 const float cosNI = dot(N, wi);
242
243 // Prepare for early exit
244 if (pdf != NULL) {
245 *pdf = 0.0f;
246 }
247
248 if (cosNO <= 0.0f || cosNI <= 0.0f) {
249 return sBlack;
250 }
251
252 // Experimental code from PIB2 hair toon slides
253 // https://docs.google.com/presentation/d/10lYRt3FpQEOu8vrCoGdc_ngJnRXkzFi2B69cEUFkM48/edit
254 // Uses acos to get a more linear response for easier ramp remapping
255 R = wi - 2.0f * scene_rdl2::math::dot(wi, N) * N;
256 float specAngle = scene_rdl2::math::max(0.0f, dot(-wo, R));
257 specAngle = 1.0f - 2.0f * (scene_rdl2::math::acos(scene_rdl2::math::clamp(specAngle, -1.0f, 1.0f)) / scene_rdl2::math::sPi);
258 const float exponent = (1.0f - mRoughness) * 100.0f;
259 specAngle = scene_rdl2::math::pow(specAngle, exponent);
260
261 if (specAngle <= 0.0f) {
262 return sBlack;
263 }
264
265 if (pdf != NULL) {
266 *pdf = 0.5f * scene_rdl2::math::sOneOverPi * specAngle;
267 }
268
269 const float ramp = mRampControl.eval1D(specAngle);
270 return mTint * ramp * mIntensity * scene_rdl2::math::sOneOverPi;
271}

319scene_rdl2::math::Color
320ToonBsdfLobe::eval(const BsdfSlice &slice,
321 const Vec3f &wi,
322 float *pdf) const
323{
324 Vec3f N = mFrame.getN();
325 if (mExtendRamp && dot(N, wi) < 0) {
326 // Make shading normal perpendicular to light direction
327 // and nudge towards light with lerp
328 N = normalize(lerp(cross(wi, cross(N, wi)),
329 wi,
330 sEpsilon)); // Must be > 1e-9
331 }
332 float cosThetaWi = max(dot(N, wi), 0.0f);
333
334 if (pdf != NULL) {
335 *pdf = cosThetaWi * sOneOverPi;
336 }
337
338 // Note: we assume this lobe has been setup with a OneMinus*Fresnel
339 // as we want to use 1 - specular_fresnel. Also notice we use
340 // cosThetaWo to evaluate the fresnel term, as an approximation of what
341 // hDotWi would be for the specular lobe.
342 float cosThetaWo = 1.0f;
343
344 if (getFresnel()) {
345 cosThetaWo = max(dot(N, slice.getWo()), 0.0f);
346 }
347
348 Color ramp = mRampControl.eval1D(1.0f - cosThetaWi);
349
350 const Color albedo = clamp(ramp*mAlbedo, sBlack, sWhite);
351 // Soften hard shadow terminator due to shading normals
352 const float Gs = slice.computeShadowTerminatorFix(N, wi);
353
354 // We use the upper bound of 1/2pi as the normalization factor
355 // for the maximum possible toon step function
356 // TODO: a more accurate normalization factor for the varying
357 // step can be computed if so desired.
358 Color result = Gs * albedo *
359 computeScaleAndFresnel(cosThetaWo) * sOneOverTwoPi;
360 return result;
361}
