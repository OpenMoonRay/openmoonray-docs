22Color
23UnderClearcoatTransmissionBsdfLobe::eval(const BsdfSlice &slice,
24 const Vec3f &wi,
25 float *pdf) const
26{
27 if (pdf != nullptr)
28 *pdf = 0.0f;
29
30 Color result;
31 Vec3f woPrime;
32 float cosThetaWo, cosThetaWoPrime;
33 if (!computeRefractionDirection(mN,
34 slice.getWo(),
35 mNeta,
36 woPrime,
37 cosThetaWo,
38 cosThetaWoPrime))
39 {
40 // Total Outward Reflection
41 return scene_rdl2::math::sBlack;
42 }
43 // Reverse refraction direction
44 woPrime = -woPrime;
45 BsdfSlice underSlice(slice.getNg(),
46 woPrime,
47 slice.getIncludeCosineTerm(),
48 slice.getEntering(),
49 slice.getShadowTerminatorFix(),
50 slice.getFlags());
51
52 result = mUnder->eval(underSlice, wi, pdf);
53
54 result *= computeScaleAndFresnel(dot(mN, slice.getWo()));
55
56 // Clearcoat Absorption
57 const Color ct = computeTransmission(cosThetaWoPrime,
58 0.0f);
59
60 result = result * ct;
61
62 return result;
63}
