29class PrimitiveTypeChecker : public PrimitiveVisitor
30{
31public:
32 PrimitiveTypeChecker(const Primitive& p): mType(PrimitiveType::UNKNOWN) {
33 // we know we are not going to modify the primitive here
34 const_cast<Primitive&>(p).accept(*this);
35 }
36
37 PrimitiveType getType() const { return mType; }
38
39 virtual void visitPrimitive(Primitive& p) override {
40 MNRY_ASSERT_REQUIRE(false);
41 }
42
43 virtual void visitCurves(Curves& c) override {
44 mType = PrimitiveType::CURVES;
45 }
46
47 virtual void visitInstance(Instance& i) override {
48 mType = PrimitiveType::INSTANCE;
49 }
50
51 virtual void visitPoints(Points& p) override {
52 mType = PrimitiveType::POINTS;
53 }
54
55 virtual void visitPolygonMesh(PolygonMesh& p) override {
56 mType = PrimitiveType::POLYGON_MESH;
57 }
58
59 virtual void visitPrimitiveGroup(PrimitiveGroup& pg) override {
60 mType = PrimitiveType::PRIMITIVE_GROUP;
61 }
62
63 virtual void visitSphere(Sphere& s) override {
64 mType = PrimitiveType::SPHERE;
65 }
66
67 virtual void visitBox(Box& b) override {
68 mType = PrimitiveType::BOX;
69 }
70
71 virtual void visitSubdivisionMesh(SubdivisionMesh& s) override {
72 mType = PrimitiveType::SUBDIVISION_MESH;
73 }
74
75 virtual void visitTransformedPrimitive(TransformedPrimitive& t) override {
76 mType = PrimitiveType::TRANSFORMED_PRIMITIVE;

80 mType = PrimitiveType::VDB_VOLUME;

93bool
94validateVertexBuffer3fa(const VertexBuffer<Vec3fa, InterleavedTraits>& vertices,
95 const scene_rdl2::rdl2::Geometry* rdlGeometry)
96{
97 bool isValid = true;
98 for (size_t v = 0; v < vertices.size(); ++v) {
99 for (size_t t = 0; t < vertices.get_time_steps(); ++t) {
100 if (std::isnan(vertices(v, t).x) ||
101 std::isnan(vertices(v, t).y) ||
102 std::isnan(vertices(v, t).z) ||
103 std::isnan(vertices(v, t).w)) {
104 isValid = false;
105 rdlGeometry->error("Vertex data has a nan value at index (", v, ", ", t, ")");
106 } else if (!std::isfinite(vertices(v, t).x) ||
107 !std::isfinite(vertices(v, t).y) ||
108 !std::isfinite(vertices(v, t).z) ||
109 !std::isfinite(vertices(v, t).w)) {
110 isValid = false;
111 rdlGeometry->error("Vertex data has a infinite value at index (", v, ", ", t, ")");
112 }
113 }
114 }
115 return isValid;
116}

141template <typename T> bool
142validateAttr(const T& attr,
143 const std::string& attrName,
144 const scene_rdl2::rdl2::Geometry* rdlGeometry) {
145 if (std::isnan(attr)) {
146 rdlGeometry->error("Primitive attribute ", attrName," has a nan value");
147 return false;
148 } else if (!std::isfinite(attr)) {
149 rdlGeometry->error("Primitive attribute ", attrName," has a infinite value");
150 return false;
151 } else {
152 return true;
153 }
154}

291template <typename T> bool
292validateAttrs(const shading::PrimitiveAttributeTable& pat,
293 const shading::AttributeKey& key,
294 const float timeSample,
295 const scene_rdl2::rdl2::Geometry* rdlGeometry) {
296 const shading::PrimitiveAttribute<T> &attr =
297 pat.getAttribute(shading::TypedAttributeKey<T>(key), timeSample);
298 for (size_t i = 0; i < attr.size(); ++i) {
299 if (!validateAttr<T>(attr[i], key.getName(), rdlGeometry)) {
300 return false;
301 }
302 }
303 return true;
304}

306bool
307validatePrimitiveAttributeTable(const shading::PrimitiveAttributeTable& pat,
308 const scene_rdl2::rdl2::Geometry* rdlGeometry)
309{
310 bool isValid = true;
311 for (const auto& p : pat) {
312
313 const shading::AttributeKey& key = p.first;
314 const size_t motionSamplesCount = pat.getTimeSampleCount(key);
315
316 for (size_t t = 0; t < motionSamplesCount; ++t) {
317
318 switch (key.getType()) {
319 case scene_rdl2::rdl2::TYPE_BOOL:
320 validateAttrs<scene_rdl2::rdl2::Bool>(pat, key, t, rdlGeometry);
321 break;
322 case scene_rdl2::rdl2::TYPE_INT:
323 validateAttrs<scene_rdl2::rdl2::Int>(pat, key, t, rdlGeometry);
324 break;
325 case scene_rdl2::rdl2::TYPE_LONG:
326 validateAttrs<scene_rdl2::rdl2::Long>(pat, key, t, rdlGeometry);
327 break;
328 case scene_rdl2::rdl2::TYPE_FLOAT:
329 validateAttrs<scene_rdl2::rdl2::Float>(pat, key, t, rdlGeometry);
330 break;
331 case scene_rdl2::rdl2::TYPE_DOUBLE:
332 validateAttrs<scene_rdl2::rdl2::Double>(pat, key, t, rdlGeometry);
333 break;
334 case scene_rdl2::rdl2::TYPE_VEC2F:
335 validateAttrs<scene_rdl2::rdl2::Vec2f>(pat, key, t, rdlGeometry);
336 break;
337 case scene_rdl2::rdl2::TYPE_VEC3F:
338 validateAttrs<scene_rdl2::rdl2::Vec3f>(pat, key, t, rdlGeometry);
339 break;
340 case scene_rdl2::rdl2::TYPE_VEC4F:
341 validateAttrs<scene_rdl2::rdl2::Vec4f>(pat, key, t, rdlGeometry);
342 break;
343 case scene_rdl2::rdl2::TYPE_VEC2D:
344 validateAttrs<scene_rdl2::rdl2::Vec2d>(pat, key, t, rdlGeometry);
345 break;
346 case scene_rdl2::rdl2::TYPE_VEC3D:
347 validateAttrs<scene_rdl2::rdl2::Vec3d>(pat, key, t, rdlGeometry);
348 break;
349 case scene_rdl2::rdl2::TYPE_VEC4D:
350 validateAttrs<scene_rdl2::rdl2::Vec4d>(pat, key, t, rdlGeometry);
351 break;
352 case scene_rdl2::rdl2::TYPE_RGB:
353 validateAttrs<scene_rdl2::rdl2::Rgb>(pat, key, t, rdlGeometry);
354 break;
355 case scene_rdl2::rdl2::TYPE_RGBA:
356 validateAttrs<scene_rdl2::rdl2::Rgba>(pat, key, t, rdlGeometry);
357 break;
358 case scene_rdl2::rdl2::TYPE_MAT4F:
359 validateAttrs<scene_rdl2::rdl2::Mat4f>(pat, key, t, rdlGeometry);
360 break;
361 case scene_rdl2::rdl2::TYPE_MAT4D:
362 validateAttrs<scene_rdl2::rdl2::Mat4d>(pat, key, t, rdlGeometry);
363 break;
364 }
365 }
366 }
367
368 return isValid;
369}

371std::unique_ptr<Curves>
372createCurves(Curves::Type type,
373 Curves::SubType subtype,
374 int tessellationRate,
375 Curves::CurvesVertexCount&& curvesVertexCount,
376 Curves::VertexBuffer&& vertices,
377 LayerAssignmentId&& layerAssignmentId,
378 shading::PrimitiveAttributeTable&& primitiveAttributeTable,
379 bool validateGeometry,
380 const scene_rdl2::rdl2::Geometry* rdlGeometry)
381{
382 if (type == Curves::Type::UNKNOWN || subtype == Curves::SubType::UNKNOWN) {
383 return nullptr;
384 }
385
386 if (validateGeometry) {
387 if (!validateVertexBuffer3fa(vertices, rdlGeometry)) {
388 return nullptr;
389 }
390 if (!validatePrimitiveAttributeTable(primitiveAttributeTable, rdlGeometry)) {
391 return nullptr;
392 }
393 }
394
395 return fauxstd::make_unique<Curves>(type,
396 subtype,
397 tessellationRate,
398 std::move(curvesVertexCount),
399 std::move(vertices),
400 std::move(layerAssignmentId),
401 std::move(primitiveAttributeTable));
402}

563static void
564removeUnassignedFaces(const scene_rdl2::rdl2::Layer* layer,
565 int constAssignmentId,
566 std::vector<Primitive::IndexType>& faceToPart,
567 std::vector<Primitive::size_type>& faceVertexCount,
568 std::vector<Primitive::IndexType>& indices,
569 shading::PrimitiveAttributeTable* primitiveAttributeTable)
570{
571 bool hasValidAssignment = constAssignmentId != -1 &&
572 (layer->lookupMaterial(constAssignmentId) != nullptr ||
573 layer->lookupVolumeShader(constAssignmentId) != nullptr);
574 // the whole mesh is a part and it's not assigned
575 if (!hasValidAssignment) {
576 faceToPart.resize(0);
577 faceVertexCount.resize(0);
578 indices.resize(0);
579 if (primitiveAttributeTable) {
580 std::vector<shading::AttributeKey> unusedAttributes;
581 for (auto& kv : *primitiveAttributeTable) {
582 shading::AttributeRate rate =
583 primitiveAttributeTable->getRate(kv.first);
584 if (rate == shading::RATE_UNIFORM || rate == shading::RATE_FACE_VARYING) {
585 unusedAttributes.push_back(kv.first);
586 }
587 }
588 for (auto& key : unusedAttributes) {
589 primitiveAttributeTable->erase(key);
590 }
591 }
592 }
593}

595static void
596removeUnassignedFaces(const scene_rdl2::rdl2::Layer* layer,
597 std::vector<int>& varyingAssignmentId,
598 std::vector<Primitive::IndexType>& faceToPart,
599 std::vector<Primitive::size_type>& faceVertexCount,
600 std::vector<Primitive::IndexType>& indices,
601 shading::PrimitiveAttributeTable* primitiveAttributeTable)
602{
603 size_t faceCount = faceVertexCount.size();
604 MNRY_ASSERT(varyingAssignmentId.size() == faceCount);
605 size_t totalFaceVertices = std::accumulate(
606 faceVertexCount.begin(), faceVertexCount.end(), 0);
607 MNRY_ASSERT(indices.size() == totalFaceVertices);
608
609 std::vector<int> newFaceIndexes(faceCount, -1);
610 std::vector<int> newFaceVertexIndexes(totalFaceVertices, -1);
611 // calculate index mapping from input face list to shrinked face list
612 // and mapping from input face vertex list to shrinked face vertex list
613 size_t currentFaceVertex = 0;
614 size_t assignedFaceCount = 0;
615 size_t assignedFaceVertices = 0;
616 for (size_t i = 0; i < faceCount; ++i) {
617 bool hasValidAssignment = varyingAssignmentId[i] != -1 &&
618 (layer->lookupMaterial(varyingAssignmentId[i]) != nullptr ||
619 layer->lookupVolumeShader(varyingAssignmentId[i]) != nullptr);
620 if (hasValidAssignment) {
621 newFaceIndexes[i] = assignedFaceCount++;
622 for (size_t j = 0; j < faceVertexCount[i]; ++j) {
623 newFaceVertexIndexes[currentFaceVertex + j] =
624 assignedFaceVertices++;
625 }
626 }
627 currentFaceVertex += faceVertexCount[i];
628 }
629 // remove the unassigned face
630 for (size_t i = 0; i < faceCount; ++i) {
631 if (newFaceIndexes[i] != -1) {
632 varyingAssignmentId[newFaceIndexes[i]] = varyingAssignmentId[i];
633 if (faceToPart.size() > 0) {
634 faceToPart[newFaceIndexes[i]] = faceToPart[i];
635 }
636 faceVertexCount[newFaceIndexes[i]] = faceVertexCount[i];
637 }
638 }
639 if (faceToPart.size() > 0) {
640 faceToPart.resize(assignedFaceCount);
641 }
642 varyingAssignmentId.resize(assignedFaceCount);
643 faceVertexCount.resize(assignedFaceCount);
644 // remove the unassigned face vertex
645 for (size_t i = 0; i < totalFaceVertices; ++i) {
646 if (newFaceVertexIndexes[i] != -1) {
647 indices[newFaceVertexIndexes[i]] = indices[i];
648 }
649 }
650 indices.resize(assignedFaceVertices);
651 // we are done if there is no primitive attributes to process
652 if (!primitiveAttributeTable) {
653 return;
654 }
655 // collect the uniform/facevarying attributes that need to process
656 std::vector<shading::AttributeKey> uniformKeys;
657 std::vector<shading::AttributeKey> faceVaryingKeys;
658 for (auto& kv : *primitiveAttributeTable) {
659 shading::AttributeRate rate =
660 primitiveAttributeTable->getRate(kv.first);
661 if (rate == shading::RATE_UNIFORM) {
662 uniformKeys.push_back(kv.first);
663 } else if (rate == shading::RATE_FACE_VARYING) {
664 faceVaryingKeys.push_back(kv.first);
665 }
666 }
667 // shrink the uniform primitive attributes
668 for (const auto& key : uniformKeys) {
669 auto& attribute = primitiveAttributeTable->find(key)->second;
670 for (size_t t = 0; t < attribute.size(); ++t) {
671 // MOONRAY-3803: Same fix as below, the size may be greater than
672 // the faceCount (it will be less than facevarying count)
673 MNRY_ASSERT(attribute[t]->size() >= faceCount);
674 for (size_t f = 0; f < faceCount; ++f) {
675 if (newFaceIndexes[f] != -1) {
676 attribute[t]->copyInPlace(f, newFaceIndexes[f]);
677 }
678 }
679 attribute[t]->resize(assignedFaceCount);
680 }
681 }
682 // shrink the facevarying primitive attributes
683 for (const auto& key : faceVaryingKeys) {
684 auto& attribute = primitiveAttributeTable->find(key)->second;
685 for (size_t t = 0; t < attribute.size(); ++t) {
686 // MOONRAY-3803: if the primvar size is larger than needed, the
687 // face-varying interpolation is selected, but the array still stays the
688 // larger size. Do not use the size(), stop at the expected count.
689 MNRY_ASSERT(attribute[t]->size() >= totalFaceVertices);
690 for (size_t fv = 0; fv < totalFaceVertices; ++fv) {
691 if (newFaceVertexIndexes[fv] != -1) {
692 attribute[t]->copyInPlace(fv, newFaceVertexIndexes[fv]);
693 }
694 }
695 attribute[t]->resize(assignedFaceVertices);
696 }
697 }
698}
