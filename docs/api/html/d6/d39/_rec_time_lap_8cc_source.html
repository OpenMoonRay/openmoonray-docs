46void
47RecTimeLap::showLastInfo(const float referenceFps, void (*msgOutFunc)(const std::string &msg)) const
48{
49 float intervalSec = mWhole.getTimeAverage();
50 float fps = 1.0f / intervalSec;
51 float miSecInterval = intervalSec * 1000.0f; // milli sec
52
53 uint64_t tickInterval = mWhole.getTickAverage();
54 float tickMiSec = miSecInterval / (float)tickInterval;
55 float allMiSec = (referenceFps > 0.0f) ? (1.0f / referenceFps * 1000.0f) : 0.0f;
56
57 std::ostringstream ostr;
58 ostr << "showLastInfo " << mName << " {\n"
59 << " <tickMiSec>" << _F30_25 << tickMiSec << " ms\n"
60 << " <interval>" << _TIC(tickInterval) << _MSEC(miSecInterval) << _FPS(fps) << '\n';
61 for (size_t i = 0; i < mSections.size(); ++i) {
62 const rec_time::RecTickManualInterval &cSection = mSections[i];
63 if (cSection.isReset()) {
64 ostr << cSection.getName() << '\n';
65 } else {
66 uint64_t cTick = cSection.getLast();
67 float cMiSec = (float)cTick * tickMiSec;
68 float cPct = (referenceFps > 0.0f)? (cMiSec / allMiSec * 100.0f): 0.0f;
69 ostr << cSection.getName() << ":" << _TIC(cTick) << _MSEC(cMiSec) << _PCT(cPct) << '\n';
70 }
71 }
72
73 if (mAuxSections.size() > 0) {
74 for (size_t i = 0; i < mAuxSections.size(); ++i) {
75 const rec_time::RecDoubleManualInterval &cSection = mAuxSections[i];
76 if (cSection.isReset()) {
77 ostr << cSection.getName() << '\n';
78 } else {
79 float cMiSec = (float)(cSection.getLast() * 1000.0);
80 float cPct = (referenceFps > 0.0f)? (cMiSec / allMiSec * 100.0f): 0.0f;
81 ostr << cSection.getName() << ":" << _MSEC(cMiSec) << _PCT(cPct) << '\n';
82 }
83 }
84 }
85
86 if (mAuxUInt64Sections.size() > 0) {
87 for (size_t i = 0; i < mAuxUInt64Sections.size(); ++i) {
88 const rec_time::RecUInt64ManualInterval &cSection = mAuxUInt64Sections[i];
89 if (cSection.isReset()) {
90 ostr << cSection.getName() << '\n';
91 } else {
92 uint64_t cVal = cSection.getLast();
93 ostr << cSection.getName() << ":" << _UINT(cVal) << '\n';
94 }

171bool
172RecTimeLap::calcNextShowIntervalSec()
173{
174 if (mLastInterval) {
175 return false;
176 }
177
178 // 5% 10% 20% 40% 80% 100%
179 mNextShowIntervalSec *= 2.0f;
180 if (mMessageIntervalSec < mNextShowIntervalSec) {
181 mNextShowIntervalSec = mMessageIntervalSec;
182 mLastInterval = true;
183 }
184 return true;
185}

187bool
188RecTimeLap::saveFile(const std::string &str) const
189{
190 if (mFileDumpId == 0xffff) {
191 return true; // skip output
192 }
193
194 float ratio = calcRatio();
195 int ratioI = (int)ratio;
196
197 std::ostringstream ostr;
198 ostr << "./recTimeLap_" << std::setw(2) << std::setfill('0') << mFileDumpId << "_" << std::setw(3) << ratioI << ".log";
199
200 std::ofstream oFile(ostr.str());
201 if (!oFile) return false;
202
203 char hostname[HOST_NAME_MAX];
204 ::gethostname(hostname, HOST_NAME_MAX);
205 oFile << "\nhostname:" << hostname << '\n';
206 oFile << str;
207 oFile.close();
208
209 return true;
210}
