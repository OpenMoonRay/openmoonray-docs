39/// This converts a lobe type to the corresponding embree compatible ray mask.
40/// Compute the proper ray mask value by shift left the lobe category bit by
41/// the surface side category bit - 1 position (0 or 1 left shift). Note that
42/// a lobe can be both transmission and reflection.
43/// Look in scene_rdl2/scene/rdl2/VisibilityFlags.h for the definition of the
44/// mask. Look in rendering/pbr/bsdf/Bsdf.h for the definition of lobe type.
45finline int
46lobeTypeToRayMask(const int lobeType)
47{
48 int mask = 0;
49 if (lobeType & shading::BsdfLobe::REFLECTION) {
50 mask |= (lobeType & shading::BsdfLobe::ALL_LOBES);
51 }
52 if (lobeType & shading::BsdfLobe::TRANSMISSION) {
53 mask |= (lobeType & shading::BsdfLobe::ALL_LOBES) << 1;
54 }
55
56 return mask;
57}
58
59finline scene_rdl2::rdl2::RaySwitchContext::RayType
60lobeTypeToRayType(const int lobeType)
61{
62 if (lobeType == 0) {
63 return scene_rdl2::rdl2::RaySwitchContext::RayType::CameraRay;
64 }
65 if (lobeType & shading::BsdfLobe::Type::DIFFUSE) {
66 return scene_rdl2::rdl2::RaySwitchContext::RayType::IndirectDiffuseRay;
67 }
68 if (lobeType & shading::BsdfLobe::Type::MIRROR) {
69 return scene_rdl2::rdl2::RaySwitchContext::RayType::IndirectMirrorRay;
70 }
71 if (lobeType & shading::BsdfLobe::Type::GLOSSY) {
72 return scene_rdl2::rdl2::RaySwitchContext::RayType::IndirectGlossyRay;
73 }
74 return scene_rdl2::rdl2::RaySwitchContext::RayType::OtherRay;
75}

77// Query the bsdf reflection lobes that their spawned rays can intersect with
78// geometries using specified ray mask
79finline int
80rayMaskToReflectionLobes(const int rayMask)
81{
82 return (rayMask & shading::BsdfLobe::ALL_LOBES) |
83 ((rayMask >> scene_rdl2::rdl2::sNumVisibilityTypes) & shading::BsdfLobe::ALL_LOBES) |
84 shading::BsdfLobe::REFLECTION;
85}
86
87// Query the bsdf transmission lobes that their spawned rays can intersect with
88// geometries using specified ray mask
89finline int
90rayMaskToTransmissionLobes(const int rayMask)
91{
92 return ((rayMask >> 1) & shading::BsdfLobe::ALL_LOBES) |
93 ((rayMask >> (scene_rdl2::rdl2::sNumVisibilityTypes + 1)) & shading::BsdfLobe::ALL_LOBES) |
94 shading::BsdfLobe::TRANSMISSION;
95}

97finline bool
98checkForNanSimple(const scene_rdl2::math::Color &radiance,
99 const char *contributionName, const Subpixel &sp)
100{
101 if (!scene_rdl2::math::isFinite(radiance)) {
102 scene_rdl2::logging::Logger::error("[MCRT-RENDER] " , contributionName ,
103 " produced NaN values at pixel (",
104 uint32ToPixelX(sp.mPixel) , ", " , uint32ToPixelY(sp.mPixel),
105 "), subpixel index (", sp.mSubpixelIndex, ")");
106 return true;
107 }
108 return false;
109}

159inline void
160addPriorityMaterial(const scene_rdl2::rdl2::Material* material,
161 const scene_rdl2::rdl2::Material* dstList[4],
162 int dstListCount[4])
163{
164 // First see if the material is already in the list. If so, increase its
165 // count.
166 for (int i = 0; i < 4; i++) {
167 if (dstList[i] == material) {
168 dstListCount[i]++;
169 return;
170 }
171 }
172
173 // Else put it in the first empty list slot
174 for (int i = 0; i < 4; i++) {
175 if (dstList[i] == nullptr) {
176 dstList[i] = material;
177 dstListCount[i] = 1;
178 return;
179 }
180 }
181}
182
183inline void
184removePriorityMaterial(const scene_rdl2::rdl2::Material* material,
185 const scene_rdl2::rdl2::Material* dstList[4],
186 int dstListCount[4])
187{
188 for (int i = 0; i < 4; i++) {
189 if (dstList[i] == material) {
190 dstListCount[i]--;
191 if (dstListCount[i] == 0) {
192 dstList[i] = nullptr;
193 }
194 return;
195 }
196 }
197}

212inline const scene_rdl2::rdl2::Material*
213getHighestPriorityMaterial(mcrt_common::RayDifferential &ray, int &highestPriority)
214{
215 const scene_rdl2::rdl2::Material *hpMat = nullptr;
216 highestPriority = 0;
217 const void** list = &ray.ext.priorityMaterial0;
218 for (int i = 0; i < 4; i++) {
219 if (list[i]) {
220 const scene_rdl2::rdl2::Material *mat = static_cast<const scene_rdl2::rdl2::Material*>(list[i]);
221 int priority = mat->priority();
222 if (hpMat == nullptr || priority < highestPriority) {
223 hpMat = mat;
224 highestPriority = priority;
225 }
226 }
227 }
228 return hpMat;
229}
