23// Returns angle in range in [-pi,pi]
24static finline float rangeAngle(float angle)
25{
26 while (angle > scene_rdl2::math::sPi) angle -= scene_rdl2::math::sTwoPi;
27 while (angle < -scene_rdl2::math::sPi) angle += scene_rdl2::math::sTwoPi;
28 return angle;
29}

61// Conversion from roughness to width (in radians)
62finline float
63roughness2Width(float roughness)
64{
65 // See rats test rats/material/base/roughness_comparison
66 static const float factor = 55.0f * scene_rdl2::math::sPi / 180.0f;
67
68 const float exponent = roughness2Exponent(roughness);
69 const float width = factor * scene_rdl2::math::pow(exponent, -0.8f);

72}

116// Dispersion Support Via Abbe Numbers
117
118// The following function computes three different IORs for R, G and B wavelengths based on a concept called
119// the "abbe number" as defined in https://en.wikipedia.org/wiki/Abbe_number.
120// I've used the following three equations to calculate the three variables
121// AbbeNumber = (etaG - 1) / (etaB - etaR) (1)
122// Assuming etaGreen as the average eta specified by the user:
123// etaG = etaT (2)
124// avgEta = etaT = (etaR + etaG + etaB) / 3.0f (3)
125// Substituing for etaG in equation (3), we get:
126// etaB + etaR = 2 * etaG (4)
127// Eqn (1) can be written as:
128// etaB - etaR = (etaG - 1) / AbbeNumber (5)
129// Eqns (4) and (5) can be solved as:
130// etaB = (etaG - 1) / (2 * AbbeNumber) + etaG
131// etaR = 2 * etaG - etaB
132finline void
133computeSpectralIOR(float eta,
134 float abbeNumber,
135 float *etaR,
136 float *etaG,
137 float *etaB)
138{
139 MNRY_ASSERT(!scene_rdl2::math::isZero(abbeNumber));
140 *etaG = eta;
141 *etaB = (*etaG - 1.0f) / (2 * abbeNumber) + *etaG;
142 *etaR = 2 * *etaG - *etaB;
143}
144
145// Sampling R, G and B wavelengths with the weights (0.35f, 0.35f, 0.30f).
146// Choosing this specific combination instead of (0.33, 0.33, 0.33) since it quite evenly sums to 1.0f
147// This function also renormalizes the random number used to select a wavelength and returns
148// the appropriate dispersionColor and PDF.
149finline void
150sampleSpectralIOR(float& r,
151 float etaR,
152 float etaG,
153 float etaB,
154 float *sampleEta,
155 float *pdf,
156 scene_rdl2::math::Color& dispersionColor)
157{
158 if (r < 0.35f) {
159 r /= 0.35f;
160 // Red
161 dispersionColor = scene_rdl2::math::Color(1.0f, 0.0f, 0.0f);
162 *sampleEta = etaR;
163 *pdf = 0.35f;
164 } else if (r < 0.7f) {
165 r = (r - 0.35f) / 0.35f;
166 // Green
167 dispersionColor = scene_rdl2::math::Color(0.0f, 1.0f, 0.0f);
168 *sampleEta = etaG;
169 *pdf = 0.35f;
170 } else {
171 r = (r - 0.7f) / 0.3f;
172 // Blue
173 dispersionColor = scene_rdl2::math::Color(0.0f, 0.0f, 1.0f);
174 *sampleEta = etaB;
175 *pdf = 0.3f;
176 }
177}

181finline scene_rdl2::math::Vec3f
182sampleLocalHemisphereCosine(float r1, float r2)
183{
184 // Sample polar coordinates
185 float cosTheta = (r1 > 1.0f ? 1.0f : scene_rdl2::math::sqrt(r1));
186 float sinTheta = scene_rdl2::math::sqrt(1.0f - r1);
187
188 float phi = r2 * scene_rdl2::math::sTwoPi;
189 float cosPhi;
190 float sinPhi;
191 scene_rdl2::math::sincos(phi, &sinPhi, &cosPhi);
192
193 // Convert to direction in the reference frame
194 scene_rdl2::math::Vec3f result;
195 result[0] = sinTheta * cosPhi;
196 result[1] = sinTheta * sinPhi;
197 result[2] = cosTheta;
198
199 return result;
200}

204{
205 float length1 = length(edge1);
206 float length2 = length(edge2);
207 float newLength = scene_rdl2::math::sqrt(length1 * length2);
208
209 edge1 *= newLength / length1;
210 edge2 *= newLength / length2;

264/// Y-up sphere parameterization: theta=0 --> y-axis, phi=0..pi/2 --> x..z axes
265finline void
266computeInverseYupSphericalDirection(const scene_rdl2::math::Vec3f &wi, float &theta, float &phi)
267{
268 // Inverse the transform from computeYupSphericalDirection()
269 float cosTheta = wi[1];
270 cosTheta = scene_rdl2::math::max(cosTheta, -1.0f);
271 cosTheta = scene_rdl2::math::min(cosTheta, 1.0f);
272 theta = scene_rdl2::math::acos(cosTheta);
273
274 phi = scene_rdl2::math::atan2(wi[2], wi[0]);
275 phi = (phi < 0.0f ? phi + scene_rdl2::math::sTwoPi : phi);
276}

292/// Assumes all vectors are normalized, and N points on the side of Ng, not Ngf
293/// Returns N.dot(wo) (which is also N.dot(wi))
294finline float
295computeReflectionDirection(const scene_rdl2::math::Vec3f &N, const scene_rdl2::math::Vec3f &wo, scene_rdl2::math::Vec3f &wi)
296{
297 float NDotWo = dot(N, wo);
298
299 wi = -wo + 2.0f * NDotWo * N;
300
301 // There is no need to normalize R here
302 MNRY_ASSERT(isNormalized(wi));
303
304 return NDotWo;
305}

308/// For details see:
309/// "Tracing ray differentials - Homan Igehy,
310/// Computer Graphics #33 (Annual Conference Series 1999)"
311finline void
312computeReflectionDirectionDifferential(
313 const scene_rdl2::math::Vec3f &wo, const scene_rdl2::math::Vec3f &wi, const scene_rdl2::math::Vec3f &H,
314 const scene_rdl2::math::Vec3f &dNdx, const scene_rdl2::math::Vec3f &dNdy, scene_rdl2::math::Vec3f &dDdx, scene_rdl2::math::Vec3f &dDdy)
315{
316 // Note: This is slightly incorrect, since we use the shading normal
317 // derivatives, not the H vector derivatives. But it is only important
318 // for this to be correct for mirror reflections (when H == N).
319 float Cx = dot(dDdx, H) - dot(wo, dNdx);
320 float Cy = dot(dDdy, H) - dot(wo, dNdy);
321
322 float dDotN = -dot(wo, H);
323 dDdx -= 2.0f * (dDotN * dNdx + Cx * H);
324 dDdy -= 2.0f * (dDotN * dNdy + Cy * H);
325}

328/// Same as computeReflectionDirectionDifferential(), but ignoring the effect
329/// of surface curvature
330finline void
331computeReflectionDirectionDifferentialNoCurvature(

340/// Assumes all vectors are normalized, and N points on the side of Ng (it has
341/// not been flipped towards wo).
342/// Returns false in case of total internal reflection, and true otherwise.
343/// In the latter case, it also returns scene_rdl2::math::abs(N.dot(wo)) and
344/// scene_rdl2::math::abs(N.dot(wi)).
345finline bool
346computeRefractionDirection(const scene_rdl2::math::Vec3f &N,
347 const scene_rdl2::math::Vec3f &wo,
348 float neta,
349 scene_rdl2::math::Vec3f &wi,
350 float &NdotWo,
351 float &NdotWi)
352{
353 // Compute refraction vector
354 NdotWo = dot(N, wo);
355 scene_rdl2::math::Vec3f Nf = (NdotWo > 0.0f ? N : -N);
356 NdotWo = scene_rdl2::math::abs(NdotWo);
357
358 NdotWi = 1.0f - neta * neta * (1.0f - (NdotWo * NdotWo));
359 if (NdotWi < 0.0f) {
360 // Total internal reflection
361 return false;
362 }
363 NdotWi = scene_rdl2::math::sqrt(NdotWi);
364
365 float negMu = neta * NdotWo - NdotWi;
366 wi = negMu * Nf - neta * wo;
367
368 // There is no need to normalize R here
369 MNRY_ASSERT(isNormalized(wi));
370
371 return true;
372}

375/// For details see:
376/// "Tracing ray differentials - Homan Igehy,
377/// Computer Graphics #33 (Annual Conference Series 1999)"
378finline void
379computeRefractionDirectionDifferential(const float neta,
380 const scene_rdl2::math::Vec3f &wo,
381 const scene_rdl2::math::Vec3f &wi,
382 const scene_rdl2::math::Vec3f &H,
383 const scene_rdl2::math::Vec3f &dNdx,
384 const scene_rdl2::math::Vec3f &dNdy,
385 scene_rdl2::math::Vec3f &dDdx,
386 scene_rdl2::math::Vec3f &dDdy)
387{
388 // Note: This is slightly incorrect, since we use the shading normal
389 // derivatives, not the H vector derivatives. But it is only important
390 // for this to be correct for mirror reflections (when H == N).
391 float HdotWo = dot(H, wo);
392
393 // TODO: Check if this test is necessary
394 scene_rdl2::math::Vec3f Hf = (HdotWo > 0.0f ? H : -H);
395 HdotWo = scene_rdl2::math::abs(HdotWo);
396
397 float Cx = dot(dDdx, Hf) - dot(wo, dNdx);
398 float Cy = dot(dDdy, Hf) - dot(wo, dNdy);
399
400 float HdotWi = dot(Hf, wi);
401 HdotWi = scene_rdl2::math::min(HdotWi, -scene_rdl2::math::sEpsilon);
402
403 float dmu = (neta + (neta * neta * HdotWo / HdotWi));
404 float dmudx = dmu * Cx;
405 float dmudy = dmu * Cy;
406
407 float mu = - neta * HdotWo - HdotWi;
408 dDdx = neta * dDdx - (mu * dNdx + dmudx * Hf);
409 dDdy = neta * dDdy - (mu * dNdy + dmudy * Hf);
410}

413/// H can be very small if wo and wi are almost the opposite vectors. This
414/// happens in the unfortunate event when the sampled H vector was almost
415/// at 90 degrees from wo. In that case we can't re-construct H
416/// and return false. We return a normalized H and true otherwise.
417finline bool
418computeNormalizedHalfVector(const scene_rdl2::math::Vec3f &wo, const scene_rdl2::math::Vec3f &wi, scene_rdl2::math::Vec3f &H)
419{
420 static const float halfVectorLengthMinSqr = 0.001f * 0.001f;
421
422 H = wo + wi;
423
424 float lengthHSqr = lengthSqr(H);
425 if (lengthHSqr > halfVectorLengthMinSqr) {
426 H *= scene_rdl2::math::rsqrt(lengthHSqr);
427 return true;
428 } else {
429 return false;
430 }
431}

434// H is always pointing towards medium with lowest IOR, but only if wo and wi
435// are pointing on opposite side of the surface (wrt. N).
436finline bool
437computeNormalizedRefractionHalfVector(const float iorWo, const scene_rdl2::math::Vec3f &wo,
438 const float iorWi, const scene_rdl2::math::Vec3f &wi, scene_rdl2::math::Vec3f &H)
439{
440 static const float halfVectorLengthMinSqr = 0.001f * 0.001f;
441
442 H = -(iorWo * wo + iorWi * wi);
443
444 float lengthHSqr = lengthSqr(H);
445 if (lengthHSqr > halfVectorLengthMinSqr) {
446 H *= scene_rdl2::math::rsqrt(lengthHSqr);
447 return true;
448 } else {
449 return false;
450 }
451}

454finline void
455localHemisphereCosineDifferentials(float r1, float r2, scene_rdl2::math::Vec3f &dDdr1, scene_rdl2::math::Vec3f &dDdr2)
456{
457 // See lib/rendering/pbr/doc/diffuse_scatter_derivatives.mw for Maple
458 // source code on how we got to this result
459
460 // The derivative gets very big close to the equator and the pole
461 // (because of the division by cosTheta and sinTheta below)
462 // Cap r1 to make sure we stay an epsilon away from both
463 static const float minR1 = scene_rdl2::math::pow(scene_rdl2::math::cos(scene_rdl2::math::deg2rad(85.0f)), 2.0f);
464 static const float maxR1 = 1.0f - minR1;
465 r1 = scene_rdl2::math::clamp(r1, minR1, maxR1);
466
467 float cosTheta = (r1 > 1.0f ? 1.0f : scene_rdl2::math::sqrt(r1));
468 float sinTheta = scene_rdl2::math::sqrt(1.0f - r1);
469
470 float phi = r2 * scene_rdl2::math::sTwoPi;
471 float cosPhi;
472 float sinPhi;
473 scene_rdl2::math::sincos(phi, &sinPhi, &cosPhi);
474
475 // We scale our derivatives down to account for multiple samples taken over
476 // the hemisphere
477 static const float scale = 1.0f / 8.0f;
478 static const float halfScale = 0.5f * scale;
479 static const float twoPiScale = scene_rdl2::math::sTwoPi * scale;
480
481 dDdr1 = scene_rdl2::math::Vec3f(-halfScale * cosPhi / sinTheta,
482 -halfScale * sinPhi / sinTheta,
483 halfScale / cosTheta);
484 dDdr2 = scene_rdl2::math::Vec3f(-twoPiScale * sinPhi * sinTheta,
485 twoPiScale * cosPhi * sinTheta,
486 0.0f);
487}

585// Converts a continuous variable in the space [0,1] (t)
586// to the nearest of n (numBins) discrete values in [0,1].
587finline static float
588discretize(const float t, const size_t numBins)
589{
590 // optimization
591 if (numBins == 1) { return 1.0f; }
592
593 return scene_rdl2::math::ceil(scene_rdl2::math::saturate(t) * numBins) / numBins;
594}
