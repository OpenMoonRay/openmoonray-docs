19class shared_mutex_pthread
20{
21 pthread_rwlock_t mRWLock = PTHREAD_RWLOCK_INITIALIZER;
22
23public:
24 shared_mutex_pthread() = default;
25 ~shared_mutex_pthread() = default;
26
27 shared_mutex_pthread(const shared_mutex_pthread&) = delete;
28 shared_mutex_pthread& operator=(const shared_mutex_pthread&) = delete;

30 void lock()
31 {
32 const int ret = pthread_rwlock_wrlock(&mRWLock);
33 if (ret == EDEADLK) {
34 throw std::system_error(ret, std::system_category(), "Resource deadlock would occur");
35 }
36 MNRY_ASSERT(ret == 0);
37 }
38
39 bool try_lock()
40 {
41 const int ret = pthread_rwlock_trywrlock(&mRWLock);
42 if (ret == EBUSY) {
43 return false;
44 }
45 MNRY_ASSERT(ret == 0);
46 return true;
47 }
48
49 void unlock()
50 {
51 const int ret __attribute__((unused)) = pthread_rwlock_unlock(&mRWLock);
52 MNRY_ASSERT(ret == 0);
53 }
54
55 void lock_shared()
56 {
57 int ret;
58 // We retry if we exceeded the maximum number of read locks supported by
59 // the POSIX implementation; this can result in busy-waiting, but this
60 // is okay based on the current specification of forward progress
61 // guarantees by the standard.
62 do {
63 ret = pthread_rwlock_rdlock(&mRWLock);
64 } while (ret == EAGAIN);
65 if (ret == EDEADLK) {
66 throw std::system_error(ret, std::system_category(), "Resource deadlock would occur");
67 }
68 MNRY_ASSERT(ret == 0);
69 }
70
71 bool try_lock_shared()
72 {
73 const int ret = pthread_rwlock_tryrdlock(&mRWLock);
74 // If the maximum number of read locks has been exceeded, we just fail
75 // to acquire the lock. Unlike for lock(), we are not allowed to throw
76 // an exception.
77 if (ret == EBUSY || ret == EAGAIN) {
78 return false;
79 }
80 MNRY_ASSERT(ret == 0);
81 return true;
82 }
83
84 void unlock_shared()
85 {
86 unlock();
87 }
88
89 void* native_handle() { return &mRWLock; }
90};
