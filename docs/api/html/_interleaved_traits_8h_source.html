112 InterleavedTraits& operator=(InterleavedTraits&& other)
113 noexcept(traits::propagate_on_container_move_assignment::value ||
114 fauxstd::is_always_equal<allocator_type>::value)
115 {
116 if (traits::propagate_on_container_move_assignment::value &&
117 this->getAllocatorInternal() != other.getAllocatorInternal()) {
118 // We get to move the allocator. This is good.
119 allocator_type::operator=(std::move(other));
120 mCapacity = other.mCapacity;
121 mSize = other.mSize;
122 mTimeSteps = other.mTimeSteps;
123 mData = std::move(other.mData);
124
125 other.mTimeSteps = 0;
126 other.mCapacity = 0;
127 other.mSize = 0;
128 other.mData = nullptr;
129 } else if (fauxstd::is_always_equal<allocator_type>::value ||
130 this->getAllocatorInternal() ==
131 other.getAllocatorInternal()) {
132 // We don't have to move the allocator. This is good.
133 mCapacity = other.mCapacity;
134 mSize = other.mSize;
135 mTimeSteps = other.mTimeSteps;
136 mData = std::move(other.mData);
137
138 other.mTimeSteps = 0;
139 other.mCapacity = 0;
140 other.mSize = 0;
141 other.mData = nullptr;
142 } else {
143 // Our only option is to copy all of the elements, because we can't
144 // do anything smart with the allocators. This is bad.
145 auto newData = copyMove(getAllocatorInternal(), other.mData,
146 other.mSize, other.mTimeSteps);
147 destroy();
148 mCapacity = other.mSize;
149 mSize = other.mSize;
150 mTimeSteps = other.mTimeSteps;
151 mData = newData;
152 }
153
154 return *this;
155 }

356 template <typename... Args>
357 static pointer create(allocator_type& alloc, size_type size,
358 size_type timeSteps, Args&&... args)
359 {
360 pointer p = traits::allocate(alloc, size * timeSteps);
361 try {
362 constructArraySamples(alloc, p, size * timeSteps,
363 std::forward<Args>(args)...);
364 } catch (...) {
365 // Any constructed elements will be destroyed in
366 // constructArraySamples, but we're in charge of the memory.
367 traits::deallocate(alloc, p, size * timeSteps);
368 throw;
369 }
370 return p;
371 }

373 static pointer copyValue(allocator_type& alloc, pointer oldData,
374 size_type size, size_type timeSteps)
375 {
376 pointer p = traits::allocate(alloc, size * timeSteps);
377 try {
378 copyArraySamples(alloc, p, oldData, size * timeSteps);
379 } catch (...) {
380 traits::deallocate(alloc, p, size * timeSteps);
381 throw;
382 }

385
386 static pointer copyMove(allocator_type& alloc, pointer oldData,
387 size_type size, size_type timeSteps)
388 {
389 pointer p = traits::allocate(alloc, size * timeSteps);
390 if (!std::is_nothrow_move_constructible<value_type>::value) {
391 try {
392 // If a move fails, we can't unwind, so we copy if move can throw.
393 copyArraySamples(alloc, p, oldData, size * timeSteps);
394 } catch (...) {
395 traits::deallocate(alloc, p, size * timeSteps);
396 throw;
397 }
398 } else {
399 for (size_type i = 0; i < size; ++i) {
400 traits::construct(alloc, p + i, std::move(oldData[i]));
401 }
402 }
403 return p;
404 }

406 void change_memory_size(size_type size)
407 {
408 using std::swap; // Allow ADL
409 pointer p = traits::allocate(getAllocatorInternal(), size * mTimeSteps);
410 try {
411 moveArraySamples(getAllocatorInternal(), p, mData,
412 mSize * mTimeSteps);
413 } catch (...) {
414 // moveArraySamples may throw, in that case, we have to
415 // deallocate the memory we've already allocated.
416 traits::deallocate(getAllocatorInternal(), p, size * mTimeSteps);
417 throw;
418 }
419 // Destroy the old values. We (justifiably) assume that
420 // destruction does not throw.
421 destroyArraySamples(getAllocatorInternal(), mData, mSize * mTimeSteps);
422 // Now that all of the throwing operations are done, we can modify
423 // the state of our object with non-throwing operations.
424 swap(p, mData);
425 traits::deallocate(getAllocatorInternal(), p, mCapacity * mTimeSteps);
426 mCapacity = size;
427 }
428
429 void destroy() noexcept
430 {
431 if (mData) {
432 destroyArraySamples(getAllocatorInternal(), mData,
433 mSize * mTimeSteps);
434 traits::deallocate(getAllocatorInternal(), mData,
435 mCapacity * mTimeSteps);
436 }
437 }
