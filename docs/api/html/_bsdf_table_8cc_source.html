86IsotropicBsdfTable::IsotropicBsdfTable(const std::string &filename) :
87 mData(nullptr)
88{
89 // Read file
90 FILE *file = fopen(filename.c_str(), "rb");
91 if (!file) {
92 throw scene_rdl2::except::IoError("Cannot open file \"" + filename + "\" ("
93 + strerror(errno) + ")");
94 }
95
96 if (fread(&mSizeThetaH, sizeof(int), 1, file) != 1 ||
97 fread(&mSizeThetaD, sizeof(int), 1, file) != 1 ||
98 fread(&mSizePhiD, sizeof(int), 1, file) != 1) {
99 fclose(file);
100 throw scene_rdl2::except::IoError("Cannot read table size in file \"" + filename + "\" ("
101 + strerror(errno) + ")");
102 }
103
104 // Sizes are bin sizes over thetaH, thetaD, phiD/2
105 int n = getColorCount();
106 double *data = new double[3 * n];
107 if (fread(data, sizeof(double), 3 * n, file) != 3 * n) {
108 delete [] data;
109 fclose(file);
110 throw scene_rdl2::except::IoError("Cannot read table in file \"" + filename + "\" ("+ strerror(errno) + ")");
111 }
112 fclose(file);
113
114 init();
115
116 // Copy to floats, re-interleaving data and applying scale
117 for (int i=0; i < n; i++) {
118 int j = 3 * i;
119 mData[j] = data[i] * sRedScale;
120 mData[j + 1] = data[i + n] * sGreenScale;
121 mData[j + 2] = data[i + n * 2] * sBlueScale;
122
123 //Color c(mData[j], mData[j + 1], mData[j + 2]);
124 //std::cout << c << std::endl;
125 }
126
127 delete [] data;
128}

131void
132IsotropicBsdfTable::saveAs(const std::string &filename) const
133{
134 // Open file
135 FILE *file = fopen(filename.c_str(), "wb");
136 if (!file) {
137 throw scene_rdl2::except::IoError("Cannot open file \"" + filename + "\" ("
138 + strerror(errno) + ")");
139 }
140
141 if (fwrite(&mSizeThetaH, sizeof(int), 1, file) != 1 ||
142 fwrite(&mSizeThetaD, sizeof(int), 1, file) != 1 ||
143 fwrite(&mSizePhiD, sizeof(int), 1, file) != 1) {
144 fclose(file);
145 throw scene_rdl2::except::IoError("Cannot write table size in file \"" + filename + "\" ("
146 + strerror(errno) + ")");
147 }
148
149 int n = getColorCount();
150 double *data = new double[3 * n];
151
152 // Copy to doubles, de-interleaving data and inverting scale
153 for (int i=0; i < n; i++) {
154 int j = 3 * i;
155 data[i] = mData[j] / sRedScale;
156 data[i + n] = mData[j + 1] / sGreenScale;
157 data[i + n * 2] = mData[j + 2] / sBlueScale;
158 }
159
160 if (fwrite(data, sizeof(double), 3 * n, file) != 3 * n) {
161 fclose(file);
162 delete [] data;
163 throw scene_rdl2::except::IoError("Cannot write table in file \"" + filename + "\" ("+ strerror(errno) + ")");
164 }
165 fclose(file);
166
167 delete [] data;
168}

173static const float sFloatIndexEpsilon = 1e-3f;
174
175float
176IsotropicBsdfTable::getIndexThetaH(float thetaH) const
177{
178 // Only do first half od the test to account for directions slightly
179 // below the surface due to interpolated normals
180 MNRY_ASSERT(thetaH >= 0.0f /* && thetaH <= sHalfPi */);
181
182 return sqrt(thetaH / sHalfPi) * (mSizeThetaH - sFloatIndexEpsilon);
183}

186float
187IsotropicBsdfTable::getThetaH(int index) const
188{
189 MNRY_ASSERT(index >= 0 && index < mSizeThetaH);
190 float tmp = float(index) / mSizeThetaH;
191 return tmp * tmp * sHalfPi;
192}

196IsotropicBsdfTable::getIndexThetaD(float thetaD) const
197{

212float
213IsotropicBsdfTable::getIndexPhiD(float phiD) const
214{
215 MNRY_ASSERT(phiD >= -sPi && phiD <= sPi);
216
217 // Because of reciprocity, the BRDF is unchanged under phiD -> phiD + sPi
218 if (phiD < 0.0f) {
219 phiD += sPi;
220 }
221
222 MNRY_ASSERT(phiD >= 0.0f);
223 return phiD / sPi * (mSizePhiD - sFloatIndexEpsilon);
224}

227float
228IsotropicBsdfTable::getPhiD(int index) const
229{
230 MNRY_ASSERT(index >= 0 && index < mSizePhiD);
231 return float(index) / mSizePhiD * sPi;
232}

237// Convert standard coordinates to half vector/difference vector coordinates
238// Returns true if ok and false if the conversion failed
239// TODO: paper reference
240bool
241IsotropicBsdfTable::localWoWi2IndexHD(const Vec3f &localWo, const Vec3f &localWi,
242 float &indexThetaH, float &indexThetaD, float &indexPhiD) const
243{
244 Vec3f H;
245 if (!computeNormalizedHalfVector(localWo, localWi, H)) {
246 return false;
247 }
248
249 // Compute thetaH, phiH
250 if (H.z < sEpsilon) {
251 return false;
252 }
253 float thetaH = acos(clamp(H.z, -1.0f, 1.0f));
254 float phiH = atan2(H.y, H.x);
255
256 // Compute difference vector
257 static const ReferenceFrame frame;
258 Vec3f D = rotate(localWi, frame.getN(), -phiH);
259 D = rotate(D, frame.getY(), -thetaH);
260
261 // Compute thetaD, phiD
262 float thetaD = acos(clamp(D.z, -1.0f, 1.0f));
263 float phiD = atan2(D.y, D.x);
264
265 // Convert to indices
266 indexThetaH = getIndexThetaH(thetaH);
267 indexThetaD = getIndexThetaD(thetaD);
268 indexPhiD = getIndexPhiD(phiD);
269
270 //std::cout << ", th/td/pd = " << thetaH << ", " << thetaD << ", " << phiD;
271 //std::cout << ", ith/itd/ipd = " << indexThetaH << ", " << indexThetaD << ", " << indexPhiD;
272
273 return true;
274}

293void
294IsotropicBsdfTable::indexHD2localWoWi(int indexThetaH, int indexThetaD, int indexPhiD,
295 Vec3f &localWo, Vec3f &localWi) const
296{
297 // TODO: Compute phiH such that phiWo = 0 ?
298 float thetaH = getThetaH(indexThetaH);
299 float phiH = 0.0f;
300 float thetaD = getThetaD(indexThetaD);
301 float phiD = getPhiD(indexPhiD);
302
303 // Compute D
304 float s, c;
305 sincos(thetaD, &s, &c);
306 Vec3f D = computeLocalSphericalDirection(c, s, phiD);
307
308 // Compute localWi
309 static const ReferenceFrame frame;
310 localWi = rotate(D, frame.getY(), thetaH);
311 // Since phiH is 0 we can ignore:
312 // localWi = rotate(localWi, frame.getN(), phiH);
313
314 // Compute H
315 sincos(thetaH, &s, &c);
316 Vec3f H = computeLocalSphericalDirection(c, s, phiH);
317
318 // Compute localWo
319 computeReflectionDirection(H, localWi, localWo);
320}

325Color
326IsotropicBsdfTable::getBsdf(const Vec3f &localWo, const Vec3f &localWi, bool useLerp) const
327{
328 // Convert (wi, wo) to thetaH, thetaD, phiD floating-point indices
329 float findexThetaH, findexThetaD, findexPhiD;
330 if (!localWoWi2IndexHD(localWo, localWi, findexThetaH, findexThetaD, findexPhiD)) {
331 return sBlack;
332 }
333 int indexThetaH = int(findexThetaH);
334 int indexThetaD = int(findexThetaD);
335 int indexPhiD = int(findexPhiD);
336
337 if (useLerp) {
338
339 // Use tri-linear interpolation
340 Color c[4];
341 float t;
342
343 int indexThetaHNext = min(indexThetaH + 1, mSizeThetaH - 1);
344 int indexThetaDNext = min(indexThetaD + 1, mSizeThetaD - 1);
345 int indexPhiDNext = indexPhiD + 1;
346 indexPhiDNext = indexPhiDNext == mSizePhiD ? 0 : indexPhiDNext;
347
348 // Interpolate over thetaH
349 t = findexThetaH - indexThetaH;
350 c[0] = lerp(getColor(indexThetaH, indexThetaD, indexPhiD),
351 getColor(indexThetaHNext, indexThetaD, indexPhiD), t);
352 c[1] = lerp(getColor(indexThetaH, indexThetaDNext, indexPhiD),
353 getColor(indexThetaHNext, indexThetaDNext, indexPhiD), t);
354 c[2] = lerp(getColor(indexThetaH, indexThetaD, indexPhiDNext),
355 getColor(indexThetaHNext, indexThetaD, indexPhiDNext), t);
356 c[3] = lerp(getColor(indexThetaH, indexThetaDNext, indexPhiDNext),
357 getColor(indexThetaHNext, indexThetaDNext, indexPhiDNext), t);
358
359 // Interpolate over thetaD
360 t = findexThetaD - indexThetaD;
361 c[0] = lerp(c[0], c[1], t);
362 c[2] = lerp(c[2], c[3], t);
363
364 // Interpolate over phiD
365 t = findexPhiD - indexPhiD;
366 c[0] = lerp(c[0], c[2], t);
367
368 return c[0];
369
370 } else {
371
372 // Use piece-wise constant lookup
373 return getColor(indexThetaH, indexThetaD, indexPhiD);
374 }
375}
