40void
41reSampleBsdf(const std::vector<int> &phiValues,
42 const SlicePtrArray &slicesBottom, const SlicePtrArray &slicesTop,
43 const scene_rdl2::math::Vec2f &smootThetaWoRange, AnisotropicBsdfTable &bsdf)
44{
45 MNRY_ASSERT(bsdf.getSizePhiH() == phiValues.size() * 2);
46
47 int size = phiValues.size();
48 for (int indexThetaH = 0; indexThetaH < bsdf.getSizeThetaH(); indexThetaH++) {
49 for (int indexPhiH = 0; indexPhiH < size; indexPhiH++) {
50 for (int indexThetaD = 0; indexThetaD < bsdf.getSizeThetaD(); indexThetaD++) {
51 for (int indexPhiD = 0; indexPhiD < bsdf.getSizePhiD(); indexPhiD++) {
52
53 scene_rdl2::math::Vec3f localWo, localWi;
54 scene_rdl2::math::Color color;
55 int iph;
56
57 iph = indexPhiH;
58 bsdf.indexHD2localWoWi(indexThetaH, iph,
59 indexThetaD, indexPhiD,
60 localWo, localWi);
61 color = slicesBottom[iph]->getBsdf(localWo, localWi, smootThetaWoRange);
62 bsdf.setBsdf(indexThetaH, iph, indexThetaD, indexPhiD, color);
63
64 iph = indexPhiH + size;
65 bsdf.indexHD2localWoWi(indexThetaH, iph,
66 indexThetaD, indexPhiD,
67 localWo, localWi);
68 color = slicesTop[indexPhiH]->getBsdf(localWo, localWi, smootThetaWoRange);
69 bsdf.setBsdf(indexThetaH, iph, indexThetaD, indexPhiD, color);
70 }
71 }
72 }
73 }
74}

79void
80sampleLightStageTriangles(const LightStageCylinderBsdfSlice &slice)
81{
82 // Use this to sample triangles out of the lights
83 scene_rdl2::math::Vec3f wo(0.0f, 0.0f, 1.0f);
84 for (int indexTheta = 0; indexTheta <= 180; indexTheta++) {
85 for (int indexPhi = 0; indexPhi <= 360; indexPhi++) {
86 float theta = float(indexTheta) * sHalfPi / 90.0f;
87 float phi = float(indexPhi) * sHalfPi / 90.0f;
88 float s, c;
89 sincos(theta, &s, &c);
90 scene_rdl2::math::Vec3f wi = moonray::shading::computeLocalSphericalDirection(c, s, phi);
91 slice.getBsdf(wo, wi, scene_rdl2::math::Vec2f(0.0f));
92 }
93 }
94}

138void
139sampleDebugBsdfPatternHD(IsotropicBsdfTable &bsdf)
140{
141 // Sample a debug pattern that varies smoothly only along thetaO
142 for (int indexThetaH = 0; indexThetaH < bsdf.getSizeThetaH(); indexThetaH++) {
143 for (int indexThetaD = 0; indexThetaD < bsdf.getSizeThetaD(); indexThetaD++) {
144 for (int indexPhiD = 0; indexPhiD < bsdf.getSizePhiD(); indexPhiD++) {
145 scene_rdl2::math::Vec3f localWo, localWi;
146 bsdf.indexHD2localWoWi(indexThetaH, indexThetaD, indexPhiD, localWo, localWi);
147
148 // ThetaO / PhiO
149 float cosThetaO = clamp(localWo.z, -1.0f, 1.0f);
150 float thetaO = acos(cosThetaO);
151 float phiO = atan2(localWo.y, localWo.x);
152
153 // ThetaI / PhiI
154 float cosThetaI = clamp(localWi.z, -1.0f, 1.0f);
155 float thetaI = acos(cosThetaI);
156 float phiI = atan2(localWi.y, localWi.x);
157
158 float phi = moonray::shading::rangeAngle(phiI - phiO);
159
160 scene_rdl2::math::Color color(thetaO / sHalfPi, thetaI / sHalfPi,
161 scene_rdl2::math::abs(phi) / sPi);
162
163 bsdf.setBsdf(indexThetaH, indexThetaD, indexPhiD, color);
164
165// std::cout << "ith, itd, ipd = " << indexThetaH << ", " << indexThetaD << ", " << indexPhiD <<
166// ", wo = " << localWo << ", wi = " << localWi << ", color = " << color << std::endl;
167
168 int ith, itd, ipd;
169 bsdf.localWoWi2IndexHD(localWo, localWi, ith, itd, ipd);
170 if (scene_rdl2::math::abs(ith - indexThetaH) > 1 ||
171 scene_rdl2::math::abs(itd - indexThetaD) > 1 ||
172 scene_rdl2::math::abs(ipd - indexPhiD) > 1) {
173
174 MNRY_ASSERT_REQUIRE(0);
175 bsdf.indexHD2localWoWi(indexThetaH, indexThetaD, indexPhiD, localWo, localWi);
176 bsdf.localWoWi2IndexHD(localWo, localWi, ith, itd, ipd);
177 }
178 }
179 }
180 }
181}

202int
203main(int argc, char* argv[])
204{
205 try {
206 //------------------------------------
207
208 // Initialize our logging
209 Logger::init();
210
211 // Check for no flags or help flag.
212 if (argc == 1 || std::string(argv[1]) == "-h") {
213 usage(argv[0], "");
214 std::exit(EXIT_FAILURE);
215 }
216
217 //------------------------------------
218
219 // Args parsing
220 Args args(argc, argv);
221 Args::StringArray values;
222
223 if (args.getFlagValues("-in", 2, values) < 0) {
224 usage(argv[0], "Missing input prefix");
225 std::exit(EXIT_FAILURE);
226 }
227 std::string inPrefix = values[0];
228 std::vector<int> phiValues;
229 stringToIntArray(values[1], phiValues);
230
231 if (args.getFlagValues("-out", 1, values) < 0) {
232 usage(argv[0], "Missing output filename");
233 std::exit(EXIT_FAILURE);
234 }
235 std::string outFilename = values[0];
236
237 scene_rdl2::math::Vec2f smoothThetaWoRange(0.05f * sHalfPi, 0.1f * sHalfPi);
238 if (args.getFlagValues("-smoothThetaWoRange", 2, values) >= 0) {
239 smoothThetaWoRange[0] = stringToFloat(values[0]);
240 smoothThetaWoRange[1] = stringToFloat(values[1]);
241 smoothThetaWoRange *= sHalfPi;
242 }
243
244 //------------------------------------
245
246 Logger::info("Processing input sequence: ");
247 SlicePtrArray slicesBottom;
248 SlicePtrArray slicesTop;
249 int size = phiValues.size();
250 for (int i = 0; i < size; i++) {
251 LightStageCylinderBsdfSlice *slice;
252
253 std::ostringstream number;
254 number << phiValues[i];
255
256 std::string filenameBottom = inPrefix + "_" + number.str() + "_bottom.slice";
257 Logger::info("Reading bsdf slice file: \"" , filenameBottom , "\"");
258 slice = new LightStageCylinderBsdfSlice(filenameBottom);
259 slicesBottom.push_back(slice);
260
261 std::string filenameTop = inPrefix + "_" + number.str() + "_top.slice";
262 Logger::info("Reading bsdf slice file: \"" , filenameTop , "\"");
263 slice = new LightStageCylinderBsdfSlice(filenameTop);
264 slicesTop.push_back(slice);
265 }
266
267 Logger::info("Re-Sampling bsdf...");
268 AnisotropicBsdfTable bsdf(90, 2 * size, 90, 360, false);
269 reSampleBsdf(phiValues, slicesBottom, slicesTop, smoothThetaWoRange, bsdf);
270 Logger::info("Writing bsdf file: \"" , outFilename , "\"");
271 bsdf.saveAs(outFilename);
272
273 } catch (const std::exception& e) {
274 Logger::error(e.what());
275 std::exit(EXIT_FAILURE);
276 }
277
278 return EXIT_SUCCESS;
279}
