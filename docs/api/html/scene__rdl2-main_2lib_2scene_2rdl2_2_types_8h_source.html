247/**
248 * Defines bitflags that affect the behavior of attributes.
249 *
250 * The "bindable" flag indicates that an attribute may have a binding
251 * registered in addition to having a value. Client code must decide what to do
252 * with the bound object. RDL does not know how to "evaluate" these bindings.
253 *
254 * The "blurrable" flag indicates that an attribute has multiple values, one at
255 * each timestep defined by the AttributeTimestep enum.
256 *
257 * The "enumerable" flag indicates that an attribute can only take on a fixed
258 * number of defined values.
259 *
260 * The "filename" flag indicates that this attribute represents a filename.
261 *
262 * The "no_geom_reload" flag indicates that an attribute update would not cause
263 * geometry to regenerate/tessellate/construct accelerator
264 */
265enum AttributeFlags
266{
267 FLAGS_NONE = 0,
268 FLAGS_BINDABLE = 1 << 0,
269 FLAGS_BLURRABLE = 1 << 1,
270 FLAGS_ENUMERABLE = 1 << 2,
271 FLAGS_FILENAME = 1 << 3,
272 FLAGS_CAN_SKIP_GEOM_RELOAD = 1 << 4
273};

460/**
461 * This struct holds the fast time rescaling coefficients, which are used by
462 * interpolated get().
463 *
464 * The basic idea is that rather than resampling the attribute data, we'd
465 * rather rescale the ray's time value from shutter interval parameter space
466 * (Sopen -> Sclose = 0 -> 1) to motion step parameter space
467 * (MotionStep[0] -> MotionStep[1] = 0 -> 1). These spaces are linked through
468 * the fact that both the motion steps and the shutter interval are defined in
469 * the same space of frame-relative time.
470 *
471 * To start, we first need to remap from ray parameter space to frame-
472 * relative time space:
473 *
474 * Tframe = (Sopen - Sclose) * Tray + Sopen
475 *
476 * (This form of linear interpolation is fine for this purpose, since the
477 * shutter open and close times (Sclose - Sopen) are always close in value. We
478 * won't accumulate much floating point error.
479 *
480 * Next, we need to do the inverse to map from frame-relative time space
481 * into motion step parameter space:
482 *
483 * Tmostep = (Tframe - MotionStep[0]) / (MotionStep[1] - MotionStep[0])
484 *
485 * Substituting Tframe and doing a bit of algebra gives us:
486 *
487 * Tmostep = M * Tray + B, where
488 * M = (Sclose - Sopen) / (MotionStep[1] - MotionStep[0])
489 * B = (Sopen - MotionStep[0]) / (MotionStep[1] - MotionStep[0])
490 *
491 * So, given Tray, M, and B, we can quickly compute Tmostep, which is the
492 * time value we should use for interpolating RDL data in motion step parameter
493 * space. Huzzah!
494 */
495struct TimeRescalingCoeffs
496{
497 float mScale;
498 float mOffset;
499};
