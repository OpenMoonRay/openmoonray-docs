89void
90McrtFbSender::initRenderOutput(const rndr::RenderOutputDriver *rod)
91{
92 if (!rod) {
93 //
94 // disable renderOutput
95 //
96 mRenderOutputName.clear();
97 mRenderOutputName.shrink_to_fit();
98 mRenderOutputSkipCondition.clear();
99 mRenderOutputSkipCondition.shrink_to_fit();
100 mActivePixelsRenderOutput.clear();
101 mActivePixelsRenderOutput.shrink_to_fit();
102 mRenderOutputBufferTiled.clear();
103 mRenderOutputBufferTiled.shrink_to_fit();
104 mRenderOutputBufferDefaultValue.clear();
105 mRenderOutputBufferDefaultValue.shrink_to_fit();
106 mRenderOutputWeightBufferTiled.clear();
107 mRenderOutputWeightBufferTiled.shrink_to_fit();
108 mRenderOutputBufferScaledByWeight.clear();
109 mRenderOutputBufferScaledByWeight.shrink_to_fit();
110 mRenderOutputBufferOrigNumChan.clear();
111 mRenderOutputBufferOrigNumChan.shrink_to_fit();
112 mRenderOutputBufferClosestFilterStatus.clear();
113 mRenderOutputBufferClosestFilterStatus.shrink_to_fit();
114 mRenderOutputBufferFinePassPrecision.clear();
115 mRenderOutputBufferFinePassPrecision.shrink_to_fit();
116 mRenderOutputBufferCoarsePassPrecision.clear();
117 mRenderOutputBufferCoarsePassPrecision.shrink_to_fit();
118
119 initHeatMap(-1); // no heatMap buffer
120 initWeightBuffer(nullptr, -1); // no weight buffer
121 initRenderBufferOdd(-1, -1); // no renderBufferOdd buffer
122 return;
123 }
124
125 const unsigned int total = rod->getNumberOfRenderOutputs();
126 mRenderOutputName.resize(total);
127 mRenderOutputSkipCondition.resize(total);
128 mActivePixelsRenderOutput.resize(total);
129 mRenderOutputBufferTiled.resize(total);
130 mRenderOutputBufferDefaultValue.resize(total);
131 mRenderOutputWeightBufferTiled.resize(total);
132 mRenderOutputBufferScaledByWeight.resize(total);
133 mRenderOutputBufferOrigNumChan.resize(total);
134 mRenderOutputBufferClosestFilterStatus.resize(total);
135 mRenderOutputBufferCoarsePassPrecision.resize(total);
136 mRenderOutputBufferFinePassPrecision.resize(total);
137
138 const pbr::AovSchema &schema = rod->getAovSchema();
139
140 int beautyId = -1;
141 int alphaId = -1;
142 int heatMapId = -1;
143 int weightBufferId = -1;
144 int beautyAuxId = -1;
145 int alphaAuxId = -1;
146 for (unsigned int roIdx = 0; roIdx < total; ++roIdx) {
147 mRenderOutputName[roIdx] = rod->getRenderOutput(roIdx)->getName(); // save buffer name
148
149 const int aovIdx = rod->getAovBuffer(roIdx);
150 if (schema.isVarianceEntry(aovIdx)) {
151 //
152 // Variance AOV
153 //
154 const int varianceSourceRoIdx = schema.getVarianceToSourceAOV(aovIdx);
155 if (rod->isVisibilityAov(varianceSourceRoIdx)) {
156 initRenderOutputVarianceVisibilityAOV(rod, roIdx);
157 } else {
158 initRenderOutputVarianceAOV(rod, roIdx);
159 }
160
161 } else if (rod->isVisibilityAov(roIdx)) {
162 //
163 // Visibility AOV
164 //
165 initRenderOutputVisibilityAOV(rod, roIdx);
166
167 } else {
168 //
169 // Regular AOV
170 //
171 initRenderOutputRegularAOV(rod, roIdx,
172 beautyId, alphaId, heatMapId, weightBufferId, beautyAuxId, alphaAuxId);
173 }
174 }
175
176 adjustRenderBufferFinePassPrecision(rod, beautyId, alphaId, beautyAuxId, alphaAuxId);
177
178 initHeatMap(heatMapId); // initialize heatMap memory information
179 initWeightBuffer(rod, weightBufferId); // initialize weight buffer memory information
180 initRenderBufferOdd(beautyAuxId, alphaAuxId); // initialize renderBufferOdd memory information
181}

183void
184McrtFbSender::fbReset()
185//
186// only reset fb related information
187//
188{
189 mRenderBufferTiled.clear();
190 mRenderBufferWeightBufferTiled.clear();
191
192 if (mPixelInfoStatus) {
193 mPixelInfoBufferTiled.clear(scene_rdl2::fb_util::PixelInfo(FLT_MAX));
194 mPixelInfoWeightBufferTiled.clear();
195 }
196
197 if (mHeatMapStatus) {
198 mHeatMapBufferTiled.clear();
199 mHeatMapWeightBufferTiled.clear();
200 mHeatMapSecBufferTiled.clear();
201 }
202
203 if (mWeightBufferStatus) {
204 mWeightBufferTiled.clear();
205 }
206
207 if (mRenderBufferOddStatus) {
208 mRenderBufferOddTiled.clear();
209 mRenderBufferOddWeightBufferTiled.clear();
210 }
211
212 for (size_t rodId = 0; rodId < mActivePixelsRenderOutput.size(); ++rodId) {
213 if (mActivePixelsRenderOutput[rodId].isActive()) {
214 mRenderOutputBufferTiled[rodId].clear(mRenderOutputBufferDefaultValue[rodId]);
215 mRenderOutputWeightBufferTiled[rodId].clear();
216 }
217 }

222void
223McrtFbSender::snapshotDelta(const rndr::RenderContext &renderContext,
224 const bool doPixelInfo, const bool doParallel, const uint32_t snapshotId,
225 std::function<bool(const std::string &bufferName)> checkOutputIntervalFunc,
226 const bool coarsePass)
227//
228// for ProgressiveFrame message
229//
230{
231 mSnapshotDeltaCoarsePass = coarsePass; // condition of coarse pass or not
232 mBeautyHDRITest = HdriTestCondition::INIT; // condition of HDRI test for beauty buffer
233
234 //
235 // Beauty
236 //
237 timeLogStart(snapshotId); // for performance analyze
238 renderContext.snapshotDelta(&mRenderBufferTiled, &mRenderBufferWeightBufferTiled, mActivePixels, doParallel);
239 mLatencyLog.enq(scene_rdl2::grid_util::LatencyItem::Key::SNAPSHOT_END_BEAUTY); // for performance analyze : finish snapshot
240
241 if (mActivePixelsArray) {
242 // record all activePixels info for analyzing purpose
243 mActivePixelsArray->set(mActivePixels, coarsePass); // record Beauty's activePixels info
244 }
245
246 //
247 // PixelInfo
248 //
249 if (doPixelInfo && mPixelInfoStatus) {
250 mLatencyLog.enq(scene_rdl2::grid_util::LatencyItem::Key::SNAPSHOT_START_PIXELINFO); // for performance analyze
251 renderContext.snapshotDeltaPixelInfo(&mPixelInfoBufferTiled,
252 &mPixelInfoWeightBufferTiled,
253 mActivePixelsPixelInfo,
254 doParallel);
255 mLatencyLog.enq(scene_rdl2::grid_util::LatencyItem::Key::SNAPSHOT_END_PIXELINFO); // for performance analyze
256 }
257
258 //
259 // HeatMap
260 //
261 if (mHeatMapStatus) {
262 if (checkOutputIntervalFunc(mRenderOutputName[mHeatMapId])) {
263 mLatencyLog.enq(scene_rdl2::grid_util::LatencyItem::Key::SNAPSHOT_START_HEATMAP); // for performance analyze
264 renderContext.snapshotDeltaHeatMap(&mHeatMapBufferTiled,
265 &mHeatMapWeightBufferTiled,
266 mActivePixelsHeatMap,
267 &mHeatMapSecBufferTiled,
268 doParallel);
269 mLatencyLog.enq(scene_rdl2::grid_util::LatencyItem::Key::SNAPSHOT_END_HEATMAP); // for performance analyze
270 mHeatMapSkipCondition = false;
271 } else {
272 mHeatMapSkipCondition = true;
273 }
274 }
275
276 //
277 // Weight buffer
278 //
279 if (mWeightBufferStatus) {
280 if (checkOutputIntervalFunc(mRenderOutputName[mWeightBufferId])) {
281 mLatencyLog.enq(scene_rdl2::grid_util::LatencyItem::Key::SNAPSHOT_START_WEIGHTBUFFER); // for performance analyze
282 renderContext.snapshotDeltaWeightBuffer(&mWeightBufferTiled, mActivePixelsWeightBuffer, doParallel);
283 mLatencyLog.enq(scene_rdl2::grid_util::LatencyItem::Key::SNAPSHOT_END_WEIGHTBUFFER); // for performance analyze
284 mWeightBufferSkipCondition = false;
285 } else {
286 mWeightBufferSkipCondition = true;
287 }
288 }
289
290 //
291 // BeautyOdd buffer
292 //
293 if (mRenderBufferOddStatus) {
294 if ((mBeautyAuxId > 0 && checkOutputIntervalFunc(mRenderOutputName[mBeautyAuxId])) ||
295 (mAlphaAuxId > 0 && checkOutputIntervalFunc(mRenderOutputName[mAlphaAuxId]))) {
296 mLatencyLog.enq(scene_rdl2::grid_util::LatencyItem::Key::SNAPSHOT_START_BEAUTYODD); // for performance analyze
297 renderContext.snapshotDeltaRenderBufferOdd(&mRenderBufferOddTiled,
298 &mRenderBufferOddWeightBufferTiled,
299 mActivePixelsRenderBufferOdd,
300 doParallel);
301 mLatencyLog.enq(scene_rdl2::grid_util::LatencyItem::Key::SNAPSHOT_END_BEAUTYODD); // for performance analyze
302 mRenderBufferOddSkipCondition = false;
303 } else {
304 mRenderBufferOddSkipCondition = true;
305 }
306 }
307
308 //
309 // RenderOutput
310 //
311 mDenoiserAlbedoInputNamePtr = nullptr;
312 mDenoiserNormalInputNamePtr = nullptr;
313 for (unsigned id = 0; id < mActivePixelsRenderOutput.size(); ++id) {
314 // We have to update buffer skip condition at every snapshotBuffer call.
315 const std::string &bufferName = mRenderOutputName[id];
316 bool skipCondition = ((checkOutputIntervalFunc(bufferName))? false: true);
317
318 setRenderOutputSkipCondition(id, skipCondition);
319 if (getRenderOutputSkipCondition(id)) {
320 // We don't need snapshot for
321 // 1) manually disabled buffer by rate control,
322 // 2) beautyRGB, beautyAlpha heatMap or weight buffer
323 continue; // skip this buffer
324 }
325
326 mLatencyLog.enq(scene_rdl2::grid_util::LatencyItem::Key::SNAPSHOT_START_RENDEROUTPUT);
327 bool denoiserAlbedoInput, denoiserNormalInput;
328 renderContext.snapshotDeltaRenderOutput(id,
329 &mRenderOutputBufferTiled[id],
330 &mRenderOutputWeightBufferTiled[id],
331 mActivePixelsRenderOutput[id],
332 doParallel,
333 denoiserAlbedoInput,
334 denoiserNormalInput);
335 if (denoiserAlbedoInput) mDenoiserAlbedoInputNamePtr = &bufferName;
336 if (denoiserNormalInput) mDenoiserNormalInputNamePtr = &bufferName;
337 mLatencyLog.enq(scene_rdl2::grid_util::LatencyItem::Key::SNAPSHOT_END_RENDEROUTPUT);
338 }
339}

367bool
368McrtFbSender::snapshotDeltaRecDump(const std::string &fileName)
369//
370// only used for debug / performance analyze purpose
371//
372{
373 if (!mActivePixelsArray) {
374 return false; // not snapshotDeltaRecStart() yet
375 }
376 if (mActivePixelsArray->isStart()) {
377 return false; // not stopped yet
378 }
379 if (!mActivePixelsArray->size()) {
380 return false; // data is empty
381 }
382
383 std::string data;
384 mActivePixelsArray->encode(data);
385
386 uint32_t machineId = mLatencyLog.getMachineId();
387 std::string outName = fileName + '.' + std::to_string(machineId);
388
389 std::ofstream fout(outName, std::ios::out | std::ios::binary | std::ios::trunc);
390 if (!fout) {
391 std::cerr << ">> McrtFbSender.cc snapshotDeltaRecDump() Can't open file:" << outName << std::endl;
392 return false;
393 }
394
395 fout.write((const char *)data.data(), data.size());
396 if (!fout) {
397 std::cerr << ">> McrtFbSender.cc snapshotDeltaRecDump() Can't write data."
398 << " file:" << outName << std::endl;
399 return false;
400 }
401
402 fout.close();
403
404 mActivePixelsArray = nullptr;
405
406 // Using std::cerr intentionally. This API is used only for debug/performance analyze purpose.
407 // And this API is called inside mcrt computations and using std::cerr always guarantees
408 // to bypass output logging system of arras_framework and minimise the delay to see the message.
409 std::cerr << ">> McrtFbSender.cc snapshotDeltaRecDump() done" << std::endl;
410
411 return true;
412}

624void
625McrtFbSender::addRenderOutputToProgressiveFrame(const bool directToClient,
626 MessageAddBuffFunc func)
627{
628 static const bool sha1HashSw = false;
629
630 for (size_t id = 0; id < mActivePixelsRenderOutput.size(); ++id) {
631 if (getRenderOutputSkipCondition(id)) {
632 //
633 // Need to skip standard output operation but still need to check bit more detail
634 // to support "reference" type buffer (i.e. "BeautyRGB", "BeautyAlpha" and "HeatMap")
635 //
636 if (!isRenderOutputDisable(id)) {
637 //
638 // Now this buffer is NOT manually disabled by rate control
639 //
640 scene_rdl2::grid_util::FbReferenceType referenceType = scene_rdl2::grid_util::FbReferenceType::UNDEF;
641 if (isRenderOutputBeauty(id)) {
642 referenceType = scene_rdl2::grid_util::FbReferenceType::BEAUTY;
643 } else if (isRenderOutputAlpha(id)) {
644 referenceType = scene_rdl2::grid_util::FbReferenceType::ALPHA;
645 } else if (isRenderOutputHeatMap(id)) {
646 referenceType = scene_rdl2::grid_util::FbReferenceType::HEAT_MAP;
647 } else if (isRenderOutputWeightBuffer(id)) {
648 referenceType = scene_rdl2::grid_util::FbReferenceType::WEIGHT;
649 } else if (isRenderOutputBeautyAux(id)) {
650 referenceType = scene_rdl2::grid_util::FbReferenceType::BEAUTY_AUX;
651 } else if (isRenderOutputAlphaAux(id)) {
652 referenceType = scene_rdl2::grid_util::FbReferenceType::ALPHA_AUX;
653 }
654
655 if (referenceType != scene_rdl2::grid_util::FbReferenceType::UNDEF) {
656 //
657 // AOV is reference type and need to output

1028void
1029McrtFbSender::initRenderOutputRegularAOV(const rndr::RenderOutputDriver *rod,
1030 const unsigned int roIdx,
1031 int &beautyId,
1032 int &alphaId,
1033 int &heatMapId,
1034 int &weightBufferId,
1035 int &beautyAuxId,
1036 int &alphaAuxId)
1037{
1038 static constexpr VariablePixelBuffer::Format fmtTbl[] = {
1039 VariablePixelBuffer::UNINITIALIZED,
1040 VariablePixelBuffer::FLOAT, // 1 channel
1041 VariablePixelBuffer::FLOAT2, // 2 channels
1042 VariablePixelBuffer::FLOAT3, // 3 channels
1043 VariablePixelBuffer::FLOAT4, // 4 channels
1044 };
1045
1046 // mRenderOutputSkipCondition[] bit mask : this is char(8bit) now but you can expand to ushort if you need.
1047 // |
1048 // 7 6 5 4 3 2 1 0
1049 // ^ ^ ^ ^ ^ ^ ^
1050 // | | | | | | +--- skip on/off by checkOutputInterval() test : update every snapshot call
1051 // | | | | | +----- buffer is Beauty RGB and skip regular AOV operation
1052 // | | | | +------- buffer is Alpha and skip regular AOV operation
1053 // | | | +--------- buffer is HeatMap and skip regular AOV operation
1054 // | | +----------- buffer is Weight and skip regular AOV operation
1055 // | +------------- buffer is BeautyAUX and skip regular AOV operation
1056 // +--------------- buffer is AlphaAUX and skip regular AOV operation
1057 //
1058 char skipST = 0x0;
1059 if (rod->requiresRenderBuffer(roIdx)) {
1060 if (rod->getRenderOutput(roIdx)->getResult() == scene_rdl2::rdl2::RenderOutput::RESULT_BEAUTY) {
1061 skipST = SKIP_CONDITION_BEAUTY_AOV;
1062 beautyId = roIdx;
1063 }
1064 if (rod->getRenderOutput(roIdx)->getResult() == scene_rdl2::rdl2::RenderOutput::RESULT_ALPHA) {
1065 skipST = SKIP_CONDITION_ALPHA_AOV;
1066 alphaId = roIdx;
1067 }
1068 } else if (rod->requiresHeatMap(roIdx)) {
1069 skipST = SKIP_CONDITION_HEATMAP_AOV;
1070 heatMapId = roIdx; // We save heatMap id for init mem and checkOutputInterval
1071 } else if (rod->requiresWeightBuffer(roIdx)) {
1072 skipST = SKIP_CONDITION_WEIGHT_AOV;
1073 weightBufferId = roIdx; // We save weight buffer id for init mem and checkOutputInterval
1074 } else if (rod->requiresRenderBufferOdd(roIdx)) {
1075 // This is a case of AOV about beauty_AUX/alpha_AUX data (moonray::rndr::Film::mRenderBufOdd)
1076 if (rod->getRenderOutput(roIdx)->getResult() == scene_rdl2::rdl2::RenderOutput::RESULT_BEAUTY_AUX) {
1077 skipST = SKIP_CONDITION_BEAUTYAUX_AOV;
1078 beautyAuxId = roIdx; // We save beautyAux id for init mem and checkOutputInterval
1079 }
1080 if (rod->getRenderOutput(roIdx)->getResult() == scene_rdl2::rdl2::RenderOutput::RESULT_ALPHA_AUX) {
1081 skipST = SKIP_CONDITION_ALPHAAUX_AOV;
1082 alphaAuxId = roIdx; // We save alphaAux id for init mem and checkOutputInterval
1083 }
1084 }
1085 mRenderOutputSkipCondition[roIdx] = skipST;
1086
1087 mRenderOutputBufferDefaultValue[roIdx] = rod->getAovDefaultValue(roIdx);
1088 if (skipST == SKIP_CONDITION_BEAUTYAUX_AOV || skipST == SKIP_CONDITION_ALPHAAUX_AOV) {
1089 mRenderOutputBufferScaledByWeight[roIdx] = true; // special case. We can not get this info from rod
1090 } else {
1091 mRenderOutputBufferScaledByWeight[roIdx] = static_cast<char>(rod->requiresScaledByWeight(roIdx));
1092 }
1093
1094 unsigned int numChans = rod->getNumberOfChannels(roIdx);
1095 mRenderOutputBufferOrigNumChan[roIdx] = numChans; // save original numChan here
1096 mRenderOutputBufferClosestFilterStatus[roIdx] = false;
1097 {
1098 auto isClosestFilter = [rod, roIdx]() {
1099 const pbr::AovSchema &schema = rod->getAovSchema();
1100 const int aovIdx = rod->getAovBuffer(roIdx);
1101 const bool result = aovIdx >= 0 && schema[aovIdx].filter() == pbr::AOV_FILTER_CLOSEST;
1102 return result;
1103 };
1104 if (isClosestFilter()) {
1105 MNRY_ASSERT(numChans <= 3);
1106 numChans = 4; // this AOV uses closestFilter, and uses FLOAT4 buffer
1107 mRenderOutputBufferClosestFilterStatus[roIdx] = true;
1108 } else {
1109 if (numChans == 0 || 3 < numChans) {
1110 skipST |= SKIP_CONDITION_MANUALLY_SKIP; // Wrong numChans, set manually skip condition
1111 }
1112 }
1113 }
1114
1115 if (skipST == 0x0) {
1116 unsigned width = mActivePixels.getWidth();
1117 unsigned height = mActivePixels.getHeight();
1118 unsigned tileAlignedWidth = mActivePixels.getAlignedWidth();
1119 unsigned tileAlignedHeight = mActivePixels.getAlignedHeight();
1120
1121 mActivePixelsRenderOutput[roIdx].init(width, height);
1122 mRenderOutputBufferTiled[roIdx].init(fmtTbl[numChans], tileAlignedWidth, tileAlignedHeight);
1123 mRenderOutputWeightBufferTiled[roIdx].init(tileAlignedWidth, tileAlignedHeight);
1124 } else {
1125 // skip this buffer
1126 mActivePixelsRenderOutput[roIdx].cleanUp();
1127 mRenderOutputBufferTiled[roIdx].cleanUp();
1128 mRenderOutputWeightBufferTiled[roIdx].cleanUp();
1129 }
1130
1131 mRenderOutputBufferCoarsePassPrecision[roIdx] =
1132 calcRenderOutputBufferCoarsePassPrecision(rod, static_cast<int>(roIdx));
1133 mRenderOutputBufferFinePassPrecision[roIdx] =
1134 calcRenderOutputBufferFinePassPrecision(rod, static_cast<int>(roIdx));
1135}

1137void
1138McrtFbSender::adjustRenderBufferFinePassPrecision(const rndr::RenderOutputDriver *rod,
1139 const int beautyId, const int alphaId,
1140 const int beautyAuxId, const int alphaAuxId)
1141//
1142// adjust renderBuffer's fine pass precision based on the channel-format (float/half) definition
1143//
1144{
1145 auto calcPrecision = [&](const rndr::RenderOutputDriver *rod,
1146 const int beautyId, const int alphaId,
1147 const int beautyAuxId, const int alphaAuxId) -> FinePassPrecision {
1148 //
1149 // This is a heuristic solution. We want to decide renderBuffer (RGB + A) fine pass precision
1150 // based on the renderOutput definition here.
1151 // We have 4 input beautyId (RGB), alphaId (A), beautyAuxId (RGB) and alphaAuxId (A).
1152 // We don't know all the information is available. (In most of the case we can get beautyId
1153 // and alphaId but it is not mandatory.)
1154 // Alpha buffer is unlikely to keep more than 1.0 value mostly and we should pick beauty buffer
1155 // condition to define renderBuffer fine pass precision if available.
1156 // This is why beautyId and beautyAuxId are checked before alphaId and alphaAuxId.
1157 // If we don't have all the input, precision is not changed by renderOutput definition.
1158 //
1159 FinePassPrecision precision = mRenderBufferFinePassPrecision;
1160 if (beautyId != -1) {
1161 precision = calcRenderOutputBufferFinePassPrecision(rod, beautyId);
1162 } else if (beautyAuxId != -1) {
1163 precision = calcRenderOutputBufferFinePassPrecision(rod, beautyAuxId);
1164 } else if (alphaId != -1) {
1165 precision = calcRenderOutputBufferFinePassPrecision(rod, alphaId);
1166 } else if (alphaAuxId != -1) {
1167 precision = calcRenderOutputBufferFinePassPrecision(rod, alphaAuxId);
1168 }
1169 return precision;
1170 };
1171
1172 mRenderBufferFinePassPrecision = calcPrecision(rod, beautyId, alphaId, beautyAuxId, alphaAuxId);
1173}
1174
1175McrtFbSender::FinePassPrecision
1176McrtFbSender::calcRenderOutputBufferFinePassPrecision(const rndr::RenderOutputDriver *rod,
1177 const int roIdx) const
1178//
1179// Define fine pass precision by channel-format definition (float/half) inside RenderOutput block.
1180//
1181{
1182 FinePassPrecision precision = FinePassPrecision::F32;
1183
1184 if (roIdx >= 0) {
1185 const scene_rdl2::rdl2::RenderOutput *ro = rod->getRenderOutput(roIdx);
1186 switch (ro->getChannelFormat()) {
1187 case scene_rdl2::rdl2::RenderOutput::ChannelFormat::CHANNEL_FORMAT_FLOAT : precision = FinePassPrecision::F32; break;
1188 case scene_rdl2::rdl2::RenderOutput::ChannelFormat::CHANNEL_FORMAT_HALF : precision = FinePassPrecision::H16; break;
1189 default : break;
1190 }
1191 }
1192 return precision;
1193}
1194
1195McrtFbSender::CoarsePassPrecision
1196McrtFbSender::calcRenderOutputBufferCoarsePassPrecision(const rndr::RenderOutputDriver *rod,
1197 const int roIdx) const
1198//
1199// This function compute best packTile precision mode for coarse pass encoding about each AOV data
1200//
1201{
1202 CoarsePassPrecision precision = CoarsePassPrecision::F32;
1203
1204 const scene_rdl2::rdl2::RenderOutput *ro = rod->getRenderOutput(roIdx);
1205 switch (ro->getResult()) {
1206 case scene_rdl2::rdl2::RenderOutput::RESULT_BEAUTY : precision = COARSE_PASS_PRECISION_BEAUTY; break;
1207 case scene_rdl2::rdl2::RenderOutput::RESULT_ALPHA : precision = COARSE_PASS_PRECISION_ALPHA; break;
1208 case scene_rdl2::rdl2::RenderOutput::RESULT_DEPTH : precision = CoarsePassPrecision::H16; break;
1209 case scene_rdl2::rdl2::RenderOutput::RESULT_STATE_VARIABLE : precision = CoarsePassPrecision::H16; break;
1210 case scene_rdl2::rdl2::RenderOutput::RESULT_PRIMITIVE_ATTRIBUTE :
1211 if (ro->getMathFilter() == scene_rdl2::rdl2::RenderOutput::MATH_FILTER_AVG ||
1212 ro->getMathFilter() == scene_rdl2::rdl2::RenderOutput::MATH_FILTER_SUM) {
1213 precision = CoarsePassPrecision::H16;
1214 } else {
1215 // This primitive attribute might be an integer base value.
1216 // If so we can not do any compression.
1217 precision = CoarsePassPrecision::F32;
1218 }
1219 break;
1220 case scene_rdl2::rdl2::RenderOutput::RESULT_HEAT_MAP :
1221 // Actually, we don't have precision control for Heat Map.
1222 // Heat Map is always encoded as H16 regardless of coarse pass or non coarse pass
1223 precision = CoarsePassPrecision::H16;
1224 break;
1225 case scene_rdl2::rdl2::RenderOutput::RESULT_WIREFRAME : precision = CoarsePassPrecision::UC8; break;

1245McrtFbSender::PackTilePrecision
1246McrtFbSender::calcPackTilePrecision(const CoarsePassPrecision coarsePassPrecision,
1247 const FinePassPrecision finePassPrecision,
1248 PackTilePrecisionCalcFunc runtimeDecisionFunc) const
1249//
1250// This function decides runtime encoding precision
1251//
1252{
1253 auto calcCoarsePassPrecision = [&]() -> PackTilePrecision {
1254 PackTilePrecision precision = PackTilePrecision::F32;
1255 switch (coarsePassPrecision) {
1256 case CoarsePassPrecision::F32 : precision = PackTilePrecision::F32; break;
1257 case CoarsePassPrecision::H16 : precision = PackTilePrecision::H16; break;
1258 case CoarsePassPrecision::UC8 : precision = PackTilePrecision::UC8; break;
1259 case CoarsePassPrecision::RUNTIME_DECISION :
1260 if (runtimeDecisionFunc) precision = runtimeDecisionFunc();
1261 break;
1262 default : break;
1263 }
1264 return precision;
1265 };
1266 auto calcFinePassPrecision = [&]() -> PackTilePrecision {
1267 PackTilePrecision precision = PackTilePrecision::F32;
1268 switch (finePassPrecision) {
1269 case FinePassPrecision::F32 : precision = PackTilePrecision::F32; break;
1270 case FinePassPrecision::H16 : precision = PackTilePrecision::H16; break;
1271 default : break;
1272 }
1273 return precision;
1274 };
1275
1276 PackTilePrecision precision = PackTilePrecision::F32;
1277
1278 switch (mPrecisionControl) {
1279 case PrecisionControl::FULL32 :
1280 // Always uses F32 for both of Coarse and Fine pass
1281 precision = PackTilePrecision::F32; // Always uses F32
1282 break;
1283
1284 case PrecisionControl::FULL16 :
1285 // Always uses H16 if possible for both of Coarse and Fine pass.
1286 // However, uses F32 if minimum precision is F32
1287 if (mSnapshotDeltaCoarsePass) {
1288 if (coarsePassPrecision == CoarsePassPrecision::F32) {
1289 precision = PackTilePrecision::F32; // This data is not able to use H16
1290 } else {
1291 precision = PackTilePrecision::H16; // Use H16 if possible
1292 }
1293 } else {
1294 if (finePassPrecision == FinePassPrecision::F32) {
1295 precision = PackTilePrecision::F32; // This data is not able to use H16
1296 } else {
1297 precision = PackTilePrecision::H16; // Use H16 if possible
1298 }
1299 }
1300 break;
1301
1302 case PrecisionControl::AUTO32 :
1303 // CoarsePass : Choose proper precision automatically based on the AOV data
1304 // FinePass : Always uses F32
1305 if (mSnapshotDeltaCoarsePass) {
1306 precision = calcCoarsePassPrecision(); // respect coarse pass precision decision
1307 } else {
1308 precision = PackTilePrecision::F32; // Always uses F32
1309 }
1310 break;
1311
1312 case PrecisionControl::AUTO16 :
1313 // CoarsePass : Choose proper precision automatically based on the AOV data
1314 // FinePass : Basically use H16. Only uses F32 if minimum precision is F32
1315 if (mSnapshotDeltaCoarsePass) {
1316 precision = calcCoarsePassPrecision(); // respect coarse pass precision decision
1317 } else {
1318 precision = calcFinePassPrecision(); // respect fine pass precision decision
1319 }
1320 break;
1321
1322 default : break;
1323 }
1324
1325 return precision;
1326}

1347bool
1348McrtFbSender::beautyHDRITest() const
1349//
1350// HDR pixel existence test for beauty buffer
1351//
1352{
1353 size_t area = mActivePixels.getAlignedWidth() * mActivePixels.getAlignedHeight();
1354 // We want to use minLimit in order to ignore small number of HDRI pixels like firefly
1355 // Use experimental number here. This cost was around 0.5ms~2ms range in my HD reso test scene.
1356 // Future enhancement related ticket is MOONRAY-3588
1357 size_t minLimit = (size_t)((float)area * 0.005f); // 0.5% of whole pixels
1358 const scene_rdl2::math::Vec4f *c = mRenderBufferTiled.getData();
1359 const float *w = mRenderBufferWeightBufferTiled.getData();
1360
1361 size_t totalHDRI = 0;
1362 for (size_t i = 0; i < area; ++i, ++c, ++w) {
1363 if (*w != 0.0f) {
1364 float scale = 1.0f / *w;
1365 if (c->x * scale > 1.0f || c->y * scale > 1.0f ||
1366 c->z * scale > 1.0f || c->w * scale > 1.0f) {
1367 if (totalHDRI > minLimit) {
1368 return true;
1369 }
1370 totalHDRI++;
1371 }
1372 }
1373 }
1374
1375 return false;
1376}

1378bool
1379McrtFbSender::renderOutputHDRITest(const ActivePixels &activePixels,
1380 const VariablePixelBuffer &buff,
1381 const FloatBuffer &weightBuff) const
1382//
1383// HDR pixel existence test for renderOutput AOV buffer.
1384// The idea is the same as beautyHDRITest() and we are not testing whole pixels.
1385//
1386{
1387 if (buff.getFormat() == scene_rdl2::fb_util::VariablePixelBuffer::Format::RGB888 ||
1388 buff.getFormat() == scene_rdl2::fb_util::VariablePixelBuffer::Format::RGBA8888) {
1389 return false; // uc8 based frame buffer
1390 }
1391 if (buff.getFormat() != scene_rdl2::fb_util::VariablePixelBuffer::Format::FLOAT &&
1392 buff.getFormat() != scene_rdl2::fb_util::VariablePixelBuffer::Format::FLOAT2 &&
1393 buff.getFormat() != scene_rdl2::fb_util::VariablePixelBuffer::Format::FLOAT3 &&
1394 buff.getFormat() != scene_rdl2::fb_util::VariablePixelBuffer::Format::FLOAT4) {
1395 return true; // We can not apply HDRI test here. return true as non 8bit range data
1396 }
1397
1398 size_t area = activePixels.getAlignedWidth() * activePixels.getAlignedHeight();
1399 size_t minLimit = static_cast<size_t>((float)area * 0.005f); // 0.5% of whole pixels
1400 unsigned pixByte = buff.getSizeOfPixel();
1401 size_t pixFloatCount = pixByte / sizeof(float);
1402 const float *p = reinterpret_cast<const float *>(buff.getData());
1403 const float *w = weightBuff.getData();
1404
1405 size_t totalHDRI = 0;
1406 for (size_t i = 0; i < area; ++i, ++w) {
1407 if (*w > 0.0f) {
1408 int currTotalHDRIchan = 0;
1409 for (size_t j = 0; j < pixFloatCount ; ++j) {
1410 if (p[j] > *w) currTotalHDRIchan++;
1411 }
1412 p += pixFloatCount;
1413 if (currTotalHDRIchan > 0) {
1414 if (totalHDRI > minLimit) {
1415 return true; // HDRI fb
1416 }
1417 totalHDRI++;
1418 }
1419 }
1420 }
1421 return false; // non HDRI fb
1422}
