55void
56CheckpointRayCostEstimator::push(float deltaSec, // Time (sec) used for the sampling operation
57 unsigned deltaSampleStartId, // tile sampleId
58 unsigned deltaSampleEndId) // tile sampleId
59{
60 // Store sampling event info to the top of the list. The front item is new and the back item is old.
61 mEventList.emplace_front
62 (std::make_shared<CheckpointRayCostEvalEvent>(deltaSec, deltaSampleStartId, deltaSampleEndId));
63
64 // We only keep up to some amount of history (i.e. we don't keep all of the histories).
65 while (oldestEventRemoveTest()) {
66 mEventList.pop_back();
67 }
68
69# ifdef DEBUG_MSG
70 std::cerr << ">> CheckpointController.cc CheckpointRayCostEstimator::push() " << show() << '\n';
71# endif // end DEBUG_MSG
72}

74float
75CheckpointRayCostEstimator::estimateRayCost() const
76//
77// Simply calculate average ray cost based on the history information
78//
79{
80 float totalSec = 0.0f;
81 unsigned totalSamples = 0;
82 for (auto itr = mEventList.begin(); itr != mEventList.end(); ++itr) {
83 totalSec += (*itr)->getDeltaSec();
84 totalSamples += (*itr)->getDeltaSamples();
85 }
86
87 if (totalSamples == 0) {
88 return 0.0f; // special case
89 }
90
91 return totalSec / static_cast<float>(totalSamples);
92}

115bool
116CheckpointRayCostEstimator::oldestEventRemoveTest()
117{
118 if (mEventList.size() <= 1) {
119 // We should try to keep at least 1 event regardless of its sample condition.
120 // If the event list size is 1 or less, we don't need to remove them.
121 return false; // not remove
122 }
123
124 // This number was defined in a heuristic way based on several test scenes.
125 // We would keep at least 4spp sampling results in the list in this case.
126 constexpr unsigned thresholdSamples = 64 * 4; // 4 SPP
127
128 unsigned totalSamples = calcDeltaSamplesTotal();
129 if (totalSamples <= thresholdSamples) {
130 return false; // keep all events if total samples are not enough
131 }
132
133 unsigned oldestSamples = mEventList.back()->getDeltaSamples();
134 if ((totalSamples - oldestSamples) < thresholdSamples) {
135 return false;
136 }
137 return true; // we should remove oldest event item
138}

193void
194CheckpointSnapshotEstimator::pushSnapshotCost(float snapshotSec)
195//
196// Stores single snapshot cost event into the history list and maintains history list size as reasonable.
197//
198{
199 if (mSnapshotIntervalSec > 0.0f) {
200 // snapshotInterval sec is defined and we don't need to tracking snapshot sec
201 return;
202 }
203
204 mEventList.emplace_front(snapshotSec);
205
206 constexpr size_t maxKeep = 10;
207 while (mEventList.size() > maxKeep) {
208 mEventList.pop_back();
209 }
210
211# ifdef DEBUG_MSG
212 std::cerr << ">> CheckpointController.cc snapshotEstimator::pushSnapshotCost() " << show() << '\n';
213# endif // end DEBUG_MSG

216float
217CheckpointSnapshotEstimator::estimateSnapshotInterval()
218//
219// Calculates the best guess of snapshot interval based on the history information.
220//
221{
222 if (mSnapshotIntervalSec > 0.0f) {
223 return mSnapshotIntervalSec; // snapshotInterval sec is defined. We don't need to estimate it.
224 }
225
226 float snapshotSec = estimateSnapshotSec();
227 if (snapshotSec == 0.0f || mSnapshotOverheadThresholdFraction == 0.0f) {
228 // We don't have any guess of the snapshot cost yet (This happens when history info is empty).
229 // Or threshold fraction is not defined (Safety logic to avoid divide by ZERO).
230 // This initial value was selected based on the 50AOVs with HD resolution scene.
231 // I hope this value is not so bad.
232 mEstimateIntervalSec = 30.0f;
233 } else {
234 // We have a cap to increase interval length from the previous value.
235 // This scaleMax value was heuristically defined based on several tests.
236 // We are concerned only if the interval is too long.
237 // Actually, we don't care about too short intervals because too short interval does not mak
238 // any serious issues to the micro checkpoint loop logic itself.
239 // (RenderDriver::checkpointRenderMicroStintLoop())
240 constexpr float scaleMax = 2.5f;
241
242 float intervalSec = snapshotSec / mSnapshotOverheadThresholdFraction;
243 mEstimateIntervalSec = std::min(mEstimateIntervalSec * scaleMax, intervalSec);
244 }
245 return mEstimateIntervalSec;
246}

265float
266CheckpointSnapshotEstimator::estimateSnapshotSec() const
267//
268// Return averaged snapshot time based on the history information
269//
270{
271 if (mEventList.size() == 0) {
272 return 0.0f; // special case
273 }
274
275 float totalSec = 0.0f;
276 for (auto itr = mEventList.begin(); itr != mEventList.end(); ++itr) {
277 totalSec += (*itr);
278 }
279 return totalSec / static_cast<float>(mEventList.size());
280}

313unsigned
314CheckpointController::estimateEndSampleId(const unsigned startSampleId, const unsigned endSampleId)
315//
316// Returns current best guess of next end tile sampleId for micro checkpoint loop based on the history.
317//
318{
319 auto sampleClip = [&](unsigned samples) -> unsigned {
320 //
321 // We have a cap to increase sample delta.
322 // The new sample total should be less than some scaled value of max sample deltas from the frame
323 // start. This scale value (i.e. maxScale) was heuristically defined based on the several test
324 // scenes.
325 // This idea is safety logic to avoid executing too many samples in the early stage of rendering.
326 // In most cases, the estimated sample delta is not accurate due to a lack of statistical information
327 // at the early stage of rendering. However, the estimated result would be enhanced time goes by and
328 // brush up to the pretty accurate value in the end.
329 //
330 constexpr float maxScale = 2.5f;
331 unsigned maxSamples =
332 static_cast<unsigned>(static_cast<float>(std::max(mMaxDeltaSamples, 1U)) * maxScale);
333 return scene_rdl2::math::clamp(samples, 1U, maxSamples);
334 };
335
336 float estimatedRayCost = mRayCostEstimator.estimateRayCost();
337 unsigned deltaSamples = 1;
338 if (estimatedRayCost > 0.0f) {
339 deltaSamples = sampleClip(mRemainingIntervalSec / estimatedRayCost);
340
341 // Tracking max delta samples value. This is important for sampleClip() operation
342 if (deltaSamples > mMaxDeltaSamples) mMaxDeltaSamples = deltaSamples;
343 }
344
345 mCurrDeltaSampleStartId = startSampleId;
346 mCurrDeltaSampleEndId = startSampleId + deltaSamples;
347 if (endSampleId < mCurrDeltaSampleEndId) mCurrDeltaSampleEndId = endSampleId;
348
349# ifdef DEBUG_MSG
350 std::cerr << ">> CheckpointController.cc estimateEndSampleId"
351 << " rayCost:" << estimatedRayCost
352 << " startId:" << mCurrDeltaSampleStartId
353 << " endId:" << mCurrDeltaSampleEndId
354 << " delta:" << (mCurrDeltaSampleEndId - mCurrDeltaSampleStartId)
355 << " maxDelta:" << mMaxDeltaSamples
356 << '\n';
357# endif // end DEBUG_MSG
358
359 return mCurrDeltaSampleEndId;
360}

368bool
369CheckpointController::microStintEnd()
370{
371 float deltaSec = mRayCostEvalTime.end();
372 mRayCostEstimator.push(deltaSec, mCurrDeltaSampleStartId, mCurrDeltaSampleEndId);
373
374 mRemainingIntervalSec -= deltaSec;
375 if (mRemainingIntervalSec <= 0.0f) {
376 mLastSnapshotIntervalSec = mSnapshotIntervalTime.end(); // for debug
377 return true; // we need snapshot
378 }
379 return false;

449//
450// CheckpointController file output main logic
451//
452// Implementation for writing logic is a bit complicated.
453// We have to consider 3 flags (scene_variables) condition.
454// They are checkpoint_bg_write, two_stage_output and checkpoint_overwrite scene_variable.
455// So, we have to think about 2 * 2 * 2 = 8 cases total.
456// Multi-version file specification does not have any impact on the output logic itself.
457// a multi-version file only affects the name of the output checkpoint filename.
458//
459// Following list is detailed information about flag condition, output files and output detail
460// procedure which includes write runMode (STD, ENQ, and DEQ) and finalize file (cp/rename).
461// We have total 8 types (A ~ H).
462// bg = checkpoint_bg_write scene_variable condition
463// two = two_stage_output scene_variable condition
464// overwrite = checkpoint_overwrite scene_variable condition.
465//
466// A) bg=off, two=off, overwrite=off
467// out = checkpoint + multi-ver-checkpoint
468// procedure : We need 2 STD actions.
469// STD-0 : standard checkpoint file
470// STD-1 : multi-version checkpoint file
471//
472// B) bg=off, two=off, overwrite=on
473// out = checkpoint (no multi-version checkpoint file)
474// procedure : We only need 1 STD action.
475// STD : standard checkpoint file
476//
477// C) bg=off, two=on, overwrite=off
478// out = checkpoint + multi-ver-checkpoint
479// procedure : We need 1 STD and 2 cp/rename actions.
480// STD : tmpFile output due to two stage output on
481// cp/rename-0 : finalize for standard checkpoint file
482// cp/rename-1 : finalize for multi-version checkpoint file
483//
484// D) bg=off, two=on, overwrite=on
485// out = checkpoint (no multi-version checkpoint file)
486// procedure : We need 1 STD and 1 cp/rename actions.
487// STD : tmpFile output due to two stage output on
488// cp/rename : finalize for standard checkpoint file
489//
490// E) bg=on, two=off, overwrite=off
491// out = checkpoint + multi-ver-checkpoint
492// procedure : We need 1 ENQ and 2 DEQ actions.
493// ENQ : standard ENQ action
494// DEQ-0 : output standard checkpoint file
495// DEQ-1 : output multi-version checkpoint file
496//
497// F) bg=on, two=off, overwrite=on
498// output = checkpoint (no multi-version checkpoint file)
499// procedure : We need 1 ENQ and 1 DEQ actions.
500// ENQ : standard ENQ action
501// DEQ : output for standard checkpoint file (multi-ver=off)
502//
503// G) bg=on, two=on, overwrite=off
504// out = checkpoint + multi-ver-checkpoint
505// procedure : We need 1 ENQ, 1 DEQ and 2 cp/rename actions.
506// ENQ : starndard ENQ action
507// DEQ : output to tmpFile due to two stage output
508// cp/rename-0 : finalize for standard checkpoint file
509// cp/rename-1 : finalize for multi-ver-checkpoint file
510//
511// H) bg=on, two=on, overwrite=on
512// out = checkpoint (no multi-version checkpoint file)
513// procedure : We need 1 ENQ, 1 DEQ and 1 cp/rename actions.
514// ENQ : standard ENQ action
515// DEQ : output to tmpFile (two stage output)
516// cp/rename : finalize for standard checkpoint file
517//
518void
519CheckpointController::fileOutputMain(bool checkpointBgWrite,
520 bool twoStageOutput,
521 bool snapshotOnly,
522 RenderContext *renderContext,
523 const std::string &checkpointPostScript,
524 const unsigned endSampleId)
525{
526 const RenderOutputDriver *renderOutputDriver = renderContext->getRenderOutputDriver();
527 if (!renderOutputDriver) return; // skip output. just in case
528
529 if (snapshotOnly) {
530 // just in case. we use checkpointBgWrite=true, twoStageOutput=true when
531 // snapshotOnly situation
532 checkpointBgWrite = true;
533 twoStageOutput = true;
534 }
535
536 // We can generate deep buffer checkpoint file data but not support deep buffer resume render yet.
537 const pbr::DeepBuffer *deepBuffer = renderContext->getDeepBuffer();
538 pbr::CryptomatteBuffer *cryptomatteBuffer = renderContext->getCryptomatteBuffer();
539 scene_rdl2::fb_util::HeatMapBuffer heatMapBuffer;
540 renderContext->snapshotHeatMapBuffer(&heatMapBuffer, true, true); // only do if it has data
541 scene_rdl2::fb_util::FloatBuffer weightBuffer;
542 renderContext->snapshotWeightBuffer(&weightBuffer, true, true); // only do if it has data
543 scene_rdl2::fb_util::RenderBuffer renderBufferOdd;
544 renderContext->snapshotRenderBufferOdd(&renderBufferOdd, true, true); // only do if it has data
545 std::vector<scene_rdl2::fb_util::VariablePixelBuffer> aovBuffers;
546 renderContext->snapshotAovBuffers(aovBuffers, true, true);
547 std::vector<scene_rdl2::fb_util::VariablePixelBuffer> displayFilterBuffers;
548 renderContext->snapshotDisplayFilterBuffers(displayFilterBuffers, true, true);
549
550 if (!snapshotOnly && checkpointBgWrite) {
551 // Non memorySnapshot only situation : conditional wait until we have enough bg cache memory capacity
552 ImageWriteDriver::get()->waitUntilBgWriteReady();
553 }
554
555 ImageWriteDriver::ImageWriteCacheUqPtr cache;
556 if (checkpointBgWrite ||
557 twoStageOutput ||
558 (!checkpointPostScript.empty())) {
559 // checkpoint file is written by background thread and run parallel with MCRT threads
560 // or post checkpoint script is set. Or two stage output mode is enable.
561 // In this case, We need ImageWriteCache object.
562 cache = ImageWriteDriver::get()->newImageWriteCache(renderOutputDriver, snapshotOnly);
563 cache->setPostCheckpointScript(checkpointPostScript);
564 cache->setTwoStageOutput(twoStageOutput);
565 if (checkpointBgWrite) {
566 cache->setupEnqMode(); // cache internal mode is ENQ
567 } else {
568 // cache internal mode is STD
569 }
570
571 } else {
572 // Stop all MCRT threads and checkpoint file write is exclusively executed.
573 // Also no post checkpoint script is specified. Also two stage output is disabled.
574 // This case cache is nullptr
575 MNRY_ASSERT(!cache);
576 }
577
578 renderOutputDriver->writeCheckpointEnq(false, // checkpointMultiVersion
579 deepBuffer,
580 cryptomatteBuffer,
581 &heatMapBuffer,
582 &weightBuffer,
583 &renderBufferOdd,
584 aovBuffers,
585 displayFilterBuffers,
586 endSampleId,
587 cache.get());
588 renderOutputDriver->loggingErrorAndInfo(cache.get());
589
590 if (cache) {
591 cache->calcFinalBlockInternalDataSize();
592 }
593
594 if (checkpointBgWrite) {
595 if (snapshotOnly) {
596 ImageWriteDriver::get()->updateSnapshotData(cache);
597 } else {
598 ImageWriteDriver::get()->resetSnapshotData(); // clear snapshot data
599 ImageWriteDriver::get()->enqImageWriteCache(cache);
600 }
601
602 } else {
603 if (!twoStageOutput) {
604 // We need to output checkpointMultiVersion data here when non-bgOutput mode
605 // with non-twoStage mode. If we are overwite=true case, following function
606 // return quickly without doing anything.
607 renderOutputDriver->writeCheckpointEnq(true, // checkpointMultiVersion
608 deepBuffer,
609 cryptomatteBuffer,
610 &heatMapBuffer,
611 &weightBuffer,
612 &renderBufferOdd,
613 aovBuffers,
614 displayFilterBuffers,
615 endSampleId,
616 nullptr);
617 renderOutputDriver->loggingErrorAndInfo(nullptr);
618 }
619
620 if (cache) {
621 // TwoStage=on
622 if (cache->getTwoStageOutput()) {
623 // copy and rename for checkpoint file
624 cache->allFinalizeCheckpointFile();
625 }
626 if (!cache->getCheckpointOverwrite()) {
627 // We need to output checkpoint multi-version files. Data already be written as tmpFile.
628 // copy and rename for checkpoint multi version file
629 cache->allFinalizeCheckpointMultiVersionFile();
630 }
631 if (cache->hasPostCheckpointScript()) cache->runPostCheckpointScript();
632
633 ImageWriteDriver::get()->setLastImageWriteCache(cache);
634 }
635 }
636}
