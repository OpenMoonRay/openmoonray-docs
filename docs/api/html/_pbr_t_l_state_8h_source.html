222//
223// Convenience function for iterating over all existing pbr TLS instances.
224//
225// Example of syntax:
226//
227// pbr::forEachTLS([&](pbr::TLState *tls)
228// {
229// tls->doWork();
230// });
231//
232template <typename Body>
233finline void forEachTLS(Body const &body)
234{
235 unsigned numTLS = mcrt_common::getNumTBBThreads();
236 mcrt_common::ThreadLocalState *tlsList = mcrt_common::getTLSList();
237 for (unsigned i = 0; i < numTLS; ++i) {
238 auto pbrTls = tlsList[i].mPbrTls.get();
239 if (pbrTls) {
240 body(pbrTls);
241 }
242 }
243}
