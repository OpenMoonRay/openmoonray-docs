33template <typename Generator, typename T>
34const T& uniformSelection(const std::vector<T>& v, Generator& g)
35{
36 typedef typename std::vector<T>::size_type size_type;
37 typedef typename std::uniform_int_distribution<size_type> dist_type;
38 typedef typename dist_type::param_type param_type;
39
40 static std::uniform_int_distribution<size_type> dist;
41 return v[dist(g, param_type(0, v.size() - 1u))];
42}

79inline std::vector<NPoint<DIMENSIONS>> generatePoissonDisk(uint32_t seed, uint32_t gridSize)
80{
81 const unsigned kCandidatePoints = 30u * DIMENSIONS;
82 const unsigned kNoSample = std::numeric_limits<unsigned>::max();
83
84 typedef NPoint<DIMENSIONS> Point;
85
86 // We want the cell size to be r/sqrt(n), where r is the radius and n is
87 // the dimensions. This allows, at most, one point per cell. We're
88 // specifying the number of cells, and not the radius, so we have to figure
89 // out r from this.
90 const float cellWidth = 1.0f / static_cast<float>(gridSize);
91 const float r = cellWidth * std::sqrt(static_cast<float>(DIMENSIONS));
92
93 std::mt19937 rng(seed);
94
95 typedef PerfectPowerArray<unsigned, DIMENSIONS> Grid;
96
97 Grid grid(gridSize, kNoSample);
98 std::vector<unsigned> activeList;
99 std::vector<Point> points;
100
101 ProgressBar progress(powu(gridSize, DIMENSIONS));
102
103 // Choose initial sample uniformly.
104 const Point initialPoint = generateRandomPoint<DIMENSIONS>(rng);
105 activeList.push_back(0);
106 points.push_back(initialPoint);
107
108 progress.update(); // For initial point.
109
110 // Insert index 0 into grid
111 addSampleToGrid(initialPoint, 0u, grid);
112
113 while (!activeList.empty()) {
114 // Choose uniformly from active list in [0, activeList.size())
115 const unsigned randIdx = uniformSelection(activeList, rng);
116 const Point& basisPoint = points[randIdx];
117
118 bool foundNewPoint = false;
119
120 // Generate candidate point
121 for (unsigned i = 0; i < kCandidatePoints && !foundNewPoint; ++i) {
122 const Point candidate = generateRandomPoint<DIMENSIONS>(basisPoint, r, rng);
123
124 // Check to see if candidate within 'r' of other points.
125 // We check ourselves and the neighbors on either side.
126 // 1D: 3 cells.
127 // 2D: 9 cells.
128 // 3D: 27 cells.
129 // etc.
130 constexpr unsigned nneighbors = powu(3, DIMENSIONS);
131 std::array<unsigned, nneighbors> otherPointIndices;
132 unsigned arrayIdx = 0;
133 auto f = [&otherPointIndices, &arrayIdx](unsigned idx) { otherPointIndices[arrayIdx++] = idx; };
134 const auto gridIdx = toGridLocation(candidate, gridSize);
135 grid.visitNeighbors(f, gridIdx.data());
136
137 bool valid = true;
138 for (unsigned idx = 0; idx < nneighbors; ++idx) {
139 const auto otherPointIdx = otherPointIndices[idx];
140 if (otherPointIdx != kNoSample) {
141 const Point& otherPoint = points[otherPointIdx];
142 if (wrappedDistanceSquared(candidate, otherPoint) < r*r) {
143 valid = false;
144 break;
145 }
146 }
147 }
148
149 if (valid) {
150 points.push_back(candidate);
151 const unsigned newindex = points.size() - 1u;
152 activeList.push_back(newindex);
153 addSampleToGrid(candidate, newindex, grid);
154 foundNewPoint = true;
155 progress.update();
156 progress.draw();
157 }
158 }
159
160 if (!foundNewPoint) {
161 // Remove i from activeList.
162 const auto it = std::find(activeList.begin(), activeList.end(), randIdx);
163 activeList.erase(it);
164 }
165 }
166 std::cout << '\n';
167
168 return points;
169}
