40 static constexpr Array8 iota8() noexcept

46 static constexpr Array8 permutationIdentity() noexcept
47 {
48 // This value, which is simply 0-7 rotated, results in the identity permutation when the factoradic number is
49 // zero. This is because of the way we do the Fisher-Yates shuffle: when zero, we still repeatedly swap with the
50 // last digit, resulting in a rotation.
51
52 // This identity value allows us to have lexicographical comparisons in line with the numbers used to generate
53 // them. E.g. a seed of 0 is the identity permutation. A seed of 1 is one permutation away in lexicographic
54 // ordering.
55
56 // This lexicographic order is important so that we can use a subset of the generated set of permutations.
57 // E.g. If we generate 8! permutations, we can access the first 5! values to get permutations of 5.
58
59 // 111 110 101 100 011 010 001 000 // We want this bit pattern [0-7]
60 // -> 110 101 100 011 010 001 000 111 // Rotated to account for the way we apply Fisher-Yates
61 return Array8(065432107ul);
62 }

64 std::uint32_t get(std::uint32_t idx) const noexcept
65 {
66 return (mData >> (idx * sBitsPerDigit)) & sMask;
67 }
68
69 std::uint32_t operator[](std::uint32_t idx) const noexcept
70 {
71 return get(idx);
72 }
73
74 void clear(std::uint32_t idx) noexcept
75 {
76 mData &= (~(sMask << (idx * sBitsPerDigit)));
77 }
78
79 void set(std::uint32_t idx, std::uint32_t val) noexcept
80 {
81 MNRY_ASSERT(val < 8);
82 MNRY_ASSERT(idx < 10);
83 clear(idx);
84 mData |= val << (idx * sBitsPerDigit);
85 }

92 void swap(std::uint32_t a, std::uint32_t b) noexcept
93 {
94 const auto t = get(a);
95 set(a, get(b));
96 set(b, t);
97 }

99private:
