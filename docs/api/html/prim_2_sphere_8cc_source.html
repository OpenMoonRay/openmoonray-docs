27Sphere::Sphere(float radius,
28 LayerAssignmentId&& layerAssignmentId,
29 PrimitiveAttributeTable&& primitiveAttributeTable)
30: NamedPrimitive(std::move(layerAssignmentId))
31, mL2P(scene_rdl2::math::one)
32, mP2L(scene_rdl2::math::one)
33, mRadius(radius)
34, mPhiMax(scene_rdl2::math::sTwoPi)
35, mZMin(-radius)
36, mZMax(radius)
37, mThetaMin(scene_rdl2::math::sPi)
38, mThetaMax(0.0f)
39, mIsSingleSided(false)
40, mIsNormalReversed(false)
41{
42 MNRY_ASSERT_REQUIRE(
43 mLayerAssignmentId.getType() == LayerAssignmentId::Type::CONSTANT);
44 // facevarying/varying/vertex are the same for Sphere:
45 // bilinear interpolation on the intersection uv across 4 specified values
46 for (auto& kv : primitiveAttributeTable) {
47 for (auto& attribute : kv.second) {
48 if (attribute->getRate() == RATE_VARYING ||
49 attribute->getRate() == RATE_FACE_VARYING) {
50 attribute->setRate(RATE_VERTEX);
51 }
52 }
53 }
54 setAttributes(Attributes::interleave(primitiveAttributeTable,
55 0, 1, 4, std::vector<size_t>(), 4));
56}

70void
71Sphere::postIntersect(mcrt_common::ThreadLocalState& tls,
72 const scene_rdl2::rdl2::Layer* pRdlLayer, const mcrt_common::Ray& ray,
73 Intersection& intersection) const
74{
75 intersection.setLayerAssignments(mLayerAssignmentId.getConstId(), pRdlLayer);
76
77 const scene_rdl2::rdl2::Material* material = intersection.getMaterial();
78 const AttributeTable *table =
79 material->get<shading::RootShader>().getPrimitiveAttributeTable();
80 intersection.setTable(&tls.mArena, table);
81 intersection.setIds(ray.primID, 0, 0);
82 overrideInstanceAttrs(ray, intersection);
83
84 // we store the local space intersect point in Ng during ray tracing time,
85 // which is the geometry normal in local space. (we need this local space
86 // information to compute st/dpds/dpdt here)
87 const Vec3f& pLocal = ray.Ng;
88 Vec3f Ng = normalize(scene_rdl2::math::transformNormal(mP2L, pLocal));
89 // calculate st, dpds, dpdt
90 float theta = scene_rdl2::math::acos(scene_rdl2::math::clamp(pLocal.z / mRadius, -1.0f, 1.0f));
91 Vec2f st(ray.u / mPhiMax, (theta - mThetaMin) / (mThetaMax - mThetaMin));
92 Vec3f dpds = scene_rdl2::math::transformVector(mL2P,
93 Vec3f(-mPhiMax * pLocal.y, mPhiMax * pLocal.x, 0.0f));
94 float invZRadius = 1.0f / sqrtf(pLocal.x * pLocal.x + pLocal.y * pLocal.y);
95 float cosPhi = pLocal.x * invZRadius;
96 float sinPhi = pLocal.y * invZRadius;
97 Vec3f dpdt = scene_rdl2::math::transformVector(mL2P, (mThetaMax - mThetaMin) *
98 Vec3f(pLocal.z * cosPhi, pLocal.z * sinPhi, -mRadius * sinf(theta)));
99 if (mIsNormalReversed) {
100 Ng = -Ng;
101 }
102 intersection.setDifferentialGeometry(Ng, Ng, st, dpds, dpdt, true);
103
104 // interpolate primitive attributes
105 QuadricInterpolator interpolator(getAttributes(), ray.time, st[0], st[1]);
106 intersection.setRequiredAttributes(interpolator);
107
108 const scene_rdl2::rdl2::Geometry* geometry = intersection.getGeometryObject();
109 MNRY_ASSERT(geometry != nullptr);
110 intersection.setEpsilonHint( geometry->getRayEpsilon() );
111
112 // wireframe AOV is blank
113 if (table->requests(StandardAttributes::sNumPolyVertices)) {
114 intersection.setAttribute(StandardAttributes::sNumPolyVertices, 0);
115 intersection.setAttribute(StandardAttributes::sPolyVertexType,
116 static_cast<int>(StandardAttributes::POLYVERTEX_TYPE_POLYGON));
117 }
118
119 // motion vectors
120 if (table->requests(StandardAttributes::sMotion)) {
121 // sphere has no motion slices, primitive motion can only come from instancing
122 Vec3f motion(0.0f);
123 if (ray.isInstanceHit()) {
124 // Motion vectors only support a single instance level, hence we only care
125 // about ray.instance0.
126 motion = computePrimitiveMotion(intersection.getP(), nullptr, ray.getTime(),
127 static_cast<const Instance *>(ray.ext.instance0OrLight));
128 }
129 intersection.setAttribute(StandardAttributes::sMotion, motion);
130 }
131}

133bool
134Sphere::computeIntersectCurvature(const mcrt_common::Ray &ray,
135 const Intersection &intersection,
136 Vec3f &dnds, Vec3f &dndt) const
137{
138 // p(s, t) = n(s, t) for local space unit sphere so reuse dpds/dpdt
139 // with radius scaling should be sufficient enough for sphere case
140 float invRadius = 1.0f / mRadius;
141 dnds = intersection.getdPds() * invRadius;
142 dndt = intersection.getdPdt() * invRadius;
143 return true;
144}

175static void
176intersectFunc(const RTCIntersectFunctionNArguments* args)
177{
178 int* valid = (int*)args->valid;
179 unsigned int N = args->N;
180 RTCRayHitN* rayhit = (RTCRayHitN*)args->rayhit;
181 RTCRayN* rays = RTCRayHitN_RayN(rayhit, N);
182 RTCHitN* hits = RTCRayHitN_HitN(rayhit, N);
183 const BVHUserData* userData = (const BVHUserData*)args->geometryUserPtr;
184 const Sphere* sphere = (const Sphere*)userData->mPrimitive;
185 // get sidedness and reverse normals
186 bool isSingleSided = sphere->getIsSingleSided();
187 bool isNormalReversed = sphere->getIsNormalReversed();
188 float radius = sphere->getRadius();
189 const Mat43& P2L = sphere->getP2L();
190 for (unsigned int index = 0; index < N; ++index) {
191 if (valid[index] == 0) {
192 continue;
193 }
194 // transform ray to object space
195 Vec3f org = scene_rdl2::math::transformPoint(P2L, Vec3f(
196 RTCRayN_org_x(rays, N, index),
197 RTCRayN_org_y(rays, N, index),
198 RTCRayN_org_z(rays, N, index)));
199 Vec3f dir = scene_rdl2::math::transformVector(P2L, Vec3f(
200 RTCRayN_dir_x(rays, N, index),
201 RTCRayN_dir_y(rays, N, index),
202 RTCRayN_dir_z(rays, N, index)));
203 float& rayTnear = RTCRayN_tnear(rays, N, index);
204 float& rayTfar = RTCRayN_tfar(rays, N, index);
205 // compute quadratic sphere coefficients
206 const float A = dot(dir, dir);
207 const float B = 2.0f * dot(org, dir);
208 const float C = dot(org, org) - scene_rdl2::math::sqr(radius);
209 // solve quadratic equation for t values
210 const float D = B * B - 4.0f * A * C;
211 if (D < 0.0f) {
212 continue;
213 }
214 const float rootDiscrim = scene_rdl2::math::sqrt(D);
215 float q = B < 0.0f ?
216 -0.5f * (B - rootDiscrim) : -0.5f * (B + rootDiscrim);
217 float t0 = q / A;
218 float t1 = C / q;
219 if (t0 > t1) {
220 std::swap(t0, t1);
221 }
222 // compute intersection distance along ray
223 if (t0 > rayTfar || t1 < rayTnear) {
224 continue;
225 }
226 float tHit = t0;
227 if (t0 < rayTnear || (isSingleSided && isNormalReversed)) {
228 tHit = t1;
229 if (tHit > rayTfar || (isSingleSided && !isNormalReversed)) {
230 continue;
231 }
232 }
233 // compute hit position and phi
234 Vec3f pHit = org + tHit * dir;
235 if (pHit.x == 0.0f && pHit.y == 0.0f) {
236 pHit.x = 1e-5f * radius;
237 }
238 float phi = atan2f(pHit.y, pHit.x);
239 if (phi < 0.0f) {
240 phi += scene_rdl2::math::sTwoPi;
241 }
242 float zMin = sphere->getZMin();
243 float zMax = sphere->getZMax();
244 float phiMax = sphere->getPhiMax();
245 // test sphere intersection against clipping parameters
246 if (pHit.z < zMin || pHit.z > zMax || phi > phiMax) {
247 if (tHit == t1) {
248 continue;
249 }
250 if (t1 > rayTfar || (isSingleSided && !isNormalReversed)) {
251 continue;
252 }
253 tHit = t1;
254 pHit = org + tHit * dir;
255 if (pHit.x == 0.0f && pHit.y == 0.0f) {
256 pHit.x = 1e-5f * radius;
257 }
258 phi = atan2f(pHit.y, pHit.x);
259 if (phi < 0.0f) {
260 phi += scene_rdl2::math::sTwoPi;
261 }
262 if (pHit.z < zMin || pHit.z > zMax || phi > phiMax) {
263 continue;
264 }
265 }
266 // TODO call intersect filter functions
267 rayTfar = tHit;
268 RTCHitN_instID(hits, N, index, 0) = args->context->instID[0];
269 RTCHitN_geomID(hits, N, index) = sphere->getGeomID();
270 RTCHitN_primID(hits, N, index) = 0;
271 // local space normal will be transformed
272 // to parent space at postIntersect
273 RTCHitN_Ng_x(hits, N, index) = pHit.x;
274 RTCHitN_Ng_y(hits, N, index) = pHit.y;
275 RTCHitN_Ng_z(hits, N, index) = pHit.z;
276 // v will be computed at postIntersect
277 RTCHitN_u(hits, N, index) = phi;
278 }
