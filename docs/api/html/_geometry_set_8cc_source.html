49void
50GeometrySet::add(Geometry* geometry)
51{
52 if (!mUpdateActive) {
53 std::stringstream errMsg;
54 errMsg << "Geometry '" << geometry->getName() << "' can only be added"
55 " to GeometrySet '" << mName << "' between beginUpdate() and"
56 " endUpdate() calls.";
57 throw except::RuntimeError(errMsg.str());
58 }
59
60 // Retrieve a mutable reference to the geometries attribute.
61 SceneObjectIndexable& geometries = getMutable(sGeometriesKey);
62
63 // Search for the geometry.
64 const auto iters = geometries.equal_range(const_cast<Geometry*>(geometry));
65
66 // If it's already in the set, we're done
67 if (iters.first != iters.second) {
68 return;
69 }
70
71 // Otherwise, append geometry.
72 geometries.push_back(geometry);
73
74 // When a geometry is added to a geometry set in a delta file,
75 // we must make a request to update it.
76 geometry->requestUpdate();
77
78 // Manually turn on the set flag and dirty flag since we didn't go through
79 // the set() method.
80 mAttributeUpdateMask.set(sGeometriesKey.mIndex, true);
81 mAttributeSetMask.set(sGeometriesKey.mIndex, true);
82 mDirty = true;
83}

85void
86GeometrySet::remove(Geometry* geometry)
87{
88 if (!mUpdateActive) {
89 std::stringstream errMsg;
90 errMsg << "Geometry '" << geometry->getName() << "' can only be removed"
91 " from GeometrySet '" << mName << "' between beginUpdate() and"
92 " endUpdate() calls.";
93 throw except::RuntimeError(errMsg.str());
94 }
95
96 // Retrieve a mutable reference to the geometries attribute.
97 SceneObjectIndexable& geometries = getMutable(sGeometriesKey);
98
99 // Search for the geometry.
100 const auto iters = geometries.equal_range(const_cast<Geometry*>(geometry));
101
102 // If found, remove it.
103 if (iters.first != iters.second) {
104 geometries.erase(geometries.begin() + *iters.first);
105
106 // Manually turn on the set flag and dirty flag since we didn't go
107 // through the set() method.
108 mAttributeUpdateMask.set(sGeometriesKey.mIndex, true);
109 mAttributeSetMask.set(sGeometriesKey.mIndex, true);
110 mDirty = true;
111 }

146bool
147GeometrySet::isStatic() const
148{
149 // Retrieve a const reference to the geometries attribute.
150 const SceneObjectIndexable& geometries = get(sGeometriesKey);
151
152 for (auto iter = geometries.begin(); iter != geometries.end(); ++iter) {
153 if (!(*iter)->asA<Geometry>()->isStatic()) {
154 return false;
155 }
156 }
157
158 return true;

161bool
162GeometrySet::updatePrepFast(UpdateHelper& sceneObjects, int depth)
163{
164 MNRY_ASSERT_REQUIRE(!mUpdateActive);
165
166 if (mUpdatePrepApplied &&
167 (sceneObjects.getDepth(this) >=depth || sceneObjects.isLeaf(this))) {
168 return updateRequired();
169 }
170 mUpdatePrepApplied = true;
171
172 const SceneObjectIndexable& geometries = get(sGeometriesKey);
173 bool attributeTreeChanged = false;
174 bool bindingTreeChanged = false;
175 // check for any geometry in this set has been updated.
176 for (auto iter = geometries.begin(); iter != geometries.end(); ++iter) {
177 Geometry* geom = (*iter)->asA<Geometry>();
178 if (geom->attributeTreeChanged()) {
179 attributeTreeChanged = true;
180 }
181 if (geom->bindingTreeChanged()) {
182 bindingTreeChanged = true;
183 }
184 // terminate early if both are true
185 if (attributeTreeChanged && bindingTreeChanged) {
186 break;
187 }
188 }
189 mAttributeTreeChanged = attributeTreeChanged || mAttributeUpdateMask.any();
190 mBindingTreeChanged = bindingTreeChanged || mBindingUpdateMask.any();
191
192 if (updateRequired()) {
193 sceneObjects.insert(this, depth);
194 }
195 return updateRequired();
196}
