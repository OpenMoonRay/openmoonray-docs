81// Used to signify that we should wrap up rendering this frame ASAP.
82// This flag used to live in RenderDriver. It doesn't anymore for two reasons:
83// * We want atomic behavior. While it's possible to do atomic operations in ISPC, doing them on the same data structure
84// would require a lot of effort (i.e. we would have to hand-roll a lot of the atomic operations and pass around a
85// normal bool). Now we just access the boolean from an extern "C" function and let C++ worry about the atomacity.
86// * We want to align this on a cache line. False sharing caused a simple render to go from 25 seconds to more than a
87// minute and a half. Unfortunately, ICC compilation results in an error when aligning this member variable, saying
88// that our dynamic allocation of RenderDriver is mis-aligned. I believe this is an ICC bug, as an aligned member
89// variable should only change the size of RenderDriver, and not the allocation alignment requirements of RenderDriver
90// itself (we're not changing the alignment of RenderDriver, only the size).
91struct alignas(64) CancelFlag
92{
93 CancelFlag() noexcept
94 : mCanceled(false)
95 {
96 }
97
98 bool isCanceled() const noexcept
99 {
100 return mCanceled.load(std::memory_order_relaxed);
101 }
102
103 void set(bool v) noexcept
104 {
105 mCanceled.store(v, std::memory_order_relaxed);
106 }
107
108 std::atomic_bool mCanceled;
109};

150 //
151 // Here are the steps which happen internally to produce a linear render
152 // buffer:
153 //
154 // 1) If we're still render coarse passes then we need to extrapolate the
155 // buffer to fill in the gaps.
156 // 2) Radiance in rgb and alpha is a is normalized by dividing by the

417 //
418 // The state machine mechanism for recording debug rays is dependent on
419 // there only being a single thread which has permission to change to a
420 // particular state. The thread permissions are shown in the comments in
421 // the DebugRayState enum. As long as we follow this one rule, we don't
422 // need a mutex around the state. For added verification this is true, the
423 // switchDebugRayState function requires the client to pass in what it
424 // expects the current state to be. By doing this we can check that no other
425 // threads have changed the value in the meantime.

912template<typename DST_PIXEL_TYPE, typename SRC_PIXEL_TYPE, typename HAS_DATA_FUNC,
913 typename PIXEL_XFORM_FUNC>
914inline void
915RenderDriver::snapshotBuffer(scene_rdl2::fb_util::PixelBuffer<DST_PIXEL_TYPE> *outputBuffer,
916 const scene_rdl2::fb_util::PixelBuffer<SRC_PIXEL_TYPE> &srcBuffer,
917 scene_rdl2::fb_util::PixelBuffer<SRC_PIXEL_TYPE> *scratchBuffer,
918 bool extrapolate,
919 bool untile,
920 bool parallel,
921 HAS_DATA_FUNC hasData,
922 PIXEL_XFORM_FUNC pixelXform) const
923{
924 scene_rdl2::fb_util::Tiler tiler(mUnalignedW, mUnalignedH);
925
926 if (!scratchBuffer) {
927 if (sizeof(SRC_PIXEL_TYPE) <= sizeof(scene_rdl2::fb_util::RenderBuffer::PixelType)) {
928 scratchBuffer = (scene_rdl2::fb_util::PixelBuffer<SRC_PIXEL_TYPE> *)&mExtrapolationBuffer;
929 } else {
930 // If this triggers you either didn't pass in a scratch buffer or
931 // the mExtrapolationBuffer can't be used since the size of its pixels
932 // aren't large enough to accommodate the request.
933 MNRY_ASSERT_REQUIRE(0);
934 }
935 }
936
937 scratchBuffer->init(tiler.mAlignedW, tiler.mAlignedH);
938 if (untile) {
939 outputBuffer->init(mUnalignedW, mUnalignedH);
940 } else {
941 outputBuffer->init(tiler.mAlignedW, tiler.mAlignedH);
942 }
943
944 // Check if we still require an extrapolation pass.
945 if (extrapolate) {
946
947 const bool viewportActive = (mCachedViewport.mMinX != 0 ||
948 mCachedViewport.mMaxX != mUnalignedW ||
949 mCachedViewport.mMinY != 0 ||
950 mCachedViewport.mMaxY != mUnalignedH);
951
952 const bool distributed = mTileScheduler->isDistributed();
953
954 // If we have a viewport set which is smaller than the render buffer
955 // then clear the extrapolation buffer beforehand since not all of it
956 // may get written to in the ExtrapolateBuffer call.
957 if (viewportActive || distributed) {
958
959 scratchBuffer->clear();
960
961 if (distributed) {
962 // Only extrapolate tiles in the list.
963 rndr::extrapolateBufferWithTileList(scratchBuffer, srcBuffer,
964 tiler,
965 mTileScheduler->getTiles(),
966 hasData,
967 parallel);
968 } else {
969 MNRY_ASSERT(viewportActive);
970
971 // Synthesize each tile and clip them to the viewport.
972 rndr::extrapolateBufferWithViewport(scratchBuffer, srcBuffer, tiler,
973 mCachedViewport, hasData,
974 parallel);
975 }
976 } else {
977 // Fast path.
978 rndr::extrapolateBufferFastPath(scratchBuffer, srcBuffer, tiler,
979 hasData, parallel);
980 }
981 }
982
983 // Write final result directly to outputBuffer.
984 const auto *srcBuf = extrapolate ? scratchBuffer : &srcBuffer;
985
986 if (untile) {
987 scene_rdl2::fb_util::untile(outputBuffer, *srcBuf, tiler, parallel, pixelXform);
988 } else {
989 unsigned area = mUnalignedW * mUnalignedH;
990 auto *__restrict dstRow = outputBuffer->getData();
991 const auto *__restrict srcRow = srcBuf->getData();
992 //
993 // This block crashes in a opt build, see comment in Film.h untile()
994 //
995#if 0
996 simpleLoop(parallel, 0u, area, [&](unsigned i) {
997 dstRow[i] = pixelXform(srcRow[i], i);
998 });
999#else
1000 for (unsigned i = 0; i < area; ++i) {
1001 dstRow[i] = pixelXform(srcRow[i], i);
1002 }
1003#endif
1004 }
1005}
