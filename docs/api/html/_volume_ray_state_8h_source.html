22/// This class stores all the volume intervals along a ray during the
23/// BVH traversal stage. Each thread owns "one" VolumeRayState so
24/// the data it stores is overwritten each time we do another ray intersection
25/// test in this thread
26class VolumeRayState
27{
28public:
29 static constexpr int ORIGIN_VOLUME_INIT = -2; // We don't have volume along the ray
30 static constexpr int ORIGIN_VOLUME_EMPTY = -1; // We have volume but ray origin is outside volume
31
32 VolumeRayState() : mIntervalCount(0), mVolumeAssignmentTable(nullptr) {}
33
34 finline void initializeVolumeLookup(
35 const VolumeAssignmentTable* volumeAssignmentTable) {
36 mVolumeAssignmentTable = volumeAssignmentTable;
37 mVolumeSampleInfo.resize(mVolumeAssignmentTable->getVolumeCount());
38 }
39
40 finline void resetState(float tEnd, bool estimateInScatter) {
41 mIntervalCount = 0;
42 mVolumeRegions.reset();
43 mVisitedVolumes.reset();
44 mTEnd = tEnd;
45 mOriginVolumeId = ORIGIN_VOLUME_INIT;
46 mEstimateInScatter = estimateInScatter;
47 }
48
49 finline void addInterval(float t, int volumeId, bool isEntry, float *tRenderSpace = nullptr) {
50 MNRY_ASSERT(mIntervalCount < sMaxIntervalCount,
51 "volume intersections along the ray exceeds "
52 "thread local storage capacity");
53 mVolumeIntervals[mIntervalCount++] =
54 VolumeTransition(t, volumeId, isEntry, tRenderSpace);
55 }

115 finline int getVolumeId(int assignmentId, int instanceState) const {
116 // instancing case
117 const VolumeIdFSM &fsm = mVolumeAssignmentTable->getInstanceVolumeIds();
118 if (instanceState > 0 && fsm.isLeaf(instanceState)) {
119 const int vId = fsm.getVolumeId(instanceState);
120 MNRY_ASSERT(vId >= 0);
121 return vId;
122 }
123
124 // non-instancing
125 const std::vector<int> &vIds = mVolumeAssignmentTable->getVolumeIds(assignmentId);
126 MNRY_ASSERT(vIds.size() == 1);
127 return vIds[0];
128 }
129
130 finline int getAssignmentId(int volumeId) const {
131 return mVolumeAssignmentTable->getAssignmentId(volumeId);
132 }
133
134 finline float getTEnd() const { return mTEnd; }
135
136
137 inline void setOriginVolumeId(int id) { mOriginVolumeId = id; }
138 inline int getOriginVolumeId() const { return mOriginVolumeId; }
139
140 inline bool getEstimateInScatter() const { return mEstimateInScatter; }
141
142private:
143 size_t mIntervalCount;
144 const VolumeAssignmentTable* mVolumeAssignmentTable;
145 VolumeRegions mVolumeRegions;
146 VolumeRegions mVisitedVolumes;
147 std::vector<VolumeSampleInfo> mVolumeSampleInfo;
148 float mTEnd;
149 static constexpr size_t sMaxIntervalCount = 4096;
150 std::array<VolumeTransition, sMaxIntervalCount> mVolumeIntervals;
151
152 // volumeId of ray origin position. 3 possible conditions.
153 // ORIGIN_VOLUME_INIT : initial condition. origin volume is not computed yet.
154 // ORIGIN_VOLUME_EMPTY : no volume at ray origin
155 // 0 or positive value : volumeId at ray origi position.
156 int mOriginVolumeId;
157
158 // true : This is a case when this class is used for PathIntegrator::estimateInScatteringSourceTerm().
159 // i.e. light transmittance computation for volume.
160 // false : all other situations.
161 bool mEstimateInScatter;
162};
