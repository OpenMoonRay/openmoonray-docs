29static finline float
30totalDiffuseReflectance(float alphaP, float A)
31{
32 const float e = scene_rdl2::math::sqrt(3.0f * (1.0f - alphaP));
33 return 0.5f * alphaP * (1.0f + scene_rdl2::math::exp(-4.0f / 3.0f * A * e)) * scene_rdl2::math::exp(-e);
34}

63DipoleBssrdf::DipoleBssrdf(const Vec3f &N,
64 float eta,
65 const Color &sigmaA,
66 const Color &sigmaSP,
67 float sceneScale,
68 const scene_rdl2::rdl2::Material* material,
69 const scene_rdl2::rdl2::EvalNormalFunc evalNormalFn) :
70 Bssrdf(N, PROPERTY_COLOR | PROPERTY_RADIUS, material, evalNormalFn),
71 mSigmaA(mmInvToWorldInv(sigmaA, sceneScale)),
72 mSigmaSP(mmInvToWorldInv(sigmaSP, sceneScale)),
73 mAlbedo(scene_rdl2::math::sBlack)
74{
75 float fdr = diffuseFresnelReflectance(eta);
76 mA = (1.0f + fdr) / (1.0f - fdr);
77
78 mSigmaTP = mSigmaA + mSigmaSP;
79 mSigmaTr = scene_rdl2::math::sqrt(3.0f * mSigmaA * mSigmaTP);
80 mAlphaP = mSigmaSP / mSigmaTP;
81
82 // Keep a note of the total diffuse reflectance
83 mDiffuseReflectance = Color(totalDiffuseReflectance(mAlphaP[0], mA),
84 totalDiffuseReflectance(mAlphaP[1], mA),
85 totalDiffuseReflectance(mAlphaP[2], mA));
86 finishInit();
87}

89DipoleBssrdf::DipoleBssrdf(const Vec3f &N,
90 float eta,
91 const Color &translucentColor,
92 const Color &radius,
93 const scene_rdl2::rdl2::Material* material,
94 const scene_rdl2::rdl2::EvalNormalFunc evalNormalFn) :
95 Bssrdf(N, PROPERTY_COLOR | PROPERTY_RADIUS | PROPERTY_PBR_VALIDITY, material, evalNormalFn),
96 mAlbedo(translucentColor)
97{
98 float fdr = diffuseFresnelReflectance(eta);
99 mA = (1.0f + fdr) / (1.0f - fdr);
100
101 // Convert from artist controls to physical controls as described in [3]
102 Color r = max(radius, Color(0.001f, 0.001f, 0.001f));
103 mSigmaTr = rcp(r);
104 for (int c = 0; c < 3; ++c) {
105 float reflectance = max(translucentColor[c], 0.0f);
106 reflectance = min(reflectance, 1.0f);
107 mAlphaP[c] = findAlphaP(reflectance, mA);
108 }
109
110 mDiffuseReflectance = Color(totalDiffuseReflectance(mAlphaP[0], mA),
111 totalDiffuseReflectance(mAlphaP[1], mA),
112 totalDiffuseReflectance(mAlphaP[2], mA));
113
114 mSigmaTP = mSigmaTr / scene_rdl2::math::sqrt(3.0f * (sWhite - mAlphaP));
115 mSigmaSP = mAlphaP * mSigmaTP;
116 mSigmaA = mSigmaTP - mSigmaSP;
117
118 finishInit();
119}

138void
139DipoleBssrdf::finishInit()
140{
141 mZr = rcp(mSigmaTP);
142 mZv = mZr * (1.0f + (4.0f / 3.0f) * mA);
143
144 // Evaluate Rd at the mean free path radius
145 Color mfp = getDiffuseMeanFreePath();
146 float maxMfp = max(mfp[0], mfp[1]);
147 maxMfp = max(maxMfp, mfp[2]);
148
149 // At 3x the mean free path Rd is very small.
150 mMaxRadius = 3.0f * maxMfp;
151
152#if !BSSRDF_SAMPLE_UNIFORM
153 // Compute sampling ratio and pdf normalization terms
154 float normalize = 0.0f;
155 for (int c=0; c < 3; c++) {
156 float sigmaTr = mSigmaTr[c];
157 float expTermR = scene_rdl2::math::exp(-sigmaTr * mZr[c]);
158 float expTermV = scene_rdl2::math::exp(-sigmaTr * mZv[c]);
159 float sumExpTerms = expTermR + expTermV;
160 float reflectance = mAlphaP[c] * 0.5f * sumExpTerms;
161 mRatio[c] = expTermR / sumExpTerms;
162 mPdfNormFactor[c] = 1.0f / reflectance;
163 mChannelCdf[c] = max(reflectance, sEpsilon);
164 normalize += mChannelCdf[c];
165 }
166
167#if BSSRDF_SAMPLE_WIDEST_TAIL
168 // The highest Rd value at maxMfp has the widest tail
169 Color rd = eval(maxMfp);
170 normalize = 1.0f;
171 if (rd[0] >= rd[1]) {
172 if (rd[0] >= rd[2]) {
173 mChannelCdf = Color(1.0f, 0.0f, 0.0f);
174 } else {
175 mChannelCdf = Color(0.0f, 0.0f, 1.0f);
176 }
177 } else {
178 if (rd[1] >= rd[2]) {
179 mChannelCdf = Color(0.0f, 1.0f, 0.0f);
180 } else {
181 mChannelCdf = Color(0.0f, 0.0f, 1.0f);
182 }
183 }
184#endif
185
186 // Normalize and integrate the cdf, all in one pass
187 normalize = 1.0f / normalize;
188 float cdf = 0.0f;
189 for (int c=0; c < 3; c++) {
190 mChannelCdf[c] *= normalize;
191 mChannelCdf[c] += cdf;
192 cdf = mChannelCdf[c];
193 }
194 MNRY_ASSERT(isOne(cdf));
195#endif
196}

201static float
202findAlphaP(float reflectance, float A)
203{
204 // TODO: Use lookup table as in Christophe's sketch
205 float low = 0.0f;
206 float high = 1.0f;
207
208 MNRY_DURING_ASSERTS(
209 float kdLow = totalDiffuseReflectance(low, A);
210 float kdHigh = totalDiffuseReflectance(high, A);
211 );
212
213 for (int i = 0; i < 16; i++) {
214 MNRY_ASSERT(kdLow <= reflectance && kdHigh >= reflectance);
215 const float mid = (low + high) * 0.5f;
216 const float kd = totalDiffuseReflectance(mid, A);
217 if (kd < reflectance) {
218 low = mid;
219 MNRY_DURING_ASSERTS(kdLow = kd);
220 } else {
221 high = mid;
222 MNRY_DURING_ASSERTS(kdHigh = kd);
223 }
224 }
225 return (low + high) * 0.5f;
226}

231// Careful, the formula in [1] is incorrect. See the other referenced papers,
232// which all have the corrected formula.
233static finline float
234Rd(float r2, float zr, float zv, float alphaP, float sigmaTr)
235{
236 const float dr = scene_rdl2::math::sqrt(r2 + zr * zr);
237 const float dv = scene_rdl2::math::sqrt(r2 + zv * zv);
238 const float rd = (alphaP * sOneOverFourPi) *
239 ((zr * (dr * sigmaTr + 1.0f) * scene_rdl2::math::exp(-sigmaTr * dr)) / (dr * dr * dr) +
240 (zv * (dv * sigmaTr + 1.0f) * scene_rdl2::math::exp(-sigmaTr * dv)) / (dv * dv * dv));
241
242 // return rd < 0.0f ? 0.0f : rd;
243 return rd;
244}

262// Newton solve for u (see [2] for details)
263static float
264solveU(float r1, float sigmaZ)
265{
266 float logfR1Term = scene_rdl2::math::log(1.0f - r1);
267 float u = 1.0f;
268
269 static const int maxSteps = 10;
270 int step = 0;
271 while (step < maxSteps) {
272 float f = sigmaZ * (u - 1) + scene_rdl2::math::log(u) + logfR1Term;
273 step = (scene_rdl2::math::abs(f) < 0.001f ? maxSteps : step + 1);
274 // TODO: Why do we hit this assert ?
275 MNRY_ASSERT(u > sEpsilon);
276 u -= f / (sigmaZ + 1.0f / u);
277 }
278
279 return u;
280}

285float
286DipoleBssrdf::sampleLocal(float r1, float r2, Vec3f &dPi, float &r) const
287{
288 // Importance sample the channel and remap r2
289 int c = (r2 < mChannelCdf[0] ? 0 :
290 (r2 < mChannelCdf[1] ? 1 : 2));
291 r2 = (c == 0 ? r2 / mChannelCdf[0] :
292 (r2 - mChannelCdf[c-1]) / (mChannelCdf[c] - mChannelCdf[c-1]));
293
294// if (c != 0) {
295// dPi = Vec3f(zero);
296// r = 0.0f;
297// return 0.0f;
298// }
299
300 // Pick real or virtual source and remap r1
301 float z;
302 if (r1 < mRatio[c]) {
303 z = mZr[c];
304 r1 /= mRatio[c];
305 } else {
306 z = mZv[c];
307 r1 = (r1 - mRatio[c]) / (1.0f - mRatio[c]);
308 }
309
310 // Solve for u using r1
311 float u = solveU(r1, mSigmaTr[c] * z);
312
313 // Compute radius r
314 r = z * scene_rdl2::math::sqrt(u * u - 1);
315
316 // Note because the above two operations correspond to statistical branching
317 // it is better to use the remapped r1 and r2 to preserve a good discrepancy of the
318 // samples drawn from each distribution out of the 2*3=6 possible
319 // distributions.
320 // TODO: Try using a good quality 3D sampler instead that down-projects well
321 // to lower dimensions.
322 float phi = r2 * sTwoPi;
323
324 // Compute position
325 float sinPhi;
326 float cosPhi;
327 sincos(phi, &sinPhi, &cosPhi);
328
329 dPi[0] = r * cosPhi;
330 dPi[1] = r * sinPhi;
331 dPi[2] = 0.0f;
332
333 return pdfLocal(r);
334}

337float
338DipoleBssrdf::pdfLocal(float r) const
339{
340 // We compute the estimator with MIS, under Veach's one-sample model.
341 // See paper: "Optimally combining sampling techniques for MC rendering"
342 // Here we use estimator (16) and therefore: pdf = ci * pi / wi
343 // With the balance heuristic it amounts to the weighted average of the
344 // pdf for each channel: pdf = sum(ci * pi)
345 float pdf = 0.0f;
346 for (int i = 0; i < 3; i++) {
347 // Compute ci, the discrete probability of picking this channel
348 float ci = (i == 0 ? mChannelCdf[i] :
349 mChannelCdf[i] - mChannelCdf[i-1]);
350
351 // Accumulate ci * pi, where pi is the pdf of sampling dPi
352 pdf += (ci < sEpsilon ? 0.0f :
353 ci * mPdfNormFactor[i] *
354 Rd(r * r, mZr[i], mZv[i], mAlphaP[i], mSigmaTr[i]));
355 }
356
357 return pdf;
358}

363float
364DipoleBssrdf::sampleLocal(float r1, float r2, Vec3f &dPi, float &r) const
365{
366 dPi[0] = (r1 - 0.5f) * 2.0f * mMaxRadius;
367 dPi[1] = (r2 - 0.5f) * 2.0f * mMaxRadius;
368 dPi[2] = 0.0f;
369 r = dPi.length();
370
371 float pdf = 1.0f / (4.0f * mMaxRadius * mMaxRadius);
372 return pdf;

386bool
387DipoleBssrdf::getProperty(Property property, float *dest) const
388{
389 bool handled = true;
390
391 switch (property)
392 {
393 case PROPERTY_COLOR:
394 *dest = totalDiffuseReflectance(mAlphaP[0], mA);
395 *(dest + 1) = totalDiffuseReflectance(mAlphaP[1], mA);
396 *(dest + 2) = totalDiffuseReflectance(mAlphaP[2], mA);
397 break;
398 case PROPERTY_RADIUS:
399 *dest = 1.f / mSigmaTr[0];
400 *(dest + 1) = 1.f / mSigmaTr[1];
401 *(dest + 2) = 1.f / mSigmaTr[2];
402 break;
403 case PROPERTY_PBR_VALIDITY:
404 {
405 Color res = computeAlbedoPbrValidity(mAlbedo);
406 *dest = res.r;
407 *(dest + 1) = res.g;
408 *(dest + 2) = res.b;
409 }
410 break;
411 default:
412 handled = Bssrdf::getProperty(property, dest);
413 }
414
415 return handled;
416}
