59void
60ActivePixelsArray::encode(std::string &outData) const
61{
62 rdl2::ValueContainerEnq vContainerEnq(&outData);
63
64 size_t total = mActivePixels.size();
65 vContainerEnq.enqVLSizeT(total);
66
67 for (size_t i = 0; i < total; ++i) {
68 vContainerEnq.enqBool(getCoarsePass(i));
69 PackTiles::encodeActivePixels(get(i), vContainerEnq);
70 }
71
72 size_t dataSize = vContainerEnq.finalize();
73
74 // Useful info to make sure do properly encode data and it's size for debugging purpose only.
75 // This API is called inside mcrt computations and using cerr always guarantees
76 // to bypass output logging system of arras_framework and minimise the delay to see the msessage.
77 std::cerr << ">> ActivePixelsArray.cc encode()"
78 << " total:" << total
79 << " size:" << dataSize << std::endl;
80}

82void
83ActivePixelsArray::decode(const std::string &inData)
84{
85 rdl2::ValueContainerDeq vContainerDeq(inData.data(), inData.size());
86
87 size_t total;
88 vContainerDeq.deqVLSizeT(total);
89 mActivePixels.resize(total);
90 mCoarsePass.resize(total);
91
92 for (size_t i = 0; i < total; ++i) {
93 bool b;
94 vContainerDeq.deqBool(b);
95 mCoarsePass[i] = b;
96 PackTiles::decodeActivePixels(vContainerDeq, mActivePixels[i]);
97 }
98
99 // This debug dump might be input of gnuplot. This is why start with #.
100 // Also intentionally used cerr. Currently this API only called from
101 // independent small debug purpose program (scene_rdl2/cmd/mcrt_cmd/snapshotDeltaDump)
102 // and using cerr is most easiest way to dump out decode information.
103 std::cerr << "#>> ActivePixelsArray.cc decode()"
104 << " total:" << total << std::endl;
105}
