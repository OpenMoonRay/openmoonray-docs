43bool
44GPUPrimitiveGroup::build(CUstream cudaStream,
45 OptixDeviceContext context,
46 std::string* errorMsg)
47{
48 if (mIsBuilt) {
49 return true;
50 }
51 mIsBuilt = true;
52
53 std::vector<OptixInstance> instances;
54
55 if (!mTriMeshes.empty()) {
56 // Create the GAS (geometry acceleration structure) for all of the triangle
57 // meshes in the scene and wrap them in an OptixInstance.
58 if (!createTrianglesGAS(cudaStream,
59 context,
60 mTriMeshes,
61 &mTrianglesGAS,
62 &mTrianglesGASBuf,
63 errorMsg)) {
64 return false;
65 }
66 OptixInstance oinstance;
67 GPUXform::identityXform().toOptixTransform(oinstance.transform);
68 oinstance.instanceId = 0;
69 oinstance.visibilityMask = 255;
70 oinstance.sbtOffset = mSBTOffset;
71 oinstance.flags = OPTIX_INSTANCE_FLAG_DISABLE_TRANSFORM;
72 oinstance.traversableHandle = mTrianglesGAS;
73 instances.push_back(oinstance);
74 }
75
76 if (!mTriMeshesMB.empty()) {
77 // Create the GAS (geometry acceleration structure) for all of the triangle
78 // meshes in the scene and wrap them in an OptixInstance.
79 if (!createTrianglesGAS(cudaStream,
80 context,
81 mTriMeshesMB,
82 &mTrianglesMBGAS,
83 &mTrianglesMBGASBuf,
84 errorMsg)) {
85 return false;
86 }
87 OptixInstance oinstance;
88 GPUXform::identityXform().toOptixTransform(oinstance.transform);
89 oinstance.instanceId = 0;
90 oinstance.visibilityMask = 255;
91 oinstance.sbtOffset = mSBTOffset + mTriMeshes.size();
92 oinstance.flags = OPTIX_INSTANCE_FLAG_DISABLE_TRANSFORM;
93 oinstance.traversableHandle = mTrianglesMBGAS;
94 instances.push_back(oinstance);
95 }
96
97 if (!mRoundCurves.empty()) {
98 // Create the GAS (geometry acceleration structure) for all of the round
99 // curves in the scene and wrap them in an OptixInstance.
100 if (!createRoundCurvesGAS(cudaStream,
101 context,
102 mRoundCurves,
103 &mRoundCurvesGAS,
104 &mRoundCurvesGASBuf,
105 errorMsg)) {
106 return false;
107 }
108 OptixInstance oinstance;
109 GPUXform::identityXform().toOptixTransform(oinstance.transform);
110 oinstance.instanceId = 0;
111 oinstance.visibilityMask = 255;
112 oinstance.sbtOffset = mSBTOffset + mTriMeshes.size() + mTriMeshesMB.size();
113 oinstance.flags = OPTIX_INSTANCE_FLAG_DISABLE_TRANSFORM;
114 oinstance.traversableHandle = mRoundCurvesGAS;
115 instances.push_back(oinstance);
116 }
117
118 if (!mRoundCurvesMB.empty()) {
119 // Create the GAS (geometry acceleration structure) for all of the motion-blurred round
120 // curves in the scene and wrap them in an OptixInstance.
121 if (!createRoundCurvesGAS(cudaStream,
122 context,
123 mRoundCurvesMB,
124 &mRoundCurvesMBGAS,
125 &mRoundCurvesMBGASBuf,
126 errorMsg)) {
127 return false;
128 }
129 OptixInstance oinstance;
130 GPUXform::identityXform().toOptixTransform(oinstance.transform);
131 oinstance.instanceId = 0;
132 oinstance.visibilityMask = 255;
133 oinstance.sbtOffset = mSBTOffset + mTriMeshes.size() + mTriMeshesMB.size() + mRoundCurves.size();
134 oinstance.flags = OPTIX_INSTANCE_FLAG_DISABLE_TRANSFORM;
135 oinstance.traversableHandle = mRoundCurvesMBGAS;
136 instances.push_back(oinstance);
137 }
138
139 if (!mCustomPrimitives.empty()) {
140 // Create the GAS for all of the custom primitives in the scene and
141 // wrap them in an OptixInstance.
142 if (!createCustomPrimitivesGAS(cudaStream,
143 context,
144 mCustomPrimitives,
145 &mCustomPrimitivesGAS,
146 &mCustomPrimitivesGASBuf,
147 errorMsg)) {
148 return false;
149 }
150 OptixInstance oinstance;
151 GPUXform::identityXform().toOptixTransform(oinstance.transform);
152 oinstance.instanceId = 0;
153 oinstance.visibilityMask = 255;
154 oinstance.sbtOffset = mSBTOffset + mTriMeshes.size() + mTriMeshesMB.size() +
155 mRoundCurves.size() + mRoundCurvesMB.size();
156 oinstance.flags = OPTIX_INSTANCE_FLAG_DISABLE_TRANSFORM;
157 oinstance.traversableHandle = mCustomPrimitivesGAS;
158 instances.push_back(oinstance);
159
160 for (auto& prim : mCustomPrimitives) {
161 prim->freeHostMemory();
162 }
163 }
164
165 if (!mInstances.empty()) {
166 // Wrap all of the instances in OptixInstances. But first we need to
167 // build the instances. This causes the referenced groups/instances to be built
168 // recursively.
169 for (size_t i = 0; i < mInstances.size(); i++) {
170 if (!mInstances[i]->build(cudaStream, context, errorMsg)) {
171 return false;
172 }
173 }
174
175 for (size_t i = 0; i < mInstances.size(); i++) {
176 OptixInstance oinstance = {};
177 oinstance.instanceId = 0;
178 oinstance.visibilityMask = 255;
179 oinstance.sbtOffset = 0;
180 if (!mInstances[i]->mHasMotionBlur) {
181 // The instance's xform is specified on the OptixInstance and the child
182 // node is the top level IAS node in the referenced group.
183 oinstance.flags = OPTIX_INSTANCE_FLAG_NONE;
184 mInstances[i]->mXforms[0].toOptixTransform(oinstance.transform);
185 oinstance.traversableHandle = mInstances[i]->mGroup->mTopLevelIAS;
186 } else {
187 // The xform is specified in the mMMTTraversable instead. The child
188 // node is the MatrixMotionTransform of the instance, which itself
189 // has the group's top level IAS node as its child.
190 oinstance.flags = OPTIX_INSTANCE_FLAG_DISABLE_TRANSFORM;
191 GPUXform::identityXform().toOptixTransform(oinstance.transform);
192 oinstance.traversableHandle = mInstances[i]->mMMTTraversable;
193 }
194 instances.push_back(oinstance);
195 }
196 }
197
198 // Upload the instance objects and AABBs to the GPU. Note this GPU data is
199 // temporary and is freed when this function returns.
200 GPUBuffer<OptixInstance> instanceBuf;
201 if (instanceBuf.allocAndUpload(instances) != cudaSuccess) {
202 *errorMsg = "Error uploading the instance objects to the GPU";
203 return false;
204 }
205
206 // We are building an IAS (instance acceleration structure), so create one that
207 // contains the two instances we just created.
208 OptixBuildInput input = {}; // zero initialize
209 input.type = OPTIX_BUILD_INPUT_TYPE_INSTANCES;
210 input.instanceArray.instances = instanceBuf.deviceptr();
211 input.instanceArray.numInstances = static_cast<unsigned int>(instanceBuf.count());
212
213 std::vector<OptixBuildInput> inputs;
214 inputs.push_back(input);
215
216 OptixAccelBuildOptions accelOptions = {};
217 accelOptions.buildFlags = OPTIX_BUILD_FLAG_ALLOW_COMPACTION | OPTIX_BUILD_FLAG_PREFER_FAST_TRACE;
218 accelOptions.motionOptions.numKeys = 0;
219 accelOptions.operation = OPTIX_BUILD_OPERATION_BUILD;
220
221 if (!createOptixAccel(context,
222 cudaStream,
223 accelOptions,
224 inputs,
225 true,
226 &mTopLevelIASBuf,
227 &mTopLevelIAS,
228 errorMsg)) {
229 return false;
230 }
231
232 return true;
233}

235void
236GPUPrimitiveGroup::getSBTRecords(std::map<std::string, OptixProgramGroup>& pgs,
237 std::vector<HitGroupRecord>& hitgroupRecs)
238{
239 // We need to create one "HitGroup" object a.k.a. record in the Shader
240 // Binding Table for each GPUPrimitive. It is important that these HitGroup
241 // records appear in memory in exactly the same order as they were added.
242 // The HitGroup records tell Optix what kind of geometry is contained in the
243 // BVH node, its properties, and what programs to call to perform ray-object
244 // intersections.
245 // HitGroup records have two parts. 1) A header that tells Optix what program
246 // group to use and 2) custom data for the primitive such as properties, vertices, etc.
247 // The properties differ between primitive types but we can only have one
248 // HitGroup struct type, and all HitGroup records must be the same size,
249 // so the differing properties are efficiently packed via an anonymous union.
250 // See GPUSBTRecord.h.
251
252 for (size_t i = 0; i < mTriMeshes.size(); i++) {
253 HitGroupRecord rec = {};
254 GPUTriMesh* triMesh = mTriMeshes[i];
255 rec.mData.mIsSingleSided = triMesh->mIsSingleSided;
256 rec.mData.mIsNormalReversed = triMesh->mIsNormalReversed;
257 rec.mData.mVisibleShadow = triMesh->mVisibleShadow;
258 rec.mData.mAssignmentIds = triMesh->mAssignmentIds.ptr();
259 rec.mData.mNumShadowLinkEntries = triMesh->mShadowLinkLightIds.count();
260 rec.mData.mShadowLinkAssignmentIds = triMesh->mShadowLinkAssignmentIds.ptr();
261 rec.mData.mShadowLinkLightIds = triMesh->mShadowLinkLightIds.ptr();
262
263 // Specify the program group to use
264 optixSbtRecordPackHeader(pgs["triMeshHG"], &rec);
265
266 hitgroupRecs.push_back(rec);
267 }
268
269 for (size_t i = 0; i < mTriMeshesMB.size(); i++) {
270 HitGroupRecord rec = {};
271 GPUTriMesh* triMesh = mTriMeshesMB[i];
272 rec.mData.mIsSingleSided = triMesh->mIsSingleSided;
273 rec.mData.mIsNormalReversed = triMesh->mIsNormalReversed;
274 rec.mData.mVisibleShadow = triMesh->mVisibleShadow;
275 rec.mData.mAssignmentIds = triMesh->mAssignmentIds.ptr();
276 rec.mData.mNumShadowLinkEntries = triMesh->mShadowLinkLightIds.count();
277 rec.mData.mShadowLinkAssignmentIds = triMesh->mShadowLinkAssignmentIds.ptr();
278 rec.mData.mShadowLinkLightIds = triMesh->mShadowLinkLightIds.ptr();
279
280 // Specify the program group to use
281 optixSbtRecordPackHeader(pgs["triMeshHG"], &rec);
282
283 hitgroupRecs.push_back(rec);
284 }
285
286 for (size_t i = 0; i < mRoundCurves.size(); i++) {
287 HitGroupRecord rec = {};
288 GPURoundCurves* curves = mRoundCurves[i];
289 rec.mData.mIsSingleSided = curves->mIsSingleSided;
290 rec.mData.mIsNormalReversed = curves->mIsNormalReversed;
291 rec.mData.mVisibleShadow = curves->mVisibleShadow;
292 rec.mData.mAssignmentIds = curves->mAssignmentIds.ptr();
293 rec.mData.mNumShadowLinkEntries = curves->mShadowLinkLightIds.count();
294 rec.mData.mShadowLinkAssignmentIds = curves->mShadowLinkAssignmentIds.ptr();
295 rec.mData.mShadowLinkLightIds = curves->mShadowLinkLightIds.ptr();
296
297 // Specify the program group to use
298 switch (curves->mType) {
299 case OPTIX_PRIMITIVE_TYPE_ROUND_LINEAR:
300 optixSbtRecordPackHeader(pgs["roundLinearCurvesHG"], &rec);
301 break;
302 case OPTIX_PRIMITIVE_TYPE_ROUND_CUBIC_BSPLINE:
303 optixSbtRecordPackHeader(pgs["roundCubicBsplineCurvesHG"], &rec);
304 break;
305 default:
306 MNRY_ASSERT_REQUIRE(false);
307 }
308
309 hitgroupRecs.push_back(rec);
310 }
311
312 for (size_t i = 0; i < mRoundCurvesMB.size(); i++) {
313 HitGroupRecord rec = {};
314 GPURoundCurves* curves = mRoundCurvesMB[i];
315 rec.mData.mIsSingleSided = curves->mIsSingleSided;
316 rec.mData.mIsNormalReversed = curves->mIsNormalReversed;
317 rec.mData.mVisibleShadow = curves->mVisibleShadow;
318 rec.mData.mAssignmentIds = curves->mAssignmentIds.ptr();
319 rec.mData.mNumShadowLinkEntries = curves->mShadowLinkLightIds.count();
320 rec.mData.mShadowLinkAssignmentIds = curves->mShadowLinkAssignmentIds.ptr();
321 rec.mData.mShadowLinkLightIds = curves->mShadowLinkLightIds.ptr();
322
323 // Specify the program group to use
324 switch (curves->mType) {
325 case OPTIX_PRIMITIVE_TYPE_ROUND_LINEAR:
326 optixSbtRecordPackHeader(pgs["roundLinearCurvesMBHG"], &rec);
327 break;
328 case OPTIX_PRIMITIVE_TYPE_ROUND_CUBIC_BSPLINE:
329 optixSbtRecordPackHeader(pgs["roundCubicBsplineCurvesMBHG"], &rec);
330 break;
331 default:
332 MNRY_ASSERT_REQUIRE(false);
333 }
334
335 hitgroupRecs.push_back(rec);
336 }
337
338 for (size_t i = 0; i < mCustomPrimitives.size(); i++) {
339 HitGroupRecord rec = {};
340
341 // Fill in the common properties
342 GPUCustomPrimitive* prim = mCustomPrimitives[i];
343 rec.mData.mIsSingleSided = prim->mIsSingleSided;
344 rec.mData.mIsNormalReversed = prim->mIsNormalReversed;
345 rec.mData.mVisibleShadow = prim->mVisibleShadow;
346 rec.mData.mAssignmentIds = prim->mAssignmentIds.ptr();
347 rec.mData.mNumShadowLinkEntries = prim->mShadowLinkLightIds.count();
348 rec.mData.mShadowLinkAssignmentIds = prim->mShadowLinkAssignmentIds.ptr();
349 rec.mData.mShadowLinkLightIds = prim->mShadowLinkLightIds.ptr();
350
351 // Fill in the primitive type-specific properties
352 GPUCurve* curve = dynamic_cast<GPUCurve*>(prim);
353 if (curve) {
354 rec.mData.curve.mMotionSamplesCount = curve->mMotionSamplesCount;
355 rec.mData.curve.mSegmentsPerCurve = curve->mSegmentsPerCurve;
356 rec.mData.curve.mIndices = curve->mIndices.ptr();
357 rec.mData.curve.mNumControlPoints = curve->mNumControlPoints;
358 rec.mData.curve.mControlPoints = curve->mControlPoints.ptr();
359 switch (curve->mBasis) {
360 case BEZIER:
361 optixSbtRecordPackHeader(pgs["flatBezierCurveHG"], &rec);
362 break;
363 case BSPLINE:
364 optixSbtRecordPackHeader(pgs["flatBsplineCurveHG"], &rec);
365 break;
366 case LINEAR:
367 optixSbtRecordPackHeader(pgs["flatLinearCurveHG"], &rec);
368 break;
369 default:
370 break;
371 }
372 }
373 GPUPoints* points = dynamic_cast<GPUPoints*>(prim);
374 if (points) {
375 rec.mData.points.mMotionSamplesCount = points->mMotionSamplesCount;
376 rec.mData.points.mPoints = points->mPoints.ptr();
377 optixSbtRecordPackHeader(pgs["pointsHG"], &rec);
378 }
379 GPUSphere* sphere = dynamic_cast<GPUSphere*>(prim);
380 if (sphere) {
381 rec.mData.sphere.mL2P = sphere->mL2P;
382 rec.mData.sphere.mP2L = sphere->mP2L;
383 rec.mData.sphere.mRadius = sphere->mRadius;
384 rec.mData.sphere.mPhiMax = sphere->mPhiMax;
385 rec.mData.sphere.mZMin = sphere->mZMin;
386 rec.mData.sphere.mZMax = sphere->mZMax;
387 optixSbtRecordPackHeader(pgs["sphereHG"], &rec);
388 }
389 GPUBox* box = dynamic_cast<GPUBox*>(prim);
390 if (box) {
391 rec.mData.box.mL2P = box->mL2P;
392 rec.mData.box.mP2L = box->mP2L;
393 rec.mData.box.mLength = box->mLength;
394 rec.mData.box.mHeight = box->mHeight;
395 rec.mData.box.mWidth = box->mWidth;
396 optixSbtRecordPackHeader(pgs["boxHG"], &rec);
397 }
398
399 hitgroupRecs.push_back(rec);
400 }
401}
