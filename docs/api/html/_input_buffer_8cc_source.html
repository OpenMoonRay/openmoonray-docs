63const scene_rdl2::math::Color
64InputBuffer::getPixel(unsigned int x, unsigned int y) const
65{
66 scene_rdl2::fb_util::VariablePixelBuffer::Format format = getFormat();
67 switch (format) {
68 case scene_rdl2::fb_util::VariablePixelBuffer::RGB888:
69 {
70 const scene_rdl2::fb_util::ByteColor& val = getRgb888Pixel(x, y);
71 return scene_rdl2::math::Color(val.r, val.g, val.b);
72 }
73 case scene_rdl2::fb_util::VariablePixelBuffer::RGBA8888:
74 {
75 const scene_rdl2::fb_util::ByteColor4& val = getRgba8888Pixel(x, y);
76 return scene_rdl2::math::Color(val.r, val.g, val.b);
77 }
78 case scene_rdl2::fb_util::VariablePixelBuffer::FLOAT:
79 {
80 const float val = getFloatPixel(x, y);
81 return scene_rdl2::math::Color(val, val, val);
82 }
83 case scene_rdl2::fb_util::VariablePixelBuffer::FLOAT2:
84 {
85 const scene_rdl2::math::Vec2f& val = getFloat2Pixel(x, y);
86 return scene_rdl2::math::Color(val.x, val.y, 0.f);
87 }
88 case scene_rdl2::fb_util::VariablePixelBuffer::FLOAT3:
89 {
90 const scene_rdl2::math::Vec3f& val = getFloat3Pixel(x, y);
91 return scene_rdl2::math::Color(val.x, val.y, val.z);
92 }
93 case scene_rdl2::fb_util::VariablePixelBuffer::FLOAT4:
94 {
95 const scene_rdl2::math::Vec4f& val = getFloat4Pixel(x, y);
96 return scene_rdl2::math::Color(val.x, val.y, val.z);
97 }
98 default:
99 MNRY_ASSERT(0);
100 return scene_rdl2::math::sBlack;
101 }
102}
