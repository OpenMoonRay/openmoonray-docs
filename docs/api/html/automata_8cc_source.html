53void
54NdfAutomata::State::getTransitions(ustring symbol, IntSet &out_states)const
55{
56 SymbolToIntList::const_iterator s = m_symbol_trans.find(symbol);
57 if (s != m_symbol_trans.end())
58 for (IntSet::const_iterator i = s->second.begin(); i != s->second.end(); ++i)
59 out_states.insert(*i);
60 if (m_wildcard && m_wildcard->matches(symbol))
61 out_states.insert(m_wildcard_trans);
62}

67std::pair <IntSet::const_iterator, IntSet::const_iterator>
68NdfAutomata::State::getLambdaTransitions ()const
69{
70 std::pair <IntSet::const_iterator, IntSet::const_iterator> res;
71 SymbolToIntList::const_iterator s = m_symbol_trans.find(lambda);
72 if (s != m_symbol_trans.end()) {
73 res.first = s->second.begin();
74 res.second = s->second.end();
75 }
76 else
77 // We use a static empty list to return an empty range
78 res.first = res.second = _emptyset.end();
79 return res;

103std::string
104NdfAutomata::State::tostr()const
105{
106 std::string s = "";
107 // output the transitions
108 for (SymbolToIntList::const_iterator i = m_symbol_trans.begin(); i != m_symbol_trans.end(); ++i) {
109 ustring sym = i->first;
110 const IntSet &dest = i->second;
111 if (s.size())
112 s += " ";
113 if (sym == lambda)
114 s += "@";
115 else
116 s += sym.c_str();
117 s += ":{";
118 for (IntSet::const_iterator j = dest.begin(); j != dest.end(); ++j) {
119 if (s[s.size()-1] != '{')
120 s += ", ";
121 s += Strutil::format("%d", *j);
122 }
123 s += "}";
124 }
125 // In case there is a wildcard transition ...
126 if (m_wildcard) {
127 if (s.size())
128 s += " ";
129 // No symbols in the black list, print just .
130 if (m_wildcard->m_minus.empty())
131 s += ".:";
132 else {
133 // Standard regexp notation [^abcd]
134 s += "[^";
135 for (SymbolSet::const_iterator i = m_wildcard->m_minus.begin(); i != m_wildcard->m_minus.end(); ++i) {
136 if (!i->c_str())
137 s += "_";
138 else
139 s += i->c_str();
140 }
141 s += "]:";
142 }
143 s += Strutil::format("%d", m_wildcard_trans);
144 }
145 // and finally the rule if we have it
146 if (m_rule) {
147 s += " | ";
148 s += Strutil::format("%lx", (long unsigned int)m_rule);
149 }
150 return s;
151}

164void
165NdfAutomata::symbolsFrom(const IntSet &states, SymbolSet &out_symbols, Wildcard *&wildcard)const
166{
167 for (IntSet::const_iterator i = states.begin(); i != states.end(); ++i) {
168 const State *state = m_states[*i];
169 // For every state we have to go thorugh all the symbols in the transition table
170 // m_symbol_trans and add them to the output
171 for (SymbolToIntList::const_iterator j = state->m_symbol_trans.begin(); j != state->m_symbol_trans.end(); ++j)
172 if (j->first != lambda)
173 out_symbols.insert(j->first);
174 if (state->m_wildcard) {
175 if (!wildcard)
176 wildcard = new Wildcard();
177 // And if the state has a wildcard movement, we need to add its exclusion list
178 // to our new computed exclusion list. So we guarantee that the returned wildcard
179 // matches will be contained in all the wildcards out of this set
180 wildcard->m_minus.insert(state->m_wildcard->m_minus.begin(), state->m_wildcard->m_minus.end());
181 }
182 }
183 if (wildcard) {
184 // We have to make sure that all the symbols covered by the wildcards
185 // are either covered by our wildcard or in out_symbols set
186 for (IntSet::const_iterator i = states.begin(); i != states.end(); ++i) {
187 const State *state = m_states[*i];
188 if (state->m_wildcard)
189 for (SymbolSet::const_iterator j = wildcard->m_minus.begin(); j != wildcard->m_minus.end(); ++j)
190 if (state->m_wildcard->matches(*j))
191 out_symbols.insert(*j);
192 }
193 // And don't forget about the symbols which are already in the transitions
194 wildcard->m_minus.insert(out_symbols.begin(), out_symbols.end());
195 }
196}

225void
226NdfAutomata::lambdaClosure(IntSet &states)const
227{
228 // This algorithm basically keeps expanding the set until no new states appear
229 // to avoid checking over and over the same states we keep a frontier pair of sets
230 // so we only expand newly discovered states
231 std::vector<int> frontier, discovered;
232 // First iterate all the states in the given set
233 // and see what lambda transitions are there
234 for (IntSet::const_iterator i = states.begin(); i != states.end(); ++i) {
235 const State *state = m_states[*i];
236 std::pair <IntSet::const_iterator, IntSet::const_iterator> lr;
237 // iterate all lambda transitions for this state
238 for (lr = state->getLambdaTransitions(); lr.first != lr.second; lr.first++) {
239 // Add them to the set, and if they were not already there add to the
240 // frontier
241 std::pair<IntSet::iterator, bool> rec = states.insert(*(lr.first));
242 if (rec.second) // newly added
243 frontier.push_back(*(lr.first));
244 }
245 }
246 // frontier becomes last discovered
247 frontier.swap(discovered); // swap discovered and frontier
248 while (discovered.size()) { // as long as there are new found states
249 frontier.clear();
250 // we do the same as in the above loop but with discovered instead of states
251 for (std::vector<int>::iterator i = discovered.begin(); i != discovered.end(); ++i) {
252 const State *state = m_states[*i];
253 std::pair <IntSet::const_iterator, IntSet::const_iterator> lr;
254 for (lr = state->getLambdaTransitions(); lr.first != lr.second; lr.first++) {
255 std::pair<IntSet::iterator, bool> rec = states.insert(*(lr.first));
256 if (rec.second)
257 frontier.push_back(*(lr.first));
258 }
259
260 }
261 // again frontier becomes last discovered
262 frontier.swap(discovered); // swap discovered and frontier
263 }
264}

302DfAutomata::State::getTransition(ustring symbol)const
303{
304 SymbolToInt::const_iterator i = m_symbol_trans.find(symbol);
305 if (i == m_symbol_trans.end())
306 // in case there is a wildcard (!= -1) and the symbol is not
307 // tagged as -1 (not found), then follow the wildcard
308 return m_wildcard_trans;
309 else
310 // it already has -1 if it is in the wildcard's black list
311 return i->second;
312}

340void
341DfAutomata::State::removeUselessTransitions()
342{
343 if (m_wildcard_trans >= 0) {
344 std::list<SymbolToInt::iterator> toremove;
345 for (SymbolToInt::iterator i = m_symbol_trans.begin(); i != m_symbol_trans.end(); ++i)
346 // If there is a transition to the same state as the wildcard, we better nuke it
347 // and just add that symbol to the wildcard be removing it from the map itself
348 if (i->second == m_wildcard_trans)
349 toremove.push_back(i);
350 for (std::list<SymbolToInt::iterator>::iterator i = toremove.begin(); i != toremove.end(); ++i)
351 m_symbol_trans.erase(*i);
352 }
353}

430bool
431DfAutomata::equivalent(const State *dfstateA, const State *dfstateB)
432{
433 // early exit if the size of the tables is different
434 if (dfstateA->m_symbol_trans.size() != dfstateB->m_symbol_trans.size())
435 return false;
436 // The pointed state by both transitions have to be the same or any of dfstateA and dfstateB
437 int destA = (dfstateA->m_wildcard_trans == dfstateA->getId() || dfstateA->m_wildcard_trans == dfstateB->getId()) ? -2 : dfstateA->m_wildcard_trans;
438 int destB = (dfstateB->m_wildcard_trans == dfstateA->getId() || dfstateB->m_wildcard_trans == dfstateB->getId()) ? -2 : dfstateB->m_wildcard_trans;
439 if (destA != destB)
440 return false;
441 // Rules have to be the same
442 if (dfstateA->m_rules != dfstateB->m_rules)
443 return false;
444 for (SymbolToInt::const_iterator i = dfstateA->m_symbol_trans.begin(); i != dfstateA->m_symbol_trans.end(); ++i) {
445 SymbolToInt::const_iterator other = dfstateB->m_symbol_trans.find(i->first);
446 if (other == dfstateB->m_symbol_trans.end())
447 return false;
448 // The pointed state by both transitions have to be the same or any of dfstateA and dfstateB
449 int destA = (i->second == dfstateA->getId() || i->second == dfstateB->getId()) ? -2 : i->second;
450 int destB = (other->second == dfstateA->getId() || other->second == dfstateB->getId()) ? -2 : other->second;
451 // when they are -1 is because they are in the wildcard black list, anyway they have to match so ...
452 if (destA != destB)
453 return false;
454 }
455 // if everything passed, they are equivalent, congratulations.
456 return true;
457}

461void
462DfAutomata::removeEquivalentStates()
463{
464 std::vector<State *> newstatelist;
465 HashIntInt newfromold;
466
467 // First go through all states and delete all those
468 // that are equivalent with a previous one
469 for (size_t i = 0; i < m_states.size(); ++i) {
470 if (!m_states[i]) // it has already been removed
471 continue;
472 // create a new state id from newstatelist.size()
473 // move the pointer there and register the translation
474 int newstate = newfromold[i] = newstatelist.size();
475 newstatelist.push_back(m_states[i]);
476 for (size_t j = i + 1; j < m_states.size(); ++j)
477 if (m_states[j] && equivalent(m_states[i], m_states[j])) {
478 // put in the record that this state will be known as
479 // the one in newstate from now on
480 newfromold[j] = newstate;
481 delete m_states[j];
482 m_states[j] = NULL;
483 }
484 }
485 // Everything has been moved now, but we still have to fix the
486 // transitions so they point to the right states!
487 for (size_t i = 0; i < newstatelist.size(); ++i) {
488 State *state = newstatelist[i];
489 for (SymbolToInt::iterator j = state->m_symbol_trans.begin(); j != state->m_symbol_trans.end(); ++j) {
490 if (j->second != -1) { // if it is -1 it is just in the wildcards black list
491 // Get the new state that maps to the oldstate pointed by the transition
492 HashIntInt::const_iterator trans = newfromold.find(j->second);
493 if (trans != newfromold.end())
494 j->second = trans->second;
495 else
496 std::cerr << "[pathexp] broken translation list between states" << std::endl;
497 }
498 }
499 // Do the same with the wildcard
500 if (state->m_wildcard_trans >=0) {
501 HashIntInt::const_iterator trans = newfromold.find(state->m_wildcard_trans);
502 if (trans != newfromold.end())
503 state->m_wildcard_trans = trans->second;
504 else
505 std::cerr << "[pathexp] broken translation list between states" << std::endl;
506 }
507 }
508 // switch to the new hopefully reduced state vector
509 m_states = newstatelist;
510}

540DfAutomata::State *
541StateSetRecord::ensureState(const IntSet &newstates, std::list<StateSetRecord::Discovery> &discovered)
542{
543 // create the key
544 StateSetKey newkey;
545 keyFromStateSet(newstates, newkey);
546 // check if it is there
547 StateSetMap::const_iterator i = m_key_to_dfstate.find(newkey);
548 if (i != m_key_to_dfstate.end())
549 return i->second;
550 else {
551 // if not in our records create a new DF state
552 DfAutomata::State *tstate = m_dfautomata.newState();
553 getRulesFromSet(tstate, m_ndfautomata, newstates);
554 m_key_to_dfstate[newkey] = tstate;
555 // Add the discovery to the list so it will be explored
556 discovered.push_back(Discovery(tstate, newstates));
557 return tstate;
558 }
559}

575void
576ndfautoToDfauto(const NdfAutomata &ndfautomata, DfAutomata &dfautomata)
577{
578 std::list<StateSetRecord::Discovery> toexplore, discovered;
579 // our initial state is the lambda closure
580 // of the initial state in the NDF automata
581 IntSet initial;
582 initial.insert(0);
583 ndfautomata.lambdaClosure(initial);
584 StateSetRecord record(ndfautomata, dfautomata);
585 // register the initial state
586 record.ensureState(initial, toexplore);
587 while (toexplore.size()) {
588 // new states that we may find when calculating transitions
589 // make sure it is empty
590 discovered.clear();
591 for (std::list<StateSetRecord::Discovery>::iterator i = toexplore.begin(); i != toexplore.end(); ++i) {
592 // get the available symbols to move from this state
593 // set (originalset) in the original automata. Plus
594 // a wildcard movement that is guaranteed to match all
595 // the wildcard transitions in the set (if any)
596 SymbolSet symbols;
597 Wildcard *wildcard = NULL;
598 ndfautomata.symbolsFrom(i->second, symbols, wildcard);
599 for (SymbolSet::iterator j = symbols.begin(); j != symbols.end(); ++j) {
600 IntSet newstates;
601 // get all the states reachable with this symbol
602 ndfautomata.transitionsFrom(i->second, *j, newstates);
603 // build or recover the associated DF state
604 DfAutomata::State *next_state = record.ensureState(newstates, discovered);
605 // and store a transition
606 i->first->addTransition(*j, next_state);
607 }
608 if (wildcard) {
609 IntSet newstates;
610 // we know they all match whatever ours match
611 ndfautomata.wildcardTransitionsFrom(i->second, newstates);
612 // build or recover the associated DF state
613 DfAutomata::State *next_state = record.ensureState(newstates, discovered);
614 // and store a transition
615 i->first->addWildcardTransition(wildcard, next_state);
616 }
617 }
618 // swap toexplore and discovered
619 toexplore.swap(discovered);
620 }
621 // final optimizations
622 dfautomata.removeEquivalentStates();
623 dfautomata.removeUselessTransitions();
624}

636void
637DfOptimizedAutomata::compileFrom(const DfAutomata &dfautomata)
638{
639 m_states.resize(dfautomata.m_states.size());
640 size_t totaltrans = 0;
641 size_t totalrules = 0;
642 for (size_t s = 0; s < m_states.size(); ++s) {
643 totaltrans += dfautomata.m_states[s]->m_symbol_trans.size();
644 totalrules += dfautomata.m_states[s]->m_rules.size();
645 }
646 m_trans.resize(totaltrans);
647 m_rules.resize(totalrules);
648 size_t trans_offset = 0;
649 size_t rules_offset = 0;
650 for (size_t s = 0; s < m_states.size(); ++s) {
651 m_states[s].begin_trans = trans_offset;
652 m_states[s].begin_rules = rules_offset;
653 for (SymbolToInt::const_iterator i = dfautomata.m_states[s]->m_symbol_trans.begin();
654 i != dfautomata.m_states[s]->m_symbol_trans.end(); ++i, ++trans_offset) {
655 m_trans[trans_offset].symbol = i->first;
656 m_trans[trans_offset].state = i->second;
657 }
658 for (RuleSet::const_iterator i = dfautomata.m_states[s]->m_rules.begin();
659 i != dfautomata.m_states[s]->m_rules.end(); ++i, ++rules_offset)
660 m_rules[rules_offset] = *i;
661 m_states[s].ntrans = dfautomata.m_states[s]->m_symbol_trans.size();
662 m_states[s].nrules = dfautomata.m_states[s]->m_rules.size();
663 std::sort(m_trans.begin() + m_states[s].begin_trans, m_trans.begin() + m_states[s].begin_trans + m_states[s].ntrans,
664 DfOptimizedAutomata::Transition::trans_comp);
665 m_states[s].wildcard_trans = dfautomata.m_states[s]->m_wildcard_trans;
666 }
667}
