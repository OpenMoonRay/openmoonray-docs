116 template <typename T> void
117 enqVector(const T &vec)
118 {
119 void *ptr =
120 getEnqDataAddr
121 (ValueContainerUtil::variableLengthLongMaxSize + sizeof(vec[0]) * vec.size());
122 ptr =
123 updatePtr
124 (ptr,
125 ValueContainerUtil::variableLengthEncoding(static_cast<unsigned long>(vec.size()), ptr));
126 VALUE_CONTAINER_ENQ_DEBUG_MSG("enqVector(" << demangle(typeid(T).name()) << ").size():>"
127 << vec.size() << "<\n");
128 // all vec items are stored in contiguous address
129 memoryCopy(ptr, &vec[0], sizeof(vec[0]) * vec.size());
130 ptr = (void *)((uintptr_t)ptr + sizeof(vec[0]) * vec.size());
131 updateId(ptr);
132 VALUE_CONTAINER_ENQ_COUNTER(vec);
133 }

207 void *getEnqDataAddrUpdate(size_t len)
208 {
209 void *ptr = getEnqDataAddr(len);
210 mId += len;
211 return ptr;
212 }
213
214 void *getEnqDataAddr(size_t len)
215 {
216 if (capacity() < len) expandBuff(len);
217 return reinterpret_cast<void *>((uintptr_t)(mBuff->data()) + (uintptr_t)mId);
218 }
219
220 void *updatePtr(void *ptr, size_t size)
221 {
222 return reinterpret_cast<void *>((uintptr_t)ptr + (uintptr_t)size);
223 }
224 void updateId(void *ptr) { mId = (size_t)((uintptr_t)ptr - (uintptr_t)(mBuff->data())); }
225
226 void expandBuff(size_t requestAddSize)
227 {
228 constexpr size_t stepIncreaseSize = 1024; // 1KByte steps
229 size_t expandSizeOrg = requestAddSize - capacity() + mBuff->size();
230 size_t expandSize = expandSizeOrg / stepIncreaseSize * stepIncreaseSize;
231 if (expandSize < expandSizeOrg) expandSize += stepIncreaseSize;
232 mBuff->resize(expandSize);
233 }
234
235 size_t capacity() const { return mBuff->size() - mId; } // current available size
236
237 size_t mStartId; // initial start position of mBuff
238 size_t mId; // current data enqueue position of mBuff
239 std::string *mBuff;

246#endif // end VALUE_CONTAINER_ENQ_DEBUG_MSG_ON

427inline unsigned short
428ValueContainerEnq::enqAlignPad(const unsigned short alignSize, const size_t addOffset)
429{
430 size_t currSize = addOffset + currentSize() + sizeof(unsigned short);
431 unsigned short padSize =
432 (unsigned short)(ValueContainerUtil::alignedSize(currSize, alignSize) - currSize);
433 enqUShort(padSize);
434 for (size_t i = 0; i < padSize; ++i) {
435 enqChar(0x0);
436 }
437 return (unsigned short)(padSize + sizeof(unsigned short));
438}
