84bool
85Arg::noNeedToEvalTest(bool parserHasArg) const
86//
87// Parser object consists of option definitions and argument definitions (See Parser.h comment). Option
88// definition looks like "optA <A> <B>" and Argument definition looks like "<A> <B>" (this is empty option
89// name version of Option definition). This parserHasArg flag indicates Parser object has an Argument type
90// definition or not.
91//
92{
93 if (emptyArg() && mComName.empty()) {
94 // completely empty input
95
96 if (!parserHasArg) {
97 // Parser does not have an argument definition and this Arg does not have any argument
98 // as well. So, we can safely return true (= no need to evaluate this arg).
99 return true;
100 }
101
102 // This is an empty Arg but a potential error case. Parser has argument definitions
103 // and we can not skip evaluation of this Arg.
104 }
105 return false;
106}

163bool
164Arg::msg(const std::string &msg) const
165{
166 bool flag = true;
167
168 if (mCerrOutput) {
169 // output string to cerr
170 std::cerr << msg;
171 }
172
173 if (mTlSvr) {
174 // output to TlSvr if necessary
175 if (!mTlSvr->send(msg)) flag = false; // blocking function
176 }
177
178 // output to msgHandler if necessary
179 if (mMsgHandler) {
180 if (!(mMsgHandler)(msg)) flag = false;
181 }
182
183 return flag;
184}

220bool
221Arg::verify(const std::string &comName,
222 const std::string &currOptName,
223 size_t currArgId,
224 size_t nextId,
225 const ArgTbl &arg,
226 const ArgTbl &org)
227//
228// This function is designed for unitTest
229//
230{
231 auto isSameStrVec = [](const ArgTbl &a, const ArgTbl &b) {
232 if (a.size() != b.size()) return false;
233 for (size_t i = 0; i < a.size(); ++i) {
234 if (a[i] != b[i]) return false;
235 }
236 return true;
237 };
238
239 if (mComName != comName ||
240 mCurrOptName != currOptName ||
241 mCurrArgId != currArgId ||
242 !isSameStrVec(mArg, arg) ||
243 mNextId != nextId ||
244 !isSameStrVec(mOrg, org)) {
245 return false;
246 }
247 return true;
248}

252Arg
253Arg::makeChildArg(const std::function<std::string()> &childCmdNameGenFunc)
254{
255 Arg arg;
256
257 arg.mNextId = mNextId;
258 arg.mTlSvr = mTlSvr;
259 arg.mMsgHandler = mMsgHandler;
260
261 for (size_t i = 0; i < mOrg.size(); ++i) {
262 if (i >= mNextId) arg.mArg.emplace_back(mOrg[i]);
263 arg.mOrg.emplace_back(mOrg[i]);
264 }
265 arg.mComName = (childCmdNameGenFunc)();
266
267 shiftArgAll(); // shift all for original arg
268
269 return arg;
270}

272void
273Arg::setupOrg(const std::string &cmdLine)
274{
275 // First argument (i.e. command name) is empty in this case because this is a top level command line
276 mOrg.emplace_back(""); // empty first entry
277
278 std::string workCmdLine = processBlankNl(cmdLine);
279 if (workCmdLine.empty()) {
280 return; // empty
281 }
282
283 std::istringstream istr(workCmdLine);
284 std::string word;
285 while (std::getline(istr, word, ' ')) {
286 mOrg.emplace_back(word.c_str());
287 }
288}

306Arg::shiftArg()
307{
308 if (!mArg.size()) return;
309
310 for (size_t i = 1; i < mArg.size(); ++i) {
311 mArg[i - 1] = std::move(mArg[i]);
312 }
313 mArg.pop_back();
314 mNextId++;

390std::string
391Arg::processBlankNl(const std::string &str) const
392{
393 std::string tmpStr = str;
394 if (!tmpStr.empty()) {
395 tmpStr = addSpaceBeforeComment(tmpStr);
396 tmpStr = scene_rdl2::str_util::replaceNlToSingleSpace(tmpStr); // nl processing
397 tmpStr = scene_rdl2::str_util::trimBlank(tmpStr); // rm front/back blanks
398 tmpStr = scene_rdl2::str_util::replaceBlankToSingleSpace(tmpStr); // blank processing
399 }
400 return tmpStr;

403std::string
404Arg::addSpaceBeforeComment(const std::string &str) const
405{
406 size_t pos = str.find("#");
407 if (pos == std::string::npos) return str;
408
409 std::string tmpStr;
410 for (size_t i = 0; i < str.size(); ++i) {
411 if (str[i] == '#') tmpStr.push_back(' ');
412 tmpStr.push_back(str[i]);
413 }
414 return tmpStr;
415}

417bool
418Arg::vaMsg(const char *fmt, va_list ap) const
419{
420 // figure out proper buffer size
421 int buffSize = 0;
422 {
423 va_list tmpAp;
424 va_copy(tmpAp, ap); // we have to use va_copy in order to keep original ap
425 buffSize = vsnprintf(nullptr, 0, fmt, tmpAp);
426 va_end(tmpAp);
427 }
428
429 std::string msgBuff(buffSize + 1, 0x0); // +1 is for terminator NULL
430
431 // create string from va_list
432 if (std::vsnprintf(static_cast<char *>(&msgBuff[0]),
433 msgBuff.size(),
434 fmt,
435 ap) != buffSize) {
436 return false;
437 }
438 msgBuff.resize(buffSize); // resize to actual length (without terminator NULL)
439
440 return msg(msgBuff);
441}

443bool
444Arg::cmpOpt(const std::string &a, const std::string &b, bool caseSensitive) const
445{
446 std::string workA, workB;
447 if (caseSensitive) {
448 workA = a;
449 workB = b;
450 } else {
451 workA = str_util::upperStr(a);
452 workB = str_util::upperStr(b);
453 }
454
455 if (workA == workB) return true;
456
457 if (workA[0] == '-') {
458 if (workA.substr(1) == workB) return true;
459 } else if (workB[0] == '-') {
460 if (workA == workB.substr(1)) return true;
461 }
462 return false;
463}

467Arg::isBool(const std::string &str)
468{

474 return false;

477}
