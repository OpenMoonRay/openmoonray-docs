22void
23TestDisplace::displace()
24{
25 mcrt_common::ThreadLocalState *topLevelTls = mcrt_common::getFrameUpdateTLS();
26 shading::TLState *tls = MNRY_VERIFY(topLevelTls->mShadingTls.get());
27 alloc::Arena *arena = MNRY_VERIFY(tls)->mArena;
28 SCOPED_MEM(arena);
29
30 // setup a simple rdl scene consisting of a displacment object with
31 // a map binding.
32 rdl2::SceneContext ctx;
33 ctx.setDsoPath(ctx.getDsoPath() + ":dso/map:dso/displacement");
34 rdl2::Map *map =
35 ctx.createSceneObject("TestMap", "/map")->asA<rdl2::Map>();
36 {
37 rdl2::SceneObject::UpdateGuard update(map);
38 map->set("mult", .5f);
39 }
40 rdl2::Displacement *displacement =
41 ctx.createSceneObject("TestDisplacement", "/displacement")->
42 asA<rdl2::Displacement>();
43 {
44 rdl2::SceneObject::UpdateGuard update(displacement);
45 displacement->set("height", .5f);
46 displacement->setBinding("height", map);
47 }
48 displacement->applyUpdates();
49
50 // process four vector bundles (16 points in SSE).
51 const int numBundles = 4;
52
53 // initialize state input. since we are displacing along the
54 // normal, we'll set each normal to (1, 0, 0), (0, 1, 0), (0, 0, 1) etc..
55 shading::Statev *statev = arena->allocArray<shading::Statev>(numBundles, SIMD_MEMORY_ALIGNMENT);
56 memset(statev, 0, sizeof(shading::Statev) * numBundles);
57
58 for (int i = 0; i < numBundles; ++i) {
59 for (int lane = 0; lane < (int)VLEN; ++lane) {
60
61 statev[i].mSt.x[lane] = 1.0;
62
63 int point = i * VLEN + lane;
64 statev[i].mN.x[lane] = (point - 0) % 3 == 0? 1.0 : 0.0;
65 statev[i].mN.y[lane] = (point - 1) % 3 == 0? 1.0 : 0.0;
66 statev[i].mN.z[lane] = (point - 2) % 3 == 0? 1.0 : 0.0;
67 }
68 }
69
70 // setup storage for displacement output
71 Vec3fv *outv = arena->allocArray<Vec3fv>(numBundles, SIMD_MEMORY_ALIGNMENT);
72
73 // displace
74 displacev(displacement, tls, numBundles, statev, outv);
75
76 // now verify the results
77 // should be (.25, 0, 0), (0, .25, 0), (0, 0, .25) ...
78 for (int i = 0; i < numBundles; ++i) {
79 for (int lane = 0; lane < (int)VLEN; ++lane) {
80 int point = i * VLEN + lane;
81 const float x = (point - 0) % 3 == 0? .25 : 0;
82 const float y = (point - 1) % 3 == 0? .25 : 0;
83 const float z = (point - 2) % 3 == 0? .25 : 0;
84 CPPUNIT_ASSERT(outv[i].x[lane] == x);
85 CPPUNIT_ASSERT(outv[i].y[lane] == y);
86 CPPUNIT_ASSERT(outv[i].z[lane] == z);
87 }
88 }
89}
