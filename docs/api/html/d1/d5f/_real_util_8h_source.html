17//
18// The following {float,double}ToleranceEqual() are useful for the equality judgment of
19// 2 float/double values with tolerance precision control.
20// Keep in mind, following functions only work with a non-zero value.
21// Tolerance precision is controlled by bit-mask which is generated by compareMaskGen{32,64}().
22//
23// If you generate compareMask by small bitSize number, compare tolerance is small and executes high
24// precision equality judgment. On the other hand, if you generate compareMask by big bitSize number,
25// compare tolerance is big and executes low precision equality judgment.
26//
27// example-1
28//
29// float a = 1.234;
30// float b = 1.2340001;
31//
32// // exact bitImage compare
33// assert(!real_util::floatToleranceEqual(a, b, real_util::compareMaskGen32(0)));
34//
35// // a and b are considered as the equal value under 1-bit tolerance
36// assert(real_util::floatToleranceEqual(a, b, real_util::compareMaskGen32(1))));
37//
38// example-2
39//
40// float a = 12340000.0;
41// float b = 12340001.0;
42//
43// // exact bitImage compare
44// assert(!real_util::floatToleranceEqual(a, b, real_util::compareMaskGen32(0)));
45//
46// // a and b are considered as the equal value under 1-bit tolerance
47// assert(real_util::floatToleranceEqual(a, b, real_util::compareMaskGen32(1))));
48//
49// example-3
50//
51// double a = 78.9;
52// double b = 78.9000000000001;
53//
54// // 2-bit tolerance returns false
55// assert(!real_util::doubleToleranceEqual(a, b, real_util::compareMaskGen64(2)));
56//
57// // a and b are considered as the equal value under 3-bit tolerance
58// assert(real_util::doubleToleranceEqual(a, b, real_util::compareMaskGen64(3)));
59//

61inline
62unsigned int
63compareMaskGen32(unsigned int maskBitSize)
64//
65// This function generate bitmask for floatToleranceEqual()¡Çs 3rd argument
66//
67{
68 unsigned int mask = 0x0;
69 for (unsigned int i = 0; i < ((maskBitSize > 32) ? 32 : maskBitSize); ++i) {
70 mask |= (0x1 << i);
71 }
72 return mask;
73}

75inline
76bool
77floatToleranceEqual(const float &a, const float &b, const unsigned int compareMask32)
78//
79// Tolerance equal compare operation for float value
80// Compare the bit-wise difference of IEEE 32bit float format.
81// Only works with a non-ZERO value.
82// return true when 2 values are the same under defined precision. otherwise, return false.
83//
84{
85 union {
86 float f;
87 unsigned int ui;
88 } uniA, uniB;
89 uniA.f = a; unsigned int ai = uniA.ui;
90 uniB.f = b; unsigned int bi = uniB.ui;
91 if (ai < bi) {
92 return compareMask32 >= (bi - ai);
93 } else {
94 return compareMask32 >= (ai - bi);
95 }
96}

153inline
154bool
155doubleToleranceEqual(const double &a, const double &b, const unsigned long compareMask64)
156//
157// Tolerance equal compare operation for double value
158// Compare the bit-wise difference of IEEE 64bit double format.
159// Only works with a non-ZERO value.
160// return true when 2 values are the same under defined precision. otherwise, return false.
161//
162{
163 union {
164 double d;
165 unsigned long ul;
166 } uniA, uniB;
167 uniA.d = a; unsigned long al = uniA.ul;
168 uniB.d = b; unsigned long bl = uniB.ul;
169 if (al < bl) {
170 return compareMask64 >= (unsigned long)(bl - al);
171 } else {
172 return compareMask64 >= (unsigned long)(al - bl);
173 }
