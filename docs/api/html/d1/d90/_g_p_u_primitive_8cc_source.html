26void
27GPUCurve::getPrimitiveAabbs(std::vector<OptixAabb>* aabbs) const
28{
29 aabbs->resize(mHostIndices.size());
30
31 int cpPerCurve = (mBasis == LINEAR) ? 2 : 4;
32
33 for (size_t i = 0; i < mHostIndices.size(); i++) {
34 OptixAabb &bb = (*aabbs)[i];
35 bb = emptyAabb();
36 for (int ms = 0; ms < mMotionSamplesCount; ms++) {
37 int idx = ms * mNumControlPoints + mHostIndices[i];
38 for (int j = 0; j < cpPerCurve; j++) {
39 float4 p = mHostControlPoints[idx + j];
40 float3 center = make_float3(p);
41 float radius = p.w;
42 bb = expandAabb(bb, center - radius);
43 bb = expandAabb(bb, center + radius);
44 }
45 }
46 }
47}

56void
57GPUPoints::getPrimitiveAabbs(std::vector<OptixAabb>* aabbs) const
58{
59 aabbs->resize(mHostPoints.size() / mMotionSamplesCount);
60
61 for (size_t i = 0; i < mHostPoints.size() / mMotionSamplesCount; i++) {
62 OptixAabb &bb = (*aabbs)[i];
63 bb = emptyAabb();
64 for (int ms = 0; ms < mMotionSamplesCount; ms++) {
65 size_t idx = i * mMotionSamplesCount + ms;
66 float3 point = make_float3(mHostPoints[idx]);
67 float radius = mHostPoints[idx].w;
68 OptixAabb sampleBB = {point.x - radius, point.y - radius, point.z - radius,
69 point.x + radius, point.y + radius, point.z + radius};
70 bb = combineAabbs(bb, sampleBB);
71 }
72 }
73}

81void
82GPUSphere::getPrimitiveAabbs(std::vector<OptixAabb>* aabbs) const
83{
84 OptixAabb localAabb = {-mRadius, -mRadius, mZMin,
85 mRadius, mRadius, mZMax};
86 // TODO: This will create a poor bounding box if the transform has any
87 // rotation. CPU-side Sphere::computeAABB() does the same thing but it's
88 // even worse because it doesn't consider z-clipping of the sphere.
89 aabbs->push_back(mL2P.transformAabb(localAabb));
90}
