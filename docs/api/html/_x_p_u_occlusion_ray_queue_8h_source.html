22/*
23Dispatching rays to the GPU efficiently is a tricky problem. You need to ensure
24that your batch of rays is large enough to run efficiently to mitigate overhead,
25and there are dozens of CPU threads that all want to submit small numbers of
26individual rays to the GPU at once, creating contention. It is not sufficient
27to have one large shared queue that each CPU thread locks when it adds a few
28rays because the other CPU threads will spend a lot of time waiting.
29
30You also need to keep the GPU as busy as possible but not cause the CPU side to
31stall by giving the GPU more work than it can process.
32
33MoonRay currently implements a fairly simple scheme that also works reasonably well,
34if a bit memory inefficient. Each CPU thread has its own large queue of occlusion
35rays to process (65536 rays). Each ray is 128 bytes, so each thread's queue is 8MB.
36This is not ideal, but the cost scales with the number of threads. I.e. a 96-thread
37render will consume 768MB of memory for XPU queues. BUT, these systems have 192GB of RAM
38so this is not a major problem.
39
40Since each CPU thread has its own queue for XPU rays, there is no contention when filling
41the queues. Previous work used a single shared queue, but the overhead of 96 threads contending
42for a lock on the queue was too high and performance was poor.
43
44When a CPU thread has completely filled its queue, it attempts to call the GPU to process
45those queued rays. First, it checks how many other CPU threads are waiting for the GPU.
46If there are more than two other threads waiting, this CPU thread will process its rays by
47itself. This is a simple automatic load-balancing scheme that ensures that the CPU side isn't
48too blocked with too many CPU threads waiting for the GPU. It also keeps the GPU fed with
49a constant uninterrupted stream of work by having a few threads queued up waiting with work ready.
50
51This system can potentially be improved upon with Keith's lock-free queue ideas,
52but this current scheme is a reasonable baseline that we can compare more sophisticated
53and efficient methods.
54
55The API design of this queue class resembles the other queues in mcrt_common/Bundle.h.
56The two main methods are addEntries() and flush().
57*/
58
59template<typename EntryType>
60class XPUAcceleratorQueue
61{
62public:
63 // The CPU handler that is called when the GPU is busy
64 typedef void (*CPUHandler)(mcrt_common::ThreadLocalState *tls,
65 unsigned numEntries,
66 const EntryType *entryData,
67 void *userData);
68
69 // The GPU handler that calls the GPU
70 typedef void (*GPUHandler)(mcrt_common::ThreadLocalState *tls,
71 unsigned numEntries,
72 EntryType *entryData,
73 const rt::GPUOcclusionRay *gpuRays,
74 tbb::spin_mutex& mutex);
75
76 XPUAcceleratorQueue(unsigned numCPUThreads,
77 unsigned cpuThreadQueueSize,
78 CPUHandler cpuThreadQueueHandler,
79 GPUHandler gpuQueueHandler,
80 void *handlerData) :
81 mNumCPUThreads(numCPUThreads),
82 mCPUThreadQueueSize(cpuThreadQueueSize),
83 mCPUThreadQueueHandler(cpuThreadQueueHandler),
84 mGPUQueueHandler(gpuQueueHandler),
85 mHandlerData(handlerData)
86 {
87 MNRY_ASSERT(numCPUThreads);
88 MNRY_ASSERT(cpuThreadQueueSize);
89 MNRY_ASSERT(mCPUThreadQueueHandler);
90 MNRY_ASSERT(mGPUQueueHandler);
91
92 // Create a queue for each CPU thread
93 mCPUThreadQueueEntries.resize(mNumCPUThreads);
94 mCPUThreadQueueNumQueued.resize(mNumCPUThreads);
95 for (size_t i = 0; i < numCPUThreads; i++) {
96 mCPUThreadQueueEntries[i] = scene_rdl2::util::alignedMallocArray<EntryType>(mCPUThreadQueueSize, CACHE_LINE_SIZE);
97 mCPUThreadQueueNumQueued[i] = 0;
98 }
99
100 mThreadsWaitingForGPU = 0;
101 }
102
103 virtual ~XPUAcceleratorQueue()
104 {
105 for (size_t i = 0; i < mNumCPUThreads; i++) {
106 MNRY_ASSERT(mCPUThreadQueueNumQueued[i] == 0);
107 scene_rdl2::util::alignedFree(mCPUThreadQueueEntries[i]);
108 }
109 }
110
111 unsigned getMemoryUsed() const
112 {
113 return (sizeof(EntryType) * mCPUThreadQueueSize * mNumCPUThreads) + sizeof(*this);
114 }
115
116 bool isValid() const
117 {
118 MNRY_ASSERT(mGPUQueueHandler);
119 return true;
120 }

122 // The CPU threads call this function to queue up rays (entries.)
123 void addEntries(mcrt_common::ThreadLocalState *tls,
124 unsigned numEntries,
125 EntryType *entries)
126 {
127 MNRY_ASSERT(tls);
128 MNRY_ASSERT(numEntries);
129 MNRY_ASSERT(entries);
130
131 EXCL_ACCUMULATOR_PROFILE(tls, EXCL_ACCUM_QUEUE_LOGIC);
132
133 while (numEntries >= mCPUThreadQueueSize) {
134 // The number of rays that have been submitted exceeds or equals the
135 // CPU thread queue size. Process them in batches of mCPUThreadQueueSize.
136 processRays(tls, mCPUThreadQueueSize, entries);
137 entries += mCPUThreadQueueSize;
138 numEntries -= mCPUThreadQueueSize;
139 }
140
141 int threadIdx = tls->mThreadIdx;
142
143 MNRY_ASSERT(mCPUThreadQueueNumQueued[threadIdx] < mCPUThreadQueueSize);
144 unsigned totalEntries = mCPUThreadQueueNumQueued[threadIdx] + numEntries;
145
146 // Is there enough room in the CPU thread queue for the new entries?
147 if (totalEntries <= mCPUThreadQueueSize) {
148 // Copy entries into CPU thread's queue as there is room for them in that queue.
149 memcpy(mCPUThreadQueueEntries[threadIdx] + mCPUThreadQueueNumQueued[threadIdx],
150 entries,
151 numEntries * sizeof(EntryType));
152 mCPUThreadQueueNumQueued[threadIdx] = totalEntries;
153 return;
154 }
155
156 // Else there isn't enough room in the CPU thread's queue for all of the new
157 // entries. We need to flush the queue to free up space.
158 processRays(tls, mCPUThreadQueueNumQueued[threadIdx], mCPUThreadQueueEntries[threadIdx]);
159 mCPUThreadQueueNumQueued[threadIdx] = 0;
160
161 // Now that the queue is empty, we can add the new entries.
162 memcpy(mCPUThreadQueueEntries[threadIdx],
163 entries,
164 numEntries * sizeof(EntryType));
165 mCPUThreadQueueNumQueued[threadIdx] = numEntries;
166
167 MNRY_ASSERT(mCPUThreadQueueNumQueued[threadIdx] < mCPUThreadQueueSize);
168 }
169
170 // Explicit flush of the CPU queues per thread.
171 unsigned flush(mcrt_common::ThreadLocalState *tls, scene_rdl2::alloc::Arena *arena)
172 {
173 EXCL_ACCUMULATOR_PROFILE(tls, EXCL_ACCUM_QUEUE_LOGIC);
174
175 int threadIdx = tls->mThreadIdx;
176
177 int numFlushed = mCPUThreadQueueNumQueued[threadIdx];
178 if (mCPUThreadQueueNumQueued[threadIdx]) {
179 processRays(tls, mCPUThreadQueueNumQueued[threadIdx], mCPUThreadQueueEntries[threadIdx]);
180 mCPUThreadQueueNumQueued[threadIdx] = 0;
181 }
182
183 return numFlushed;
184 }
185
186protected:
187
188 void processRays(mcrt_common::ThreadLocalState *tls,
189 unsigned numRays,
190 EntryType *rays)
191 {
192 EXCL_ACCUMULATOR_PROFILE(tls, EXCL_ACCUM_QUEUE_LOGIC);
193
194 MNRY_ASSERT(numRays);
195
196 if (mThreadsWaitingForGPU.load() < 3 && numRays > 1024) {
197 // There are an acceptable number of threads waiting to access the GPU, so we
198 // just wait our turn. But, before we try to acquire the GPU lock, we get the
199 // buffer of rays ready for the GPU.
200 // The value 2 was determined empirically. Higher values do not provide benefit.
201
202 // Another thread can sneak in here before we increment, but it doesn't matter
203 // because 2 is just a heuristic anyway and it doesn't matter if there's a small
204 // chance we end up with 3.
205 mThreadsWaitingForGPU++;
206
207 pbr::TLState *pbrTls = tls->mPbrTls.get();
208 scene_rdl2::alloc::Arena *arena = &tls->mArena;
209 rt::GPUOcclusionRay* gpuRays = arena->allocArray<rt::GPUOcclusionRay>(numRays, CACHE_LINE_SIZE);
210
211 // Now that the thread knows it's running on the GPU, it still needs to wait its
212 // turn for the GPU. Before waiting, it needs to prepare the GPUOcclusionRays.
213 for (size_t i = 0; i < numRays; ++i) {
214 const BundledOcclRay &occlRay = rays[i];
215 MNRY_ASSERT(occlRay.isValid());
216
217 gpuRays[i].mOriginX = occlRay.mOrigin.x;
218 gpuRays[i].mOriginY = occlRay.mOrigin.y;
219 gpuRays[i].mOriginZ = occlRay.mOrigin.z;
220 gpuRays[i].mDirX = occlRay.mDir.x;
221 gpuRays[i].mDirY = occlRay.mDir.y;
222 gpuRays[i].mDirZ = occlRay.mDir.z;
223 gpuRays[i].mMinT = occlRay.mMinT;
224 gpuRays[i].mMaxT = occlRay.mMaxT;
225 gpuRays[i].mTime = occlRay.mTime;
226 const scene_rdl2::rdl2::Light* light = static_cast<BundledOcclRayData *>(
227 pbrTls->getListItem(occlRay.mDataPtrHandle, 0))->mLight->getRdlLight();
228 gpuRays[i].mLightId = reinterpret_cast<intptr_t>(light);
229 }
230
231 // Acquire the GPU.
232 mGPUDeviceMutex.lock();
233
234 // This thread is no longer waiting to access the GPU, because it has the GPU.
235 mThreadsWaitingForGPU--;
236
237 // The handler unlocks the GPU device mutex internally once the GPU
238 // is finished but there is still some CPU code left for this thread to run.
239 (*mGPUQueueHandler)(tls,
240 numRays,
241 rays,
242 gpuRays,
243 mGPUDeviceMutex);
244
245 } else {
246 // There's too many threads already waiting for the GPU, and we would need to wait
247 // too long. Process these rays on the CPU instead.
248 (*mCPUThreadQueueHandler)(tls, numRays, rays, mHandlerData);
249 }
250 }
251
252 unsigned mNumCPUThreads;
253 unsigned mCPUThreadQueueSize;
254 std::vector<EntryType*> mCPUThreadQueueEntries;
255 std::vector<unsigned> mCPUThreadQueueNumQueued;
256 CPUHandler mCPUThreadQueueHandler;
257 std::atomic<int> mThreadsWaitingForGPU;
258 GPUHandler mGPUQueueHandler;
259 tbb::spin_mutex mGPUDeviceMutex;
260 void * mHandlerData;
261};
