15// This class is used to drill down into a multi-level instancing
16// hierarchy looking for volume or surface assignments on a leaf
17// node reference. It assumes all leaf node references have already
18// had their assignments set.
19class GetAssignments : public geom::PrimitiveVisitor
20{
21public:
22 GetAssignments(bool &hasVolume, bool &hasSurface):
23 mHasVolume(hasVolume),
24 mHasSurface(hasSurface)
25 {
26 }
27
28 void visitPrimitiveGroup(geom::PrimitiveGroup& pg) override
29 {
30 bool isParallel = false;
31 pg.forEachPrimitive(*this, isParallel);
32 }
33
34 void visitInstance(geom::Instance& i) override
35 {
36 geom::SharedPrimitive *ref = i.getReference().get();
37 mHasVolume |= ref->getHasVolumeAssignment();
38 mHasSurface |= ref->getHasSurfaceAssignment();
39 // drill down in the case of multi-level instancing
40 ref->getPrimitive()->accept(*this);
41 }
42
43 bool &mHasVolume;
44 bool &mHasSurface;
45};
