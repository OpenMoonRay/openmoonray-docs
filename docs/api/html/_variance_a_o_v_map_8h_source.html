13/// @class VarianceAOVMap
14/// @brief Links AovSchema members by index to/from corresponding variance aovs
15///
16/// When calculating the variance of an aov, we create another aov that
17/// references the original aov (the source). This class allows us to look up
18/// the corresponding aovs. A aov may have more than one variance aovs using it
19/// as the source, but a variance aov can only reference a single source.
20class VarianceAOVMap
21{
22 using SourceToVarianceMap = std::unordered_multimap<int, int>;
23 using VarianceToSourceMap = std::unordered_map<int, int>;

25public:

92 /// @brief Get the indices of aovs that compute the the variance of an aov
93 /// \param sourceAovIdx the aov index of a potential variance source
94 /// \return an iterator range of indices of aovs that compute the variance of this aov
95 SourceToVarianceRange getSourceToVarianceAOVs(int sourceAovIdx) const
96 {
97 const auto range = mSourceToVariance.equal_range(sourceAovIdx);
98 return { SourceToVarianceIterator{range.first}, SourceToVarianceIterator{range.second} };
99 }
100
101 /// @brief Get the indices of the aov that is the source of the variance being computed by an aov
102 /// \param varianceAovIdx the variance aov
103 /// \return the index of the source aov from which \p varianceAovIdx is computing the variance. -1 if this isn't a variance aov.
104 int getVarianceToSourceAOV(int varianceAovIdx) const
105 {
106 const auto itr = mVarianceToSource.find(varianceAovIdx);
107 if (itr == mVarianceToSource.end()) {
108 return -1;
109 } else {
110 return itr->second;
111 }
112 }
113
114 // Does this particular aov have another aov that is using it as the source of its variance calculation?
115 bool hasVarianceEntry(int sourceAovIdx) const
116 {
117 return mSourceToVariance.find(sourceAovIdx) != mSourceToVariance.cend();
118 }
119
120 // Is this aov computing the variance of another aov?
121 bool isVarianceEntry(int varianceAovIdx) const
122 {
123 return mVarianceToSource.find(varianceAovIdx) != mVarianceToSource.cend();
124 }
125
126 void linkVarianceOutput(int source, int variance)
127 {
128 mSourceToVariance.emplace(source, variance);
129 const auto r __attribute__ ((unused)) = mVarianceToSource.emplace(variance, source);
130 MNRY_ASSERT(r.second);
131 }
132
133private:
134 SourceToVarianceMap mSourceToVariance;
135 VarianceToSourceMap mVarianceToSource;
136};
