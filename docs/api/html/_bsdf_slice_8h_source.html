28///
29/// @class BsdfSlice Bsdf.h <pbr/Bsdf.h>
30/// @brief A BsdfSlice object is used to both represent a lower-dimensional
31/// slice of a multi-lobe bsdf and to evaluate multi-lobe bsdfs.
32///
33class BsdfSlice
34{
35public:
36 /// A bsdf slice is defined given a surface geometric normal Ng and
37 /// observer direction wo. We are also narrowing down on only the lobes that
38 /// match the given flags. It also controls when a Bsdf is evaluated or
39 /// sampled, whether or not its result includes the cosine term,
40 /// and what if any shadow terminator smoothing should be applied.
41 BsdfSlice(const scene_rdl2::math::Vec3f Ng, const scene_rdl2::math::Vec3f &wo, bool includeCosineTerm,
42 bool entering, ShadowTerminatorFix shadowTerminatorFix, BsdfLobe::Type flags = BsdfLobe::ALL) :
43 mNg(Ng),
44 mWo(wo),
45 mFlags(flags),
46 mShadowTerminatorFix(shadowTerminatorFix),
47 mIncludeCosineTerm(includeCosineTerm),
48 mEntering(entering) {}
49
50 virtual ~BsdfSlice() {}

62 /// This method returns true if a given sample is compatible with the given
63 /// lobe and according to the geometric surface. It tests if
64 /// wo and wi are in the same / opposite hemispheres wrt. Ng and checks
65 /// whether this is compatible with the lobe type REFLECTION vs. TRANSMISSION.
66 /// This method always returns false for spherical lobes.
67 finline bool isSampleGeometricallyInvalid(const BsdfLobe &lobe, const scene_rdl2::math::Vec3f &wi) const
68 {
69 return (lobe.getIsSpherical() ? false :
70 !(lobe.getType() & ((dot(mNg, mWo) * dot(mNg, wi) > 0.0f) ?
71 BsdfLobe::REFLECTION : BsdfLobe::TRANSMISSION)));
72 }
73
74 /// Computes a lobe type flag that lets integrators ignore btdfs or brdfs
75 /// based on whether wo and wi are respectively on the same or opposite side
76 /// of the geometric surface (wrt. Ng).
77 /// For spherical Bsdfs, this always returns the unmodified slice flags.
78 finline BsdfLobe::Type getSurfaceFlags(const Bsdf &bsdf, const scene_rdl2::math::Vec3f &wi) const
79 {
80 return (bsdf.getIsSpherical() ? mFlags :
81 BsdfLobe::Type(mFlags & ((dot(mNg, mWo) * dot(mNg, wi) > 0.0f) ?
82 ~BsdfLobe::TRANSMISSION : ~BsdfLobe::REFLECTION)));
83 }
84 finline BsdfLobe::Type getSurfaceFlags(const BsdfLobe &lobe, const scene_rdl2::math::Vec3f &wi) const
85 {
86 return (lobe.getIsSpherical() ? mFlags :
87 BsdfLobe::Type(mFlags & ((dot(mNg, mWo) * dot(mNg, wi) > 0.0f) ?
88 ~BsdfLobe::TRANSMISSION : ~BsdfLobe::REFLECTION)));
89 }
90
91 /// Compute a visibility multipilier based on shadow terminator fix mode
92 /// N should be the lobe's shading normal and wi is the incoming light direction
93 float computeShadowTerminatorFix(const scene_rdl2::math::Vec3f &N, const scene_rdl2::math::Vec3f &wi) const;
94
95
96protected:
97 scene_rdl2::math::Vec3f mNg;
98 scene_rdl2::math::Vec3f mWo;
99 BsdfLobe::Type mFlags;
100 ShadowTerminatorFix mShadowTerminatorFix;
101 bool mIncludeCosineTerm;
102 bool mEntering;
103};
