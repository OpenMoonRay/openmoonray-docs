35 void setVelocityGrid(openvdb::VectorGrid::ConstPtr velocityGrid,
36 const std::vector<int>& volumeIds)
37 {
38 mVolumeIdCount = volumeIds.size();
39 for (uint32_t samplerId = 0; samplerId < mVolumeIdCount; ++samplerId) {
40 mVolumeIdToSamplerId[volumeIds[samplerId]] = samplerId;
41 }
42 mVelocityGrid = velocityGrid;
43 unsigned samplerCount = mVolumeIdCount * mcrt_common::getNumTBBThreads();
44 mVelocitySamplers.reserve(samplerCount);
45 for (unsigned i = 0; i < samplerCount; ++i) {
46 mVelocitySamplers.emplace_back(*mVelocityGrid);
47 }
48 mIsMotionBlurOn = true;
49 }

63 // return a vdb position for grid value access, the input position
64 // may get advected by velocity in motion blur case
65 openvdb::Vec3d getEvalPosition(mcrt_common::ThreadLocalState *tls,
66 uint32_t volumeId,
67 const scene_rdl2::math::Vec3f& pSample,
68 float time) const
69 {
70 if (mIsMotionBlurOn) {
71 tls->mGeomTls->mStatistics.incCounter(STATS_VELOCITY_GRID_SAMPLES);
72 uint32_t threadIdx = tls->mThreadIdx;
73 // Use backward advection to retrieve velocity v at time t,
74 // and apply v on position p with backward advection again to
75 // retrieve the final resolved position
76 // For detailed reference see:
77 // "Eulerian Motion Blur"
78 // EGWNP2007 Doyub Kim and Hyeong-Seok Ko
79 // equation (6) and (7)
80 openvdb::Vec3d p(pSample[0], pSample[1], pSample[2]);
81 unsigned samplerIdx = threadIdx * mVolumeIdCount + (mVolumeIdToSamplerId.at(volumeId));
82 openvdb::Vec3s velocity = mVelocitySamplers[samplerIdx].sample(p);
83 float dt = mTShutterOpen + time * mTShutterRange;
84 velocity = mVelocitySamplers[samplerIdx].sample(p - velocity * dt);
85 return p - velocity * dt;
86 } else {
87 return openvdb::Vec3d(pSample[0], pSample[1], pSample[2]);
88 }
89 }
90
91 size_t getMemory()
92 {
93 size_t mem = 0;
94 if (mVelocityGrid) {
95 mem += mVelocityGrid->memUsage();
96 }
97
98 mem += scene_rdl2::util::getVectorElementsMemory(mVelocitySamplers);
99
100 return mem;
101 }
102
103private:
104 openvdb::VectorGrid::ConstPtr mVelocityGrid;
105 std::vector<openvdb::tools::VelocitySampler<openvdb::VectorGrid>> mVelocitySamplers;
106 bool mIsMotionBlurOn;
107 float mTShutterOpen, mTShutterRange;
108 std::unordered_map<uint32_t, uint32_t> mVolumeIdToSamplerId;
109 unsigned mVolumeIdCount;
110};
