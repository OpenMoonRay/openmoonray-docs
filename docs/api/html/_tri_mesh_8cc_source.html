43bool
44TriMesh::bakePosMap(int width, int height, int udim,
45 TypedAttributeKey<Vec2f> stKey,
46 Vec3fa *posResult, Vec3f *nrmResult) const
47{
48 const Attributes *primitiveAttributes = getAttributes();
49 MNRY_ASSERT_REQUIRE(primitiveAttributes->isSupported(stKey));
50 // throw out the anchor boys....
51 if (mIsTessellated) {
52 // triangle got tessellated to quads
53 for (size_t faceId = 0; faceId < getTessellatedMeshFaceCount(); ++faceId) {
54 int baseFaceId = getBaseFaceId(faceId);
55 const uint32_t id0 = mIndices[faceId * sQuadVertexCount];
56 const uint32_t id1 = mIndices[faceId * sQuadVertexCount + 1];
57 const uint32_t id2 = mIndices[faceId * sQuadVertexCount + 2];
58 const uint32_t id3 = mIndices[faceId * sQuadVertexCount + 3];
59
60 const Vec3f pos0 = mVertices(id0).asVec3f();
61 const Vec3f pos1 = mVertices(id1).asVec3f();
62 const Vec3f pos2 = mVertices(id2).asVec3f();
63 const Vec3f pos3 = mVertices(id3).asVec3f();
64
65 // FIXME: we are forced to fetch st1 and st3 twice.
66 Vec2f st0, st1, st2, st3;
67 getTriangleAttributes(stKey, baseFaceId, faceId, st0, st1, st3,
68 0.0f, true);
69 getTriangleAttributes(stKey, baseFaceId, faceId, st2, st3, st1,
70 0.0f, false);
71
72 // check if this face exists in this udim and if so
73 // transform the st coordinates into the normalized range
74 if (udimxform(udim, st0) && udimxform(udim, st1) &&
75 udimxform(udim, st2) && udimxform(udim, st3)) {
76 const Vec3f *nrm0 = nullptr;
77 const Vec3f *nrm1 = nullptr;
78 const Vec3f *nrm2 = nullptr;
79 const Vec3f *nrm3 = nullptr;
80 Vec3f nrmData0, nrmData1, nrmData2, nrmData3;
81 if (nrmResult) {
82 getTriangleNormal(baseFaceId, faceId,
83 nrmData0, nrmData1, nrmData3,
84 0.0f, /* isFirst = */ true);
85 getTriangleNormal(baseFaceId, faceId,
86 nrmData2, nrmData3, nrmData1,
87 0.0f, /* isFirst = */ false);
88 nrm0 = &nrmData0;
89 nrm1 = &nrmData1;
90 nrm2 = &nrmData2;
91 nrm3 = &nrmData3;
92 }
93 // first triangle: <0, 1, 3>
94 scene_rdl2::math::BBox2f roiST(
95 Vec2f(std::min(st0.x, std::min(st1.x, st3.x)),
96 std::min(st0.y, std::min(st1.y, st3.y))),
97 Vec2f(std::max(st0.x, std::max(st1.x, st3.x)),
98 std::max(st0.y, std::max(st1.y, st3.y))));
99 rasterizeTrianglePos(roiST, width, height, st0, st1, st3,
100 pos0, pos1, pos3, nrm0, nrm1, nrm3, posResult, nrmResult);
101 // second triangle: <2, 3, 1>
102 roiST = scene_rdl2::math::BBox2f(
103 Vec2f(std::min(st2.x, std::min(st3.x, st1.x)),
104 std::min(st2.y, std::min(st3.y, st1.y))),
105 Vec2f(std::max(st2.x, std::max(st3.x, st1.x)),
106 std::max(st2.y, std::max(st3.y, st1.y))));
107 rasterizeTrianglePos(roiST, width, height, st2, st3, st1,
108 pos2, pos3, pos1, nrm2, nrm3, nrm1, posResult, nrmResult);
109 }
110 }
111 } else {
112 for (size_t faceId = 0; faceId < getTessellatedMeshFaceCount(); ++faceId) {
113 const uint32_t id0 = mIndices[faceId * sTriangleVertexCount];
114 const uint32_t id1 = mIndices[faceId * sTriangleVertexCount + 1];
115 const uint32_t id2 = mIndices[faceId * sTriangleVertexCount + 2];
116
117 const Vec3f pos0 = mVertices(id0).asVec3f();
118 const Vec3f pos1 = mVertices(id1).asVec3f();
119 const Vec3f pos2 = mVertices(id2).asVec3f();
120
121 Vec2f st0, st1, st2;
122 getTriangleAttributes(stKey, faceId, faceId, st0, st1, st2, 0.0f, true);
123
124 // check if this face exists in this udim and if so,
125 // transform the st coordinates into the normalized range
126 if (udimxform(udim, st0) && udimxform(udim, st1) &&
127 udimxform(udim, st2)) {
128 scene_rdl2::math::BBox2f roiST(
129 Vec2f(std::min(st0.x, std::min(st1.x, st2.x)),
130 std::min(st0.y, std::min(st1.y, st2.y))),
131 Vec2f(std::max(st0.x, std::max(st1.x, st2.x)),
132 std::max(st0.y, std::max(st1.y, st2.y))));
133
134 const Vec3f *nrm0 = nullptr;
135 const Vec3f *nrm1 = nullptr;
136 const Vec3f *nrm2 = nullptr;
137 Vec3f nrmData0, nrmData1, nrmData2;
138 if (nrmResult) {
139 getTriangleNormal(faceId, faceId,
140 nrmData0, nrmData1, nrmData2, 0.0f, true);
141 nrm0 = &nrmData0;
142 nrm1 = &nrmData1;
143 nrm2 = &nrmData2;
144 }
145 rasterizeTrianglePos(roiST, width, height, st0, st1, st2,
146 pos0, pos1, pos2, nrm0, nrm1, nrm2, posResult, nrmResult);
147 }
148 }
149 }
150
151 return true;
152}

154void
155TriMesh::setRequiredAttributes(int primId, float time, float u, float v,
156 float w, bool isFirst, Intersection& intersection) const
157{
158 int tessFaceId = primId;
159 int baseFaceId = getBaseFaceId(tessFaceId);
160 uint32_t id1, id2, id3, id4;
161 uint32_t isecId1, isecId2, isecId3;
162 uint32_t varyingId1, varyingId2, varyingId3;
163 // weights for interpolator
164 float varyingW[3];
165
166 // isFirst only matters when triangle got tessellated to quads
167 if (mIsTessellated) {
168 // triangle got tessellated to quads
169 id1 = mIndices[sQuadVertexCount * tessFaceId ];
170 id2 = mIndices[sQuadVertexCount * tessFaceId + 1];
171 id3 = mIndices[sQuadVertexCount * tessFaceId + 2];
172 id4 = mIndices[sQuadVertexCount * tessFaceId + 3];
173 // the surface uv of intersection point on base face
174 Vec2f uv;
175 if (isFirst) {
176 // Triangle (0,1,3)
177 intersection.setIds(id1, id2, id4);
178 // remap tessellated face barycentric coordinate to
179 // base face barycentric coordinate
180 uv = w * mFaceVaryingUv[sQuadVertexCount * tessFaceId + 0] +
181 u * mFaceVaryingUv[sQuadVertexCount * tessFaceId + 1] +
182 v * mFaceVaryingUv[sQuadVertexCount * tessFaceId + 3];
183 } else {
184 // Triangle (2,3,1)
185 intersection.setIds(id3, id4, id2);
186 // remap tessellated face barycentric coordinate to
187 // base face barycentric coordinate
188 uv = w * mFaceVaryingUv[sQuadVertexCount * tessFaceId + 2] +
189 u * mFaceVaryingUv[sQuadVertexCount * tessFaceId + 3] +
190 v * mFaceVaryingUv[sQuadVertexCount * tessFaceId + 1];
191 }
192 intersection.getIds(isecId1, isecId2, isecId3);
193 varyingId1 = mBaseIndices[sTriangleVertexCount * baseFaceId ];
194 varyingId2 = mBaseIndices[sTriangleVertexCount * baseFaceId + 1];
195 varyingId3 = mBaseIndices[sTriangleVertexCount * baseFaceId + 2];
196 varyingW[0] = 1.0f - uv[0] - uv[1];
197 varyingW[1] = uv[0];
198 varyingW[2] = uv[1];
199 } else {
200 id1 = mIndices[sTriangleVertexCount * tessFaceId ];
201 id2 = mIndices[sTriangleVertexCount * tessFaceId + 1];
202 id3 = mIndices[sTriangleVertexCount * tessFaceId + 2];
203 intersection.setIds(id1, id2, id3);
204 intersection.getIds(isecId1, isecId2, isecId3);
205 varyingId1 = isecId1;
206 varyingId2 = isecId2;
207 varyingId3 = isecId3;
208 varyingW[0] = w;
209 varyingW[1] = u;
210 varyingW[2] = v;
211 }
212
213 int partId = (mFaceToPart.size() > 0) ? mFaceToPart[baseFaceId] : 0;
214 const Attributes* primitiveAttributes = getAttributes();
215 // interpolate PrimitiveAttribute
216 MeshInterpolator interpolator(primitiveAttributes,
217 time, partId, baseFaceId,
218 varyingId1, varyingId2, varyingId3,
219 varyingW[0], varyingW[1], varyingW[2], tessFaceId,
220 isecId1, isecId2, isecId3,
221 w, u, v);
222 intersection.setRequiredAttributes(interpolator);
223
224 if (primitiveAttributes->isSupported(StandardAttributes::sNormal)) {
225 Vec3f N;
226 interpolator.interpolate(StandardAttributes::sNormal,
227 reinterpret_cast<char*>(&N));
228 N.normalize();
229 intersection.setN(N);
230 }

234void
235TriMesh::postIntersect(mcrt_common::ThreadLocalState& tls,
236 const scene_rdl2::rdl2::Layer* pRdlLayer, const mcrt_common::Ray& ray,
237 Intersection& intersection) const
238{
239 int tessFaceId = ray.primID;
240 int baseFaceId = getBaseFaceId(tessFaceId);
241 int assignmentId = getFaceAssignmentId(tessFaceId);
242 intersection.setLayerAssignments(assignmentId, pRdlLayer);
243
244 const auto& material =
245 intersection.getMaterial()->get<shading::RootShader>();
246 const AttributeTable *table = material.getPrimitiveAttributeTable();
247 intersection.setTable(&tls.mArena, table);
248 overrideInstanceAttrs(ray, intersection);
249
250 // barycentric coordinate
251 float u = ray.u;
252 float v = ray.v;
253 float w = 1.0f - u - v;
254 bool isFirst = w > 0.0f;
255 if (!isFirst && mIsTessellated) {
256 u = 1.0f - u;
257 v = 1.0f - v;
258 w = -w;
259 }
260
261 setRequiredAttributes(tessFaceId, ray.time, u, v, w, isFirst, intersection);
262 uint32_t isecId1, isecId2, isecId3;
263 intersection.getIds(isecId1, isecId2, isecId3);
264
265 const Attributes* primitiveAttributes = getAttributes();
266 Vec3f Ng, N, dPds, dPdt;
267 Vec2f St;
268 Ng = normalize(ray.getNg());
269 // shading normal
270 if (primitiveAttributes->isSupported(StandardAttributes::sNormal)) {
271 N = intersection.getN();
272 } else {
273 N = Ng;
274 }
275
276 Vec2f st1, st2, st3;
277 getTriangleST(baseFaceId, tessFaceId, st1, st2, st3, isFirst);
278 St = w * st1 + u * st2 + v * st3;
279
280 if (isMotionBlurOn()) {
281 float t = ray.time;
282 const Vec3f& p10 = mVertices(isecId1, 0);
283 const Vec3f& p20 = mVertices(isecId2, 0);
284 const Vec3f& p30 = mVertices(isecId3, 0);
285 // form a ReferenceFrame and use its tangent/binormal as dpds dpdt
286 // if we fail to calculate the st partial derivatives
287 Vec3f dp0[2];
288 if (!computeTrianglePartialDerivatives(
289 p10, p20, p30, st1, st2, st3, dp0)) {
290 scene_rdl2::math::ReferenceFrame frame(N);
291 dp0[0] = frame.getX();
292 dp0[1] = frame.getY();
293 }
294 const Vec3f& p11 = mVertices(isecId1, 1);
295 const Vec3f& p21 = mVertices(isecId2, 1);
296 const Vec3f& p31 = mVertices(isecId3, 1);
297 Vec3f dp1[2];
298 if (!computeTrianglePartialDerivatives(
299 p11, p21, p31, st1, st2, st3, dp1)) {
300 scene_rdl2::math::ReferenceFrame frame(N);
301 dp1[0] = frame.getX();
302 dp1[1] = frame.getY();
303 }
304 dPds = (1.0f - t) * dp0[0] + t * dp1[0];
305 dPdt = (1.0f - t) * dp0[1] + t * dp1[1];
306 } else {
307 const Vec3f& p1 = mVertices(isecId1);
308 const Vec3f& p2 = mVertices(isecId2);
309 const Vec3f& p3 = mVertices(isecId3);
310 // form a ReferenceFrame and use its tangent/binormal as dpds dpdt
311 // if we fail to calculate the st partial derivatives
312 Vec3f dp[2];
313 if (!computeTrianglePartialDerivatives(p1, p2, p3, st1, st2, st3, dp)) {
314 scene_rdl2::math::ReferenceFrame frame(N);
315 dp[0] = frame.getX();
316 dp[1] = frame.getY();
317 }
318 dPds = dp[0];
319 dPdt = dp[1];
320 }
321
322 intersection.setDifferentialGeometry(Ng, N, St, dPds, dPdt, true);
323
324 // calculate dfds/dfdt for primitive attributes that request differential
325 if (table->requestDerivatives()) {
326 computeAttributesDerivatives(table, st1, st2, st3,
327 baseFaceId, tessFaceId, ray.time, isFirst, intersection);
328 }
329
330 // fill in ray epsilon to avoid self intersection
331 const scene_rdl2::rdl2::Geometry* geometry = intersection.getGeometryObject();
332 MNRY_ASSERT(geometry != nullptr);
333 intersection.setEpsilonHint(geometry->getRayEpsilon());
334
335 // polygon vertices
336 int numVertices = mIsTessellated ? 4 : 3;
337 if (table->requests(StandardAttributes::sNumPolyVertices)) {
338 intersection.setAttribute(StandardAttributes::sNumPolyVertices, numVertices);
339 }
340 if (table->requests(StandardAttributes::sPolyVertexType)) {
341 intersection.setAttribute(StandardAttributes::sPolyVertexType,
342 static_cast<int>(StandardAttributes::POLYVERTEX_TYPE_POLYGON));
343 }
344 for (int iVert = 0; iVert < numVertices; iVert++) {
345 if (table->requests(StandardAttributes::sPolyVertices[iVert])) {
346 uint32_t id = mIndices[numVertices * tessFaceId + iVert];
347 // may need to move the vertices to render space
348 // for instancing object since they are ray traced in local space
349 const Vec3f v = ray.isInstanceHit() ? transformPoint(ray.ext.l2r, mVertices(id).asVec3f())
350 : mVertices(id).asVec3f();
351 intersection.setAttribute(StandardAttributes::sPolyVertices[iVert], v);
352 }
353 }
354
355 // motion vectors
356 if (table->requests(StandardAttributes::sMotion)) {
357 const Vec3f motion = computeMotion(mVertices, isecId1, isecId2, isecId3, w, u, v, ray);
358 intersection.setAttribute(StandardAttributes::sMotion, motion);
359 }
360}

362bool
363TriMesh::computeIntersectCurvature(const mcrt_common::Ray& ray,
364 const Intersection& intersection, Vec3f& dnds, Vec3f& dndt) const
365{
366 int tessFaceId = ray.primID;
367 int baseFaceId = getBaseFaceId(tessFaceId);
368 uint32_t id1, id2, id3;
369 intersection.getIds(id1, id2, id3);
370 bool isFirst = ray.u + ray.v <= 1.0f;
371 Vec2f st1, st2, st3;
372 getTriangleST(baseFaceId, tessFaceId, st1, st2, st3, isFirst);
373 Vec3f n1, n2, n3;
374 getTriangleNormal(baseFaceId, tessFaceId, n1, n2, n3, ray.time, isFirst);
375 Vec3f dn[2];
376 if (computeTrianglePartialDerivatives(n1, n2, n3, st1, st2, st3, dn)) {
377 dnds = dn[0];
378 dndt = dn[1];
379 return true;
380 } else {
381 return false;
382 }

385void
386TriMesh::getST(int tessFaceId, float u, float v, Vec2f& st) const
387{
388 int baseFaceId = getBaseFaceId(tessFaceId);
389 bool isFirst = u + v <= 1.0f;
390 float w = 1.0f - u - v;
391 Vec2f st1, st2, st3;
392 getTriangleST(baseFaceId, tessFaceId, st1, st2, st3, isFirst);
393 st = w * st1 + u * st2 + v * st3;
394}

396void
397TriMesh::getNeighborVertices(int baseFaceId, int tessFaceId,
398 int tessFaceVIndex, int& vid, int& vid1, int& vid2, int& vid3,
399 Vec2f& st, Vec2f& st1, Vec2f& st2, Vec2f& st3) const
400{
401 if (mIsTessellated) {
402 size_t offset = tessFaceId * sQuadVertexCount;
403 if (tessFaceVIndex == 2) {
404 if (mIndices[offset + 2] == mIndices[offset + 3]) {
405 // the degened quad introduced during tessellation
406 // first triangle on quad (0, 1, 3), which is (0, 1, 2)
407 getTriangleST(baseFaceId, tessFaceId, st1, st2, st3, true);
408 vid1 = mIndices[offset + 0];
409 vid2 = mIndices[offset + 1];
410 vid3 = mIndices[offset + 2];
411 vid = vid3;
412 st = st3;
413 } else {
414 // second triangle on quad (2, 3, 1)
415 getTriangleST(baseFaceId, tessFaceId, st1, st2, st3, false);
416 vid1 = mIndices[offset + 2];
417 vid2 = mIndices[offset + 3];
418 vid3 = mIndices[offset + 1];
419 vid = vid1;
420 st = st1;
421 }
422 } else {
423 // first triangle on quad (0, 1, 3)
424 getTriangleST(baseFaceId, tessFaceId, st1, st2, st3, true);
425 vid1 = mIndices[offset + 0];
426 vid2 = mIndices[offset + 1];
427 vid3 = mIndices[offset + 3];
428 if (tessFaceVIndex == 0) {
429 vid = vid1;
430 st = st1;
431 } else if (tessFaceVIndex == 1) {
432 vid = vid2;
433 st = st2;
434 } else if (tessFaceVIndex == 3) {
435 vid = vid3;
436 st = st3;
437 }
438 }
439 } else {
440 size_t offset = tessFaceId * sTriangleVertexCount;
441 getTriangleST(baseFaceId, tessFaceId, st1, st2, st3, true);
442 vid1 = mIndices[offset + 0];
443 vid2 = mIndices[offset + 1];
444 vid3 = mIndices[offset + 2];
445 if (tessFaceVIndex == 0) {
446 vid = vid1;
447 st = st1;
448 } else if (tessFaceVIndex == 1) {
449 vid = vid2;
450 st = st2;
451 } else if (tessFaceVIndex == 2) {
452 vid = vid3;
453 st = st3;
454 }
455 }

458void
459TriMesh::splitNGons(size_t outputFaceCount,
460 PolygonMesh::FaceToPartBuffer& faceToPart,
461 const PolygonMesh::FaceVertexCount& faceVertexCount,
462 PolygonMesh::IndexBuffer& indices,
463 LayerAssignmentId& layerAssignmentId,
464 PrimitiveAttributeTable& primitiveAttributeTable)
465{
466 size_t inputFaceCount = faceVertexCount.size();
467 size_t inputIndexCount = indices.size();
468 indices.resize(sTriangleVertexCount * outputFaceCount);
469 if (faceToPart.size() > 0) {
470 faceToPart.resize(outputFaceCount);
471 }
472
473 size_t inputIndexOffset = inputIndexCount;
474 size_t outputIndexOffset = indices.size() - 1;
475 size_t outputF2POffset = faceToPart.size() - 1;
476 // triangulate indices
477 for (int f = inputFaceCount - 1; f >= 0; --f) {
478 size_t fvCount = faceVertexCount[f];
479 inputIndexOffset -= fvCount;
480 for (size_t v = fvCount; v >= sTriangleVertexCount; --v) {
481 // triangulate a ngon into triangles fan
482 indices[outputIndexOffset--] = indices[inputIndexOffset + v - 1];
483 indices[outputIndexOffset--] = indices[inputIndexOffset + v - 2];
484 indices[outputIndexOffset--] = indices[inputIndexOffset];
485 if (faceToPart.size() > 0) {
486 faceToPart[outputF2POffset--] = faceToPart[f];
487 }
488 }
489 }
490
491 // triangulate layerAssignmentId if it's in per face frequency
492 if (layerAssignmentId.getType() == LayerAssignmentId::Type::VARYING) {
493 auto& faceAssignmentIds = layerAssignmentId.getVaryingId();
494 faceAssignmentIds.resize(outputFaceCount);
495 size_t outputFaceOffset = outputFaceCount - 1;
496 for (int f = inputFaceCount - 1; f >= 0; --f) {
497 size_t fvCount = faceVertexCount[f];
498 for (size_t v = fvCount; v >= sTriangleVertexCount; --v) {
499 faceAssignmentIds[outputFaceOffset--] = faceAssignmentIds[f];
500 }
501 }
502 }
503
504 // triangulate uniform and facevarying primitive attributes
505 for (auto& kv : primitiveAttributeTable) {
506 for (auto& attribute : kv.second) {
507 AttributeRate rate = attribute->getRate();
508 if (rate == RATE_UNIFORM) {
509 attribute->resize(outputFaceCount);
510 size_t outputFaceOffset = outputFaceCount - 1;
511 for (int f = inputFaceCount - 1; f >= 0; --f) {
512 size_t fvCount = faceVertexCount[f];
513 for (size_t v = fvCount; v >= sTriangleVertexCount; --v) {
514 attribute->copyInPlace(f, outputFaceOffset--);
515 }
516 }
517 } else if (rate == RATE_FACE_VARYING) {
518 attribute->resize(sTriangleVertexCount * outputFaceCount);
519 size_t inputIndexOffset = inputIndexCount;
520 size_t outputIndexOffset = indices.size() - 1;
521 for (int f = inputFaceCount - 1; f >= 0; --f) {
522 size_t fvCount = faceVertexCount[f];
523 inputIndexOffset -= fvCount;
524 for (size_t v = fvCount; v >= sTriangleVertexCount; --v) {
525 // triangulate a ngon into triangles fan
526 attribute->copyInPlace(inputIndexOffset + v - 1,
527 outputIndexOffset--);
528 attribute->copyInPlace(inputIndexOffset + v - 2,
529 outputIndexOffset--);
530 attribute->copyInPlace(inputIndexOffset,
531 outputIndexOffset--);
532 }
533 }
534 }
535 }
536 }

539void
540TriMesh::generateIndexBufferAndSurfaceSamples(
541 const std::vector<PolyFaceTopology>& faceTopologies,
542 const PolyTessellatedVertexLookup& tessellatedVertexLookup,
543 PolygonMesh::IndexBuffer& indices,
544 std::vector<PolyMesh::SurfaceSample>& surfaceSamples,
545 std::vector<int>& tessellatedToBaseFace,
546 std::vector<Vec2f>* faceVaryingUv) const
547{
548 tessellatedToBaseFace.clear();
549 // we need to use arena to allocate temporary data structure for
550 // outer/interior ring stitching
551 mcrt_common::ThreadLocalState *tls = mcrt_common::getFrameUpdateTLS();
552 scene_rdl2::alloc::Arena *arena = &tls->mArena;
553
554 size_t tessellatedVertexCount =
555 tessellatedVertexLookup.getTessellatedVertexCount();
556 surfaceSamples.resize(tessellatedVertexCount);
557 // please refer to OpenGL spec for triangle tessellation detail
558 // 11.2.2.1 Triangle Tessellation p371
559 // https://www.khronos.org/registry/OpenGL/specs/gl/glspec44.core.pdf
560 for (size_t f = 0; f < faceTopologies.size(); ++f) {
561 const PolyFaceTopology& faceTopology = faceTopologies[f];
562 int maxEdgeVertexCount =
563 tessellatedVertexLookup.getMaxEdgeVertexCount(faceTopology);
564 // simplest case
565 if (maxEdgeVertexCount == 0) {
566 int vid0 = tessellatedVertexLookup.getTessellatedVertexId(
567 faceTopology.mCornerVertexId[0]);
568 int vid1 = tessellatedVertexLookup.getTessellatedVertexId(
569 faceTopology.mCornerVertexId[1]);
570 int vid2 = tessellatedVertexLookup.getTessellatedVertexId(
571 faceTopology.mCornerVertexId[2]);
572 indices.push_back(vid0);
573 indices.push_back(vid1);
574 indices.push_back(vid2);
575 indices.push_back(vid2);
576 if (faceVaryingUv) {
577 faceVaryingUv->emplace_back(0.0f, 0.0f);
578 faceVaryingUv->emplace_back(1.0f, 0.0f);
579 faceVaryingUv->emplace_back(0.0f, 1.0f);
580 faceVaryingUv->emplace_back(0.0f, 1.0f);
581 }
582 tessellatedToBaseFace.push_back(f);
583 surfaceSamples[vid0].mFaceId = f;
584 surfaceSamples[vid0].mUv = Vec2f(0.0f, 0.0f);
585 surfaceSamples[vid1].mFaceId = f;
586 surfaceSamples[vid1].mUv = Vec2f(1.0f, 0.0f);
587 surfaceSamples[vid2].mFaceId = f;
588 surfaceSamples[vid2].mUv = Vec2f(0.0f, 1.0f);
589 continue;
590 }
591 // TODO handle uniform tessellation in more optimized topology
592 // interior region index buffer first.
593 int interiorRingCount =
594 tessellatedVertexLookup.getTriangleRingCount(faceTopology) - 1;
595 // how many rings to pass through from outer vertex to central
596 // with barycentric coordinate shifting from 0 to 1/3
597 float ringUvDelta = 2.0f / (3.0f * (float)(maxEdgeVertexCount + 1));
598 for (int r = 0; r < interiorRingCount - 1; ++r) {
599 // use the same barycentric coordinate encoding embree does
600 // (the weight of v1 and v2, so three corner vertices v0, v1, v2
601 // with barycentric coordinates (1, 0, 0), (0, 1, 0), (0, 0, 1) are
602 // encoded as (0, 0), (1, 0), (0, 1)
603
604 // cornerUv0 is the corner barycentric coordinates in the
605 // current interior ring
606 // cornerUv1 is the corner barycentric coordinates in the
607 // next interior ring (closer to center)
608 Vec2f cornerUv0[3] = {
609 Vec2f((r + 1) * ringUvDelta, (r + 1) * ringUvDelta),
610 Vec2f(1.0f - 2 * (r + 1) * ringUvDelta, (r + 1) * ringUvDelta),
611 Vec2f((r + 1) * ringUvDelta, 1.0f - 2 * (r + 1) * ringUvDelta)};
612 Vec2f cornerUv1[3] = {
613 Vec2f((r + 2) * ringUvDelta, (r + 2) * ringUvDelta),
614 Vec2f(1.0f - 2 * (r + 2) * ringUvDelta, (r + 2) * ringUvDelta),
615 Vec2f((r + 2) * ringUvDelta, 1.0f - 2 * (r + 2) * ringUvDelta)};
616 int edgeVertexCount = maxEdgeVertexCount - 2 * (r + 1);
617 float w0 = 1.0f / (float)(edgeVertexCount + 1);
618 float w1 = 1.0f / (float)(edgeVertexCount - 1);
619 for (size_t c = 0; c < sTriangleVertexCount; ++c) {
620 // corner quad
621 indices.push_back(
622 tessellatedVertexLookup.getTriangleInteriorVertexId(
623 f, maxEdgeVertexCount, r, c, 0));
624 indices.push_back(
625 tessellatedVertexLookup.getTriangleInteriorVertexId(
626 f, maxEdgeVertexCount, r, c, 1));
627 indices.push_back(
628 tessellatedVertexLookup.getTriangleInteriorVertexId(
629 f, maxEdgeVertexCount, r + 1, c, 0));
630 indices.push_back(
631 tessellatedVertexLookup.getTriangleInteriorVertexId(
632 f, maxEdgeVertexCount, r, (c + 2) % sTriangleVertexCount,
633 edgeVertexCount));
634 if (faceVaryingUv) {
635 faceVaryingUv->push_back(cornerUv0[c]);
636 faceVaryingUv->push_back(
637 (1.0f - w0) * cornerUv0[c] +
638 ( w0) * cornerUv0[(c + 1) % sTriangleVertexCount]);
639 faceVaryingUv->push_back(cornerUv1[c]);
640 faceVaryingUv->push_back(
641 (1.0f - w0) * cornerUv0[c] +
642 ( w0) * cornerUv0[(c + 2) % sTriangleVertexCount]);
643 }
644 tessellatedToBaseFace.push_back(f);
645 // edge quads
646 for (int n = 1; n < edgeVertexCount; ++n) {
647 indices.push_back(
648 tessellatedVertexLookup.getTriangleInteriorVertexId(
649 f, maxEdgeVertexCount, r, c, n));
650 indices.push_back(
651 tessellatedVertexLookup.getTriangleInteriorVertexId(
652 f, maxEdgeVertexCount, r, c, n + 1));
653 indices.push_back(
654 tessellatedVertexLookup.getTriangleInteriorVertexId(
655 f, maxEdgeVertexCount, r + 1, c, n));
656 indices.push_back(
657 tessellatedVertexLookup.getTriangleInteriorVertexId(
658 f, maxEdgeVertexCount, r + 1, c, n - 1));
659 if (faceVaryingUv) {
660 faceVaryingUv->push_back(
661 (1.0f - (n ) * w0) * cornerUv0[c] +
662 ( (n ) * w0) * cornerUv0[(c + 1) % sTriangleVertexCount]);
663 faceVaryingUv->push_back(
664 (1.0f - (n + 1) * w0) * cornerUv0[c] +
665 ( (n + 1) * w0) * cornerUv0[(c + 1) % sTriangleVertexCount]);
666 faceVaryingUv->push_back(
667 (1.0f - (n ) * w1) * cornerUv1[c] +
668 ( (n ) * w1) * cornerUv1[(c + 1) % sTriangleVertexCount]);
669 faceVaryingUv->push_back(
670 (1.0f - (n - 1) * w1) * cornerUv1[c] +
671 ( (n - 1) * w1) * cornerUv1[(c + 1) % sTriangleVertexCount]);
672 }
673 tessellatedToBaseFace.push_back(f);
674 }
675 for (int n = 0; n < edgeVertexCount + 1; ++n) {
676 int vid =
677 tessellatedVertexLookup.getTriangleInteriorVertexId(
678 f, maxEdgeVertexCount, r, c, n);
679 surfaceSamples[vid].mFaceId = f;
680 surfaceSamples[vid].mUv =
681 (1.0f - n * w0) * cornerUv0[c] +
682 ( n * w0) * cornerUv0[(c + 1) % sTriangleVertexCount];
683 }
684 }
685 }
686 // few remaining interior central edge cases
687 if (maxEdgeVertexCount % 2 == 0) {
688 int vid0 =
689 tessellatedVertexLookup.getTriangleInteriorVertexId(
690 f, maxEdgeVertexCount, interiorRingCount - 1, 0, 0);
691 int vid1 =
692 tessellatedVertexLookup.getTriangleInteriorVertexId(
693 f, maxEdgeVertexCount, interiorRingCount - 1, 1, 0);
694 int vid2 =
695 tessellatedVertexLookup.getTriangleInteriorVertexId(
696 f, maxEdgeVertexCount, interiorRingCount - 1, 2, 0);
697 Vec2f uv0(
698 interiorRingCount * ringUvDelta,
699 interiorRingCount * ringUvDelta);
700 Vec2f uv1(
701 1.0f - 2 * interiorRingCount * ringUvDelta,
702 interiorRingCount * ringUvDelta);
703 Vec2f uv2(
704 interiorRingCount * ringUvDelta,
705 1.0f - 2 * interiorRingCount * ringUvDelta);
706 // one more central triangle
707 indices.push_back(vid0);
708 indices.push_back(vid1);
709 indices.push_back(vid2);
710 indices.push_back(vid2);
711 if (faceVaryingUv) {
712 faceVaryingUv->push_back(uv0);
713 faceVaryingUv->push_back(uv1);
714 faceVaryingUv->push_back(uv2);
715 faceVaryingUv->push_back(uv2);
716 }
717 tessellatedToBaseFace.push_back(f);
718
719 surfaceSamples[vid0].mFaceId = f;
720 surfaceSamples[vid0].mUv = uv0;
721 surfaceSamples[vid1].mFaceId = f;
722 surfaceSamples[vid1].mUv = uv1;
723 surfaceSamples[vid2].mFaceId = f;
724 surfaceSamples[vid2].mUv = uv2;
725 } else {
726 // one more central point
727 int vid =
728 tessellatedVertexLookup.getTriangleInteriorVertexId(
729 f, maxEdgeVertexCount, interiorRingCount - 1, 0, 0);
730 surfaceSamples[vid].mFaceId = f;
731 surfaceSamples[vid].mUv = Vec2f(1.0f / 3.0f, 1.0f / 3.0f);
732 }
733
734 // stitch interior and outer ring edge
735 // *---*--*--*---*
736 // / \ / \ | / \ / \
737 // *---*----*----*---*
738 const Vec2f cornerUv[3] = {
739 Vec2f(0.0f, 0.0f), Vec2f(1.0f, 0.0f), Vec2f(0.0f, 1.0f)};
740 for (size_t c = 0; c < sTriangleVertexCount; ++c) {
741 SCOPED_MEM(arena);
742 StitchPoint* innerRing = nullptr;
743 StitchPoint* outerRing = nullptr;
744 Vec2f* innerRingUv = nullptr;
745 Vec2f* outerRingUv = nullptr;
746 int innerRingVertexCount, outerRingVertexCount;
747 generateTriangleStitchRings(arena, faceTopology, f, c,
748 tessellatedVertexLookup,
749 maxEdgeVertexCount, ringUvDelta,
750 innerRing, innerRingUv, innerRingVertexCount,
751 outerRing, outerRingUv, outerRingVertexCount);
752 stitchRings(innerRing, innerRingUv, innerRingVertexCount,
753 outerRing, outerRingUv, outerRingVertexCount, indices,
754 f, &tessellatedToBaseFace, faceVaryingUv);
755 // surface samples
756 float w = 1.0f / (float)(outerRingVertexCount - 1);
757 for (int v = 0; v < outerRingVertexCount - 1; ++v) {
758 int vid = outerRing[v].mVertexId;
759 surfaceSamples[vid].mFaceId = f;
760 surfaceSamples[vid].mUv =
761 (1.0f - v * w) * cornerUv[c] +
762 ( v * w) * cornerUv[(c + 1) % sTriangleVertexCount];
763 }
764 }
765 }
766}

768PolygonMesh::VertexBuffer
769TriMesh::generateVertexBuffer(
770 const PolygonMesh::VertexBuffer& baseVertices,
771 const PolygonMesh::IndexBuffer& baseIndices,
772 const std::vector<PolyMesh::SurfaceSample>& surfaceSamples) const
773{
774 // allocate tessellated vertex/index buffer to hold the evaluation result
775 size_t tessellatedVertexCount = surfaceSamples.size();
776 size_t motionSampleCount = baseVertices.get_time_steps();
777 PolygonMesh::VertexBuffer tessellatedVertices(
778 tessellatedVertexCount, motionSampleCount);
779 tbb::blocked_range<size_t> range =
780 tbb::blocked_range<size_t>(0, tessellatedVertexCount);
781 tbb::parallel_for(range, [&](const tbb::blocked_range<size_t> &r) {
782 for (size_t i = r.begin(); i < r.end(); ++i) {
783 int fid = surfaceSamples[i].mFaceId;
784 int vid0 = baseIndices[sTriangleVertexCount * fid + 0];
785 int vid1 = baseIndices[sTriangleVertexCount * fid + 1];
786 int vid2 = baseIndices[sTriangleVertexCount * fid + 2];
787 // bilinear interpolate the new position
788 Vec2f uv = surfaceSamples[i].mUv;
789 float u = uv[0];
790 float v = uv[1];
791 float w = 1.0f - u - v;
792 for (size_t t = 0; t < motionSampleCount; ++t) {
793 tessellatedVertices(i, t) =
794 w * baseVertices(vid0, t) +
795 u * baseVertices(vid1, t) +
796 v * baseVertices(vid2, t);
797 }
798 }
799 });
800 return tessellatedVertices;
801}

945template <typename T> T
946TriMesh::getAttribute(const TypedAttributeKey<T>& key,
947 int baseFaceId, int tessFaceId, int vIndex, float time) const
948{
949 const Attributes* attributes = getAttributes();
950 MNRY_ASSERT(attributes->isSupported(key));
951
952 T result;
953 AttributeRate rate = attributes->getRate(key);
954 switch (rate) {
955 case RATE_CONSTANT:
956 result = getConstantAttribute(key, time);
957 break;
958 case RATE_UNIFORM:
959 result = getUniformAttribute(key, baseFaceId, time);
960 break;
961 case RATE_VARYING:
962 {
963 if (mIsTessellated) {
964 size_t offset = sQuadVertexCount * tessFaceId;
965 Vec2f uv = mFaceVaryingUv[offset + vIndex];
966 size_t baseOffset = sTriangleVertexCount * baseFaceId;
967 T a1 = getVaryingAttribute(
968 key, mBaseIndices[baseOffset + 0], time);
969 T a2 = getVaryingAttribute(
970 key, mBaseIndices[baseOffset + 1], time);
971 T a3 = getVaryingAttribute(
972 key, mBaseIndices[baseOffset + 2], time);
973 result = (1.0f - uv[0] - uv[1]) * a1 + uv[0] * a2 + uv[1] * a3;
974 } else {
975 result = getVaryingAttribute(
976 key, mIndices[sTriangleVertexCount * baseFaceId + vIndex], time);
977 }
978 }
979 break;
980 case RATE_FACE_VARYING:
981 {
982 if (mIsTessellated) {
983 size_t tessOffset = sQuadVertexCount * tessFaceId;
984 Vec2f uv = mFaceVaryingUv[tessOffset + vIndex];
985 T a1 = getFaceVaryingAttribute(key, baseFaceId, 0, time);
986 T a2 = getFaceVaryingAttribute(key, baseFaceId, 1, time);
987 T a3 = getFaceVaryingAttribute(key, baseFaceId, 2, time);
988 result = (1.0f - uv[0] - uv[1]) * a1 + uv[0] * a2 + uv[1] * a3;
989 } else {
990 result = getFaceVaryingAttribute(key, baseFaceId, vIndex, time);
991 }
992 }
993 break;
994 case RATE_VERTEX:
995 {
996 if (mIsTessellated) {
997 result = getVertexAttribute(
998 key, mIndices[sQuadVertexCount * tessFaceId + vIndex], time);
999 } else {
1000 result = getVertexAttribute(
1001 key, mIndices[sTriangleVertexCount * baseFaceId + vIndex], time);
1002 }
1003 }
1004 break;
1005 default:
1006 MNRY_ASSERT(false, "unknown attribute rate");
1007 break;
1008 }
1009 return result;
1010}

1012template <typename T> bool
1013TriMesh::getTriangleAttributes(const TypedAttributeKey<T>& key,
1014 int baseFaceId, int tessFaceId, T& v1, T& v2, T& v3,
1015 float time, bool isFirst) const
1016{
1017 int index1, index2, index3;
1018 if (mIsTessellated) {
1019 if (isFirst) {
1020 index1 = 0;
1021 index2 = 1;
1022 index3 = 3;
1023 } else {
1024 index1 = 2;
1025 index2 = 3;
1026 index3 = 1;
1027 }
1028 } else {
1029 index1 = 0;
1030 index2 = 1;
1031 index3 = 2;
1032 }
1033 const Attributes* attributes = getAttributes();
1034 if (attributes->isSupported(key)) {
1035 AttributeRate rate = attributes->getRate(key);
1036 switch (rate) {
1037 case RATE_CONSTANT:
1038 v3 = v2 = v1 = getConstantAttribute(key, time);
1039 break;
1040 case RATE_UNIFORM:
1041 v3 = v2 = v1 = getUniformAttribute(key, baseFaceId, time);
1042 break;
1043 case RATE_VARYING:
1044 {
1045 if (mIsTessellated) {
1046 size_t offset = sQuadVertexCount * tessFaceId;
1047 Vec2f uv1 = mFaceVaryingUv[offset + index1];
1048 Vec2f uv2 = mFaceVaryingUv[offset + index2];
1049 Vec2f uv3 = mFaceVaryingUv[offset + index3];
1050 size_t baseOffset = sTriangleVertexCount * baseFaceId;
1051 T a1 = getVaryingAttribute(
1052 key, mBaseIndices[baseOffset + 0], time);
1053 T a2 = getVaryingAttribute(
1054 key, mBaseIndices[baseOffset + 1], time);
1055 T a3 = getVaryingAttribute(
1056 key, mBaseIndices[baseOffset + 2], time);
1057 v1 = (1.0f - uv1[0] - uv1[1]) * a1 +
1058 uv1[0] * a2 + uv1[1] * a3;
1059 v2 = (1.0f - uv2[0] - uv2[1]) * a1 +
1060 uv2[0] * a2 + uv2[1] * a3;
1061 v3 = (1.0f - uv3[0] - uv3[1]) * a1 +
1062 uv3[0] * a2 + uv3[1] * a3;
1063 } else {
1064 size_t offset = sTriangleVertexCount * baseFaceId;
1065 v1 = getVaryingAttribute(
1066 key, mIndices[offset + index1], time);
1067 v2 = getVaryingAttribute(
1068 key, mIndices[offset + index2], time);
1069 v3 = getVaryingAttribute(
1070 key, mIndices[offset + index3], time);
1071 }
1072 }
1073 break;
1074 case RATE_FACE_VARYING:
1075 {
1076 if (mIsTessellated) {
1077 size_t tessOffset = sQuadVertexCount * tessFaceId;
1078 Vec2f uv1 = mFaceVaryingUv[tessOffset + index1];
1079 Vec2f uv2 = mFaceVaryingUv[tessOffset + index2];
1080 Vec2f uv3 = mFaceVaryingUv[tessOffset + index3];
1081 T a1 = getFaceVaryingAttribute(key, baseFaceId, 0, time);
1082 T a2 = getFaceVaryingAttribute(key, baseFaceId, 1, time);
1083 T a3 = getFaceVaryingAttribute(key, baseFaceId, 2, time);
1084 v1 = (1.0f - uv1[0] - uv1[1]) * a1 +
1085 uv1[0] * a2 + uv1[1] * a3;
1086 v2 = (1.0f - uv2[0] - uv2[1]) * a1 +
1087 uv2[0] * a2 + uv2[1] * a3;
1088 v3 = (1.0f - uv3[0] - uv3[1]) * a1 +
1089 uv3[0] * a2 + uv3[1] * a3;
1090 } else {
1091 v1 = getFaceVaryingAttribute(key, baseFaceId, 0, time);
1092 v2 = getFaceVaryingAttribute(key, baseFaceId, 1, time);
1093 v3 = getFaceVaryingAttribute(key, baseFaceId, 2, time);
1094 }
1095 }
1096 break;
1097 case RATE_VERTEX:
1098 {
1099 size_t offset = mIsTessellated ?
1100 sQuadVertexCount * tessFaceId :
1101 sTriangleVertexCount * tessFaceId;
1102 v1 = getVertexAttribute(key, mIndices[offset + index1], time);
1103 v2 = getVertexAttribute(key, mIndices[offset + index2], time);
1104 v3 = getVertexAttribute(key, mIndices[offset + index3], time);
1105 }
1106 break;
1107 default:
1108 MNRY_ASSERT(false, "unknown attribute rate");
1109 break;
1110 }
1111 return true;
1112 }
1113 return false;
1114}

1116void
1117TriMesh::getTriangleST(int baseFaceId, int tessFaceId,
1118 Vec2f& st1, Vec2f& st2, Vec2f& st3, bool isFirst) const
1119{
1120 if (!getTriangleAttributes(StandardAttributes::sSurfaceST,
1121 baseFaceId, tessFaceId, st1, st2, st3, 0.0f, isFirst)) {
1122 if (mIsTessellated) {
1123 if (isFirst) {
1124 st1 = Vec2f(0.0f, 0.0f);
1125 st2 = Vec2f(1.0f, 0.0f);
1126 st3 = Vec2f(0.0f, 1.0f);
1127 } else {
