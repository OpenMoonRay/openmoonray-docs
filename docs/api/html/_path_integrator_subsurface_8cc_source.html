57// Utility Function to decide if we can substitute expensive subsurface computation
58// with a cheaper approximation instead. Right now, we check if
59// the ray-footprint is bigger than the 'scattering radius' and if so,
60// we replace subsurface with Lambertian.
61// Note a similar user-controlled substitution happens with a
62// 'max subsurface per path' parameter.
63// This substitution is an optimization for when the full subsurface computation
64// has no visual effect since the ray-footprint being approximated is
65// bigger than the scattering radius.
66finline bool
67canSubstituteSubsurface(float sssRadius,
68 const shading::Intersection &isect)
69{
70 const scene_rdl2::math::Vec3f dPds = isect.getdPds();
71 const scene_rdl2::math::Vec3f dPdt = isect.getdPdt();
72 const scene_rdl2::math::Vec3f dPdx = dPds * isect.getdSdx() + dPdt * isect.getdTdx();
73 const scene_rdl2::math::Vec3f dPdy = dPds * isect.getdSdy() + dPdt * isect.getdTdy();
74 // Avg ray footprint length (faster than actual area computation)
75 float rayFootprintRSqr = (lengthSqr(dPdx) + lengthSqr(dPdy)) * 0.5f;
76 float ratioFootprintToRadius = rayFootprintRSqr / (sssRadius * sssRadius);
77 // 0.85f is an empirical threshold based on local tests.
78 // It could be modified in the future if we encounter any issues.
79 // So far, disabling subsurface when the ray-footprint's around 85% of the
80 // scattering radius seems like a good enough approximation for optimum speed.
81 static const float SSS_THRESHOLD = 0.85f;
82 return (ratioFootprintToRadius > SSS_THRESHOLD);
83}

92// Sample direct and indirect lighting at a given subsurface sample and compute
93// the reflected radiance. We apply the given pathThroughput and
94// transmissionFresnel, if any, to the contribution of the samples.
95scene_rdl2::math::Color
96PathIntegrator::computeRadianceSubsurfaceSample(pbr::TLState *pbrTls,
97 const Bsdf &bsdf, const Subpixel &sp, int cameraId,
98 const PathVertex &pv, const RayDifferential &parentRay,
99 const scene_rdl2::math::Vec3f &dNdx, const scene_rdl2::math::Vec3f &dNdy,
100 const scene_rdl2::math::Color &pathThroughput, const Fresnel *transmissionFresnel,
101 const LightSet &lightSet, BsdfLobe &lobe,
102 const BsdfSlice &slice, const scene_rdl2::math::Vec3f &P, const scene_rdl2::math::Vec3f &N,
103 int subsurfaceSplitFactor, int computeRadianceSplitFactor,
104 int subsurfaceIndex, bool doIndirect, float rayEpsilon, float shadowRayEpsilon,
105 unsigned sssSampleID, unsigned &sequenceID, bool isLocal,
106 float *aovs) const
107{
108 scene_rdl2::math::Color radiance = scene_rdl2::math::sBlack;
109 // Estimate emissive volume region energy contribution
110 radiance += computeRadianceEmissiveRegionsSSS(pbrTls, sp, cameraId, pv, parentRay,
111 pathThroughput, transmissionFresnel, bsdf, lobe, slice, P, N,
112 subsurfaceSplitFactor, subsurfaceIndex,
113 rayEpsilon, sssSampleID, isLocal, aovs);
114 MNRY_ASSERT(pbrTls->isIntegratorAccumulatorRunning());
115
116 Statistics &stats = pbrTls->mStatistics;
117 scene_rdl2::math::Color pt = pathThroughput / float(computeRadianceSplitFactor);
118
119 // Sample from the lights
120 stats.addToCounter(STATS_LIGHT_SAMPLES,
121 lightSet.getLightCount() * computeRadianceSplitFactor);
122
123 IntegratorSample3D lightSamples;
124 IntegratorSample2D lightFilterSamples;
125 IntegratorSample3D lightFilterSamples3D;
126
127 const Scene *scene = MNRY_VERIFY(pbrTls->mFs->mScene);
128 bool lightFilterNeedsSamples = scene->lightFilterNeedsSamples();
129
130 for (int lightIndex = 0; lightIndex < lightSet.getLightCount(); lightIndex++) {
131 const Light* const light = lightSet.getLight(lightIndex);
132
133 // Sampling WIP
134 // We want one shared sequence for depth 0 across subpixels and local
135 // or global sub-surface samples
136 const SequenceIDIntegrator lightSid(
137 pv.nonMirrorDepth,
138 sp.mPixel,
139 light->getHash(),
140 (isLocal) ? SequenceType::BssrdfLocalLight : SequenceType::BssrdfGlobalLight,
141 (pv.nonMirrorDepth == 0) ? 0 : sp.mSubpixelIndex * subsurfaceSplitFactor + subsurfaceIndex,
142 sssSampleID);
143
144 // Using SSS / SP / L
145 // Note: using SP / SSS / L caused more correlation
146 int samplesSoFar = (pv.nonMirrorDepth == 0) ?
147 subsurfaceIndex * sp.mPixelSamples * computeRadianceSplitFactor +
148 sp.mSubpixelIndex * computeRadianceSplitFactor : 0;
149 lightSamples.resume(lightSid, samplesSoFar);
150
151 if (lightFilterNeedsSamples) {
152 const SequenceIDIntegrator lightFilterSid(
153 pv.nonMirrorDepth,
154 sp.mPixel,
155 light->getHash(),
156 SequenceType::LightFilter,
157 (pv.nonMirrorDepth == 0) ? 0 : sp.mSubpixelIndex * subsurfaceSplitFactor + subsurfaceIndex,
158 sssSampleID);
159 const SequenceIDIntegrator lightFilter3DSid(
160 pv.nonMirrorDepth,
161 sp.mPixel,
162 light->getHash(),
163 SequenceType::LightFilter3D,
164 (pv.nonMirrorDepth == 0) ? 0 : sp.mSubpixelIndex * subsurfaceSplitFactor + subsurfaceIndex,
165 sssSampleID);
166
167 lightFilterSamples.resume(lightFilterSid, samplesSoFar);
168 lightFilterSamples3D.resume(lightFilter3DSid, samplesSoFar);
169 }
170
171 // if light's visibility mask does not match, skip this light.
172 if (!(scene_rdl2::rdl2::DIFFUSE_REFLECTION & light->getVisibilityMask())) {
173 continue;
174 }
175
176 const LightFilterList *lightFilterList = lightSet.getLightFilterList(lightIndex);
177
178 for (int sampleIndex = 0; sampleIndex < computeRadianceSplitFactor;
179 sampleIndex++) {
180 // Draw a sample for this light
181 scene_rdl2::math::Vec3f lightSample;
182 lightSamples.getSample(&lightSample[0], pv.nonMirrorDepth);
183 LightFilterRandomValues lightFilterSample;
184 if (lightFilterNeedsSamples) {
185 lightFilterSamples.getSample(&lightFilterSample.r2[0], pv.nonMirrorDepth);
186 lightFilterSamples3D.getSample(&lightFilterSample.r3[0], pv.nonMirrorDepth);
187 }
188
189 // Warp the sample from this light
190 scene_rdl2::math::Vec3f wi;
191 LightIntersection lightIsect;
192 if (!light->sample(P, &N, parentRay.getTime(), lightSample,
193 wi, lightIsect, parentRay.getDirFootprint())) {
194 continue;
195 }
196
197 // Evaluate the light sample and pdf
198 float lightPdf;
199 scene_rdl2::math::Color Li = light->eval(pbrTls->mTopLevelTls, wi, P, lightFilterSample,
200 parentRay.getTime(), lightIsect, false, lightFilterList, parentRay.getDirFootprint(), &lightPdf);
201
202 if (isSampleInvalid(Li, lightPdf)) {
203 continue;
204 }
205 // Evaluate the lobe pdf. Note we only use the lobe for sampling
206 // to apply MIS, not for its contribution.
207 // Note: We don't want a geometric surface check like we do when
208 // evaluating bsdf because the subsurface transport makes wo irrelevant
209 float bsdfPdf;
210 scene_rdl2::math::Color f = lobe.eval(slice, wi, &bsdfPdf);
211 if (isSampleInvalid(f, bsdfPdf)) {
212 continue;
213 }
214 // Check orientation with surface normal. This is a bit unneccessary
215 // due to the call to isSampleInvalid() above.
216 float cosWi = dot(N, wi);
217 if (cosWi < scene_rdl2::math::sEpsilon) {
218 continue;
219 }
220 // Compute contribution applying incoming fresnel transmission
221 scene_rdl2::math::Color fresnel = evalFresnel(transmissionFresnel, N, wi);
222 // Apply MIS power heuristic
223 scene_rdl2::math::Color contribution = pt * Li * fresnel * f / lightPdf *
224 powerHeuristic(lightPdf, bsdfPdf);
225 // TODO: Apply russian roulette
226 // Check shadowing
227 float tfar = lightIsect.distance * sHitEpsilonEnd;
228 float time = parentRay.getTime();
229 int rayDepth = parentRay.getDepth() + 1;
230 Ray shadowRay(P, wi, rayEpsilon, tfar, time, rayDepth);
231 float presence = 0.0f;
232 if (isRayOccluded(pbrTls, light, shadowRay, rayEpsilon, shadowRayEpsilon, presence)) {
233 // LPE
234 if (aovs) {
235 EXCL_ACCUMULATOR_PROFILE(pbrTls, EXCL_ACCUM_AOVS);
236 const FrameState &fs = *pbrTls->mFs;
237 const LightAovs &lightAovs = *fs.mLightAovs;
238 // transition
239 int lpeStateId = pv.lpeStateId;
240 lpeStateId = lightAovs.subsurfaceEventTransition(pbrTls,
241 lpeStateId, bsdf);
242 // intentionally ignore the the internal lambert
243 // lobe event (see MOONRAY-1957).
244 lpeStateId = lightAovs.lightEventTransition(pbrTls,
245 lpeStateId, light);
246 // accumulate matching aovs
247 aovAccumVisibilityAovs(pbrTls, *fs.mAovSchema, cameraId, lightAovs,
248 scene_rdl2::math::Vec2f(0.0f, 1.0f), lpeStateId, aovs);
249 }
250 continue;
251 }
252 // Raydb debug
253 if (lightIsect.distance < sDistantLightDistance) {
254 Ray debugRay(P, wi, 0.0f);
255 RAYDB_EXTEND_RAY_NO_HIT(pbrTls, debugRay, lightIsect.distance);
256 RAYDB_SET_CONTRIBUTION(pbrTls, Li);
257 RAYDB_ADD_TAGS(pbrTls, TAG_AREALIGHT);
258 } else {
259 Ray debugRay(P, wi, 0.0f);
260 RAYDB_EXTEND_RAY_NO_HIT(pbrTls, debugRay, 40.0f);
261 RAYDB_SET_CONTRIBUTION(pbrTls, Li);
262 RAYDB_ADD_TAGS(pbrTls, TAG_ENVLIGHT);
263 }
264 // shadowRay can be modified in occlusion query
265 Ray trRay(P, wi, scene_rdl2::math::max(rayEpsilon, shadowRayEpsilon), tfar, time, rayDepth);
266 // volume transmittance from P to light intersection
267 scene_rdl2::math::Color volumeTransmittance = transmittance(pbrTls, trRay, sp.mPixel, sp.mSubpixelIndex, sequenceID, light);
268 contribution *= (1.0f - presence) * volumeTransmittance;
269 // Sum things up
270 radiance += contribution;
271 // LPE
272 if (aovs) {
273 EXCL_ACCUMULATOR_PROFILE(pbrTls, EXCL_ACCUM_AOVS);
274 const FrameState &fs = *pbrTls->mFs;
275 const LightAovs &lightAovs = *fs.mLightAovs;
276 const AovSchema &aovSchema = *fs.mAovSchema;
277 // transition
278 int lpeStateId = pv.lpeStateId;
279 lpeStateId = lightAovs.subsurfaceEventTransition(pbrTls, lpeStateId, bsdf);
280 // intentionally ignore the the internal lambert
281 // lobe event (see MOONRAY-1957).
282 lpeStateId = lightAovs.lightEventTransition(pbrTls, lpeStateId, light);
283 // accumulate matching aovs
284 aovAccumLightAovs(pbrTls, aovSchema, cameraId, lightAovs, contribution, nullptr,
285 AovSchema::sLpePrefixNone, lpeStateId, aovs);
286 aovAccumVisibilityAovs(pbrTls, aovSchema, cameraId, lightAovs,
287 scene_rdl2::math::Vec2f((1.0f - presence) * reduceTransparency(volumeTransmittance), 1.0f),
288 lpeStateId, aovs);
289 }
290 }
291 }

293 // Sample from the lobe, sharing the sample for direct and indirect estimator
294 IntegratorSample2D bsdfSamples;
295 IntegratorSample2D bsdfLightFilterSamples;
296 IntegratorSample3D bsdfLightFilterSamples3D;
297 // Sampling WIP
298 // We want one shared sequence for depth 0 across subpixels and
299 // local sub-surface samples
300 // We want one shared sequence for depth 0 across subpixels and
301 // global sub-surface samples
302 // TODO: We still get weird visible bias over progressive passes
303 const SequenceIDIntegrator bsdfSid(
304 pv.nonMirrorDepth,
305 sp.mPixel,
306 0,
307 (isLocal) ? SequenceType::BssrdfLocalBsdf : SequenceType::BssrdfGlobalBsdf,
308 (pv.nonMirrorDepth == 0) ? 0 : sp.mSubpixelIndex * subsurfaceSplitFactor + subsurfaceIndex,
309 sssSampleID);
310
311 // Using SSS / SP / Bsdf
312 // Note: using SP / SSS / Bsdf caused more correlation
313 const int samplesSoFar = (pv.nonMirrorDepth == 0) ?
314 (subsurfaceIndex * sp.mPixelSamples +
315 sp.mSubpixelIndex) * computeRadianceSplitFactor : 0;
316
317 bsdfSamples.resume(bsdfSid, samplesSoFar);
318
319 if (lightFilterNeedsSamples) {
320 const SequenceIDIntegrator bsdfLightFilterSid(
321 pv.nonMirrorDepth,
322 sp.mPixel,
323 0,
324 SequenceType::LightFilter,
325 (pv.nonMirrorDepth == 0) ? 0 : sp.mSubpixelIndex * subsurfaceSplitFactor + subsurfaceIndex,
326 sssSampleID);
327 const SequenceIDIntegrator bsdfLightFilter3DSid(
328 pv.nonMirrorDepth,
329 sp.mPixel,
330 0,
331 SequenceType::LightFilter3D,
332 (pv.nonMirrorDepth == 0) ? 0 : sp.mSubpixelIndex * subsurfaceSplitFactor + subsurfaceIndex,
333 sssSampleID);
334
335 bsdfLightFilterSamples.resume(bsdfLightFilterSid, samplesSoFar);
336 bsdfLightFilterSamples3D.resume(bsdfLightFilter3DSid, samplesSoFar);
337 }
338
339 // Set up sample sequence for stochastic light choice (overlapping lights)
340 SequenceIDIntegrator lightChoiceSid(pv.nonMirrorDepth, sp.mPixel,
341 SequenceType::IndexSelection,
342 sp.mSubpixelIndex * subsurfaceSplitFactor + subsurfaceIndex,
343 sssSampleID);
344 IntegratorSample1D lightChoiceSamples(lightChoiceSid);
345
346 // Prepare our next potential path vertex
347 PathVertex nextPv;
348 nextPv.pathPixelWeight = pv.pathPixelWeight;
349 nextPv.aovPathPixelWeight = pv.pathPixelWeight;
350 nextPv.pathDistance = pv.pathDistance + parentRay.getEnd();
351 nextPv.minRoughness = computeMinRoughness(lobe, mRoughnessClampingFactor,
352 pv.minRoughness);
353 nextPv.diffuseDepth = pv.diffuseDepth + 1;
354 nextPv.subsurfaceDepth = pv.subsurfaceDepth;
355 nextPv.glossyDepth = pv.glossyDepth;
356 nextPv.mirrorDepth = pv.mirrorDepth;
357 nextPv.nonMirrorDepth = pv.nonMirrorDepth + 1;
358 nextPv.presenceDepth = pv.presenceDepth;
359 nextPv.totalPresence = pv.totalPresence;
360 nextPv.volumeDepth = pv.volumeDepth;
361 nextPv.hairDepth = pv.hairDepth;
362 nextPv.lobeType = lobe.getType();
363
364 // LPE
365 if (aovs) {
366 EXCL_ACCUMULATOR_PROFILE(pbrTls, EXCL_ACCUM_AOVS);
367 const FrameState &fs = *pbrTls->mFs;
368 const LightAovs &lightAovs = *fs.mLightAovs;
369 // transition
370 int lpeStateId = pv.lpeStateId;
371 lpeStateId = lightAovs.subsurfaceEventTransition(pbrTls, lpeStateId, bsdf);
372 // intentionally ignore the the internal lambert
373 // lobe event (see MOONRAY-1957).
374 nextPv.lpeStateId = lpeStateId;
375 nextPv.lpeStateIdLight = -1;
376 }
377
378 stats.addToCounter(STATS_BSDF_SAMPLES, computeRadianceSplitFactor);
379 for (int sampleIndex = 0; sampleIndex < computeRadianceSplitFactor; sampleIndex++) {
380 // Draw a sample
381 float bsdfSample[2];
382 bsdfSamples.getSample(bsdfSample, pv.nonMirrorDepth);
383 LightFilterRandomValues bsdfLightFilterSample;
384 if (lightFilterNeedsSamples) {
385 bsdfLightFilterSamples.getSample(&bsdfLightFilterSample.r2[0], pv.nonMirrorDepth);
386 bsdfLightFilterSamples3D.getSample(&bsdfLightFilterSample.r3[0], pv.nonMirrorDepth);
387 }
388
389 // Warp the sample and evaluate lobe pdf. Note we only use the lobe for
390 // sampling to apply BIS/MIS, not for its contribution.
391 // Note: We don't want a geometric surface check like we do when
392 // evaluating bsdf because the subsurface transport makes wo irrelevant
393 scene_rdl2::math::Vec3f wi;
394 float bsdfPdf;
395 scene_rdl2::math::Color f = lobe.sample(slice, bsdfSample[0], bsdfSample[1], wi, bsdfPdf);
396 if (isSampleInvalid(f, bsdfPdf)) {
397 continue;
398 }
399 // Check orientation with surface normal. This is a bit unneccessary
400 // due to the call to isSampleInvalid() above.
401 float cosWi = dot(N, wi);
402 if (cosWi < scene_rdl2::math::sEpsilon) {
403 continue;
404 }
405 int numLightsHit = 0;
406 LightIntersection lightIsect;
407 int hitIdx = lightSet.intersect(P, &N, wi, parentRay.getTime(),
408 scene_rdl2::math::sMaxValue, false, lightChoiceSamples, pv.nonMirrorDepth,
409 scene_rdl2::rdl2::DIFFUSE_REFLECTION, lightIsect, numLightsHit);
410 const Light* hitLight = hitIdx >= 0 ? lightSet.getLight(hitIdx) : nullptr;
411 const LightFilterList* hitLightFilterList = hitIdx >= 0 ? lightSet.getLightFilterList(hitIdx) : nullptr;
412 bool doDirect = (numLightsHit > 0);
413 // Evaluate the light contribution and pdf
414 float lightPdf = 0.0f;
415 scene_rdl2::math::Color Li = scene_rdl2::math::sBlack;
416 if (doDirect) {
417 // We multiply by numLightsHit because we're stochastically sampling
418 // just one, thus computing an average
419 Li = hitLight->eval(pbrTls->mTopLevelTls, wi, P, bsdfLightFilterSample,
420 parentRay.getTime(), lightIsect, false, hitLightFilterList, parentRay.getDirFootprint(), &lightPdf) *
421 (float)numLightsHit;
422 if (isSampleInvalid(Li, lightPdf)) {
423 doDirect = false;
424 }
425 }
426 // Compute incoming fresnel transmission
427 scene_rdl2::math::Color fresnel = evalFresnel(transmissionFresnel, N, wi);
428 // TODO: Apply russian roulette
429 // the volume attenuation along this ray to the first hit (or infinity)
430
431 VolumeTransmittance vt;
432 vt.reset();
433
434 float presence = 0.0f;
435 // Compute indirect illumination if needed, trace a shadow ray if not
436 if (doIndirect) {
437 // Apply BIS
438 nextPv.pathThroughput = pt * fresnel * f / bsdfPdf;
439 // Build our next ray differential to see if there is an indirect
440 // lighting contribution closer than the intersected light, if any.
441 // TODO: We had some self-intersections when rays leave at grazing
442 // angle, so we adjust the rayEpsilon accordingly
443 const float denom = scene_rdl2::math::abs(dot(slice.getNg(), wi));
444 // slice.getNg() itself or the dot product above can be zero.
445 const float start = scene_rdl2::math::isZero(denom) ? rayEpsilon : rayEpsilon / denom;
446 RayDifferential ray(
447 parentRay, start,
448 lightIsect.distance * sHitEpsilonEnd);
449 scatterAndScale(dNdx, dNdy,
450 lobe,
451 -parentRay.getDirection(),
452 wi,
453 1.0f,
454 bsdfSample[0],
455 bsdfSample[1],
456 ray);
457
458 // Recurse
459 scene_rdl2::math::Color contribution;
460 float transparency;
461 ++sequenceID;
462 bool hitVolume;
463
464 IndirectRadianceType indirectRadianceType = computeRadianceRecurse(
465 pbrTls, ray, sp, cameraId, nextPv, &lobe,
466 contribution, transparency, vt,
467 sequenceID, aovs, nullptr, nullptr, nullptr, false, hitVolume);
468 if (indirectRadianceType != NONE) {
469 // Accumulate radiance, but only accumulate indirect or direct
470 // contribution
471 radiance += contribution;
472 }
473 }
474 //------------------------------
475 // Compute direct lighting contribution with MIS if needed.
476 if (doDirect) {
477 float tfar = lightIsect.distance * sHitEpsilonEnd;
478 float time = parentRay.getTime();
479 int rayDepth = parentRay.getDepth() + 1;
480 const FrameState &fs = *pbrTls->mFs;
481 Ray shadowRay(P, wi, rayEpsilon, tfar, time, rayDepth);
482 const bool hasUnoccludedFlag = fs.mAovSchema->hasLpePrefixFlags(AovSchema::sLpePrefixUnoccluded);
483 if (isRayOccluded(pbrTls, hitLight, shadowRay, rayEpsilon, shadowRayEpsilon, presence)) {
484 // LPE
485 if (aovs) {
486 EXCL_ACCUMULATOR_PROFILE(pbrTls, EXCL_ACCUM_AOVS);
487 const AovSchema &aovSchema = *fs.mAovSchema;
488 const LightAovs &lightAovs = *fs.mLightAovs;
489 // transition
490 int lpeStateId = pv.lpeStateId;
491 lpeStateId = lightAovs.subsurfaceEventTransition(pbrTls,
492 lpeStateId, bsdf);
493 // intentionally ignore the the internal lambert
494 // lobe event (see MOONRAY-1957).
495 lpeStateId = lightAovs.lightEventTransition(pbrTls,
496 lpeStateId, hitLight);
497 // accumulate matching aovs
498 aovAccumVisibilityAovs(pbrTls, aovSchema, cameraId, lightAovs,
499 scene_rdl2::math::Vec2f(0.0f, 1.0f), lpeStateId, aovs);
500
501 // unoccluded prefix LPEs
502 if (hasUnoccludedFlag) {
503 // If it's occluded but we have the unoccluded flag set, only contribute this to any
504 // pre-occlusion aovs.
505 scene_rdl2::math::Color contribution = pt * Li * fresnel * f / bsdfPdf * powerHeuristic(bsdfPdf, lightPdf);
506 aovAccumLightAovs(pbrTls, aovSchema, cameraId, lightAovs, contribution, nullptr,
507 AovSchema::sLpePrefixUnoccluded, lpeStateId, aovs);
508 }
509 }
510 continue;
511 }
512 // shadowRay can be modified in occlusion query
513 Ray trRay(P, wi, scene_rdl2::math::max(rayEpsilon, shadowRayEpsilon), tfar, time, rayDepth);
514 vt.mTransmittanceE = transmittance(pbrTls, trRay, sp.mPixel, sp.mSubpixelIndex, sequenceID, hitLight);
515 // Apply MIS power heuristic
516 scene_rdl2::math::Color contribution = pt * Li * fresnel * f / bsdfPdf *
517 powerHeuristic(bsdfPdf, lightPdf);
518 const scene_rdl2::math::Color unoccludedContribution = contribution;
519 contribution *= (1.0f - presence) * vt.mTransmittanceE;
520 // Sum things up
521 radiance += contribution;
522 // LPE
523 if (aovs) {
524 EXCL_ACCUMULATOR_PROFILE(pbrTls, EXCL_ACCUM_AOVS);
525 const FrameState &fs = *pbrTls->mFs;
526 const AovSchema &aovSchema = *fs.mAovSchema;
527 const LightAovs &lightAovs = *fs.mLightAovs;
528 // transition
529 int lpeStateId = nextPv.lpeStateId;
530 lpeStateId = lightAovs.lightEventTransition(pbrTls, lpeStateId, hitLight);
531 // accumulate matching aovs
532 aovAccumVisibilityAovs(pbrTls, aovSchema, cameraId, lightAovs,
533 scene_rdl2::math::Vec2f((1-presence) * reduceTransparency(vt.mTransmittanceE), 1.0f),
534 lpeStateId, aovs);
535
536 // Accumulate aovs depending on whether or not the unoccluded flag is set.
537 if (hasUnoccludedFlag) {
538 // If the unoccluded flag is set we have to add occluded and unoccluded
539 // (without presence and volume transmittance) separately.
540 aovAccumLightAovs(pbrTls, aovSchema, cameraId, lightAovs, unoccludedContribution, &contribution,
541 AovSchema::sLpePrefixUnoccluded, lpeStateId, aovs);
542 } else {
543 // Otherwise, just add the contribution to all non-pre-occlusion aovs.
544 aovAccumLightAovs(pbrTls, aovSchema, cameraId, lightAovs, contribution, nullptr,
545 AovSchema::sLpePrefixNone, lpeStateId, aovs);
546 }
547 }
548 // Raydb debug
549 if (lightIsect.distance < sDistantLightDistance) {
550 Ray debugRay(P, wi, 0.0f);
551 RAYDB_EXTEND_RAY_NO_HIT(pbrTls, debugRay, lightIsect.distance);
552 RAYDB_SET_CONTRIBUTION(pbrTls, Li);
553 RAYDB_ADD_TAGS(pbrTls, TAG_AREALIGHT);
554 } else {
555 Ray debugRay(P, wi, 0.0f);
556 RAYDB_EXTEND_RAY_NO_HIT(pbrTls, debugRay, 40.0f);
557 RAYDB_SET_CONTRIBUTION(pbrTls, Li);
558 RAYDB_ADD_TAGS(pbrTls, TAG_ENVLIGHT);
559 }
560 }
561 }
562 return radiance;
563}

565scene_rdl2::math::Color
566PathIntegrator::computeRadianceDiffusionSubsurface(pbr::TLState *pbrTls,
567 const Bsdf &bsdf, const Subpixel &sp, int cameraId, const PathVertex &pv,
568 const RayDifferential &ray, const Intersection &isect,
569 const BsdfSlice &slice, const Bssrdf &bssrdf,
570 const LightSet &lightSet, bool doIndirect,
571 float rayEpsilon, float shadowRayEpsilon,
572 unsigned &sequenceID, scene_rdl2::math::Color &ssAov, float *aovs) const
573{
574 scene_rdl2::math::Color radiance = scene_rdl2::math::sBlack;
575 if (mBssrdfSamples < 1 || !mEnableSSS) {
576 return radiance;
577 }
578
579 EXCL_ACCUMULATOR_PROFILE(pbrTls, EXCL_ACCUM_SSS_INTEGRATION);
580
581 // TODO: We need to use a "lobe type" that tells computeRadianceRecurse() and
582 // the rest of the system that this is a bssrdf "lobe type" (used for ray
583 // masking and ray debugging)
584 const shading::BsdfLobe::Type lobeType =
585 shading::BsdfLobe::Type(shading::BsdfLobe::REFLECTION | shading::BsdfLobe::DIFFUSE);
586 Statistics &stats = pbrTls->mStatistics;
587
588 const Scene *scene = MNRY_VERIFY(pbrTls->mFs->mScene);
589
590 // Search limits
591 float maxRadius = bssrdf.getMaxRadius();
592 float minRadius = maxRadius * 0.1f;
593
594 // Where are we ?
595 const scene_rdl2::math::Vec3f P = isect.getP();
596 const scene_rdl2::math::Vec3f N = isect.getN();
597 const scene_rdl2::math::Vec3f wo = -(ray.getDirection());
598 // Local Reference Frame
599 scene_rdl2::math::ReferenceFrame localF(N);
600
601 // The scale * outgoing fresnel
602 const Fresnel *transmissionFresnel = bssrdf.getTransmissionFresnel();
603 const scene_rdl2::math::Color scaleFresnelWo = evalFresnel(transmissionFresnel, N, wo) *
604 bssrdf.getScale();
605
606 // We disable the transmission fresnel on the wi side.
607 // If the material is using a OneMinusRoughSchlickFresnel, this matches the
608 // energy conservation we are doing with other types of "under" lobes.
609 // Otherise it darkens the SSS somewhat a bit when the specular factor of
610 // the "top" lobe is in the 0.2 .. 0.7 range.
611 transmissionFresnel = nullptr;
612
613 //---------------------------------------------------------------------
614 // TODO: We ignore single scattering for now since its relative contribution
615 // is small. We'll implement this in the future, its relatively very easy
616 // compared to multiple scattering below.
617
618 // Check if we can substitute subsurface with a cheaper approximation
619 if (canSubstituteSubsurface(maxRadius, isect)) {
620
621 shading::BsdfSlice sliceLocal(isect.getNg(), slice.getWo(), true, true, slice.getShadowTerminatorFix());
622 LambertBsdfLobe lobeLocal(N, scene_rdl2::math::sWhite, true);
623
624 // Not doing any subsurface computations
625 int subsurfaceSplitFactor = 1;
626 // But split to compute Radiance
627 int computeRadianceSplitFactor = (pv.nonMirrorDepth > 0) ?
628 1 : mBsdfSamples;
629
630 /// SSS throughput
631 scene_rdl2::math::Color throughput = pv.pathThroughput * bssrdf.diffuseReflectance() *
632 scaleFresnelWo;
633
634 // Lambertian Reflection instead of SSS
635 radiance += computeRadianceSubsurfaceSample(
636 pbrTls, bsdf, sp, cameraId, pv,
637 ray, isect.getdNdx(), isect.getdNdy(), throughput,
638 transmissionFresnel, lightSet, lobeLocal, sliceLocal, P, N,
639 subsurfaceSplitFactor, //subsurfaceSplitFactor
640 computeRadianceSplitFactor, //regular split factor
641 0, //split index
642 doIndirect, rayEpsilon, shadowRayEpsilon, sequenceID, sequenceID, true, aovs);
643
644
645 // We cannot approximate the forward scattering with a
646 // Lambertian lobe, so evaluate this explicitly.
647 radiance += computeDiffusionForwardScattering(pbrTls, bsdf, sp, cameraId, pv,
648 ray, isect, slice, transmissionFresnel, scaleFresnelWo, lightSet,
649 bssrdf, P, N, localF, subsurfaceSplitFactor, doIndirect,
650 rayEpsilon, shadowRayEpsilon, sequenceID, sequenceID, ssAov, aovs);
651
652 return radiance;
653 }
654
655
656 // Compute Full Subsurface
657
658 //---------------------------------------------------------------------
659 // Compute local multiple scattering, as in paper [2]
660 bool dontSplit = (pv.nonMirrorDepth > 0);
661
662 // Make sure we do all local sss sampling with input sequenceID, which will
663 // get incremented in the process.
664 unsigned sssSampleID = sequenceID;
665
666 // Triplanar projection sampling needs at least 2 samples.
667 int subsurfaceSplitFactor = (dontSplit ? 2 : mBssrdfSamples);
668
669 scene_rdl2::math::Color ptSubsurface = pv.pathThroughput / subsurfaceSplitFactor;
670
671 // Setup Bssrdf local samples.
672 // We want one shared sequence for non-mirror-depth 0 across sub-pixels
673 IntegratorSample2D bssrdfLocalSamples;
674 const SequenceIDBssrdf bssrdfLocalSid(
675 pv.nonMirrorDepth, sp.mPixel, SequenceType::BssrdfLocal,
676 pv.nonMirrorDepth == 0 ? 0 : sp.mSubpixelIndex, sssSampleID);
677 bssrdfLocalSamples.resume(bssrdfLocalSid,
678 pv.nonMirrorDepth == 0 ? sp.mSubpixelIndex * subsurfaceSplitFactor : 0);
679
680 stats.addToCounter(STATS_SSS_SAMPLES, subsurfaceSplitFactor);
681
682 // We measure the diffuse reflectance through sampling to
683 // calculate the area compensation factor based on how much we
684 // deviate from the semi-infinite plane assumption
685 scene_rdl2::math::Color measuredDiffuseReflectance = scene_rdl2::math::sBlack;
686
687 // Collect a list of sample points to evaluate the subsurface integral
688 struct SubsurfaceSample {
689 scene_rdl2::math::Vec3f mP, mN, mNg;
690 scene_rdl2::math::Vec3f mdNdx, mdNdy;
691 scene_rdl2::math::Color mBssrdfEval;
692 };
693 std::vector<SubsurfaceSample> subsurfaceSamples;
694
695 // Bssrdf input normal
696 const scene_rdl2::rdl2::Material* sssMaterial = bssrdf.getMaterial();
697 const scene_rdl2::rdl2::EvalNormalFunc evalSubsurfaceNormal = bssrdf.getEvalNormalFunc();
698 bool validInputNormal = sssMaterial && evalSubsurfaceNormal;
699
700 for (int sampleIndex = 0; sampleIndex < subsurfaceSplitFactor; sampleIndex++) {
701
702 // Draw a low discrepancy sample
703 float sample[2];
704 bssrdfLocalSamples.getSample(sample, pv.nonMirrorDepth);
705
706 // Step 1 - Select a projection axis & remap the random number to be
707 // in [0,1) Axis of Projection
708 scene_rdl2::math::Vec3f directionProj;
709 int axisIndex = bssrdfSelectAxisAndRemapSample(
710 localF, sample[0], directionProj);
711
712 // Sample position according to Bssrdf
713 scene_rdl2::math::Vec3f PiTangent;
714 float r;
715 float pdf = bssrdf.sampleLocal(sample[0], sample[1], PiTangent, r);
716 if (scene_rdl2::math::isZero(pdf, 0.0f)) { // exactly zero
717 continue;
718 }
719
720 PiTangent = P + bssrdfOffsetLocalToGlobal(localF, axisIndex, PiTangent);
721
722 // We need to limit the distance to search for nearby surfaces to
723 // project onto. Otherwise we might go through and get blocked by
724 // other nearby objects.
725 float search = scene_rdl2::math::min(r, maxRadius);
726 search = scene_rdl2::math::max(search, minRadius);
727 scene_rdl2::math::Vec3f originProj = PiTangent - directionProj * search;
728
729 shading::Intersection isectProj;
730 Ray rayProj(originProj,
731 directionProj,
732 0.0f,
733 2.0f * search,
734 ray.getTime(),
735 ray.getDepth() + 1);
736
737 // get trace set for sss
738 auto geomTls = pbrTls->mTopLevelTls->mGeomTls.get();
739 geomTls->mSubsurfaceTraceSet = bssrdf.getTraceSet();
740 rayProj.ext.geomTls = (void*)geomTls;
741
742 // set default material for trace set;
743 const scene_rdl2::rdl2::Material* isectMaterial = isect.getMaterial();
744 if (isectMaterial) {
745 const shading::Material *materialExt =
746 &isectMaterial->get<const shading::Material>();
747 rayProj.ext.materialID = materialExt->getMaterialId();
748 }
749
750 // TODO: We need position and normal only in the intersection
751 bool intersected = scene->intersectRay(pbrTls->mTopLevelTls, rayProj,
752 isectProj, lobeType);
753
754 const scene_rdl2::rdl2::Material* isectProjMaterial = isectProj.getMaterial();
755
756 if (!intersected || isectProj.getMaterial() == nullptr) {
757 continue;
758 }
759 geom::initIntersectionPhase2(isectProj,
760 pbrTls->mTopLevelTls,
761 pv.mirrorDepth,
762 pv.glossyDepth,
763 pv.diffuseDepth,
764 isSubsurfaceAllowed(pv.subsurfaceDepth),
765 pv.minRoughness,
766 -rayProj.getDirection());
767 isectProj.setCameraId(cameraId);
768 RayDifferential rayDiff(ray, isectProj.getEpsilonHint(), rayProj.tfar);
769 isectProj.transferAndComputeDerivatives(pbrTls->mTopLevelTls, &rayDiff,
770 sp.mTextureDiffScale);
771
772 // Keep track of projected position and normal
773 scene_rdl2::math::Vec3f PiProj = isectProj.getP();
774 scene_rdl2::math::Vec3f NiProj = isectProj.getN();
775
776 scene_rdl2::math::Vec3f NiProjMap = NiProj;
777 // The material of the intersected point must match the
778 // isectMaterial to evaluate the normal map. This ensures that
779 // the intersection state has all the attributes requested by the
780 // normal map. There may not be a match when using trace sets.
781 // In case of a mismatch, we do not attempt to evaluate the subsurface normal map.
782 if (validInputNormal && isectProjMaterial == isectMaterial) {
783 // We evaluate using the sssMaterial, which does not always match the isectMaterial.
784 // Example: sssMaterial is undermaterial of isectMaterial, which is a GlitterFlakeMaterial.
785 NiProjMap = evalSubsurfaceNormal(sssMaterial,
786 pbrTls->mTopLevelTls->mShadingTls.get(),
787 shading::State(&isectProj));
788 }
789
790 // Only count surface hits facing the direction of projection.
791 // This avoids double contribution across local / global scattering
792 const float cosTheta = -dot(NiProj, directionProj);
793 if (cosTheta < scene_rdl2::math::sEpsilon) {
794 continue;
795 }
796
797 RAYDB_EXTEND_RAY(pbrTls, rayProj, isectProj);
798 RAYDB_SET_TAGS(pbrTls, lobeType);
799
800 // Update r from projected position and make sure the projected
801 // position is still within maxRadius, otherwise this may clash
802 // with light culling (see where the lightSet is computed).
803 scene_rdl2::math::Vec3f dp = P - PiProj;
804 r = dp.length();
805 if (r > maxRadius) {
806 continue;
807 }
808 // MIS One-Sample Projection Weight
809 float misWeight = bssrdfGetMISAxisWeight(localF, NiProj, dp, bssrdf);
810 scene_rdl2::math::Color bssrdfEval = bssrdf.eval(r) * misWeight;
811
812 // Calculate the BSSRDF DiffuseReflectance Integral via Sampling
813 measuredDiffuseReflectance += bssrdfEval;
814
815 // Subsurface Integral
816 scene_rdl2::math::Color pt = ptSubsurface * bssrdfEval;
817
818 // Apply the out-point transmission fresnel
819 pt *= scaleFresnelWo;
820
821 // used for subsurface material aovs
822 ssAov += pt;
823
824 // Collect this sample
825 SubsurfaceSample s;
826 s.mP = PiProj;
827 s.mN = NiProjMap;
828 s.mNg = isectProj.getNg();
829 s.mdNdx = isectProj.getdNdx();
830 s.mdNdy = isectProj.getdNdy();
831 s.mBssrdfEval = pt;
832 subsurfaceSamples.push_back(std::move(s));
833 }
834
835 // DiffuseReflectance Integral
836 measuredDiffuseReflectance /= subsurfaceSplitFactor;
837 // Area Compensation Term
838 const scene_rdl2::math::Color areaCompensationFactor = bssrdfAreaCompensation(
839 measuredDiffuseReflectance, bssrdf.diffuseReflectance());
840 ssAov *= areaCompensationFactor;
841
842 //For each sample, evaluate irradiance and compute the subsurface integral
843 // TODO: For now, we don't want to split here at all
844 #if 0
845 int computeRadianceSplitFactor = pv.nonMirrorDepth == 0 ? mDirectSamples:1;
846 #else
847 int computeRadianceSplitFactor = 1;
848 #endif
849
850 for (size_t i = 0; i < subsurfaceSamples.size(); ++i) {
851 // Compute the direct irradiance for this sample
852 // TODO: compute woiProj
853 shading::BsdfSlice sliceLocal(subsurfaceSamples[i].mNg, slice.getWo(), true, true,
854 slice.getShadowTerminatorFix());
855 LambertBsdfLobe lobeLocal(subsurfaceSamples[i].mN, scene_rdl2::math::sWhite, true);
856
857 // TODO: Need to pass rayProj and updated pv instead of ray, pv
858 radiance += computeRadianceSubsurfaceSample(
859 pbrTls, bsdf, sp, cameraId, pv, ray,
860 subsurfaceSamples[i].mdNdx, subsurfaceSamples[i].mdNdy,
861 subsurfaceSamples[i].mBssrdfEval * areaCompensationFactor,
862 transmissionFresnel, lightSet, lobeLocal, sliceLocal,
863 subsurfaceSamples[i].mP, subsurfaceSamples[i].mN,
864 subsurfaceSplitFactor, computeRadianceSplitFactor, i,
865 doIndirect, rayEpsilon, shadowRayEpsilon, sssSampleID,
866 sequenceID, true, aovs);
867 RAYDB_SET_CONTRIBUTION(pbrTls, radiance / pv.pathThroughput);
868 }
869
870 /// Back-scattering Term
871 radiance += computeDiffusionForwardScattering(pbrTls, bsdf, sp, cameraId, pv, ray,
872 isect, slice, transmissionFresnel, scaleFresnelWo, lightSet, bssrdf,
873 P, N, localF, subsurfaceSplitFactor, doIndirect, rayEpsilon, shadowRayEpsilon,
874 sssSampleID, sequenceID, ssAov, aovs);
875
876 return radiance;
877}

879scene_rdl2::math::Color
880PathIntegrator::computeDiffusionForwardScattering(pbr::TLState *pbrTls, const Bsdf &bsdf,
881 const Subpixel &sp, int cameraId, const PathVertex &pv,
882 const RayDifferential &ray, const Intersection &isect,
883 const BsdfSlice &slice, const Fresnel *transmissionFresnel,
884 const scene_rdl2::math::Color& scaleFresnelWo, const LightSet &lightSet,
885 const Bssrdf &bssrdf, const scene_rdl2::math::Vec3f &P, const scene_rdl2::math::Vec3f &N,
886 const scene_rdl2::math::ReferenceFrame &localF, int subsurfaceSplitFactor,
887 bool doIndirect, float rayEpsilon, float shadowRayEpsilon,
888 unsigned sssSampleID, unsigned &sequenceID, scene_rdl2::math::Color &ssAov, float *aovs) const
889{
890 scene_rdl2::math::Color radiance = scene_rdl2::math::sBlack;
891 const Scene *scene = MNRY_VERIFY(pbrTls->mFs->mScene);
892 // Bssrdf input normal
893 const scene_rdl2::rdl2::Material* sssMaterial = bssrdf.getMaterial();
894 const scene_rdl2::rdl2::EvalNormalFunc evalSubsurfaceNormal = bssrdf.getEvalNormalFunc();
895 bool validInputNormal = sssMaterial && evalSubsurfaceNormal;
896
897 scene_rdl2::math::Color ptSubsurface = pv.pathThroughput / subsurfaceSplitFactor;
898
899 // Compute global multiple scattering (back of the ear, leaf, etc.)
900 /// Ref - "Efficient rendering of human skin", D'Eon et al., EGSR 2007
901 // Setup Bssrdf global samples.
902 // We want one shared sequence for non-mirror-depth 0 across sub-pixels
903 IntegratorSample2D bssrdfGlobalSamples;
904
905 Statistics &stats = pbrTls->mStatistics;
906 float maxRadius = bssrdf.getMaxRadius();
907 // TODO: We need to use a "lobe type" that tells computeRadianceRecurse() and
908 // the rest of the system that this is a bssrdf "lobe type" (used for ray
909 // masking and ray debugging)
910 const BsdfLobe::Type lobeType =
911 BsdfLobe::Type(BsdfLobe::REFLECTION | BsdfLobe::DIFFUSE);
912
913 const SequenceIDBssrdf bssrdfGlobalSid(pv.nonMirrorDepth, sp.mPixel,
914 SequenceType::BssrdfGlobal,
915 (pv.nonMirrorDepth == 0 ? 0 : sp.mSubpixelIndex), sssSampleID);
916 bssrdfGlobalSamples.resume(bssrdfGlobalSid,
917 (pv.nonMirrorDepth == 0 ? sp.mSubpixelIndex * subsurfaceSplitFactor : 0));
918
919 // Prepare a path vertex
920 PathVertex nextPv;
921 nextPv.pathThroughput = scene_rdl2::math::sWhite; // Unused
922 nextPv.pathPixelWeight = 0.0f;
923 // Use previous path pixel weight for aovPathPixelWeight as there's existing logic
924 // in vector mode that sometimes assumes that pv.pathPixelWeight = 0. Thus, we must seperately
925 // keep track of the pathPixelWeight for aovs. See comment in PathIntegratorMultiSampler.ispc::
926 // addIndirectOrDirectVisibleContributionsBundled().
927 nextPv.aovPathPixelWeight = pv.pathPixelWeight;
928 nextPv.pathDistance = pv.pathDistance + ray.getEnd();
929 nextPv.minRoughness = computeMinRoughness(
930 1.0f, mRoughnessClampingFactor, pv.minRoughness);
931 nextPv.diffuseDepth = pv.diffuseDepth + 1;
932 nextPv.subsurfaceDepth = pv.subsurfaceDepth;
933 nextPv.glossyDepth = pv.glossyDepth;
934 nextPv.mirrorDepth = pv.mirrorDepth;
935 nextPv.nonMirrorDepth = pv.nonMirrorDepth + 1;
936 nextPv.presenceDepth = pv.presenceDepth;
937 nextPv.totalPresence = pv.totalPresence;
938 nextPv.volumeDepth = pv.volumeDepth;
939 nextPv.hairDepth = pv.hairDepth;
940
941 // LPE: not sure how to characterize this event
942 if (aovs) {
943 nextPv.lpeStateId = pv.lpeStateId;
944 nextPv.lpeStateIdLight = -1;
945 }
946
947 stats.addToCounter(STATS_SSS_SAMPLES, subsurfaceSplitFactor);
948
949 // Max Spherical Cap
950 static const float cosThetaMax = scene_rdl2::math::cos(85.0f / 180.0f * scene_rdl2::math::sPi);
951
952 for (int sampleIndex = 0; sampleIndex < subsurfaceSplitFactor; sampleIndex++) {
953 // Draw a low discrepancy sample
954 float sample[2];
955 bssrdfGlobalSamples.getSample(sample, pv.nonMirrorDepth);
956 // Use the bssrdf to sample a direction into the subsurface
957 scene_rdl2::math::Vec3f direction;
958 float pdf = bssrdf.sampleGlobal(sample[0], sample[1],
959 cosThetaMax, direction);
960 direction = -localF.localToGlobal(direction);
961 // Trace to search a nearby back-facing surface
962 Intersection isectBack;
963 Ray rayBack(P, direction, rayEpsilon, maxRadius,
964 ray.getTime(), ray.getDepth() + 1);
965 // get trace set for sss
966 geom::internal::TLState* geomTls = pbrTls->mTopLevelTls->mGeomTls.get();
967 geomTls->mSubsurfaceTraceSet = bssrdf.getTraceSet();
968 rayBack.ext.geomTls = geomTls;
969 // get default material for trace set
970 const scene_rdl2::rdl2::Material* isectMaterial = isect.getMaterial();
971 if (isectMaterial) {
972 const shading::Material *materialExt =
973 &isectMaterial->get<const shading::Material>();
974 rayBack.ext.materialID = materialExt->getMaterialId();
975 }
976 // TODO: We need position and normal only in the intersection
977 if (!scene->intersectRay(pbrTls->mTopLevelTls, rayBack, isectBack, lobeType) ||
978 isectBack.getMaterial() == nullptr) {
979 continue;
980 }
981 geom::initIntersectionPhase2(isectBack,
982 pbrTls->mTopLevelTls,
983 pv.mirrorDepth,
984 pv.glossyDepth,
985 pv.diffuseDepth,
986 isSubsurfaceAllowed(pv.subsurfaceDepth),
987 pv.minRoughness,
988 -rayBack.getDirection());
989 RayDifferential rayDiff(ray, isectBack.getEpsilonHint(), rayBack.tfar);
990 isectBack.setCameraId(cameraId);
991 isectBack.transferAndComputeDerivatives(pbrTls->mTopLevelTls, &rayDiff,
992 sp.mTextureDiffScale);
993
994 // TODO: Should we use the un-flipped N ?
995 scene_rdl2::math::Vec3f Ni = -isectBack.getN();
996 scene_rdl2::math::Vec3f Pi = isectBack.getP();
997 float r = rayBack.getEnd();
998
999 scene_rdl2::math::Vec3f NiMap = Ni;
1000 // The material of the intersected point must match the
1001 // isectMaterial to evaluate the normal map. This ensures that
1002 // the intersection state has all the attributes requested by the
1003 // normal map. There may not be a match when using trace sets.
1004 // In case of a mismatch, we do not attempt to evaluate the subsurface normal map.
1005 if (validInputNormal && isectBack.getMaterial() == isectMaterial) {
1006 // We evaluate using the sssMaterial, which does not always match the isectMaterial.
1007 // Example: sssMaterial is undermaterial of isectMaterial, which is a GlitterFlakeMaterial.
1008 NiMap = -evalSubsurfaceNormal(sssMaterial,
1009 pbrTls->mTopLevelTls->mShadingTls.get(),
1010 shading::State(&isectBack));
1011 }
1012 // Only count surface hits facing away from the direction of projection
1013 // that we used for local scattering.
1014 // This avoids double contribution across local / global scattering
1015 float cosTheta = -dot(Ni, N);
1016 if (cosTheta < scene_rdl2::math::sEpsilon) {
1017 continue;
1018 }
1019
1020 RAYDB_EXTEND_RAY(pbrTls, rayBack, isectBack);
1021 RAYDB_SET_TAGS(pbrTls, lobeType);
1022
1023 // We compute the jacobian to convert between this integral over solid
1024 // angle into the integral we want over surface area
1025 cosTheta = scene_rdl2::math::abs(dot(Ni, direction));
1026 float jacobian = r * r / scene_rdl2::math::max(cosTheta, 0.05f);
1027
1028 // Account for Bssrdf
1029 scene_rdl2::math::Color pt = ptSubsurface * bssrdf.eval(r, true) * jacobian / pdf;
1030 // Apply the out-point transmission fresnel and 1/pi term
1031 pt *= scaleFresnelWo;
1032 // used for subsurface aov
1033 ssAov += pt;
1034 // TODO: For now, we don't want to split here at all
1035 const int computeRadianceSplitFactor = 1;
1036 // Compute the direct irradiance for this sample
1037 // TODO: compute woiBack
1038 shading::BsdfSlice sliceGlobal(-isectBack.getNg(), slice.getWo(), true, true,
1039 slice.getShadowTerminatorFix());
1040 LambertBsdfLobe lobeGlobal(NiMap, scene_rdl2::math::sWhite, true);
1041 // TODO: Need to pass rayBack and updated pv instead of ray, pv
1042 radiance += computeRadianceSubsurfaceSample(pbrTls, bsdf,
1043 sp, cameraId, nextPv, ray, isectBack.getdNdx(), isectBack.getdNdy(),
1044 pt, transmissionFresnel, lightSet, lobeGlobal, sliceGlobal,
1045 Pi, NiMap, subsurfaceSplitFactor, computeRadianceSplitFactor,
1046 sampleIndex, doIndirect, rayEpsilon, shadowRayEpsilon,
1047 sssSampleID, sequenceID, false, aovs);
1048
1049 RAYDB_SET_CONTRIBUTION(pbrTls, radiance / pv.pathThroughput);
1050 }
1051 return radiance;
1052}

1056// utility function to sample distance t from 0 to inf with distribution
1057// propotional to exp(-sigmaT[channel] * t)
1058// channel is selected based on discrete pdf provided through channelWeights
1059// this is similar to the bsdf one sample idea that picking out a lobe first
1060// and then eval the total lobes contribution
1061// the idea is explored by multiple publication addressing chromatic volume
1062// coefficient importance sampling, see
1063// "Practical and Controllable Subsurface Scattering for Production Path Tracing"
1064// Matt Jen-Yuan Chiang el al
1065// 3. Sampling for detail reference
1066// pbrt3 also contains an implementation that selecting channel with uniform pdf
1067// http://www.pbr-book.org/3ed-2018/Light_Transport_II_Volume_Rendering/Sampling_Volume_Scattering.html#HomogeneousMedium
1068float
1069sampleDistanceZeroScatter(const scene_rdl2::math::Color& channelWeights, const scene_rdl2::math::Color& sigmaT,
1070 float uChannel, float uDistance, scene_rdl2::math::Color& tr, float& pdfDistance)
1071{
1072 float normalization =
1073 scene_rdl2::math::rcp(channelWeights[0] + channelWeights[1] + channelWeights[2]);
1074 scene_rdl2::math::Color pdfChannel = normalization * channelWeights;
1075
1076 float distance;
1077 if (uChannel < pdfChannel[0]) {
1078 distance = -scene_rdl2::math::log(1.0f - uDistance) / sigmaT[0];
1079 } else if (uChannel < (pdfChannel[0] + pdfChannel[1])) {
1080 distance = -scene_rdl2::math::log(1.0f - uDistance) / sigmaT[1];
1081 } else {
1082 distance = -scene_rdl2::math::log(1.0f - uDistance) / sigmaT[2];
1083 }
1084 tr = scene_rdl2::math::exp(-sigmaT * distance);
1085 pdfDistance =
1086 pdfChannel[0] * sigmaT[0] * tr[0] +
1087 pdfChannel[1] * sigmaT[1] * tr[1] +
1088 pdfChannel[2] * sigmaT[2] * tr[2];
1089 return distance;
1090}

1092// Distance Sampling with either:
1093// (a) sampling transmittance or (b) Dwivedi distance sampling
1094// For Dwivedi Distance Sampling, ref "Improved Dwivedi Sampling"
1095// Return the resulting transmittance for that distance
1096// & the PDF for both methods for one-sample PDF computation
1097float
1098sampleDistanceMultiScatter(const scene_rdl2::math::Color& sigmaT, const scene_rdl2::math::Color& sigmaTPrime,
1099 const scene_rdl2::math::Color& channelSamplingWeights, int chIndex, float uDistance,
1100 float NdotDir, const scene_rdl2::math::Color& dwivediV0, bool choseDwivediSampling,
1101 scene_rdl2::math::Color& tr, scene_rdl2::math::Color& pdfDistance, scene_rdl2::math::Color& pdfDistanceDwivedi)
1102{
1103 float distance;
1104 if (choseDwivediSampling) {
1105 distance = -scene_rdl2::math::log(1.0f - uDistance) / sigmaTPrime[chIndex];
1106 } else {
1107 distance = -scene_rdl2::math::log(1.0f - uDistance) / sigmaT[chIndex];
1108 }
1109 tr = scene_rdl2::math::exp(-sigmaT * distance);
1110 pdfDistance = sigmaT*tr;
1111 pdfDistanceDwivedi = sigmaTPrime*exp(-sigmaTPrime*distance);
1112 return distance;
1113}

1136// One-Sample Between PhaseFunction and Dwivedi Direction Sampling
1137// Returns a boolean for that choice, the outputcosine between chosen direction
1138// and the referenceFrame normal and the output direction
1139void
1140sampleDirection(const scene_rdl2::math::Vec2f& uDir,
1141 float backwardDirectionSample, int chIndex, float phaseSamplingWeight,
1142 float wBackwardSampling, const VolumePhase& sssPhaseFunction,
1143 const Ray& sssRay, const scene_rdl2::math::Color& dwivediV0,
1144 const scene_rdl2::math::ReferenceFrame& inReferenceFrame,
1145 bool& choseDwivediSampling, float& outputCosine,
1146 scene_rdl2::math::Vec3f& outputDir)
1147{
1148 if (uDir[0] < phaseSamplingWeight) {
1149 // choose isotropic sampling
1150 choseDwivediSampling = false;
1151 // re-normalize random number
1152 float r = uDir[0] / phaseSamplingWeight;
1153
1154 outputDir = sssPhaseFunction.sample(-sssRay.dir,
1155 r,
1156 uDir[1]);
1157 // choose appropriate cosine for one-sample afterwards
1158 if (backwardDirectionSample < wBackwardSampling) {
1159 outputCosine = dot(outputDir, inReferenceFrame.getN());
1160 } else {
1161 // go 'forwards' along -N
1162 outputCosine = -dot(outputDir, inReferenceFrame.getN());
1163 }
1164 } else {
1165 // choose Dwivedi Sampling
1166 choseDwivediSampling = true;
1167 // re-normalize random number
1168 float r = (uDir[0] - phaseSamplingWeight) / (1.0f - phaseSamplingWeight);
1169 const float v0 = dwivediV0[chIndex];
1170 // Ref "Improved Dwivedi Sampling" Eqn 10
1171 outputCosine = v0 - (v0+1.0f)*scene_rdl2::math::pow((v0-1.0f)/(v0+1.0f), r);
1172 float sine = scene_rdl2::math::sqrt(scene_rdl2::math::max(0.0f, 1.0f - outputCosine*outputCosine));
1173 float phi = scene_rdl2::math::sTwoPi * uDir[1];
1174 float cosPhi, sinPhi;
1175 scene_rdl2::math::sincos(phi, &sinPhi, &cosPhi);
1176 scene_rdl2::math::Vec3f localDir(sine * cosPhi,
1177 sine * sinPhi,
1178 outputCosine);
1179
1180 if (backwardDirectionSample < wBackwardSampling) {
1181 // go backwards along N
1182 outputDir = normalize(inReferenceFrame.localToGlobal(localDir));
1183 } else {
1184 // go 'forwards' along -N
1185 outputDir = -normalize(inReferenceFrame.localToGlobal(localDir));
1186 }
1187 }
1188}

1190// One Sample PDF from sampling Random-Walk between
1191// RGB, Direction and Distance sampling
1192float
1193calculateOneSamplePDF(float NdotDir, const scene_rdl2::math::Vec3f& wo, const scene_rdl2::math::Vec3f& wi,
1194 const scene_rdl2::math::Color& dwivediV0, const scene_rdl2::math::Color& dwivediNormalization,
1195 const VolumePhase& sssPhaseFunction, const scene_rdl2::math::Color& wChannelSampling,
1196 float phaseSamplingWeight, const scene_rdl2::math::Color& pdfDistance, const scene_rdl2::math::Color& pdfDistanceDwivedi)
1197{
1198 const float pdfPhaseFunction = sssPhaseFunction.pdf(wo, wi);
1199 // dwivedi PDF
1200 // Ref "Improved Dwivedi Sampling" Eqn 12
1201 const scene_rdl2::math::Color pdfDwivedi = scene_rdl2::math::sOneOverTwoPi * dwivediNormalization /
1202 (dwivediV0 - NdotDir);
1203
1204 float oneSamplePdf = 0.0f;
1205 for (int c = 0; c < 3; ++c) {
1206 oneSamplePdf += (wChannelSampling[c]*
1207 (phaseSamplingWeight*pdfPhaseFunction*pdfDistance[c] +
1208 (1.0f-phaseSamplingWeight)*pdfDwivedi[c]*pdfDistanceDwivedi[c]));
1209 }
1210 return oneSamplePdf;
1211}

1229scene_rdl2::math::Color
1230PathIntegrator::computeRadiancePathTraceSubsurface(pbr::TLState* pbrTls,
1231 const Bsdf& bsdf, const Subpixel &sp, int cameraId, const PathVertex &pv,
1232 const RayDifferential& ray, const Intersection& isect,
1233 const VolumeSubsurface& volumeSubsurface, const LightSet& lightSet,
1234 bool doIndirect, float rayEpsilon, float shadowRayEpsilon, unsigned &sequenceID,
1235 scene_rdl2::math::Color &ssAov, float* aovs) const
1236{
1237 scene_rdl2::math::Color radiance(0.0f);
1238 // Check if samples are zero OR
1239 // Global Subsurface is Toggled OFF
1240 if (mBssrdfSamples < 1 || !mEnableSSS) {
1241 return radiance;
1242 }
1243
1244 // -----------------------------------------------------------------------------------------------------------------
1245 // Fix (part 1) for MOONRAY-3352:
1246 // Backup the current aov values. For explanation, search "MOONRAY-3352" lower in this function.
1247 float *backupAovs = nullptr;
1248 const int aovNumChannels = pbrTls->mFs->mAovSchema->numChannels();
1249 scene_rdl2::alloc::Arena *arena = pbrTls->mArena;
1250 SCOPED_MEM(arena);
1251
1252 if (aovs != nullptr) {
1253 backupAovs = arena->allocArray<float>(aovNumChannels);
1254 memcpy(backupAovs, aovs, aovNumChannels * sizeof(float));
1255 }
1256 // -----------------------------------------------------------------------------------------------------------------
1257
1258 EXCL_ACCUMULATOR_PROFILE(pbrTls, EXCL_ACCUM_SSS_INTEGRATION);
1259 const Scene *scene = MNRY_VERIFY(pbrTls->mFs->mScene);
1260 // volume coefficient for scatter event sampling and transmittance
1261 const scene_rdl2::math::Color albedo = volumeSubsurface.getAlbedo();
1262 const scene_rdl2::math::Color sigmaT = volumeSubsurface.getSigmaT();
1263 const scene_rdl2::math::Color sigmaS = sigmaT * albedo;
1264
1265 // Dwivedi Sampling Params
1266 // Ref Improved Dwivedi Sampling
1267 // Ref A Zero-Variance based Sampling Scheme for Monte Carlo SSS
1268 // Largest Discrete Eigenvalue of the transport operator
1269 const scene_rdl2::math::Color dwivediV0 = volumeSubsurface.getDwivediV0();
1270
1271 // subsurface scattering is considered as diffuse reflection for now
1272 const BsdfLobe::Type lobeType =
1273 BsdfLobe::Type(BsdfLobe::REFLECTION | BsdfLobe::DIFFUSE);
1274 int nSubsurfaceSample = (pv.nonMirrorDepth == 0) ? mBssrdfSamples : 1;
1275 float time = ray.time;
1276 // potential normal map binding
1277 const scene_rdl2::rdl2::Material* sssMaterial = volumeSubsurface.getMaterial();
1278 const scene_rdl2::rdl2::EvalNormalFunc evalNormalFunc = volumeSubsurface.getEvalNormalFunc();
1279 bool evalNormalBinding = sssMaterial && evalNormalFunc;
1280 // trace set for sss
1281 auto geomTls = pbrTls->mTopLevelTls->mGeomTls.get();
1282 geomTls->mSubsurfaceTraceSet = volumeSubsurface.getTraceSet();
1283 // Fresnel evaluation
1284 scene_rdl2::math::Vec3f nIn = evalNormalBinding ?
1285 evalNormalFunc(sssMaterial, pbrTls->mTopLevelTls->mShadingTls.get(),
1286 shading::State(&isect)) :
1287 isect.getN();
1288 scene_rdl2::math::Color fresnel = evalFresnel(volumeSubsurface.getTransmissionFresnel(),
1289 nIn, -ray.dir);
1290
1291 // We disable the transmission fresnel on the wi side.
1292 // If the material is using a OneMinusRoughSchlickFresnel, this matches the
1293 // energy conservation we are doing with other types of "under" lobes.
1294 // Otherise it darkens the SSS somewhat a bit when the specular factor of
1295 // the "top" lobe is in the 0.2 .. 0.7 range.
1296 const Fresnel* transmissionFresnel = nullptr;
1297
1298 // Make sure we do all local sss sampling with input sequenceID, which will
1299 // get incremented in the process.
1300 unsigned sssSampleID = sequenceID;
1301
1302 // Local Reference Frame
1303 // Using the smooth shading normal here instead of Ng to avoid
1304 // faceting artifacts (similar to the diffusion integrator above)
1305 scene_rdl2::math::ReferenceFrame inReferenceFrame(isect.getN());
1306 const scene_rdl2::math::ReferenceFrame inReferenceFrameNg(isect.getNg());
1307 pbrTls->mStatistics.addToCounter(STATS_SSS_SAMPLES, nSubsurfaceSample);
1308
1309
1310 // Counter to calculate all the *valid* subsurface samples evaluated to
1311 // Divide the throughput correctly
1312 unsigned int samplesEvaluated = 0;
1313 // Weight for Isotropic Sampling
1314 float phaseSamplingWeight = 1.0f;
1315 // Weight for sampling 'back' towards the surface along N for Dwivedi
1316 // Sampling. Save some samples for going 'forwards'
1317 // for thin-geom/back-lighting scenes.
1318 // Ref "Improved Dwivedi Sampling"
1319 const float wBackwardSampling = 0.9f;
1320
1321 for (int s = 0; s < nSubsurfaceSample; ++s) {
1322 // sample sequence for random walk direction
1323 IntegratorSample2D directionSamples(SequenceIDIntegrator(
1324 sp.mPixel, sp.mSubpixelIndex,
1325 SequenceType::BssrdfGlobal,
1326 SequenceType::VolumePhase,
1327 s, sssSampleID));
1328 // sample sequence for selecting RGB channel to do distance sampling
1329 IntegratorSample1D channelSamples(SequenceIDIntegrator(
1330 sp.mPixel, sp.mSubpixelIndex,
1331 SequenceType::BssrdfLocalChannel,
1332 SequenceType::VolumeScattering,
1333 s, sssSampleID));
1334 // sample sequence for random walk scatter distance
1335 IntegratorSample1D distanceSamples(SequenceIDIntegrator(
1336 sp.mPixel, sp.mSubpixelIndex,
1337 SequenceType::BssrdfGlobal,
1338 SequenceType::VolumeScattering,
1339 s, sssSampleID));
1340 // sample sequence for random walk forward/backward sampling
1341 IntegratorSample1D dwivediDirectionSamples(SequenceIDIntegrator(
1342 sp.mPixel, sp.mSubpixelIndex,
1343 SequenceType::BssrdfGlobal,
1344 SequenceType::VolumeScattering,
1345 s, sssSampleID));
1346#if 0
1347 // NOT using RR at the moment
1348 // sample sequence for russian roulette
1349 IntegratorSample1D rrSamples(SequenceIDIntegrator(
1350 sp.mPixel, sp.mSubpixelIndex,
1351 SequenceType::BssrdfGlobal,
1352 SequenceType::RussianRouletteBsdf,
1353 s, sssSampleID));
1354#endif
1355 // setup initial path throughput
1356 scene_rdl2::math::Color pt = volumeSubsurface.getScale() * pv.pathThroughput *
1357 fresnel / nSubsurfaceSample;
1358 if (isBlack(pt)) {
1359 continue;
1360 }
1361 // sample a random walk direction using hemisphere cosine distribution
1362 // along the flip side of shading Normal
1363 scene_rdl2::math::Vec2f uDir;
1364 directionSamples.getSample(&uDir[0], pv.nonMirrorDepth);
1365 scene_rdl2::math::Vec3f dir = -inReferenceFrame.localToGlobal(
1366 sampleLocalHemisphereCosine(uDir[0], uDir[1]));
1367
1368 // Use reverse geometric normal with minimum of rayEpsilon and sHitEpsilonStart (10e-5)
1369 // min(rayEpsilon, sHitEpsilonStart) is used instead of just rayEpsilon for the following reason
1370 // If camera is far away from surface, rayEpsilon becomes much larger than the least scatter_color component
1371 // that's non zero. This results in loss of samples when selecting that color channel during importance sampling.
1372 // This ends up "tinting" the transmittance more to the largest scatter_color component.
1373 // MOONRAY-4260 is an example of this
1374 // Reverse geometric normal is used to avoid self intersection with same triangle.
1375 // This is better than using rayDir or shading normal as grazing angles can cause more self intersections.
1376 // We could compensate by increasing epsilon based on rayDir and geometric normal but grazing angles can cause
1377 // us to use a much larger epsilon leading back to the original issue rayEpsilon issue above.
1378 // https://link.springer.com/content/pdf/10.1007%2F978-1-4842-4427-2_6.pdf has more insight on this
1379 // TODO: do the random walk in volumesubsurface space so we can reduce precision loss even further when camera is
1380 // far away. this involves generating a BVH with just the volume subsurface at render prep time.
1381 scene_rdl2::math::Vec3f org = isect.getP() - inReferenceFrameNg.getN() * scene_rdl2::math::min(rayEpsilon, sHitEpsilonStart);
1382 bool reachSurface = false;
1383 scene_rdl2::math::Vec3f pOut, nOut;
1384 Intersection isectOut;
1385
1386 // ZERO Scatter Integrator
1387 // Direct Transmission Events using mZeroScatterSigmaT
1388 constexpr size_t maxZeroScatterEvents = 1;
1389 for (size_t nScatter = 0; nScatter < maxZeroScatterEvents; ++nScatter) {
1390 if (isBlack(pt)) {
1391 break;
1392 }
1393 // sample the next scattering event based on path throughput and sigmaT
1394 float uChannel, uDistance;
1395 channelSamples.getSample(&uChannel, pv.nonMirrorDepth);
1396 distanceSamples.getSample(&uDistance, pv.nonMirrorDepth);
1397 scene_rdl2::math::Color tr;
1398 float pdfT;
1399 scene_rdl2::math::Color zeroScatterSigmaT = volumeSubsurface.getZeroScatterSigmaT();
1400 float tfar = sampleDistanceZeroScatter(pt, zeroScatterSigmaT,
1401 uChannel, uDistance, tr, pdfT);
1402 // setup random walk ray
1403 Ray sssRay(org, dir, 0.0f, tfar, time);
1404 // setup trace set
1405 sssRay.ext.geomTls = (void*)geomTls;
1406 const scene_rdl2::rdl2::Material* isectMaterial = isect.getMaterial();
1407 if (isectMaterial) {
1408 const shading::Material *materialExt =
1409 &isectMaterial->get<const shading::Material>();
1410 sssRay.ext.materialID = materialExt->getMaterialId();
1411 }
1412 bool intersected = scene->intersectRay(pbrTls->mTopLevelTls, sssRay,
1413 isectOut, lobeType);
1414 if (intersected && isValidIntersection(isectOut, volumeSubsurface)) {
1415 // update throughput
1416 tr = exp(-zeroScatterSigmaT * sssRay.tfar);
1417 // the pdf of tfar sample from the intersection point
1418 // see pbrt3
1419 // http://www.pbr-book.org/3ed-2018/Light_Transport_II_Volume_Rendering/Sampling_Volume_Scattering.html#eq:homogeneous-medium-psurf
1420 // for derivation
1421 // 15.11, 1/n is for uniform sampling
1422 // we use MIS between channels based on throughput instead
1423 float normalization = 1.0f / (pt[0] + pt[1] + pt[2]);
1424 float pdfExitSurf =
1425 pt[0] * normalization * tr[0] +
1426 pt[1] * normalization * tr[1] +
1427 pt[2] * normalization * tr[2];
1428 pt *= tr / pdfExitSurf;
1429
1430 pOut = isectOut.getP();
1431 // construct intersection
1432
1433 // The material of the exiting intersection must match the
1434 // sssMaterial to evaluate the normal map. This ensures that
1435 // the intersection state has all the attributes requested by the
1436 // normal map. There may not be a match when using trace sets.
1437 // In case of a mismatch, we do not attempt to evaluate the subsurface normal map.
1438 if (evalNormalBinding && isectOut.getMaterial() == isectMaterial) {
1439
1440 // Invert Intersection Normals here because even though we'll hit the surfaces
1441 // from the back-side, we want to shade the point on the 'outside' facing geo.
1442 // One of the applications is computing the correct ReferenceFrame to
1443 // for evaluating tangent space normal maps and for
1444 // correct transformations to render space. This should also helps for projected
1445 // normals where we use the surface normal for calculating the correct projection.
1446 // Note that computing tangent space normal on the back side of the geo (-N)
1447 // and inverting the resulting normal is not mathematically equivalent.
1448 isectOut.invertIntersectionNormals();
1449 geom::initIntersectionPhase2(isectOut, pbrTls->mTopLevelTls,
1450 pv.mirrorDepth, pv.glossyDepth, pv.diffuseDepth,
1451 isSubsurfaceAllowed(pv.subsurfaceDepth), pv.minRoughness,
1452 sssRay.dir);
1453 RayDifferential rayDiff(ray,
1454 isectOut.getEpsilonHint(),
1455 sssRay.tfar);
1456 isectOut.transferAndComputeDerivatives(pbrTls->mTopLevelTls,
1457 &rayDiff,
1458 sp.mTextureDiffScale);
1459
1460 // We evaluate using the sssMaterial, which does not always match the isectMaterial.
1461 // Example: sssMaterial is undermaterial of isectMaterial, which is a GlitterFlakeMaterial.
1462 nOut = evalNormalFunc(sssMaterial,
1463 pbrTls->mTopLevelTls->mShadingTls.get(),
1464 shading::State(&isectOut));
1465 } else {
1466 geom::initIntersectionPhase2(isectOut, pbrTls->mTopLevelTls,
1467 pv.mirrorDepth, pv.glossyDepth, pv.diffuseDepth,
1468 isSubsurfaceAllowed(pv.subsurfaceDepth), pv.minRoughness,
1469 -sssRay.dir);
1470 nOut = -isectOut.getN();
1471 }
1472 isectOut.setCameraId(cameraId);
1473 // used for subsurface material aovs
1474 ssAov += pt;
1475 reachSurface = true;
1476 samplesEvaluated += 1;
1477 // we are getting out of surface
1478 // calculate the radiance contribution of random walk exit point
1479 // Count The Samples Evaluated
1480 const bool includeCosineTerm = true, isEntering = true;
1481 shading::BsdfSlice sliceLocal(nOut, -ray.getDirection(),
1482 includeCosineTerm, isEntering,
1483 pbrTls->mFs->mShadowTerminatorFix);
1484 const bool isReflection = true;
1485 LambertBsdfLobe lobeLocal(nOut, scene_rdl2::math::sWhite, isReflection);
1486 radiance += computeRadianceSubsurfaceSample(
1487 pbrTls, bsdf, sp, cameraId, pv, ray, isectOut.getdNdx(), isectOut.getdNdy(),
1488 pt, transmissionFresnel, lightSet, lobeLocal, sliceLocal, pOut, nOut,
1489 nSubsurfaceSample, 1, s, doIndirect, rayEpsilon, shadowRayEpsilon,
1490 sequenceID, sequenceID, true, aovs);
1491 }
1492 }
1493
1494 if (reachSurface == true) {
1495 // Had a Zero Scatter Event
1496 // Try another sample
1497 continue;
1498 }
1499
1500 // 1+ Scatter Event
1501 // For now using isotropic phase function (what Hyperion does)
1502 VolumePhase sssPhaseFunction = VolumePhase(0.0f);
1503 // set a maximum scattering depth in case we go into
1504 // infinite random walk (due to bad geometry setup)
1505 // the hardcoded number is mentioned in Pixar's memo
1506
1507 scene_rdl2::math::Color tr;
1508 bool choseDwivediSampling = false;
1509 // set a maximum scattering depth in case we go into
1510 // infinite random walk (due to bad geometry setup)
1511 // the hardcoded number is mentioned in Pixar's memo
1512 constexpr size_t maxScatterEvents = 256;
1513 bool hadScatterEvent = false;
1514 for (size_t nScatter = 0; nScatter < maxScatterEvents; ++nScatter) {
1515 if (isBlack(pt)) {
1516 break;
1517 }
1518
1519 // Start using Dwivedi Sampling after a few
1520 // Phase-Function related bounces inside
1521 // Ref : A Zero-Variance Based Sampling for MonteCarlo SSS
1522 // Paper suggests a weight distribution of (0.25f, 0.75f) towards
1523 // Dwivedi Sampling, but in our tests (0.1f, 0.9f) seems to
1524 // work best after a couple of steps inside. Starting Dwivedi
1525 // at the first steps seems to result in much less saturated results
1526 phaseSamplingWeight = (nScatter > 1) ? 0.1f : 1.0f;
1527
1528 // sample the next scattering event based on path throughput and sigmaT
1529 // Sample between RGB
1530 float uChannel;
1531 channelSamples.getSample(&uChannel, pv.nonMirrorDepth);
1532 scene_rdl2::math::Color channelSamplingWeights;
1533 int chIndex = sampleChannel(pt, uChannel, channelSamplingWeights);
1534
1535 // Sample Distance
1536 float uDistance;
1537 distanceSamples.getSample(&uDistance, pv.nonMirrorDepth);
1538 float NdotDir = dot(dir, inReferenceFrame.getN());
1539 // sigmaTP for Dwivedi Sampling
1540 // Dwivedi Distance Sampling - take larger steps when going back
1541 // towards the normal
1542 // Ref "Improved Dwivedi Sampling" Eqn 13
1543 scene_rdl2::math::Color sigmaTPrime = sigmaT*(1.0f - NdotDir / dwivediV0);
1544 scene_rdl2::math::Color pdfDistance, pdfDistanceDwivedi;
1545 float tfar = sampleDistanceMultiScatter(sigmaT, sigmaTPrime,
1546 channelSamplingWeights, chIndex, uDistance, NdotDir, dwivediV0,
1547 choseDwivediSampling, tr, pdfDistance, pdfDistanceDwivedi);
1548
1549 // setup random walk ray
1550 Ray sssRay(org, dir, 0.0f, tfar, time);
1551 // setup trace set
1552 sssRay.ext.geomTls = (void*)geomTls;
1553 const scene_rdl2::rdl2::Material* isectMaterial = isect.getMaterial();
1554 if (isectMaterial) {
1555 const shading::Material *materialExt =
1556 &isectMaterial->get<const shading::Material>();
1557 sssRay.ext.materialID = materialExt->getMaterialId();
1558 }
1559 bool intersected = scene->intersectRay(pbrTls->mTopLevelTls, sssRay,
1560 isectOut, lobeType);
1561 if (intersected && isValidIntersection(isectOut, volumeSubsurface)) {
1562 // update throughput
1563 if (hadScatterEvent == false) {
1564 // we've already accounted for zero scattering in the loop above.
1565 // we should not count this sample because the sigmaT here is not
1566 // calibrated for zeroScatter and will cause hue-shifts for thin geo.
1567 break;
1568 } else {
1569 tr = exp(-sigmaT * sssRay.tfar);
1570 }
1571
1572 reachSurface = true;
1573
1574 // transmittance per dwivedi sampling
1575 scene_rdl2::math::Color trp = exp(-sigmaTPrime*sssRay.tfar);

1716 // few samples can potentially not reach the surface.

1718 radiance *= scaleFactor;

1721 // -------------------------------------------------------------------------------------------------------------
1722 // Fix for MOONRAY-3352:
1723 // Use the backed-up aov values to modify the aovs consistently with the scaled radiance.
1724 //
1725 // Essentially, we only scale the part of each aov that's been accumulated since the start of this function.
1726 // But there's an additional consideration - this function can be called recursively. So lets discuss how this
1727 // scaling needs to work.
1728 //
1729 // Consider 2 alternative software models for accumulating radiance across a sequence of nested function calls:
1730 //
1731 // (1) Each function has a local radiance variable which it initialises to 0 and then proceeds to add to it.
1732 // Each summand could either be a simple directly evaluated expression, or the value returned by a nested
1733 // function call. At the end of the function, the locally accumulated radiance is returned, so that the calling
1734 // function can accumulate it to its own scoped value using radiance += func(); where func() is the function
1735 // we just exited and which returned the locally accumulated radiance.
1736 //
1737 // (2) There's a single global radiance accumulator, which lives in the topmost function, and is passed by
1738 // reference into lower level functions. All functions modify this radiance variable directly, and do not
1739 // therefore need to return a radiance value.
1740 //
1741 // Moonray actually uses a mixture of both models to compute a final radiance value, but clearly the present
1742 // function is using model 1. The additional complication it introduces is that the radiance is being scaled
1743 // (sometimes) by a scale factor just before being returned.
1744 //
1745 // Let's consider how we would perform the same scaling if we were to (hypothetically) modify this function
1746 // to use model 2 instead. If the global radiance has value r0 at the beginning of the function call, and after
1747 // the various accumulations performed by the function, it has value r1 at the end, then the difference in
1748 // radiance r1-r0 corresponds to the locally accumulated value r that gets returned by the function under model
1749 // 1. If model 1 now applies a scale factor s to r, and returns the final value r*s, under model 2 the portion
1750 // of the accumulated radiance to be scaled by s is r1-r0, so the adjusted value at the end of the function
1751 // should be r0 + s*(r1-r0) = lerp(r0, r1, s). Note that this is true regardless of what expressions any
1752 // lower-level functions are using to compute their radiance contributions, including, crucially, the same
1753 // type of scaling performed by lower level recursive instantiations of the same function. These details are
1754 // irrelevant to the present scope of the function because they are all hidden internal details of the nested
1755 // function calls. So the lerp is still the appropriate thing to do even in the recursive case.
1756 //
1757 // The reason it was important to work through that hypothetical case is that the aovs exclusively use model 2
1758 // to perform their accumulations. This makes good sense because some renders use large numbers of aovs, and it
1759 // would require lots of memory plus the additional execution cost of zeroing that memory at the start of each
1760 // function to support model 1 for aovs. Hence to support the local scaling of aovs, using model 2, we apply
1761 // the lerp indicated above on each aov channel. Note that we backed up the values of all aovs on entry to this
1762 // function.
1763
1764 if (aovs != nullptr) {
1765 for (int i = 0; i < aovNumChannels; ++i) {
1766 aovs[i] = scene_rdl2::math::lerp(backupAovs[i], aovs[i], scaleFactor);
1767 }
1768 }
1769
1770 // -------------------------------------------------------------------------------------------------------------
1771 }
1772 return radiance;
1773}
