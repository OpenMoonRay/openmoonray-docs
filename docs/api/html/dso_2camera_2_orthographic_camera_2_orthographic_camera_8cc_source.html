24 math::Mat4f computeProjectionMatrix(float t, const std::array<float, 4>& window, float /*interocularOffset*/) const override
25 {
26 // Get film-back width (mm)
27 float filmWidth = get(attrFilmWidthApertureKey);
28 filmWidth = max(filmWidth, 0.01f);
29 float halfFilmWidth = filmWidth * 0.5f;
30
31 // Get near plane
32 float near = get(rdl2::Camera::sNearKey);
33 near = max(near, 0.01f);
34
35 // Get far plane
36 float far = get(rdl2::Camera::sFarKey);
37 far = max(far, 0.01f);
38
39 // Get film back offset
40 float filmOffsetX = get(attrHorizontalFilmOffset);
41 float filmOffsetY = get(attrVerticalFilmOffset);
42
43 // Get pixel aspect ratio
44 float par = get(attrPixelAspectRatio);
45 if (isEqual(par, 0.0f)) {
46 par = 1.0f;
47 }
48
49 // Compute window corners on the near plane
50 float windowNear[4];
51 windowNear[0] = (window[0] * halfFilmWidth + filmOffsetX);
52 windowNear[1] = (window[1] * halfFilmWidth + filmOffsetY) / par;
53 windowNear[2] = (window[2] * halfFilmWidth + filmOffsetX);
54 windowNear[3] = (window[3] * halfFilmWidth + filmOffsetY) / par;
55
56 // See Appendix C.5 of the SGI Graphics Library Programming Guide
57 // for format of projection matrix.
58 Mat4f c2s;
59 c2s[0][0] = 2.0f / (windowNear[2] - windowNear[0]);
60 c2s[0][1] = 0.0f;
61 c2s[0][2] = 0.0f;
62 c2s[0][3] = 0.0f;
63
64 c2s[1][0] = 0.0f;
65 c2s[1][1] = 2.0f / (windowNear[3] - windowNear[1]);
66 c2s[1][2] = 0.0f;
67 c2s[1][3] = 0.0f;
68
69 c2s[2][0] = 0.0f;
70 c2s[2][1] = 0.0f;
71 c2s[2][2] = -2.0f / (far - near);
72 c2s[2][3] = 0.0f;
73
74 c2s[3][0] = -(windowNear[2] + windowNear[0])/(windowNear[2] - windowNear[0]);
75 c2s[3][1] = -(windowNear[3] + windowNear[1])/(windowNear[3] - windowNear[1]);
76 c2s[3][2] = -(far + near) / (far - near);
77 c2s[3][3] = 1.0f;
78
79 return c2s;
80 }
