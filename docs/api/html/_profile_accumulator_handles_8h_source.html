141inline bool
142ExclusiveAccumulators::push(ExclAccType type)
143{
144 MNRY_ASSERT(type < NUM_EXCLUSIVE_ACC);
145 MNRY_ASSERT_REQUIRE(mStackSize < MAX_EXCL_ACCUM_STACK_SIZE);
146
147 if (mStackSize) {
148 // Check if the desired accumulator is already running and if so,
149 // treat this as a no-op.
150 ThreadLocalAccumulator *stackTop = mStack[mStackSize - 1];
151 if (stackTop == mAccumulators[type])
152 return false;
153
154 stopAccumulator(stackTop);
155 }
156
157 ThreadLocalAccumulator *acc = mAccumulators[type];
158 startAccumulator(acc);
159
160 mStack[mStackSize++] = acc;
161
162 return true;
163}
164
165inline void
166ExclusiveAccumulators::pop()
167{
168 MNRY_ASSERT(mStackSize);
169
170 stopAccumulator(mStack[mStackSize - 1]);
171
172 --mStackSize;
173
174 if (mStackSize) {
175 startAccumulator(mStack[mStackSize - 1]);
176 }
177}

207inline bool

209{
210#ifdef PROFILE_ACCUMULATORS_ENABLED

221class ScopedExclAccumulator
222{
223public:
224 ScopedExclAccumulator(ExclusiveAccumulators *exclAcc, ExclAccType type) :
225 mExclAcc(nullptr)
226 {
227 if (exclAcc && exclAcc->push(type)) {
228 mExclAcc = exclAcc;
229
230#ifdef DEBUG
231 mDebugStackSize = exclAcc->getStackSize();
232#endif
233 }
234 }
235
236 ~ScopedExclAccumulator()
237 {
238 pop();
239 }
240
241 void pop()
242 {
243 if (mExclAcc) {
244#ifdef DEBUG
245 MNRY_ASSERT(mDebugStackSize == mExclAcc->getStackSize());
246#endif
247 mExclAcc->pop();
248 mExclAcc = nullptr;
249 }
250 }
251
252private:
253 ExclusiveAccumulators * mExclAcc;
254
255#ifdef DEBUG
256 unsigned mDebugStackSize; // For debugging only.
257#endif
258};
