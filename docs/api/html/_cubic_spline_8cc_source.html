45template <typename CvType>
46static __forceinline CvType evalCubicFromWeights(
47 const float& w0, const float& w1, const float& w2, const float& w3,
48 const CvType& cv0, const CvType& cv1, const CvType& cv2, const CvType& cv3)
49{
50 return w0 * cv0 + w1 * cv1 + w2 * cv2 + w3 * cv3;
51}

53template <typename AttributeType> bool
54getAttribute(const Attributes* attributes,
55 const TypedAttributeKey<AttributeType> key,
56 const uint32_t vertex,
57 const int primID,
58 AttributeType& value)
59{
60 if (!attributes->isSupported(key)) {
61 return false;
62 }
63
64 // Data for points is typically stored as constant
65 // (one value for all points) or varying (one value
66 // per point).
67 AttributeRate rate = attributes->getRate(key);
68 switch (rate) {
69 case RATE_CONSTANT:
70 value = attributes->getConstant(key);
71 break;
72 case RATE_VARYING:
73 value = attributes->getVarying(key, primID);
74 break;
75 case RATE_VERTEX:
76 value = attributes->getVertex(key, vertex);
77 break;
78 default:
79 return false;
80 }
81
82 return true;
83}

85void
86CubicSpline::postIntersect(mcrt_common::ThreadLocalState &tls,
87 const scene_rdl2::rdl2::Layer* layer, const mcrt_common::Ray& ray,
88 shading::Intersection& intersection) const
89{
90 int spanId = ray.primID;
91 const IndexData& index = mIndexBuffer[spanId];
92 const uint32_t chain = index.mChain;
93
94 int assignmentId =
95 mLayerAssignmentId.getType() == LayerAssignmentId::Type::CONSTANT ?
96 mLayerAssignmentId.getConstId() :
97 mLayerAssignmentId.getVaryingId()[chain];
98 intersection.setLayerAssignments(assignmentId, layer);
99
100 const AttributeTable *table =
101 intersection.getMaterial()->get<shading::RootShader>().getPrimitiveAttributeTable();
102 intersection.setTable(&tls.mArena, table);
103 overrideInstanceAttrs(ray, intersection);
104
105 const uint32_t vertexOffset = index.mVertex;
106
107 Vec3fa cv[4];
108 if (!isMotionBlurOn()) {
109 // Grab the CVs
110 cv[0] = mVertexBuffer(vertexOffset + 0);
111 cv[1] = mVertexBuffer(vertexOffset + 1);
112 cv[2] = mVertexBuffer(vertexOffset + 2);
113 cv[3] = mVertexBuffer(vertexOffset + 3);
114 } else {
115 // only support two time samples for motionblur at this moment
116 MNRY_ASSERT(getMotionSamplesCount() == 2);
117 float wt = ray.time;
118 cv[0] = lerp(mVertexBuffer(vertexOffset + 0, 0), mVertexBuffer(vertexOffset + 0, 1), wt);
119
120 cv[1] = lerp(mVertexBuffer(vertexOffset + 1, 0), mVertexBuffer(vertexOffset + 1, 1), wt);
121
122 cv[2] = lerp(mVertexBuffer(vertexOffset + 2, 0), mVertexBuffer(vertexOffset + 2, 1), wt);
123
124 cv[3] = lerp(mVertexBuffer(vertexOffset + 3, 0), mVertexBuffer(vertexOffset + 3, 1), wt);
125 }
126
127 Vec3f Ng, N, dPds, dPdt;
128 Vec2f St;
129
130 // Partial with respect to S is easy enough -- its the tangent vector
131 // TODO: This isn't varying with R down the curve in the direction of
132 // the tangent vector, which it should. We should slope down by R'.
133 dPds = evalCubicDeriv(ray.u, cv[0], cv[1], cv[2], cv[3]);
134
135 if (mSubType == SubType::RAY_FACING) {
136 // Use the flipped ray direction to compute the normal. But, this may be an
137 // instance, so we need to transform the ray direction into the local
138 // space. It gets transformed back to render space later on.
139 // Note that it looks like we should be using r2l to get from render to
140 // local space, but recall that when transforming normals we use the transpose
141 // of the inverse, which is l2r transposed. The transformNormal() function
142 // uses the inverse of the transform (l2r) and applies the transpose.
143 const Vec3f flippedRayDir = ray.isInstanceHit() ?
144 normalize(transformNormal(ray.ext.l2r, -ray.dir)) :
145 normalize(-ray.dir);
146
147 // Lacking much better options, we'll construct dPdt wrt the other two
148 // components of the ostensibly orthonormal frame we have
149 dPdt = cross(dPds, flippedRayDir);
150
151 if (unlikely(lengthSqr(dPdt) < scene_rdl2::math::sEpsilon)) {
152 // Degened dPds or dPdt. The reason can be either clustered cv points
153 // or ray direction is parallel to dPds. Use ReferenceFrame as fallback
154 N = flippedRayDir;
155 scene_rdl2::math::ReferenceFrame frame(N);
156 dPds = frame.getX();
157 dPdt = frame.getY();
158 } else {
159 // Cross the partials to get the normal
160 N = cross(dPdt, dPds);
161 }
162 N.normalize();
163 } else if (mSubType == SubType::ROUND) {
164 // round curves just use the geom normal
165 N = ray.getNg();
166 N.normalize();
167
168 dPdt = cross(dPds, N);
169 if (unlikely(lengthSqr(dPdt) < scene_rdl2::math::sEpsilon)) {
170 // Degened dPds or dPdt. The reason can be either clustered cv points
171 // or ray direction is parallel to dPds. Use ReferenceFrame as fallback
172 scene_rdl2::math::ReferenceFrame frame(N);
173 dPds = frame.getX();
174 dPdt = frame.getY();
175 }
176 }
177
178 // Geometric normal equals shading normal for a cubic spline
179 Ng = N;
180
181 const uint32_t spansInChain = getSpansInChain(chain);
182 const float stSpanRange = 1.0f / spansInChain;
183 const uint32_t indexInChain = index.mSpan;
184 Vec3f uv;
185 const Attributes* attributes = getAttributes();
186 if (intersection.isProvided(StandardAttributes::sUv)) {
187 uv = intersection.getAttribute<Vec3f>(StandardAttributes::sUv);
188 St.x = uv.x;
189 St.y = uv.y;
190 } else if (getAttribute<Vec3f>(attributes,
191 StandardAttributes::sUv,
192 vertexOffset,
193 ray.primID,
194 uv)) {
195 St.x = uv.x;
196 St.y = uv.y;
197 } else {
198 const float stStart = stSpanRange * indexInChain;
199 St[0] = stStart + ray.u * stSpanRange;
200 // ray.v is in (-1,1) so we need to remap to (0,1)
201 St[1] = ray.v * 0.5f + 0.5f;
202 }
203
204 intersection.setDifferentialGeometry(Ng, N, St, dPds, dPdt, true);
205
206 float w0, w1, w2, w3;
207 evalWeights(ray.u, w0, w1, w2, w3);
208
209 const scene_rdl2::rdl2::Geometry* geometry = intersection.getGeometryObject();
210 MNRY_ASSERT(geometry != nullptr);
211 if (geometry->getRayEpsilon() <= 0.0f) {
212 // Use the 1.5 times the curve radius as the next ray epsilon hint.
213 // This lets shadow and continuation rays escape
214 // the intersected hair, instead of self-intersecting.
215 float curveRadius = w0 * cv[0].w + w1 * cv[1].w + w2 * cv[2].w + w3 * cv[3].w;
216 intersection.setEpsilonHint( 1.5 * curveRadius );
217 } else {
218 intersection.setEpsilonHint( geometry->getRayEpsilon() );
219 }
220
221 int varyingOffset = chain + spanId;
222 CubicSplineInterpolator interpolator(getAttributes(), ray.time, chain,
223 varyingOffset, indexInChain, ray.u, vertexOffset, w0, w1, w2, w3);
224 intersection.setRequiredAttributes(interpolator);
225
226 // calculate dfds/dfdt for primitive attributes that request derivatives
227 if (table->requestDerivatives()) {
228 // ds = stSpanRange -> 1 / ds = 1 / stSpanRange = spansInChain
229 computeAttributesDerivatives(table, ray.u, (float)spansInChain,
230 chain, varyingOffset, indexInChain, vertexOffset,
231 ray.time, intersection);
232 }
233
234 // "polygon" vertices
235 if (table->requests(StandardAttributes::sNumPolyVertices)) {
236 intersection.setAttribute(StandardAttributes::sNumPolyVertices, 4);
237 }
238 if (table->requests(StandardAttributes::sPolyVertexType)) {
239 intersection.setAttribute(StandardAttributes::sPolyVertexType,
240 static_cast<int>(StandardAttributes::POLYVERTEX_TYPE_CUBIC_SPLINE));
241 }
242 for (int iVert = 0; iVert < 4; iVert++) {
243 if (table->requests(StandardAttributes::sPolyVertices[iVert])) {
244 // may need to move the vertices to render space
245 // for instancing object since they are ray traced in local space
246 const Vec3f v = ray.isInstanceHit() ? transformPoint(ray.ext.l2r, cv[iVert].asVec3f())
247 : cv[iVert].asVec3f();
248 intersection.setAttribute(StandardAttributes::sPolyVertices[iVert], v);
249 }
250 }
251
252 // motion vectors
253 if (table->requests(StandardAttributes::sMotion)) {
254 Vec3f pos0, pos1;
255 const Vec3f *pos1Ptr = nullptr;
256 if (!isMotionBlurOn()) {
257 const Vec3fa v0 = evalCubic(ray.u,
258 mVertexBuffer(vertexOffset + 0, 0), mVertexBuffer(vertexOffset + 1, 0),
259 mVertexBuffer(vertexOffset + 2, 0), mVertexBuffer(vertexOffset + 3, 0));
260 pos0 = v0.asVec3f();
261 pos1Ptr = nullptr;
262 } else {
263 float wc = ray.getTime() - sHalfDt;
264 const Vec3fa v0 = evalCubic(ray.u,
265 lerp(mVertexBuffer(vertexOffset + 0, 0), mVertexBuffer(vertexOffset + 0, 1), wc),
266 lerp(mVertexBuffer(vertexOffset + 1, 0), mVertexBuffer(vertexOffset + 1, 1), wc),
267 lerp(mVertexBuffer(vertexOffset + 2, 0), mVertexBuffer(vertexOffset + 2, 1), wc),
268 lerp(mVertexBuffer(vertexOffset + 3, 0), mVertexBuffer(vertexOffset + 3, 1), wc));
269 wc = ray.getTime() + sHalfDt;
270 const Vec3fa v1 = evalCubic(ray.u,
271 lerp(mVertexBuffer(vertexOffset + 0, 0), mVertexBuffer(vertexOffset + 0, 1), wc),
272 lerp(mVertexBuffer(vertexOffset + 1, 0), mVertexBuffer(vertexOffset + 1, 1), wc),
273 lerp(mVertexBuffer(vertexOffset + 2, 0), mVertexBuffer(vertexOffset + 2, 1), wc),
274 lerp(mVertexBuffer(vertexOffset + 3, 0), mVertexBuffer(vertexOffset + 3, 1), wc));
275 pos0 = v0.asVec3f();
276 pos1 = v1.asVec3f();
277 pos1Ptr = &pos1;
278 }
279
280 // Motion vectors only support a single instance level, hence we only care
281 // about ray.instance0.
282 const Instance *instance = (ray.isInstanceHit())?
283 static_cast<const Instance *>(ray.ext.instance0OrLight) : nullptr;
284 const Vec3f motion = computePrimitiveMotion(pos0, pos1Ptr, ray.getTime(), instance);
285 intersection.setAttribute(StandardAttributes::sMotion, motion);
286 }
287}

290bool
291CubicSpline::computeIntersectCurvature(const mcrt_common::Ray& ray,
292 const shading::Intersection& intersection, Vec3f& dNds, Vec3f& dNdt) const
293{
294 // dNds and dNdt are not geometrically useful because the cuve is a flat
295 // ribbon whose normal faces the ray. This means N is dependent on
296 // ray direction rather than on the geometry of the curve alone.
297 dNds = Vec3f(0.0f, 0.0f, 0.0f);
298 dNdt = Vec3f(0.0f, 0.0f, 0.0f);
299 return true;
300}

302void
303CubicSpline::computeAttributesDerivatives(const AttributeTable* table,
304 float u, float invDs, int chain,
305 int varyingOffset, int faceVaryingOffset, int vertexOffset,
306 float time, shading::Intersection& intersection) const
307{
308 Attributes* attrs = getAttributes();
309 for (auto key: table->getDifferentialAttributes()) {
310 if (!attrs->isSupported(key)) {
311 continue;
312 }
313 switch (key.getType()) {
314 case scene_rdl2::rdl2::TYPE_FLOAT:
315 computeAttributeDerivatives(TypedAttributeKey<float>(key),
316 u, invDs, chain, varyingOffset, faceVaryingOffset, vertexOffset,
317 time, intersection);
318 break;
319 case scene_rdl2::rdl2::TYPE_RGB:
320 computeAttributeDerivatives(TypedAttributeKey<scene_rdl2::math::Color>(key),
321 u, invDs, chain, varyingOffset, faceVaryingOffset, vertexOffset,
322 time, intersection);
323 break;
324 case scene_rdl2::rdl2::TYPE_RGBA:
325 computeAttributeDerivatives(TypedAttributeKey<scene_rdl2::math::Color4>(key),
326 u, invDs, chain, varyingOffset, faceVaryingOffset, vertexOffset,
327 time, intersection);
328 break;
329 case scene_rdl2::rdl2::TYPE_VEC2F:
330 computeAttributeDerivatives(TypedAttributeKey<scene_rdl2::math::Vec2f>(key),
331 u, invDs, chain, varyingOffset, faceVaryingOffset, vertexOffset,
332 time, intersection);
333 break;
334 case scene_rdl2::rdl2::TYPE_VEC3F:
335 computeAttributeDerivatives(TypedAttributeKey<scene_rdl2::math::Vec3f>(key),
336 u, invDs, chain, varyingOffset, faceVaryingOffset, vertexOffset,
337 time, intersection);
338 break;
339 case scene_rdl2::rdl2::TYPE_MAT4F:
340 computeAttributeDerivatives(TypedAttributeKey<scene_rdl2::math::Mat4f>(key),
341 u, invDs, chain, varyingOffset, faceVaryingOffset, vertexOffset,
342 time, intersection);
343 break;
344 default:
345 break;
346 }
347 }
