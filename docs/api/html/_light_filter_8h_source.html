115 void init(std::unique_ptr<const LightFilter* []>&& lightFilters, int count)
116 {
117 // LightFilterList now owns the array of LightFilters.
118 mLightFilters = std::move(lightFilters);
119 mLightFilterCount = count;
120 mNeedsLightXform = false;
121
122 for (int i = 0; i < mLightFilterCount; i++) {
123 if (mLightFilters[i]->needsLightXform()) {
124 mNeedsLightXform = true;
125 break;
126 }
127 }
128 }
129
130 int getLightFilterCount() const { return mLightFilterCount; }
131 const LightFilter *getLightFilter(int idx) const { return mLightFilters[idx]; }
132 bool needsLightXform() const { return mNeedsLightXform; }
133
134private:
135 LIGHT_FILTER_LIST_MEMBERS;
136};

140inline bool canIlluminateLightFilterList(const LightFilterList* lightFilterList,
141 const LightFilter::CanIlluminateData& data)
142{
143 MNRY_ASSERT(lightFilterList);
144 size_t lightFilterCount = lightFilterList->getLightFilterCount();
145 for (size_t i = 0; i < lightFilterCount; ++i) {
146 const LightFilter *lightFilter = lightFilterList->getLightFilter(i);
147 MNRY_ASSERT(lightFilter);
148 if (!lightFilter->canIlluminate(data)) {
149 return false;
150 }
151 }
152 return true;
153}
154
155inline void evalLightFilterList(const LightFilterList* lightFilterList,
156 const LightFilter::EvalData& data,
157 scene_rdl2::math::Color& radiance)
158{
159 MNRY_ASSERT(lightFilterList);
160 size_t lightFilterCount = lightFilterList->getLightFilterCount();
161 for (size_t i = 0; i < lightFilterCount; ++i) {
162 const LightFilter *lightFilter = lightFilterList->getLightFilter(i);
163 MNRY_ASSERT(lightFilter);
164 radiance *= lightFilter->eval(data);
165 }
166}
