42RectLight::computeCorners(Vec3f *corners, float time) const
43{

50float
51RectLight::planeDistance(const Vec3f &p, float time) const
52{
53 Plane pl = isMb() ? Plane(getPosition(time), getDirection(time)) : mRenderPlane;
54
55 float distance = pl.getDistance(p);
56 if (mSidedness == LightSidednessType::LIGHT_SIDEDNESS_REVERSE) {
57 distance = -distance;
58 } else if (mSidedness == LightSidednessType::LIGHT_SIDEDNESS_2_SIDED) {
59 distance = scene_rdl2::math::abs(distance);
60 }
61
62 return distance;
63}

65bool
66RectLight::getOverlapBounds(const Vec2f& localP, float localLength,
67 Vec2f& center, float& width, float& height) const
68{
69 float minX = math::max(localP.x - localLength, -mHalfWidth);
70 float maxX = math::min(localP.x + localLength, mHalfWidth);
71 float minY = math::max(localP.y - localLength, -mHalfHeight);
72 float maxY = math::min(localP.y + localLength, mHalfHeight);
73
74 if (maxX <= minX || maxY <= minY) {
75 // region does not overlap with RectLight
76 return false;
77 }
78
79 center.x = 0.5f * (maxX + minX);
80 center.y = 0.5f * (maxY + minY);
81 width = maxX - minX;
82 height = maxY - minY;
83
84 return true;
85}

97bool
98RectLight::updateTransforms(const Mat4f &local2Render, int ti)
99{
100 // update transforms at time index (ti)
101 // ti == 0 is at normalized rayTime = 0.f
102 // ti == 1 is at normalized rayTime = 1.f;
103 MNRY_ASSERT(getRdlLight());
104
105 Mat4f local2RenderCopy = local2Render;
106 if (extractUniformScale(local2Render, &mLocal2RenderScale[ti])) {
107 mRender2LocalScale[ti] = 1.0f / math::max(sEpsilon, mLocal2RenderScale[ti]);
108 mWidth = mRdlLight->get<scene_rdl2::rdl2::Float>(sWidthKey);
109 mHeight = mRdlLight->get<scene_rdl2::rdl2::Float>(sHeightKey);
110 } else {
111 mRender2LocalScale[ti] = 1.0f;
112 Vec3f nonUniformScale = extractNonUniformScale(local2Render);
113 mWidth = mRdlLight->get<scene_rdl2::rdl2::Float>(sWidthKey) * nonUniformScale.x;
114 mHeight = mRdlLight->get<scene_rdl2::rdl2::Float>(sHeightKey) * nonUniformScale.y;
115 local2RenderCopy = Mat4f(Vec4f(normalize(asVec3(local2Render.row0()))),
116 Vec4f(normalize(asVec3(local2Render.row1()))),
117 Vec4f(normalize(asVec3(local2Render.row2()))),
118 local2Render.row3());
119 }
120 mHalfWidth = mWidth * 0.5f;
121 mHalfHeight = mHeight * 0.5f;
122
123 mLocal2Render[ti] = Xform3f(Mat3f(asVec3(local2RenderCopy.row0()),
124 asVec3(local2RenderCopy.row1()),
125 asVec3(local2RenderCopy.row2())),
126 asVec3(local2RenderCopy.row3()));
127
128 // WARNING: this can be inaccurate, may lead to float precision jitter
129 mRender2Local[ti] = mLocal2Render[ti].inverse();
130
131 Mat4f local2RenderRot = local2RenderCopy * Mat4f::scale(Vec4f(mRender2LocalScale[ti]));
132
133 // WARNING: Includes translation, ok because only used with transformVector
134 mLocal2RenderRot[ti] = Xform3f(Mat3f(asVec3(local2RenderRot.row0()),
135 asVec3(local2RenderRot.row1()),
136 asVec3(local2RenderRot.row2())),
137 asVec3(local2RenderRot.row3()));
138
139 // WARNING: Includes translation, ok because only used with transformVector
140 // and also getXformRender2Local which uses the translation.
141 mRender2LocalRot[ti] = mRender2Local[ti] * Xform3f::scale(Vec3f(mLocal2RenderScale[ti]));
142
143 return true;
144}

149bool
150RectLight::update(const Mat4d& world2render)
151{
152 MNRY_ASSERT(mRdlLight);
153
154 mOn = mRdlLight->get(scene_rdl2::rdl2::Light::sOnKey);
155 if (!mOn) {
156 return false;
157 }
158
159 updateVisibilityFlags();
160 updatePresenceShadows();
161 updateRayTermination();
162 updateTextureFilter();
163 updateMaxShadowDistance();
164
165 mWidth = mRdlLight->get<scene_rdl2::rdl2::Float>(sWidthKey);
166 mHeight = mRdlLight->get<scene_rdl2::rdl2::Float>(sHeightKey);
167 mHalfWidth = mWidth * 0.5f;
168 mHalfHeight = mHeight * 0.5f;
169
170 // mSpread: 1 = 90 degree cone angle, 0 = 0 degree cone angle (cross section is square).
171 mSpread = math::clamp(mRdlLight->get<scene_rdl2::rdl2::Float>(sSpreadKey), math::sEpsilon, 1.f);
172 mTanSpreadTheta = math::tan(mSpread * math::sHalfPi);
173
174 UPDATE_ATTRS_CLEAR_RADIUS
175
176 const Mat4d l2w0 = mRdlLight->get(scene_rdl2::rdl2::Node::sNodeXformKey, /* rayTime = */ 0.f);
177 const Mat4d l2w1 = mRdlLight->get(scene_rdl2::rdl2::Node::sNodeXformKey, /* rayTime = */ 1.f);
178
179 const Mat4f local2Render0 = sRotateX180 * toFloat(l2w0 * world2render);
180 const Mat4f local2Render1 = sRotateX180 * toFloat(l2w1 * world2render);
181
182 if (!updateParamAndTransforms(local2Render0, local2Render1, mHalfWidth, mHalfHeight)) {
183 return false;
184 }
185
186 // Compute render space corners of light.
187 // Cached at rayTime = 0
188 computeCorners(mRenderCorners, 0.f);
189
190 // Compute render space quantities.
191 const Vec3f xAxis = mRenderCorners[2] - mRenderCorners[0];
192 const Vec3f yAxis = mRenderCorners[1] - mRenderCorners[0];
193 mArea = xAxis.length() * yAxis.length();
194 mInvArea = 1.0f / mArea;
195
196 // Compute radiance.
197 mRadiance = computeLightRadiance(mRdlLight, scene_rdl2::rdl2::Light::sColorKey,
198 scene_rdl2::rdl2::Light::sIntensityKey, scene_rdl2::rdl2::Light::sExposureKey, sNormalizedKey,
199 sApplySceneScaleKey, mInvArea);
200 if (isBlack(mRadiance)) {
201 mOn = false;
202 return false;
203 }
204
205 // Compute render space plane of light.
206 // Cached at rayTime = 0.f
207 mRenderPlane = Plane(getPosition(0.f), getDirection(0.f));
208
209 if (!updateImageMap(Distribution2D::PLANAR)) {
210 return false;
211 }
212
213 // Sidedness
214 mSidedness = static_cast<LightSidednessType>(mRdlLight->get<scene_rdl2::rdl2::Int>(sSidednessKey));
215
216 return true;
217}

221bool
222RectLight::canIlluminate(const Vec3f p, const Vec3f *n, float time, float radius,
223 const LightFilterList* lightFilterList) const
224{
225 MNRY_ASSERT(mOn);
226
227 // Cull points which are completely on the backside of the light, taking sidedness into account
228 const float threshold = sEpsilon - radius;
229 if ((mSidedness != LightSidednessType::LIGHT_SIDEDNESS_2_SIDED) && (planeDistance(p, time) < threshold)) {
230 return false;
231 }
232
233 // Cull lights which are completely on the backside of the point.
234 if (n) {
235 Plane pl(p, *n);
236 float d0, d1, d2, d3;
237 if (!isMb()) {
238 d0 = pl.getDistance(mRenderCorners[0]);
239 d1 = pl.getDistance(mRenderCorners[1]);
240 d2 = pl.getDistance(mRenderCorners[2]);
241 d3 = pl.getDistance(mRenderCorners[3]);
242 } else {
243 Vec3f corners[4];
244 computeCorners(corners, time);
245 d0 = pl.getDistance(corners[0]);
246 d1 = pl.getDistance(corners[1]);
247 d2 = pl.getDistance(corners[2]);
248 d3 = pl.getDistance(corners[3]);
249 }
250 bool canIllum = d0 > threshold || d1 > threshold ||
251 d2 > threshold || d3 > threshold;
252 if (!canIllum) return false;
253 }
254
255 if (lightFilterList) {
256 return canIlluminateLightFilterList(lightFilterList,
257 { getPosition(time),
258 xformLocal2RenderScale(math::sqrt(mWidth * mWidth + mHeight * mHeight) / 2, time),
259 p, getXformRender2Local(time, lightFilterList->needsLightXform()),
260 radius, time
261 });
262 }
263
264 return true;
265}

285// Computing a tight bounding box for a rect light
286// -----------------------------------------------
287//
288// Let w and h be the half-width and half-height of the rectangle in local space,
289// so that the 4 corners lie at (x,y,z) = (+/-w, +/-h, 0).
290// If we denote the local2render matrix by (Mij) with i=0,1,2,3 and j=0,1,2,3,
291// then the corners transform to the points (x',y',z') in render space, where
292//
293// x' = +/-w * M00 +/-h * M10 + M30
294// y' = +/-w * M01 +/-h * M11 + M31
295// z' = +/-w * M02 +/-h * M12 + M32
296//
297// Considering just the x' values, the min and max are given respectively by
298//
299// xMin' = M30 - (w * |M00| + h * |M10|)
300// xMax' = M30 + (w * |M00| + h * |M10|)
301//
302// Similar expression arise for the min & max y' and z' values, and the 3
303// expressions can be combined into vector form using Vec3f.
304
305BBox3f
306RectLight::getBounds() const
307{
308 const Vec3f pos = getPosition(0.f);
309 const Vec3f absXRow = scene_rdl2::math::abs(mLocal2Render[0].l.vx);
310 const Vec3f absYRow = scene_rdl2::math::abs(mLocal2Render[0].l.vy);
311 const Vec3f halfDims = max(mHalfWidth * absXRow + mHalfHeight * absYRow,
312 Vec3f(sEpsilon)); // ensure non-degeneracy
313 BBox3f bounds(pos - halfDims, pos + halfDims);
314
315 if (isMb()) {
316 const Vec3f pos = getPosition(1.f);
317 const Vec3f absXRow = scene_rdl2::math::abs(mLocal2Render[1].l.vx);
318 const Vec3f absYRow = scene_rdl2::math::abs(mLocal2Render[1].l.vy);
319 const Vec3f halfDims = max(mHalfWidth * absXRow + mHalfHeight * absYRow,
320 Vec3f(sEpsilon)); // ensure non-degeneracy
321 bounds.extend(BBox3f(pos - halfDims, pos + halfDims));
322 }
323
324 return bounds;
325}

327bool
328RectLight::intersect(const Vec3f &p, const Vec3f *n, const Vec3f &wi, float time,
329 float maxDistance, LightIntersection &isect) const
330{
331 // Get local direction and position
332 Vec3f localWi = xformVectorRender2LocalRot(wi, time);
333 Vec3f localP = xformVectorRender2Local(p - getPosition(time), time);
334
335 // Trivial rejection of pos/dir/sidedness cases
336 if (rejectPosDirSidedness(localP.z, localWi.z)) return false;
337
338 // Reject if intersection with plane is beyond maxDistance
339 float localDistance = -localP.z / localWi.z;
340 float renderDistance = xformLocal2RenderScale(localDistance, time);
341 if (renderDistance > maxDistance) {
342 return false;
343 }
344
345 // Reject if intersection is outside rectangle
346 Vec3f localHit = localP + localWi * localDistance;
347 if (scene_rdl2::math::abs(localHit.x) >= mHalfWidth || scene_rdl2::math::abs(localHit.y) >= mHalfHeight) {
348 return false;
349 }
350
351 if (mSpread < 1.f) {
352 // Reject if intersection point is outside spread
353 Vec3f absLocalWi = scene_rdl2::math::abs(localWi);
354 if (max(absLocalWi.x, absLocalWi.y) > absLocalWi.z * mTanSpreadTheta) {
355 return false;
356 }
357
358 // compute pdf
359 // get half side length of square spread region
360 float spreadLength = localP.z * mTanSpreadTheta;
361 Vec2f center;
362 float width, height;
363 getOverlapBounds(Vec2f(localP.x, localP.y), spreadLength, center, width, height);
364 // store pdf if not rejected
365 isect.pdf = 1.f / (xformLocal2RenderScale(width, time) * xformLocal2RenderScale(height, time));
366 }
367
368 // Fill in isect members.
369 // We will only reach this far if p is on the illuminated side of the light's plane (either side, for 2-sided).
370 // The light's mDirection is its local z-axis, which is only guaranteed to concide with the normal for lights
371 // with regular sidedness. In the general case, the correct normal is obtained by ensuring the ray is travelling
372 // against the normal, i.e. the ray's direction has negative local z-coordinate.
373 Vec3f normal = getDirection(time);
374 isect.N = (localWi.z < 0.0f) ? normal : -normal;
375 isect.uv = (mDistribution ? asVec2(localHit) * mUvScale + mUvOffset
376 : math::zero);
377 isect.distance = renderDistance;
378
379 return true;
380}

382bool
383RectLight::sample(const Vec3f &p, const Vec3f *n, float time, const Vec3f& r,
384 Vec3f &wi, LightIntersection &isect, float rayDirFootprint) const
385{
386 MNRY_ASSERT(mOn);
387
388 // We don't cast light on points behind the light. Also we can't always
389 // rely on canIlluminate() to test for that case, as it's an optional
390 // culling function. canIlluminate() can also be passed a non-zero radius,
391 // defining a p-region partially in front and partially behind the light.
392 // So we have to test for it per sample, here too.
393 float distance = planeDistance(p, time);
394 if ((mSidedness != LightSidednessType::LIGHT_SIDEDNESS_2_SIDED) && (distance < sEpsilon)) {
395 return false;
396 }
397
398 Vec3f localHit;
399
400 // There are 3 possible sampling modes:
401 // 1) If there is an image distribution, sample that.
402 // 2) If there is no image distribution, and the spread parameter is less than 1 and
403 // we are in a good region to sample the spread, then sample that region.
404 // 3) Otherwise, sample the shape of the light.
405 if (mDistribution) {
406 // sample the image distribution if any
407 mDistribution->sample(r[0], r[1], 0, &isect.uv, nullptr, mTextureFilter);
408 localHit = Vec3f((isect.uv[0] - mUvOffset.x) / mUvScale.x,
409 (isect.uv[1] - mUvOffset.y) / mUvScale.y,
410 0.0f);
411
412 if (mSpread < 1.f) {
413 // Reject samples that are outside of the spread region.
414 Vec3f localP = xformPointRender2Local(p, time);
415 Vec3f localWi = localHit - localP;
416 Vec3f absLocalWi = scene_rdl2::math::abs(localWi);
417 if (max(absLocalWi.x, absLocalWi.y) > absLocalWi.z * mTanSpreadTheta) {
418 return false;
419 }
420 }
421
422 } else if (mSpread < 1.f) {
423 // Uniformly sample the rectangular region of overlap between the spread region and the RectLight.
424
425 // get p in local space
426 Vec3f localP = xformPointRender2Local(p, time);
427
428 // get length of square region of influence
429 float spreadLength = distance * mTanSpreadTheta;
430 float localSpreadLength = xformRender2LocalScale(spreadLength, time);
431
432 Vec2f center;
433 float width, height;
434 if (!getOverlapBounds(Vec2f(localP.x, localP.y), localSpreadLength, center, width, height)) {
435 return false;
436 }
437
438 // sample rectangular region of influence
439 localHit = Vec3f((0.5f - r[0]) * width + center.x,
440 (0.5f - r[1]) * height + center.y,
441 0.0f);
442
443 isect.uv = zero;
444 // store pdf
445 isect.pdf = 1.f / (xformLocal2RenderScale(width, time) * xformLocal2RenderScale(height, time));
446
447 } else {
448 // Otherwise uniformly distribute samples on rectangle area
449 isect.uv = zero;
450 localHit = Vec3f((0.5f - r[0]) * mWidth,
451 (0.5f - r[1]) * mHeight,
452 0.0f);
453 }
454
455 Vec3f renderHit = xformPointLocal2Render(localHit, time);
456
457 // Compute wi and d
458 wi = renderHit - p;
459 isect.distance = length(wi);
460 if (isect.distance < sEpsilon) {
461 return false;
462 }
463 wi *= rcp(isect.distance);
464 if (n && dot(*n, wi) < sEpsilon) {
465 return false;
466 }
467
468 // We will only reach this far if p is on the illuminated side of the light's plane (either side, for 2-sided).
469 // The light's mDirection is its local z-axis, which is only guaranteed to concide with the normal for lights
470 // with regular sidedness. The correct normal is obtained by ensuring the ray is travelling against the normal,
471 // i.e. the ray's direction has negative dot product with it.
472 Vec3f normal = getDirection(time);
473 isect.N = (dot(normal, wi) < 0.0f) ? normal : -normal;
474
475 return true;
476}

478Color
479RectLight::eval(mcrt_common::ThreadLocalState* tls, const Vec3f &wi, const Vec3f &p, const LightFilterRandomValues& filterR, float time,
480 const LightIntersection &isect, bool fromCamera, const LightFilterList *lightFilterList, float rayDirFootprint,
481 float *pdf) const
482{
483 MNRY_ASSERT(mOn);
484
485 // Point sample the texture
486 Color radiance = mRadiance;
487
488 if (mSpread < 1.f) {
489 // modify radiance as if it were physically blocked by an eggcrate light filter.
490 Vec3f localWi = xformVectorRender2LocalRot(wi, time);
491 radiance *= math::max(1.f - math::abs(localWi.x / (localWi.z * mTanSpreadTheta)), 0.f) *
492 math::max(1.f - math::abs(localWi.y / (localWi.z * mTanSpreadTheta)), 0.f);
493 }
494
495 if (mDistribution) {
496 radiance *= mDistribution->eval(isect.uv[0], isect.uv[1], 0, mTextureFilter);
497 }
498
499 if (lightFilterList) {
500 evalLightFilterList(lightFilterList,
501 { tls, &isect, getPosition(time),
502 getDirection(time), p,
503 filterR, time,
504 getXformRender2Local(time, lightFilterList->needsLightXform()),
505 wi
506 },
507 radiance);
508 }
509
510 if (pdf) {
511 if (mDistribution) {
512 // we sampled the image distribution
513 *pdf = mInvArea * mDistribution->pdf(isect.uv[0], isect.uv[1], 0, mTextureFilter);
514 } else if (mSpread < 1.f) {
515 // we sampled the spread region, grab the stored pdf.
516 *pdf = isect.pdf;
517 } else {
518 // we uniformly sampled the rectangle
519 *pdf = mInvArea;
520 }
521 *pdf *= areaToSolidAngleScale(wi, isect.N, isect.distance);
522 }
523
524 return radiance;
525}

527Vec3f
528RectLight::getEquiAngularPivot(const Vec3f& r, float time) const
529{
530 Vec2f uv;
531 Vec3f hit;
532 // Sample the image distribution if any
533 if (mDistribution) {
534 mDistribution->sample(r[0], r[1], 0, &uv, nullptr, mTextureFilter);
535 hit = Vec3f((uv[0] - mUvOffset.x) / mUvScale.x,
536 (uv[1] - mUvOffset.y) / mUvScale.y,
537 0.0f);
538 } else {
539 hit = Vec3f((0.5f - r[0]) * mWidth,
540 (0.5f - r[1]) * mHeight,
541 0.0f);
542 }
543 return xformPointLocal2Render(hit, time);
544}
