84 // Widen roughness at grazing angles, to reduce maximum weight
85 // Also called the "Walter trick". See sect. 5.3 "Modified Sampling Distrubtion"
86 static float
87 widenRoughness(float roughness, float cosNO)
88 {
89 #if PBR_CT_SAMPLE_WIDENED_ROUGHNESS
90 return (1.2f - 0.2f * scene_rdl2::math::sqrt(cosNO)) * roughness;
91 #else
92 return roughness;
93 #endif
94 }
