86math::Mat4f
87CookieLightFilter::computePerspectiveProjectionMatrix(float t) const
88{
89 // Get focal length (mm)
90 float focal = mRdlLightFilter->get(sProjectorFocalKey, t);
91 focal = max(focal, 0.1f);
92
93 // Get film-back width (mm)
94 float filmWidth = mRdlLightFilter->get(sProjectorFilmWidthApertureKey);
95 filmWidth = max(filmWidth, 0.01f);
96 float halfFilmWidth = filmWidth * 0.5f;
97
98 // Get near plane
99 float near = 1.f;
100
101 // Get far plane
102 float far = 10000.f;
103
104 // Note that the cookie filter doesn't clip to near/far planes but reasonable
105 // defaults need to be specified for the projection matrix to be constructed.
106 // Other values would also work and produce an equivalent projection.
107
108 // Get pixel aspect ratio
109 float par = mRdlLightFilter->get(sProjectorPixelAspectRatioKey);
110 if (isEqual(par, 0.0f)) {
111 par = 1.0f;
112 }
113
114 // Compute window corners on the near plane
115 float factor = near / focal;
116 float windowNear[4];
117 windowNear[0] = (-2.f * halfFilmWidth) * factor;
118 windowNear[1] = (-2.f * halfFilmWidth) * factor / par;
119 windowNear[2] = (2.f * halfFilmWidth) * factor;
120 windowNear[3] = (2.f * halfFilmWidth) * factor / par;
121
122 // See Appendix C.5 of the SGI Graphics Library Programming Guide
123 // for format of projection matrix.
124 Mat4f c2s;
125 c2s[0][0] = (2.0f * near) / (windowNear[2] - windowNear[0]);
126 c2s[0][1] = 0.0f;
127 c2s[0][2] = 0.0f;
128 c2s[0][3] = 0.0f;
129
130 c2s[1][0] = 0.0f;
131 c2s[1][1] = (2.0f * near) / (windowNear[3] - windowNear[1]);
132 c2s[1][2] = 0.0f;
133 c2s[1][3] = 0.0f;
134
135 c2s[2][0] = (windowNear[2] + windowNear[0]) / (windowNear[2] - windowNear[0]);
136 c2s[2][1] = (windowNear[3] + windowNear[1]) / (windowNear[3] - windowNear[1]);
137 c2s[2][2] = -(far + near) / (far - near);
138 c2s[2][3] = -1.0f;
139
140 c2s[3][0] = 0.0f;
141 c2s[3][1] = 0.0f;
142 c2s[3][2] = -(2.0f * far * near) / (far - near);
143 c2s[3][3] = 0.0f;
144
145 return c2s;
146}

148math::Mat4f
149CookieLightFilter::computeOrthoProjectionMatrix(float t) const
150{
151 // Get film-back width (mm)
152 float filmWidth = mRdlLightFilter->get(sProjectorFilmWidthApertureKey);
153 filmWidth = max(filmWidth, 0.01f);
154 float halfFilmWidth = filmWidth * 0.5f;
155
156 // Get near plane
157 float near = 1.f;
158
159 // Get far plane
160 float far = 10000.f;
161
162 // Get pixel aspect ratio
163 float par = mRdlLightFilter->get(sProjectorPixelAspectRatioKey);
164 if (isEqual(par, 0.0f)) {
165 par = 1.0f;
166 }
167
168 // Compute window corners on the near plane
169 float windowNear[4];
170 windowNear[0] = (-2.f * halfFilmWidth);
171 windowNear[1] = (-2.f * halfFilmWidth) / par;
172 windowNear[2] = (2.f * halfFilmWidth);
173 windowNear[3] = (2.f * halfFilmWidth) / par;
174
175 // See Appendix C.5 of the SGI Graphics Library Programming Guide
176 // for format of projection matrix.
177 Mat4f c2s;
178 c2s[0][0] = 2.0f / (windowNear[2] - windowNear[0]);
179 c2s[0][1] = 0.0f;
180 c2s[0][2] = 0.0f;
181 c2s[0][3] = 0.0f;
182
183 c2s[1][0] = 0.0f;
184 c2s[1][1] = 2.0f / (windowNear[3] - windowNear[1]);
185 c2s[1][2] = 0.0f;
186 c2s[1][3] = 0.0f;
187
188 c2s[2][0] = 0.0f;
189 c2s[2][1] = 0.0f;
190 c2s[2][2] = -2.0f / (far - near);
191 c2s[2][3] = 0.0f;
192
193 c2s[3][0] = -(windowNear[2] + windowNear[0])/(windowNear[2] - windowNear[0]);
194 c2s[3][1] = -(windowNear[3] + windowNear[1])/(windowNear[3] - windowNear[1]);
195 c2s[3][2] = -(far + near) / (far - near);
196 c2s[3][3] = 1.0f;
197
198 return c2s;
199}

201bool
202isValidXform(const Mat4d& xf)
203{
204 // Check for zero scale
205 return !(isZero(xf.vx[0]) && isZero(xf.vx[1]) && isZero(xf.vx[2]) &&
206 isZero(xf.vy[0]) && isZero(xf.vy[1]) && isZero(xf.vy[2]) &&
207 isZero(xf.vz[0]) && isZero(xf.vz[1]) && isZero(xf.vz[2]));

210void
211CookieLightFilter::update(const LightFilterMap& /*lightFilters*/,
212 const Mat4d& world2Render)
213{
214 if (!mRdlLightFilter) {
215 return;
216 }
217
218 // Update projector render2Screen. This transform is the guts of the filter.
219 rdl2::SceneObject* projectorSo = mRdlLightFilter->get(sProjectorKey);
220 if (projectorSo) {
221 rdl2::Camera *projector = projectorSo->asA<rdl2::Camera>();
222 if (projector != nullptr && projector->doesSupportProjectionMatrix()) {
223
224 Mat4d render2World = world2Render.inverse();
225 Mat4d projector2World0 = projector->get(rdl2::Node::sNodeXformKey, 0.0f);
226 Mat4d projector2World1 = projector->get(rdl2::Node::sNodeXformKey, 1.0f);
227 Mat4d world2Projector0 = projector2World0.inverse();
228 Mat4d world2Projector1 = projector2World1.inverse();
229
230 Mat4d projector2Screen = Mat4d(projector->computeProjectionMatrix(0.f, // t
231 {-2.f, -2.f, 2.f, 2.f}, // window
232 0.f)); // interocularOffset
233
234 mProjectorR2S[0] = Mat4f(render2World * world2Projector0 * projector2Screen);
235 mProjectorR2S[1] = Mat4f(render2World * world2Projector1 * projector2Screen);
236
237 Mat4d projector2Render0 = projector2World0 * world2Render;
238 Mat4d projector2Render1 = projector2World1 * world2Render;
239 mProjectorPos[0] = projector2Render0.row3().toVec3();
240 mProjectorPos[1] = projector2Render1.row3().toVec3();
241 mProjectorDir[0] = -projector2Render0.row2().toVec3();
242 mProjectorDir[1] = -projector2Render1.row2().toVec3();
243
244 } else {
245 mRdlLightFilter->error("Projector is invalid or does not have a projection matrix");
246 mProjectorR2S[0] = Mat4f(math::one);
247 mProjectorR2S[1] = Mat4f(math::one);
248 mProjectorPos[0] = {0.f, 0.f, 0.f};
249 mProjectorPos[1] = {0.f, 0.f, 0.f};
250 mProjectorDir[0] = {0.f, 0.f, 1.f};
251 mProjectorDir[1] = {0.f, 0.f, 1.f};
252 }
253 } else {
254 // use the projector attributes
255 Mat4d render2World = world2Render.inverse();
256 Mat4d projector2World0 = mRdlLightFilter->get(sProjectorXformKey, 0.f);
257 Mat4d projector2World1 = mRdlLightFilter->get(sProjectorXformKey, 1.f);
258
259 if (!isValidXform(projector2World0) || !isValidXform(projector2World1)) {
260 mRdlLightFilter->error("node_xform is invalid");
261 }
262
263 Mat4d world2Projector0 = projector2World0.inverse();
264 Mat4d world2Projector1 = projector2World1.inverse();
265
266 Mat4d projector2Screen;
267
268 switch (mRdlLightFilter->get(sProjectorTypeKey)) {
269 case PERSPECTIVE:
270 projector2Screen = Mat4d(computePerspectiveProjectionMatrix(0.f));
271 break;
272 case ORTHOGRAPHIC:
273 projector2Screen = Mat4d(computeOrthoProjectionMatrix(0.f));
274 break;
275 default:
276 MNRY_ASSERT(false);
277 }
278
279 mProjectorR2S[0] = Mat4f(render2World * world2Projector0 * projector2Screen);
280 mProjectorR2S[1] = Mat4f(render2World * world2Projector1 * projector2Screen);
281
282 Mat4d projector2Render0 = projector2World0 * world2Render;
283 Mat4d projector2Render1 = projector2World1 * world2Render;
284 mProjectorPos[0] = projector2Render0.row3().toVec3();
285 mProjectorPos[1] = projector2Render1.row3().toVec3();
286 mProjectorDir[0] = -projector2Render0.row2().toVec3();
287 mProjectorDir[1] = -projector2Render1.row2().toVec3();
288 }
289
290 // update image map
291 rdl2::SceneObject* mapSo = mRdlLightFilter->get(sMapShaderKey);
292 if (mapSo) {
293 // ispc ptr is int64
294 mMapShader = (const int64_t *) mapSo->asA<rdl2::Map>();
295 if (mMapShader == nullptr) {
296 mRdlLightFilter->error("Map shader is invalid");
297 }
298 } else {
299 mRdlLightFilter->error("Map shader does not exist");
300 mMapShader = nullptr;
301 }
302
303 mBlurNearDistance = mRdlLightFilter->get<rdl2::Float>(sBlurNearDistanceKey);
304 mBlurMidpoint = mRdlLightFilter->get<rdl2::Float>(sBlurMidpointKey);
305 mBlurFarDistance = mRdlLightFilter->get<rdl2::Float>(sBlurFarDistanceKey);
306 mBlurNearValue = mRdlLightFilter->get<rdl2::Float>(sBlurNearValueKey);
307 mBlurMidValue = mRdlLightFilter->get<rdl2::Float>(sBlurMidValueKey);
308 mBlurFarValue = mRdlLightFilter->get<rdl2::Float>(sBlurFarValueKey);
309 mBlurType = mRdlLightFilter->get<rdl2::Int>(sBlurType);
310
311 mOutsideProjection = mRdlLightFilter->get<rdl2::Int>(sOutsideProjection);
312
313 mDensity = mRdlLightFilter->get<rdl2::Float>(sDensityKey);
314 mInvert = mRdlLightFilter->get<rdl2::Bool>(sInvertKey);
315}

317bool
318CookieLightFilter::canIlluminate(const CanIlluminateData& data) const
319{
320 Vec3f projectorPos = lerp(mProjectorPos[0], mProjectorPos[1], data.time);
321 Vec3f projectorDir = lerp(mProjectorDir[0], mProjectorDir[1], data.time);
322 Vec3f shadingVec = data.shadingPointPosition - projectorPos;
323
324 if (dot(shadingVec, projectorDir) < 0.f) {
325 // The shading point is behind the projector.
326 switch (mOutsideProjection) {
327 case BLACK:
328 return false;
329 case WHITE:
330 return true;
331 default:
332 return false;
333 }
334 }
335
336 return true;
337}

344Color
345CookieLightFilter::eval(const EvalData& data) const
346{
347 if (!mMapShader) {
348 return Color(1.f);
349 }
350
351 Vec3f projectorPos = lerp(mProjectorPos[0], mProjectorPos[1], data.time);
352 Vec3f projectorDir = lerp(mProjectorDir[0], mProjectorDir[1], data.time);
353 Vec3f shadingVec = data.shadingPointPosition - projectorPos;
354 float distanceToProjector = length(shadingVec);
355
356 if (dot(shadingVec, projectorDir) < 0.f) {
357 // The shading point is behind the projector, we don't need to evaluate the
358 // map shader.
359 switch (mOutsideProjection) {
360 case BLACK:
361 return Color(0.f);
362 case WHITE:
363 return Color(1.f);
364 default:
365 return Color(0.f);
366 }
367 }
368
369 // We need the screen space position of the render space point.
370 Vec3f screenP0 = transformH(mProjectorR2S[0], data.shadingPointPosition);
371 Vec3f screenP1 = transformH(mProjectorR2S[1], data.shadingPointPosition);
372 Vec3f screenP = lerp(screenP0, screenP1, data.time);
373
374 // Right now we are in screen space where (0, 0) is the center of the projection.
375 // However, for maps, the center is (0.5, 0.5) and we want the map centered
376 // in the center of the projection. So we need to add an 0.5 offset.
377 screenP.x += 0.5f;
378 screenP.y += 0.5f;
379
380 // Based on the distance to the projector, we interpolate the filter radius
381 // depending on whether it is in the near/mid/far interval.
382 float filterRadius = 0.f; // aka blur value
383 if (distanceToProjector <= mBlurNearDistance) {
384 filterRadius = mBlurNearValue;
385 } else if (distanceToProjector <= mBlurMidpoint) {
386 float dt = mBlurMidpoint - mBlurNearDistance;
387 if (dt > 0.f) {
388 float t = (distanceToProjector - mBlurNearDistance) / dt;
389 filterRadius = lerp(mBlurNearValue, mBlurMidValue, t);
390 } else {
391 filterRadius = mBlurMidValue;
392 }
393 } else if (distanceToProjector <= mBlurFarDistance) {
394 float dt = mBlurFarDistance - mBlurMidpoint;
395 if (dt > 0.f) {
396 float t = (distanceToProjector - mBlurMidpoint) / dt;
397 filterRadius = lerp(mBlurMidValue, mBlurFarValue, t);
398 } else {
399 filterRadius = mBlurFarValue;
400 }
401 } else {
402 filterRadius = mBlurFarValue;
403 }
404
405 // Move the screen space position based on random sampling the filter
406
407 // Box filter, kept here for debugging.
408 // Vec2f offset((r.x - 0.5f) * filterRadius, (r.y - 0.5f) * filterRadius;
409
410 Vec2f offset;
411
412 switch (mBlurType) {
413 case GAUSSIAN:
414 // Quadratic bspline "gaussian" filter, which is the same filter the regular camera uses.
415 offset.x = quadraticBSplineWarp(data.randVar.r2.x) * filterRadius;
416 offset.y = quadraticBSplineWarp(data.randVar.r2.y) * filterRadius;
417 break;
418 case CIRCULAR:
419 offset.x = data.randVar.r2.x;
420 offset.y = data.randVar.r2.y;
421 toUnitDisk(offset.x, offset.y);
422 offset.x *= filterRadius;
423 offset.y *= filterRadius;
424 break;
425 default:
426 MNRY_ASSERT(false);
427 }
428
429 Vec2f st(screenP.x + offset.x, screenP.y + offset.y);
430
431 if (st.x < 0.f || st.x > 1.f ||
432 st.y < 0.f || st.y > 1.f) {
433 switch (mOutsideProjection) {
434 case BLACK:
435 return Color(0.f);
436 case WHITE:
437 return Color(1.f);
438 }
439 }
440
441 // Lookup the map value using the displaced screen space position
442 Color mapValue = sampleMapShader(data.tls, st);
443
444 // Apply density scaling to allow partial light filtering
445 mapValue = Color(1.f - mDensity) + mapValue * mDensity;
446
447 // Invert the filter?
448 if (mInvert) {
449 mapValue = 1.f - mapValue;
450 }
451
452 return mapValue;
453}

455Color
456CookieLightFilter::sampleMapShader(mcrt_common::ThreadLocalState* tls,
457 const Vec2f& st) const
458{
459 Vec3f p(0.f);
460 Vec3f n(0.f);
461
462 shading::Intersection isect;
463 isect.initMapEvaluation(&tls->mArena,
464 nullptr, // attr table
465 nullptr, // geom
466 nullptr, // layer
467 0, // assignment id
468 p,
469 n,
470 0, 0, // All texture derivatives are set to 0.
471 st,
472 0, 0,
473 0, 0);
474
475 Color result;
476 const rdl2::Map * map = reinterpret_cast<const rdl2::Map *>(mMapShader);
477 map->sample(tls->mShadingTls.get(), shading::State(&isect), &result);
478 return result;
479}
