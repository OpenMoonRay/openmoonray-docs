109 CPPUNIT_ASSERT_EQUAL(a0.at(i).mVal, 99);
110 }
111
112 moonray::util::AlignedElementArray<T> a1(std::move(a0));
113 for (int i = 0; i < length; ++i) {
114 CPPUNIT_ASSERT_EQUAL(a1[i].mVal, 99);
115 CPPUNIT_ASSERT_EQUAL(a1.at(i).mVal, 99);
116 }
117
118 for (const auto& x: a1) {
119 CPPUNIT_ASSERT_EQUAL(x.mVal, 99);
120 }
121 for (auto it = a1.begin(); it != a1.end(); ++it) {
122 CPPUNIT_ASSERT_EQUAL(it->mVal, 99);
123 }
124 for (auto it = a1.cbegin(); it != a1.cend(); ++it) {
125 CPPUNIT_ASSERT_EQUAL(it->mVal, 99);
126 }
127 }
128 CPPUNIT_ASSERT_EQUAL(T::mCount, 0);

270void TestAlignedElementArray::testAlignment()
271{
272 constexpr size_t desiredAlignment = 32;
273
274 auto isAligned = [desiredAlignment](auto& value) {
275 return reinterpret_cast<std::uintptr_t>(std::addressof(value)) % desiredAlignment == 0;
276 };
277
278 constexpr int numValues = 101;
279 std::vector<int> initValues(numValues);
280 std::iota(initValues.begin(), initValues.end(), 0);
281
282 CPPUNIT_ASSERT_EQUAL(NonDefaultConstructable::mCount, 0);
283 {
284 moonray::util::AlignedElementArray<NonDefaultConstructable,desiredAlignment> array(initValues.cbegin(),
285 initValues.cend());
286
287 for (int i = 0; i < numValues; ++i) {
288 CPPUNIT_ASSERT_EQUAL(i, array[i].mVal);
289 CPPUNIT_ASSERT(isAligned(array[i].mVal));
290 }
291 }
292 CPPUNIT_ASSERT_EQUAL(NonDefaultConstructable::mCount, 0);
293}
