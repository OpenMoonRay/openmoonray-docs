19/**
20 * The ObjectFactory manages function pointers to the critical functions for
21 * declaring, creating, and destroying objects of a particular type.
22 *
23 * You create an ObjectFactory through one of the static creation functions,
24 * depending on how you want the objects created. At the moment we support
25 * built in types, DSO types, and proxy objects (which use DSOs for their
26 * attribute declarations, but are constructed from built in proxy types).
27 *
28 * The ObjectFactory also takes ownership of a Dso object, if loading symbols
29 * from any DSO is required.
30 *
31 * Thread Safety:
32 * - Creating ObjectFactories for the same SceneClass from different threads
33 * simultaneously is not thread safe, because we don't enforce thread
34 * safety of any DSO operations.
35 * - Creating and manipulating ObjectFactories for different SceneClasses
36 * from multiple threads should be safe, as there is no global shared data
37 * to synchronize.
38 */
39class ObjectFactory
40{
41public:
42 /**
43 * Invoke the declare function, no matter where it came from. This
44 * effectively forwards the call to the underlying function pointer.
45 *
46 * @param sceneClass The SceneClass that will be filled in with attribute
47 * declarations.
48 * @return The SceneObjectInterface this SceneClass intends to implement.
49 */
50 finline SceneObjectInterface declare(SceneClass& sceneClass);
51
52 /**
53 * Invoke the create function, no matter where it came from. This
54 * effectively forwards the call to the underlying function pointer.
55 *
56 * @param sceneClass The SceneClass that this new SceneObject is to
57 * belong to.
58 * @param name The name of the new SceneObject.
59 * @return A pointer to the newly allocated SceneObject. Ownership of this
60 * pointer is transferred to the caller.
61 */
62 finline SceneObject* create(const SceneClass& sceneClass, const std::string& name);
63
64 /**
65 * Invoke the destroy function, no matter where it came from. This
66 * effectively forwards the call to the underlying function pointer.
67 *
68 * @param sceneObject The SceneObject that we wish to destroy.
69 */
70 finline void destroy(SceneObject* sceneObject);
71
72 /**
73 * Returns the path to where this SceneClass came from. If the factory is a
74 * DsoFactory or a ProxyFactory, it returns the file system path of the DSO
75 * (or proxy DSO) that was loaded. If the factory is a built-in factory, it
76 * returns an empty string.
77 *
78 * @return File system path to the source of this SceneClass or an empty
79 * string if it's built-in.
80 */
81 std::string getSourcePath() const;
82
83 /**
84 * Create an ObjectFactory for the built in type specified by the template
85 * parameter.
86 *
87 * @return An ObjectFactory that can declare, create, and destroy these
88 * DSO objects.
89 */
90 template <typename T>
91 static std::unique_ptr<ObjectFactory> createBuiltInFactory();
92
93 /**
94 * Create an ObjectFactory that sources its function pointers from a DSO.
95 * The DSO should be named "className.so" and be in the dsoPath.
96 *
97 * @param className The SceneClass name. The DSO is expected to be
98 * named "className.so".
99 * @param dsoPath A colon seperated path to search for DSOs.
100 * @return An ObjectFactory that can declare, create, and destroy these
101 * DSO objects.
102 */
103 static std::unique_ptr<ObjectFactory> createDsoFactory(const std::string& className, const std::string& dsoPath);
104
105 /**
106 * Create an ObjectFactory that sources its declare function pointer from
107 * a DSO, but creates and destroys objects through built in proxy objects.
108 *
109 * @param className The SceneClass name. The DSO is expected to be
110 * named "className.so".
111 * @param dsoPath A colon seperated path to search for DSOs.
112 * @return An ObjectFactory that can declare, create, and destroy these
113 * DSO objects.
114 */
115 static std::unique_ptr<ObjectFactory> createProxyFactory(const std::string& className, const std::string& dsoPath);
116
117private:
118 ObjectFactory(ClassDeclareFunc declareFunc, ObjectCreateFunc createFunc,
119 ObjectDestroyFunc destroyFunc, std::unique_ptr<Dso> dso = nullptr);
120
121 std::unique_ptr<Dso> mDso;
122 ClassDeclareFunc mDeclareFunc;
123 ObjectCreateFunc mCreateFunc;
124 ObjectDestroyFunc mDestroyFunc;
125};
