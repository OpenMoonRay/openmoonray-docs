23// PolyFaceTopology describes a triangle or quad face in polygon mesh
24// that is going to be further tessellated (for displacement purpose)
25// with a set of vertex/edge id.
26// It is used for input quad -> tessellated face lookup with the help of
27// PolyTessellatedVertexLookup.
28//
29// The representation for quad face looks like:
30//
31// corner[3] corner[2]
32// *-------------*
33// | edge[2] |
34// | |
35// edge[3] | | edge[1]
36// | |
37// | edge[0] |
38// *-------------*
39// corner[0] corner[1]
40//
41// The presentation for triangle face looks like:
42//
43// corner[2]
44// /*
45// / |
46// edge[2] / |
47// / | edge[1]
48// / |
49// / edge[0] |
50// *------------*
51// corner[0] corner[1]
52//
53struct PolyFaceTopology
54{
55 std::array<int, sQuadVertexCount> mCornerVertexId;
56 std::array<int, sQuadVertexCount> mEdgeId;
57};

59// PolyTessellationFactor describes how many segments each edge on input face
60// will be tessellated (the value can be different from face to face if we
61// do adaptive tessellation) for polygon mesh
62struct PolyTessellationFactor

64 // edge id is only used during computation stage,
65 // the computation result is stored in edge factor

67 std::array<int, sQuadVertexCount> mEdgeId;

69 };

73// SubdQuadTopology describes a control quad in subdivision mesh
74// with a set of vertex/edge id.
75// It is used for control quad -> tessellated face lookup with the help of
76// SubdTessellatedVertexLookup.
77// If a quad is regular control quad (major case for well modeled Catmull-Clark
78// control mesh), its representation looks like:
79//
80// corner[3] mid[2] corner[2]
81// *--------------x--------------*
82// | edge1[2] edge0[2] |
83// | |
84// | |
85// | edge0[3] | edge1[1]
86// | |
87// | |
88// x mid[3] x mid[1]
89// | |
90// | |
91// | edge1[3] | edge0[1]
92// | |
93// | |
94// | edge0[0] mid[0] |
95// *--------------x--------------*
96// corner[0] edge1[0] corner[1]
97//
98// If a quad comes from quadrangulated n-gons, its representation looks like:
99//
100// corner[3] corner[2]
101// *-------------*
102// | edge0[2] |
103// | |
104// edge0[3] | | edge0[1]
105// | |
106// | edge0[0] |
107// *-------------*
108// corner[0] corner[1]
109struct SubdQuadTopology
110{
111 bool nonQuadParent() const
112 {
113 return mNonQuadParent;
114 }
115 std::array<int, sQuadVertexCount> mCornerVertexId;
116 std::array<int, sQuadVertexCount> mEdgeId0;
117 // only regular control quad (instead of quads from quadrangulated n-gons)
118 // has valid values for mMidEdgeVertexId and mEdgeId1
119 std::array<int, sQuadVertexCount> mMidEdgeVertexId;
120 std::array<int, sQuadVertexCount> mEdgeId1;
121 int mControlFaceId;
122 bool mHasAssignment;
123 bool mNonQuadParent;
124};

126// SubdTessellationFactor describes how many segments each edge on control quad
127// will be tessellated (the value can be different from face to face if we
128// do adaptive tessellation) for subdivision mesh
129// The number of SubdTessellationFactor should match the number of
130// SubdQuadTopology. The reason we separate these two helpers is
131// because SubdTessellationFactor can be reused across multiple channels of
132// face varying attributes, position and texture buffer, while SubdQuadTopology
133// can be different for each of them since they use different index buffer
134// (and thus different overlap edges/vertices)
135struct SubdTessellationFactor
136{
137 // edge id is only used during computation stage,
138 // the computation result is stored in edge factor
139 union {
140 std::array<int, sQuadVertexCount> mEdgeId0;
141 std::array<int, sQuadVertexCount> mEdge0Factor;
142 };
143 // only regular control quad (instead of quads from quadrangulated n-gons)
144 // has valid values for the following members
145 union {
146 std::array<int, sQuadVertexCount> mEdgeId1;
147 std::array<int, sQuadVertexCount> mEdge1Factor;
148 };
149};

152// see the following link for the heuristic detail explanation
153// https://developer.nvidia.com/content/dynamic-hardware-tessellation-basics
154// this is the same method OpenSubdiv GPU evaluator uses
155// in util function OsdGetTessLevelsAdaptiveLimitPoints()
156finline int
157computeEdgeVertexCount(const Vec3f& v0, const Vec3f& v1,
158 float edgesPerScreenHeight, const scene_rdl2::math::Mat4f& c2s)
159{
160 // calculate the bounding sphere of this edge
161 Vec3f pCenter = 0.5f * (v0 + v1);
162 float diameter = distance(v0, v1);
163 // projection[1][1] = 1 / tan(fov / 2);
164 // calculate edge tessellation factor based on frustum information
165 int edgeVertexCount = 0.5f * edgesPerScreenHeight *
166 scene_rdl2::math::abs(diameter * c2s[1][1] / pCenter.z) - 1;
167 return edgeVertexCount;
168}

197// TopologyIdLookup provides edge id lookup for tessellation utilities usage
198// by analyzing the input control face topology
199class TopologyIdLookup
200{
201public:
202 int getEdgeId(int v0, int v1) const
203 {
204 const auto& result = mEdgeIds.find(getEdgeKey(v0, v1));
205 return result == mEdgeIds.end() ? -1 : result->second;
206 }
207
208 int getCoarseVertexCount() const
209 {
210 return mCoarseVertexCount;
211 }
212
213 int getCoarseEdgeCount() const
214 {
215 return mCoarseEdgeCount;
216 }
217
218protected:
219 // TopologyIdLookup is not intended for direct use but holding shared data
220 // across subclasses
221 TopologyIdLookup(const int coarseVertexCount):
222 mCoarseVertexCount(coarseVertexCount), mCoarseEdgeCount(0)
223 {}
224
225 virtual ~TopologyIdLookup() = default;
226
227 int64_t getEdgeKey(int64_t v0, int64_t v1) const
228 {
229 return v0 < v1 ? (v1 << 32) + v0 : (v0 << 32) + v1;
230 }
231
232protected:
233 std::unordered_map<int64_t, int> mEdgeIds;
234 int mCoarseVertexCount;
235 int mCoarseEdgeCount;
236};

356// PolyTessellatedVertexLookup implements polygon mesh specific vertex id
357// lookup through PolyFaceTopology
358class PolyTessellatedVertexLookup : public TessellatedVertexLookup
359{
360public:
361 PolyTessellatedVertexLookup(

363 const PolyTopologyIdLookup& topologyIdLookup,
364 const std::vector<PolyTessellationFactor>& tessellationFactors,
365 size_t faceVertexCount);
366
367 // the convention for edge tessellation vertex id lookup is:
368 // the edge vertex id increase from the cornervertex with smaller vertex id
369 // for example:
370 // v0 v1 v2
371 // x0----*-----*-----*-----x1
372 //
373 // x0 < x1
374 // if reverseEdge:
375 // getEdgeVertexId(edgeId, 0, reverseEdge) = v2
376 // getEdgeVertexId(edgeId, 1, reverseEdge) = v1
377 // getEdgeVertexId(edgeId, 2, reverseEdge) = v0
378 // else:
379 // getEdgeVertexId(edgeId, 0, reverseEdge) = v0
380 // getEdgeVertexId(edgeId, 1, reverseEdge) = v1
381 // getEdgeVertexId(edgeId, 2, reverseEdge) = v2
382 //
383 int getEdgeVertexId(int edgeId, int n, bool reverseEdge) const
384 {
385 MNRY_ASSERT(n < mOuterEdgeVertexIdsCount[edgeId]);
386 int start = mOuterEdgeVertexIdsStart[edgeId];
387 return reverseEdge ?
388 start + mOuterEdgeVertexIdsCount[edgeId] - 1 - n : start + n;
389 }
