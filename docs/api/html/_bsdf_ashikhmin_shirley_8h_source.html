104public:

126 // no surface curvature required
127 finline void differentials(const scene_rdl2::math::Vec3f &wo, const scene_rdl2::math::Vec3f &wi,
128 float r1, float r2, const scene_rdl2::math::Vec3f &dNdx, const scene_rdl2::math::Vec3f &dNdy,
129 scene_rdl2::math::Vec3f &dDdx, scene_rdl2::math::Vec3f &dDdy) const override
130 {
131 // See BsdfLambert.h for details
132 localHemisphereCosineDifferentials(r1, r2, dDdx, dDdy);

136 dDdy = mFrame.localToGlobal(dDdy);
137 }
