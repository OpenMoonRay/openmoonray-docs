301 struct SortChildren
302 {
303 finline bool operator()(Node const *a, Node const *b) const
304 {
305 DebugRayVertex const *vertA = a->mVertex;
306 DebugRayVertex const *vertB = b->mVertex;
307
308 if (vertA->mScreenY != vertB->mScreenY) return vertA->mScreenY < vertB->mScreenY;
309 if (vertA->mScreenX != vertB->mScreenX) return vertA->mScreenX < vertB->mScreenX;
310 if (vertA->mDepth != vertB->mDepth) return vertA->mDepth < vertB->mDepth;
311 if (vertA->mThreadId != vertB->mThreadId) return vertA->mThreadId < vertB->mThreadId;
312
313 MNRY_ASSERT(vertA->mId != vertB->mId);
314 return vertA->mId < vertB->mId;
315 }
316 };

489finline void
490addRayVertex(pbr::TLState *tls,
491 scene_rdl2::math::Vec3f const &hitPoint,
492 scene_rdl2::math::Vec3f const &hitNormal,
493 scene_rdl2::math::Vec3f const &rayDiffX,
494 scene_rdl2::math::Vec3f const &rayDiffY)
495{
496 MNRY_ASSERT(!tls->mRayVertexStack.empty());
497
498 tls->mRayVertexStack.push_back(tls->mRayRecorder->extendRay(tls->mRayVertexStack.back()));
499
500 DebugRayVertex *vert = tls->mRayVertexStack.back();
501 vert->mHitPoint = hitPoint;
502 vert->mNormal = hitNormal;
503 vert->mRayDiffX = rayDiffX;
504 vert->mRayDiffY = rayDiffY;
505
506#ifdef DEBUG
507 // Debug functionality to isolate individual vertices of a ray differential.
508 // You can find the coordinate of a debug ray mesh in in mm_view and set
509 // vertToFind to that value to retrieve the callstack for how we got there.
510 static scene_rdl2::math::Vec3f vertToFind = scene_rdl2::math::Vec3f(FLT_MAX, FLT_MAX, FLT_MAX);
511
512 if (vertToFind != scene_rdl2::math::Vec3f(FLT_MAX, FLT_MAX, FLT_MAX)) {
513
514 scene_rdl2::math::Vec3f halfDx = (rayDiffX - hitPoint) * 0.5f;
515 scene_rdl2::math::Vec3f halfDy = (rayDiffY - hitPoint) * 0.5f;
516 scene_rdl2::math::Vec3f p0 = hitPoint - halfDx - halfDy;
517 scene_rdl2::math::Vec3f p1 = hitPoint + halfDx - halfDy;
518 scene_rdl2::math::Vec3f p2 = hitPoint - halfDx + halfDy;
519 scene_rdl2::math::Vec3f p3 = hitPoint + halfDx + halfDy;
520
521 scene_rdl2::math::Vec3f hit = transformPoint(DebugRayRecorder::getRenderToWorldMatrix(), hitPoint);
522
523 p0 = transformPoint(DebugRayRecorder::getRenderToWorldMatrix(), p0);
524 p1 = transformPoint(DebugRayRecorder::getRenderToWorldMatrix(), p1);
525 p2 = transformPoint(DebugRayRecorder::getRenderToWorldMatrix(), p2);
526 p3 = transformPoint(DebugRayRecorder::getRenderToWorldMatrix(), p3);
527
528 const float eps = 0.01f;
529
530 if (isEqual(vertToFind, hit, eps) ||
531 isEqual(vertToFind, p0, eps) ||
532 isEqual(vertToFind, p1, eps) ||
533 isEqual(vertToFind, p2, eps) ||
534 isEqual(vertToFind, p3, eps)) {
535 scene_rdl2::logging::Logger::info("Hit debug ray vertex at " , vertToFind ,
536 ", hit dx = " , (rayDiffX - hitPoint) ,
537 ", hit dy = " , (rayDiffY - hitPoint));
538 }
539 }
540#endif

544struct ExtendDebugRay
545{
546 // assumes transferred ray
547 finline ExtendDebugRay(pbr::TLState *pbrTls,
548 mcrt_common::RayDifferential const &ray, shading::Intersection const &isect) :
549 mTls(pbrTls)
550 {
551 if (DebugRayRecorder::isRecordingEnabled()) {
552 addRayVertex(mTls,
553 ray.getOrigin(),
554 isect.getN(),
555 ray.hasDifferentials() ? ray.getOriginX() : ray.getOrigin(),
556 ray.hasDifferentials() ? ray.getOriginY() : ray.getOrigin());
557 }
558 }
559
560 // assumes untransferred ray
561 finline ExtendDebugRay(pbr::TLState *pbrTls,
562 mcrt_common::RayDifferential const &ray, float dist) :
563 mTls(pbrTls)
564 {
565 if (DebugRayRecorder::isRecordingEnabled()) {
566 MNRY_ASSERT(dist < sDistantLightDistance);
567 mcrt_common::RayDifferential rd = ray;
568 rd.transfer(ray.getOrigin() + ray.getDirection() * dist, -ray.getDirection());
569 addRayVertex(mTls,
570 rd.getOrigin(),
571 -rd.getDirection(),
572 rd.hasDifferentials() ? rd.getOriginX() : rd.getOrigin(),
573 rd.hasDifferentials() ? rd.getOriginY() : rd.getOrigin());
574 }
575 }
576
577 // overload version of "with hit" constructor which accepts a mcrt_common::Ray
578 // instead of a mcrt_common::RayDifferential
579 finline ExtendDebugRay(pbr::TLState *pbrTls,
580 mcrt_common::Ray const &ray, shading::Intersection const &isect) :
581 mTls(pbrTls)
582 {
583 if (DebugRayRecorder::isRecordingEnabled()) {
584 scene_rdl2::math::ReferenceFrame fr(isect.getN());
585 float dist = ray.getEnd();
586 float scale = dist * 0.1f;
587 addRayVertex(mTls,
588 isect.getP(),
589 isect.getN(),
590 isect.getP() + fr.getX() * scale,
591 isect.getP() + fr.getY() * scale);
592 }
593 }
594
595 // overload version of "NO_HIT" constructor which accepts a mcrt_common::Ray
596 // instead of a mcrt_common::RayDifferential
597 finline ExtendDebugRay(pbr::TLState *pbrTls,
598 mcrt_common::Ray const &ray, float dist) :
599 mTls(pbrTls)
600 {
601 if (DebugRayRecorder::isRecordingEnabled()) {
602 MNRY_ASSERT(dist < sDistantLightDistance);
603 scene_rdl2::math::Vec3f hitPoint = ray.getOrigin() + ray.getDirection() * dist;
604 scene_rdl2::math::Vec3f hitNormal = -ray.getDirection();
605 scene_rdl2::math::ReferenceFrame fr(hitNormal);
606 float scale = dist * 0.1f;
607 addRayVertex(mTls,
608 hitPoint,
609 hitNormal,
610 hitPoint + fr.getX() * scale,
611 hitPoint + fr.getY() * scale);
612 }
613 }
614
615
616 finline ~ExtendDebugRay()
617 {
618 if (DebugRayRecorder::isRecordingEnabled()) {
619 mTls->mRayVertexStack.pop_back();
620 MNRY_ASSERT(!mTls->mRayVertexStack.empty());
621 }
622 }
623
624 pbr::TLState *mTls;
625};
