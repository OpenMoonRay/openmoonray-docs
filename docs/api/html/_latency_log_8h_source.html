55 void setOffsetByMs(const float offsetMs) { mOffsetMs = offsetMs; } // milli-sec
56
57 bool isPositive() const { return (mOffsetMs >= 0.0f)? true: false; }

191finline uint64_t
192LatencyItem::getCurrentMicroSec()
193{
194 struct timeval tv;
195 gettimeofday(&tv, 0x0);
196 uint64_t microSec = static_cast<uint64_t>(tv.tv_sec) * 1000 * 1000 + static_cast<uint64_t>(tv.tv_usec);
197 if (LatencyClockOffset::getInstance().isPositive()) {
198 microSec += LatencyClockOffset::getInstance().getAbsOffsetMicroSec();
199 } else {
200 microSec -= LatencyClockOffset::getInstance().getAbsOffsetMicroSec();
201 }
202 return microSec;
203}
204
205finline uint64_t
206LatencyItem::getLatencyMicroSec(const uint64_t startTime)
207{
208 return getCurrentMicroSec() - startTime;
209}
210
211finline float
212LatencyItem::getLatencySec(const uint64_t startTime)
213{
214 uint64_t currentTime = getCurrentMicroSec();
215
216 float deltaSec = 0.0f;
217 if (startTime < currentTime) {
218 uint64_t deltaTime = currentTime - startTime;
219 deltaSec = (float)deltaTime / 1000.0f / 1000.0f;
220 } else if (currentTime < startTime) {
221 uint64_t deltaTime = startTime - currentTime;
222 deltaSec = (float)deltaTime / 1000.0f / 1000.0f;
223 deltaSec *= -1.0f;
224 }
225
226 return deltaSec;
227}

229finline void
230LatencyItem::encode(VContainerEnq &vContainerEnq) const
231{
232#ifdef USE_VLCODEC
233 vContainerEnq.enqVLUInt(static_cast<unsigned int>(mTime));
234 vContainerEnq.enqVLUInt(static_cast<unsigned int>(mKey));
235 if (mKey == Key::RECV_PROGRESSIVEFRAME_START) {
236 vContainerEnq.enqVLUInt(mData[0]);
237 vContainerEnq.enqVLUInt(mData[1]);
238 }
239#else // else USE_VLCODEC
240 vContainerEnq.enqMask64(static_cast<uint64_t>(mTime));
241 vContainerEnq.enqInt(static_cast<int>(mKey));
242 if (mKey == Key::RECV_PROGRESSIVEFRAME_START) {
243 vContainerEnq.enqInt(static_cast<int>(mData[0]));
244 vContainerEnq.enqInt(static_cast<int>(mData[1]));
245 }
246#endif // end !USE_VLCODEC
247}
248
249finline void
250LatencyItem::decode(VContainerDeq &vContainerDeq)
251{
252#ifdef USE_VLCODEC
253 vContainerDeq.deqVLUInt(static_cast<unsigned int &>(mTime));
254 vContainerDeq.deqVLUInt(reinterpret_cast<unsigned int &>(mKey));
255 if (mKey == Key::RECV_PROGRESSIVEFRAME_START) {
256 mData.resize(2);
257 vContainerDeq.deqVLUInt(mData[0]);
258 vContainerDeq.deqVLUInt(mData[1]);
259 }
260#else // else USE_VLCODEC
261 uint64_t tmpTime;
262 vContainerDeq.deqMask64(static_cast<uint64_t &>(tmpTime));
263 mTime = static_cast<uint32_t>(tmpTime);
264 int tmpKey;
265 vContainerDeq.deqInt(static_cast<int &>(tmpKey));
266 mKey = static_cast<Key>(tmpKey);
267 if (mKey == Key::RECV_PROGRESSIVEFRAME_START) {
268 mData.resize(2);
269 int tmpInt;
270 vContainerDeq.deqInt(tmpInt); mData[0] = static_cast<unsigned int>(tmpInt);
271 vContainerDeq.deqInt(tmpInt); mData[1] = static_cast<unsigned int>(tmpInt);
272 }
273#endif // end !USE_VLCODEC
274}

350finline void
351LatencyLog::encode(VContainerEnq &vContainerEnq) const
352{
353#ifdef USE_VLCODEC
354 vContainerEnq.enqString(mName);
355 vContainerEnq.enqVLInt(mMachineId);
356 vContainerEnq.enqVLUInt(mSnapshotId);
357 vContainerEnq.enqVLSizeT(mDataSize);
358 vContainerEnq.enqMask64(mTimeBase);
359 vContainerEnq.enqVLSizeT(mLog.size());
360 for (size_t id = 0; id < mLog.size(); ++id) {
361 mLog[id].encode(vContainerEnq);
362 }
363#else // else USE_VLCODEC
364 vContainerEnq.enqString(mName);
365 vContainerEnq.enqInt(mMachineId);
366 vContainerEnq.enqInt((int)mSnapshotId);
367 vContainerEnq.enqMask64(mDataSize);
368 vContainerEnq.enqMask64(mTimeBase);
369 vContainerEnq.enqLong(mLog.size());
370 for (size_t id = 0; id < mLog.size(); ++id) {
371 mLog[id].encode(vContainerEnq);
372 }
373#endif // end !USE_VLCODEC
374}

376finline void
377LatencyLog::decode(VContainerDeq &vContainerDeq)
378{
379#ifdef USE_VLCODEC
380 vContainerDeq.deqString(mName);
381 vContainerDeq.deqVLInt(mMachineId);
382 vContainerDeq.deqVLUInt(mSnapshotId);
383 vContainerDeq.deqVLSizeT(mDataSize);
384 vContainerDeq.deqMask64(mTimeBase);
385
386 size_t total;
387 vContainerDeq.deqVLSizeT(total);
388 mLog.resize(total);
389 for (size_t id = 0; id < total; ++id) {
390 mLog[id].decode(vContainerDeq);
391 }
392#else // else USE_VLCODEC
393 vContainerDeq.deqString(mName);
394 vContainerDeq.deqInt(mMachineId);
395 int tmpI;
396 vContainerDeq.deqInt(tmpI); mSnapshotId = static_cast<uint32_t>(tmpI);
397 vContainerDeq.deqMask64(mDataSize);
398 vContainerDeq.deqMask64(mTimeBase);
399
400 long total;
401 vContainerDeq.deqLong(total);
402 mLog.resize(total);
403 for (size_t id = 0; id < static_cast<size_t>(total); ++id) {
404 mLog[id].decode(vContainerDeq);
405 }
406#endif // end !USE_VLCODEC
407}
