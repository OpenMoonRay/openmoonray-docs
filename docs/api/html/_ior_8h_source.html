16/**
17 * This structure stores incident and transmitted Index Of Refraction (Ior)
18 * as well as their ratio
19 */
20class ShaderIor
21{
22public:
23
24 /// @brief Initialize a ShaderIor object given the current shading state and
25 /// material index of refraction. It automatically handles computing the correct
26 /// incident and transmitted indices of refraction, as well as their ratio.
27 ShaderIor(const shading::State &state, const float materialIor, bool isThinGeometry = false)
28 {
29 // If the geometry is infinitely "thin" (non-manifold, not water-tight) then we should always treat
30 // the state as 'entering' regardless of what state.isEntering() reports. There can never be
31 // a case where the ray origin is "inside" the geometry because it is infinitely thin, so we treat the
32 // observer as always "outside" the thin geometry.
33 mIncident = (state.isEntering() || isThinGeometry) ? state.getMediumIor() : materialIor;
34 mTransmitted = (state.isEntering() || isThinGeometry) ? materialIor : state.getMediumIor();
35
36 mRatio = mIncident * scene_rdl2::math::rcp(mTransmitted);
37 }
38
39 /// @brief This special constructor is only for cases where we need to explicitly specify
40 /// the mediumIor using a value that comes from the shader rather than the state.
41 /// This usage case should be extremely rare and therefore this constructor should
42 /// very rarely be used.
43 ShaderIor(const shading::State &state, const float mediumIor, const float materialIor, bool isThinGeometry = false)
44 {
45 mIncident = state.isEntering() ? mediumIor : isThinGeometry ? mediumIor : materialIor;
46 mTransmitted = state.isEntering() ? materialIor : isThinGeometry ? materialIor : mediumIor;
47 mRatio = mIncident * scene_rdl2::math::rcp(mTransmitted);
48 }
49
50 /// Accessors
51 float getIncident() const { return mIncident; }
52 float getTransmitted() const { return mTransmitted; }
53 float getRatio() const { return mRatio; }
54
55private:
56 float mIncident;
57 float mTransmitted;
58 float mRatio; // mIncident / mTransmitted
59};

61/**
62 * This class stores complex index of refraction that is generally used to represent
63 * metallic surface interactions.
64 */
65class ShaderComplexIor
66{
67public:
68
69 /// @brief Initialize a ShaderComplexIor object given the material complex index of refraction.
70 ShaderComplexIor(const scene_rdl2::math::Color& eta,
71 const scene_rdl2::math::Color& absorption) :
72 mEta(eta),
73 mAbsorption(absorption)
74 { }
75
76 static ShaderComplexIor
77 createComplexIorFromColor(const scene_rdl2::math::Color& color,
78 const scene_rdl2::math::Color& edgeColor)
79 {
80 // Make sure no color channel is at 1.0
81 const scene_rdl2::math::Color maxRefl = scene_rdl2::math::Color(0.999f);
82 const scene_rdl2::math::Color clampedColor = clamp(color, scene_rdl2::math::sBlack, maxRefl);
83 const scene_rdl2::math::Color clampedEdgeColor = clamp(edgeColor, scene_rdl2::math::sBlack, maxRefl);
84 // Convert the colors to complex IOR
85 const scene_rdl2::math::Color eta = computeEta(clampedColor,
86 clampedEdgeColor);
87 const scene_rdl2::math::Color k = computeK(clampedColor,
88 eta);
89
90 return ShaderComplexIor(eta, k);
91 }
92
93 scene_rdl2::math::Color getEta() const { return mEta; }
94 scene_rdl2::math::Color getAbsorption() const { return mAbsorption; }
95
96private:
97 // functions for computing complex IOR values for conductor Fresnel
98 // from 'reflectivity' and 'edge tint' colors.
99 // See paper: "Artist Friendly Metallic Fresnel", by Ole Gulbrandsen
100 // from Framestore, published at JCGT in 2014 (http://jcgt.org)
101 finline static scene_rdl2::math::Color
102 nMin(const scene_rdl2::math::Color &r)
103 {
104 return (scene_rdl2::math::sWhite - r) / (scene_rdl2::math::sWhite + r);
105 }
106
107 finline static scene_rdl2::math::Color
108 nMax(const scene_rdl2::math::Color &r)
109 {
110 scene_rdl2::math::Color rSqrt = sqrt(r);
111 return (scene_rdl2::math::sWhite + rSqrt) / (scene_rdl2::math::sWhite - rSqrt);
112 }
113
114 finline static scene_rdl2::math::Color
115 computeEta(const scene_rdl2::math::Color &r, const scene_rdl2::math::Color &g)
116 {
117 return g * nMin(r) + (scene_rdl2::math::sWhite - g) * nMax(r);
118 }
119
120 finline static scene_rdl2::math::Color
121 computeK(const scene_rdl2::math::Color &r, const scene_rdl2::math::Color &n)
122 {
123 const scene_rdl2::math::Color a = n + scene_rdl2::math::sWhite;
124 const scene_rdl2::math::Color b = n - scene_rdl2::math::sWhite;
125 // Take a max() here to get rid of any numerical -0 etc
126 const scene_rdl2::math::Color nr = max(scene_rdl2::math::sBlack, r * a * a - b * b);
127 return sqrt(nr / (scene_rdl2::math::sWhite - r));
128 }
129
130private:
131 scene_rdl2::math::Color mEta;
132 scene_rdl2::math::Color mAbsorption;
133};
