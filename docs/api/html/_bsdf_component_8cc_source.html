19MicrofacetAnisotropicBRDF::MicrofacetAnisotropicBRDF(
20 const scene_rdl2::math::Vec3f& N,
21 float eta,
22 float roughnessU,
23 float roughnessV,
24 const scene_rdl2::math::Vec3f& shadingTangent,
25 ispc::MicrofacetDistribution microfacetDistribution,
26 ispc::MicrofacetGeometric microfacetGeometric,
27 const Iridescence* const iridescence) :
28 BsdfComponent(iridescence),
29 mN(N),
30 mEta(scene_rdl2::math::Color(eta)),
31 mK(scene_rdl2::math::sBlack),
32 mRoughnessU(roughnessU),
33 mRoughnessV(roughnessV),
34 mShadingTangent(shadingTangent),
35 mMicrofacetDistribution(microfacetDistribution),
36 mMicrofacetGeometric(microfacetGeometric),
37 mIsConductor(false)
38{
39 // Average Dielectric Fresnel Reflectance
40 // Ref:"Revisiting Physically Based Shading", Kulla'17
41 float favg = shading::averageFresnelReflectance(eta);
42 mFavg = scene_rdl2::math::sWhite * favg;
43}
