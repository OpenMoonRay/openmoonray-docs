34/**
35 * An AttributeKey is a lightweight object for retrieving the value of an
36 * attribute from a SceneObject.
37 *
38 * AttributeKeys are templated on a C++ type corresponding to their attribute
39 * type. This allows us to do static typechecking wherever possible, and most
40 * importantly, do fast, typesafe gets and sets on attribute values.
41 * Unfortunately we can't statically check everything, so sometimes those
42 * sanity type checks will happen at runtime and throw an except::TypeError if
43 * you've done something wrong.
44 *
45 * AttributeKeys are lightweight (16 bytes), and can be compared for equality.
46 * However, comparing AttributeKeys from different SceneClasses is invalid, and
47 * the result of such a comparison is undefined.
48 *
49 * AttributeKeys that are default constructed (not assigned from a valid
50 * AttributeKey or constructed from an Attribute) are invalid until a valid
51 * AttributeKey is assigned into them.
52 *
53 * Thread Safety:
54 * - All data members are baked in at construction time. Since AttributeKey
55 * objects are immutable after construction, reading their members from
56 * multiple threads without synchronization is safe.
57 */
58template <typename T>
59class AttributeKey
60{
61public:
62 typedef T Type;
63
64 /**
65 * Construct an AttributeKey directly from an Attribute object.
66 *
67 * @param attribute The attribute object this AttributeKey will refer to.
68 */
69 finline explicit AttributeKey(const Attribute& attribute);
70
71 /**
72 * Default constructor (for convenience). A default constructed
73 * AttributeKey is invalid until it is assigned to by a valid AttributeKey.
74 */
75 finline AttributeKey();
76
77 /**
78 * Test two AttributeKeys for equality.
79 *
80 * @warning It is invalid to compare AttributeKeys from different
81 * SceneClasses, and the result of such a comparison is undefined.
82 *
83 * @param other The other AttributeKey to compare with this one.
84 */
85 finline bool operator==(const AttributeKey<T>& other) const;
86
87 /**
88 * Test two AttributeKeys for inequality.
89 *
90 * @warning It is invalid to compare AttributeKeys from different
91 * SceneClasses, and the result of such a comparison is undefined.
92 *
93 * @param other The other AttributeKey to compare with this one.
94 */
95 finline bool operator!=(const AttributeKey<T>& other) const;
96
97 /// Returns true if the attribute key is valid. Default constructed
98 /// AttributeKeys are not valid.
99 finline bool isValid() const;
100
101 /// Returns true if the underlying attribute is bindable.
102 finline bool isBindable() const;
103
104 /// Returns true if the underlying attribute is blurrable.
105 finline bool isBlurrable() const;
106
107 /// Returns true if the underlying attribute is an enumeration.
108 finline bool isEnumerable() const;
109
110 /// Returns true if the underlying attribute represents a filename.
111 finline bool isFilename() const;
112
113private:
114 // The index into the vector of attributes in the SceneClass.
115 uint32_t mIndex;
116
117 // The offset of this attribute in its memory chunk.
118 uint32_t mOffset;
119
120 // The flags of this attribute.
121 AttributeFlags mFlags;
122
123 // The mask of object types allowed (if the attribute is a SceneObject).
124 SceneObjectInterface mObjectType;
125
126 // The SceneClass needs to access the index for attribute lookup.
127 friend class SceneClass;
128
129 // The SceneObject needs to access the offset for attribute lookup.
130 friend class SceneObject;
131
132 // SceneObject derived classes which need access to the index for manually
133 // setting the set flags.
134 friend class GeometrySet;
135 friend class Layer;
136 friend class LightFilterSet;
137 friend class LightSet;
138 friend class TraceSet;
139
140 // Classes which need access for unit testing purposes.
141 friend class unittest::TestAttributeKey;
142 friend class unittest::TestSceneClass;
143 friend class unittest::TestSceneObject;
144};

146template <typename T>
147AttributeKey<T>::AttributeKey(const Attribute& attribute) :
148 mIndex(attribute.mIndex),
149 mOffset(attribute.mOffset),
150 mFlags(attribute.mFlags),
151 mObjectType(attribute.mObjectType)
152{
153 // Unfortunately we have to fall back on a runtime type check here.
154 if (attributeType<T>() != attribute.getType()) {
155 std::stringstream errMsg;
156 errMsg << "Type mismatch between AttributeKey of type '" <<
157 attributeTypeName<T>() << "' and Attribute '" << attribute.getName() <<
158 "' of type '" << attributeTypeName(attribute.getType()) << "'.";
159 throw except::TypeError(errMsg.str());
160 }
161}

163template <typename T>
164AttributeKey<T>::AttributeKey() :
165 mIndex(std::numeric_limits<uint32_t>::max()),
166 mOffset(std::numeric_limits<uint32_t>::max()),
167 mFlags(FLAGS_NONE),
168 mObjectType(INTERFACE_GENERIC)
169{
170}

172template <typename T>
173bool
174AttributeKey<T>::operator==(const AttributeKey<T>& other) const
175{
176 // Invalid AttributeKeys are not equal to anything.
177 if (!isValid() || !other.isValid()) {
178 return false;
179 }
180
181 // No need to check the offset. The index is enough.
182 return mIndex == other.mIndex;
183}
184
185template <typename T>
186bool
187AttributeKey<T>::operator!=(const AttributeKey<T>& other) const
188{
189 // Inequality is defined in terms of equality.
190 return !(*this == other);
191}
192
193template <typename T>
194bool
195AttributeKey<T>::isValid() const
196{
197 return (mIndex != std::numeric_limits<uint32_t>::max()) ||
198 (mOffset != std::numeric_limits<uint32_t>::max());
199}
