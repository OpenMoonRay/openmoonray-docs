146template <typename T>
147AttributeKey<T>::AttributeKey(const Attribute& attribute) :
148 mIndex(attribute.mIndex),
149 mOffset(attribute.mOffset),
150 mFlags(attribute.mFlags),
151 mObjectType(attribute.mObjectType)
152{
153 // Unfortunately we have to fall back on a runtime type check here.
154 if (attributeType<T>() != attribute.getType()) {
155 std::stringstream errMsg;
156 errMsg << "Type mismatch between AttributeKey of type '" <<
157 attributeTypeName<T>() << "' and Attribute '" << attribute.getName() <<
158 "' of type '" << attributeTypeName(attribute.getType()) << "'.";
159 throw except::TypeError(errMsg.str());
160 }
161}

172template <typename T>
173bool
174AttributeKey<T>::operator==(const AttributeKey<T>& other) const
175{
176 // Invalid AttributeKeys are not equal to anything.
177 if (!isValid() || !other.isValid()) {
178 return false;
179 }
180
181 // No need to check the offset. The index is enough.
182 return mIndex == other.mIndex;
183}

185template <typename T>
186bool
187AttributeKey<T>::operator!=(const AttributeKey<T>& other) const
188{
189 // Inequality is defined in terms of equality.
190 return !(*this == other);
191}
