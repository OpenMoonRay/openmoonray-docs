1138static bool
1139intersect(const Vec3f &P, const Vec3f &wi, const Vec3f Plight[3],
1140 Vec3f &weights)
1141{
1142 Vec3f mV0(Plight[0]);
1143 Vec3f mE1 = Plight[1] - mV0;
1144 Vec3f mE2 = Plight[2] - mV0;
1145 Vec3f mE1CrossE2 = cross(mE1, mE2);
1146
1147 float det = -dot(wi, mE1CrossE2);
1148 if (scene_rdl2::math::abs(det) < sEpsilon) {
1149 return false;
1150 }
1151 float invDet = 1.0f / det;
1152
1153 Vec3f tvec = P - mV0;
1154 float distance = dot(mE1CrossE2, tvec) * invDet;
1155 if (distance < 0.0f) {
1156 return false;
1157 }
1158
1159 Vec3f txr = cross(tvec, wi);
1160
1161 float alpha = dot(txr, mE2) * invDet; // alpha
1162 if (alpha < 0.0f || alpha > 1.0f) {
1163 return false;
1164 }
1165
1166 float beta = -dot(txr, mE1) * invDet; // beta
1167 if (beta < 0.0f || beta > 1.0f) {
1168 return false;
1169 }
1170
1171 if (alpha + beta > 1.0f) {
1172 return false;
1173 }
1174
1175 weights[0] = 1.0f - (alpha + beta);
1176 weights[1] = alpha;
1177 weights[2] = beta;
1178
1179 return true;
1180}
