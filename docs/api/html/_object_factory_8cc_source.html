62SceneObject*
63proxyCreate(const SceneClass& sceneClass, const std::string& name)
64{
65 auto interface = sceneClass.getDeclaredInterface();
66
67 // Create the proper proxy class based on the declared interface of the
68 // DSO type.
69 if (interface & INTERFACE_CAMERA) {
70 return new CameraProxy(sceneClass, name);
71 } else if (interface & INTERFACE_ENVMAP) {
72 return new EnvMapProxy(sceneClass, name);
73 } else if (interface & INTERFACE_GEOMETRY) {
74 return new GeometryProxy(sceneClass, name);
75 } else if (interface & INTERFACE_LIGHT) {
76 return new LightProxy(sceneClass, name);
77 } else if (interface & INTERFACE_MAP) {
78 return new MapProxy(sceneClass, name);
79 } else if (interface & INTERFACE_NORMALMAP) {
80 return new NormalMapProxy(sceneClass, name);
81 } else if (interface & INTERFACE_MATERIAL) {
82 if (interface & INTERFACE_DWABASELAYERABLE) {
83 return new DwaBaseLayerableProxy(sceneClass, name);
84 } else if (interface & INTERFACE_DWABASEHAIRLAYERABLE) {
85 return new DwaBaseHairLayerableProxy(sceneClass, name);
86 } else {
87 return new MaterialProxy(sceneClass, name);
88 }
89 } else if (interface & INTERFACE_DISPLACEMENT) {
90 return new DisplacementProxy(sceneClass, name);
91 } else if (interface & INTERFACE_VOLUMESHADER) {
92 return new VolumeShaderProxy(sceneClass, name);
93 } else if (interface & INTERFACE_LIGHTFILTER) {
94 return new LightFilterProxy(sceneClass, name);
95 } else if (interface & INTERFACE_DISPLAYFILTER) {
96 return new DisplayFilterProxy(sceneClass, name);
97 } else {
98 std::string errStr ="Undefined Scene Object Interface: " + name + "\n";
99 MNRY_ASSERT(false, errStr.c_str());
100 }
101
102
103 // If none of these interfaces match, we can't get any more specific than a
104 // generic SceneObject.
105 return new SceneObjectProxy(sceneClass, name);
106}

111 delete sceneObject;

158std::unique_ptr<ObjectFactory>
159ObjectFactory::createDsoFactory(const std::string& className, const std::string& dsoPath)
160{
161 // Open the DSO.
162 std::unique_ptr<Dso> dso(new Dso(className, dsoPath, false));
163
164 // Extract the declare, create, and destroy function pointers.
165 ClassDeclareFunc declarer = dso->getDeclare();
166 ObjectCreateFunc creator = dso->getCreate();
167 ObjectDestroyFunc destroyer = dso->getDestroy();
168
169 // Create the ObjectFactory.
170 return std::unique_ptr<ObjectFactory>(
171 new ObjectFactory(declarer, creator, destroyer, std::move(dso)));
172}

183 // Create the ObjectFactory.
184 return std::unique_ptr<ObjectFactory>(
185 new ObjectFactory(declarer, proxyCreate, proxyDestroy, std::move(dso)));
186}
