36void
37TestTraceSet::testSerialize()
38{
39 // Create some geometries.
40 Geometry* teapot1 = mContext->createSceneObject("FakeTeapot", "/seq/shot/teapot1")->asA<Geometry>();
41 Geometry* teapot2 = mContext->createSceneObject("FakeTeapot", "/seq/shot/teapot2")->asA<Geometry>();
42
43 // Create a trace set
44 TraceSet* traceSet = mContext->createSceneObject("TraceSet", "/seq/shot/traceset")->asA<TraceSet>();
45
46 // Make some assignments in the trace set, verify they get new IDs.
47 traceSet->beginUpdate();
48 CPPUNIT_ASSERT(traceSet->assign(teapot1, "lid") == 0);
49 CPPUNIT_ASSERT(traceSet->assign(teapot1, "spout") == 1);
50 CPPUNIT_ASSERT(traceSet->assign(teapot2, "lid") == 2);
51 CPPUNIT_ASSERT(traceSet->assign(teapot2, "body") == 3);
52 traceSet->endUpdate();
53
54 // Serialize
55 BinaryWriter writer(*mContext);
56 std::stringstream ss;
57 writer.toStream(ss);
58
59 // Create a copy of the context
60 SceneContext context;
61 BinaryReader reader(context);
62 reader.fromStream(ss);
63 context.commitAllChanges();
64
65 // Perform an update of the trace set
66 mContext->commitAllChanges();
67 traceSet->beginUpdate();
68 CPPUNIT_ASSERT(traceSet->assign(teapot2, "base") == 4);
69 traceSet->endUpdate();
70
71 // Serialize with delta encoding on
72 BinaryWriter writer2(*mContext);
73 writer2.setDeltaEncoding(true);
74 std::stringstream ss2;
75 writer2.toStream(ss2);
76
77 // Deserialize to the copy
78 SceneContext context2;
79 BinaryReader reader2(context2);
80 reader2.fromStream(ss2);
81
82 // Get the trace set from the copy
83 teapot1 = context2.getSceneObject("/seq/shot/teapot1")->asA<Geometry>();
84 teapot2 = context2.getSceneObject("/seq/shot/teapot2")->asA<Geometry>();
85 TraceSet * traceSetRead = context2.getSceneObject("/seq/shot/traceset")->asA<TraceSet>();
86
87 // check attributes
88 CPPUNIT_ASSERT(traceSetRead->getAssignmentCount() == 5);
89 CPPUNIT_ASSERT(traceSetRead->getAssignmentId(teapot1, "lid") == 0);
90 CPPUNIT_ASSERT(traceSetRead->getAssignmentId(teapot1, "spout") == 1);
91 CPPUNIT_ASSERT(traceSetRead->getAssignmentId(teapot2, "lid") == 2);
92 CPPUNIT_ASSERT(traceSetRead->getAssignmentId(teapot2, "body") == 3);
93 CPPUNIT_ASSERT(traceSetRead->getAssignmentId(teapot2, "base") == 4);
94 CPPUNIT_ASSERT(traceSetRead->contains(teapot1));
95 CPPUNIT_ASSERT(traceSetRead->contains(teapot2));
96 CPPUNIT_ASSERT(traceSetRead->lookupGeomAndPart(0).first == teapot1);
97 CPPUNIT_ASSERT(traceSetRead->lookupGeomAndPart(0).second == "lid");
98 CPPUNIT_ASSERT(traceSetRead->lookupGeomAndPart(1).first == teapot1);
99 CPPUNIT_ASSERT(traceSetRead->lookupGeomAndPart(1).second == "spout");
100 CPPUNIT_ASSERT(traceSetRead->lookupGeomAndPart(2).first == teapot2);
101 CPPUNIT_ASSERT(traceSetRead->lookupGeomAndPart(2).second == "lid");
102 CPPUNIT_ASSERT(traceSetRead->lookupGeomAndPart(3).first == teapot2);
103 CPPUNIT_ASSERT(traceSetRead->lookupGeomAndPart(3).second == "body");
104 CPPUNIT_ASSERT(traceSetRead->lookupGeomAndPart(4).first == teapot2);
105 CPPUNIT_ASSERT(traceSetRead->lookupGeomAndPart(4).second == "base");
106}
