40// From http://extremelearning.com.au/unreasonable-effectiveness-of-quasirandom-sequences
41// The unreasonable effectiveness of quasi-random sequences
42// Martin Roberts
43inline Sample2D sampleR2(uint32_t i) noexcept
44{
45 // We define g as the generalized version of the golden ratio in d dimensions as the unique positive root
46 // x^(d + 1) = x + 1
47 // Here we're in 2 dimensions.
48 constexpr float g = 1.324717957244746f; // special math constant
49 Sample2D a{1.0f/g, 1.0f/(g*g)};
50 return Sample2D{mod1(a.u * (i + 1)), mod1(a.v * (i + 1))};
51}

53// From http://extremelearning.com.au/a-simple-method-to-construct-isotropic-quasirandom-blue-noise-point-sequences/
54// A simple method to construct isotropic quasi-random blue noise point sequences
55// Martin Roberts
56inline Sample2D jitteredR2(uint32_t n, uint32_t scramble) noexcept
57{
58 constexpr float lambda = 0.4f; // User parameter
59 constexpr float delta = 0.76f; // Empirically optimized parameter
60
61 // To jitter a point in the Fibonacci lattice of N points, you can offset the points as below:
62 // p_i(x, y) + epsilon_i(delta_0/sqrt(N))
63 // where delta_0 =~ 0.78
64 // The author comes to the following value of i0 value by taking the mapping of two random variables to a disk of
65 // radius r: epsilon_i(r) = (r*sqrt(u1) * cos(2*pi*u2), r*sqrt(ui) * sin(2*pi*u2))
66 // With the coefficient of 1/sqrt(N) above, the author asserts that an infinite sequence can be substituted as
67 // follows:
68 // 1/sqrt(N) -> 1/(2 * sqrt(i - i0)), where i0 = 0.700 for i > 0
69 // However, we're using zero-based indexing. Let
70 // n = i - 1
71 // Therefore
72 // i - i0 = (n + 1) - i0 = n + (1 - i0)
73 // Let n0 = 1 - i0
74 // Taking n in [0, inf) leads to 1/(2 * sqrt(n + n0))
75 constexpr float i0 = 0.700f; // Special math constant
76 constexpr float n0 = 1.0f - i0;
77
78 const Sample2D p = sampleR2(n);
79 const Sample2D u = randomPoint(n + scramble);
80
81 const float k = lambda * delta * sqrt_pi<float>/(4.0f * std::sqrt(n + n0));
82 const Sample2D j{k*u.u, k*u.v};
83 const Sample2D result{mod1(p.u + j.u), mod1(p.v + j.v)};
84 return result;
85}

87// http://extremelearning.com.au/unreasonable-effectiveness-of-quasirandom-sequences/
88inline float jitteredR1(uint32_t i, uint32_t scramble) noexcept
89{
90 using scene_rdl2::util::bitsToFloat;
91 constexpr float lambda = 0.4f; // User parameter
92 // Guessed parameter: TODO: we should do an analysis as done for 2D points here:
93 // http://extremelearning.com.au/a-simple-method-to-construct-isotropic-quasirandom-blue-noise-point-sequences/
94 constexpr float delta = 0.26f;
95 constexpr float i0 = 1.010f;
96
97 // We want three scramble values: the starting offset (s0), the hash index, and our random value. We could pass in
98 // three separate values, but I worry about correlation since the hash values will be generated by the same sequence
99 // identifier and the same hash function. Here we're still using the same initial value, but our hash functions are
100 // at least unique. I am not worried about loss of precision in the floating point values (notably in s0), as these
101 // only serve to add some additional randomness.
102 const uint16_t scramble0(scramble & 0xFFFFu);
103 const uint16_t scramble1((scramble >> 16u) & 0xFFFFu);
104
105 static_assert(std::numeric_limits<decltype(scramble1)>::max() >= kMoebiusTransformationSize,
106 "For a good distribution, we should be at least as big as our array");
107 const uint32_t idx = hash16_s6(scramble1) % kMoebiusTransformationSize;
108 const float s0 = bitsToFloat(static_cast<uint32_t>(hash16_s6(scramble0) << 16u));
109
110 const float p = s0 + i * moebiusTransformationValues[idx];
111 const float k = lambda * delta / (i + i0);
112 const float u = bitsToFloat(triple32inc(scramble));
113 const float j = k*u;
114
115 return mod1(p + j);
116}
