13bool
14OiioReader::readData(const int subImageId)
15{
16 if (!mIn) return false;
17
18 if (!mIn->seek_subimage(subImageId, mSpec)) {
19 return false; // failed to seek
20 }
21
22 const OIIO::ImageSpec &spec = mSpec;
23 mPixels.resize(spec.width * spec.height * spec.nchannels);
24 if (mPixels.empty()) {
25 return false; // internal pixels buffer resize failed
26 }
27
28 mIn->read_image(OIIO::TypeDesc::FLOAT, mPixels.data());
29 return true;
30}
