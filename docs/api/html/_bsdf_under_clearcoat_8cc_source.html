22Color
23UnderClearcoatBsdfLobe::eval(const BsdfSlice &slice,
24 const Vec3f &wi,
25 float *pdf) const
26{
27 if (pdf != nullptr)
28 *pdf = 0.0f;
29
30 Vec3f woPrime;
31 float cosThetaWo, cosThetaWoPrime;
32 if (!computeRefractionDirection(mN,
33 slice.getWo(),
34 mNeta,
35 woPrime,
36 cosThetaWo,
37 cosThetaWoPrime))
38 {
39 // Total Outward Reflection handled by the clearcoat layer on top
40 return scene_rdl2::math::sBlack;
41 }
42 if (cosThetaWo <= scene_rdl2::math::sEpsilon) return scene_rdl2::math::sBlack;
43
44 // Reverse refraction direction
45 woPrime = -woPrime;
46
47 BsdfSlice underSlice(slice.getNg(),
48 woPrime,
49 slice.getIncludeCosineTerm(),
50 slice.getEntering(),
51 slice.getShadowTerminatorFix(),
52 slice.getFlags());
53
54 Vec3f wiPrime;
55 float pdfPrime = 0.0f;
56 float cosThetaWi, cosThetaWiPrime;
57 if (!computeRefractionDirection(mN,
58 wi,
59 mNeta,
60 wiPrime,
61 cosThetaWi,
62 cosThetaWiPrime))
63 {
64 // Total Outward Reflection handled by the clearcoat layer on top
65 return scene_rdl2::math::sBlack;
66 }
67
68 if (cosThetaWi <= scene_rdl2::math::sEpsilon) return scene_rdl2::math::sBlack;
69
70 // Reverse the refraction vector to face outward
71 wiPrime = -wiPrime;
72 Color result = mUnder->eval(underSlice, wiPrime, &pdfPrime);
73
74 // portion of light transmitted through air->clearcoat interface approximated using cosThetaWo
75 result *= computeScaleAndFresnel(cosThetaWo);
76
77 // Clearcoat Absorption
78 const Color ct = computeTransmission(cosThetaWoPrime, cosThetaWiPrime);
79
80 // The light that bounces off the under lobe does not entirely exit the clearcoat immediately.
81 // Here we model multiple 'bounces' within the clearcoat layer itself.
82 // At each bounce some portion of light escapes into the 'air', while the rest is reflected
83 // inward. The portion that is reflected inward is attenuated by clearcoat absorption, and
84 // then again reflected off the under lobe, and this process repeats. We use a geometric
85 // series to approximate the result.
86 // The series looks like:
87 // Ti*UnderBSDF*ct*To + Ti*UnderBSDF*ct^2*Ro*AvgReflectanceUnderLobe*To + Ti*UnderBSDF*ct^3*Ro^2*AvgReflectanceUnderLobe^2*To + ...(so on)
88 // This is equivalent to the following sum:
89 // Ti*UnderBSDF*ct*To * 1/(1 - ct*Ro*AveReflectanceUnderLobe)
90 const Color Fr = mExitingFresnel->eval(cosThetaWoPrime);
91 const Color Ft = scene_rdl2::math::sWhite - Fr;
92 const Color albedo = mUnder->albedo(underSlice);
93 result = (result * ct * Ft) * rcp(maxColor(scene_rdl2::math::sWhite - (ct * Fr * albedo), 0.01f));
94
95 // Now we account for the change in variables because of bending rays.
96 // For the derivation, please look in:
97 // /work/gshad/moonshine/papers/ClearcoatChangeOfVariables.pdf
98 float jacobian = scene_rdl2::math::abs(cosThetaWi) / scene_rdl2::math::max(scene_rdl2::math::abs(cosThetaWiPrime),0.01f);
99 if (mUnder->matchesFlag(BsdfLobe::GLOSSY)) {
100 jacobian *= mNeta*mNeta;
101 }
102 if (pdf != nullptr)
103 *pdf = pdfPrime * jacobian;
104
105 result *= jacobian;
106
107 return result;
108}

110Color
111UnderClearcoatBsdfLobe::sample(const BsdfSlice &slice,
112 float r1, float r2,
113 Vec3f &wi,
114 float &pdf) const
115{
116 pdf = 0.0f;
117 Vec3f woPrime;
118 Color result;
119 float cosThetaWo, cosThetaWoPrime;
120 if (!computeRefractionDirection(mN,
121 slice.getWo(),
122 mNeta,
123 woPrime,
124 cosThetaWo,
125 cosThetaWoPrime))
126 {
127 // Total Outward Reflection
128 return scene_rdl2::math::sBlack;
129 }
130 // Reverse refraction direction
131 woPrime = -woPrime;
132 BsdfSlice underSlice(slice.getNg(),
133 woPrime,
134 slice.getIncludeCosineTerm(),
135 slice.getEntering(),
136 slice.getShadowTerminatorFix(),
137 slice.getFlags());
138
139 // sample the under lobe
140 Vec3f wiPrime;
141 float pdfPrime;
142 result = mUnder->sample(underSlice, r1, r2, wiPrime, pdfPrime);
143
144 if (scene_rdl2::math::isZero(pdfPrime)) {
145 return scene_rdl2::math::sBlack;
146 }
147
148 float cosThetaWi, cosThetaWiPrime;
149 if (!computeRefractionDirection(mN,
150 wiPrime,
151 scene_rdl2::math::rcp(mNeta),
152 wi,
153 cosThetaWiPrime,
154 cosThetaWi))
155 {
156 // Total Internal Reflection
157 if (mPassThroughTIRWhenSampling) {
158 // Assume this ray bounces around inside the clearcoat interface
159 // and eventually gets back out in the same direction.
160 // Pass the ray through
161 wi = wiPrime;
162 cosThetaWi = cosThetaWiPrime;
163 }
164 else {
165 // Discard this sample
166 return scene_rdl2::math::sBlack;
167 }
168 }
169 else {
170 // Reverse the refracted vector to point outward
171 wi = -wi;
172 }
173
174 // portion of light transmitted through air->clearcoat interface approximated using cosThetaWo
175 result *= computeScaleAndFresnel(cosThetaWo);
176
177 // Clearcoat Absorption
178 const Color ct = computeTransmission(cosThetaWoPrime, cosThetaWiPrime);
179
180 // The light that bounces off the under lobe does not entirely exit the clearcoat immediately.
181 // Here we model multiple 'bounces' within the clearcoat layer itself.
182 // At each bounce some portion of light escapes into the 'air', while the rest is reflected
183 // inward. The portion that is reflected inward is attenuated by clearcoat absorption, and
184 // then again reflected off the under lobe, and this process repeats. We use a geometric
185 // series to approximate the result.
186 // The series looks like:
187 // Ti*UnderBSDF*ct*To + Ti*UnderBSDF*ct^2*Ro*AvgReflectanceUnderLobe*To + Ti*UnderBSDF*ct^3*Ro^2*AvgReflectanceUnderLobe^2*To + ...(so on)
188 // This is equivalent to the following sum:
189 // Ti*UnderBSDF*ct*To * 1/(1 - ct*Ro*AveReflectanceUnderLobe)
190 const Color Fr = mExitingFresnel->eval(cosThetaWoPrime);
191 const Color Ft = scene_rdl2::math::sWhite - Fr;
192 const Color albedo = mUnder->albedo(underSlice);
193 result = (result * ct * Ft) * rcp(maxColor(scene_rdl2::math::sWhite - (ct * Fr * albedo), 0.01f));
194
195 // Now we account for the change in variables because of bending rays.
196 // For the derivation, please look in:
197 // /work/gshad/moonshine/papers/ClearcoatChangeOfVariables.pdf
198 float jacobian = scene_rdl2::math::abs(cosThetaWi) / scene_rdl2::math::max(scene_rdl2::math::abs(cosThetaWiPrime),0.01f);
199 if (mUnder->matchesFlag(BsdfLobe::GLOSSY)) {
200 jacobian *= mNeta*mNeta;
201 }
202 pdf = pdfPrime * jacobian;
203
204 result *= jacobian;
205
206 return result;
207}
