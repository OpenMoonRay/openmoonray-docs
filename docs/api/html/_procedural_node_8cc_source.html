58void
59ProceduralNode::forEachPrimitive(PrimitiveVisitor& visitor,
60 bool parallel)
61{
62 if (parallel) {
63 tbb::blocked_range<size_t> range(0, mImpl->mChildren.size());
64 tbb::parallel_for(range, [&](const tbb::blocked_range<size_t> &r) {
65 for (size_t i = r.begin(); i < r.end(); ++i) {
66 mImpl->mChildren[i]->forEachPrimitive(visitor, parallel);
67 }
68 });
69 } else {
70 for (auto& child : mImpl->mChildren) {
71 child->forEachPrimitive(visitor, parallel);
72 }
73 }
74}
