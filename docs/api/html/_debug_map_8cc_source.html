89void
90DebugMap::update()
91{
92 if (hasChanged(attrDebugMapType) ||
93 hasChanged(attrPrimitiveAttributeType) ||
94 hasChanged(attrPrimitiveAttributeName)) {
95 mRequiredAttributes.clear();
96 if (get(attrDebugMapType) == ispc::DebugMapType::PRIMITIVE_ATTRIBUTE) {
97 int type = get(attrPrimitiveAttributeType);
98 const std::string& name = get(attrPrimitiveAttributeName);
99 if (type == ispc::TYPE_FLOAT) {
100 mData.mPrimitiveAttributeType = ispc::TYPE_FLOAT;
101 moonray::shading::TypedAttributeKey<float> attributeKey(name);
102 mData.mPrimitiveAttributeIndex = attributeKey;
103 mRequiredAttributes.push_back(attributeKey);
104 } else if (type == ispc::TYPE_VEC2F) {
105 mData.mPrimitiveAttributeType = ispc::TYPE_VEC2F;
106 moonray::shading::TypedAttributeKey<Vec2f> attributeKey(name);
107 mData.mPrimitiveAttributeIndex = attributeKey;
108 mRequiredAttributes.push_back(attributeKey);
109 } else if (type == ispc::TYPE_VEC3F) {
110 mData.mPrimitiveAttributeType = ispc::TYPE_VEC3F;
111 moonray::shading::TypedAttributeKey<Vec3f> attributeKey(name);
112 mData.mPrimitiveAttributeIndex = attributeKey;
113 mRequiredAttributes.push_back(attributeKey);
114 } else if (type == ispc::TYPE_RGB) {
115 mData.mPrimitiveAttributeType = ispc::TYPE_RGB;
116 moonray::shading::TypedAttributeKey<Color> attributeKey(name);
117 mData.mPrimitiveAttributeIndex = attributeKey;
118 mRequiredAttributes.push_back(attributeKey);
119 } else {
120 MNRY_ASSERT(false, "unsupported primitive attribute type");
121 }
122 }
123 }

126void
127DebugMap::sample(const scene_rdl2::rdl2::Map* self, moonray::shading::TLState *tls,
128 const moonray::shading::State& state, Color* sample)
129{
130 const DebugMap* me = static_cast<const DebugMap*>(self);
131
132 if (me->get(attrDebugMapType) == ispc::DebugMapType::N) {
133 sample3D(moonray::shading::evalNormal(me, attrInputNormal, attrInputNormalDial, attrInputNormalSpace, tls, state),
134 me->get(attrCheckerboard), *sample);
135 } else if (me->get(attrDebugMapType) == ispc::DebugMapType::ST) {
136 sample2D(state.getSt(), me->get(attrCheckerboard), *sample);
137 } else if (me->get(attrDebugMapType) == ispc::DebugMapType::P) {
138 sample3D(state.getP(), me->get(attrCheckerboard), *sample);
139 } else if (me->get(attrDebugMapType) == ispc::DebugMapType::NG) {
140 sample3D(state.getNg(), me->get(attrCheckerboard), *sample);
141 } else if (me->get(attrDebugMapType) == ispc::DebugMapType::DPDS) {
142 sample3D(state.getdPds(), me->get(attrCheckerboard), *sample);
143 } else if (me->get(attrDebugMapType) == ispc::DebugMapType::DPDT) {
144 sample3D(state.getdPdt(), me->get(attrCheckerboard), *sample);
145 } else if (me->get(attrDebugMapType) == ispc::DebugMapType::PRIMITIVE_ATTRIBUTE) {
146 int key = me->mData.mPrimitiveAttributeIndex;
147 int type = me->mData.mPrimitiveAttributeType;
148 if (type == ispc::TYPE_FLOAT) {
149 float f = state.getAttribute(moonray::shading::TypedAttributeKey<float>(key));
150 *sample = Color(f);
151 } else if (type == ispc::TYPE_VEC2F) {
152 const Vec2f& v2 = state.getAttribute(moonray::shading::TypedAttributeKey<Vec2f>(key));
153 sample2D(v2, me->get(attrCheckerboard), *sample);
154 } else if (type == ispc::TYPE_VEC3F) {
155 const Vec3f& v3 = state.getAttribute(moonray::shading::TypedAttributeKey<Vec3f>(key));
156 sample3D(v3, me->get(attrCheckerboard), *sample);
157 } else if (type == ispc::TYPE_RGB) {
158 const Color& c = state.getAttribute(moonray::shading::TypedAttributeKey<Color>(key));
159 sample3D(Vec3f(c.r, c.g, c.b), me->get(attrCheckerboard), *sample);
160 }
161 } else {
162 sample->r = 0.f;
163 sample->g = 0.f;
164 sample->b = 0.f;
165 return;
166 }
167}
