32unsigned int
33selectIndexFromCdf(const float* cdfs, unsigned int count, const float r, float &remapped_r)
34{
35 const float *ptr = std::upper_bound(cdfs, cdfs+count, r);
36 unsigned int index = ptr - cdfs;
37 index = scene_rdl2::math::min(index, count - 1);
38 float cdfRange = cdfs[index];
39 if (index >= 1) {
40 cdfRange = cdfs[index] - cdfs[index-1];
41 remapped_r = (r-cdfs[index-1]) * scene_rdl2::math::rcp(cdfRange);
42 } else {
43 remapped_r = r * scene_rdl2::math::rcp(cdfRange);
44 }
45 return index;
46}

59// algorithm for rendering mirror-like flakes. In ACM SIGGRAPH
60// 2016 Talks, ACM, New York, NY, USA, SIGGRAPH ’16, 67:1–

86 // BsdfLobe API
87 finline scene_rdl2::math::Color eval(const BsdfSlice &slice, const scene_rdl2::math::Vec3f &wi, float *pdf = NULL) const override
88 {
89 // The bsdf from [2016] is not reciprocal (evaluating reflection cache based on Wo)
90 // Using an additional reflection cache computed using Wi and
91 // averaging the D term in the bsdf is a cheat we use to make
92 // this bsdf reciprocal.
93 FlakeCaches evalCachesWo, evalCachesWi;
94 bool validCache = computeReflectionCache(slice.getWo(), evalCachesWo, false);
95 validCache &= computeReflectionCache(wi, evalCachesWi, false);
96
97 if (!validCache) {
98 if (pdf) *pdf = 0.0f;
99 return scene_rdl2::math::sBlack;
100 }
101
102 return computeBrdf(slice, wi, evalCachesWo, evalCachesWi, pdf);
103 }

105 finline scene_rdl2::math::Color sample(const BsdfSlice &slice, float r1, float r2,
106 scene_rdl2::math::Vec3f &wi, float &pdf) const override
107 {
108 FlakeCaches sampleCachesWo;
109
110 // generate reflection cache and weights with weighted CDF
111 bool validCache = computeReflectionCache(slice.getWo(), sampleCachesWo, true);
112
113 if (!validCache) {
114 pdf = 0.0f;
115 return scene_rdl2::math::sBlack;
116 }
117#if UNIFORM_SAMPLING
118 scene_rdl2::math::Vec3f wiLocal = sampleLocalHemisphereUniform(r1, r2);
119 wi = mFrame.localToGlobal(wiLocal);
120
121#else
122 if (sampleCachesWo.size == 0) {
123 scene_rdl2::math::Vec3f wiLocal = sampleLocalHemisphereUniform(r1, r2);
124 wi = mFrame.localToGlobal(wiLocal);
125 } else {
126 float remapped_r1 = r1;
127 unsigned int reflIdx = selectIndexFromCdf(sampleCachesWo.wCDF, sampleCachesWo.size, r1, remapped_r1);
128
129 scene_rdl2::math::Vec3f reflSample = sampleCachesWo.reflections[reflIdx];
130
131 // Sample the roughness of the individual flake
132 // generate local wi using cos(gamma) as cut off
133 scene_rdl2::math::Vec3f wiLocal = sampleLocalSphericalCapUniform(remapped_r1, r2, mCosGamma);
134
135 // transform local wi to reflection sample reference frame
136 scene_rdl2::math::ReferenceFrame reflReferenceFrame(reflSample);
137 wi = reflReferenceFrame.localToGlobal(wiLocal);
138 }
139#endif
140
141 // Same as in eval() - we use an additional reflection cache created
142 // using Wi to enforce reciprocity in bsdf computation
143 FlakeCaches sampleCachesWi;
144 validCache = computeReflectionCache(wi, sampleCachesWi, false);
145
146 if (!validCache) {
147 pdf = 0.0f;
148 return scene_rdl2::math::sBlack;
149 }
150
151 return computeBrdf(slice, wi, sampleCachesWo, sampleCachesWi, &pdf);
152 }

173 {
174 bool handled = true;
175
176 switch (property) {
177 case PROPERTY_ROUGHNESS:
178 {
179 *dest = mFlakeRandomness;
180 *(dest + 1) = mFlakeRandomness;
181 }
182 break;
183 case PROPERTY_NORMAL:
184 {
185 const scene_rdl2::math::Vec3f &N = mFrame.getN();
186 *dest = N.x;
187 *(dest + 1) = N.y;
188 *(dest + 2) = N.z;
189 }
190 break;
191 case PROPERTY_PBR_VALIDITY:
192 {
193 const Fresnel* fresnel = getFresnel();
194 // Get the property from the fresnel object
195 scene_rdl2::math::Color res = scene_rdl2::math::sBlack;
196 if (fresnel) {
197 res = fresnel->computePbrValidity();
198 }
199 *dest = res.r;
200 *(dest + 1) = res.g;
201 *(dest + 2) = res.b;
202 }
203 break;
204 default:
205 handled = BsdfLobe::getProperty(property, dest);
206 }

232 struct FlakeCaches {
233 scene_rdl2::math::Vec3f reflections[sMaxFlakes];
234 float weights[sMaxFlakes];
235 scene_rdl2::math::Color scales[sMaxFlakes];
236 float wCDF[sMaxFlakes];

241 size_t mFlakeCount;
242 const scene_rdl2::math::Vec3f* mFlakeNormals; // pointer to array containing normals and colors

245 // Compute the D term as explained in "Reflection Cache" section of [2016]
246 // This is refactored into a function since we compute D twice for the bsdf
247 // calculation, once with Wo and again with Wi and use the average
248 finline float computeD(const FlakeCaches& caches, const scene_rdl2::math::Vec3f& w,
249 scene_rdl2::math::Color& weightedColor,
250 float* reflectionWeight = nullptr,
251 float* totalWeights = nullptr) const
252 {
253 if (totalWeights != nullptr) {
254 *totalWeights = 0.0f;
255 }
256 float reflectWt = 0.0f;
257 weightedColor = scene_rdl2::math::sBlack;
258
259 for(unsigned int i = 0; i < caches.size; ++i) {
260 // Formulating the pdf and brdf weights calculation according to Chaos Group paper
261 float wDotRj = dot(w, caches.reflections[i]);
262 if(wDotRj >= mCosGamma) {
263 reflectWt += caches.weights[i];
264 weightedColor += caches.scales[i] * caches.weights[i];
265 }
266 if (totalWeights != nullptr) {
267 *totalWeights += caches.weights[i];
268 }
269 }
270
271 if (reflectionWeight != nullptr) {
272 *reflectionWeight = reflectWt; // k for the pdf in [2016]
273 }
274 if (!scene_rdl2::math::isZero(reflectWt)) {
275 weightedColor /= reflectWt; // normalize the weighted color
276 }
277 return reflectWt / static_cast<float>(caches.size); // Return discrete D
278 }

280 /// Supporting private member functions
281 finline scene_rdl2::math::Color computeBrdf(
282 const BsdfSlice& slice, const scene_rdl2::math::Vec3f& wi,
283 const FlakeCaches& cachesWo, const FlakeCaches& cachesWi, float *pdf = NULL) const
284 {
285 // calculate pdf as
286 // Sum(weight[i], such that dot(wi,reflection[i]) < cos(gamma)) / pi*(1-cos(gamma))*sum(all weights)
287 //
288 // calculate return color value as
289 // sum(scale[i]*weight[i], such that dot(wi,reflection[i]) < cos(gamma)) / sum(all weights)
290
291 // prepare for possible early return
292 if(pdf != NULL) {
293 *pdf = 0.0f;
294 }
295
296 scene_rdl2::math::Vec3f m;
297 if (!computeNormalizedHalfVector(slice.getWo(), wi, m)) {
298 return scene_rdl2::math::sBlack;
299 }
300
301 float kIntersect = 0.0f; // k in [2016] - "optimal importance sampling"
302 float kTotalSum = 0.0f; // c_n in [2016] - "optimal importance sampling"
303 scene_rdl2::math::Color weightedColor1 = scene_rdl2::math::sBlack, weightedColor2 = scene_rdl2::math::sBlack;
304
305 // **Cheat** for reciprocity. The D term evaluates how many of the flakes
306 // reflect light from Wo into a cone around Wi. We however, also compute D
307 // by flipping Wo and Wi and averaging the two results.
308 // kIntersect and kTotalSum are only used for evaluating the pdf,
309 // so we only accumulate them when we compute D as per the talk
310 // (we retain the importance sampling scheme and thus the pdf from the talk)
311 float discreteD = 0.5f * (computeD(cachesWo, wi, weightedColor1, &kIntersect, &kTotalSum) +
312 computeD(cachesWi, slice.getWo(), weightedColor2, nullptr, nullptr));
313
314 const float cosNI = std::min(dot(mFrame.getN(), wi), scene_rdl2::math::sOneMinusEpsilon);
315 if (cosNI <= scene_rdl2::math::sEpsilon) return scene_rdl2::math::sBlack;
316
317 const float cosNO = std::min(dot(mFrame.getN(), slice.getWo()), scene_rdl2::math::sOneMinusEpsilon);
318 if (cosNO <= scene_rdl2::math::sEpsilon) return scene_rdl2::math::sBlack;
319
320 const float cosMI = dot(m, wi);
321 if (cosMI <= scene_rdl2::math::sEpsilon) return scene_rdl2::math::sBlack;
322
323 // Computing D as described in the Chaos Group paper
324 //
325 // D is broken up into two parts. D~ and factor 4/(a(A)*sigma(SolidAngle))
326 //
327 // D = (D~)*4/(a(A)*sigma(SolidAngle))
328 //
329 // a(A) is the area of the pixel footprint
330 // sigma(SolidAngle) is the solid angle area
331 //
332 // and
333 // D~ = (1/N)*(Sum of all weights contributing to the reflection direction)
334 //
335 // Sum of all weights = kIntersect
336 // N = number of flakes per unit area, i.e. it is constant
337 // See Jakob, Haan, et al "Discrete Stochastic Micofacet Models" for a clear description of N
338 // since it is not clear in the Chaos group paper and can be confused with the number of flakes
339 // *within* the footprint
340 //
341 // So then
342 //
343 // D = 4/(a(A)*sigma(SolidAngle))*(1/N)*kIntersect
344 // Re-organizing, moving N into the denominator...
345 // D = 4/(N*a(A)*sigma(SolidAngle))*kIntersect
346 //
347 // Note the factor N*a(A) = E(n), which is the expected value for the number of flakes
348 // E(n) is then the number of flakes within the pixel footprint, mFlakeCount
349 //
350 // So D = (4/(mFlakeCount*SolidAngleArea))*kIntersect
351 //
352 // Since the CookTorrance model divides by 4, the factor of 4 is left out in
353 // both parts since they cancel.
354
355 const float solidAngleArea = scene_rdl2::math::sTwoPi * (1.0 - mCosGamma); // must be 2*pi, the Chaos paper has only pi and is incorrect.
356
357 const float D = discreteD * scene_rdl2::math::rcp(solidAngleArea);
358
359 scene_rdl2::math::Color F = computeScaleAndFresnel(cosMI);
360
361 // compute G for CookTorrance
362 const float G = computeGeometryTerm(cosNO, cosNI);
363
364 const float invBrdfDenom = scene_rdl2::math::rcp( cosNO * (slice.getIncludeCosineTerm() ? 1.0f : cosNI) );
365 scene_rdl2::math::Color flakesBrdf = F*D*G*invBrdfDenom;
366
367 if(pdf != NULL) {
368#if UNIFORM_SAMPLING
369 *pdf = scene_rdl2::math::sOneOverTwoPi;
370#else
371 *pdf = kIntersect * scene_rdl2::math::rcp(solidAngleArea * kTotalSum);
372#endif
373 }
374 // Create a conductor fresnel object using the weighted (reflection cache weights)
375 // color of the flakes that contribute to the final bsdf
376 // **Note: This is a special case use of the fresnel object inside the bsdf lobe itself -
377 // since we know the color to use only after evaluating the reflection cache
378 scene_rdl2::math::Color avgWeightedColor = scene_rdl2::math::min(
379 scene_rdl2::math::Color(0.999f), 0.5f * (weightedColor1+weightedColor2));
380 scene_rdl2::math::Color eta = computeEta(avgWeightedColor, avgWeightedColor);
381 scene_rdl2::math::Color k = computeK(avgWeightedColor, eta);
382 ConductorFresnel fresnel(eta, k);
383 scene_rdl2::math::Color result = flakesBrdf * fresnel.eval(cosMI);
384 return result;
385 }

387 finline bool computeReflectionCache(const scene_rdl2::math::Vec3f& w, FlakeCaches& caches,
388 bool buildCDF) const
389 {
390 scene_rdl2::math::Vec3f reflection;
391 float weight;
392
393 unsigned int k = 0;
394 float totalWeight = 0;
395 for(unsigned int i = 0; i < mFlakeCount; ++i) {
396 weight = scene_rdl2::math::max(0.0f, computeReflectionDirection(mFlakeNormals[i], w, reflection));
397 caches.weights[k] = weight;
398 caches.reflections[k] = reflection;
399 caches.scales[k] = mFlakeColors[i];
400 caches.flakeNormals[k] = mFlakeNormals[i];
401 totalWeight += weight;
402 k++;
403 }
404
405 if (totalWeight > 0) {
406 totalWeight = scene_rdl2::math::rcp(totalWeight);
407 } else {
408 return false;
409 }
410
411 if (buildCDF) {
412 float cdf = 0.0f;
413 for (unsigned int i = 0; i < k; i++) {
414 cdf += caches.weights[i] * totalWeight;
415 caches.wCDF[i] = cdf;
416 }
417 }
418
419 caches.size = k;
420 return true;
421 }

423 finline float
424 computeGeometryTerm(const float cosNO, const float cosNI) const
425 {
426 const float invCosNO = 1.0f / cosNO;
427 const float invCosNI = 1.0f / cosNI;
428 const float ao = 1.0f / (mFlakeRandomness * scene_rdl2::math::sqrt((1.0f - cosNO * cosNO) *
429 (invCosNO * invCosNO)));
430 const float ai = 1.0f / (mFlakeRandomness * scene_rdl2::math::sqrt((1.0f - cosNI * cosNI) *
431 (invCosNI * invCosNI)));
432 const float G1o = (ao < 1.6f) ? (3.535f * ao + 2.181f * ao * ao) /
433 (1.0f + 2.276f * ao + 2.577f * ao * ao) : 1.0f;
434 const float G1i = (ai < 1.6f) ? (3.535f * ai + 2.181f * ai * ai) /
435 (1.0f + 2.276f * ai + 2.577f * ai * ai) : 1.0f;
436 return G1o * G1i;
437 }

439 // ** Note: The below functions have been duplicated from
440 // ShaderComplexIor class in lib/shaders/shading_util/Ior.h
441 // and we cannot make the StochasticFlakesBsdfLobe class dependent on a
442 // class in lib/shaders/shading_util.
443 // Also, they cannot be moved to lib/rendering/pbr because of their dependency
444 // on lib/rendering/shading (becomes cyclic).

448 // Functions for computing complex IOR values for conductor Fresnel
449 // from 'reflectivity' and 'edge tint' colors.
450 // See paper: "Artist Friendly Metallic Fresnel", by Ole Gulbrandsen
451 // from Framestore, published at JCGT in 2014 (http://jcgt.org)
452 finline static scene_rdl2::math::Color
453 nMin(const scene_rdl2::math::Color &r)
454 {
455 return (scene_rdl2::math::sWhite - r) / (scene_rdl2::math::sWhite + r);
456 }
457
458 finline static scene_rdl2::math::Color
459 nMax(const scene_rdl2::math::Color &r)
460 {
461 scene_rdl2::math::Color rSqrt = sqrt(r);
462 return (scene_rdl2::math::sWhite + rSqrt) / (scene_rdl2::math::sWhite - rSqrt);
463 }
464
465 finline static scene_rdl2::math::Color
466 computeEta(const scene_rdl2::math::Color &r, const scene_rdl2::math::Color &g)
467 {
468 return g * nMin(r) + (scene_rdl2::math::sWhite - g) * nMax(r);
469 }
470
471 finline static scene_rdl2::math::Color
472 computeK(const scene_rdl2::math::Color &r, const scene_rdl2::math::Color &n)
473 {
474 const scene_rdl2::math::Color a = n + scene_rdl2::math::sWhite;
475 const scene_rdl2::math::Color b = n - scene_rdl2::math::sWhite;
476 // Take an abs() here to get rid of any numerical -0 etc
477 const scene_rdl2::math::Color nr = max(scene_rdl2::math::sBlack, r * a * a - b * b);
478 return sqrt(nr / (scene_rdl2::math::sWhite - r));
479 }
