76bool
77RodLightFilter::updateTransforms(const Mat4f &local2Render, int ti)
78{
79 // update transforms at time index (ti)
80 // ti == 0 is at normalized rayTime = 0.f
81 // ti == 1 is at normalized rayTime = 1.f;
82 MNRY_ASSERT(mRdlLightFilter);
83
84
85 Quaternion3f rot;
86 Mat3f scale;
87 Xform3f local2RenderXform = xform<Xform3f>(local2Render);
88 decompose(local2RenderXform.l, scale, rot);
89
90 mPosition[ti] = asVec3(local2Render.row3());
91
92 mLocal2RenderScale[ti] = scale;
93
94 // Single shutter-time items (for when there is no MB)
95 if (ti == 0) {
96 mRender2Local0 = local2RenderXform.inverse();
97 mRender2LocalRot0 = Mat3f(rot).transposed();
98 mRender2LocalScale0 = scale.inverse();
99 }
100
101 Xform3f local2RenderRot = Xform3f(Mat3f(rot), zero);
102 mOrientation[ti] = normalize(
103 scene_rdl2::math::Quaternion3f(local2RenderRot.row0(),
104 local2RenderRot.row1(),
105 local2RenderRot.row2()));
106
107 // After setting the second quaternion (ti==1), check if the angle from the
108 // first quaternion is negative and flip accordingly.
109 if (ti==1 && dot(mOrientation[0], mOrientation[1]) < 0) {
110 mOrientation[1] *= -1.f;
111 }
112
113 return true;
114}

116bool
117RodLightFilter::updateParamAndTransforms(const Mat4f &local2Render0,
118 const Mat4f &local2Render1)
119{
120 if (!updateTransforms(local2Render0, /* ti = */ 0)) return false;
121
122 // setup mMb
123 mMb = LIGHTFILTER_MB_NONE;
124 const scene_rdl2::rdl2::SceneVariables &vars =
125 mRdlLightFilter->getSceneClass().getSceneContext()->getSceneVariables();
126 const bool mb = vars.get(scene_rdl2::rdl2::SceneVariables::sEnableMotionBlur) &&
127 !isEqual(local2Render0, local2Render1);
128
129 if (mb) {
130 if (!updateTransforms(local2Render1, /* ti = */ 1)) return false;
131
132 // set specific mb bits
133 if (!isEqual(mPosition[0], mPosition[1]))
134 mMb |= LIGHTFILTER_MB_TRANSLATION;
135 if (!isEqual(mOrientation[0], mOrientation[1]))
136 mMb |= LIGHTFILTER_MB_ROTATION;
137 if (!isEqual(mLocal2RenderScale[0], mLocal2RenderScale[1]))
138 mMb |= LIGHTFILTER_MB_SCALE;
139 }
140
141 return true;
142}

144void
145RodLightFilter::update(const LightFilterMap& /*lightFilters*/,
146 const Mat4d& world2render)
147{
148 if (!mRdlLightFilter) {
149 return;
150 }
151
152 const Mat4d l2w0 = mRdlLightFilter->get<scene_rdl2::rdl2::Mat4d>(sNodeXformKey, 0.0f);
153 const Mat4d l2w1 = mRdlLightFilter->get<scene_rdl2::rdl2::Mat4d>(sNodeXformKey, 1.0f);
154
155 const Mat4f local2Render0 = toFloat(l2w0 * world2render);
156 const Mat4f local2Render1 = toFloat(l2w1 * world2render);
157
158 if (!updateParamAndTransforms(local2Render0, local2Render1)) {
159 return;
160 }
161
162 mWidth = mRdlLightFilter->get<scene_rdl2::rdl2::Float>(sWidthKey);
163 mDepth = mRdlLightFilter->get<scene_rdl2::rdl2::Float>(sDepthKey);
164 mHeight = mRdlLightFilter->get<scene_rdl2::rdl2::Float>(sHeightKey);
165 mRadius = mRdlLightFilter->get<scene_rdl2::rdl2::Float>(sRadiusKey);
166 mEdge = mRdlLightFilter->get<scene_rdl2::rdl2::Float>(sEdgeKey);
167 mEdgeInv = mEdge ? 1.f/mEdge : 1.f;
168 mColor = mRdlLightFilter->get<scene_rdl2::rdl2::Rgb>(sColorKey);
169 mIntensity = mRdlLightFilter->get<scene_rdl2::rdl2::Float>(sIntensityKey);
170 mColor *= mIntensity;
171 mDensity = mRdlLightFilter->get<scene_rdl2::rdl2::Float>(sDensityKey);
172 mInvert = mRdlLightFilter->get<scene_rdl2::rdl2::Bool>(sInvertKey);
173 mBoxCorner = abs(Vec3f(mWidth, mHeight, mDepth)*0.5f);
174 mRender2LocalRotAndScale = mRender2LocalRot0.l * mRender2LocalScale0;
175 mRadiusEdgeSum = mRadius+mEdge;
176
177 std::vector<float> inDistancesVec = mRdlLightFilter->get<rdl2::FloatVector>(sRampInKey);
178 std::vector<float> outDistancesVec = mRdlLightFilter->get<rdl2::FloatVector>(sRampOutKey);
179 std::vector<int> interpolationTypesVec = mRdlLightFilter->get<rdl2::IntVector>(sRampInterpolationTypesKey);
180
181 if (inDistancesVec.size() != outDistancesVec.size() || inDistancesVec.size() != interpolationTypesVec.size()) {
182 mRdlLightFilter->error(
183 "Rod ramp light filter ramp_in_distances, ramp_out_distances and ramp_interpolation_types are different sizes, using defaults");
184 outDistancesVec = {0.f, 1.f};
185 inDistancesVec = {0.f, 1.f};
186 interpolationTypesVec = {ispc::RAMP_INTERPOLATOR_MODE_MONOTONECUBIC, ispc::RAMP_INTERPOLATOR_MODE_MONOTONECUBIC};
187 }
188
189 mRamp.init(
190 inDistancesVec.size(),
191 inDistancesVec.data(),
192 outDistancesVec.data(),
193 reinterpret_cast<const ispc::RampInterpolatorMode*>(interpolationTypesVec.data()));
194}

196Vec3f
197RodLightFilter::slerpPointRender2Local(const Vec3f &p, float time) const
198{
199 // inverse of Local2Render is (p + t) * r * s
200 // where t = -position, r = render2localRot.l, s = render2localScale
201 const Vec3f t = (mMb & LIGHTFILTER_MB_TRANSLATION) ?
202 -1.f * lerp(mPosition[0], mPosition[1], time) :
203 -1.f * mPosition[0];
204 const Mat3f r = (mMb & LIGHTFILTER_MB_ROTATION)?
205 // transpose is inverse
206 Mat3f(slerp(mOrientation[0], mOrientation[1], time)).transposed() :
207 mRender2LocalRot0.l;
208 const Mat3f s = (mMb & LIGHTFILTER_MB_SCALE) ?
209 // The inverse() call below, while unfortunate, is necessary.
210 // Reversing the order of operations (and lerping precomputed inverses)
211 // does not look correct.
212 lerp(mLocal2RenderScale[0], mLocal2RenderScale[1], time).inverse() :
213 mRender2LocalScale0;
214
215 return ((p + t) * r) * s;
216}

218// This function is for quickly determining if a filter can be skipped.
219// It returns true if the sphere at point p with radius rad is completely
220// outside the region of influence of a filter. Since filters can
221// be inverted, this means the sphere is either out of the volume, or
222// if the filter is inverted, entirely within the interior of the volume.
223// It accounts for the radius and edge zones.
224//
225// Limitation: Currently this function can only quickly reject if the filter is
226// not moving. If the filter is moving (has motion blur, i.e. mMb is true)
227// this function does not help. Camera motion blur is fine, however.
228//
229// TODO: Add support for motion blur
230//
231// I have not profiled this code. Currently the scalar version incurs a
232// significant slow-down with many (43) RLF filters, compared to vector. The
233// reason for the difference is unknown. However, each filter incurs an
234// overhead for each light sample, as it must do a bounding box check in this
235// function (for static filters) and a very-slow full-quality check (using
236// slerpPointRender2Local and signedDistanceRoundBox) for moving filters. We
237// should be smarter about knowing when a filter applies, using something like
238// a BVH.
239//
240// TODO: Profiling and performance improvements. Possibly use a BVH.
241//
242bool
243RodLightFilter::isOutsideInfluence(const Vec3f &p, float rad) const
244{
245 if (mMb) return false;
246
247 Vec3f localP = abs((p - mPosition[0]) * mRender2LocalRotAndScale);
248 float buf = mRadiusEdgeSum + rad;
249
250 if (mInvert) {
251 buf = -buf;
252 }
253
254 bool out = false;
255 if (localP.x > mBoxCorner.x + buf ||
256 localP.y > mBoxCorner.y + buf ||
257 localP.z > mBoxCorner.z + buf) {
258 out = true;
259 }
260
261 if (mInvert) {
262 out = !out;
263 }
264
265 return out;
266}

268// Signed distance to a round box.
269//
270// Original implementation (for reference) taken from:
271// https://www.iquilezles.org/www/articles/distfunctions/distfunctions.htm
272//
273// Provides exact distance to boundary (rather than conservative minimum)
274// float sdRoundBox( vec3 p, vec3 b, float r )
275// {
276// vec3 q = abs(p) - b;
277// return length(max(q,0.0)) + min(max(q.x,max(q.y,q.z)),0.0) - r;
278// }
279float RodLightFilter::signedDistanceRoundBox( const Vec3f &p ) const
280{
281 Vec3f q = abs(p) - mBoxCorner;
282 return length(max(q, Vec3f(0.f))) + min(reduce_max(q), 0.f) - mRadius;
283}

285// TODO: Replace signedDistanceRoundBox with the function below
286// that avoids a sqrt(). In my small test with 43 filters
287// it reduced render time 1.5%. However, it is less readable.
288// Perhaps that could be improved.
289//
290//
291// Optimized version of signedDistanceRoundBox(). Not in use!
292//
293// Using this version of the signedDistanceRoundBox function lets us avoid a
294// sqrt in canIlluminate(). It returns a squared part, and a regular part.
295//
296// For reference, canIlluminate() does a test like:
297// r = signedDistanceRoundBox();
298// ...
299// if (r < -data.shadingPointRadius)
300// ... ^^^^^^^^^^^^^^^^^^^^^^^^ call this "a"
301//
302// r becomes sqrt(sqrPart)+regPart
303// r < a becomes sqrt(sqrPart)+regPart < a
304// sqrt(sqrPart) < a-regPart
305// sqrPart < (a-regPart)^2
306//
307// To use (untested):
308//
309// float sqrPart, regPart;
310// signedDistanceRoundBoxSqrReg(localPoint, sqrPart, regPart);
311//
312// if (sqrPart < (a-regPart)*(a-regPart))
313// ...
314//
315// void RodLightFilter::signedDistanceRoundBoxSqrReg(
316// const Vec3f &p, float &sqrPart, float &regPart ) const
317// {
318// Vec3f q = abs(p) - mBoxCorner;
319// sqrPart = lengthSqr(max(q, Vec3f(0.f)));
320// regPart = min(reduce_max(q), 0.f) - mRadius;
321// }

323bool
324RodLightFilter::canIlluminate(const CanIlluminateData& data) const
325{
326 // Approximate light and shading point as two spheres. We can compute the
327 // minimum and maximum distances between any 2 points in those two spheres.
328
329 // If the filter is less than full effect, it cannot
330 // completely block light.
331 if (mDensity < 1.f) {
332 return true;
333 }
334
335 if (isOutsideInfluence(data.shadingPointPosition,
336 data.shadingPointRadius)) {
337 return true;
338 }
339
340 Vec3f localPoint = xformPointRender2Local(data.shadingPointPosition,
341 data.time);
342
343 float r = signedDistanceRoundBox(localPoint);
344
345 if (mInvert) {
346 // Everything outside the volume is unlit
347 if (isBlack(mColor)) {
348 if (r > (data.shadingPointRadius + mEdge)) {
349 //if (sqrPart > sqr(data.shadingPointRadius + mEdge - regPart))
350 return false;
351 }
352 }
353 } else {
354 // The only place that is completely unlit
355 // is within the volume and if mColor = 0
356 if (isBlack(mColor)) {
357 if (r < -data.shadingPointRadius) {
358 //if (sqrPart < sqr(-data.shadingPointRadius - regPart))
359 return false;
360 }
361 }
362 }
363
364 return true;
365}

367Color
368RodLightFilter::eval(const EvalData& data) const
369{
370 if (isOutsideInfluence(data.shadingPointPosition, 0)) {
371 return Color(1.f);
372 }
373
374 Vec3f localPoint = xformPointRender2Local(data.shadingPointPosition, data.time);
375
376 float r = signedDistanceRoundBox(localPoint);
377
378 float scale = 1.f;
379 if (r <= 0.f) {
380 scale = 0.f;
381 } else if (r < mEdge) {
382 r *= mEdgeInv;
383 scale = mRamp.eval1D(r);
384 }
385
386 if (mInvert) {
387 scale = 1.f - scale;
388 }
389
390 // apply density (lerp)
391 scale = scene_rdl2::math::lerp(1.f, scale, mDensity);
392
393 Color result = scene_rdl2::math::lerp(mColor, Color(1.f), scale);
394
395 return result;
396}
