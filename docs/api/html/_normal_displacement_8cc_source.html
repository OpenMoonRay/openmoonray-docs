57void NormalDisplacement::displace(const Displacement *self, moonray::shading::TLState *tls,
58 const State &state, Vec3f *displace)
59{
60 const NormalDisplacement* me = static_cast<const NormalDisplacement*>(self);
61
62 float zeroValue = me->get(attrZeroValue);
63 float height = evalFloatWithPreAdd(me, attrHeight, tls, state, -zeroValue);
64 float heightMult = evalFloat(me, attrHeightMultiplier, tls, state);
65
66 *displace = height * heightMult * state.getN();
67}
