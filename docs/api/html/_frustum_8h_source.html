16// Some camera types may implement a frustum, i.e. orthographic and perspective
17// cameras. A frustum is defined by right, left, bottom, top, far, and near
18// clipping planes, plus the viewport in raster space and the camera to screen xform.
19struct Frustum
20{
21 // Performs a simple test to check if a BBox overlaps the frustum. This test
22 // may produce false positives, but may NOT produce false negatives.
23 bool testBBoxOverlaps(const scene_rdl2::math::BBox3f& bbox) const;
24
25 // Projects a camera-space 3D point to a 3D point on the z=-1 plane, where
26 // the result's x, y coordinates are the pixel's raster coordinates.
27 scene_rdl2::math::Vec3f projectToViewport(const scene_rdl2::math::Vec3f& p) const;
28
29 // Computes a 6-bit Cohen-Sutherland outcode value where each bit is the
30 // result of a point/plane half-space test with one of the clip planes.
31 // The bit is set if the point is behind the plane (on the side opposite the
32 // normal defined by the plane equation coefficients).
33 // A point on the plane is classified as 'inside'.
34 // Bit i corresponds to mClipPlanes[i].
35 int computeOutcode(const scene_rdl2::math::Vec3f& p) const;
36
37 // Clip the convex planar poly defined by (xyzIn, numIn) to one plane of the frustum,
38 // outputting results in xyzOut. Returns the number of points in the clipped poly.
39 // The output array xyzOut must be big enough to hold numIn+1 points.
40 int clipPolyToPlane(scene_rdl2::math::Vec3f* xyzOut, const scene_rdl2::math::Vec3f* xyzIn, int numIn, int planeIdx) const;
41
42 // Clip the convex planar poly defined by (xyzIn, stIn, numIn) to one plane of the frustum,
43 // outputting results in (xyzOut, stOut). Returns the number of points in the clipped poly.
44 // The output arrays (xyzOut, stOut) must be big enough to hold numIn+1 points.
45 int clipPolyToPlane(scene_rdl2::math::Vec3f* xyzOut, scene_rdl2::math::Vec2f* stOut,
46 const scene_rdl2::math::Vec3f* xyzIn, const scene_rdl2::math::Vec2f* stIn, int numIn, int planeIdx) const;
47
48 // Clip the convex planar poly defined by (xyzIn, numIn) to the planes of the frustum specified by
49 // whichPlanes, outputting results in xyzOut. Returns the number of points in the clipped poly.
50 // The output array xyzOut must be big enough to hold numIn+6 points.
51 // If the whichPlanes mask is omitted, the function will clip to all 6 planes.
52 // If the clipping operation is preceded by outcode testing, and polygons whose AND-of-outcodes have
53 // been trivially rejected, then the OR-of-outcodes can be supplied as the whichPlanes value, assuming
54 // outcodes have been generated using Frustum::computeOutcode().
55 // If supplied, whichPlanes must be non-zero (otherwise why bother clippng?)
56 int clipPoly(scene_rdl2::math::Vec3f* xyzOut, const scene_rdl2::math::Vec3f* xyzIn, int numIn, int whichPlanes = 0x3F) const;
57
58 // Clip the convex planar poly defined by (xyzIn, stIn, numIn) to the planes of the frustum specified by
59 // whichPlanes, outputting results in (xyzOut, stOut). Returns the number of points in the clipped poly.
60 // The output arrays (xyzOut, stOut) must be big enough to hold numIn+6 points.
61 // If the whichPlanes mask is omitted, the function will clip to all 6 planes.
62 // If the clipping operation is preceded by outcode testing, and polygons whose AND-of-outcodes have
63 // been trivially rejected, then the OR-of-outcodes can be supplied as the whichPlanes value, assuming
64 // outcodes have been generated using Frustum::computeOutcode().
65 // If supplied, whichPlanes must be non-zero (otherwise why bother clippng?)
66 int clipPoly(scene_rdl2::math::Vec3f* xyzOut, scene_rdl2::math::Vec2f* stOut,
67 const scene_rdl2::math::Vec3f* xyzIn, const scene_rdl2::math::Vec2f* stIn, int numIn, int whichPlanes = 0x3F) const;
68
69 // Camera space to screen space xform
70 scene_rdl2::math::Mat4f mC2S;
71
72 // 6 planes with 4 coefficients each, i.e. Ax + By + Cz + D = 0;
73 // Ordering: right, left, bottom, top, far, near
74 float mClipPlanes[6][4];
75
76 // Viewport in raster space (pixels)
77 std::array<int, 4> mViewport;
78};
