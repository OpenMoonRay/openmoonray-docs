144template <utype Dims>
145inline void IntegratorSample<Dims>::getSample(float returnValue[Dims], unsigned depth) const
146{
147 // mTotalSamples gets set based on a constructor overload. In theory, this can
148 // be computed at compile-time, if we want to put that kind of work in to
149 // remove the conditional.
150
151 if (mUsePseudoRandom || depth >= sNumDistributionBounces) {
152 static_assert(Dims <= 4, "These random engines only support up to four values.");
153 moonray::util::StatelessRandomEngine reng(mSequenceID.getHash(0xa511e9b3));
154 const auto result = reng.asFloat(mSampleNumber++);
155 for (utype i = 0; i < Dims; ++i) {
156 returnValue[i] = result[i];
157 }
158 } else if (mTotalSamples != sInvalidTotalSamples) {
159 getSample(returnValue, [=](utype n) {
160 getIntegratorSamples<SIMDFloatArray>(mSequenceID, n, mCache, mTotalSamples);
161 }, depth);
162 } else {
163 getSample(returnValue, [=](utype n) {
164 getIntegratorSamples<SIMDFloatArray>(mSequenceID, n, mCache);
165 }, depth);
166 }
167#if defined(DEBUG)
168 for (utype i = 0; i < Dims; ++i) {
169 MNRY_ASSERT(returnValue[i] >= 0.0f);
170 MNRY_ASSERT(returnValue[i] < 1.0f);
171 }
172#endif
173}

175template <utype Dims>
176template <typename F>
177inline void IntegratorSample<Dims>::getSample(float returnValue[Dims], F f, unsigned depth) const
178{
179 // Ask the cache to fill up, if needed, and return the index into our
180 // array.
181 const utype idx = mCacheIdx.getIndex(f, mSampleNumber++);
182 for (utype i = 0; i < Dims; ++i) {
183 returnValue[i] = mCache[i][idx];
184 }
185}
