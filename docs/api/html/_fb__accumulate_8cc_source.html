69void
70Fb::accumulateRenderOutput(const PartialMergeTilesTbl *partialMergeTilesTbl,
71 const Fb &srcFb)
72// This function is used on progmcrt_merge computation
73{
74 if (!srcFb.getRenderOutputStatus()) return;
75
76 accumulateAllActiveAov(srcFb, [&](const FbAovShPtr &srcFbAov, FbAovShPtr &dstFbAov) {
77 // activeAovFunc
78 if (srcFbAov->getReferenceType() == FbReferenceType::UNDEF) {
79 // Non-Reference type buffer
80 // We have to update fbAov information and accumulate data based on activeTile information
81
82 // need to setup default value before call setup()
83 dstFbAov->setDefaultValue(srcFbAov->getDefaultValue());
84
85 // We always need to process numSampleData on merge computation
86 constexpr bool storeNumSampleData = true;
87
88 dstFbAov->setup(partialMergeTilesTbl,
89 srcFbAov->getFormat(),
90 srcFbAov->getWidth(),
91 srcFbAov->getHeight(), // setup memory and clean if needed
92 storeNumSampleData);
93
94 // setup closestFilter condition
95 dstFbAov->setClosestFilterStatus(srcFbAov->getClosestFilterStatus());
96
97 switch (srcFbAov->getFormat()) {
98 case VariablePixelBuffer::FLOAT :
99 accumulatePartialTiles(partialMergeTilesTbl, [&](int tileId) {
100 accumulateFloat1AovOneTile(dstFbAov, srcFbAov, tileId);
101 });
102 break;
103 case VariablePixelBuffer::FLOAT2 :
104 accumulatePartialTiles(partialMergeTilesTbl, [&](int tileId) {
105 accumulateFloat2AovOneTile(dstFbAov, srcFbAov, tileId);
106 });
107 break;
108 case VariablePixelBuffer::FLOAT3 :
109 accumulatePartialTiles(partialMergeTilesTbl, [&](int tileId) {
110 accumulateFloat3AovOneTile(dstFbAov, srcFbAov, tileId);
111 });
112 break;
113 case VariablePixelBuffer::FLOAT4 :
114 accumulatePartialTiles(partialMergeTilesTbl, [&](int tileId) {
115 accumulateFloat4AovOneTile(dstFbAov, srcFbAov, tileId);
116 });
117 break;
118 default :
119 break;
120 }
121 } else {
122 // Reference type buffer
123 // Just setup fbAov w/ referenceType information. We don't have any actual data
124 // for reference buffer type inside fbAov.
125 dstFbAov->setup(srcFbAov->getReferenceType());
126 }
127 });
128}

130void
131Fb::accumulateAllFbs(const int numMachines,
132 const std::vector<char> &received,
133 const std::vector<grid_util::Fb> &srcFbs)
134//
135// Test function for tile base MT task distribution. Still work in progress.
136// This function is used on progmcrt_merge computation
137//
138{
139 auto bufferSetupFunc = [&](unsigned bufferId, const Fb &src) {
140 switch (bufferId) {
141 case 0 : {
142 if (src.getPixelInfoStatus()) setupPixelInfo(nullptr, src.getPixelInfoName());
143 } break;
144 case 1 : {
145 if (src.getHeatMapStatus()) setupHeatMap(nullptr, src.getHeatMapName());
146 } break;
147 case 2 : {
148 if (src.getWeightBufferStatus()) setupWeightBuffer(nullptr, src.getWeightBufferName());
149 } break;
150 case 3 : {
151 if (src.getRenderBufferOddStatus()) setupRenderBufferOdd(nullptr);
152 } break;
153 case 4 : {
154 if (src.getRenderOutputStatus()) {
155 accumulateAllActiveAov
156 (src,
157 [&](const FbAovShPtr &srcFbAov, FbAovShPtr &dstFbAov) {
158 if (srcFbAov->getReferenceType() == FbReferenceType::UNDEF) {
159 // Non-Reference type buffer
160 // We have to update fbAov information and accumulate data based on
161 // activeTile information
162
163 // We always need to process numSampleData on merge computation
164 constexpr bool storeNumSampleData = true;
165
166 // need to setup default value before call setup()
167 dstFbAov->setDefaultValue(srcFbAov->getDefaultValue());
168 dstFbAov->setup(nullptr,
169 srcFbAov->getFormat(),
170 srcFbAov->getWidth(),
171 srcFbAov->getHeight(), // setup memory and clean if needed
172 storeNumSampleData);
173
174 // setup closestFilter condition
175 dstFbAov->setClosestFilterStatus(srcFbAov->getClosestFilterStatus());
176
177 } else {
178 // Reference type buffer
179 // Just setup fbAov w/ referenceType information.
180 // We don't have any actual data for reference buffer type inside fbAov.
181 dstFbAov->setup(srcFbAov->getReferenceType());
182 }
183 });
184 }
185 } break;
186 }
187 };
188
189 // setup all buffer memory first. This should be done for every machineId once
190 // (not for every tile of each machineId).
191 for (int machineId = 0; machineId < numMachines; ++machineId) {
192 if (!received[machineId]) continue;
193 const Fb &src = srcFbs[machineId];
194
195# ifdef SINGLE_THREAD
196 for (unsigned int bufferId = 0; bufferId < 5; ++bufferId) {
197 bufferSetupFunc(bufferId, src);
198 }
199# else // else SINGLE_THREAD
200 // probably we should try to parallel run for machineId loop as well. : TODO
201 tbb::parallel_for(0, 5, [&](unsigned bufferId) { bufferSetupFunc(bufferId, src); });
202# endif // end !SINGLE_THREAD
203 }
204
205 // merge all buffers
206 accumulatePartialTiles(nullptr, [&](int tileId) {
207 for (int machineId = 0; machineId < numMachines; ++machineId) {
208 if (!received[machineId]) continue;
209
210 const Fb &src = srcFbs[machineId];
211
212 accumulateRenderBufferOneTile(src, tileId);
213 if (src.getPixelInfoStatus()) accumulatePixelInfoOneTile(src, tileId);
214 if (src.getHeatMapStatus()) accumulatePixelInfoOneTile(src, tileId);
215 if (src.getWeightBufferStatus()) accumulateWeightBufferOneTile(src, tileId);
216 if (src.getRenderBufferOddStatus()) accumulateRenderBufferOddOneTile(src, tileId);
217 if (src.getRenderOutputStatus()) {
218 accumulateAllActiveAov(src, [&](const FbAovShPtr &srcFbAov, FbAovShPtr &dstFbAov) {
219 if (srcFbAov->getReferenceType() != FbReferenceType::UNDEF) return;
220
221 switch (srcFbAov->getFormat()) {
222 case VariablePixelBuffer::FLOAT :
223 accumulateFloat1AovOneTile(dstFbAov, srcFbAov, tileId);
224 break;
225 case VariablePixelBuffer::FLOAT2 :
226 accumulateFloat2AovOneTile(dstFbAov, srcFbAov, tileId);
227 break;
228 case VariablePixelBuffer::FLOAT3 :
229 accumulateFloat3AovOneTile(dstFbAov, srcFbAov, tileId);
230 break;
231 case VariablePixelBuffer::FLOAT4 :
232 accumulateFloat4AovOneTile(dstFbAov, srcFbAov, tileId);
233 break;
234 default :
235 break;
236 }
237 });
238 }
239 }
240 });
241}

245#ifdef SINGLE_THREAD
246template <typename F>
247void
248Fb::accumulatePartialTiles(const PartialMergeTilesTbl *partialMergeTilesTbl,
249 F accumTileFunc) const
250{
251 if (!partialMergeTilesTbl) {
252 // If partialMergeTilesTbl is empty, we accumulate all the tiles.
253 for (int tileId = 0; tileId < static_cast<int>(getTotalTiles()); ++tileId) {
254 accumTileFunc(tileId);
255 }
256 } else {
257 // Only accumulate tile which specified by partialMergeTilesTbl
258 for (int tileId = 0; tileId < static_cast<int>(getTotalTiles()); ++tileId) {
259 if ((*partialMergeTilesTbl)[tileId]) {
260 accumTileFunc(tileId);
261 }
262 }
263 }
264}
265#else // else SINGLE_THREAD
266template <typename F>
267void
268Fb::accumulatePartialTiles(const PartialMergeTilesTbl *partialMergeTilesTbl,
269 F accumTileFunc) const
270{
271 if (!partialMergeTilesTbl) {
272 // If partialMergeTilesTbl is empty, we accumulate all the tiles.
273 if (!getTotalTiles()) return;
274 // Based on several different grain size test (2,4,16,32,64,128,256,512,1024,2048,4096)
275 // and found 64 is somehow reasonable for 1K or more resolution image in this parallel_for loop
276 tbb::blocked_range<size_t> range(0, getTotalTiles(), 64);
277 tbb::parallel_for(range, [&](const tbb::blocked_range<size_t> &tileRange) {
278 for (size_t tileId = tileRange.begin(); tileId < tileRange.end(); ++tileId) {
279 accumTileFunc(tileId);
280 }
281 });
282 } else {
283 // Only accumulate tile which specified by partialMergeTilesTbl
284 std::vector<unsigned> partialMergeTilesId;
285 for (size_t tileId = 0; tileId < partialMergeTilesTbl->size(); ++tileId) {
286 if ((*partialMergeTilesTbl)[tileId]) {
287 partialMergeTilesId.push_back(tileId);
288 }
289 }
290 if (!partialMergeTilesId.size()) return;
291
292 // Based on several different grain size test (2,4,16,32,64,128,256,512,1024,2048,4096)
293 // and found 16 is somehow reasonable for 1K or more resolution image in this parallel_for loop
294 tbb::blocked_range<size_t> range(0, partialMergeTilesId.size(), 16);
295 tbb::parallel_for(range, [&](const tbb::blocked_range<size_t> &idRange) {
296 for (size_t id = idRange.begin(); id < idRange.end(); ++id) {
297 accumTileFunc(partialMergeTilesId[id]);
298 }
299 });
300 }
301}
302#endif // end !SINGLE_THREAD

304#ifdef SINGLE_THREAD
305template <typename F>
306void
307Fb::accumulateAllActiveAov(const Fb &srcFb, F activeAovFunc)
308{
309 for (const auto &itr : srcFb.mRenderOutput) {
310 const FbAovShPtr &srcFbAov = itr.second;
311 if (!srcFbAov->getStatus()) continue; // skip non active aov
312 // real data AOV buffer or Reference type
313 const std::string &aovName = srcFbAov->getAovName();
314
315 FbAovShPtr &dstFbAov = getAov(aovName);
316 activeAovFunc(srcFbAov, dstFbAov);
317 mRenderOutputStatus = true;
318 }
319}
320#else // else SINGLE_THREAD
321template <typename F>
322void
323Fb::accumulateAllActiveAov(const Fb &srcFb, F activeAovFunc)
324{
325 std::vector<std::string> activeAovNameArray;
326 for (const auto &itr : srcFb.mRenderOutput) {
327 const FbAovShPtr &srcFbAov = itr.second;
328 if (!srcFbAov->getStatus()) continue; // skip non active aov
329 // real data AOV buffer or Reference type
330 activeAovNameArray.push_back(srcFbAov->getAovName());
331 }
332 if (!activeAovNameArray.size()) return;
333
334 tbb::blocked_range<size_t> range(0, activeAovNameArray.size());
335 tbb::parallel_for(range, [&](const tbb::blocked_range<size_t> &r) {
336 for (size_t activeAovNameId = r.begin(); activeAovNameId < r.end(); ++activeAovNameId) {
337 const std::string &aovName = activeAovNameArray[activeAovNameId];
338 if (!srcFb.findAov(aovName)) {
339 std::ostringstream ostr;
340 ostr << ">> ============ Fb.h findAov failed. aovName:>" << aovName << "<";
341 logging::Logger::error(ostr.str());
342 continue;
343 }
344 const FbAovShPtr &srcFbAov = srcFb.mRenderOutput.at(aovName);
345
346 FbAovShPtr dstFbAov = getAov(aovName);
347 activeAovFunc(srcFbAov, dstFbAov);
348 mRenderOutputStatus = true;
349 }
350 });
351}
352#endif // end !SINGLE_THREAD

354template <typename F>
355void
356Fb::accumulateActiveOneTile(ActivePixels &dstActivePixels,
357 const ActivePixels &srcActivePixels,
358 const int tileId,
359 F accumTileFunc) const
360{
361 int pixOffset = tileId << 6;
362
363 uint64_t srcMask = srcActivePixels.getTileMask(tileId);
364 if (srcMask) {
365 uint64_t dstMask = dstActivePixels.getTileMask(tileId);
366 dstMask |= srcMask; // update destination activePixels mask
367 dstActivePixels.setTileMask(tileId, dstMask);
368
369 accumTileFunc(srcMask, pixOffset);
370 }
371}

373template <typename T>
374void
375Fb::accumulateTile(T *dstFirstValOfTile,
376 unsigned int *dstFirstNumSampleTotalOfTile,
377 uint64_t srcMask,
378 const T *srcFirstValOfTile,
379 const unsigned int *srcFirstNumSampleTotalOfTile) const
380{
381 T ave;
382 for (unsigned int pixId = 0; pixId < sPixelsPerTile; ++pixId) {
383 if (!srcMask) break; // early exit
384 if (srcMask & static_cast<uint64_t>(0x1)) {
385 T &currDstVal = dstFirstValOfTile[pixId];
386 unsigned int &currDstNumSampleTotal = dstFirstNumSampleTotalOfTile[pixId];
387 const T &currSrcVal = srcFirstValOfTile[pixId];
388 const unsigned int &currSrcNumSampleTotal = srcFirstNumSampleTotalOfTile[pixId];
389
390 unsigned int totalSample = currDstNumSampleTotal + currSrcNumSampleTotal;
391 if (totalSample > 0) {
392 ave =
393 (currDstVal * static_cast<float>(currDstNumSampleTotal) +
394 currSrcVal * static_cast<float>(currSrcNumSampleTotal)) /
395 static_cast<float>(totalSample);
396 } else {
397 // just in case
398 std::memset(reinterpret_cast<void *>(&ave), 0x0, sizeof(T));
399 }
400
401 currDstVal = ave;
402 currDstNumSampleTotal = totalSample;
403 }
404 srcMask >>= 1;
405 }
406}

408template <typename T>
409void
410Fb::accumulateTileClosestFilter(T *dstFirstValOfTile,
411 unsigned int *dstFirstNumSampleTotalOfTile,
412 uint64_t srcMask,
413 const T *srcFirstValOfTile,
414 const unsigned int *srcFirstNumSampleTotalOfTile) const
415//
416// special accumulateTile function for the case of using closestFilter
417//
418{
419 unsigned int depthId = T::N - 1; // depth value is last component
420
421 for (unsigned int pixId = 0; pixId < sPixelsPerTile; ++pixId) {
422 if (!srcMask) break; // early exit
423 if (srcMask & static_cast<uint64_t>(0x1)) {
424 T &currDstVal = dstFirstValOfTile[pixId];
425 unsigned int &currDstNumSampleTotal = dstFirstNumSampleTotalOfTile[pixId];
426
427 const T &currSrcVal = srcFirstValOfTile[pixId];
428 const unsigned int &currSrcNumSampleTotal = srcFirstNumSampleTotalOfTile[pixId];
429
430 unsigned int totalSample = currDstNumSampleTotal + currSrcNumSampleTotal;
431 if (totalSample > 0) {
432 if (currDstNumSampleTotal == 0) {
433 currDstVal = currSrcVal;
434 } else {
435 if (currSrcVal[depthId] < currDstVal[depthId]) {
436 // replace currDstVal if currSrcVal's closestFilter depth is closer
437 currDstVal = currSrcVal;
438 }
439 }
440 currDstNumSampleTotal = totalSample;
441 }
442 }
443 srcMask >>= 1;
444 }
445}

619void
620Fb::accumulatePixelInfoTile(PixelInfo *dstFirstPixelInfoOfTile,
621 uint64_t srcMask,
622 const PixelInfo *srcFirstPixelInfoOfTile) const
623{
624 for (unsigned int pixId = 0; pixId < sPixelsPerTile; ++pixId) {
625 if (!srcMask) break; // early exit
626 if (srcMask & static_cast<uint64_t>(0x1)) {
627 PixelInfo &currDstPixelInfo = dstFirstPixelInfoOfTile[pixId];
628 const PixelInfo &currSrcPixelInfo = srcFirstPixelInfoOfTile[pixId];
629
630 if (currDstPixelInfo.depth > currSrcPixelInfo.depth) {
631 currDstPixelInfo.depth = currSrcPixelInfo.depth;
632 }
633 }
634 srcMask >>= 1;
635 }
636}

638void
639Fb::accumulateWeightBufferTile(float *dstFirstPixelInfoOfTile,
640 uint64_t srcMask,
641 const float *srcFirstPixelInfoOfTile) const
642{
643 for (unsigned int pixId = 0; pixId < sPixelsPerTile; ++pixId) {
644 if (!srcMask) break; // early exit
