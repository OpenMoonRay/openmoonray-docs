35void
36TestBsdfConsistencyTask::testBsdfSampler()
37{
38 int zeroSamplePdfCount = 0;
39 int zeroEvalPdfCount = 0;
40 int zeroEvalRecipPdfCount = 0;
41 int sampleCount = 0;
42
43 mcrt_common::ThreadLocalState *tls = mcrt_common::getFrameUpdateTLS();
44 pbr::TLState* const pbrTls = MNRY_VERIFY(tls->mPbrTls.get());
45 scene_rdl2::alloc::Arena &arena = tls->mArena;
46 SCOPED_MEM(&arena);
47
48 scene_rdl2::util::Random random(randomSeed, randomStream);
49 shading::Bsdf* const bsdf = inTest->bsdfFactory(arena, inTest->frame);
50 BsdfSample bsmp;
51 const PathGuide pg;
52 const Vec3f p(0.f);
53
54 for (int sample = sampleFirst; sample != sampleLast; ++sample) {
55
56 bool foundError = false;
57
58 // Generate a random eye direction
59 Vec3f wo;
60 do {
61 wo = shading::sampleSphereUniform(random.getNextFloat(), random.getNextFloat());
62 } while(dot(wo, inTest->frame.getN()) < sEpsilon);
63 testAssert(isOne(wo.length()), "wo is not normalized");
64
65 // Get ready to sample, but don't include cosine term since we want to
66 // test bsdf reciprocity.
67 shading::BsdfSlice slice(inTest->frame.getN(), wo, false, true, ispc::SHADOW_TERMINATOR_FIX_OFF);
68 BsdfSampler sampler(pbrTls->mArena, *bsdf, slice,
69 TestBsdfSettings::sMaxSamplesPerLobe, true, pg);
70 sampleCount += sampler.getSampleCount();
71
72 for (int lobeIndex = 0; lobeIndex < sampler.getLobeCount(); ++lobeIndex) {
73 for (int i = 0; i < sampler.getLobeSampleCount(lobeIndex); ++i) {
74
75 // Draw the sample and test validity
76 const float r1 = random.getNextFloat();
77 const float r2 = random.getNextFloat();
78 sampler.sample(pbrTls, lobeIndex, p, r1, r2, bsmp);
79 if (bsmp.pdf == 0.0f) {
80 zeroSamplePdfCount++;
81 continue;
82 }
83
84 // TODO: work under fixed reference frame and slice, and record
85 // distribution of sampled lobes and make sure they match their albedos
86
87 // Check the sample
88 foundError |= testAssert(isValidPdf(bsmp.pdf),
89 "sample() returned invalid pdf (%f)", bsmp.pdf);
90 foundError |= testAssert(isValidColor(bsmp.f),
91 "sample() returned invalid color (%f %f %f)",
92 bsmp.f[0], bsmp.f[1], bsmp.f[2]);
93 foundError |= testAssert(isValidDirection(bsmp.wi),
94 "sample() returned invalid direction (%f %f %f)",
95 bsmp.wi[0], bsmp.wi[1], bsmp.wi[2]);
96
97
98 float error;
99
100 // Check consistency with eval()
101 // TODO: If we sampled from a mirror lobe, then calling eval here
102 // won't be consistent with bsdf.sample(), since sample() will return
103 // only the mirror lobe contribution and eval() will return all
104 // non-mirror lobes.
105 float checkPdf;
106 shading::BsdfLobe *lobe = sampler.getLobe(lobeIndex);
107 Color checkF = lobe->eval(slice, bsmp.wi, &checkPdf);
108 if (checkPdf == 0.0f) {
109 zeroEvalPdfCount++;
110 continue;
111 }
112
113 foundError |= testAssert(isValidPdf(checkPdf),
114 "eval() returned invalid pdf (%f)", checkPdf);
115 error = computeError(bsmp.pdf, checkPdf);
116 foundError |= testAssert(error < inTest->toleranceConsistency,
117 "eval() returned inconsistent pdf (error=%f)", error);
118
119 // Check consistency with eval()
120 foundError |= testAssert(isValidColor(checkF),
121 "eval() returned invalid color (%f %f %f)",
122 checkF[0], checkF[1], checkF[2]);
123 for (int i=0; i < 3; i++) {
124 error = computeError(bsmp.f[i], checkF[i]);
125 foundError |= testAssert(error < inTest->toleranceConsistency,
126 "eval() returned inconsistent color (error=%f)", error);
127 }
128
129
130 // Check eval() reciprocity
131 shading::BsdfSlice recipSlice(inTest->frame.getN(), bsmp.wi, false, true,
132 ispc::SHADOW_TERMINATOR_FIX_OFF);
133 if (inTest->testReciprocity) {
134 checkF = lobe->eval(recipSlice, wo, &checkPdf);
135 if (checkPdf == 0.0f) {
136 zeroEvalRecipPdfCount++;
137 continue;
138 }
139 foundError |= testAssert(isValidPdf(checkPdf),
140 "eval() reciprocal returned invalid pdf (%f)", checkPdf);
141 foundError |= testAssert(isValidColor(checkF),
142 "eval() reciprocal returned invalid color (%f %f %f)",
143 checkF[0], checkF[1], checkF[2]);
144 // TODO: The reciprocity test should divide by the square of the index
145 // of refraction for each medium side to handle glossy transmission
146 // properly
147 for (int i=0; i < 3; i++) {
148 error = computeError(bsmp.f[i], checkF[i]);
149 foundError |= testAssert(error < inTest->toleranceConsistency,
150 "eval() reciprocal returned inconsistent color (err=%f)",
151 error);
152 }
153 }
154
155 // TODO: check other "Rules" from Bsdf API (i.e. mirror case).
156
157 // For your debugging needs
158 if (foundError) {
159 sampler.sample(pbrTls, lobeIndex, p, r1, r2, bsmp);
160 lobe->eval(slice, bsmp.wi, &checkPdf);
161 lobe->eval(recipSlice, wo, &checkPdf);
162 }
163 }
164 }
165 }
166
167 mResult.mZeroSamplePdfCount += zeroSamplePdfCount;
168 mResult.mZeroEvalPdfCount += zeroEvalPdfCount;
169 mResult.mZeroEvalRecipPdfCount += zeroEvalRecipPdfCount;
170 mResult.mSampleCount += sampleCount;
171}

176void
177TestBsdfPdfIntegralTask::testBsdfSampler()
178{
179 mcrt_common::ThreadLocalState *tls = mcrt_common::getFrameUpdateTLS();
180 pbr::TLState *pbrTls = MNRY_VERIFY(tls->mPbrTls.get());
181 scene_rdl2::alloc::Arena &arena = tls->mArena;
182 SCOPED_MEM(&arena);
183
184 scene_rdl2::util::Random random(randomSeed, randomStream);
185 shading::Bsdf *bsdf = inTest->bsdfFactory(arena, inTest->frame);
186 const PathGuide pg;
187 BsdfSampler sampler(pbrTls->mArena, *bsdf, *inSlice,
188 TestBsdfSettings::sMaxSamplesPerLobe, true, pg);
189
190 // Compute the integrated probability using uniform sampling
191 float integral = 0.0f;
192 for (int sample = sampleFirst; sample != sampleLast; ++sample) {
193
194 // Compute the integrated probability using uniform sampling
195 for (int lobeIndex = 0; lobeIndex < sampler.getLobeCount(); ++lobeIndex) {
196 for (int i = 0; i < sampler.getLobeSampleCount(lobeIndex); ++i) {
197
198 shading::BsdfLobe *lobe = sampler.getLobe(lobeIndex);
199
200 Vec3f wi;
201 if (mResult.mSpherical) {
202 wi = shading::sampleSphereUniform(random.getNextFloat(), random.getNextFloat());
203 } else {
204 wi = shading::sampleLocalHemisphereUniform(random.getNextFloat(), random.getNextFloat());
205 if (lobe->matchesFlag(shading::BsdfLobe::TRANSMISSION)) {
206 wi = -wi;
207 }
208 wi = inTest->frame.localToGlobal(wi);
209 }
210 testAssert(isNormalized(wi), "wi is not normalized");
211
212 float pdf;
213 lobe->eval(*inSlice, wi, &pdf);
214 if (pdf > 0.0f) {
215 testAssert(isValidPdf(pdf), "pdf() returned invalid pdf (%f)", pdf);
216 integral += pdf * sampler.getInvLobeSampleCount(lobeIndex);
217 }
218 }
219 }
220 }
221
222 mResult.mIntegral += integral / sampler.getLobeCount();
223 mResult.mSampleCount = sampleLast - sampleFirst;
224}

229void
230TestBsdfEvalIntegralTask::testBsdfSampler()
231{
232 mcrt_common::ThreadLocalState *tls = mcrt_common::getFrameUpdateTLS();
233 pbr::TLState *pbrTls = MNRY_VERIFY(tls->mPbrTls.get());
234 scene_rdl2::alloc::Arena &arena = tls->mArena;
235 SCOPED_MEM(&arena);
236
237 scene_rdl2::util::Random random(randomSeed, randomStream);
238 shading::Bsdf *bsdf = inTest->bsdfFactory(arena, inTest->frame);
239 const PathGuide pg;
240 const Vec3f p(0.f);
241 BsdfSampler sampler(pbrTls->mArena, *bsdf, *inSlice,
242 TestBsdfSettings::sMaxSamplesPerLobe, true, pg);
243 BsdfSample bsmp;
244
245 const float pdfUniform = 1.0f /
246 (bsdf->getIsSpherical() ? sFourPi : sTwoPi);
247
248 Color integralUniform(sBlack);
249 Color integralImportance(sBlack);
250 for (int sample = sampleFirst; sample != sampleLast; ++sample) {
251 for (int lobeIndex = 0; lobeIndex < sampler.getLobeCount(); ++lobeIndex) {
252 for (int i = 0; i < sampler.getLobeSampleCount(lobeIndex); ++i) {
253
254 shading::BsdfLobe *lobe = sampler.getLobe(lobeIndex);
255
256 const float r1 = random.getNextFloat();
257 const float r2 = random.getNextFloat();
258
259 // Compute the integrated bsdf using uniform sampling
260 Vec3f wi;
261 if (bsdf->getIsSpherical()) {
262 wi = shading::sampleSphereUniform(r1, r2);
263 } else {
264 wi = shading::sampleLocalHemisphereUniform(r1, r2);
265 if (lobe->matchesFlag(shading::BsdfLobe::TRANSMISSION)) {
266 wi = -wi;
267 }
268 wi = inTest->frame.localToGlobal(wi);
269 }
270 testAssert(isOne(wi.length()), "wi is not normalized");
271 Color f = lobe->eval(*inSlice, wi);
272 testAssert(isValidColor(f), "eval() returned invalid color (%f %f %f)",
273 f[0], f[1], f[2]);
274 integralUniform += f / pdfUniform *
275 sampler.getInvLobeSampleCount(lobeIndex);
276
277 // Compute the integrated bsdf using importance sampling
278 sampler.sample(pbrTls, lobeIndex, p, r1, r2, bsmp);
279 if (bsmp.isInvalid()) {
280 continue;
281 }
282 testAssert(isValidPdf(bsmp.pdf), "sample() returned invalid pdf (%f)", bsmp.pdf);
283 testAssert(isValidColor(bsmp.f), "sample() returned invalid color "
284 "(%f %f %f)", bsmp.f[0], bsmp.f[1], bsmp.f[2]);
285 testAssert(isValidDirection(bsmp.wi), "sample() returned invalid direction "
286 "(%f %f %f)", bsmp.wi[0], bsmp.wi[1], bsmp.wi[2]);
287 integralImportance += bsmp.f / bsmp.pdf *
288 sampler.getInvLobeSampleCount(lobeIndex);
289 }
290 }
291 }
292
293 mResult.mIntegralUniform += integralUniform;
294 mResult.mIntegralImportance += integralImportance;
295 mResult.mSampleCount = sampleLast - sampleFirst;
296}
