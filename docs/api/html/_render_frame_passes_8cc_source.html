83// static function
84RenderDriver::RenderPassesResult
85RenderDriver::renderPasses(RenderDriver *driver, const FrameState &fs,
86 bool allowCancelation)
87{
88# ifdef RUNTIME_VERIFY1
89 if (!PixSampleSpanRuntimeVerify::get()) {
90 PixSampleSpanRuntimeVerify::init(fs.mWidth, fs.mHeight);
91 }
92# endif // end RUNTIME_VERIFY1
93
94 MNRY_ASSERT(fs.mNumRenderThreads <= mcrt_common::getNumTBBThreads());
95
96 // This is a part of the termination logic of multi-machine mcrt-computation
97 if (driver->mRenderStopAtPassBoundary) {
98 return RenderPassesResult::STOP_AT_PASS_BOUNDARY; // render completed at pass boundary
99 }
100 bool stopAtPassBoundary = false;
101
102 tbb::task_group taskGroup;
103
104 // This counter verifies that we don't leave this function until all threads
105 // have started working.
106 CACHE_ALIGN tbb::atomic<unsigned> numTBBThreads;
107 CACHE_ALIGN tbb::atomic<bool> canceled;
108
109 numTBBThreads = 0;
110 canceled = false;
111
112 TileWorkQueue *workQueue = &driver->mTileWorkQueue;
113 RenderFrameTimingRecord &timingRec = driver->getRenderFrameTimingRecord();
114
115 // Setup samplesPerTile value
116 timingRec.setRenderPassesSamplesPerTile(driver->mTileWorkQueue.getTotalTileSamples());
117
118 // Record start time
119 RenderPassesTimingRecord &renderPassesTimingRec = timingRec.startRenderPasses(fs.mNumRenderThreads);
120
121 // Hand out one TLS per TBB thread from this list.
122 mcrt_common::ThreadLocalState *topLevelTlsList = MNRY_VERIFY(mcrt_common::getTLSList());
123
124 std::mutex checkpointEstimationTimeMutex;
125 std::atomic<size_t> finishedTilesCount;
126 finishedTilesCount = 0;
127 size_t parallelInitFrameUpdateMinTileTotal = 0;
128 if (driver->mParallelInitFrameUpdate) {
129 parallelInitFrameUpdateMinTileTotal =
130 std::max(1UL,
131 static_cast<size_t>(workQueue->getNumTiles()) / driver->mParallelInitFrameUpdateMcrtCount);
132 /* useful debug message
133 std::cerr << ">> RenderFramePasses.cc init minTileTotal:" << parallelInitFrameUpdateMinTileTotal
134 << " mParallelInitFrameUpdateMcrtCount:" << driver->mParallelInitFrameUpdateMcrtCount << '\n';
135 */
136 }
137
138 // Spawn one task for each tbb thread.
139 for (unsigned ithread = 0; ithread < fs.mNumRenderThreads; ++ithread) {
140
141 mcrt_common::ThreadLocalState *topLevelTls = topLevelTlsList + ithread;
142
143 // The topLevelTls pointer must be captured by value to prevent it changing out
144 // from under us as other threads are subsequently spawned.
145
146#ifdef FORCE_SINGLE_THREADED_RENDERING
147 callLambda([&, topLevelTls]()
148#else
149 taskGroup.run([&, topLevelTls]()
150#endif
151 {
152 double timeStart = scene_rdl2::util::getSeconds(); // get current time
153
154 ++numTBBThreads;
155
156 pbr::TLState *tls = MNRY_VERIFY(topLevelTls->mPbrTls.get());
157
158 EXCL_ACCUMULATOR_PROFILE(tls, EXCL_ACCUM_RENDER_DRIVER_OVERHEAD);
159 ACCUMULATOR_PROFILE(tls, ACCUM_RENDER_DRIVER_PARALLEL);
160
161 // Wait barrier until all threads are wake up and ready to go.
162 // This is a busy loop and cost is big concern for REALTIME renderMode case.
163 // (It's negligible when BATCH and PROGRESSIVE case).
164 // Why all threads are sync to start ? This is a fastest and reliable solution to boot
165 // all threads. If we boot threads without this sync barrier logic, booted thread is start working
166 // and increase CPU load then we need longer time to complete boot for all render threads.
167 // As a result, we need longer duration to get full speed.
168 while (numTBBThreads < fs.mNumRenderThreads) {
169 if (isRenderCanceled()) {
170 // There is some risk to get deadlock situation on this loop.
171 // This is a safety logic to exit by cancel.
172 canceled = true;
173 return;
174 }
175
176 // Actually this loop can not get constant performance even carefully controlled nano sleep value.
177 // Because render has lots of sleeped (not active) threads actually and kernel need to check
178 // about them constantly. Shortest context switching interval is around 10ms (I guess).
179 // So sometimes some threads need to wait extra 10ms or so to wake up from suspended condition.
180 // Obviously 10ms loss is big for realtime render case but this solution is a best result so far.
181 struct timespec tm;
182 tm.tv_sec = 0;
183 tm.tv_nsec = 1000; // 0.001ms : tested several different range but could not find any difference.
184 nanosleep(&tm, NULL); // yield CPU resource
185 }
186
187 double timeReady = scene_rdl2::util::getSeconds(); // get current time
188
189
190 // Embree wants these modes set on each thread it uses.
191 _MM_SET_FLUSH_ZERO_MODE(_MM_FLUSH_ZERO_ON);
192 _MM_SET_DENORMALS_ZERO_MODE(_MM_DENORMALS_ZERO_ON);
193
194 // Initialize for texturing, this needs to be called once per
195 // render frame.
196 shading::TLState *shadingTls = topLevelTls->mShadingTls.get();
197 if (shadingTls) {
198 shadingTls->initTexturingSupport();
199 }
200
201 if (allowCancelation) {
202 tls->enableCancellation((fs.mRenderMode == RenderMode::BATCH) ? false : true);
203 } else {
204 tls->disableCancellation();
205 }
206
207 const unsigned threadIdx = getThreadIdx(tls);
208
209 double timeInitTexturing = scene_rdl2::util::getSeconds(); // get current time
210
211 TileGroup group;
212
213 int processedTilesTotal = 0;
214 unsigned processedSampleTotal = 0;
215
216 // We have to track the condition of stopAtPassBoundary each thread independently
217 // in order to properly flush the radiance queue under vector mode.
218 bool stopAtPassBoundaryThreadLocal = false;
219 while (!tls->isCanceled() && workQueue->getNextTileGroup(threadIdx, &group, driver->getLastCoarsePassIdx())) {
220 if (group.mFirstFinePass && (fs.mRenderMode == RenderMode::PROGRESSIVE ||
221 fs.mRenderMode == RenderMode::PROGRESSIVE_FAST ||
222 fs.mRenderMode == RenderMode::PROGRESS_CHECKPOINT)) {
223 // Increase all queue sizes in the system if we've finished coarse pass
224 pbr::forEachTLS([](pbr::TLState *tls) {
225 tls->setAllQueueSizes(1.0f);
226 });
227 }
228
229 // Record tiles currently being rendered.
230 tls->mCurrentPassIdx = group.mPassIdx;
231 if (fs.mRenderMode == RenderMode::PROGRESS_CHECKPOINT) {
232 // If tls->mCurrentPassIdx is 0, we can not cancel this pass (see pbr::TLState::isCanceled()).
233 // So we set passId + 1 for tls when PROGRESS_CHECKPOINT mode.
234 tls->mCurrentPassIdx++;
235 }
236 processedSampleTotal += renderTiles(driver, topLevelTls, group);
237 ++processedTilesTotal;
238
239 if (driver->mParallelInitFrameUpdate) {
240 // Under parallel init frame update mode, we have to check the total number of processed
241 // tiles exceeds threshold. If we processed tile more than the threshold, we turn on the
242 // snapshot ready flag and set cancellation available enable in order to enhance
243 // interactive performance under multi-machine context.
244 finishedTilesCount += (group.mEndTileIdx - group.mStartTileIdx);
245 if (finishedTilesCount >= parallelInitFrameUpdateMinTileTotal) {
246 if (!driver->isReadyForDisplay()) {
247 driver->setReadyForDisplay();
248 tls->enableCancellation(true);
249 { // statistical information for debugging purposes
250 std::lock_guard<std::mutex> lock(checkpointEstimationTimeMutex);
251 float time = driver->mParallelInitFrameUpdateTime.end();
252 driver->mCheckpointEstimationTime.set(time);
253 /* useful debug message
254 std::ostringstream ostr;
255 ostr << ">> RenderFramePasses.cc READY-FOR-DISPLAY parallel"
256 << " time:" << time << " sec"
257 << " parallelInitFrameUpdateMinTileTotal:" << parallelInitFrameUpdateMinTileTotal
258 << " finishedTilesCount:" << finishedTilesCount;
259 std::cerr << ostr.str() << '\n';
260 */
261 }
262 }
263 }
264 }
265
266 // Update progress.
267 driver->transferAllProgressFromSingleTLS(tls);
268
269 if (fs.mSamplingMode == SamplingMode::ADAPTIVE) {
270 if (driver->mFilms[0].getAdaptiveDone()) {
271 break;
272 }
273 }
274 }
275
276 double timeRenderTiles = scene_rdl2::util::getSeconds(); // get current time
277
278 //
279 // Queue draining phase for bundled mode.
280 //
281 // TODO: track progress for this phase also if it starts taking
282 // significant time.
283 //
284 if (fs.mExecutionMode == mcrt_common::ExecutionMode::VECTORIZED ||
285 fs.mExecutionMode == mcrt_common::ExecutionMode::XPU) {
286
287 ACCUMULATOR_PROFILE(tls, ACCUM_DRAIN_QUEUES);
288
289 if (tls->isCanceled() || stopAtPassBoundaryThreadLocal) {
290
291 ACCUMULATOR_PROFILE(tls, ACCUM_NON_RENDER_DRIVER);
292
293 // Radiance values are already computed so just write them into
294 // the frame buffer. This gives a more complete image even if
295 // a cancellation was requested (useful for progressive mode).
296 tls->flushRadianceQueues();
297
298 } else {
299
300 ACCUMULATOR_PROFILE(tls, ACCUM_NON_RENDER_DRIVER);
301
302 unsigned flushed;
303
304 // Note, this flushing logic depends on one thread not being able
305 // to add work to another threads queues.
306 do {
307 flushed = shading::Material::flushNonEmptyShadeQueue(topLevelTls);
308 CHECK_CANCELLATION(tls, break);
309
310 flushed += driver->flushXPUQueue(topLevelTls, &topLevelTls->mArena);
311
312 flushed += tls->flushLocalQueues();
313 CHECK_CANCELLATION(tls, break);
314
315 } while (flushed || numTBBThreads < fs.mNumRenderThreads);
316 }
317 }
318
319 double timeQueueDraining = scene_rdl2::util::getSeconds(); // get current time
320
321 if (tls->isCanceled()) {
322 canceled = true;
323 }
324
325 tls->disableCancellation();
326
327 // Update progress.
328 driver->transferAllProgressFromSingleTLS(tls);
329
330 double timeEnd = scene_rdl2::util::getSeconds(); // get current time
331
332 // Update duration record information
333 RenderEngineTimingRecord &rec = renderPassesTimingRec.getEngineTimingRecord(tls->mThreadIdx);
334 rec.setId(tls->mThreadIdx);
335 rec.set(RenderEngineTimingRecord::Tag::START, timeStart - renderPassesTimingRec.getStartTime());
336 rec.set(RenderEngineTimingRecord::Tag::READY, timeReady - renderPassesTimingRec.getStartTime());
337 rec.set(RenderEngineTimingRecord::Tag::INITTEXTURING, timeInitTexturing - timeReady);
338 rec.set(RenderEngineTimingRecord::Tag::RENDERTILES, timeRenderTiles - timeInitTexturing);
339 rec.set(RenderEngineTimingRecord::Tag::QUEUEDRAINING, timeQueueDraining - timeRenderTiles);
340 rec.set(RenderEngineTimingRecord::Tag::FINISHUP, timeEnd - timeQueueDraining);
341 rec.set(RenderEngineTimingRecord::Tag::ENDGAP, timeEnd); // set timeEnd at this moment. update later.
342 rec.set(RenderEngineTimingRecord::Tag::ACTIVE, timeEnd - timeReady);
343 rec.set(RenderEngineTimingRecord::Tag::PROCESSEDTILESTOTAL, static_cast<double>(processedTilesTotal));
344 rec.set(RenderEngineTimingRecord::Tag::PROCESSEDSAMPLETOTAL, static_cast<double>(processedSampleTotal));
345 });
346 }
347
348 taskGroup.wait();
349
350 timingRec.finalizeRenderPasses(); // End record timing : compute average thread timing and other info
351 driver->mProgressEstimation.updatePassInfo(timingRec); // update pass info
352
353# ifdef DEBUG
354 // Verify that all queues have been completely drained by now.
355 if (!canceled) {
356 MNRY_ASSERT(verifyNoOutstandingWork(fs));
357 }
358# endif // end DEBUG
359
360 if (stopAtPassBoundary) return RenderPassesResult::STOP_AT_PASS_BOUNDARY;
361 if (canceled) return RenderPassesResult::ERROR_OR_CANCEL;
362 return RenderPassesResult::OK_AND_TRY_NEXT;
363}

365void
366RenderDriver::runDisplayFiltersTile(RenderDriver *driver,
367 size_t tileIdx,
368 size_t threadId)
369{
370 if (driver->mDisplayFilterMutex.try_lock()) {
371 // This thread will try to snapshot aovs.
372 // The aov buffers are stored in mDisplayFilterDriver and all threads
373 // read from those buffers to run the display filters.
374 driver->getFrameState().mRenderContext->snapshotAovsForDisplayFilters(true, false);
375 driver->mDisplayFilterMutex.unlock();
376 }
377
378 // Filter just this tile. It may require reading input buffer data from adjacent tiles.
379 // It is ok if that data is out of date. This function is called only in progressive mode
380 // and the display filter buffers get refined with each progessive pass.
381 driver->mDisplayFilterDriver.runDisplayFilters(tileIdx, threadId);
382}

419unsigned
420RenderDriver::renderTiles(RenderDriver *driver,
421 mcrt_common::ThreadLocalState *tls,
422 const TileGroup &group) // return processed primary ray total
423{
424 pbr::TLState *pbrTls = tls->mPbrTls.get();
425 const rndr::FrameState &fs = driver->getFrameState();
426 scene_rdl2::alloc::Arena *arena = pbrTls->mArena;
427 SCOPED_MEM(arena);
428
429 // Used to track all the time not spent in the RenderDriver.
430 ACCUMULATOR_GET_PAUSED(pbrTls, ACCUM_NON_RENDER_DRIVER, nonRenderDriverAccumulator);
431
432 RenderSamplesParams params;
433 { // setup params members
434 // CPPCHECK -- Using memset() on struct which contains a floating point number.
435 // cppcheck-suppress memsetClassFloat
436 memset(&params, 0, sizeof(params));
437 params.mDriver = driver;
438 {
439 float *aovs = nullptr;
440 float *deepAovs = nullptr;
441 float *deepVolumeAovs = nullptr;
442 float *localAovs = nullptr;
443 unsigned aovNumChannels = 0;
444 if (!fs.mAovSchema->empty()) {
445 aovNumChannels = fs.mAovSchema->numChannels();
446 aovs = arena->allocArray<float>(aovNumChannels);
447 deepAovs = arena->allocArray<float>(aovNumChannels);
448 deepVolumeAovs = arena->allocArray<float>(aovNumChannels);
449 localAovs = arena->allocArray<float>(aovNumChannels);
450 }
451 params.mAovNumFloats = aovNumChannels;
452 params.mAovs = aovs;
453 params.mDeepAovs = deepAovs;
454 params.mDeepVolumeAovs = deepVolumeAovs;
455 params.mLocalAovs = localAovs;
456 }
457
458 params.mNonRenderDriverAccumulator = &nonRenderDriverAccumulator;
459
460 params.mConsistentSamplesPerPixel = fs.mRenderContext->getNumConsistentSamples();
461
462 params.mNumCameras = fs.mScene->getCameraCount();
463 for (int i = 0; i < params.mNumCameras; i++) {
464 // shutter bias modifies the time of the primary ray sample
465 params.mShutterBias[i] = fs.mScene->getCamera(i)->getShutterBias();
466 }
467
468 params.mRenderNodeTotal = fs.mNumRenderNodes;
469 params.mRenderNodeSampleOfs = fs.mRenderNodeIdx;
470 }
471
472 unsigned processedSampleTotalFilm0 = 0;
473 const unsigned numActiveFilms = driver->mLastNumActiveFilms;
474
475 for (unsigned ifilm = 0; ifilm != numActiveFilms; ++ifilm) {
476 params.mFilmIdx = ifilm;
477 params.mFilm = &driver->mFilms[ifilm];
478 pbrTls->mNonBundledFilmIdx = ifilm;
479
480 pbr::DeepBuffer *deepBuffer = params.mFilm->getDeepBuffer();
481 // This is not thread safe - it should really be done during frame setup!
482 if (deepBuffer != nullptr) {
483 unsigned totalNumSamples = fs.mMaxSamplesPerPixel;
484 deepBuffer->setSamplesPerPixel(totalNumSamples);
485 if (totalNumSamples >= 64) {
486 deepBuffer->setSubpixelRes(8);
487 } else if (totalNumSamples >= 16) {
488 deepBuffer->setSubpixelRes(4);
489 } else if (totalNumSamples >= 4) {
490 deepBuffer->setSubpixelRes(2);
491 } else {
492 deepBuffer->setSubpixelRes(1);
493 }
494 }
495
496 pbr::CryptomatteBuffer *cryptomatteBuffer = params.mFilm->getCryptomatteBuffer();
497
498 // Loop over current batch of tiles, we execute tile batches in parallel.
499 unsigned processedSampleTotal = 0;
500 for (unsigned itile = group.mStartTileIdx; itile != group.mEndTileIdx; ++itile) {
501 params.mTileIdx = itile;
502 if (!renderTile(driver, tls, group, params, deepBuffer, cryptomatteBuffer, processedSampleTotal)) {
503 return 0; // cancel return
504 }
505 }
506 if (ifilm == 0) processedSampleTotalFilm0 = processedSampleTotal;
507 }
508
509
510 return processedSampleTotalFilm0; // return processed sample total of film = 0
511}

513// static function
514bool
515RenderDriver::renderTile(RenderDriver *driver,
516 mcrt_common::ThreadLocalState *tls,
517 const TileGroup &group,
518 RenderSamplesParams &params,
519 pbr::DeepBuffer *deepBuffer,
520 pbr::CryptomatteBuffer *cryptomatteBuffer,
521 unsigned &processedSampleTotal)
522//
523// return cancel render condition : true=non-cancel false=canceled
524//
525{
526 const rndr::FrameState &fs = driver->getFrameState();
527
528 if (fs.mSamplingMode != SamplingMode::UNIFORM) {
529 if (params.mFilm->getAdaptiveRenderTilesTable()->getTile(params.mTileIdx).isCompleted()) {
530 return true; // already completed tile.
531 }
532 }
533
534 const Pass &pass = driver->mTileWorkQueue.getPass(group.mPassIdx);
535 if (fs.mSamplingMode == SamplingMode::UNIFORM || pass.isCoarsePass()) {
536 //
537 // Uniform sampling tile mode
538 //
539 if (!renderTileUniformSamples<false>(driver, tls, group, params, deepBuffer, cryptomatteBuffer,
540 pass.mStartSampleIdx, pass.mEndSampleIdx, processedSampleTotal)) {
541 return false; // canceled render
542 }
543 } else {
544 //
545 // Adaptive sampling tile mode
546 //
547 const Film* const film = params.mFilm;
548 if (!film->getAdaptiveDone()) {
549 if (!renderTileAdaptiveStage(driver, tls, group, params, deepBuffer, cryptomatteBuffer,
550 processedSampleTotal)) {
551 return false; // canceled render
552 }
553 }
554 }
555 driver->mDisplayFilterDriver.requestTileUpdate(params.mTileIdx);
556
557 // Update display filters after each tile is rendered in progressive mode.
558 // This is done for non coarse passes because each of the neighboring
559 // tiles are guaranteed to have valid data, though it might be outdated data.
560 if (driver->areCoarsePassesComplete() && fs.mRenderMode == RenderMode::PROGRESSIVE) {
561 runDisplayFiltersTile(driver, params.mTileIdx, tls->mThreadIdx);
562 }
563
564 return true;
565}

567// static function
568bool
569RenderDriver::renderTileAdaptiveStage(RenderDriver* driver,
570 mcrt_common::ThreadLocalState* tls,
571 const TileGroup& group,
572 RenderSamplesParams& params,
573 pbr::DeepBuffer* deepBuffer,
574 pbr::CryptomatteBuffer* cryptomatteBuffer,
575 unsigned& processedSampleTotal)
576{
577 const Pass &pass = driver->mTileWorkQueue.getPass(group.mPassIdx);
578# ifdef PRINT_DEBUG_MESSAGE_ADAPTIVE_STAGE
579 bool debug = params.mFilm->getAdaptiveRenderTilesTable()->isDebugTile(params.mTileIdx);
580 if (debug) {
581 std::cerr << ">> RenderFrame.cc renderTileAdaptiveStage() start ... group.mPassIdx:" << group.mPassIdx
582 << " pass(pix(" << pass.mStartPixelIdx << '~' << pass.mEndPixelIdx << ")"
583 << " smp(" << pass.mStartSampleIdx << '~' << pass.mEndSampleIdx << "))" << std::endl;
584 }
585# endif // end PRINT_DEBUG_MESSAGE_ADAPTIVE_STAGE
586 if (pass.mStartSampleIdx == pass.mEndSampleIdx) {
587 return true;
588 }
589
590 Film* const film = params.mFilm;
591 const scene_rdl2::fb_util::Tile &tile = (*driver->getTiles())[params.mTileIdx];
592
593 scene_rdl2::math::BBox2i adaptiveRegion;
594 const scene_rdl2::math::BBox2i* adaptiveRegionPtr;
595 switch (updateTileCondition(driver, group, params, pass.mStartSampleIdx)) {
596 case AdaptiveRenderTileInfo::Stage::COMPLETED:
597 return true;
598 case AdaptiveRenderTileInfo::Stage::UNIFORM_STAGE:
599 adaptiveRegionPtr = nullptr;
600 break;
601 case AdaptiveRenderTileInfo::Stage::ADAPTIVE_STAGE:
602 adaptiveRegion = film->getAdaptiveSampleArea(tile, tls);
603 adaptiveRegionPtr = std::addressof(adaptiveRegion);
604 break;
605
606 }
607 if (!renderTileUniformSamples<true>(driver,
608 tls,
609 group,
610 params,
611 deepBuffer,
612 cryptomatteBuffer,
613 pass.mStartSampleIdx,
614 pass.mEndSampleIdx,
615 processedSampleTotal,
616 adaptiveRegionPtr)) {
617# ifdef PRINT_DEBUG_MESSAGE_ADAPTIVE_STAGE
618 if (debug) { std::cerr << ">> RenderFrame.cc renderTileAdaptiveStage() canceled" << std::endl; }
619# endif // end PRINT_DEBUG_MESSAGE_ADAPTIVE_STAGE
620 return false; // canceled render
621 }
622 const rndr::FrameState &fs = driver->getFrameState();
623 film->updateAdaptiveError(tile, film->getRenderBuffer(), *film->getRenderBufferOdd(),
624 fs.mTargetAdaptiveError, tls, pass.mEndSampleIdx);
625
626# ifdef PRINT_DEBUG_MESSAGE_ADAPTIVE_STAGE
627 if (debug) { std::cerr << ">> RenderFrame.cc renderTileAdaptiveStage() done" << std::endl; }
628# endif // end PRINT_DEBUG_MESSAGE_ADAPTIVE_STAGE
629
630 return true;
631}

633// static function
634AdaptiveRenderTileInfo::Stage
635RenderDriver::updateTileCondition(RenderDriver *driver,
636 const TileGroup &group,
637 RenderSamplesParams &params,
638 const unsigned startSampleIdx)
639{
640 const rndr::FrameState &fs = driver->getFrameState();
641 if (startSampleIdx < fs.mMinSamplesPerPixel) {
642 // We have to do uniform sampling until minSamplesPerPixel
643 return AdaptiveRenderTileInfo::Stage::UNIFORM_STAGE;
644 }
645 if (startSampleIdx >= fs.mMaxSamplesPerPixel) { // hit maximum samples per pixel -> complete this tile
646 params.mFilm->getAdaptiveRenderTilesTable()->setTileCompleteAdaptiveStage(params.mTileIdx);
647 return AdaptiveRenderTileInfo::Stage::COMPLETED;
648 }
649
650 // Still need to add sample uniformly to all pixels of this tile
651 return AdaptiveRenderTileInfo::Stage::ADAPTIVE_STAGE;
652}

654// static function
655// _adaptive_ does not strictly have to be a template parameter, but it's known at compile-time, so we might as well
656// optimize it out.
657template <bool adaptive>
658bool
659RenderDriver::renderTileUniformSamples(RenderDriver *driver,
660 mcrt_common::ThreadLocalState *tls,
661 const TileGroup &group,
662 RenderSamplesParams &params,
663 pbr::DeepBuffer *deepBuffer,
664 pbr::CryptomatteBuffer *cryptomatteBuffer,
665 const unsigned startSampleIdx,
666 const unsigned endSampleIdx,
667 unsigned &processedSampleTotal,
668 const scene_rdl2::math::BBox2i* inputRegion)
669// return cancel render condition : true=non-cancel false=canceled
670{
671 if (adaptive && inputRegion && inputRegion->empty()) {
672 return true;
673 }
674
675 pbr::TLState *pbrTls = tls->mPbrTls.get();
676 const rndr::FrameState &fs = *reinterpret_cast<const rndr::FrameState *>(pbrTls->mFs);
677 const Pass &pass = driver->mTileWorkQueue.getPass(group.mPassIdx);
678 Film &film = *params.mFilm;
679 const scene_rdl2::fb_util::Tile &tile = (*driver->getTiles())[params.mTileIdx];
680
681 for (unsigned ipix = pass.mStartPixelIdx; ipix != pass.mEndPixelIdx; ++ipix) {
682 // Note we wrap around values over 64. This subtlety is actively used for realtime mode.
683 unsigned pixelPerm = Film::getPixelFillOrder(params.mTileIdx, ipix);
684 unsigned px = (tile.mMinX & ~0x07) + (pixelPerm & 7);
685 unsigned py = (tile.mMinY & ~0x07) + (pixelPerm / 8);
686 if (!fs.mViewport.contains(px, py)) continue; // Check this pixel is inside the viewport.
687
688 if (adaptive && inputRegion && disjoint(*inputRegion, scene_rdl2::math::Vec2i{static_cast<int>(px), static_cast<int>(py)})) {
689 continue;
690 }
691
692 scene_rdl2::alloc::Arena *pixelArena = pbrTls->mPixelArena;
693 SCOPED_MEM(pixelArena);
694
695 // update pixelInfo(=pixel center depth) if required
696 computePixelInfo(driver, tls, film, params.mNumCameras, px, py);
697
698 unsigned currStartSampleIdx = startSampleIdx;
699 unsigned currEndSampleIdx = endSampleIdx;
700 if (film.getResumeStartSampleIdBuff().isValid()) {
701 // resume rendering by adaptive sampled resume file
702 unsigned initSampleIdx = film.getResumeStartSampleIdBuff().getSampleId(px, py);
703 if (initSampleIdx > 0) {
704 // This is a case of resume rendering.
705 // First of all, we have to compare the sample range with the inital-start-sampleId which
706 // came from the resume file.
707 if (currEndSampleIdx <= initSampleIdx) continue; // this pixel already done.
708 if (currStartSampleIdx < initSampleIdx) {
709 currStartSampleIdx = initSampleIdx; // We don't want to run already finished sample range.
710 }
711 }
712 }
713 if (fs.mSamplingMode == SamplingMode::ADAPTIVE) {
714 // In an adaptive sampling case, we have to schedule sample range to the MCRT threads as
715 // continuous order. We have trouble for checkpoint data if some pixel does not sample some
716 // middle sample range. This wrong pixel's last sampleId is not equal to weight value and
717 // resume rendering will make a mistake start samplingId in this case.
718 // The order of the scheduled sample span by each MCRT thread does not matter.
719 // It is pretty important to get a continuous sampleId sequence without any missing span in
720 // the middle. In order to keep all samplingId spans keeps continuous condition,
721 // we use CAS access to the current pixel sampleId value. This makes it safe to get the next
722 // sampleId range by multiple MCRT threads even 2 or more threads are working on the same tile
723 // simultaneously.
724 const unsigned nsamples = currEndSampleIdx - currStartSampleIdx;
725 currStartSampleIdx = film.getCurrSampleIdBuff().atomicReserveSampleId(px, py, nsamples);
726 currEndSampleIdx = currStartSampleIdx + nsamples;
727 }
728
729# ifdef RUNTIME_VERIFY0
730 PixSampleCountRuntimeVerify::get()->add(px, py, currEndSampleIdx - currStartSampleIdx);
731# endif // end RUNTIME_VERIFY0
732# ifdef RUNTIME_VERIFY1
733 PixSampleSpanRuntimeVerify::get()->add(px, py, currStartSampleIdx, currEndSampleIdx);
734# endif // end RUNTIME_VERIFY1
735
736 // Get the number of samples for this pixel.
737 unsigned totalNumSamples = computeTotalNumSamples(fs, params.mFilmIdx, px, py);
738 if (totalNumSamples < currStartSampleIdx) continue; // skip this pixel if it has enough samples
739
740 // setup params
741 params.mPx = px;
742 params.mPy = py;
743 params.mTotalNumSamples = totalNumSamples;
744
745 // Each pixel + camera gets its own sampler.
746 pbr::Sampler samplers[params.mNumCameras];
747 for (int cameraId = 0; cameraId < params.mNumCameras; cameraId++) {
748 const auto stereo = fs.mScene->getCamera(cameraId)->getStereoView();
749 pbr::DeepBuffer *deepBuffer = film.getDeepBuffer();
750 bool use8x8Grid = deepBuffer ?
751 (deepBuffer->getFormat() == pbr::DeepFormat::OpenDCX2_0) : false;
752 samplers[cameraId] = pbr::Sampler(pbr::PixelScramble(px, py, fs.mFrameNumber, stereo),
753 fs.mPixelFilter, use8x8Grid, totalNumSamples);
754 params.mSamplers[cameraId] = &samplers[cameraId];
755 }
756
757 // Used for generating good sampling index values in the case of realtime.
758 params.mRealtimeSampleOfs = (fs.mRenderMode == RenderMode::REALTIME) ? ipix : 0;
759 bool renderPixCondition = false;
760 if (fs.mRenderMode != RenderMode::PROGRESSIVE_FAST) {
761 renderPixCondition = (fs.mExecutionMode == mcrt_common::ExecutionMode::SCALAR) ?
762 renderPixelScalarSamples(pbrTls, currStartSampleIdx, currEndSampleIdx, &params) :
763 renderPixelVectorSamples(pbrTls, currStartSampleIdx, currEndSampleIdx, &params,
764 group, deepBuffer, cryptomatteBuffer);
765 } else {
766 // renderPixelScalarSamplesFast is run for both SCALAR and VECTOR mode
767 renderPixCondition = renderPixelScalarSamplesFast(pbrTls, currStartSampleIdx, currEndSampleIdx, &params);
768 }
769 if (!renderPixCondition) {
770 return false; // canceled
771 }
772
773 if (deepBuffer) {
774 deepBuffer->finishPixel(pbrTls->mThreadIdx);
775 }
776
777 // Update progress.
778 if (params.mFilmIdx == 0) {
779 pbrTls->mTilesRenderedTo.setBit(params.mTileIdx); // for moonray_gui debugging. for showTileProgress()
780 // NonCheckpoint or nonAdaptiveCheckpoint case, we uses this code for progress update
781 // Currently all adaptive sampling progress update information is provided by
782 // film.getAdaptiveRenderTilesTable()
783 unsigned samplesRendered = currEndSampleIdx - currStartSampleIdx;
784 pbrTls->mPrimaryRaysSubmitted[group.mPassIdx] += samplesRendered; // used by non checkpoint case
785 processedSampleTotal += samplesRendered;
786 if (fs.mRenderMode == RenderMode::PROGRESS_CHECKPOINT) {
787 driver->mProgressEstimation.atomicAddSamples(samplesRendered);
788 }
789 }
790 if (film.getAdaptiveRenderTilesTable()) { // adaptive sampling case
791 unsigned addedSamples = currEndSampleIdx - currStartSampleIdx;
792 film.getAdaptiveRenderTilesTable()->setTileUpdate(params.mTileIdx, addedSamples); // update
793 }
794 } // ipix
795
796 return true;
797}

799// static function
800bool
801RenderDriver::renderPixelScalarSamples(pbr::TLState *pbrTls,
802 const unsigned startSampleIdx,
803 const unsigned endSampleIdx,
804 RenderSamplesParams *params)
805{
806 //
807 // Single ray execution start:
808 //
809
810 MNRY_ASSERT(startSampleIdx < endSampleIdx);
811
812 scene_rdl2::alloc::Arena *arena = pbrTls->mArena;
813 SCOPED_MEM(arena);
814
815 unsigned const px = params->mPx;
816 unsigned const py = params->mPy;
817 Film *film = params->mFilm;
818 float *aovs = params->mAovs;
819
820 float *localAovs = params->mLocalAovs;
821 float *deepAovs = params->mDeepAovs;
822 float localDepths[params->mNumCameras];
823 for (int cameraId = 0; cameraId < params->mNumCameras; cameraId++) {
824 localDepths[cameraId] = scene_rdl2::math::pos_inf;
825 }
826
827 const rndr::FrameState &fs = *reinterpret_cast<const rndr::FrameState *>(pbrTls->mFs);
828 const auto& schema = *fs.mAovSchema;
829
830 // conditional heat map collection
831 MNRY_ASSERT((fs.mRequiresHeatMap && film->getHeatMapBuffer()) ||
832 !fs.mRequiresHeatMap);
833 MCRT_COMMON_CLOCK_OPEN(fs.mRequiresHeatMap ? &(film->getHeatMap(px, py)) : nullptr);
834
835 scene_rdl2::fb_util::RenderColor accRadiance = scene_rdl2::fb_util::RenderColor(scene_rdl2::math::ZeroTy());
836 scene_rdl2::fb_util::RenderColor accRadiance2 = scene_rdl2::fb_util::RenderColor(scene_rdl2::math::ZeroTy());
837 uint32_t numAccSamples = 0;
838 if (params->mLocalAovs) fs.mAovSchema->initFloatArray(params->mLocalAovs);
839
840#ifdef DEBUG_SAMPLE_REC_MODE
841 DebugSamplesRecArray *debugSamplesRecArray = film->getDebugSamplesRecArray();
842#endif // end DEBUG_SAMPLE_REC_MODE
843
844 // loop over all cameras for this pixel
845 for (int cameraId = 0; cameraId < params->mNumCameras; cameraId++) {
846 // Loop over samples in current pixel. It is important to note that all samples within
847 // a single pass are guaranteed to be executed on the same thread. No single tile can
848 // be executed on different threads at one point in time (the TileWorkQueue ensures
849 // this), so we don't need to do any locking here.
850 for (unsigned isamp = startSampleIdx; isamp != endSampleIdx; ++isamp) {
851
852 // This line is for supporting the pixel sample map functionality.
853 if (isamp >= params->mTotalNumSamples) break;
854
855 CHECK_CANCELLATION(pbrTls, return false);
856
857 // Push/pop memory arena for each sample.
858 SCOPED_MEM(arena);
859
860 if (aovs) {
861 fs.mAovSchema->initFloatArray(aovs);
862 fs.mAovSchema->initFloatArray(deepAovs);
863 }
864
865 float depth = scene_rdl2::math::pos_inf;
866 float *depthPtr = nullptr;
867 if (schema.hasClosestFilter()) {
868 // we'll need a depth result
869 depthPtr = &depth;
870 }
871
872 pbr::ComputeRadianceAovParams aovParams { /* alpha = */ 0.f,
873 depthPtr, aovs, deepAovs, params->mDeepVolumeAovs };
874
875 // We add offset based on the machineId under multi-machine context.
876 // This is a core idea of how do we compute images by multi-machine.
877 // (relataed United States Patent : 11,176,721 : Nov/16/2021)
878 unsigned offset =
879 (isamp + params->mRealtimeSampleOfs) * params->mRenderNodeTotal
880 + params->mRenderNodeSampleOfs;
881
882 const pbr::Sample sample =
883 params->mSamplers[cameraId]->getPrimarySample(px, py,
884 fs.mFrameNumber,
885 offset,
886 fs.mDofEnabled,
887 params->mShutterBias[cameraId]);
888
889 ACCUMULATOR_UNPAUSE(*(params->mNonRenderDriverAccumulator));
890
891 scene_rdl2::math::Color subSample =
892 fs.mIntegrator->computeRadiance(pbrTls,
893 px, py,
894 int(offset),
895 cameraId,
896 params->mTotalNumSamples,
897 sample,
898 aovParams,
899 film->getDeepBuffer(),
900 film->getCryptomatteBuffer());
901
902 ACCUMULATOR_PAUSE(*(params->mNonRenderDriverAccumulator));
903
904#ifdef DEBUG_SAMPLE_REC_MODE
905 if (debugSamplesRecArray) {
906 debugSamplesRecArray->pixSamples(px, py, isamp,
907 subSample.r, subSample.g, subSample.b,
908 aovParams.mAlpha); // for debug
909 }
910 if (film->getAdaptiveRenderTilesTable()->isDebugPixByPos(px, py)) {
911 std::cerr << ">> RenderFrame.cc computeRadiance() result debug-pix(" << px << ',' << py << ')'
912 << " isamp:" << isamp
913 << " c(" << subSample.r << ',' << subSample.g << ',' << subSample.b << ')'
914 << " a:" << aovParams.mAlpha << std::endl;
915 }
916#endif // end DEBUG_SAMPLE_REC_MODE
917
918 // If alpha is negative then we're dealing with an
919 // invalid sample and we should skip adding it and
920 // incrementing the sampleCount.
921 const float alpha = aovParams.mAlpha;
922
923 if (alpha < 0.f) continue;
924 if (cameraId == 0) {
925 // Render buffer values are only computed by the primary camera
926
927 // copy color from unaligned subSample into
928 // aligned sample
929 scene_rdl2::fb_util::RenderColor sampleResult(subSample.r, subSample.g, subSample.b, alpha);
930
931 accRadiance += sampleResult;
932
933 // Record odd indexed samples in a secondary radiance buffer.
934 if (isamp & 1) {
935 accRadiance2 += sampleResult;
936 }
937
938 ++numAccSamples;
939 }
940
941 if (params->mAovNumFloats) {
942 unsigned aovFloatIndex = 0;
943
944 // Fill in the AOVs
945
946 for (std::size_t entryIdx = 0; entryIdx < schema.size(); ++entryIdx) {
947 const pbr::AovSchema::Entry& entry = schema[entryIdx];
948 aovFloatIndex += (entryIdx > 0) ? schema[entryIdx - 1].numChannels() : 0;
949
950 if (cameraId != entry.cameraId()) {
951 // This AOV is rendered with a different camera, skip.
952 continue;
953 }
954
955#ifdef DEBUG_SAMPLE_REC_MODE
956 if (debugSamplesRecArray &&
957 debugSamplesRecArray->mode() == DebugSamplesRecArray::Mode::LOAD) {
958 if (entry.type() == pbr::AOV_TYPE_BEAUTY) {
959 std::memcpy((void *)(aovs + aovFloatIndex),
960 (const void *)&sampleResult, sizeof(float) * 3);
961 }
962 if (entry.type() == pbr::AOV_TYPE_ALPHA) {
963 aovs[aovFloatIndex] = alpha;
964 }
965 }
966#endif // end DEBUG_SAMPLE_REC_MODE
967
968 if (schema.hasVarianceEntry(entryIdx)) {
969 // This entry has another aov keeping track of its variance. Let's add its variance.
970
971 // Map from aov to variance aov
972 for (auto varidx : schema.getSourceToVarianceAOVs(entryIdx)) {
973 MNRY_ASSERT(varidx >= 0);
974 film->addSampleStatistics(px, py, varidx, aovs + aovFloatIndex);
975 }
976 }
977
978 // if entry IS variance...skip. Its values
979 // get added through addSampleStatistics*
980 // calls.
981 if (schema.isVarianceEntry(entryIdx)) {
982 continue;
983 }
984
985 if (!schema.hasAovFilter()) {
986 // if there is no special aov filter, sum
987 // the values
988 for (unsigned j = 0; j < entry.numChannels(); ++j) {
989 const unsigned i = aovFloatIndex + j;
990 localAovs[i] += aovs[i];
991 }
992 } else {
993 switch (entry.filter()) {
994 case pbr::AOV_FILTER_AVG:
995 case pbr::AOV_FILTER_SUM:
996 for (unsigned j = 0; j < entry.numChannels(); ++j) {
997 const unsigned i = aovFloatIndex + j;
998 localAovs[i] += aovs[i];
999 }
1000 break;
1001 case pbr::AOV_FILTER_MIN:
1002 for (unsigned j = 0; j < entry.numChannels(); ++j) {
1003 const unsigned i = aovFloatIndex + j;
1004 localAovs[i] = std::min(localAovs[i], aovs[i]);
1005 }
1006 break;
1007 case pbr::AOV_FILTER_MAX:
1008 for (unsigned j = 0; j < entry.numChannels(); ++j) {
1009 const unsigned i = aovFloatIndex + j;
1010 localAovs[i] = std::max(localAovs[i], aovs[i]);
1011 }
1012 break;
1013 case pbr::AOV_FILTER_FORCE_CONSISTENT_SAMPLING:
1014 if (isamp < params->mConsistentSamplesPerPixel) {
1015 for (unsigned j = 0; j < entry.numChannels(); ++j) {
1016 const unsigned i = aovFloatIndex + j;
1017 localAovs[i] += aovs[i];
1018 }
1019 }
1020 break;
1021 case pbr::AOV_FILTER_CLOSEST:
1022 if (depth < localDepths[cameraId]) {
1023 for (unsigned j = 0; j < entry.numChannels(); ++j) {
1024 const unsigned i = aovFloatIndex + j;
1025 localAovs[i] = aovs[i];
1026 }
1027 }
1028 break;
1029 default:
1030 MNRY_ASSERT(0 && "unexpected filter type");
1031 break;
1032 }
1033 }
1034 }
1035
1036 // update localDepth after all aovs are filled
1037 if (depth < localDepths[cameraId]) {
1038 localDepths[cameraId] = depth;
1039 }
1040
1041 } // if (params->mAovNumFloats) {
1042 } // end sample loop
1043 } // end camera loop
1044
1045 if (numAccSamples) {
1046 // Update frame buffer. Scale the weights so that they are in
1047 // the same space as radiance.
1048 {
1049 EXCL_ACCUMULATOR_PROFILE(pbrTls, EXCL_ACCUM_ADD_SAMPLE_HANDLER);
1050 film->addSamplesToRenderBuffer(px, py, accRadiance,
1051 numAccSamples, &accRadiance2);
1052 }
1053 // update aovs
1054 if (params->mAovNumFloats) {
1055 EXCL_ACCUMULATOR_PROFILE(pbrTls, EXCL_ACCUM_AOVS);
1056 film->addSamplesToAovBuffer(px, py, localDepths, localAovs);
1057 }
1058 }
1059
1060 // conditional heat map collection
1061 MCRT_COMMON_CLOCK_CLOSE();
1062
1063 return true;
1064}

1202// static function
1203bool
1204RenderDriver::renderPixelVectorSamples(pbr::TLState *pbrTls,
1205 const unsigned startSampleIdx,
1206 const unsigned endSampleIdx,
1207 RenderSamplesParams *params,
1208 const TileGroup &group,
1209 const pbr::DeepBuffer *deepBuffer,
1210 pbr::CryptomatteBuffer *cryptomatteBuffer)
1211{
1212 unsigned px = params->mPx;
1213 unsigned py = params->mPy;
1214 const rndr::FrameState &fs = *reinterpret_cast<const rndr::FrameState *>(pbrTls->mFs);
1215
1216 //
1217 // Bundled execution start:
1218 //
1219
1220 // loop over all cameras for this pixel
1221 for (int cameraId = 0; cameraId < params->mNumCameras; cameraId++) {
1222
1223 uint32_t pixel = pbr::pixelLocationToUint32(px, py);
1224
1225 // Loop over samples in current pixel.
1226 unsigned numSamples = endSampleIdx - startSampleIdx;
1227 if (params->mTotalNumSamples < endSampleIdx) {
1228 numSamples = params->mTotalNumSamples - startSampleIdx;
1229 }
1230 pbr::RayState **rayStates;
1231 if (numSamples > 0) {
1232 rayStates = pbrTls->allocRayStates(numSamples);
1233 for (unsigned i = 0; i < numSamples; i++) {
1234 pbr::RayState *rs = rayStates[i];
1235 rs->mDeepDataHandle = pbr::nullHandle;
1236 rs->mCryptomatteDataHandle = pbr::nullHandle;
1237 }
1238 }
1239
1240 // invalid ray states will not be queued, we need to keep
1241 // a list of these and free them in bulk.
1242 scene_rdl2::alloc::Arena *arena = pbrTls->mArena;
1243 SCOPED_MEM(arena);

1248

1258 + params->mRenderNodeSampleOfs;

1273 rs->mSubpixel.mSubpixelX = sample.pixelX;
1274 rs->mSubpixel.mSubpixelY = sample.pixelY;
1275 rs->mTilePassAndFilm = pbr::makeTilePassAndFilm(params->mTileIdx, group.mPassIdx, params->mFilmIdx);
1276

1294 }

1297 ACCUMULATOR_UNPAUSE(*(params->mNonRenderDriverAccumulator));

1299 fs.mIntegrator->queuePrimaryRay(pbrTls,
1300 px, py,
1301 int(offset),
1302 cameraId,
1303 params->mTotalNumSamples,
1304 sample,
1305 rs);

1307 rayStatesToFree[numRayStatesToFree++] = rs;
1308 }
1309 ACCUMULATOR_PAUSE(*(params->mNonRenderDriverAccumulator));
1310 } // isub
1311
1312 // Bulk free of raystates that were not queued
1313 pbrTls->freeRayStates(numRayStatesToFree, rayStatesToFree);
1314
1315 } // end cameras loop
1316 //
1317 // Bundled execution end!
1318 //
1319 return true;
