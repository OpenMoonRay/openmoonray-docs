23RunLenBitTable::DumpMode
24RunLenBitTable::finalize()
25//
26// figure out which DumpMode is a best
27//
28{
29 unsigned totalAllMask = 0; // data size if all data use MASK mode
30 unsigned totalAllId = 0; // data size if all data use ID mode
31 for (unsigned itemId = 0; itemId < mMask.size(); ++itemId) {
32 unsigned currActivePixTotal = countBit64(itemId);
33 mActiveBitCount[itemId] = currActivePixTotal;
34
35 totalAllMask += 8;
36 totalAllId += (1 + currActivePixTotal);
37 }
38 // if runlength size is more than this, runlength is not a best choice.
39 unsigned totalLimit = std::min(totalAllMask, totalAllId);
40
41 unsigned totalRunLen = 0;
42 {
43 unsigned currMode = calcInitialMode(0);
44 unsigned nextMode;
45 unsigned startId = 0;
46 unsigned endId = findRunLenEnd(startId, currMode, nextMode);
47 while (1) {
48 totalRunLen++; // runLenCtrl
49 if (currMode == MODE_MASK) {
50 totalRunLen += ((endId - startId + 1) * 8); // full mask
51 } else {
52 for (unsigned id = startId; id <= endId; ++id) {
53 totalRunLen += (1 + mActiveBitCount[id]); // id dump
54 }
55 }
56 if (totalRunLen >= totalLimit) {
57 // runLen is bigger, we don't need to test runLen any more.
58 break; // early exit
59 }
60
61 currMode = nextMode;
62 if ((startId = endId + 1) >= mMask.size()) break;
63
64 endId = findRunLenEnd(startId, currMode, nextMode);
65 }
66 }
67
68 DumpMode mode;
69 if (totalLimit <= totalRunLen) {
70 if (totalAllMask < totalAllId) mode = DumpMode::ALLMASK_DUMP;
71 else mode = DumpMode::ALLID_DUMP;
72 } else {
73 mode = DumpMode::RUNLEN_DUMP;
74 if (totalAllMask < totalAllId) {
75 if (totalAllMask < totalRunLen) mode = DumpMode::ALLMASK_DUMP;
76 } else {
77 if (totalAllId < totalRunLen) mode = DumpMode::ALLID_DUMP;
78 }
79 }
80
81 // keep data size for statistical info dump purpose
82 mDataSize = 0;
83 switch (mode) {
84 case DumpMode::ALLMASK_DUMP: mDataSize = totalAllMask; break;
85 case DumpMode::ALLID_DUMP: mDataSize = totalAllId; break;
86 case DumpMode::RUNLEN_DUMP: mDataSize = totalRunLen; break;
87 default : break; // never happened
88 }
89
90# ifdef DEBUG_MSG_FINALIZE
91 size_t ver1Size = totalAllMask; // ver1 is using allMask logic
92 int deltaSize = (int)mDataSize - (int)ver1Size; // how small is this compare with ver1 result
93 float ratio = (float)mDataSize / (float)ver1Size; // ratio of how small from ver1
94 std::cerr << ">> RunLenBitTable.cc finalize() "
95 << " AllMask:" << totalAllMask
96 << " AllId:" << totalAllId
97 << " Limit:" << totalLimit
98 << " RunLen:" << totalRunLen
99 << " (ver1:" << deltaSize << ' '
100 << std::setw(5) << std::fixed << std::setprecision(3) << ratio << ')'
101 << " -> " << showDumpMode(mode) << ':' << mDataSize
102 << std::endl;
103# endif // end DEBUG_MSG_FINALIZE
104
105 return mode;
106}

169void
170RunLenBitTable::randomTestData(const unsigned minActiveTotal, const unsigned maxActiveTotal)
171// for debug function
172{
173 std::random_device rnd;
174 std::mt19937 mt(rnd());
175 std::uniform_int_distribution<> randGen(minActiveTotal, maxActiveTotal);
176 std::uniform_int_distribution<> randGenPix(0, 63);
177
178 for (unsigned maskId = 0; maskId < mMask.size(); ++maskId) {
179 unsigned activePixTotal = std::min(randGen(mt), 64);
180 if (activePixTotal >= 1) {
181 uint64_t &currMask = mMask[maskId];
182 while (1) {
183 unsigned currPixId = randGenPix(mt);
184 currMask |= ((uint64_t)0x1 << currPixId);
185 if (countBit64(maskId) == activePixTotal) break;
186 }
187 }
188 }

318unsigned
319RunLenBitTable::calcInitialMode(const unsigned startId) const
320{
321 unsigned currCount = static_cast<unsigned>(mActiveBitCount[startId]);
322 if (currCount < THRESH_ACTIVE_PIX_TOTAL) {
323 return MODE_ID;
324 }
325 else if (currCount > THRESH_ACTIVE_PIX_TOTAL) {
326 return MODE_MASK;
327 }
328
329 //
330 // Current count is on the border of thresh number and we can pick up both of MODE_ID and MODE_MASK.
331 // However best mode is depend on the next item's condition. So we should check next item's activeBitCount
332 // and make sure which mode is the best for current item.
333 //
334 unsigned nextCount = THRESH_ACTIVE_PIX_TOTAL;
335 for (unsigned currId = startId + 1; currId < mActiveBitCount.size(); ++currId) {
336 nextCount = mActiveBitCount[currId];
337 if (nextCount != THRESH_ACTIVE_PIX_TOTAL) break;
338 }
339
340 if (nextCount < THRESH_ACTIVE_PIX_TOTAL) return MODE_ID;
341 return MODE_MASK;
342}

344unsigned
345RunLenBitTable::findRunLenEnd(const unsigned startId, const unsigned currMode, unsigned &nextMode) const
346//
347// find current run length section end index and also find next start section mode
348//
349{
350 unsigned endId = startId;
351
352 unsigned maxId = mMask.size() - 1;
353 if (maxId - startId + 1 > MAX_RUNLEN) {
354 maxId = startId + MAX_RUNLEN - 1;
355 }
356
357 if (currMode == MODE_MASK) {
358 for (unsigned itemId = startId + 1; itemId <= maxId; ++itemId) {
359 if (mActiveBitCount[itemId] < THRESH_ACTIVE_PIX_TOTAL) {
360 nextMode = MODE_ID;
361 return endId;
362 }
363 endId = itemId;
364 }
365 } else {
366 for (unsigned itemId = startId + 1; itemId <= maxId; ++itemId) {
367 if (mActiveBitCount[itemId] > THRESH_ACTIVE_PIX_TOTAL) {
368 nextMode = MODE_MASK;
369 return endId;
370 }
371 endId = itemId;
372 }
373 }
374
375 if (endId + 1 < mMask.size()) {
376 nextMode = calcInitialMode(endId + 1);
377 }
378 return endId;
379}

381void
382RunLenBitTable::enqSingleRunLenChunk(const unsigned currMode,
383 const unsigned startId, const unsigned endId,
384 VContainerEnq &vContainerEnq) const
385{
386 unsigned runLen = endId - startId + 1;
387 MNRY_ASSERT(0 < runLen && runLen <= 128);
388
389 // enqueue runlength control code first.
390 unsigned char runLenCtrl = (unsigned char)currMode | (unsigned char)((runLen - 1) & 0x7f);
391 vContainerEnq.enqUChar(runLenCtrl);
392
393# ifdef DEBUG_MSG_RUNLEN_ENQ
394 std::cerr << ">> RunLenBitTable.cc enqSingleRunLenChunk() runLen:" << runLen
395 << " startId:" << startId << " endId:" << endId
396 << " runLenCtrl:0x" << std::hex << std::setw(2) << std::setfill('0') << (int)runLenCtrl << std::dec
397 << std::endl;
398# endif // end DEBUG_MSG_RUNLEN_ENQ
399
400 if (currMode == MODE_MASK) {
401 for (unsigned id = startId; id <= endId; ++id) {
402 vContainerEnq.enqMask64(mMask[id]);
403 }
404 } else {
405 for (unsigned id = startId; id <= endId; ++id) {
406 enqSingleMaskById(id, vContainerEnq);
407 }
408 }

411void
412RunLenBitTable::enqSingleMaskById(const unsigned id,
413 VContainerEnq &vContainerEnq) const
414{
415 // enqueue number of active pixels inside one active pixel mask.
416 // one active pixel mask is 8x8 pixels and max is 64.
417 vContainerEnq.enqUChar(static_cast<unsigned char>(mActiveBitCount[id]));
418
419 uint64_t currMask = mMask[id];
420 for (unsigned shift = 0; shift < 64; ++shift) {
421 if (!currMask) break; // early exit
422 if (currMask & (uint64_t)0x1) {
423 // uchar is enough to keep pixel location (=shift value)
424 vContainerEnq.enqUChar(static_cast<unsigned char>(shift));
425 }
426 currMask >>= 1;
427 }
428}

430unsigned
431RunLenBitTable::deqSingleRunLenChunk(const unsigned startId,
432 VContainerDeq &vContainerDeq)
433{
434 unsigned char runLenCtrl;
435 vContainerDeq.deqUChar(runLenCtrl);
436
437 unsigned currMode = runLenCtrl & ~(0x7f);
438 unsigned runLen = (runLenCtrl & 0x7f) + 1;
439
440 unsigned endId = startId + runLen - 1;
441
442# ifdef DEBUG_MSG_RUNLEN_DEQ
443 std::cerr << ">> RunLenBitTable.cc deqSingleRunLenChunk()"
444 << " runLenCtrl:0x" << std::hex << std::setw(2) << std::setfill('0') << (int)runLenCtrl << std::dec
445 << " runLen:" << runLen << " startId:" << startId << " endId:" << endId << std::endl;
446# endif // end DEBUG_MSG_RUNLEN_DEQ
447
448 if (currMode == MODE_MASK) {
449 for (unsigned id = startId; id <= endId; ++id) {
450 vContainerDeq.deqMask64(mMask[id]);
451 }
452 } else {
453 for (unsigned id = startId; id <= endId; ++id) {
454 deqSingleMaskById(id, vContainerDeq);
455 }
456 }
457
458 return endId + 1;
459}

461void
462RunLenBitTable::deqSingleMaskById(const unsigned id,
463 VContainerDeq &vContainerDeq)
464{
465 vContainerDeq.deqUChar(mActiveBitCount[id]);
466
467 uint64_t currMask = (uint64_t)0x0;
468 for (unsigned i = 0; i < (unsigned)mActiveBitCount[id]; ++i) {
469 unsigned char shift;
470 vContainerDeq.deqUChar(shift);
471 currMask |= ((uint64_t)0x1 << shift);
472 }
473 mMask[id] = currMask;
474}
