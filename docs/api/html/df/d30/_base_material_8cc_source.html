67void
68BaseMaterial::shade(const Material* self, moonray::shading::TLState *tls,
69 const State& state, BsdfBuilder& bsdfBuilder)
70{
71 const BaseMaterial* me = static_cast<const BaseMaterial*>(self);
72 moonray::shading::Bsdf *bsdf = const_cast<moonray::shading::Bsdf*>(bsdfBuilder.getBsdf());
73
74 scene_rdl2::alloc::Arena *arena = getArena(tls);
75
76 // Fully opaque and no specular or other non-diffuse terms by default
77 float opacityFactor = 1.0f;
78 Color directionalDiffuseColor(0.0f);
79 Color specularColor(0.0f);
80 float specularRoughness = 0.0f;
81 float retroreflectivity(0.0f);
82 Color transmissionColor(0.0f);
83 Color omTransmissionColor(1.0f);
84
85 Vec3f anisotropicDirection(0.0f);
86 float anisotropy = evalFloat(me, attrAnisotropy, tls, state);
87
88 // Get minimum roughness used to apply roughness clamping.
89 const Vec2f minRoughnessAniso = state.getMinRoughness();
90 const float minRoughness = min(minRoughnessAniso.x, minRoughnessAniso.y);
91
92 // Only do opacity, specular, etc. if we are not on a caustics path or unless
93 // we really want caustics
94 if (!state.isCausticPath() || me->get(attrCastsCaustics)) {
95
96 if (me->get(attrOpacity)) {
97 // Transparency (== 1 - opacity) takes priority over everything else. All
98 // the other terms of the Bsdf will be proportional to opacity
99 opacityFactor = evalFloat(me, attrOpacityFactor, tls, state);
100 if (!isEqual(opacityFactor, 1.0f)) {
101 auto lobe = arena->allocWithArgs<moonray::shading::MirrorTransmissionBsdfLobe>(state.getN(),
102 1.f,
103 1.f,
104 sWhite);
105 lobe->setScale(Color(1.0f - opacityFactor));
106 bsdf->addLobe(lobe);
107 }
108
109 // Early out if fully transparent
110 if (isZero(opacityFactor)) {
111 return;
112 }
113 }
114
115 // Evaluate specular and other caustic-sensitive components
116 specularColor = evalColorComponent(me, attrSpecular, attrSpecularFactor,
117 attrSpecularColor, tls, state);
118
119 specularRoughness = evalFloat(me, attrSpecularRoughness, tls, state);
120
121 retroreflectivity = evalFloat(me, attrRetroreflectivity, tls, state);
122
123 directionalDiffuseColor = evalColorComponent(me, attrDirectionalDiffuse,
124 attrDirectionalDiffuseFactor, attrDirectionalDiffuseColor, tls, state);
125
126 transmissionColor = evalColorComponent(me, attrTransmission,
127 attrTransmissionFactor, attrTransmissionColor, tls, state);
128 omTransmissionColor = sWhite - transmissionColor;
129
130 // Anisotropic direction
131 if (!isZero(anisotropy)) {
132 const Vec3f T = normalize(state.getdPds());
133 const ReferenceFrame frame(state.getN(), T);
134 const Vec2f dir = evalVec2f(me, attrAnisotropicDirection, tls, state);
135 anisotropicDirection = frame.localToGlobal(
136 normalize(Vec3f(dir.x, dir.y, 0.0f)));
137 }
138 }
139
140
141 // We use a top specular lobe and use it to attenuate energy from lower lobes
142 moonray::shading::Fresnel *omSpecFresnel = nullptr;
143
144 // TODO: avoid doing these eval() calls if the result is unused
145 const Vec3f N = evalNormal(me, attrInputNormal, attrInputNormalDial,
146 attrInputNormalSpace, tls, state);
147 const float fresnelFactor = (me->get(attrUseFresnel) ?
148 evalFloat(me, attrFresnelFactor, tls, state) : 0.0);
149 const ShaderIor ior(state, me->get(attrIndexOfRefraction));
150
151 const bool isTransmissive = !isBlack(transmissionColor);
152 // Energy Compensation Params
153 float favg, favgInv;
154 moonray::shading::averageFresnelReflectance(
155 ior.getTransmitted()/ior.getIncident(),
156 favg, favgInv);
157
158 // Top Specular lobe
159 if (!isBlack(specularColor) && (retroreflectivity < 1.0f)) {
160 moonray::shading::BsdfLobe *lobe;
161
162 // Apply roughness clamping
163 float roughness = max(specularRoughness, minRoughness);
164
165 if (roughness < sEpsilon) {
166 lobe = arena->allocWithArgs<moonray::shading::MirrorReflectionBsdfLobe>(N);
167 } else {
168 if (isZero(anisotropy)) {
169 lobe = arena->allocWithArgs<moonray::shading::CookTorranceBsdfLobe>(N, roughness,
170 sWhite*favg, sWhite*favgInv,
171 ior.getIncident(),
172 ior.getTransmitted(),
173 isTransmissive); //coupledWithTransmission
174
175 } else {
176 // Compute anisotropy using un-clamped roughness, then clamp
177 float uRoughness = anisotropy > 0.0f ?
178 specularRoughness * (1.0f - anisotropy) : specularRoughness;
179 float vRoughness = anisotropy < 0.0f ?
180 specularRoughness * (1.0f + anisotropy) : specularRoughness;
181 uRoughness = max(uRoughness, minRoughness);
182 vRoughness = max(vRoughness, minRoughness);
183 lobe = arena->allocWithArgs<moonray::shading::AnisoCookTorranceBsdfLobe>(N,
184 anisotropicDirection, uRoughness, vRoughness);
185 roughness = 0.5f * (uRoughness + vRoughness);
186 }
187 }
188
189#if USE_SCHLICK
190 auto specFresnel = arena->allocWithArgs<moonray::shading::SchlickFresnel>(
191 specularColor, fresnelFactor, ior.getRatio());
192#else
193 auto specFresnel = arena->allocWithArgs<moonray::shading::DielectricFresnel>(
194 ior.getIncident(), ior.getTransmitted());
195#endif
196 lobe->setFresnel(specFresnel);
197
198 // Energy conservation: dim specular based on opacity/retro-reflectivity
199 lobe->setScale(Color((1.0f - retroreflectivity) * opacityFactor));
200#if USE_SCHLICK
201 omSpecFresnel = arena->allocWithArgs<moonray::shading::OneMinusRoughSchlickFresnel>(
202 specFresnel, roughness);
203#else
204 omSpecFresnel = arena->allocWithArgs<moonray::shading::OneMinusFresnel>(specFresnel);
205#endif
206 if (me->get(attrIridescence)) {
207 const float iridescenceFac = me->get(attrIridescenceFactor);
208 const Color iridescencePrimary = me->get(attrIridescencePrimaryColor);
209 const Color iridescenceSecondary = me->get(attrIridescenceSecondaryColor);
210 const bool iridescenceFlipHue = me->get(attrIridescenceFlipHueDirection);
211 const float iridescenceThickness = evalFloat(me, attrIridescenceThickness, tls, state);
212 const float iridescenceExponent = evalFloat(me, attrIridescenceExponent, tls, state);
213 const float iridescenceAt0 = evalFloat(me, attrIridescenceAt0, tls, state);
214
215 // Pass along the mirror/glossy lobe created above to the iridescence lobe
216 lobe = arena->allocWithArgs<moonray::shading::IridescenceBsdfLobe>(lobe,
217 N,
218 iridescenceFac,
219 static_cast<ispc::SHADING_IridescenceColorMode>(0), // use hue interpolation
220 iridescencePrimary,
221 iridescenceSecondary,
222 iridescenceFlipHue,
223 static_cast<ispc::ColorRampControlSpace>(0), // ramp interpolation mode
224 0, // ramp num points
225 nullptr, // ramp positions
226 nullptr, // ramp interpolators
227 nullptr, // ramp colors
228 iridescenceThickness,
229 iridescenceExponent,
230 iridescenceAt0,
231 1.0f);
232 }
233
234 lobe->setLabel(aovSpecular);
235 bsdf->addLobe(lobe);
236 }
237
238 // Retroreflection Lobe
239 if (retroreflectivity > 0.0f) {
240 moonray::shading::BsdfLobe *lobe;
241 const float roughness = max(specularRoughness, minRoughness);
242 if (roughness < sEpsilon) {
243 lobe = arena->allocWithArgs<moonray::shading::MirrorRetroreflectionBsdfLobe>(N);
244 } else {
245 lobe = arena->allocWithArgs<moonray::shading::RetroreflectionBsdfLobe>(N, roughness);
246 }
247 lobe->setScale(Color(opacityFactor * retroreflectivity));
248 lobe->setLabel(aovSpecular);
249 bsdf->addLobe(lobe);
250 }
251
252 // Directional Diffuse
253 if (!isBlack(directionalDiffuseColor)) {
254 moonray::shading::BsdfLobe *lobe;
255
256 // Evaluate roughness, then apply roughness clamping
257 const float ddRoughness = evalFloat(me, attrDirectionalDiffuseRoughness,
258 tls, state);
259 float roughness = max(ddRoughness, minRoughness);
260
261 if (roughness < sEpsilon) {
262 lobe = arena->allocWithArgs<moonray::shading::MirrorReflectionBsdfLobe>(N);
263 } else {
264 if (isZero(anisotropy)) {
265 lobe = arena->allocWithArgs<moonray::shading::CookTorranceBsdfLobe>(N, roughness);
266 } else {
267 // Compute anisotropy using un-clamped roughness, then clamp
268 float uRoughness = anisotropy > 0.0f ?
269 ddRoughness * (1.0f - anisotropy) : ddRoughness;
270 float vRoughness = anisotropy < 0.0f ?
271 ddRoughness * (1.0f + anisotropy) : ddRoughness;
272 uRoughness = max(uRoughness, minRoughness);
273 vRoughness = max(vRoughness, minRoughness);
274 lobe = arena->allocWithArgs<moonray::shading::AnisoCookTorranceBsdfLobe>(N,
275 anisotropicDirection, uRoughness, vRoughness);
276 }
277 }
278 auto fresnel = arena->allocWithArgs<moonray::shading::SchlickFresnel>(
279 directionalDiffuseColor, fresnelFactor, ior.getRatio());
280 lobe->setFresnel(fresnel);
281
282 // Energy conservation: dim based on transmission and opacity
283 lobe->setScale(opacityFactor * omTransmissionColor);
284
285 if (omSpecFresnel) {
286 // Energy conservation: dim Dd based on 1-F(top lobe)
287 moonray::shading::BsdfLobe *underLobe = arena->allocWithArgs<moonray::shading::UnderBsdfLobe>(lobe, N);
288 underLobe->setFresnel(omSpecFresnel);
289 underLobe->setLabel(aovDirectionalDiffuse);
290 bsdf->addLobe(underLobe);
291 } else {
292 lobe->setLabel(aovDirectionalDiffuse);
293 bsdf->addLobe(lobe);
294 }
295 }
296
297 // Translucency
298 const Color translucencyColor = evalColorComponent(me, attrTranslucency,
299 attrTranslucencyFactor, attrTranslucencyColor, tls, state);
300 if (!isBlack(translucencyColor)) {
301 // Check diffuse_depth and use the a lambert bsdf approximation
302 // TODO: Should we check if the ray footprint is larger than the
303 // scattering radius to use the approx instead ?
304 if (!state.isHifi()) {
305
306 // TODO: implement and use Jensen et al. 2001 Brdf approximation
307 // TODO: Apply omSpecFresnel
308 auto lobe = arena->allocWithArgs<moonray::shading::LambertBsdfLobe>(N, translucencyColor, true);
309 // Energy conservation: dim based on transmission and opacity
310 lobe->setScale(opacityFactor * omTransmissionColor);
311 if (omSpecFresnel) {
312 // Energy conservation: dim translucency based on 1-F(top lobe)
313 lobe->setFresnel(omSpecFresnel);
314 }
315 lobe->setLabel(aovTranslucency);
316 bsdf->addLobe(lobe);
317 } else {
318 // TODO: modify DipoleBssrdf xtor to accommodate this;
319 const Color translucencyRadius =
320 evalColor(me, attrTranslucencyFalloff, tls, state) *
321 evalFloat(me, attrTranslucencyRadius, tls, state);
322 // Energy conservation: dim based on transmission and opacity
323 auto bssrdf = moonray::shading::createBSSRDF(ispc::SUBSURFACE_DIPOLE_DIFFUSION, arena, N,
324 opacityFactor * omTransmissionColor * translucencyColor,
325 translucencyRadius, me, nullptr);
326#if 0
327 // Use physical parameters interface
328 bssrdf = new DipoleBssrdf(N, 1.0f, translucencyColor,
329 translucencyRadius, 0.001);
330
331 // Skin1 from Jensen et al. 2001.
332 bssrdf = new DipoleBssrdf(N, 1.0f,
333 Color3s(0.032f, 0.17f, 0.48f), Color3s(0.74f, 0.88f, 1.01f), 0.001);
334#endif
335 if (omSpecFresnel) {
336 // Energy conservation: dim translucency based on 1-F(top lobe)
337 bssrdf->setTransmissionFresnel(omSpecFresnel);
338 }
339 bssrdf->setLabel(aovTranslucency);
340 bsdf->setBssrdf(bssrdf);
341 }
342 }
343
344 // Transmission
345 if (isTransmissive) {
346
347 moonray::shading::BsdfLobe *lobe;
348 // Specular Transmission
349 // Apply roughness clamping
350 float roughness = max(specularRoughness, minRoughness);
351
352 if (roughness < sEpsilon || isOne(ior.getRatio())) {
353 lobe = arena->allocWithArgs<moonray::shading::MirrorTransmissionBsdfLobe>(N,
354 ior.getIncident(),
355 ior.getTransmitted(),
356 transmissionColor);
357 } else {
358 lobe = arena->allocWithArgs<moonray::shading::TransmissionCookTorranceBsdfLobe>(
359 N, roughness,
360 ior.getIncident(),
361 ior.getTransmitted(),
362 transmissionColor,
363 favg, favgInv);
364 }
365 if (omSpecFresnel) {
366 // Energy conservation: dim transmission based on 1-F(top lobe)
367 lobe->setFresnel(omSpecFresnel);
368 }
369 // Energy conservation: dim based on opacity
370 lobe->setScale(Color(opacityFactor));
371 lobe->setLabel(aovTransmission);
372 bsdf->addLobe(lobe);
373 }
374
375 // Diffuse
376 const Color diffuseColor = evalColorComponent(me, attrDiffuse,
377 attrDiffuseFactor, attrDiffuseColor, tls, state);
378 if (!isBlack(diffuseColor)) {
379 auto lobe = arena->allocWithArgs<moonray::shading::LambertBsdfLobe>(N, diffuseColor, true);
380 if (omSpecFresnel) {
381 // Energy conservation: dim diffuse based on 1-F(top lobe)
382 lobe->setFresnel(omSpecFresnel);
383 }
384 // Energy conservation: dim based on transmission and opacity
385 lobe->setScale(opacityFactor * omTransmissionColor);
386 lobe->setLabel(aovDiffuse);
387 bsdf->addLobe(lobe);
388 }
389
390 // Translucent Diffuse
391 // NOTE - We are 'adding' this lobe directly to support legacy material
392 // conversions where a translucent diffuse material could be 'added' with
393 // a base material.
394 // Ideally, there'd be support for energy conservation similar to the 'specular transmission'
395 // lobe in terms of reducing the lambertian diffuse by 'oneMinusColor'
396 const Color translucentDiffuseColor = evalColorComponent(me,
397 attrTranslucentDiffuse,
398 attrTranslucentDiffuseFactor,
399 attrTranslucentDiffuseColor,
400 tls,
401 state);
402 if (!isBlack(translucentDiffuseColor)) {
403 auto lobe = arena->allocWithArgs<moonray::shading::LambertBsdfLobe>(-N, translucentDiffuseColor, false);
404 if (omSpecFresnel) {
405 // Energy conservation: dim transmission based on 1-F(top lobe)
406 lobe->setFresnel(omSpecFresnel);
407 }
408 lobe->setScale(Color(opacityFactor));
409 bsdf->addLobe(lobe);
410 }
411
412
413 // Self-emission
414 const Color emissionColor = evalColorComponent(me, attrEmission,
415 attrEmissionFactor, attrEmissionColor, tls, state);
416 // Energy conservation: dim based on opacity
417 bsdf->setSelfEmission(opacityFactor * emissionColor);
418}
