40void
41RenderOutputDriver::Impl::write(const bool checkpointOutput,
42 const bool checkpointOutputMultiVersion,
43 const pbr::DeepBuffer *deepBuffer,
44 pbr::CryptomatteBuffer *cryptomatteBuffer,
45 const scene_rdl2::fb_util::HeatMapBuffer *heatMap,
46 const scene_rdl2::fb_util::FloatBuffer *weightBuffer,
47 const scene_rdl2::fb_util::RenderBuffer *renderBufferOdd,
48 const std::vector<scene_rdl2::fb_util::VariablePixelBuffer> *aovBuffers,
49 const std::vector<scene_rdl2::fb_util::VariablePixelBuffer> *displayFilterBuffers,
50 const unsigned checkpointTileSampleTotals,
51 ImageWriteCache *cache,
52 scene_rdl2::grid_util::Sha1Gen::Hash *hashOut) const
53//
54// weightBuffer : only has valid info when RenderOutputDriver has weightAOV.
55// checkpointOutputMultiVersion : runtime multi-version checkpoint file output condition.
56// This flag only used under checkpointOutput = on
57//
58// -- Write image by background thread --
59// Now this write() function has 3 different run mode. STD, ENQ and DEQ.
60// STD : standard output mode which is not using background write mode.
61// ENQ : same logic of STD but not create file and just create ImageWriteCache data instead.
62// DEQ : create file from ImageWriteCache.
63//
64// STD runMode is still used for final image output stage (i.e. not a checkpoint file out).
65// And also used by checkpoint write with "checkpoint_bg_write" = false situation.
66//
67// Background checkpoint write is done by 2 passes. ENQ pass and DEQ pass.
68// ENQ pass is done by renderDriver thread and create ImageWriteCache data. This is pretty quick.
69// DEQ pass is done by ImageWriteDriver thread independent from renderDriver thread tree by parallel.
70// Scene variable "checkpoint_bg_write" is used for disable/enable of background write.
71// Default is "true".
72//
73// DeepFile output is only working under STD mode at this moment.
74//
75// -- Two stage output mode --
76// In order to reduce the risk of kill/terminate signal during multiple image write situation,
77// we have special logic calls "two stage output" mode. If multiple images are not update
78// correctly due to terminate render process, all generated image files are not sync and
79// we can not resume from this output images.
80// This "two stage output" creates temporary file first and after that copy them to the final
81// destination as temporary name. Then rename from destination temporary name to destination final
82// name. If your renderOutput creates multiple files, all files are created as temporary files first
83// then after finish all files are ready, copy to the final destination as temporary name files.
84// After that, all destination temporary files are renamed to final name files.
85// Typically final rename stage is much faster than directory create that files. So this solution can
86// greatly reduce the risk of generate out of sync situation of the multiple image output.
87//
88// If "two stage output" mode is enabled, this write() function try to create file as tmp file.
89// copy and rename stage is done by later. (See ImageWriteCache::allFinalize*File() API).
90// moonray clean up the temporary files by itself but some case there is some possibility to leave
91// the tmp file unfortunately.
92// if you find the file like "*.part" with old timestamp, this is a garbage file and need
93// to be removed. This file is destination temp file and usually renamed to final filename after
94// copy is done.
95//
96// This "two stage output" is only support non DeepFile output so far.
97//
98{
99 if (checkpointOutputMultiVersion && mOverwriteCheckpoint) {
100 // checkpointOutputMultiVersion only available when checkpointOverwrite=false situation.
101 return; // early exit. This behavior is requested. (See RenderDriver::checkpointFileOutput()).
102 }
103
104 std::vector<std::string> &errors = (cache)? cache->getErrors(): mErrors;
105 std::vector<std::string> &infos = (cache)? cache->getInfos(): mInfos;
106
107 scene_rdl2::grid_util::Sha1Gen sha1Gen;
108 scene_rdl2::grid_util::Sha1Gen *sha1GenPtr = (hashOut) ? &sha1Gen : nullptr;
109
110 if (cache) cache->timeStart();
111
112 ImageWriteCache::Mode runMode = ImageWriteCache::runMode(cache);
113
114 if (runMode != ImageWriteCache::Mode::DEQ) { // STD/ENQ
115 int predefinedWidth = 0;
116 int predefinedHeight = 0;
117 if (deepBuffer) {
118 predefinedWidth = deepBuffer->getWidth();
119 predefinedHeight = deepBuffer->getHeight();
120 }
121
122 //------------------------------
123 // A 2D buffer needed by DeepBuffer and Cryptomatte that holds
124 // the number of samples per pixel, which can vary when using adaptive sampling.
125 scene_rdl2::fb_util::PixelBuffer<unsigned> samplesCount;
126 if (deepBuffer || cryptomatteBuffer) {
127 const Film& film = rndr::getRenderDriver()->getFilm(0);
128 film.fillPixelSampleCountBuffer(samplesCount);
129 }
130 if (cache) cache->timeRec(0); // record timing into position id = 0
131
132 //------------------------------
133 if (cryptomatteBuffer) {
134 cryptomatteBuffer->finalize(samplesCount);
135 }
136 if (cache) cache->timeRec(1); // record timing into position id = 1
137
138 //------------------------------
139 FileNameParam fileNameParam(checkpointOutput,
140 checkpointOutputMultiVersion,
141 mOverwriteCheckpoint,
142 mFinalMaxSamplesPerPixel);
143
144 RenderOutputWriter::CallBackCheckpointResumeMetadata callBackMetadata =
145 [&](const unsigned checkpointTileSampleTotals) -> std::vector<std::string> {
146 // callback for checkpointResumeMetaData
147 std::vector<std::string> attrTbl;
148 if (mCheckpointRenderActive) {
149 attrTbl = writeCheckpointResumeMetadata(checkpointTileSampleTotals);
150 }
151 return attrTbl;
152 };
153
154 RenderOutputWriter writer(runMode,
155 cache,
156 checkpointTileSampleTotals,
157 predefinedWidth, predefinedHeight,
158 &mFiles,
159 &fileNameParam,
160 callBackMetadata,
161 cryptomatteBuffer, heatMap, weightBuffer, renderBufferOdd,
162 aovBuffers, displayFilterBuffers,
163 errors, infos,
164 sha1GenPtr);
165 if (cache) cache->timeRec(2); // record timing into position id = 2
166
167 writer.main();
168 if (cache) cache->timeRec(3); // record timing into position id = 3
169
170 // Here we "unfinalize" the cryptomatte buffer, putting it into a state ready for further accumulation
171 // of samples following a checkpoint (which would have incurred a finalize operation)
172 if (cryptomatteBuffer) {
173 cryptomatteBuffer->unfinalize(samplesCount);
174 }
175
176 if (deepBuffer && runMode == ImageWriteCache::Mode::STD) {
177 deepWrite(checkpointOutput, checkpointOutputMultiVersion, checkpointTileSampleTotals,
178 samplesCount, deepBuffer);
179 }
180
181 } else {
182 if (cache) cache->timeRec(0); // record timing into position id = 0
183 if (cache) cache->timeRec(1); // record timing into position id = 1
184
185 FileNameParam fileNameParam(checkpointOutputMultiVersion);
186 RenderOutputWriter writer(cache,
187 &mFiles,
188 &fileNameParam,
189 errors, infos,
190 sha1GenPtr);
191 if (cache) cache->timeRec(2); // record timing into position id = 2
192
193 writer.main();
194 if (cache) cache->timeRec(3); // record timing into position id = 3
195 }
196
197 if (cache) {
198 cache->timeRec(4); // record timing into position id = 4
199 cache->timeEnd();
200 // std::cerr << ">> RenderOutputDriver.cc " << cache->timeShow() << std::endl; // useful debug dump
201 }
202
203 if (hashOut) {
204 *hashOut = sha1GenPtr->finalize();
205 }
206}

208void
209RenderOutputDriver::Impl::deepWrite(const bool checkpointOutput,
210 const bool checkpointOutputMultiVersion,
211 const unsigned checkpointTileSampleTotals,
212 const scene_rdl2::fb_util::PixelBuffer<unsigned> &samplesCount,
213 const pbr::DeepBuffer *deepBuffer) const
214{
215 std::vector<std::string> &errors = mErrors;
216 std::vector<std::string> &infos = mInfos;
217
218 int deepAOV = 0;
219 for (const auto &f: mFiles) {
220
221 // We are not supporting resume render of deep data but we support generating checkpoint deep file.
222 std::string filename = ((checkpointOutput) ?
223 ((checkpointOutputMultiVersion) ?
224 RenderOutputWriter::generateCheckpointMultiVersionFilename(f,
225 mOverwriteCheckpoint,
226 mFinalMaxSamplesPerPixel,
227 checkpointTileSampleTotals) :
228 f.mCheckpointName) :
229 f.mName);
230
231 // Make sure all of this file's outputs are "deep"
232 // You cannot mix "deep" and non-deep outputs in the same file! Deep
233 // images are output using OpenDCX and non-deep images may be output with
234 // a different library, e.g. OIIO.
235 bool hasDeep = false;
236 bool hasNonDeep = false;
237 int numOutputs = 0;
238 for (const auto &i: f.mImages) {
239 for (const auto &e: i.mEntries) {
240 const scene_rdl2::rdl2::RenderOutput *ro = e.mRenderOutput;
241 if (ro->getOutputType() == std::string("deep")) {
242 hasDeep = true;
243 } else {
244 hasNonDeep = true;
245 }
246 numOutputs++;
247 }
248 }
249 if (!hasDeep && hasNonDeep) {
250 // non-deep images are output elsewhere
251 // need to skip over non-deep outputs but increment aov index
252 for (const auto &i: f.mImages) {
253 for (const auto &e: i.mEntries) {
254 if (e.mAovSchemaId != pbr::AOV_SCHEMA_ID_UNKNOWN) {
255 deepAOV++;
256 }
257 }
258 }
259 continue;
260 }
261 if (hasDeep && hasNonDeep) {
262 errors.push_back
263 (scene_rdl2::util::buildString("ERROR: output file '", filename, "' has a mixture of deep and non-deep images."));
264 continue; // next!
265 }
266
267 // Figure out what we're outputting in this deep file.
268 // It can be any combination of aovs, collect these up from the
269 // images + entries.
270 std::vector<int> outputAOVs;
271 std::vector<std::string> outputAOVChannelNames;
272 const scene_rdl2::rdl2::Metadata *rdlMetadata = nullptr;
273
274 const scene_rdl2::rdl2::SceneVariables& vars = f.mImages[0].mEntries[0].mRenderOutput->getSceneClass().getSceneContext()->getSceneVariables();
275 const scene_rdl2::math::HalfOpenViewport aperture = vars.getRezedApertureWindow();
276 const scene_rdl2::math::HalfOpenViewport region = vars.getRezedRegionWindow();
277
278 for (const auto &i: f.mImages) {
279 for (const auto &e: i.mEntries) {
280 const scene_rdl2::rdl2::RenderOutput *ro = e.mRenderOutput;
281 const scene_rdl2::rdl2::SceneObject *metadata = ro->getExrHeaderAttributes();
282 if (metadata) {
283 // use first valid metadata we find, same as flat case above
284 rdlMetadata = metadata->asA<scene_rdl2::rdl2::Metadata>();
285 }
286
287 outputAOVs.push_back(deepAOV);
288 for (size_t cn = 0; cn < e.mChannelNames.size(); cn++) {
289 // need the channel names so they are named correctly in the deep file
290 outputAOVChannelNames.push_back(e.mChannelNames[cn]);
291 }
292 deepAOV++;
293 }
294 }
295
296 if (!filename.empty()) {
297 deepBuffer->write(filename, outputAOVs, outputAOVChannelNames, samplesCount, aperture, region, rdlMetadata);
298
299 infos.push_back(scene_rdl2::util::buildString("Wrote deep: ", filename));
300 }
301 }

304std::vector<std::string>
305RenderOutputDriver::Impl::writeCheckpointResumeMetadata(const unsigned checkpointTileSampleTotals) const
306{
307 //
308 // Dump checkpoint/resume related internal information into metadata.
309 // This information is always write out when checkpoint mode is on
310 // regardless of resumable_output condition.
311 // If you need to resume from this output image, you need to generate this
312 // output image with resumable_output = on.
313 // Keep in mind these metadata information is not guarantee that this file
314 // can be used for input of resume render.
315 //
316
317 std::vector<std::string> outTbl;
318
319 unsigned tileSampleTotal = 0;
320 {
321 //
322 // This is a information for uniform/adaptive sampling version of progress checkpoint rendering.
323 // Keep to store how many samples are completed in each tile. This information is used by resumed
324 // process and skip previously completed part of samplings and start resume render properly.
325 //
326 tileSampleTotal = checkpointTileSampleTotals;
327 if (tileSampleTotal == 0) {
328 if (mLastTileSamples > 0) {
329 // Checkpoint render completed with normal/TimeCap situation, renderer
330 // sets mLastTileSamples properly. Otherwise it's 0 (= non checkpoint render).
331 tileSampleTotal = mLastTileSamples;
332 } else {
333 // This is a case of regular output file. We should re-compute tilebase sample total based on
334 // finalMaxSamplesPerPixel information.
335 tileSampleTotal = mFinalMaxSamplesPerPixel * 64; // tile = 8*8
336 }
337 }
338 outTbl.emplace_back("progressCheckpointTileSamples");
339 outTbl.emplace_back("int");
340 outTbl.emplace_back(std::to_string(tileSampleTotal));
341 }
342
343 {
344 //
345 // This is a information of numConsistentSamples number for FORCE_CONSISTENT_SAMPLING Aov filter.
346 // Regardress of this output uses this AOV filter or not, just in case we should output
347 // this number into metadata.
348 // If this AOV filter is used, at resume Film object stage, denormalize weight operation is used
349 // this minAdaptiveSampes value insted of "on the fly" computation value at resumed process.
350 // "On the fly" computation might have some risk to have different number due to change of sampling
351 // parameters for re-run environment.
352 //
353 unsigned numConsistentSamples = mRenderContext->getNumConsistentSamples();
354 outTbl.emplace_back("AovFilterNumConsistentSamples");
355 outTbl.emplace_back("int");
356 outTbl.emplace_back(std::to_string(numConsistentSamples));
357 }
358
359 {
360 if (mRenderContext->getSamplingMode() != SamplingMode::UNIFORM) {
361 //
362 // This is a information of sampling type. We record sampling type and it's parameters
363 // into metadata for resume render when non uniform sampling case.
364 //
365 if (mRenderContext->getSamplingMode() == SamplingMode::ADAPTIVE) {
366 unsigned minSamples, maxSamples;
367 float targetError;
368 mRenderContext->getAdaptiveSamplingParam(minSamples, maxSamples, targetError);
369
370 std::ostringstream ostr;
371 ostr << minSamples << ' ' << maxSamples << ' ' << targetError;
372 outTbl.emplace_back("adaptiveSamplingV1");
373 outTbl.emplace_back("v3f");
374 outTbl.emplace_back(ostr.str());
375 }
376 }
377 }
378
379 {
380 //
381 // This is a resume render history information. We should add current render information
382 //
383 ResumeHistoryMetaData *resumeHistoryMetaData =
384 const_cast<ResumeHistoryMetaData *>(mRenderContext->getResumeHistoryMetaData());
385 if (resumeHistoryMetaData) {
386 outTbl.emplace_back("resumeHistory");
387 outTbl.emplace_back("string");
388 outTbl.emplace_back(resumeHistoryMetaData->
389 resumeHistory(mResumeHistory,
390 mRenderContext->accumulatePbrStatistics()));
391 }
392 }
393
394 return outTbl;
395}

418void
419RenderOutputDriver::writeCheckpointEnq(const bool checkpointMultiVersion,
420 const pbr::DeepBuffer *deepBuffer,
421 pbr::CryptomatteBuffer *cryptomatteBuffer,
422 const scene_rdl2::fb_util::HeatMapBuffer *heatMap,
423 const scene_rdl2::fb_util::FloatBuffer *weightBuffer,
424 const scene_rdl2::fb_util::RenderBuffer *renderBufferOdd,
425 const std::vector<scene_rdl2::fb_util::VariablePixelBuffer> &aovBuffers,
426 const std::vector<scene_rdl2::fb_util::VariablePixelBuffer> &displayFilterBuffers,
427 const unsigned tileSampleTotals,
428 ImageWriteCache *outCache) const
429{
430 scene_rdl2::grid_util::Sha1Gen::Hash *hashPtr = nullptr;
431# ifdef RUNTIME_VERIFY_HASH
432 if (outCache) {
433 scene_rdl2::grid_util::Sha1Util::init(outCache->dataHash());
434 hashPtr = &outCache->dataHash();
435 }
436# endif // end RUNTIME_VERIFY_HASH
437
438 if (hashPtr) {
439 if (checkpointMultiVersion && mImpl->getOverwriteCheckpoint()) {
440 // checkpointMultiVersion only available when checkpointOverwrite=false situation.
441 } else {
442 mImpl->write(false, false,
443 deepBuffer, cryptomatteBuffer, heatMap, weightBuffer,
444 renderBufferOdd, &aovBuffers, &displayFilterBuffers, tileSampleTotals, nullptr, hashPtr);
445 }
446 }
447
448 mImpl->write(true, checkpointMultiVersion,
449 deepBuffer, cryptomatteBuffer, heatMap, weightBuffer,
450 renderBufferOdd, &aovBuffers, &displayFilterBuffers, tileSampleTotals, outCache, nullptr);
451}

453void
454RenderOutputDriver::writeCheckpointDeq(ImageWriteCache *cache,
455 const bool checkpointMultiVersion) const
456{
457 cache->setupDeqMode();
458
459 scene_rdl2::grid_util::Sha1Gen::Hash hash;
460 scene_rdl2::grid_util::Sha1Gen::Hash *hashPtr = nullptr;
461 if (!scene_rdl2::grid_util::Sha1Util::isInit(cache->dataHash())) {
462 // cache has Enq dataHash, so we will try to compute Deq data hash for verify
463 hashPtr = &hash;
464 }
465
466 mImpl->writeDeq(cache, checkpointMultiVersion, hashPtr);
467
468 if (hashPtr) {
469 if (cache->dataHash() == *hashPtr) {
470 std::cerr << "RenderOutputDriverImplWrite.cc : runtime Enq/Deq hash compare OK\n";
471 } else {
472 std::cerr << "RenderOutputDriverImplWrite.cc : runtime Enq/Deq hash compare failed\n";
473 }
474 std::cerr << "RenderOutputDriverImplWrite.cc Enq() " << scene_rdl2::grid_util::Sha1Util::show(cache->dataHash()) << '\n';
475 std::cerr << "RenderOutputDriverImplWrite.cc Deq() " << scene_rdl2::grid_util::Sha1Util::show(*hashPtr) << '\n';
476 }
477}
