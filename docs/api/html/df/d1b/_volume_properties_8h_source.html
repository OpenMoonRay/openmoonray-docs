19/// This class is the building block for decoupled ray marching. A per ray
20/// array of VolumeProperties can help us do quick transmittance lookup and
21/// scattering sampling. It does introduces some bias because it tries to
22/// approximate heterogeneous transmittance with discretized 1D step function

25public:
26 VolumeProperties() = default;

30 const scene_rdl2::math::Color& transmittance, const scene_rdl2::math::Color& transmittanceH,
31 float surfaceOpacityThreshold,
32 float g, float tStart, float delta,
33 int assignmentId):
34 mSigmaT(sigmaT), mSigmaS(sigmaS),
35 mSigmaTh(sigmaTh), mSigmaSh(sigmaSh),
36 mTransmittance(transmittance), mTransmittanceH(transmittanceH),
37 mSurfaceOpacityThreshold(surfaceOpacityThreshold),
38 mG(g), mTStart(tStart), mDelta(delta), mAssignmentId(assignmentId) {}
39
40 scene_rdl2::math::Color mSigmaT; // extinction coefficient
41 scene_rdl2::math::Color mSigmaS; // scattering coefficient
42 scene_rdl2::math::Color mSigmaTh; // cutouts (holdouts) extinction coefficient
43 scene_rdl2::math::Color mSigmaSh; // cutouts (holdouts) scattering coefficient
44 scene_rdl2::math::Color mTransmittance; // transmittance due to regular extinction
45 scene_rdl2::math::Color mTransmittanceH; // transmittance due to cutouts (holdouts)

48 float mTStart;
