76bool
77FabricBsdfLobe::getProperty(Property property,
78 float *dest) const
79{
80 bool handled = true;
81 switch (property)
82 {
83 case PROPERTY_COLOR:
84 {
85 *dest = mColor[0];
86 *(dest + 1) = mColor[1];
87 *(dest + 2) = mColor[2];
88 break;
89 }
90 case PROPERTY_PBR_VALIDITY:
91 {
92 // Same as Albedo PBR Validity
93 scene_rdl2::math::Color res = computeAlbedoPbrValidity(mColor);
94 *dest = res.r;
95 *(dest + 1) = res.g;
96 *(dest + 2) = res.b;
97 break;
98 }
99 case PROPERTY_ROUGHNESS:
100 {
101 *dest = mRoughness;
102 *(dest + 1) = mRoughness;
103 break;
104 }
105 case PROPERTY_NORMAL:
106 {
107 const Vec3f &N = mFrame.getN();
108 *dest = N.x;
109 *(dest + 1) = N.y;
110 *(dest + 2) = N.z;
111 }
112 break;
113 default:
114 handled = BsdfLobe::getProperty(property, dest);
115 break;
116 }
117 return handled;
118}

120// Initialize the thread direction vector for fabric
121// 'thread elevation' is used to rotate the fabric thread vertically (useful for silk)
122void
123FabricBsdfLobe::calculateThreadDirection(const Vec3f& threadDirection,
124 float threadElevation)
125{
126 // Rotate the thread direction in the X,Z (T, N) plane
127 Vec3f newDir;
128 float sinTheta, cosTheta;
129 scene_rdl2::math::sincos(scene_rdl2::math::deg2rad(threadElevation), &sinTheta, &cosTheta);
130 newDir.x = threadDirection.x * cosTheta - threadDirection.z * sinTheta;
131 newDir.y = threadDirection.y;
132 newDir.z = threadDirection.x * sinTheta + threadDirection.z * cosTheta;
133
134 mThreadDirection = normalize(mFrame.localToGlobal(newDir));
135
136 // Center The frame along the thread direction now
137 if (isOne(scene_rdl2::math::dot(mFrame.getN(), mThreadDirection))) {
138 // If the two are colinear, just use the normal
139 mFrame = ReferenceFrame(mFrame.getN());
140 } else {
141 mFrame = ReferenceFrame(mFrame.getN(), mThreadDirection);
142 }
143}
