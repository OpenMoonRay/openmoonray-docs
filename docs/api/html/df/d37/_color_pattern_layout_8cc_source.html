48void
49ColorPatternLayout::formatAndAppend(log4cplus::tostream& output,
50 const log4cplus::spi::InternalLoggingEvent& event)
51{
52 log4cplus::LogLevel level = event.getLogLevel();
53 Color clr = color::NORMAL;
54
55 ColorMap::reverse_iterator iter;
56 for (iter = mColors.rbegin(); iter != mColors.rend(); ++iter) {
57 if (level >= iter->first) {
58 clr = iter->second;
59 break;
60 }
61 }
62
63 // If the specified color is NORMAL then do not write out any tty
64 // color codes. Otherwise write the specified color, then the
65 // event, then put the tty back into NORMAL mode.
66 //
67 // Note that when using this layout the corresponding appender
68 // should be configured with immediateFlush=true to avoid leaving
69 // the output device in a colored state.
70 if (std::string(clr) == color::NORMAL) {
71 PatternLayout::formatAndAppend(output, event);
72 } else {
73 output << clr;
74 PatternLayout::formatAndAppend(output, event);
75 output << color::NORMAL;
76 }
77}
