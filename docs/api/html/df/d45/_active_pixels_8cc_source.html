17bool
18ActivePixels::compare(const ActivePixels &target) const // for debug
19{
20 if (target.mOriginalWidth != mOriginalWidth ||
21 target.mOriginalHeight != mOriginalHeight ||
22 target.mAlignedWidth != mAlignedWidth ||
23 target.mAlignedHeight != mAlignedHeight ||
24 target.mNumTilesX != mNumTilesX ||
25 target.mNumTilesY != mNumTilesY) {
26 return false;
27 }
28
29 if (target.mTiles.size() != mTiles.size()) {
30 return false;
31 }
32
33 const uint64_t * __restrict ptrA = &target.mTiles[0];
34 const uint64_t * __restrict ptrB = &mTiles[0];
35 for (size_t tileId = 0; tileId < mTiles.size(); ++tileId) {
36 if (*ptrA != *ptrB) return false;
37 ptrA++;
38 ptrB++;
39 }
40 return true;
41}

105bool
106ActivePixels::verifyReset(const std::vector<char> *partialMergeTilesTbl) const
107{
108 if (!partialMergeTilesTbl) return true;
109
110 if (partialMergeTilesTbl->size() != mTiles.size()) {
111 return false;
112 }
113
114 for (size_t id = 0; id < partialMergeTilesTbl->size(); ++id) {
115 if ((*partialMergeTilesTbl)[id]) {
116 if (mTiles[id] != 0x0) {
117 return false;
118 }
119 }
120 }
121
122 return true;
123}
