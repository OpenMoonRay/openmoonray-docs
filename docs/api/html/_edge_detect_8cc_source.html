14// Compute edge sample location using a weighted average over 5 pixels
15// horizontally, centered around yEdge
16static Vec2f
17computeEdgeSample(const Image &image, int yEdge, int x, float threshold)
18{
19 const int yMin = max(yEdge - 2, 0);
20 const int yMax = min(yEdge + 2, image.getHeight() - 1);
21
22 // Compute weighted average: y = sum(y_i * g_i) / sum(g_i) + 0.5
23 float sumYG = 0.0f;
24 float sumG = 0.0f;
25 for (int y = yMin; y <= yMax; y++) {
26 const int index = image.getPixelIndex(x, y);
27 const Color c = image.getPixel(index);
28 const float g = c.r - threshold;
29 if (g > 0.0f) {
30 sumYG += y * g;
31 sumG += g;
32 }
33 }
34
35 Vec2f sample(x + 0.5f, sumYG / sumG + 0.5f);
36 return sample;
37}

44void
45detectEdgeSamples(const Image &image, int xMin, int xMax, float threshold,
46 std::vector<Vec2f> &topEdgeSamples, std::vector<Vec2f> &bottomEdgeSamples)
47{
48 const int yCenter = image.getHeight() / 2;
49
50 // Loop over pixel columns in the center portion
51 for (int x = xMin; x < xMax; x++) {
52 Vec2i yEdge(-1, -1);
53 Vec2f gMax(0.0f, 0.0f);
54
55 // Detect approximate top-most edge
56 for (int y = yCenter; y < image.getHeight(); y++) {
57 int index = image.getPixelIndex(x, y);
58 const Color c = image.getPixel(index);
59 if (c.r > threshold && c.r > gMax[0]) {
60 gMax[0] = c.r;
61 yEdge[0] = y;
62 }
63 }
64
65 // Detect approximate bottom-most edge
66 for (int y = yCenter; y >= 0; y--) {
67 int index = image.getPixelIndex(x, y);
68 const Color c = image.getPixel(index);
69 if (c.r > threshold && c.r > gMax[1]) {
70 gMax[1] = c.r;
71 yEdge[1] = y;
72 }
73 }
74
75 // Compute floating point (x,y) coordinates for each edge point sample
76 if (yEdge[0] > 0) {
77 Vec2f edgeSample = computeEdgeSample(image, yEdge[0], x, threshold);
78 topEdgeSamples.push_back(edgeSample);
79 //std::cout << "Top = (" << x << ", " << yEdge[0] << ") --> " << edgeSample << std::endl;
80 }
81 if (yEdge[1] > 0) {
82 Vec2f edgeSample = computeEdgeSample(image, yEdge[1], x, threshold);
83 bottomEdgeSamples.push_back(edgeSample);
84 //std::cout << "Bottom = (" << x << ", " << yEdge[1] << ") --> " << edgeSample << std::endl;
85 }
86 }
87}

102void
103fitEdge(const std::vector<Vec2f> &edgeSamples, float &resultA, float &resultB)
104{
105 resultA = resultB = 0.0f;
106
107 // Average various line samples
108 const int count = edgeSamples.size() / 4;
109 MNRY_ASSERT(count > 0);
110
111 for (int i=0; i < count; i++) {
112 int j = edgeSamples.size() - count + i;
113
114 float a = (edgeSamples[j].y - edgeSamples[i].y) /
115 (edgeSamples[j].x - edgeSamples[i].x);
116 float b = edgeSamples[i].y - a * edgeSamples[i].x;
117
118 resultA += a;
119 resultB += b;
120 }
121 resultA /= count;
122 resultB /= count;
123}
