89//
90// Orthogonal to the concept of tiles are the concept of passes. This allows us to specify
91// how many passes we want to make over the scene to produce the final frame. For example,
92// for interactive rendering we may want to start with a very coarse pass so we can get
93// an approximation of the image very quickly, and then do more passes over the image to
94// refine it. For minimum time to the final frame, it would be more optimal to only specify
95// a single pass. So it's a tradeoff depending on application requirements.
96//
97struct Pass
98{
99 bool isFinePass() const
100 {
101 // Obviously 1 <= startSampleIdx is a finePass.
102 // However we also need to include pixRange=0~64, sample=0~N case into finePass as well.
103 // This case we sample start from coarse pass and process all 64 pixels. We should
104 // categorize this case as finePass actually. Toshi (Feb/12/2020)
105 return (mStartPixelIdx == 0 && mEndPixelIdx == 64) || (1 <= mStartSampleIdx);
106 }
107 bool isCoarsePass() const { return !isFinePass(); }
108
109 unsigned getNumSamplesPerPixel() const { return mEndSampleIdx - mStartSampleIdx; }
110 unsigned getNumSamplesPerTile() const { return (mEndPixelIdx - mStartPixelIdx) * getNumSamplesPerPixel(); }
111
112 bool isValid() const
113 {
114 MNRY_ASSERT(mEndPixelIdx > mStartPixelIdx && mEndPixelIdx <= 64);
115
116 if (isCoarsePass()) {
117 // This is a "coarse" pass, meaning we're still filling in top level pixels on each tile.
118 MNRY_ASSERT(mStartSampleIdx == 0);
119 MNRY_ASSERT(mEndSampleIdx == 1);
120 } else {
121 // This is a "fine" pass, meaning we're refining on a subsample basis.
122 MNRY_ASSERT(mEndSampleIdx > mStartSampleIdx);
123 }
124
125 return true;
126 }
127
128 // There are 64 pixels within each tile, indexed 0-63. You can decide which subset of pixels
129 // you want filled in by this pass here. For example, by setting mStartPixelIdx to 0, and
130 // mEndPixelIdx to 64, you would fill in all pixels of all active tiles in this pass.
131 unsigned mStartPixelIdx;
132 unsigned mEndPixelIdx;
133
134 // All samples for a single pixel within a single pass can be assumed to be
135 // executed on the same thread. The number of samples rendered for this pass
136 // *per tile* would be: (mEndSampleIdx - mStartSampleIdx) * (mEndPixelIdx - mStartPixelIdx).
137 unsigned mStartSampleIdx;
138 unsigned mEndSampleIdx;
139};
