27BSpline::BSpline(
28 Curves::Type type,
29 geom::Curves::SubType subtype,
30 geom::Curves::CurvesVertexCount&& curvesVertexCount,
31 geom::Curves::VertexBuffer&& vertices,
32 LayerAssignmentId&& layerAssignmentId,
33 PrimitiveAttributeTable&& primitiveAttributeTable):
34 CubicSpline(type,
35 subtype,
36 std::move(curvesVertexCount),
37 std::move(vertices),
38 std::move(layerAssignmentId),
39 std::move(primitiveAttributeTable))
40{
41 // allocate/calculate index buffer
42 size_t curvesCount = getCurvesCount();
43 std::vector<uint32_t> spansInChain(curvesCount, 0);
44 size_t varyingCount = 0;
45 std::vector<size_t> faceVaryingCount;
46 faceVaryingCount.reserve(curvesCount);
47 size_t spanCount = 0;
48 for (size_t i = 0; i < curvesCount; ++i) {
49 spansInChain[i] = mCurvesVertexCount[i] - 3;
50 spanCount += spansInChain[i];
51 varyingCount += spansInChain[i] + 1;
52 faceVaryingCount.push_back(spansInChain[i] + 1);
53 }
54 mSpanCount = spanCount;
55 // allocate/fill the index buffer
56 mIndexBuffer.reserve(mSpanCount);
57 size_t vertexOffset = 0;
58 for (size_t i = 0; i < curvesCount; ++i) {
59 for (size_t j = 0; j < spansInChain[i]; ++j) {
60 mIndexBuffer.emplace_back(vertexOffset, i, j);
61 // A span has 4 control points, however it shares 3
62 // control points with the subsequent span. The index
63 // buffer points to the first vertex in a span, so
64 // we only increment the vertexOffset by 1.
65 vertexOffset += 1;
66 }
67 vertexOffset += 3;
68 }
69
70 if (mLayerAssignmentId.getType() == LayerAssignmentId::Type::VARYING) {
71 MNRY_ASSERT_REQUIRE(
72 mLayerAssignmentId.getVaryingId().size() == curvesCount);
73 }
74 setAttributes(Attributes::interleave(mPrimitiveAttributeTable,
75 0, curvesCount, varyingCount, faceVaryingCount, getVertexCount()));
76}
