24void
25DebugConsoleDriver::initialize(unsigned short port)
26{
27 if (mThreadState != ThreadState::INIT) {
28 return; // already initialized
29 }
30
31 parserConfigure(mParser);
32
33 // open telnet server
34 // If you set port as 0, kernel find available port for you.
35 mTlSvrPortNum = mTlSvr.open(static_cast<int>(port), tlSvrMsgCallBackFunc, tlSvrMsgCallBackFunc);
36 if (!mTlSvrPortNum) {
37 return; // early exit if failed to open telnet server
38 }
39 // This message is important if you set port=0 and this is the only information that
40 // the user gets which port was opened.
41 std::cerr << ">> DebugConsoleDriver.cc mTlSvr port:" << mTlSvrPortNum << '\n';
42
43 // We have to build thread after finish mMutex and mCvBoot initialization completed
44 mThread = std::move(std::thread(threadMain, this));
45
46 // Wait until main thread is booted
47 std::unique_lock<std::mutex> uqLock(mMutexBoot);
48 mCvBoot.wait(uqLock,
49 [&]{
50 return (mThreadState != ThreadState::INIT); // Not wait if already non INIT condition
51 });
52}

70// static function
71void
72DebugConsoleDriver::threadMain(DebugConsoleDriver *driver)
73//
74// DebugConsole main thread. This thread receives incoming command lines and executes them.
75//
76{
77 // First, change driver's threadState condition and do notify_one to caller.
78 driver->mThreadState = ThreadState::IDLE;
79 driver->mCvBoot.notify_one(); // notify to DebugConsoleDriver's constructor
80
81 std::cerr << ">> DebugConsoleDriver.cc threadMain() booted\n";
82
83 while (true) {
84 if (driver->mThreadShutdown) {
85 break;
86 }
87
88 driver->mThreadState = ThreadState::BUSY;
89 std::string recvBuff;
90 int recvByte = driver->mTlSvr.recv(recvBuff, tlSvrMsgCallBackFunc, tlSvrMsgCallBackFunc);
91 if (recvByte == 0 || recvByte == -1) {
92 // empty or EOF
93 driver->mThreadState = ThreadState::IDLE;
94 usleep(10000); // 10000us = 10ms : wake up every 10ms and check TlSvr
95
96 } else if (recvByte < -1) { // error
97 std::cerr << "telnet server failed\n";
98 break;
99 } else {
100 Arg arg(recvBuff, &(driver->mTlSvr)); // construct Arg by received command-line
101 if (!driver->mParser.main(arg)) { // evaluate command-line by predefined command
102 std::cerr << ">> DebugConsoleDriver.cc eval() failed\n";
103 }
104 driver->mThreadState = ThreadState::IDLE;
105 }
106 }
107
108 driver->mThreadState = ThreadState::DONE;
109 driver->mCvBoot.notify_one();
110
111 std::cerr << ">> DebugConsoleDriver.cc threadMain() shutdown\n";
112}
