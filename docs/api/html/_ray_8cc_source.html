29bool
30Ray::isValid() const
31{
32 MNRY_ASSERT(isFinite(getOrigin()));
33 MNRY_ASSERT(isNormalized(getDirection()));
34 MNRY_ASSERT(isfinite(getStart()));
35 MNRY_ASSERT(isfinite(getEnd()) && getStart() >= 0.f);
36 MNRY_ASSERT(getStart() <= getEnd());
37 MNRY_ASSERT(isfinite(getTime()));
38 return true;
39}
