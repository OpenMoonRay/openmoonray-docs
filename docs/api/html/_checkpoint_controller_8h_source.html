76class CheckpointSnapshotEstimator
77//
78// This class computes the best snapshot intervals based on the user-defined information
79// and history of already executed snapshot costs.
80// Snapshot costs are stored in the list. It is maintained as a reasonable size automatically
81// in order to avoid using a huge memory.
82//
83{
84public:
85 CheckpointSnapshotEstimator() :
86 mSnapshotIntervalSec(0.0f),
87 mSnapshotOverheadThresholdFraction(0.0f),
88 mEstimateIntervalSec(0.0f)
89 {}
90
91 void set(float snapshotIntervalMinute, float snapshotOverheadFraction);
92
93 bool isActive() const;
94
95 void reset();
96
97 void pushSnapshotCost(float snapshotSec);
98
99 float estimateSnapshotInterval(); // return sec
100
101 std::string show() const;
102
103private:
104 float estimateSnapshotSec() const;
105
106 //------------------------------
107 //
108 // User-defined parameters in order to control snapshot intervals.
109 // We use mSnapshotOverheadThresholdFraction if mSnapshotIntervalSec is ZERO or negative.
110 // This is a fraction. For example, If you set 0.01, this means snapshot interval is automatically
111 // adjusted and snapshot cost overhead would be less than 1% of entire MCRT calculation.
112 // We use mSnapshotIntervalSec if mSnapshotIntervalSec value is more than ZERO and ignore
113 // mSnapshotOverheadThresholdFraction. This is basically used for debugging purposes.
114 // We disable snapshot operation when both of the parameters are ZERO or negative.
115 //
116 float mSnapshotIntervalSec;
117 float mSnapshotOverheadThresholdFraction;
118
119 //------------------------------
120
121 float mEstimateIntervalSec; // last estimated snapshot interval (sec)
122
123 std::list<float> mEventList; // snapshot interval history : unit is sec
124};
