90 // Copy existing attributes into a cache
91 void cacheAttributes() const {
92 tbb::mutex::scoped_lock lock(mCachedAttributesMutex);
93
94 mCachedRequiredAttributes.clear();
95 if (!mRequiredAttributes.empty()) {
96 std::copy(mRequiredAttributes.begin(), mRequiredAttributes.end(),
97 std::inserter(mCachedRequiredAttributes, mCachedRequiredAttributes.end()));
98 }
99
100 mCachedOptionalAttributes.clear();
101 if (!mOptionalAttributes.empty()) {
102 std::copy(mOptionalAttributes.begin(), mOptionalAttributes.end(),
103 std::inserter(mCachedOptionalAttributes, mCachedOptionalAttributes.end()));
104 }
105 }
106
107 // Check if existing attribute lists match the cache
108 bool hasChangedAttributes() const {
109 if (mRequiredAttributes.size() != mCachedRequiredAttributes.size() ||
110 mOptionalAttributes.size() != mCachedOptionalAttributes.size()) {
111 return true;
112 }
113
114 size_t numKeys = mRequiredAttributes.size();
115 for (size_t i = 0; i < numKeys; ++i) {
116 if (mCachedRequiredAttributes.find(mRequiredAttributes[i]) ==
117 mCachedRequiredAttributes.end()) {
118 return true;
119 }
120 }
121
122 numKeys = mOptionalAttributes.size();
123 for (size_t i = 0; i < numKeys; ++i) {
124 if (mCachedOptionalAttributes.find(mOptionalAttributes[i]) ==
125 mCachedOptionalAttributes.end()) {
126 return true;
127 }
128 }
129 return false;
130 }
131
132 void clearCachedAttributes() const {
133 tbb::mutex::scoped_lock lock(mCachedAttributesMutex);
134
135 mCachedRequiredAttributes.clear();
136 mCachedOptionalAttributes.clear();
137 }
