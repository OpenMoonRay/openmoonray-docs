116void
117NamedPrimitive::resetShadowLinking()
118{
119 for (auto it = mShadowLinkings.begin(); it != mShadowLinkings.end(); ++it) {
120 if (it->second != nullptr) {
121 delete it->second;
122 it->second = nullptr;
123 }
124 }
125 mShadowLinkings.clear();
126 std::unordered_set<int> casterIds;
127 getUniqueAssignmentIds(casterIds);
128 for (int casterId : casterIds) {
129 if (casterId < 0) {
130 continue;
131 }
132 mShadowLinkings[casterId] = nullptr;
133 }
134}
