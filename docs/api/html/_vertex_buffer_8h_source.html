205 void ensure_capacity(size_type nelements)
206 {
207 auto cap = capacity();
208 auto sz = size();
209 if (sz + nelements > cap) {
210 // This has to be at least 2 for our integer math recurrence
211 // relation to work, but we'll start out at four, as a VertexBuffer
212 // will probably always hold at least three, and a good chance at
213 // four.
214 cap = std::max<decltype(cap)>(cap, 4);
215 while (sz + nelements > cap) {
216 cap = cap * 3 / 2;
217 }
218 traits_type::reserve(cap);
219 }
220 }
