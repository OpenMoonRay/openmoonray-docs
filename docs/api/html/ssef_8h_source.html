433 // Intel provides an atan and atan2 intrinsic in their short vector math library.

438 // This emulator was appropriated from this implementation and modified
439 // to achieve compatibility with our ssef type.

445 #define ALIGN16_END __attribute__((aligned(16)))

447 // Masks
448 static const ALIGN16_BEG int _mm_cst_sign_mask[4] ALIGN16_END =
449 { static_cast<int>(0x80000000), static_cast<int>(0x80000000), static_cast<int>(0x80000000), static_cast<int>(0x80000000) };
450
451 // Numerical constants
452 static const __m128 _mm_cst_zero = _mm_setzero_ps();
453 static const __m128 _mm_cst_one = _mm_set1_ps( 1.0f);
454 static const __m128 _mm_cst_mone = _mm_set1_ps(-1.0f);
455 static const __m128 _mm_cst_two = _mm_set1_ps( 2.0f);
456 static const __m128 _mm_cst_mtwo = _mm_set1_ps(-2.0f);
457
458 // Tangent-based numerical constants
459 static const __m128 _mm_cst_tan3pio8 = _mm_set1_ps(2.414213562373095f);
460 static const __m128 _mm_cst_tanpio8 = _mm_set1_ps(0.4142135623730950f);
461
462 // Pi-based numerical constants
463 static const __m128 _mm_cst_pi = _mm_set1_ps(3.14159265358979f);
464 static const __m128 _mm_cst_mpi = _mm_set1_ps(-3.14159265358979f);
465 static const __m128 _mm_cst_pio2 = _mm_set1_ps(1.5707963267948966f);
466 static const __m128 _mm_cst_mpio2 = _mm_set1_ps(-1.5707963267948966f);
467 static const __m128 _mm_cst_pio4 = _mm_set1_ps(0.7853981633974483f);
468
469 // Minimax coefficients
470 static const __m128 _mm_cst_atancof_p0 = _mm_set1_ps( 8.05374449538e-2f);
471 static const __m128 _mm_cst_atancof_p1 = _mm_set1_ps(-1.38776856032e-1f);
472 static const __m128 _mm_cst_atancof_p2 = _mm_set1_ps( 1.99777106478e-1f);
473 static const __m128 _mm_cst_atancof_p3 = _mm_set1_ps(-3.33329491539e-1f);
474
475 __forceinline
476 ssef atan( __m128 x)
477 {
478 // Figure out if x is negative and if so flip it
479 __m128 signbit = _mm_and_ps(x, *(const __m128*) _mm_cst_sign_mask); // Determine the signbit for each element
480 x = _mm_andnot_ps(*(const __m128*) _mm_cst_sign_mask, x); // Get the absolute value
481
482 __m128 y = _mm_cst_zero; // y = <0, 0, 0, 0, 0, 0, 0, 0>
483
484 // Range Reduction
485 __m128 x2 = _mm_div_ps(_mm_cst_mone, x); // x2[n] = (1/x[n])
486 __m128 x3 = _mm_div_ps(_mm_sub_ps(x, _mm_cst_one),
487 _mm_add_ps(x, _mm_cst_one)); // x3[n] = (x[n] - 1)/(x[n] + 1)
488
489 // if (x > tan(3*pi/8) {
490 // x = x2;
491 // y = pi/2;
492 // }
493 __m128 mask = _mm_cmp_ps(x, _mm_cst_tan3pio8, _CMP_GT_OQ); // mask[n] = (x[n] > 2.4142)
494 x = _mm_blendv_ps(x, x2, mask);
495 y = _mm_blendv_ps(y, _mm_cst_pio2, mask);
496
497 // If (x > tan(pi/8) {
498 // x = x3;
499 // y = pi/4;
500 // }
501 mask = _mm_cmp_ps(x, _mm_cst_tanpio8, _CMP_GT_OQ); // mask[n] = (x[n] > 0.4142)
502 x = _mm_blendv_ps(x, x3, mask);
503 y = _mm_blendv_ps(y, _mm_cst_pio4, mask);
504
505 // Polynomial Computation
506 __m128 z = _mm_mul_ps(x, x); // z = x^2
507 __m128 num;
508
509#if defined (__FMA__)
510 num = _mm_fmadd_ps(z, _mm_cst_atancof_p0, _mm_cst_atancof_p1); // num = (p0 * x^2) + p1
511 num = _mm_fmadd_ps(num, z, _mm_cst_atancof_p2); // num = (p0 * x^4) + (p1 * x^2) + p2
512 num = _mm_fmadd_ps(num, z, _mm_cst_atancof_p3); // num = (p0 * x^6) + (p1 * x^4) + (p2 * x^2) + p3
513 num = _mm_mul_ps(num, z); // num = (p0 * x^8) + (p1 * x^6) + (p2 * x^4) + (p3 * x^2)
514 num = _mm_fmadd_ps(num, x, x); // num = (p0 * x^9) + (p1 * x^7) + (p2 * x^5) + (p3 * x^3) + x
515#else
516 num = _mm_mul_ps(_mm_cst_atancof_p0, z); // num = p0 * x^2
517 num = _mm_add_ps(num, _mm_cst_atancof_p1); // num = (p0 * x^2) + p1
518 num = _mm_mul_ps(num, z); // num = (p0 * x^4) + (p1 * x^2)
519 num = _mm_add_ps(num, _mm_cst_atancof_p2); // num = (p0 * x^4) + (p1 * x^2) + p2
520 num = _mm_mul_ps(num, z); // num = (p0 * x^6) + (p1 * x^4) + (p2 * x^2)
521 num = _mm_add_ps(num, _mm_cst_atancof_p3); // num = (p0 * x^6) + (p1 * x^4) + (p2 * x^2) + p3
522 num = _mm_mul_ps(num, z); // num = (p0 * x^8) + (p1 * x^6) + (p2 * x^4) + (p3 * x^2)
523 num = _mm_mul_ps(num, x); // num = (p0 * x^9) + (p1 * x^7) + (p2 * x^5) + (p3 * x^3)
524 num = _mm_add_ps(num, x); // num = (p0 * x^9) + (p1 * x^7) + (p2 * x^5) + (p3 * x^3) + x
525#endif
526
527 y = _mm_add_ps(y, num); // y += num
528 y = _mm_xor_ps(y, signbit); // y *= -1 (maybe)
529
530 return y;
531 }

533 __forceinline
534 ssef atan2 (const ssef& y, const ssef& x) {
535 __m128 w = _mm_blendv_ps(_mm_cst_pi, _mm_cst_mpi, y);
536 w = _mm_blendv_ps(_mm_cst_zero, w, x);
537
538 __m128 q = _mm_div_ps(y, x); // q = y/x - Possible NaN, handled via masks
539 q = _mm_add_ps(w, atan(q)); // q = w + atan(q)
540
541 __m128 mask; // x-dependent mask
542 __m128 mask2; // y-dependent mask
543
544 // atan2(-y, 0) -> -pi/2
545 mask = _mm_cmp_ps(x, _mm_cst_zero, _CMP_EQ_OQ); // mask[n] = (x[n] == 0)
546 mask2 = _mm_and_ps(mask, _mm_cmp_ps(y, _mm_cst_zero, _CMP_LT_OQ)); // mask2[n] = (mask[n] & (y[n] < 0))
547 q = _mm_blendv_ps(q, _mm_cst_mpio2, mask2);
548
549 // atan2(+y, 0) -> pi/2
550 mask2 = _mm_and_ps(mask, _mm_cmp_ps(y, _mm_cst_zero, _CMP_GT_OQ)); // mask2[n] = (mask[n] & (y[n] > 0))
551 q = _mm_blendv_ps(q, _mm_cst_pio2, mask2);
552
553 // atan2(0, 0) -> 0
554 mask2 = _mm_and_ps(mask, _mm_cmp_ps(y, _mm_cst_zero, _CMP_EQ_OQ)); // mask2[n] = (mask & y[n] == 0)
555 q = _mm_blendv_ps(q, _mm_cst_zero, mask2);
556
557 // atan2(0, -x) -> pi
558 mask = _mm_cmp_ps(x, _mm_cst_zero, _CMP_LT_OQ); // mask[n] = (x[n] < 0)
559 mask2 = _mm_and_ps(mask, _mm_cmp_ps(y, _mm_cst_zero, _CMP_EQ_OQ)); // mask2[n] = (mask[n] & y[n] == 0)
560 q = _mm_blendv_ps(q, _mm_cst_pi, mask2);
561
562 return q;
563 }
