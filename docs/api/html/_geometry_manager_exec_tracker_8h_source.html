27class GeometryManagerExecTracker
28//
29// This class is used to track down GeometryManager execution status.
30// RenderprepStatsCallBack is a call back function to send stats to the downstream computation.
31// RenderprepStatsCallBack is properly called from each {start,end}* function.
32// We can set nullptr to RenderPrepStatsCallBack and disable report stats logic when we are not under
33// the arras context.
34//
35{
36public:
37 using RenderPrepStatsCallBack = std::function<void(const scene_rdl2::grid_util::RenderPrepStats &rPrepStats)>;
38 using RenderPrepCancelCallBack = std::function<bool()>;

40 using Parse = scene_rdl2::grid_util::Parse;

43 enum class RESULT // function execution result
44 {
45 CANCELED, // canceled middle of the function
46 FINISHED // function has been completed
47 };

49 GeometryManagerExecTracker() :
50 mRenderPrepStatsCallBack(nullptr),
51 mRenderPrepCancelCallBack(nullptr),
52 mMsgHandlerCallBack(nullptr),
53 mStageId(0),
54 mRunLoadGeometries{Condition::INIT, Condition::INIT},
55 mRunLoadGeometriesTotal{0, 0},
56 mRunFinalizeChange{Condition::INIT, Condition::INIT},
57 mRunTessellation{Condition::INIT, Condition::INIT},
58 mRunTessellationTotal{0, 0},
59 mRunBVHConstruction{Condition::INIT, Condition::INIT},
60 mCancelCodePos(CancelCodePos::EMPTY),
61 mCancelCodePosLoadGeomCounter(std::numeric_limits<int>::max()),
62 mCancelCodePosTessellationCounter(std::numeric_limits<int>::max())
63 {
64 // The following parameters need initialize here in order to avoid GCC compile error.
65 for (int i = 0; i < mStageMax; ++i) {
66 mRunLoadGeometriesItem[i] = Condition::INIT;
67 mRunLoadGeometriesProcessed[i] = 0;
68 mRunTessellationItem[i] = Condition::INIT;
69 mRunTessellationProcessed[i] = 0;
70 }
71 }

86 mCancelCodePosTessellationCounter(src.mCancelCodePosTessellationCounter)
87 {
88 // The following parameters need initialize here in order to avoid GCC compile error.
89 for (int i = 0; i < mStageMax; ++i) {
90 Condition condition;
91 int processed;
92 condition = src.mRunLoadGeometriesItem[i]; mRunLoadGeometriesItem[i] = condition;
93 processed = src.mRunLoadGeometriesProcessed[i]; mRunLoadGeometriesProcessed[i] = processed;
94 condition = src.mRunTessellationItem[i]; mRunTessellationItem[i] = condition;
95 processed = src.mRunTessellationProcessed[i]; mRunTessellationProcessed[i] = processed;
96 }
97
98 // parseConfigure() is only executed inside copy constructor so far.
99 // This is very dependent on the logic of how GeometryManager is initialized.
100 // We always use copy constructed version of GeometryManagerExecTracker instead of the
101 // very 1st initialized one.
102 parseConfigure();
103 }

178 template <typename T0, typename T1>
179 RESULT updateRunStatus(const CancelCodePos &callerCodePos0,
180 const CancelCodePos &callerCodePos1,
181 T0 &updateTarget,
182 const T1 &finishCondition,
183 const T1 &cancelCondition) const {
184 const CancelCodePos &callerCodePos =
185 (mStageId == 0) ? callerCodePos0 : callerCodePos1;
186
187 RESULT result = checkRunStatus(callerCodePos);
188 if (result == RESULT::FINISHED) {
189 updateTarget = finishCondition;
190 } else {
191 updateTarget = cancelCondition;
192 }
193
194 renderPrepStatsUpdate();
195
196 if (mMsgHandlerCallBack && result == RESULT::CANCELED) {
197 updateRunStatusCancelMessage(callerCodePos, result);
198 }
199
200 return result;
201 }
