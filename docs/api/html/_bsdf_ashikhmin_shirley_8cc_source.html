38AshikhminShirleyGlossyBsdfLobe::AshikhminShirleyGlossyBsdfLobe(
39 const Vec3f &N, const Vec3f &anisoDirection,
40 float roughnessU, float roughnessV) :
41 BsdfLobe(Type(REFLECTION | GLOSSY), DifferentialFlags(0), false,
42 PROPERTY_NORMAL | PROPERTY_ROUGHNESS),
43 mFrame(N, anisoDirection),
44 mInputRoughnessU(roughnessU),
45 mInputRoughnessV(roughnessV)
46{
47 // Apply roughness squaring to linearize roughness response
48 // See "Physically-Based Shading at Disney" Siggraph course notes.
49 roughnessU *= roughnessU;
50 roughnessV *= roughnessV;
51
52 // Convert roughness to exponent
53 mExponentU = shading::roughness2Exponent(roughnessU);
54 mExponentV = shading::roughness2Exponent(roughnessV);
55
56 mScaleFactor = scene_rdl2::math::sqrt((mExponentU + 1.0f) * (mExponentV + 1.0f)) / (8.0f * sPi);
57
58 mSampleFactor = scene_rdl2::math::sqrt((mExponentU + 1.0f) / (mExponentV + 1.0f));
59
60 // Use the sharper of the two roughnesses to derive a directional differential
61 // scale that varies with roughness
62 float minRoughness = min(roughnessU, roughnessV);
63 mdDFactor = sdDFactorMin + minRoughness * sdDFactorSlope;
64
65 // TODO: Set lobe category based on roughness
66}

98Color
99AshikhminShirleyGlossyBsdfLobe::eval(const BsdfSlice &slice, const Vec3f &wi,
100 float *pdf) const
101{
102 // Prepare for early exit
103 if (pdf != NULL) {
104 *pdf = 0.0f;
105 }
106
107 // Compute normalized half-vector
108 Vec3f H;
109 if (!computeNormalizedHalfVector(slice.getWo(), wi, H)) {
110 return Color(zero);
111 }
112
113 float NdotWi = dot(mFrame.getN(), wi);
114 float maxNdotW = max(dot(mFrame.getN(), slice.getWo()), NdotWi);
115 if (maxNdotW < sCosGrazingAngle) {
116 return Color(zero);
117 }
118
119 float HdotN = dot(H, mFrame.getN());
120 if (HdotN < sCosGrazingAngle) {
121 return Color(zero);
122 }
123
124 float sinThetaSqrd = 1.0f - HdotN * HdotN;
125 if (sinThetaSqrd < sSinNormalAngleSqrd) {
126 return Color(zero);
127 }
128
129 float HdotWi = dot(H, wi);
130 HdotWi = max(HdotWi, sEpsilon);
131
132 float HdotX = dot(H, mFrame.getX());
133 float HdotY = dot(H, mFrame.getY());
134
135 float exponent = (mExponentU * HdotX * HdotX +
136 mExponentV * HdotY * HdotY) / sinThetaSqrd;
137 float common = mScaleFactor * scene_rdl2::math::pow(HdotN, exponent);
138
139
140 // Compute bsdf contribution
141 float result = common / (HdotWi * maxNdotW);
142
143 Color f = result * computeScaleAndFresnel(HdotWi) *
144 (slice.getIncludeCosineTerm() ? max(NdotWi, 0.0f) : 1.0f);
145
146
147 // Compute pdf
148 if (pdf != NULL) {
149 float HdotWo = HdotWi;
150 *pdf = common / HdotWo;
151 }
152
153 // Soften hard shadow terminator due to shading normals
154 const float Gs = slice.computeShadowTerminatorFix(mFrame.getN(), wi);
155
156 return Gs * f;
157}

169Color
170AshikhminShirleyGlossyBsdfLobe::sample(const BsdfSlice &slice, float r1, float r2,
171 Vec3f &wi, float &pdf) const
172{
173 // Sample phi according to equation (9)
174 float phi;
175 if (r2 < 0.25f) {
176 r2 = 4.0f * r2;
177 phi = samplePhi(r2, mSampleFactor);
178 } else if (r2 < 0.5f) {
179 r2 = 4.0f * (0.5f - r2);
180 phi = sPi - samplePhi(r2, mSampleFactor);
181 } else if (r2 < 0.75f) {
182 r2 = 1.0f - 4.0f * (0.75f - r2);
183 phi = sPi + samplePhi(r2, mSampleFactor);
184 } else {
185 r2 = 4.0f * (1.0f - r2);
186 phi = sTwoPi - samplePhi(r2, mSampleFactor);
187 }
188 float cosPhi;
189 float sinPhi;
190 sincos(phi, &sinPhi, &cosPhi);
191
192 // Sample theta according to equation (10)
193 // Note: Above this value for r1 and the math below sends samples in
194 // regions where the brdf decreases very sharply, causing a lot of noise
195 // with sharper glossy reflections. This only introduces a very small error.
196 // TODO: we should probably stay away from the Bsdf due to its poor
197 // normalization properties, so I won't spend too much time figuring this
198 // one out at this point.
199 float exponent = mExponentU * cosPhi * cosPhi +
200 mExponentV * sinPhi * sinPhi;
201 float cosTheta = scene_rdl2::math::pow(1.0f - r1, 1.0f / (exponent + 1.0f));
202 if (cosTheta < sCosGrazingAngleSample || cosTheta > sCosNormalAngleSample) {
203 pdf = 0.0f;
204 return Color(zero);
205 }
206 float cosThetaSqrd = cosTheta * cosTheta;
207 float sinTheta = scene_rdl2::math::sqrt(1.0f - cosThetaSqrd);
208
209 // Map the canonical half vector into the local space
210 Vec3f H(cosPhi * sinTheta, sinPhi * sinTheta, cosTheta);
211 MNRY_ASSERT(isNormalized(H));
212 H = mFrame.localToGlobal(H);
213
214 // Compute reflection direction about the half vector
215 float HdotWo = computeReflectionDirection(H, slice.getWo(), wi);
216
217 // If the sampled microfacet is facing away from wo, we return a
218 // zero probability sample
219 // TODO: Should this be using: HdotWo = abs(HdotWo);
220 if (HdotWo < sCosGrazingAngleSample) {
221 pdf = 0.0f;
222 return Color(zero);
223 }
224
225 float NdotWi = dot(mFrame.getN(), wi);
226 float maxNdotW = max(dot(mFrame.getN(), slice.getWo()), NdotWi);
227 if (maxNdotW < sCosGrazingAngleSample) {
228 pdf = 0.0f;
229 return Color(zero);
230 }
231
232 float common = mScaleFactor * scene_rdl2::math::pow(cosTheta, exponent);
233
234 // Compute the probability of selecting wi
235 pdf = common / HdotWo;
236
237 // Compute the brdf
238 float result = pdf / maxNdotW;
239
240 // Apply scale and fresnel if any
241 float HdotWi = HdotWo;
242 Color f = result * computeScaleAndFresnel(HdotWi) *
243 (slice.getIncludeCosineTerm() ? max(NdotWi, 0.0f) : 1.0f);
244
245 return f;
246}

249void
250AshikhminShirleyGlossyBsdfLobe::differentials(const Vec3f &wo, const Vec3f &wi,
251 float r1, float r2, const Vec3f &dNdx, const Vec3f &dNdy,
252 Vec3f &dDdx, Vec3f &dDdy) const
253{
254 // It's complex to capture the full derivative. Instead we use the
255 // derivative of a mirror reflection about the H vector, and scale the
256 // length of the directional derivative proportionally with roughness.
257 Vec3f H;
258 if (!computeNormalizedHalfVector(wo, wi, H)) {
259 H = mFrame.getN();
260 }
261 computeReflectionDirectionDifferential(wo, wi, H, dNdx, dNdy, dDdx, dDdy);
262 dDdx *= mdDFactor;
263 dDdy *= mdDFactor;
264}

304Color
305AshikhminShirleyDiffuseBsdfLobe::eval(const BsdfSlice &slice, const Vec3f &wi,
306 float *pdf) const
307{
308 float tmp, tmp2;
309
310 float NdotWo = max(dot(mFrame.getN(), slice.getWo()), 0.0f);
311 tmp = 1.0f - 0.5f * NdotWo;
312 tmp2 = tmp * tmp;
313 float powerWo = tmp2 * tmp2 * tmp;
314
315 float NdotWi = max(dot(mFrame.getN(), wi), 0.0f);
316 tmp = 1.0f - 0.5f * NdotWi;
317 tmp2 = tmp * tmp;
318 float powerWi = tmp2 * tmp2 * tmp;
319
320 float result = sDiffuseConstant * (1.0f - powerWo) * (1.0f - powerWi);
321
322 Color f = result * computeScaleAndFresnel(NdotWi) *
323 (slice.getIncludeCosineTerm() ? NdotWi : 1.0f);
324
325 if (pdf != NULL) {
326 *pdf = NdotWi * sOneOverPi;
327 }
328
329 // Soften hard shadow terminator due to shading normals
330 const float Gs = slice.computeShadowTerminatorFix(mFrame.getN(), wi);
331
332 return Gs * f;
333}
