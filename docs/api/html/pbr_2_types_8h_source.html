60finline uint32_t
61makeTilePassAndFilm(uint32_t tile, uint32_t pass, uint32_t film)
62{
63 // 22 bits for the tile
64 // 7 bits for the pass
65 // 3 bits for the film
66 MNRY_ASSERT(tile <= 0x3fffff && pass < MAX_RENDER_PASSES && film <= 0x07);
67 return (tile << 10) | (pass << 3) | film;
68}
69
70finline uint32_t
71getTile(uint32_t tilePassAndFilm)
72{
73 return (tilePassAndFilm >> 10) & 0x3fffff;
74}
75
76finline uint32_t
77getPass(uint32_t tilePassAndFilm)
78{
79 return (tilePassAndFilm >> 3) & 0x7f;
80}
81
82finline uint32_t
83getFilm(uint32_t tilePassAndFilm)
84{
85 return tilePassAndFilm & 0x07;
86}

177struct ALIGN(32) BundledAov
178{
179 static const unsigned MAX_AOV = 5; // up to 5 aovs in a bunedled aov entry
180 static const unsigned MAX_AOV_IDX = 0xffe; // 0xfff indicates an unused slot
181 static const unsigned AOV_NUM_BITS = 12;
182 static const uint64_t AOV_ALL_BITS = 0xfff;
183 static const unsigned FILM_BITS = 4;
184 static const uint64_t FILM_ALL_BITS = 0xf;
185 static const unsigned FILM_BIT_SHIFT = 60;
186 static const uint64_t AOV_ALL_UNUSED = 0xfffffffffffffff;
187
188 finline BundledAov(uint32_t pixel, uint32_t deepDataHandle, uint32_t film)
189 {
190 init(pixel, deepDataHandle, film);
191 }
192 finline void init(uint32_t pixel, uint32_t deepDataHandle, uint32_t film)
193 {
194 mPixel = pixel;
195 mDeepDataHandle = deepDataHandle;
196 mIndices = ((film & FILM_ALL_BITS) << FILM_BIT_SHIFT) | AOV_ALL_UNUSED;
197 }
198 finline uint32_t filmIdx() const { return uint32_t(mIndices >> FILM_BIT_SHIFT); }
199 finline uint32_t aovIdx(unsigned aov) const
200 {
201 MNRY_ASSERT(aov < MAX_AOV);
202 return uint32_t((mIndices >> (aov * AOV_NUM_BITS)) & AOV_ALL_BITS);
203 }
204 finline void setAov(unsigned aov, float val, uint32_t aovIdx)
205 {
206 MNRY_ASSERT(aovIdx <= MAX_AOV_IDX);
207 mAovs[aov] = val;
208 mIndices &= ~(AOV_ALL_BITS << (aov * AOV_NUM_BITS)); // clear
209 mIndices |= ((aovIdx & AOV_ALL_BITS) << (aov * AOV_NUM_BITS)); // set
210 }
211 finline void setAov(unsigned aov, float val)
212 {
213 // Use this function to set a value in mAovs that does not have
214 // an aovIdx. Normally, any value in mAovs should correspond to a valid aov index.
215 // For closest filtering, we store depth in the aov slot just past
216 // the aov value. The index is set to 0xfff to indicate this.
217 mAovs[aov] = val;
218 mIndices |= ((AOV_ALL_BITS) << (aov * AOV_NUM_BITS));
219 }
220
221 float mAovs[MAX_AOV]; // up to MAX_AOV values
222 uint32_t mPixel; // Screen coordinates of pixel.
223 uint32_t mDeepDataHandle;
224 uint64_t mIndices; // 4 bit film indx + 5 * 12 bits for aov indices
225};
226
227MNRY_STATIC_ASSERT(sizeof(BundledAov) == 64);

229struct ALIGN(16) BundledHeatMapSample
230{
231 int64_t mTicks;
232 uint32_t mPixel; // Screen coordinates of pixel
233 uint32_t mFilmIdx; // Which set of film buffers are we rendering to.
234
235 finline uint32_t filmIdx() const { return mFilmIdx; }

237 // helper function to initialize an array of bundled heat map samples
238 finline static void initArray(BundledHeatMapSample *b, unsigned &bIndx,
239 uint32_t pixel, uint32_t filmIdx)
240 {
241 bIndx = 0;
242 b[bIndx].mTicks = 0;
243 b[bIndx].mPixel = pixel;
244 b[bIndx].mFilmIdx = filmIdx;
245 }
246
247 // helper function for adding ticks to an array of bundled heat map samples
248 finline static void addTicksToArray(BundledHeatMapSample *b, unsigned &bIndx,
249 int64_t ticks, uint32_t pixel, uint32_t filmIdx)
250 {
251 if (b[bIndx].mPixel != pixel || b[bIndx].mFilmIdx != filmIdx) {
252 bIndx++;
253 b[bIndx].mTicks = 0;
254 b[bIndx].mPixel = pixel;
255 b[bIndx].mFilmIdx = filmIdx;
256 }
257 b[bIndx].mTicks += ticks;
258 }
259};
