64// static function
65std::string
66ValueContainerUtil::hexDump(const std::string &hd, const std::string &titleMsg, const void *buff, const size_t size)
67//
68// general purpose hexadecimal dump
69//
70{
71 const char *startBuff = static_cast<const char *>(buff);
72 const char *endBuff = &startBuff[size];
73
74 std::ostringstream ostr;
75
76 ostr << hd << "hexDump";
77 if (titleMsg.size() > 0) ostr << " " << titleMsg;
78 ostr << " size:" << size << " {\n";
79
80 const char *cPtrA = startBuff;
81 const char *cPtrB = startBuff;
82
83 static const char *SEPARATOR = "-";
84
85 int itemCount = 0;
86 while (1) {
87 if (cPtrA == endBuff) {
88 if (itemCount != 0) {
89 for (int i = itemCount + 1; i <= 16; i++) {
90 ostr << " ";
91 if (i == 8) {
92 ostr << " " << SEPARATOR << " ";
93 } else {
94 ostr << " ";
95 }
96 }
97 ostr << " | ";
98 for (int i = 0; i < itemCount; i++) {
99 if (isprint(static_cast<int>(*cPtrB))) ostr << *cPtrB << " ";
100 else ostr << " ";
101 if (i == 7) ostr << " " << SEPARATOR << " ";
102 cPtrB++;
103 }
104 ostr << std::endl;
105 }
106 break;
107 }
108
109 itemCount++;
110
111 if (itemCount == 1) {
112 size_t cSize = (uintptr_t)cPtrA - (uintptr_t)startBuff;
113 ostr << hd << " 0x" << std::hex << std::setw(4) << std::setfill('0') << cSize << std::dec << ": ";
114 }
115
116 ostr << std::setw(2) << std::setfill('0') << std::hex << ((static_cast<int>(*cPtrA)) & 0xff) << std::dec;
117
118 if (itemCount == 16) {
119 ostr << " | ";
120 for (int i = 0; i < 16; ++i) {
121 if (isprint(static_cast<int>(*cPtrB))) ostr << *cPtrB << " ";
122 else ostr << " ";
123 if (i == 7) ostr << " " << SEPARATOR << " ";
124 cPtrB++;
125 }
126 ostr << std::endl;
127 itemCount = 0;
128 } else if (itemCount == 8) {
129 ostr << " " << SEPARATOR << " ";
130 } else {
131 ostr << " ";
132 }
133
134 cPtrA++;
135 }
136 ostr << hd << "}";
137
138 return ostr.str();
139}
