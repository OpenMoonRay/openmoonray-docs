171 // Derive a directional differential size that varies according to width
172 // Experimentally, we found that we want around 0.075 when width is
173 // 0 degrees and 0.25 when it is 15 degrees. Also taking care that
174 // mLongitudinalWidth is expressed in radians, not degrees...
175 // 15.0f/180.f = 0.0833f
176 static constexpr float sdDFactorMin = 0.075f;
177 static constexpr float sdDFactorMax = 0.25f;
178 static constexpr float sdDFactorSlope = (sdDFactorMax - sdDFactorMin)
