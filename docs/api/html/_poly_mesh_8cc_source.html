50void
51PolyMesh::tessellate(const TessellationParams& tessellationParams)
52{
53 if (mIsMeshFinalized) {
54 return;
55 }
56
57 MeshIndexType baseFaceType = getBaseFaceType();
58 size_t baseFaceVertexCount = baseFaceType == MeshIndexType::QUAD ?
59 sQuadVertexCount : sTriangleVertexCount;
60 auto& primitiveAttributeTable = mPolyMeshData->mPrimitiveAttributeTable;
61 // make input mesh all quads or all triangles
62 splitNGons(mPolyMeshData->mEstiFaceCount, mFaceToPart,
63 mPolyMeshData->mFaceVertexCount, mIndices,
64 mLayerAssignmentId, primitiveAttributeTable);
65
66 MNRY_ASSERT_REQUIRE(!mIndices.empty() &&
67 mIndices.size() % baseFaceVertexCount == 0);
68 MNRY_ASSERT_REQUIRE(mVertices.size() > 0);
69 size_t baseFaceCount = mIndices.size() / baseFaceVertexCount;
70 // varying and vertex are the same thing for polygon mesh if
71 // there is no displacement tessellation involved, otherwise
72 // vertex rate means "per tessellated vertex" and
73 // varying rate means "per base vertex"
74 // we make the distinction here that all per vertex
75 // input primitive attributes are classified as RATE_VARYING
76 // and make RATE_VERTEX storing primitive attributes that will be
77 // tessellated to "per tessellated vertex"
78 // Currently the only primitive attribute that is classified as
79 // RATE_VERTEX is shading normal (StandardAttributes::sNormal)
80 for (auto& kv : primitiveAttributeTable) {
81 for (auto& attribute : kv.second) {
82 if (attribute->getRate() == RATE_VERTEX) {
83 attribute->setRate(RATE_VARYING);
84 }
85 }
86 }
87 if (mLayerAssignmentId.getType() == LayerAssignmentId::Type::VARYING) {
88 MNRY_ASSERT_REQUIRE(
89 mLayerAssignmentId.getVaryingId().size() == baseFaceCount);
90 }
91 size_t baseVertexCount = mVertices.size();
92 const scene_rdl2::rdl2::Layer* pRdlLayer = tessellationParams.mRdlLayer;
93 if (shouldTessellate(tessellationParams.mEnableDisplacement, pRdlLayer)) {
94 PolyTopologyIdLookup topologyIdLookup(mVertices.size(),
95 baseFaceVertexCount, mIndices);
96 // calculate edge tessellation factor based on either user specified
97 // resolution (uniform) or camera frustum info (adaptive)
98 std::vector<PolyTessellationFactor> tessellationFactors =
99 computeTessellationFactor(pRdlLayer, tessellationParams.mFrustums, topologyIdLookup);
100 std::vector<PolyFaceTopology> faceTopologies =
101 generatePolyFaceTopology(topologyIdLookup);
102 MNRY_ASSERT_REQUIRE(
103 tessellationFactors.size() == faceTopologies.size());
104 PolyTessellatedVertexLookup tessellatedVertexLookup(faceTopologies,
105 topologyIdLookup, tessellationFactors, baseFaceVertexCount);
106 // generate the tessellated index buffer and
107 // sample points for generating tessellated vertex buffer
108 std::vector<PolyMesh::SurfaceSample> surfaceSamples;
109 size_t estimatedFaceCount =
110 tessellatedVertexLookup.getEstimatedFaceCount();
111 PolygonMesh::IndexBuffer tessellatedIndices;
112 tessellatedIndices.reserve(sQuadVertexCount * estimatedFaceCount);
113 std::vector<int> tessellatedToBaseFace;
114 tessellatedToBaseFace.reserve(estimatedFaceCount);
115 std::vector<Vec2f> faceVaryingUv;
116 faceVaryingUv.reserve(sQuadVertexCount * estimatedFaceCount);
117 generateIndexBufferAndSurfaceSamples(faceTopologies,
118 tessellatedVertexLookup, tessellatedIndices,
119 surfaceSamples, tessellatedToBaseFace, &faceVaryingUv);
120 MNRY_ASSERT_REQUIRE(
121 tessellatedIndices.size() == faceVaryingUv.size());
122 mTessellatedToBaseFace = std::move(tessellatedToBaseFace);
123 mFaceVaryingUv = std::move(faceVaryingUv);
124 // generate tessellated vertex buffer
125 PolygonMesh::VertexBuffer tessellatedVertices =
126 generateVertexBuffer(mVertices, mIndices, surfaceSamples);
127
128 mBaseIndices = std::move(mIndices);
129 mIndices = std::move(tessellatedIndices);
130 std::swap(mVertices, tessellatedVertices);
131 mIsTessellated = true;
132 } else {
133 mIsTessellated = false;
134 }
135 initAttributesAndDisplace(pRdlLayer, baseFaceCount,
136 baseVertexCount, tessellationParams.mEnableDisplacement,
137 tessellationParams.mFastGeomUpdate, tessellationParams.mIsBaking,
138 tessellationParams.mWorld2Render);
139
140 // reverse normals reverses orientation and negates normals
141 if (mIsNormalReversed ^ mIsOrientationReversed) {
142 size_t faceVertexCount = mIsTessellated ?
143 sQuadVertexCount : baseFaceVertexCount;
144 reverseOrientation(faceVertexCount, mIndices, mAttributes);
145 }
146
147 if (mIsNormalReversed) mAttributes->negateNormal();
148
149 // For the baked volume shader grid, we want to set the transform
150 // to world2render if the primitive is shared.
151 if (getIsReference()) {
152 mPrimToRender = scene_rdl2::math::toFloat(tessellationParams.mWorld2Render);
153 }
154
155 mIsMeshFinalized = true;
156}

168void
169PolyMesh::getTessellatedMesh(TessellatedMesh& tessMesh) const
170{
171 if (mIsTessellated) {
172 // we always tessellate poly mesh to quads
173 tessMesh.mIndexBufferType = MeshIndexType::QUAD;
174 tessMesh.mIndexBufferDesc.mStride =
175 sQuadVertexCount * sizeof(geom::Primitive::IndexType);
176 } else {
177 MeshIndexType meshIndexType = getBaseFaceType();
178 tessMesh.mIndexBufferType = meshIndexType;
179 tessMesh.mIndexBufferDesc.mStride =
180 meshIndexType == MeshIndexType::QUAD ?
181 sQuadVertexCount * sizeof(geom::Primitive::IndexType) :
182 sTriangleVertexCount * sizeof(geom::Primitive::IndexType);
183 }
184 tessMesh.mFaceCount = getTessellatedMeshFaceCount();
185 tessMesh.mIndexBufferDesc.mData = static_cast<const void*>(mIndices.data());
186 tessMesh.mIndexBufferDesc.mOffset = 0;
187 tessMesh.mVertexCount = getVertexCount();
188 size_t motionSampleCount = getMotionSamplesCount();
189 size_t vertexSize = sizeof(geom::PolygonMesh::VertexBuffer::value_type);
190 size_t vertexStride = motionSampleCount * vertexSize;
191 const void* data = mVertices.data();
192 for (size_t t = 0; t < motionSampleCount; ++t) {
193 size_t offset = t * vertexSize;
194 tessMesh.mVertexBufferDesc.emplace_back(data, offset, vertexStride);
195 }

198// Utility interpolation function for attribute baking
199template <typename T>
200T
201bilinearInterpolate(const Vec2f& uv,
202 const T& a0, const T& a1, const T& a2, const T& a3)
203{
204 return (1.0f - uv[0]) * (1.0f - uv[1]) * a0 +
205 ( uv[0]) * (1.0f - uv[1]) * a1 +
206 ( uv[0]) * ( uv[1]) * a2 +
207 (1.0f - uv[0]) * ( uv[1]) * a3;
208}

210template <typename T>
211void*
212PolyMesh::getBakedAttributeData(const TypedAttributeKey<T>& key,
213 size_t& numElements, AttributeRate &newRate) const
214{
215 int vertsPerFace = (mIsTessellated || getBaseFaceType() == MeshIndexType::QUAD) ?
216 sQuadVertexCount : sTriangleVertexCount;
217 size_t faceCount = getTessellatedMeshFaceCount();
218
219 Attributes *attributes = getAttributes();
220 size_t timeSamples = attributes->getTimeSampleCount(key);
221
222 void* data;
223 newRate = attributes->getRate(key);
224
225 switch (attributes->getRate(key)) {
226 case RATE_CONSTANT:
227 {
228 numElements = timeSamples;
229 T* tdata = new T[numElements];
230 data = tdata;
231 for (size_t t = 0; t < timeSamples; t++) {
232 tdata[t] = attributes->getConstant(key, t);
233 }
234 break;
235 }
236 case RATE_UNIFORM:
237 {
238 numElements = faceCount * timeSamples;
239 T* tdata = new T[numElements];
240 data = tdata;
241 for (size_t f = 0, dstIdx = 0; f < faceCount; f++) {
242 for (size_t t = 0; t < timeSamples; t++) {
243 int baseFaceIdx = mIsTessellated ? mTessellatedToBaseFace[f] : f;
244 tdata[dstIdx++] = attributes->getUniform(key, baseFaceIdx, t);
245 }
246 }
247 break;
248 }
249 case RATE_FACE_VARYING:
250 {
251 numElements = faceCount * vertsPerFace * timeSamples;
252 T* tdata = new T[numElements];
253 data = tdata;
254 for (size_t f = 0, dstIdx = 0; f < faceCount; f++) {
255 int baseFaceIdx = mIsTessellated ? mTessellatedToBaseFace[f] : f;
256 for (int v = 0; v < vertsPerFace; v++) {
257 for (size_t t = 0; t < timeSamples; t++) {
258 if (mIsTessellated) {
259 Vec2f uv = mFaceVaryingUv[f * sQuadVertexCount + v];
260 T a1 = attributes->getFaceVarying(key, baseFaceIdx, 0, t);
261 T a2 = attributes->getFaceVarying(key, baseFaceIdx, 1, t);
262 T a3 = attributes->getFaceVarying(key, baseFaceIdx, 2, t);
263 T a4 = attributes->getFaceVarying(key, baseFaceIdx, 3, t);
264 tdata[dstIdx++] = bilinearInterpolate(uv, a1, a2, a3, a4);
265 } else {
266 tdata[dstIdx++] = attributes->getFaceVarying(key, baseFaceIdx, v, t);
267 }
268 }
269 }
270 }
271 break;
272 }
273 case RATE_VARYING:
274 {
275 numElements = getVertexCount() * timeSamples;
276 T* tdata = new T[numElements];
277 data = tdata;
278 if (mIsTessellated) {
279 // for tessellated mesh we need to do some tricky interpolation as the
280 // varying attrs are on the base mesh's vertices
281
282 // for each tessellated face's verts we interpolate the vertex value
283 // on the base face and then write the value to the array of baked vertex data
284 // note that this will visit the same vertex multiple times but with the same value
285
286 for (size_t tessFaceIdx = 0; tessFaceIdx < faceCount; tessFaceIdx++) {
287 size_t tessFaceOffset = sQuadVertexCount * tessFaceIdx;
288
289 // find the base face
290 int baseFaceIdx = mTessellatedToBaseFace[tessFaceIdx];
291 size_t baseFaceOffset = sQuadVertexCount * baseFaceIdx;
292
293 for (size_t t = 0; t < timeSamples; t++) {
294 // get the varying values of the base face's four verts
295 T a1 = attributes->getVarying(key, mBaseIndices[baseFaceOffset + 0], t);
296 T a2 = attributes->getVarying(key, mBaseIndices[baseFaceOffset + 1], t);
297 T a3 = attributes->getVarying(key, mBaseIndices[baseFaceOffset + 2], t);
298 T a4 = attributes->getVarying(key, mBaseIndices[baseFaceOffset + 3], t);
299
300 // for each tessellated face vertex
301 for (size_t vert = 0; vert < sQuadVertexCount; vert++) {
302 // get the UVs on the base face for the current tessellated face vertex
303 Vec2f uv = mFaceVaryingUv[tessFaceOffset + vert];
304 // interpolate the point on the base face that corresponds to the
305 // tessellated face vertex
306 T result = bilinearInterpolate(uv, a1, a2, a3, a4);
307 // update the value of the tessellated vertex
308 size_t tessVtxIdx = mIndices[tessFaceOffset + vert];
309 tdata[tessVtxIdx * timeSamples + t] = result;
310 }
311 }
312 }
313 } else {
314 // for untessellated mesh we just look up the varying value with the vertex index
315 for (size_t i = 0, dstIdx = 0; i < getVertexCount(); i++) {
316 for (size_t t = 0; t < timeSamples; t++) {
317 tdata[dstIdx++] = attributes->getVarying(key, i, t);
318 }
319 }
320 }
321 break;
322 }
323 case RATE_VERTEX:
324 {
325 if (key == StandardAttributes::sNormal) {
326 // reformat to FACE_VARYING if the normals are vertex rate
327 numElements = mIndices.size() * timeSamples;
328 T* tdata = new T[numElements];
329 data = tdata;
330 for (size_t i = 0, dstIdx = 0; i < mIndices.size(); i++) {
331 for (size_t t = 0; t < timeSamples; t++) {
332 tdata[dstIdx++] = attributes->getVertex(key, mIndices[i], t);
333 }
334 }
335 newRate = RATE_FACE_VARYING;
336 } else {
337 numElements = getVertexCount() * timeSamples;
338 T* tdata = new T[numElements];
339 data = tdata;
340 for (size_t i = 0, dstIdx = 0; i < getVertexCount(); i++) {
341 for (size_t t = 0; t < timeSamples; t++) {
342 tdata[dstIdx++] = attributes->getVertex(key, i, t);
343 }
344 }
345 }
346 break;
347 }
348 default:
349 MNRY_ASSERT(false, "unknown attribute rate");
350 break;
351 }
352
353 return data;
354}

453std::unique_ptr<BakedAttribute>
454PolyMesh::getBakedAttribute(const AttributeKey& key) const
455{
456 int vertsPerFace = (mIsTessellated || getBaseFaceType() == MeshIndexType::QUAD) ?
457 sQuadVertexCount : sTriangleVertexCount;
458
459 Attributes *attributes = getAttributes();
460 size_t timeSamples = attributes->getTimeSampleCount(key);
461
462 std::unique_ptr<BakedAttribute> battr = fauxstd::make_unique<BakedAttribute>();
463
464 battr->mName = key.getName();
465 battr->mTimeSampleCount = timeSamples;
466 battr->mType = key.getType();
467 battr->mData = nullptr;
468
469 switch (battr->mType) {
470 case AttributeType::TYPE_BOOL:
471 battr->mData = getBakedAttributeData(TypedAttributeKey<bool>(key),
472 battr->mNumElements, battr->mRate);
473 break;
474 case AttributeType::TYPE_INT:
475 battr->mData = getBakedAttributeData(TypedAttributeKey<int>(key),
476 battr->mNumElements, battr->mRate);
477 break;
478 case AttributeType::TYPE_LONG:
479 battr->mData = getBakedAttributeData(TypedAttributeKey<long>(key),
480 battr->mNumElements, battr->mRate);
481 break;
482 case AttributeType::TYPE_FLOAT:
483 battr->mData = getBakedAttributeData(TypedAttributeKey<float>(key),
484 battr->mNumElements, battr->mRate);
485 break;
486 case AttributeType::TYPE_DOUBLE:
487 battr->mData = getBakedAttributeData(TypedAttributeKey<double>(key),
488 battr->mNumElements, battr->mRate);
489 break;
490 case AttributeType::TYPE_STRING:
491 battr->mData = getBakedAttributeData(TypedAttributeKey<std::string>(key),
492 battr->mNumElements, battr->mRate);
493 break;
494 case AttributeType::TYPE_RGB:
495 battr->mData = getBakedAttributeData(TypedAttributeKey<scene_rdl2::math::Color>(key),
496 battr->mNumElements, battr->mRate);
497 break;
498 case AttributeType::TYPE_RGBA:
499 battr->mData = getBakedAttributeData(TypedAttributeKey<scene_rdl2::math::Color4>(key),
500 battr->mNumElements, battr->mRate);
501 break;
502 case AttributeType::TYPE_VEC2F:
503 battr->mData = getBakedAttributeData(TypedAttributeKey<Vec2f>(key),
504 battr->mNumElements, battr->mRate);
505 break;
506 case AttributeType::TYPE_VEC3F:
507 battr->mData = getBakedAttributeData(TypedAttributeKey<Vec3f>(key),
508 battr->mNumElements, battr->mRate);
509 break;
510 case AttributeType::TYPE_VEC4F:
511 battr->mData = getBakedAttributeData(TypedAttributeKey<Vec4f>(key),
512 battr->mNumElements, battr->mRate);
513 break;
514 case AttributeType::TYPE_MAT4F:
515 battr->mData = getBakedAttributeData(TypedAttributeKey<scene_rdl2::math::Mat4f>(key),
516 battr->mNumElements, battr->mRate);
517 break;
518 default:
519 MNRY_ASSERT(false, (std::string("unsupported attribute type ") +
520 std::string(attributeTypeName(key.getType())) +
521 std::string(" for attribute ") + std::string(key.getName())).c_str());
522 break;
523 }
524
525 return battr;
526}

528void
529PolyMesh::getBakedMesh(BakedMesh& bakedMesh) const
530{
531 bakedMesh.mName = mName;
532
533 int vertsPerFace = (mIsTessellated || getBaseFaceType() == MeshIndexType::QUAD) ?
534 sQuadVertexCount : sTriangleVertexCount;
535 bakedMesh.mVertsPerFace = vertsPerFace;
536 bakedMesh.mIndexBuffer.resize(mIndices.size());
537 for (size_t i = 0; i < mIndices.size(); i++) {
538 bakedMesh.mIndexBuffer[i] = mIndices[i];
539 }
540
541 bakedMesh.mMotionSampleCount = getMotionSamplesCount();
542
543 bakedMesh.mVertexCount = getVertexCount();
544 bakedMesh.mVertexBuffer.resize(bakedMesh.mVertexCount * bakedMesh.mMotionSampleCount);
545 for (size_t v = 0; v < bakedMesh.mVertexCount; v++) {
546 for (size_t t = 0; t < bakedMesh.mMotionSampleCount; t++) {
547 bakedMesh.mVertexBuffer[v * bakedMesh.mMotionSampleCount + t] = mVertices(v, t);
548 }
549 }
550
551 bakedMesh.mTessellatedToBaseFace = mTessellatedToBaseFace;
552
553 bakedMesh.mFaceToPart.resize(mFaceToPart.size());
554 for (size_t i = 0; i < mFaceToPart.size(); i++) {
555 bakedMesh.mFaceToPart[i] = static_cast<int>(mFaceToPart[i]);
556 }
557
558 Attributes *attributes = getAttributes();
559 const PrimitiveAttributeTable* patable = getPrimitiveAttributeTable();
560 for (const auto& entry : *patable) {
561 const AttributeKey& key = entry.first;
562 if (attributes->hasAttribute(key)) {
563 bakedMesh.mAttrs.push_back(getBakedAttribute(key));
564 }
565 }
566}

574void
575PolyMesh::updateVertexData(const std::vector<float>& vertexData,
576 const XformSamples& prim2render)
577{
578 // TODO the updateVertexData mechanics is an adhoc solution
579 // and need to be updated, it doesn't support motion blur
580 // VertexBuffer so we can only update the first time sample,
581 // this function does not guaranteed work before we revise
582 // the whole update pipeline
583 size_t vertexCount = mVertices.size();
584 if (vertexData.size() != 3 * vertexCount) {
585 size_t updateVertexCount = vertexData.size() / 3;
586 scene_rdl2::logging::Logger::error("PolygonMesh ", getName(), " contains ",
587 vertexCount, " vertices"
588 " while update data contains ", updateVertexCount,
589 " vertices. Mesh topology should remain unchanged"
590 " during geometry update.");
591 vertexCount = std::min(vertexCount, updateVertexCount);
592 }
593 for (size_t i = 0; i < vertexCount; ++i) {
594 Vec3f p(vertexData[3 * i],
595 vertexData[3 * i + 1],
596 vertexData[3 * i + 2]);
597 mVertices(i) = Vec3fa(transformPoint(prim2render[0], p), 0.f);
598 }
599}

601void
602PolyMesh::setupRecomputeVertexNormals(bool fixInvalid)
603{
604 if (mIsTessellated) {
605 mPolyMeshCalcNv = std::unique_ptr<QuadMeshCalcNv>(
606 new QuadMeshCalcNv(fixInvalid));
607 } else {
608 if (getBaseFaceType() == MeshIndexType::QUAD) {
609 mPolyMeshCalcNv = std::unique_ptr<QuadMeshCalcNv>(
610 new QuadMeshCalcNv(fixInvalid));
611 } else {
612 mPolyMeshCalcNv = std::unique_ptr<TriMeshCalcNv>(
613 new TriMeshCalcNv(fixInvalid));
614 }
615 }
616 // set new vertex/face info and create vn

679std::vector<PolyTessellationFactor>
680PolyMesh::computeTessellationFactor(const scene_rdl2::rdl2::Layer *pRdlLayer,
681 const std::vector<mcrt_common::Frustum>& frustums,
682 const PolyTopologyIdLookup& topologyIdLookup) const
683{
684 size_t faceVertexCount = getBaseFaceType() == MeshIndexType::QUAD ?
685 sQuadVertexCount : sTriangleVertexCount;
686 size_t baseFaceCount = mIndices.size() / faceVertexCount;
687
688 const PolygonMesh::VertexBuffer& vertices = mVertices;
689 const PolygonMesh::IndexBuffer& indices = mIndices;
690 std::vector<PolyTessellationFactor> tessellationFactors;
691 tessellationFactors.reserve(baseFaceCount);
692 // only do adaptive tessellation when adaptiveError > 0
693 if (mAdaptiveError > scene_rdl2::math::sEpsilon && !frustums.empty()) {
694 float pixelsPerScreenHeight =
695 frustums[0].mViewport[3] - frustums[0].mViewport[1] + 1;
696 float pixelsPerEdge = mAdaptiveError;
697 float edgesPerScreenHeight = pixelsPerScreenHeight / pixelsPerEdge;
698
699 std::unordered_map<int, int> edgeTessellationFactor;
700 size_t motionSampleCount = vertices.get_time_steps();
701 int indexOffset = 0;
702 for (size_t f = 0; f < baseFaceCount; ++f) {
703 scene_rdl2::math::BBox3f bbox(vertices(indices[indexOffset]).asVec3f());
704 for (size_t v = 0; v < faceVertexCount; ++v) {
705 for (size_t t = 0; t < motionSampleCount; ++t) {
706 bbox.extend(vertices(indices[indexOffset + v], t));
707 }
708 }
709 const scene_rdl2::rdl2::Displacement* displacement =
710 pRdlLayer->lookupDisplacement(getFaceAssignmentId(f));
711 PolyTessellationFactor factor;
712 if (displacement == nullptr) {
713 // set tessellation factor to 0 if no displacement
714 // it doesn't make sense to tessellate polygon mesh if
715 // we are not doing displacement
716 for (size_t v = 0; v < faceVertexCount; ++v) {
717 int vid0 = indices[indexOffset + v];
718 int vid1 = indices[indexOffset + (v + 1) % faceVertexCount];
719 int eid = topologyIdLookup.getEdgeId(vid0, vid1);
720 factor.mEdgeId[v] = eid;
721 edgeTessellationFactor[eid] = scene_rdl2::math::max(
722 0, edgeTessellationFactor[eid]);
723 }
724 } else {
725 // enlarge the bounding box with its diag length and
726 // optional user provide displacement bound padding to avoid
727 // case that undisplaced face got culled out unintionally
728 Vec3f pCenter = scene_rdl2::math::center(bbox);
729 float padding = displacement->get(
730 scene_rdl2::rdl2::Displacement::sBoundPadding);
731 if (padding < 0.0f) {
732 padding = 0.0f;
733 }
734 Vec3f radius(0.5f * scene_rdl2::math::length(bbox.size()) + padding);
735 bbox.lower = pCenter - radius;
736 bbox.upper = pCenter + radius;
737 // frustum culling test
738 bool inFrustum = false;
739 for (size_t i = 0; i < frustums.size(); ++i) {
740 if (frustums[i].testBBoxOverlaps(bbox)) {
741 inFrustum = true;
742 break;
743 }
744 }
745 for (size_t v = 0; v < faceVertexCount; ++v) {
746 int vid0 = indices[indexOffset + v];
747 int vid1 = indices[indexOffset + (v + 1) % faceVertexCount];
748 int eid = topologyIdLookup.getEdgeId(vid0, vid1);
749 factor.mEdgeId[v] = eid;
750 for (size_t t = 0; t < motionSampleCount; ++t) {
751 const Vec3f& v0 = vertices(vid0, t).asVec3f();
752 const Vec3f& v1 = vertices(vid1, t).asVec3f();
753 Vec3f vMid = 0.5f * (v0 + v1);
754 int edgeFactor = 0;
755 if (inFrustum) {
756 edgeFactor = computeEdgeVertexCount(v0, v1,
757 edgesPerScreenHeight, frustums[0].mC2S);
758 }
759 edgeTessellationFactor[eid] = scene_rdl2::math::max(
760 edgeFactor, edgeTessellationFactor[eid]);
761 }
762 }
763 }
764 tessellationFactors.push_back(factor);
765 indexOffset += faceVertexCount;
766 }
767 // Clamp the maximum tessellation factor based on user specified
768 // mesh resolution. Otherwise the tessellation factor can get out
769 // of control when the edge is extremely close to camera near plane
770 int maxEdgeVertexCount = mMeshResolution - 1;
771 for (size_t i = 0; i < tessellationFactors.size(); ++i) {
772 for (size_t e = 0; e < faceVertexCount; ++e) {
773 int eid = tessellationFactors[i].mEdgeId[e];
774 tessellationFactors[i].mEdgeFactor[e] = scene_rdl2::math::clamp(
775 edgeTessellationFactor[eid], 0, maxEdgeVertexCount);
776 }
777 }
778 } else {
779 // uniform tessellation
780 int edgeVertexCount = scene_rdl2::math::max(0, mMeshResolution - 1);
781 for (size_t f = 0; f < baseFaceCount; ++f) {
782 PolyTessellationFactor factor;
783 for (size_t v = 0; v < faceVertexCount; ++v) {
784 factor.mEdgeFactor[v] = edgeVertexCount;
785 }
786 tessellationFactors.push_back(factor);
787 }
788 }
789 return tessellationFactors;
790}

816void
817PolyMesh::initAttributesAndDisplace(const scene_rdl2::rdl2::Layer *pRdlLayer,
818 size_t baseFaceCount, size_t varyingsCount, bool enableDisplacement,
819 bool realtimeMode, bool isBaking,
820 const scene_rdl2::math::Mat4d& world2render)
821{
822 auto& primitiveAttributeTable = mPolyMeshData->mPrimitiveAttributeTable;
823 MeshIndexType baseFaceType = getBaseFaceType();
824 size_t faceVertexCount = baseFaceType == MeshIndexType::QUAD ?
825 sQuadVertexCount : sTriangleVertexCount;
826 std::vector<size_t> faceVaryingsCount(baseFaceCount, faceVertexCount);
827 size_t verticesCount = mVertices.size();
828 // figure out whether we need to calculate smooth shading normal
829 bool calculateSmoothNormal = false;
830 bool hasDisplacement = false;
831 if (realtimeMode) {
832 // for real time mode we always calculate smooth normal,
833 // and we don't apply displacement in this mode now
834 calculateSmoothNormal = true;
835 } else {
836 // if there is displacement, we need to calculate smooth shading normal
837 hasDisplacement = enableDisplacement && hasDisplacementAssignment(pRdlLayer);
838 calculateSmoothNormal = hasDisplacement;
839 if (getSmoothNormal() &&
840 !primitiveAttributeTable.hasAttribute(StandardAttributes::sNormal)) {
841 calculateSmoothNormal = true;
842 }
843 }
844 if (calculateSmoothNormal) {
845 // when we decide to calculate the smooth shading normal,
846 // the attribute rate need to be explicit to RATE_VERTEX
847 // due to the normal computation utility logic
848 if (primitiveAttributeTable.hasAttribute(StandardAttributes::sNormal)) {
849 primitiveAttributeTable.erase(StandardAttributes::sNormal);
850 }
851 // Create empty shading normal buffer that we will calculate during tessellation stage.
852 // (Note: we only support up to 2 motion samples at present)
853 size_t motionSampleCount = std::min((int)getMotionSamplesCount(), 2);
854 std::vector<std::vector<Vec3f>> shadingNormal(motionSampleCount);
855 for (size_t t = 0; t < motionSampleCount; ++t) {
856 shadingNormal[t].resize(verticesCount);
857 }
858 primitiveAttributeTable.addAttribute(StandardAttributes::sNormal,
859 RATE_VERTEX, std::move(shadingNormal));
860 }
861 // interleave PrimitiveAttributeTable
862 setAttributes(Attributes::interleave(primitiveAttributeTable,
863 mPartCount, baseFaceCount, varyingsCount,
864 faceVaryingsCount, verticesCount));
865 getAttributes()->transformAttributes(mPolyMeshData->mXforms,
866 mPolyMeshData->mShutterOpenDelta, mPolyMeshData->mShutterCloseDelta,
867 {{StandardAttributes::sNormal, Vec3Type::NORMAL}});
868
869 if (calculateSmoothNormal) {
870 setupRecomputeVertexNormals(!realtimeMode);
871 }
872
873 if (hasDisplacement) {
874 displaceMesh(pRdlLayer, world2render);
875 }
876 // if it's not real time mode, we don't need the smooth normal utility
877 // after shading normal computation is done
878 if (!realtimeMode) {
879 mPolyMeshCalcNv.reset();
880 }
881
882 if (!isBaking) {
883 // Don't need this data anymore after primitive attributes are initialized,
884 // except if we are baking geometry.
885 mPolyMeshData.reset();
886 }

889void
890PolyMesh::displaceMesh(const scene_rdl2::rdl2::Layer* pRdlLayer,
891 const scene_rdl2::math::Mat4d& world2render)
892{
893 struct VertexToDisplace {
894 VertexToDisplace(): mAssignmentId(-1), mFaceId(0), mVIndex(0)
895 {}
896
897 int mAssignmentId;
898 uint32_t mFaceId;
899 uint32_t mVIndex;
900 };
901
902 MeshIndexType baseFaceType = getBaseFaceType();
903 size_t faceVertexCount =
904 (mIsTessellated || baseFaceType == MeshIndexType::QUAD) ?
905 sQuadVertexCount : sTriangleVertexCount;
906
907 size_t vertexCount = mVertices.size();
908 std::vector<VertexToDisplace> toDisplace(vertexCount);
909 for (size_t f = 0; f < getTessellatedMeshFaceCount(); ++f) {
910 int assignmentId = getFaceAssignmentId(f);
911 if (assignmentId == -1 ||
912 pRdlLayer->lookupDisplacement(assignmentId) == nullptr) {
913 continue;
914 }
915 for (size_t v = 0; v < faceVertexCount; ++v) {
916 uint32_t vid = mIndices[f * faceVertexCount + v];
917 if (toDisplace[vid].mAssignmentId == -1) {
918 toDisplace[vid].mAssignmentId = assignmentId;
919 toDisplace[vid].mFaceId = f;
920 toDisplace[vid].mVIndex = v;
921 }
922 }
923 }
924
925 size_t motionSampleCount = getMotionSamplesCount();
926 PolygonMesh::VertexBuffer displacementResult(vertexCount, motionSampleCount);
927
928 tbb::blocked_range<size_t> range =
929 tbb::blocked_range<size_t>(0, vertexCount);
930
931 tbb::parallel_for(range, [&](const tbb::blocked_range<size_t> &r) {
932 mcrt_common::ThreadLocalState *tls = mcrt_common::getFrameUpdateTLS();
933 shading::TLState *shadingTls = MNRY_VERIFY(tls->mShadingTls.get());
934 Intersection isect;
935 for (size_t v = r.begin(); v < r.end(); ++v) {
936 int assignmentId = toDisplace[v].mAssignmentId;
937 if (assignmentId == -1) {
938 continue;
939 }
940 const scene_rdl2::rdl2::Displacement* displacement =
941 pRdlLayer->lookupDisplacement(assignmentId);
942 const scene_rdl2::rdl2::Geometry* geometry =
943 pRdlLayer->lookupGeomAndPart(assignmentId).first;
944
945 // get primitive attribute table
946 const AttributeTable* table = nullptr;
947 if (displacement->hasExtension()) {
948 const shading::RootShader& rootShader =
949 displacement->get<shading::RootShader>();
950 table = rootShader.getPrimitiveAttributeTable();
951 }
952
953 int tessFaceId = toDisplace[v].mFaceId;
954 int baseFaceId = getBaseFaceId(tessFaceId);
955 int vIndex = toDisplace[v].mVIndex;
956 int vid, vid1, vid2, vid3;
957 Vec2f st, st1, st2, st3;
958 // need to pick one of the triangles on quad to calculate
959 // dpds/dpdt through positions, st coordinates on three vertices
960 getNeighborVertices(baseFaceId, tessFaceId, vIndex,
961 vid, vid1, vid2, vid3, st, st1, st2, st3);
962
963 for (size_t t = 0; t < motionSampleCount; ++t) {
964 Vec3f position = mVertices(vid, t);
965 Vec3f p1 = mVertices(vid1, t);
966 Vec3f p2 = mVertices(vid2, t);
967 Vec3f p3 = mVertices(vid3, t);
968
969
970 Vec3f normal = scene_rdl2::math::normalize(scene_rdl2::math::cross(p2 - p1, p3 - p1));
971 // degenerated surface, assign a valid but meaningless
972 // value as fallback
973 if (!scene_rdl2::math::isFinite(normal)) {
974 normal = Vec3f(0, 0, 1);
975 }
976 Vec3f dpdst[2];
977 if (!computeTrianglePartialDerivatives(
978 p1, p2, p3, st1, st2, st3, dpdst)) {
979 scene_rdl2::math::ReferenceFrame frame(normal);
980 dpdst[0] = frame.getX();
981 dpdst[1] = frame.getY();
982 }
983 Vec2f dSt0 = 0.5f * (st2 - st1);
984 Vec2f dSt1 = 0.5f * (st3 - st1);
985
986 if (getIsReference()) {
987 // If this primitive is referenced by another geometry (i.e. instancing)
988 // then these attributes are given to us in local/object space. Our
989 // shaders expect all State attributes to be given in render space
990 // so we transform them here by world2render. The local2world transform
991 // is assumed to be identity for references.
992 position = scene_rdl2::math::transformPoint(world2render, position);
993 dpdst[0] = scene_rdl2::math::transformVector(world2render, dpdst[0]);
994 dpdst[1] = scene_rdl2::math::transformVector(world2render, dpdst[1]);
995 }
996
997 isect.initDisplacement(tls, table, geometry,
998 pRdlLayer, assignmentId, position, normal,
999 dpdst[0], dpdst[1], st, dSt0[0], dSt1[0], dSt0[1], dSt1[1]);
1000 fillDisplacementAttributes(tessFaceId, vIndex, isect);
1001
1002 Vec3f displace;
1003 shading::displace(displacement, shadingTls, shading::State(&isect), &displace);
1004 // can't add this result directly to vertex buffer here...
1005 // otherwise it will cause wrong dpds/dpdt calculation since
1006 // displacement shaders expect pre-displaced dpds/dpdt values
1007 displacementResult(vid, t) = Vec3fa(displace, 0.f);
1008 }
1009 }
1010 });
1011 // now add the displacement result back to vertex buffer
1012 tbb::parallel_for(range, [&](const tbb::blocked_range<size_t> &r) {
1013 for (size_t v = r.begin(); v < r.end(); ++v) {
1014 int assignmentId = toDisplace[v].mAssignmentId;
1015 if (assignmentId == -1) {
1016 continue;
1017 }
1018 uint32_t faceId = toDisplace[v].mFaceId;
1019 uint32_t vIndex = toDisplace[v].mVIndex;
1020 uint32_t vid = mIndices[faceVertexCount * faceId + vIndex];
1021 for (size_t t = 0; t < motionSampleCount; ++t) {
1022 mVertices(vid, t) += displacementResult(vid, t);
1023 }
1024 }
1025 });
1026
1027 // recompute the shading normal after displacement
1028 if (mPolyMeshCalcNv) {
1029 recomputeVertexNormals();
1030 } else {
1031 bool fixInvalid = true;
1032 setupRecomputeVertexNormals(fixInvalid);
1033 }
1034}
