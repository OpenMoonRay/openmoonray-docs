29///
30/// @class LayerAssignmentId
31/// @brief This class represents the mapping from each sub-primitive
32/// (face/curve) to a corresponding layer assignment id.
33/// Each primitive is constructed with one LayerAssignmentId so that it can
34/// locate the material shader assignment for each sub-primitive. If the whole
35/// primitive shares one material assignment, one ValueType is used
36/// to represent the id (CONSTANT). If the primitive is composed of several
37/// parts, a std::vector<ValueType> with size of sub-primitive count
38/// (for example, curves count in Curves primitive) is used to represent
39/// the id (VARYING)
40class LayerAssignmentId
41{
42public:
43 typedef int ValueType;
44 typedef std::vector<ValueType> ContainerType;
45
46 enum class Type
47 {
48 CONSTANT, ///<all sub-primitives share one assignment id
49 VARYING ///<each sub-primitive maps to its corresponding assignment id
50 };

54 mType(Type::CONSTANT),
55 mConstId(id)
56 {
57 }

61 mType(Type::VARYING),

135 void swap(LayerAssignmentId& other) noexcept
136 {
137 using std::swap; // Allow ADL
138
139 const Type thisType = this->getType();
140 const Type otherType = other.getType();
141
142 if (thisType == Type::CONSTANT && otherType == Type::CONSTANT) {
143 swap(mConstId, other.mConstId);
144 } else if (thisType == Type::VARYING && otherType == Type::VARYING) {
145 swap(mVaryingId, other.mVaryingId);
146 } else if (thisType == Type::CONSTANT && otherType == Type::VARYING) {
147 swapConstVarying(*this, other);
148 } else if (thisType == Type::VARYING && otherType == Type::CONSTANT) {
149 swapConstVarying(other, *this);
150 }
151
152 swap(mType, other.mType);
153 }

184 bool hasValidAssignment() const {
185 if (mType == Type::CONSTANT) {
186 return mConstId >= 0;
187 } else {
188 return std::any_of(mVaryingId.begin(), mVaryingId.end(),
189 [](ValueType id) { return id >= 0; });

193private:
194 // Initially, "a" is const, and "b" is varying.
195 static void swapConstVarying(
196 LayerAssignmentId& a, LayerAssignmentId& b) noexcept
197 {
198 // Technically, we don't have to destroy an int, but it doesn't hurt.
199 // Let's be consistent.
200 auto id = std::move(a.mConstId);
201 detail::destroy(a.mConstId);
202
203 // Move the vector from b to a, making sure to call the constructor of
204 // the new vector.
205 new (&a.mVaryingId) ContainerType(std::move(b.mVaryingId));
206 detail::destroy(b.mVaryingId);
207
208 // Move the saved int to b.
209 new (&b.mConstId) ValueType(std::move(id));
210 }
