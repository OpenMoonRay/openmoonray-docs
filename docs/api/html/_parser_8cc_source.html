60bool
61Parser::main(Arg &arg) const
62{
63 if (arg.noNeedToEvalTest(hasArgument())) return true;
64
65 if (arg.isHelp()) {
66 return arg.msg(usage(arg.comName(), true) + '\n');
67 }
68
69 try {
70 size_t argId = 0;
71 do {
72 bool find = false;
73
74 // option parameter
75 for (auto &itr: mParserItemTbl) {
76 if (itr.isOpt() && arg.isOpt(itr.name())) {
77 if (itr.getArgCount() > arg.size()) {
78 throw(arg.warnMsgLastNext("option argument count error"));
79 }
80 arg.setCurrOptName(itr.name());
81 if (!itr(arg)) { // evaluate option
82 throw(arg.warnMsgEvalOpt("eval option error"));
83 }
84 find = true;
85 }
86 }
87
88 // argument parameter
89 if (!find) {
90 size_t localArgId = 0;
91 for (auto &itr: mParserItemTbl) {
92 if (itr.isArg()) {
93 if (argId == localArgId) {
94 if (itr.getArgCount() > arg.size()) {
95 throw(arg.warnMsgLastNext("command argument count error"));
96 }
97 arg.setCurrArgId(argId);
98 if (!itr(arg)) { // evaluate argument
99 throw(arg.warnMsgEvalArg("eval argument error"));
100 }
101 argId += itr.getArgCount();
102 find = true;
103 } else {
104 localArgId += itr.getArgCount();
105 }
106 }
107 }
108 }
109
110 if (!find) {
111 if (mErrorUnknownOption) {
112 throw(arg.warnMsgCurrVal("Unknown option/argument"));
113 } else {
114 break; // this is not an error
115 }
116 }
117 } while (!arg.emptyArg());
118
119 if (argId != mTotalArgCount) {
120 arg.setCurrArgId(argId);
121 throw(arg.warnMsgEvalArg("needs more argument"));
122 }
123 }
124 catch (std::string error) {
125 arg.msg(error + '\n');
126 return false;
127 }
128 catch (const std::invalid_argument& e) {
129 arg.msg(arg.warnMsgPrevVal("invalid argument") + ' ' + e.what() + " failed\n");
130 return false;
131 }
132 catch (const std::out_of_range& e) {
133 arg.msg(arg.warnMsgPrevVal("out of range") + ' ' + e.what() + " failed\n");
134 return false;
135 }
136 catch (const std::exception& e) {
137 arg.msg(arg.warnMsgPrevVal("exception") + ' ' + e.what() + " failed\n");
138 return false;
139 }
140 catch (...) {
141 arg.msg("unknown error\n");
142 return false;
143 }
144
145 return true;
146}

161std::string
162Parser::usage(const std::string &comName, const bool sort) const
163{
164 std::ostringstream ostr;
165
166 auto newLineIfNeed = [&]() -> std::string { return ostr.str().size() ? "\n" : ""; };
167 auto usageTop = [&]() {
168 ostr << "[Usage] : " << comName;
169 if (hasOptions()) {
170 ostr << " [options]";
171 }
172 if (itemCount(ParserItem::ItemType::ARG)) {
173 ostr << argListOneLine();
174 }
175 };
176
177 if (!comName.empty()) usageTop();
178
179 if (!mDescription.empty()) {
180 ostr << newLineIfNeed() << "[Description] : " << mDescription;
181 }
182 if (hasArgument()) {
183 ostr << newLineIfNeed() << "[Argument]\n" << str_util::addIndent(argListDetail());
184 }
185 if (hasOptions()) {
186 ostr << newLineIfNeed()
187 << (comName.empty() ? "[Command]" : "[Options]") << '\n'
188 << str_util::addIndent(optList(sort));
189 }
190
191 return ostr.str();
192}

204std::string
205Parser::argListDetail() const
206{
207 std::vector<const ParserItem *> argTbl;
208 size_t maxArgMsgLen = 0;
209 for (auto &itr: mParserItemTbl) {
210 if (itr.isArg()) {
211 argTbl.push_back(&itr);
212 maxArgMsgLen = std::max(maxArgMsgLen, itr.getArgMsgLen());
213 }
214 }
215
216 std::ostringstream ostr;
217 for (size_t i = 0; i < argTbl.size(); ++i) {
218 const ParserItem &currItem = *argTbl[i];
219 ostr << currItem.argMsg() << spaces(maxArgMsgLen - currItem.getArgMsgLen());
220 ostr << " : " << currItem.shortMsg();
221 if (i != argTbl.size() - 1) ostr << '\n';
222 }
223
224 return ostr.str();
225}

227std::string
228Parser::optList(const bool sort) const
229{
230 std::vector<const ParserItem *> optTbl;
231 size_t maxNameLen = 0;
232 size_t maxArgMsgLen = 0;
233 for (auto &itr: mParserItemTbl) {
234 if (itr.isOpt()) {
235 optTbl.push_back(&itr);
236 maxNameLen = std::max(maxNameLen, itr.getNameLen());
237 maxArgMsgLen = std::max(maxArgMsgLen, itr.getArgMsgLen());
238 }
239 }
240 if (sort) {
241 std::sort(optTbl.begin(), optTbl.end(),
242 [](const ParserItem *a, const ParserItem *b) {
243 return a->name() < b->name();
244 });
245 }
246
247 std::ostringstream ostr;
248 for (size_t i = 0; i < optTbl.size(); ++i) {
249 const ParserItem &currItem = *optTbl[i];
250 ostr << spaces(maxNameLen - currItem.getNameLen()) << currItem.name();
251 if (maxArgMsgLen) ostr << ' ';
252 ostr << currItem.argMsg() << spaces(maxArgMsgLen - currItem.getArgMsgLen());
253 ostr << " : " << currItem.shortMsg();
254 if (i != optTbl.size() - 1) ostr << '\n';
255 }
256
257 return ostr.str();
258}
