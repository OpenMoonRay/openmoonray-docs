59// bounds callback required by Embree.
60// Accesses the light's bounding volume and copies it to bounds_o.
61static void
62boundsCallback(const struct RTCBoundsFunctionArguments* args)
63{
64 const Light*const* lights = (const Light*const*)args->geometryUserPtr;
65 const BBox3f bbox = lights[args->primID]->getBounds();
66 args->bounds_o->lower_x = bbox.lower.x;
67 args->bounds_o->lower_y = bbox.lower.y;
68 args->bounds_o->lower_z = bbox.lower.z;
69 args->bounds_o->upper_x = bbox.upper.x;
70 args->bounds_o->upper_y = bbox.upper.y;
71 args->bounds_o->upper_z = bbox.upper.z;
72}

74// intersect callback required by Embree.
75static void
76intersectCallback(const RTCIntersectFunctionNArguments* args)
77{
78 int* valid = (int*)args->valid;
79 unsigned int N = args->N;
80 unsigned int primID = args->primID;
81 RTCRayHitN* rayhit = (RTCRayHitN*)args->rayhit;
82 RTCRayN* rays = RTCRayHitN_RayN(rayhit, N);
83 RTCHitN* hits = RTCRayHitN_HitN(rayhit, N);
84 const Light*const* lights = (const Light*const*)args->geometryUserPtr;
85 // get light
86 const Light* light = lights[primID];
87 LightIntersectContext* context = (LightIntersectContext*)args->context;
88 for (unsigned int index = 0; index < N; ++index) {
89 // If the lane is not valid in vector mode or if the light in the accelerator
90 // does not have a corresponding light in the LightSet, then we can skip this
91 // intersection test.
92 if (valid[index] == 0 || context->mLightIdMap[primID] == -1) {
93 continue;
94 }
95
96 if (!(RTCRayN_mask(rays, N, index) & light->getVisibilityMask())) {
97 // skip light if it is masked
98 continue;
99 }
100
101 int rayId = RTCRayN_id(rays, N, index);
102 bool includeRayTerminationLights = context->mIncludeRayTerminationLights[rayId];
103 if (!includeRayTerminationLights && light->getIsRayTerminator()) {
104 // Skip any ray termination lights if we were told not to include them
105 continue;
106 }
107
108 const Vec3f p(
109 RTCRayN_org_x(rays, N, index),
110 RTCRayN_org_y(rays, N, index),
111 RTCRayN_org_z(rays, N, index));
112 const Vec3f wi(
113 RTCRayN_dir_x(rays, N, index),
114 RTCRayN_dir_y(rays, N, index),
115 RTCRayN_dir_z(rays, N, index));
116 float maxDistance = RTCRayN_tfar(rays, N, index);
117
118 const Vec3f* n = nullptr;
119 if (context->mShadingNormal) {
120 // is mShadingNormal == nullptr, it is an invalid shading normal
121 // in c++ code derived from a subsurface material
122 n = &context->mShadingNormal[index];
123 if (n->x > 1.f) {
124 // invalid normal coming from ispc
125 n = nullptr;
126 }
127 }
128
129 LightIntersection isect;
130 bool hit = light->intersect(p, n, wi, RTCRayN_time(rays, N, index),
131 maxDistance, isect);
132 if (!hit) {
133 continue;
134 }
135
136 context->mNumHits[rayId]++;
137
138 if (chooseThisLight(context->mSamples[rayId], context->mDepth[rayId], context->mNumHits[rayId])) {
139 RTCHitN_instID(hits, N, index, 0) = context->mRtcContext.instID[0];
140 RTCHitN_geomID(hits, N, index) = 0;
141 RTCHitN_primID(hits, N, index) = primID;
142 RTCHitN_u(hits, N, index) = isect.uv[0];
143 RTCHitN_v(hits, N, index) = isect.uv[1];
144 RTCHitN_Ng_x(hits, N, index) = isect.N.x;
145 RTCHitN_Ng_y(hits, N, index) = isect.N.y;
146 RTCHitN_Ng_z(hits, N, index) = isect.N.z;
147 context->mData0[rayId] = isect.data[0];
148 context->mData1[rayId] = isect.data[1];
149 // We cannot set the RTCHit distance parameter to the isect.distance. If we do, the ray
150 // traversal will stop at the closest ray hit. Therefore we store the distance of the chosen
151 // light in the context instead.
152 context->mDistance[rayId] = isect.distance;
153 context->mPdf[rayId] = isect.pdf;
154 context->mMeshGeomID[rayId] = isect.geomID;
155 context->mMeshPrimID[rayId] = isect.primID;
156 }
157 }
158}

160// Initialise LightAccelerator using the given light list. We also pass in the Embree device
161// created by the geometry manager so that we don't need to create our own.
162// This function sets all of the callbacks required by Embree.
163
164void
165LightAccelerator::init(const Light*const* lights, int lightCount, const RTCDevice& rtcDevice)
166{
167 // Deal with boundary cases
168 if (lights == nullptr) {
169 mRtcScene = nullptr;
170 mLights = nullptr;
171 mLightCount = 0;
172 mBoundedLights = nullptr;
173 mBoundedLightCount = 0;
174 mUnboundedLights = nullptr;
175 mUnboundedLightCount = 0;
176 return;
177 }
178
179 // Bounded lights should have been sorted to the front - count them
180 int boundedLightCount = 0;
181 for (int l=0; l<lightCount; l++) {
182 if (!lights[l]->isBounded()) {
183 break;
184 }
185 boundedLightCount++;
186 }
187
188 // Check that the rest are all unbounded
189 for (int l=boundedLightCount; l<lightCount; l++) {
190 MNRY_ASSERT(!lights[l]->isBounded());
191 }
192
193 mLights = lights;
194 mLightCount = lightCount;
195
196 mBoundedLights = lights;
197 mBoundedLightCount = boundedLightCount;
198
199 mUnboundedLights = lights + boundedLightCount;
200 mUnboundedLightCount = lightCount - boundedLightCount;
201
202 // Create an Embree scene as long as there are enough lights to put in it.
203 // We check the SCALAR_THRESHOLD_COUNT because even in vector mode,
204 // we fall into scalar code when computing subsurface radiance.
205 // All bounded lights are stored in embree as one UserGeometry
206 if (boundedLightCount >= SCALAR_THRESHOLD_COUNT) {
207 mRtcScene = rtcNewScene(rtcDevice);
208 RTCGeometry rtcGeom = rtcNewGeometry(rtcDevice, RTC_GEOMETRY_TYPE_USER);
209 rtcSetGeometryBuildQuality(rtcGeom, RTC_BUILD_QUALITY_MEDIUM);
210 rtcSetGeometryUserPrimitiveCount(rtcGeom, boundedLightCount);
211 rtcSetGeometryTimeStepCount(rtcGeom, 1);
212 rtcSetGeometryUserData(rtcGeom, (void *)const_cast<Light**>(mBoundedLights));
213 rtcSetGeometryBoundsFunction(rtcGeom, boundsCallback, nullptr);
214 rtcSetGeometryIntersectFunction(rtcGeom, intersectCallback);
215 rtcAttachGeometry(mRtcScene, rtcGeom);
216 rtcCommitGeometry(rtcGeom);
217 rtcReleaseGeometry(rtcGeom);
218 rtcCommitScene(mRtcScene);
219 } else {
220 mRtcScene = nullptr;
221 }
222}

225// The main intersection function for the Embree light acceleration structure.
226// The intention is to avoid a brute-force search through the array of lights to find a randomly hit light.
227// Only the bounded lights (cylinder, disk, rect, sphere, spot) are put in the acceleration structure.
228// Unbounded lights by definition do not have a bounding volume so they must still be iterated through
229// in search of a hit.
230
231int
232LightAccelerator::intersect(const Vec3f &P, const Vec3f* N, const Vec3f &wi,
233 float time, float maxDistance, bool includeRayTerminationLights, int visibilityMask,
234 IntegratorSample1D &samples, int depth, LightIntersection &isect, int &numHits,
235 const int* lightIdMap) const
236{
237 MNRY_ASSERT(mBoundedLightCount >= SCALAR_THRESHOLD_COUNT);
238 MNRY_ASSERT(mRtcScene != nullptr);
239 if (mBoundedLightCount < SCALAR_THRESHOLD_COUNT || mRtcScene == nullptr) {
240 return -1;
241 }
242
243 LightIntersection boundedIsect;
244 int boundedLightIdx = intersectBounded(P, N, wi, time, maxDistance, includeRayTerminationLights,
245 visibilityMask, samples, depth, boundedIsect, numHits, lightIdMap);
246
247 LightIntersection unboundedIsect;
248 int unboundedLightIdx = intersectUnbounded(P, wi, time, maxDistance, includeRayTerminationLights,
249 visibilityMask, samples, depth, unboundedIsect, numHits, lightIdMap);
250
251 if (unboundedLightIdx >= 0) {
252 isect = unboundedIsect;
253 return unboundedLightIdx;
254 }
255
256 isect = boundedIsect;
257 return boundedLightIdx;
258}

261// A simple wrapper for the Embree intersection test against the
262// light acceleration structure. Its main role is to convert between
263// the Moonray-style parameters and the Embree Ray struct.
264int
265LightAccelerator::intersectBounded(const Vec3f &P, const Vec3f* N, const Vec3f &wi,
266 float time, float maxDistance, bool includeRayTerminationLights, int visibilityMask,
267 IntegratorSample1D &samples, int depth, LightIntersection &isect, int &numHits, const int* lightIdMap) const
268{
269 RTCRayHit rayHit;
270 rayHit.ray.org_x = P.x;
271 rayHit.ray.org_y = P.y;
272 rayHit.ray.org_z = P.z;
273 rayHit.ray.tnear = 0.0f;
274 rayHit.ray.dir_x = wi.x;
275 rayHit.ray.dir_y = wi.y;
276 rayHit.ray.dir_z = wi.z;
277 rayHit.ray.time = time;
278 rayHit.ray.tfar = maxDistance;
279 rayHit.ray.mask = visibilityMask;
280 rayHit.ray.id = 0;
281 rayHit.hit.geomID = RTC_INVALID_GEOMETRY_ID;
282 rayHit.hit.instID[0] = RTC_INVALID_GEOMETRY_ID;
283
284 LightIntersectContext context;
285 int iIncludeRayTerminationLights = includeRayTerminationLights;
286 float distance = maxDistance;
287 context.mIncludeRayTerminationLights = &iIncludeRayTerminationLights;
288 context.mData0 = &isect.data[0];
289 context.mData1 = &isect.data[1];
290 context.mDistance = &distance;
291 context.mPdf = &isect.pdf;
292 context.mMeshGeomID = &isect.geomID;
293 context.mMeshPrimID = &isect.primID;
294 context.mShadingNormal = N;
295 context.mSamples = &samples;
296 context.mNumHits = &numHits;
297 context.mDepth = &depth;
298 context.mLightIdMap = lightIdMap;
299 // Call Embree intersection test
300 rtcIntersect1(mRtcScene, &context.mRtcContext, &rayHit);
301 if (rayHit.hit.geomID == RTC_INVALID_GEOMETRY_ID) {
302 isect.distance = maxDistance;
303 return -1;
304 }
305
306 isect.N.x = rayHit.hit.Ng_x;
307 isect.N.y = rayHit.hit.Ng_y;
308 isect.N.z = rayHit.hit.Ng_z;
309 isect.distance = *context.mDistance;
310 isect.uv.x = rayHit.hit.u;
311 isect.uv.y = rayHit.hit.v;
312
313 return rayHit.hit.primID;
314}

317// Randomly intersect a ray against the LightAccelerator's list of unbounded lights.
318
319int
320LightAccelerator::intersectUnbounded(const Vec3f &P, const Vec3f &wi, float time,
321 float maxDistance, bool includeRayTerminationLights, int visibilityMask, IntegratorSample1D &samples,
322 int depth, LightIntersection &isect, int &numHits, const int* lightIdMap) const
323{
324 isect.distance = maxDistance;
325
326 int chosenLightIdx = -1;
327 for (int idx = 0; idx < mUnboundedLightCount; idx++) {
328 // skip lights that do not exist in LightSet
329 if (lightIdMap[idx + mBoundedLightCount] == -1) {
330 continue;
331 }
332 const Light *light = mUnboundedLights[idx];
333 LightIntersection currentIsect;
334
335 if (!(visibilityMask & light->getVisibilityMask())) {
336 // skip light if it is masked
337 continue;
338 }
339
340 if (!includeRayTerminationLights && light->getIsRayTerminator()) {
341 // Skip any ray termination lights if we were told not to include them
342 continue;
343 }
344
345 if (light->intersect(P, nullptr /*unbounded lights do not depend on the shade point normal*/,
346 wi, time, maxDistance, currentIsect)) {
347
348 numHits++;
349
350 if (chooseThisLight(samples, depth, numHits)) {
351 chosenLightIdx = idx + mBoundedLightCount;
352 isect = currentIsect;
353 }
354 }
355 }
356
357 return chosenLightIdx;
358}

361// Wrappers for the vector versions of rtcIntersect. These are called from the ISPC intersection code.
362// We don't currently use the Embree ISPC API due to build system issues: Embree's ISPC build uses CMake
363// to generate the various vector versions of the library; since we don't use CMake this would entail
364// porting over this mechanism to our SCons-based build system, which would involve considerable CM work.
365// For now, the mechanism used here offers the path of least resistance.
366// See VLEN switch above for vector type information.
367
368extern "C" void CPP_lightIntersect(RTCScene scene, RTCRayHitv& rayHitv,
369 int* includeRayTerminationLights, float* isectData0, float* isectData1, SequenceID* sequenceID,
370 uint32_t* totalSamples, uint32_t* sampleNumber, int* depth, float* isectDistance, int* numHits,
371 float* pdf, int* meshGeomId, int* meshPrimId, const scene_rdl2::math::Vec3fv* shadingNormalv,
372 const int* lightIdMap)
373{
374 // Create IntegratorSample1D from ispc array data
375 IntegratorSample1D samples[VLEN];
376 for (int i = 0; i < VLEN; ++i) {
377 samples[i] = IntegratorSample1D(sequenceID[i], totalSamples[i], sampleNumber[i]);
378 }
379
380 LightIntersectContext context;
381 context.mNumHits = numHits;
382 context.mDepth = depth;
383 context.mSamples = (const IntegratorSample1D *)&samples;
384 context.mIncludeRayTerminationLights = includeRayTerminationLights;
385 context.mLightIdMap = lightIdMap;
386
387 // light type specific data passed to eval from intersect or sample
388 context.mData0 = isectData0;
389 context.mData1 = isectData1;
390 // we get the distance to the chosen light here
391 context.mDistance = isectDistance;
392 // mesh light specific parameters
393 context.mPdf = pdf;
394 context.mMeshGeomID = meshGeomId;
395 context.mMeshPrimID = meshPrimId;
396 // create shading normals
397 Vec3f shadingNormals[VLEN];
398 for (int i = 0; i < VLEN; ++i) {
399 shadingNormals[i] = Vec3f(shadingNormalv->x[i], shadingNormalv->y[i], shadingNormalv->z[i]);
400 }
401 context.mShadingNormal = shadingNormals;
402 rtcIntersectv(sAllValidMask, scene, &context.mRtcContext, &rayHitv);
403
404 // need to update the sample number for the ispc SampleIntegrator1D struct
405 for (int i = 0; i < VLEN; ++i) {
406 sampleNumber[i] = samples[i].getSampleNumber();
407 }
408}
