94void
95TestDebugRays::testPrimaryRays()
96{
97 CPPUNIT_ASSERT(mDb);
98
99 std::vector<uint32_t> const &primaryRays = mDb->getPrimaryRayIndices();
100 size_t numPrimaryRays = primaryRays.size();
101 CPPUNIT_ASSERT(numPrimaryRays);
102
103 // count the number of primary rays using brute force iteration
104 size_t numPrimaryRays2 = 0;
105 DebugRayDatabase::Iterator it(*mDb);
106 while (!it.isDone()) {
107 DebugRayVertex const *vertex = &(*it);
108 if (vertex->isRoot())
109 ++numPrimaryRays2;
110 ++it;
111 }
112
113 // iterate through the db and verify the mNumSubnodes member is correct,
114 // this is the number of children at this particular subtree including the
115 // current node
116
117 // get first primary ray via iteration
118 DebugRayVertex const *startVert = &(*DebugRayDatabase::Iterator(*mDb));
119 DebugRayVertex const *currVert = startVert;
120 CPPUNIT_ASSERT(startVert == mDb->getRay(primaryRays[0]));
121
122 for (size_t i = 0; i < numPrimaryRays; ++i, ++currVert) {
123
124 CPPUNIT_ASSERT(currVert->isRoot());
125
126 // make sure primary indices are consistent with list of rays
127 size_t ofs = currVert - startVert;
128 CPPUNIT_ASSERT(ofs == primaryRays[i]);
129
130 // loop over all children
131 uint32_t numSubnodes = currVert->mNumSubnodes;
132 CPPUNIT_ASSERT(numSubnodes >= 1);
133
134 for (uint32_t j = 1; j < numSubnodes; ++j) {
135 ++currVert;
136 CPPUNIT_ASSERT(!currVert->isRoot());
137 }
138 }
139}

143void
144TestDebugRays::testRectFilter()
145{
146 CPPUNIT_ASSERT(mDb);
147
148 int minX = 123;
149 int maxX = 232;
150 int minY = 12;
151 int maxY = 121;
152
153 // do manual iteration to get baseline count of ray vertices
154 unsigned refResult = 0;
155
156 {
157 DebugRayDatabase::Iterator it(*mDb);
158 while (!it.isDone()) {
159
160 DebugRayVertex const *vertex = &(*it);
161
162 if (vertex->mScreenX >= minX && vertex->mScreenX <= maxX &&
163 vertex->mScreenY >= minY && vertex->mScreenY <= maxY)
164 ++refResult;
165
166 ++it;
167 }
168 }
169
170 // verify that we get the same results when using a filter
171 unsigned filterResult = 0;
172
173 {
174 DebugRayFilter filter;
175 filter.mViewport = scene_rdl2::math::BBox2i(scene_rdl2::math::Vec2i(minX, minY), scene_rdl2::math::Vec2i(maxX, maxY));
176
177 DebugRayDatabase::Iterator it(*mDb, &filter);
178
179 while (!it.isDone()) {
180 ++filterResult;
181 ++it;
182 }
183 }
184
185 CPPUNIT_ASSERT(refResult == filterResult);
186}

190void
191TestDebugRays::testTagFilter()
192{
193 CPPUNIT_ASSERT(mDb);
194
195 // do manual iteration to get baseline count of tagged rays
196 unsigned refResultAny = 0;
197 unsigned refResultAll = 0;
198
199 // pick some arbitrary tag
200 uint32_t tag = 0x14200;
201
202 {
203 DebugRayDatabase::Iterator it(*mDb);
204 while (!it.isDone()) {
205 DebugRayVertex const *vertex = &(*it);
206
207 if (vertex->mUserTags & tag)
208 ++refResultAny;
209
210 if ((vertex->mUserTags & tag) == tag)
211 ++refResultAll;
212
213 ++it;
214 }
215 }
216
217 // verify that we get the same results when using a filter
218 unsigned filterResultAny = 0;
219 unsigned filterResultAll = 0;
220
221 {
222 DebugRayFilter filter;
223 filter.mTags = tag;
224
225 filter.mMatchAll = false;
226 DebugRayDatabase::Iterator it(*mDb, &filter);
227 while (!it.isDone()) {
228 ++filterResultAny;
229 ++it;
230 }
231
232 filter.mMatchAll = true;
233 it = DebugRayDatabase::Iterator(*mDb, &filter);
234 while (!it.isDone()) {
235 ++filterResultAll;
236 ++it;
237 }
238
239 }
240
241 CPPUNIT_ASSERT(refResultAny == filterResultAny);
242 CPPUNIT_ASSERT(refResultAll == filterResultAll);
243}

279void
280TestDebugRays::populateRecorder(DebugRayRecorder *recorder)
281{
282 recorder->record();
283
284 static uint32_t seed = 0x0eadbeef;
285 scene_rdl2::util::Random rnd(seed);
286
287 MOONRAY_THREADSAFE_STATIC_WRITE(++seed);
288
289 DebugRayVertex *parent = nullptr;
290 uint32_t sx = rnd.getNextUInt(WIDTH);
291 uint32_t sy = rnd.getNextUInt(HEIGHT);
292 for (size_t i = 0; i < NUM_VERTICES_PER_RECORDER; ++i) {
293
294 uint32_t tags = rnd.getNextUInt();
295
296 DebugRayVertex *vertex = nullptr;
297 if (!parent) {
298 parent = recorder->startNewRay(Vec3f(scene_rdl2::math::zero), sx, sy, tags);
299 }
300 else {
301 vertex = recorder->extendRay(parent);
302
303 // in a real app, we'd fill in the all the rest of the
304 // vertex user fields here also
305 vertex->mUserTags = tags;
306 }
307
308 float x = rnd.getNextFloat();
309 if (x < 0.05f || parent->mDepth == 254) {
310 // start new ray
311 parent = nullptr;
312 sx = rnd.getNextUInt(WIDTH);
313 sy = rnd.getNextUInt(HEIGHT);
314 } else if (x < 0.6f) {
315 // continue ray
316 parent = vertex;
317 } else {
318 // leave parent alone which will create a sibling
319 }
320 }
321
322 recorder->stopRecording();
323}
