43Curves::Curves(Type type,
44 SubType subtype,
45 int tessellationRate,
46 CurvesVertexCount&& curvesVertexCount,
47 VertexBuffer&& vertices,
48 LayerAssignmentId&& layerAssignmentId,
49 shading::PrimitiveAttributeTable&& primitiveAttributeTable)
50{
51 if (type == Type::LINEAR) {
52 mImpl = fauxstd::make_unique<Impl>(new internal::LineSegments(
53 static_cast<internal::Curves::Type>(type),
54 subtype,
55 std::move(curvesVertexCount),
56 std::move(vertices), std::move(layerAssignmentId),
57 std::move(primitiveAttributeTable)),
58 type, subtype, tessellationRate);
59 } else if (type == Type::BEZIER) {
60 mImpl = fauxstd::make_unique<Impl>(new internal::BezierSpanChains(
61 static_cast<internal::Curves::Type>(type),
62 subtype,
63 std::move(curvesVertexCount),
64 std::move(vertices), std::move(layerAssignmentId),
65 std::move(primitiveAttributeTable)),
66 type, subtype, tessellationRate);
67 } else if (type == Type::BSPLINE) {
68 mImpl = fauxstd::make_unique<Impl>(new internal::BSpline(
69 static_cast<internal::Curves::Type>(type),
70 subtype,
71 std::move(curvesVertexCount),
72 std::move(vertices), std::move(layerAssignmentId),
73 std::move(primitiveAttributeTable)),
74 type, subtype, tessellationRate);
75 } else {
76 // only support linear/bezier/bspline curve at this moment
77 MNRY_ASSERT_REQUIRE(false);
78 }
79}

161static Primitive::DataValidness
162checkLineSegmentsData(
163 const Curves::CurvesVertexCount& curvesVertexCount,
164 const Curves::VertexBuffer& vertices,
165 const shading::PrimitiveAttributeTable& primitiveAttributeTable,
166 std::string* message)
167{
168 size_t curvesCount = curvesVertexCount.size();
169 size_t verticesCount = 0;
170 for (size_t i = 0; i < curvesCount; ++i) {
171 size_t n = curvesVertexCount[i];
172 // need at leat 2 vertices to form a line segment
173 if (n < 2) {
174 if (message) {
175 std::stringstream errMsg;
176 errMsg << "Invalid number of vertices (" << n <<
177 ") in a curve. A linear line segment should contain at least"
178 " two vertices per curve.";
179 *message = errMsg.str();
180 }
181 return Primitive::DataValidness::INVALID_TOPOLOGY;
182 }
183 verticesCount += n;
184 }
185 if (vertices.size() != verticesCount) {
186 if (message) {
187 std::stringstream errMsg;
188 errMsg << "Vertex buffer size (" << vertices.size() <<
189 ") doesn't match the total number of vertices " <<
190 verticesCount << " index buffer specified.";
191 *message = errMsg.str();
192 }
193 return Primitive::DataValidness::INVALID_VERTEX_BUFFER;
194 }
195 return Primitive::DataValidness::VALID;
196}

198static Primitive::DataValidness
199checkCubicCurvesData(
200 const Curves::CurvesVertexCount& curvesVertexCount,
201 const Curves::VertexBuffer& vertices,
202 const shading::PrimitiveAttributeTable& primitiveAttributeTable,
203 const Curves::Type type,
204 const int tessellationRate,
205 std::string* message)
206{
207 size_t curvesCount = curvesVertexCount.size();
208 size_t varyingsCount = 0;
209 size_t faceVaryingsCount = 0;
210 size_t verticesCount = 0;
211 // each bezier curve should have 3 * k + 1 vertices
212 // since each span has 4 vertices and the neighbor
213 // spans share one vertex
214 for (size_t i = 0; i < curvesCount; ++i) {
215 size_t n = curvesVertexCount[i];
216 // need at least 4 vertices to form a cubic spline span
217 if (type == Curves::Type::BEZIER && (n - 1) % 3 != 0) {
218 if (message) {
219 std::stringstream errMsg;
220 errMsg << "Invalid number of vertices (" << n <<
221 ") in a curve. A Bezier curve should be 3 * k + 1 vertices.";
222 *message = errMsg.str();
223 }
224 return Primitive::DataValidness::INVALID_TOPOLOGY;
225 }
226 if (n < 4) {
227 if (message) {
228 std::stringstream errMsg;
229 errMsg << "Invalid number of vertices (" << n <<
230 ") in a curve. A CubicSpline curve should have at least 4 vertices.";
231 *message = errMsg.str();
232 }
233 return Primitive::DataValidness::INVALID_TOPOLOGY;
234 }
235 size_t spansInCurve = 0;
236 if (type == Curves::Type::BEZIER) {
237 spansInCurve = (n - 1) / 3;
238 } else if (type == Curves::Type::BSPLINE) {
239 spansInCurve = n - 3;
240 }
241 varyingsCount += spansInCurve + 1;
242 faceVaryingsCount += spansInCurve + 1;
243 verticesCount += n;
244 }
245 if (vertices.size() != verticesCount) {
246 if (message) {
247 std::stringstream errMsg;
248 errMsg << "Vertex buffer size (" << vertices.size() <<
249 ") doesn't match the total number of vertices " <<
250 verticesCount << " index buffer specified.";
251 *message = errMsg.str();
252 }
253 return Primitive::DataValidness::INVALID_VERTEX_BUFFER;
254 }
255 if (tessellationRate <= 0) {
256 if (message) {
257 std::stringstream errMsg;
258 errMsg << "Invalid tessellation rate (" << tessellationRate <<
259 "). Must be > 0.";
260 *message = errMsg.str();
261 }
262 return Primitive::DataValidness::INVALID_TOPOLOGY;
263 }
264 return Primitive::DataValidness::VALID;
265}

267Primitive::DataValidness
268Curves::checkPrimitiveData(Type type,
269 SubType subtype,
270 int tessellationRate,
271 const CurvesVertexCount& curvesVertexCount,
272 const VertexBuffer& vertices,
273 const shading::PrimitiveAttributeTable& primitiveAttributeTable,
274 std::string* message)
275{
276 Primitive::DataValidness result;
277 switch (type) {
278 case Type::LINEAR:
279 result = checkLineSegmentsData(curvesVertexCount, vertices,
280 primitiveAttributeTable, message);
281 break;
282 case Type::BEZIER:
283 case Type::BSPLINE:
284 result = checkCubicCurvesData(curvesVertexCount, vertices,
285 primitiveAttributeTable, type, tessellationRate, message);
286 break;
287 default:
288 {
289 if (message) {
290 *message = "Invalid curves type";
291 }
292 result = DataValidness::INVALID_TOPOLOGY;
293 }
294 break;
295 }
296
297 return result;
298}

300void
301Curves::transformPrimitive(
302 const MotionBlurParams& motionBlurParams,
303 const shading::XformSamples& prim2render)
304{
305 size_t motionSamplesCount = getMotionSamplesCount();
306 shading::XformSamples p2r = prim2render;
307 if (motionSamplesCount > 1 && prim2render.size() == 1) {
308 p2r.resize(motionSamplesCount, prim2render[0]);
309 }
310
311 const shading::PrimitiveAttributeTable* primAttrTab = mImpl->mCurves->getPrimitiveAttributeTable();
312 transformVertexBuffer(mImpl->mCurves->getVertexBuffer(), p2r, motionBlurParams,
313 mImpl->mCurves->getMotionBlurType(), mImpl->mCurves->getCurvedMotionBlurSampleCount(),
314 primAttrTab);
315
316 float shutterOpenDelta, shutterCloseDelta;
317 motionBlurParams.getMotionBlurDelta(shutterOpenDelta, shutterCloseDelta);
318 mImpl->mCurves->getAttributes()->transformAttributes(p2r, shutterOpenDelta, shutterCloseDelta, {});
319}
