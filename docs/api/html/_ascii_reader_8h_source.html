150 // Constructs a new object of type T, but allocates the memory as a full
151 // userdata on Lua's GC heap. This means its lifetime will be handled by
152 // the Lua GC. The named metatable will be set as the userdata's metatable,
153 // and any additional arguments will be forwarded to T's constructor. When
154 // this function returns, the newly constructed object will be on the top
155 // of the Lua stack.
156 template <typename T, typename... Args>
157 T* boxNew(const char* metatable, Args&&... args);
158
159 // Unboxes a full userdata that you constructed with boxNew() and returns
160 // a typed pointer.
161 template <typename T>
162 T* unbox(void* boxPtr);
163
164 // Boxes a pointer for an object which is *NOT* owned by Lua's GC heap.
165 // This is a pointer owned by the host C++ program, whose lifetime is not
166 // dictated by Lua. The named metatable will be set as the boxed pointer's
167 // metatable. This is more or less a lightuserdata (just a boxed pointer),
168 // with the primary difference that you can attach a metatable to this
169 // boxed pointer and not to a lightuserdata.
170 template <typename T>
171 void boxPtr(const char* metatable, T* objectPtr);
172
173 // Unboxes a userdata which was boxed with boxPtr() and returns a typed

205 // Pushes a new table onto the stack which represents a vector (type T)

248 // If the value at the given index on the stack is a userdata which matches
249 // the named metatable, it unboxes the userdata as a BoxedT, then passes
250 // that value to the constructor of an AttrT, and returns the AttrT. This
251 // allows for an explicit type conversion between boxed types and returned
252 // types.
253 //
254 // For example, When extracting a Vec3d, both AttrT and BoxedT are Vec3d,
255 // because all Vec3's are boxed as Vec3d's. However, when extracting a
256 // Vec3f, AttrT is Vec3f while BoxedT is Vec3d. This requires that Vec3f's
257 // are constructible from Vec3d's.
258 template <typename AttrT, typename BoxedT>
259 AttrT extractComplex(int index, const char* metatable);

261 // If the value at the given index on the stack is a userdata which

265 // does not pass the metatable check, throws a TypeError.

296 // Helper function similar to setSingleAttr(), but instead it sets a vector
297 // attribute from a table of values. The arguments so and attr specify the
298 // SceneObject and attribute, while index specifies the index on the stack
299 // where the *TABLE* of values can be found. The type parameter VecT is the
300 // vector type to set (e.g. FloatVector). The extractor callable is
301 // identical to setSingleAttr(). It should take a Lua stack index and return
302 // a *SINGLE* value of the appropriate C++ type. This extractor function is
303 // called on each element of the vector.
304 //
305 // Example:
306 // lua_State* luaState = ...;
307 // SceneObject* so = ...;
308 // const Attribute* attr = ...;
309 // int valueIndex = 3; // table of float values at stack index 3
310 // setVectorAttr<FloatVector>(so, attr, valueIndex, [this](int index) {
311 // return extractNumeric<Float>(index);
312 // });

371// Macro for declaring both a Lua callback handler and its static dispatch
372// function in one go. This is necessary because we can't pass a function
373// pointer to an instance member function when registering callbacks with Lua.
374// The static dispatch function forwards to a non-static function once we
375// resolve the AsciiReader instance with loadInstancePtr().
376//
377// Any member function which you intend to register as a Lua callback must be
378// declared here with RDL2_LUA_DECLARE() and defined in AsciiReader.cc with
379// RDL2_LUA_DEFINE().
380#define RDL2_LUA_DECLARE(func_name) \
381 static int func_name##_DISPATCHER(lua_State* state); \
382 int func_name##_HANDLER()
