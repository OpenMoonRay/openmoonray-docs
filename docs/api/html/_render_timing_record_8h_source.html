42 //
43 // Measurement timing inside RenderDriver::renderPasses() function
44 //
45 // |<------ ACTIVE : active working time ----->|
46 // +-+-----+---+----------------------------+-------+--+-----+ -> time+
47 // ^ ^ : ^ ^ render tiles ^ queue ^ ^ : ^
48 // | | : | | | drain | | : |
49 // | | : | | | | | : +-- allEnd : finish thread (ENDGAP = allEnd - timeEnd)
50 // | | : | | | | | +... waiting for other all threads are finished
51 // | | : | | | | |
52 // | | : | | | | +-- timeEnd : completed rendering
53 // | | : | | | | (FINISHUP = timeEnd - timeQueueDraining)
54 // | | : | | | |
55 // | | : | | | +-- timeQueueDraining : completed queue draining stage
56 // | | : | | | (QUEUEDRAINING = timeQueueDraining - timeRenderTiles)
57 // | | : | | |
58 // | | : | | +-- timeRenderTiles : completed tiles
59 // | | : | | (RENDERTILES = timeRenderTiles - timeInitTexturing)
60 // | | : | |
61 // | | : | +-- timeInitTexturing : completed texture init (INITTEXTURING = timeInitTextureing - timeReady)
62 // | | : |
63 // | | : +-- timeReady : thread is read to work (READY = timeReady - start)
64 // | | +... waiting for other all threads are booted
65 // | |
66 // | +-- timeStart : render engine thread booted and start working (START = timeStart - start)
67 // +-- start
68 //
69 enum class Tag : int {
70 START = 0, // duration of start to thread is start working
71 READY, // duration of start to ready to working
72 INITTEXTURING, // duration of initialize texture staff
73 RENDERTILES, // duration of rendering tiles
74 QUEUEDRAINING, // duration of queue draining
75 FINISHUP, // duration of finish up
76 ENDGAP, // duration of end gap (waiting other threads are finished)
77 ACTIVE, // actively working time
78 PROCESSEDTILESTOTAL, // processed tiles total number by double
79 PROCESSEDSAMPLETOTAL, // processed sample total number by double
80
81 SIZE // total tag size
82 };

130 void init() {
131 mTimeBudget = 0.0;
132 mStartTime = 0.0;
133 mEndTime = 0.0;
134 mDuration = 0.0;
135
136 mSamplesPerTile = 0;
137 mSamplesAll = 0;
138
139 for (auto &currEng : mRenderEngines) currEng.init();
140 mRenderEngineMin.init(DBL_MAX);
141 mRenderEngineMax.init();
142 mRenderEngineAverage.init();
143
144 mSampleCost = 0.0;

153 void startRenderPasses(const size_t totalEngine) {
154 mStartTime = scene_rdl2::util::getSeconds();
155 if (mRenderEngines.size() < totalEngine) mRenderEngines.resize(totalEngine);
156 }
157 void finalizeRenderPasses() {
158 double now = scene_rdl2::util::getSeconds();
159
160 mRenderEngineMin.init(DBL_MAX);
161 mRenderEngineMax.init();
162 mRenderEngineAverage.init();
163 mSamplesAll = 0;
164 for (auto &currEng : mRenderEngines) {
165 double currGap = now - currEng.get(RenderEngineTimingRecord::Tag::ENDGAP);
166 currEng.set(RenderEngineTimingRecord::Tag::ENDGAP, currGap); // update by proper gap duration.
167
168 mRenderEngineMin != currEng; // min value update
169 mRenderEngineMax ^= currEng; // max value update
170 mRenderEngineAverage += currEng; // add
171 mSamplesAll += static_cast<unsigned>(currEng.get(RenderEngineTimingRecord::Tag::PROCESSEDSAMPLETOTAL));
172 }
173 mRenderEngineAverage *= static_cast<double>(1.0 / (double)(std::max(mRenderEngines.size(), (size_t)1)));
174
175 mEndTime = scene_rdl2::util::getSeconds();
176 mDuration = mEndTime - mStartTime;
177
178 double averageActiveDuration = getActiveDuration();
179 mSampleCost = averageActiveDuration / static_cast<double>(std::max(mSamplesPerTile, (unsigned)1));
180 mOverheadCost = mDuration - averageActiveDuration;
181 }

245 void setRenderPassesSamplesPerTile(const unsigned newSamplesPerTile) {
246#ifdef KEEP_ALL_PASSES_INFO
247 if (mRenderPasses.size() < mNumPassesRendered + 1) mRenderPasses.resize(mNumPassesRendered + 1); // Just in case
248 RenderPassesTimingRecord &currPass = mRenderPasses[mNumPassesRendered];
249#else // else KEEP_ALL_PASSES_INFO
250 updateRenderPassesTable(newSamplesPerTile, 4096); // maintaine passes for last 4096 samples per tile
251 RenderPassesTimingRecord &currPass = mRenderPasses[0];
252#endif // end !KEEP_ALL_PASSES_INFO
253 currPass.init();
254 currPass.setTimeBudget(mTimeBudget);
255 currPass.setRenderPassesSamplesPerTile(newSamplesPerTile);

257 RenderPassesTimingRecord &startRenderPasses(const size_t totalEngine) {
258#ifdef KEEP_ALL_PASSES_INFO
259 if (mRenderPasses.size() < mNumPassesRendered + 1) mRenderPasses.resize(mNumPassesRendered + 1); // Just in case
260 RenderPassesTimingRecord &currPass = mRenderPasses[mNumPassesRendered];
261#else // else KEEP_ALL_PASSES_INFO
262 if (mRenderPasses.empty()) mRenderPasses.resize(1); // This is for Batch/Progress render mode
263 RenderPassesTimingRecord &currPass = mRenderPasses[0];
264#endif // end !KEEP_ALL_PASSES_INFO
265 currPass.startRenderPasses(totalEngine);
266 return currPass;
267 }
268 void finalizeRenderPasses() {
269#ifdef KEEP_ALL_PASSES_INFO
270 RenderPassesTimingRecord &currPass = mRenderPasses[mNumPassesRendered];
271#else // else KEEP_ALL_PASSES_INFO
272 RenderPassesTimingRecord &currPass = mRenderPasses[0];
273#endif // end !KEEP_ALL_PASSES_INFO
274 currPass.finalizeRenderPasses();
275
276 mNumSamplesPerTile += currPass.getSamplesPerTile();
277 mNumSamplesAll += currPass.getSamplesAll();
278 mSampleCost = currPass.getSampleCost();
279 mOverheadCost = currPass.getOverheadCost();
280 mTotalOverheadDuration += currPass.getOverheadCost();
281 mTotalActiveDuration += currPass.getActiveDuration();
282 mAveragedSampleCost = mTotalActiveDuration / static_cast<double>(std::max(mNumSamplesPerTile, (unsigned)1));
283 /* Useful debug message
284 std::cerr << ">> RenderTimingRecord.h addSamples:" << currPass.getSamplesPerTile()
285 << " addActiveDuration:" << currPass.getActiveDuration()
286 << " sampleCost:" << currPass.getSampleCost()
287 << std::endl;
288 std::cerr << ">> RenderTimingRecord.h mNumSamplesPerTile:" << mNumSamplesPerTile
289 << " mTotalActiveDuration:" << mTotalActiveDuration
290 << " mAveragedSampleCost: " << mAveragedSampleCost
291 << std::endl;
292 */
293
294 if (mNumPassesRendered == 0) {
295 mEstimatedSampleCost = currPass.getSampleCost(); // estimation phase result
296 }
297
298 ++mNumPassesRendered;
299 }

317 bool isComplete(const double timeBudgetSec, const double now)
318 // Do we have time to render more ?
319 {
320 if (timeBudgetSec < (mOverheadCost + mSampleCost)) {
321 mRenderFrameEndTime = now;
322 mRenderFrameDuration = mRenderFrameEndTime - mRenderFrameStartTime;
323 return true; // this frame is completed. out of time budget.
324 }
325 return false; // we should render more
326 }
327 double estimateSamples(const double timeBudgetSec)
328 // Compute how many new samples we can safely render within the remaining time interval.
329 // Estimation is done based on updated sampling cost by at least last 64 samples per tile.
330 {
331 mTimeBudget = timeBudgetSec;
332 double remainingTime = mTimeBudget - mOverheadCost;
333
334 double sampleCost = 0.0;
335 {
336 // We have to estimate sampleCost by history of last several passes.
337 // mRenderPasses already maintained and keeps up to at least 64 samples per tile information.
338 double totalActiveDuration = 0.0;
339 unsigned totalSamplesPerTile = 0;
340 for (size_t i = 0; i < mRenderPasses.size(); ++i) {
341 totalActiveDuration += mRenderPasses[i].getActiveDuration();
342 totalSamplesPerTile += mRenderPasses[i].getSamplesPerTile();
343 }
344 sampleCost = totalActiveDuration / static_cast<double>(std::max(totalSamplesPerTile, (unsigned)1));
345# ifdef PRINT_DEBUG_MESSAGE
346 std::ostringstream ostr;
347 ostr << "estimateSamples mRenderPasses.size():" << mRenderPasses.size() << " {\n";
348 for (size_t i = 0; i < mRenderPasses.size(); ++i) {
349 ostr << " i:" << i
350 << " activeDuration:" << mRenderPasses[i].getActiveDuration()
351 << " samplePerTile:" << mRenderPasses[i].getSamplesPerTile() << '\n';
352 }
353 ostr << "} totalActiveDuration:" << totalActiveDuration
354 << " totalSamplesPerTile:" << totalSamplesPerTile
355 << " sampleCost:" << sampleCost;
356 std::cerr << ostr.str() << std::endl;
357# endif // end PRINT_DEBUG_MESSAGE
358 }
359 return (remainingTime < 0.0)? 0.0: (remainingTime / sampleCost);
360 }
361 double estimateSamplesAllStint(const double timeBudgetSec)
362 // Compute how many new samples we can safely render within the remaining time interval.
363 // Estimation is done based on updated sampling cost by all executed passes from reset().
364 {
365 mTimeBudget = timeBudgetSec;
366 double remainingTime = mTimeBudget - mOverheadCost;
367 /* Useful debug message
368 std::cerr << ">> RenderTimingRecord.h remainingTime:" << remainingTime
369 << " mAveragedSampleCost:" << mAveragedSampleCost << std::endl;
370 */
371 return (remainingTime < 0.0)? 0.0: (remainingTime / mAveragedSampleCost);
372 }
373 double actualSampleCost() const { // compute actual sample cost based on actual active duration of engine thread
374 return (mNumSamplesPerTile <= 1)? 0.0: mTotalActiveDuration / static_cast<double>(mNumSamplesPerTile - 1);
375 }
