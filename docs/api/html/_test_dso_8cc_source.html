43void
44TestDso::testIsValidDso()
45{
46 // Reject bad filenames.
47 CPPUNIT_ASSERT(!Dso::isValidDso("", false));
48 CPPUNIT_ASSERT(!Dso::isValidDso(".so", false));
49 CPPUNIT_ASSERT(!Dso::isValidDso("rdl2_test", false));
50 CPPUNIT_ASSERT(!Dso::isValidDso("", true));
51 CPPUNIT_ASSERT(!Dso::isValidDso(".so.proxy", true));
52 CPPUNIT_ASSERT(!Dso::isValidDso("rdl2_test", true));
53
54 // Non-RDL .so's should not be valid.
55 CPPUNIT_ASSERT(!Dso::isValidDso("BadObject.so", false));
56 CPPUNIT_ASSERT(!Dso::isValidDso("BadObject.so.proxy", true));
57
58 // Good DSOs should work, regardless of path or extension case.
59 CPPUNIT_ASSERT(Dso::isValidDso("ExampleObject.so", false));
60 CPPUNIT_ASSERT(Dso::isValidDso("./ExampleObject.so", false));
61 CPPUNIT_ASSERT(Dso::isValidDso("ExampleObject.SO", false));
62 CPPUNIT_ASSERT(Dso::isValidDso("ExampleObject.so.proxy", true));
63 CPPUNIT_ASSERT(Dso::isValidDso("./ExampleObject.so.proxy", true));
64 CPPUNIT_ASSERT(Dso::isValidDso("ExampleObject.SO.PROXY", true));
65}

67void
68TestDso::testFindDso()
69{
70 // Location of the correct path shouldn't matter.
71 CPPUNIT_ASSERT(util::findFile("ExampleObject.so", "..:.:ref") == "./ExampleObject.so");
72 CPPUNIT_ASSERT(util::findFile("ExampleObject.so", ".:..:ref") == "./ExampleObject.so");
73 CPPUNIT_ASSERT(util::findFile("ExampleObject.so", "ref:..:.") == "./ExampleObject.so");
74
75 // Single directory search paths should work.
76 CPPUNIT_ASSERT(util::findFile("ExampleObject.so", ".") == "./ExampleObject.so");
77
78 // If the DSO can't be found, it should return an empty string.
79 CPPUNIT_ASSERT(util::findFile("ExampleObject.so", "ref").empty());
80 CPPUNIT_ASSERT(util::findFile("Nonexistent.so", ".").empty());
81
82 // An empty search path shouldn't find anything.
83 CPPUNIT_ASSERT(util::findFile("ExampleObject.so", "").empty());
84}

86void
87TestDso::testLazyLoading()
88{
89 // A proper DSO should resolve each symbol on demand.
90 Dso example("ExampleObject", ".");
91 CPPUNIT_ASSERT(example.mDeclareFunc == nullptr);
92 CPPUNIT_ASSERT(example.mCreateFunc == nullptr);
93 CPPUNIT_ASSERT(example.mDestroyFunc == nullptr);
94 example.getDeclare();
95 CPPUNIT_ASSERT(example.mDeclareFunc != nullptr);
96 CPPUNIT_ASSERT(example.mCreateFunc == nullptr);
97 CPPUNIT_ASSERT(example.mDestroyFunc == nullptr);
98 example.getCreate();
99 CPPUNIT_ASSERT(example.mDeclareFunc != nullptr);
100 CPPUNIT_ASSERT(example.mCreateFunc != nullptr);
101 CPPUNIT_ASSERT(example.mDestroyFunc == nullptr);
102 example.getDestroy();
103 CPPUNIT_ASSERT(example.mDeclareFunc != nullptr);
104 CPPUNIT_ASSERT(example.mCreateFunc != nullptr);
105 CPPUNIT_ASSERT(example.mDestroyFunc != nullptr);
106}
