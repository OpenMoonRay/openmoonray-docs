252 bool getProperty(Property property, float *dest) const override
253 {
254 bool handled = true;
255 switch (property)
256 {
257 case PROPERTY_COLOR:
258 {
259 *dest = mColor[0];
260 *(dest + 1) = mColor[1];
261 *(dest + 2) = mColor[2];
262 break;
263 }
264 case PROPERTY_PBR_VALIDITY:
265 {
266 // Same as Albedo PBR Validity
267 scene_rdl2::math::Color res = computeAlbedoPbrValidity(mColor);
268 *dest = res.r;
269 *(dest + 1) = res.g;
270 *(dest + 2) = res.b;
271 break;
272 }
273 case PROPERTY_ROUGHNESS:
274 {
275 *dest = mRoughness;
276 *(dest + 1) = mRoughness;
277 break;
278 }
279 case PROPERTY_NORMAL:
280 {
281 const scene_rdl2::math::Vec3f &N = mFrame.getN();
282 *dest = N.x;
283 *(dest + 1) = N.y;
284 *(dest + 2) = N.z;
285 }
286 break;
287 default:
288 handled = BsdfLobe::getProperty(property, dest);
289 break;
290 }
291 return handled;
292 }
