46#define _mm_mullo_epi32 __emu_mm_mullo_epi32
47__forceinline __m128i _mm_mullo_epi32( __m128i value, __m128i input ) {
48 __m128i rvalue;
49 char* _r = (char*)(&rvalue + 1);
50 char* _v = (char*)(& value + 1);
51 char* _i = (char*)(& input + 1);
52 for ( ssize_t i = -16 ; i != 0 ; i += 4 ) *((int32*)(_r + i)) = *((int32*)(_v + i))* *((int32*)(_i + i));
53 return rvalue;
54}

67#define _mm_extract_epi32 __emu_mm_extract_epi32
68__forceinline int _mm_extract_epi32( __m128i input, const int index ) {
69 switch ( index ) {
70 case 0: return _mm_cvtsi128_si32(input);
71 case 1: return _mm_cvtsi128_si32(_mm_shuffle_epi32(input, _MM_SHUFFLE(1, 1, 1, 1)));
72 case 2: return _mm_cvtsi128_si32(_mm_shuffle_epi32(input, _MM_SHUFFLE(2, 2, 2, 2)));
73 case 3: return _mm_cvtsi128_si32(_mm_shuffle_epi32(input, _MM_SHUFFLE(3, 3, 3, 3)));
74 default: assert(false); return 0;
75 }
76}

92#define _mm_round_ps __emu_mm_round_ps
93__forceinline __m128 _mm_round_ps( __m128 value, const int flags )
94{
95 switch ( flags )
96 {
97 case _MM_FROUND_TO_NEAREST_INT: return _mm_cvtepi32_ps(_mm_cvtps_epi32(value));
98 case _MM_FROUND_TO_NEG_INF : return _mm_cvtepi32_ps(_mm_cvtps_epi32(_mm_add_ps(value, _mm_set1_ps(-0.5f))));
99 case _MM_FROUND_TO_POS_INF : return _mm_cvtepi32_ps(_mm_cvtps_epi32(_mm_add_ps(value, _mm_set1_ps( 0.5f))));
100 case _MM_FROUND_TO_ZERO : return _mm_cvtepi32_ps(_mm_cvttps_epi32(value));
101 }
102 return value;
103}
