26void
27Camera::update(const Mat4d& world2render)
28{
29 float near = getNear();
30 float far = getFar();
31 if (near < 0.f || far <= near) {
32 mRdlCamera->error("Camera has invalid near/far values. ",
33 "near: " , near , " far: " , far);
34 }
35
36 // Recompute world and trace transforms
37 mCamera2World = computeC2W(0.0f);
38 bool isRotTrans;
39 mWorld2Camera = rtInverse(mCamera2World, isRotTrans);
40 if (!isRotTrans) {
41 mRdlCamera->warn("Camera has scale or shear, results may be non-deterministic\n");
42 }
43 mCamera2Render = toFloat(mCamera2World * world2render);
44 mRender2Camera = rtInverse(mCamera2Render);
45
46 // Compute camera space transforms between shutter close
47 // and shutter open
48 Mat4d cameraClose2World = computeC2W(1.0f);
49 const Mat4f cameraClose2CameraOpen = toFloat(cameraClose2World * getWorld2Camera());
50 const Mat4f cameraOpen2CameraClose = toFloat((cameraClose2World * getWorld2Camera()).inverse());
51 // rotations
52 mCameraClose2CameraOpenRot = Quaternion3f(asVec3(cameraClose2CameraOpen.vx),
53 asVec3(cameraClose2CameraOpen.vy),
54 asVec3(cameraClose2CameraOpen.vz));
55 // This quaternion could be computed as <r, -i, -j, -k> of
56 // the previous one. But for robustness reasons, compute it directly
57 // from the source matrix.
58 mCameraOpen2CameraCloseRot = Quaternion3f(asVec3(cameraOpen2CameraClose.vx),
59 asVec3(cameraOpen2CameraClose.vy),
60 asVec3(cameraOpen2CameraClose.vz));
61 // When slerping two generic quaternions, a and b, this check is "if (dot(a, b) < 0.f)".
62 // But since we are slerping these quaternions always and only with <1, 0, 0, 0>,
63 // checking against the .r component is sufficient. Both of these conditionals
64 // should always be either both true or both false, but it is conceivably possible
65 // that rounding errors could cause this to not be the case.
66 if (mCameraClose2CameraOpenRot.r < 0.f) mCameraClose2CameraOpenRot *= -1.0f;
67 if (mCameraOpen2CameraCloseRot.r < 0.f) mCameraOpen2CameraCloseRot *= -1.0f;
68 // translations
69 mCameraClose2CameraOpenTrans = asVec3(cameraClose2CameraOpen.vw);
70 mCameraOpen2CameraCloseTrans = asVec3(cameraOpen2CameraClose.vw);
71
72 const scene_rdl2::rdl2::SceneVariables& vars =
73 getRdlCamera()->getSceneClass().getSceneContext()->getSceneVariables();
74
75 scene_rdl2::math::HalfOpenViewport aperture = vars.getRezedApertureWindow();
76 scene_rdl2::math::HalfOpenViewport region = vars.getRezedRegionWindow();
77
78 mApertureWindowWidth = aperture.width();
79 mApertureWindowHeight = aperture.height();
80 mRegionWindowWidth = region.width();
81 mRegionWindowHeight = region.height();
82 mRegionToApertureOffsetX = region.mMinX - aperture.mMinX;
83 mRegionToApertureOffsetY = region.mMinY - aperture.mMinY;
84
85 updateImpl(world2render);
86}

88Mat4f
89Camera::computeCamera2Render(float time) const
90{
91 // "camera space at ray time" --> render space
92 if (getMotionBlur()) {
93 // slerp rotation, lerp translation to compute camera(time) to camera(0) matrix
94 const Quaternion3f r = slerp(Quaternion3f(scene_rdl2::math::one), mCameraClose2CameraOpenRot, time);
95 const Vec3f t = mCameraClose2CameraOpenTrans * time;
96 const Mat4f cameraTime2CameraOpen = Mat4f(r, Vec4f(t.x, t.y, t.z, 1.f));
97 // Then apply camera to render transform
98 return cameraTime2CameraOpen * getCamera2Render();
99 } else {
100 return getCamera2Render();
101 }
102}

104Mat4f
105Camera::computeRender2Camera(float time) const
106{
107 // render space -> "camera space at ray time"
108 if (getMotionBlur()) {
109 // slerp rotation, lerp translation to compute camera(0) to camera(time) matrix
110 const Quaternion3f r = slerp(Quaternion3f(scene_rdl2::math::one), mCameraOpen2CameraCloseRot, time);
111 const Vec3f t = mCameraOpen2CameraCloseTrans * time;
112 const Mat4f cameraOpen2CameraTime = Mat4f(r, Vec4f(t.x, t.y, t.z, 1.f));
113 // Then apply camera to render transform
114 return getRender2Camera() * cameraOpen2CameraTime;
115 } else {
116 return getRender2Camera();
117 }
118}

157float
158Camera::computeZDistanceImpl(const Vec3f &p, const Vec3f &o, float time) const
159{
160 // The default implemenation computes the euclidean distance between
161 // the ray hit point p, and the ray origin o. This should be a good
162 // default for all but projective cameras. It assumes that the camera
163 // space has no scales, which ensures that distance in render space is equal
164 // to distance in all camera spaces. So the time parameter is ignored.
165 return (p - o).length();
166}
