38class ActiveBitTable
39//
40// Keep multiple item's on/off information as array of block(bitmask=uint64_t) and
41// provides APIs to access by blockId
42//
43{
44public:
45 ActiveBitTable(unsigned totalItems) :
46 mTotalItems(totalItems),
47 mTable(((totalItems > 0)? ((totalItems - 1) / 64 + 1): 0), 0x0)
48 {}
49
50 static unsigned calcBlockTotal(const unsigned totalItems) { return (totalItems - 1) / 64 + 1; }
51
52 void reset() { memset(mTable.data(), 0x0, mTable.size() * sizeof(uint64_t)); }
53
54 finline void setOn(const unsigned itemId);
55 finline void setOff(const unsigned itemId);
56 finline bool get(const unsigned itemId) const;
57
58 unsigned getTotalBlock() const { return mTable.size(); }
59 uint64_t getBlock(const unsigned blockId) const { return mTable[blockId]; }
60 void setBlock(const unsigned blockId, uint64_t block) { mTable[blockId] = block; }
61
62 finline unsigned getActiveTotalBlock() const;
63
64 std::string show(const std::string &hd) const;
65
66private:
67 unsigned mTotalItems;
68 std::vector<uint64_t> mTable;
69
70 template <typename F>
71 void accessItem(const unsigned itemId, F accessFunc) const
72 {
73 unsigned blockId = itemId / 64; // uint64_t = 64bit
74 if (blockId < mTable.size()) {
75 unsigned shiftId = itemId % 64;
76 accessFunc(blockId, shiftId);
77 }
78 }
79
80 std::string showBlock(const std::string &hd, const unsigned blockId) const;
81};

83finline void
84ActiveBitTable::setOn(const unsigned itemId)
85{
86 accessItem(itemId,
87 [&](const unsigned blockId, const unsigned shiftId) {
88 mTable[blockId] |= ((uint64_t)0x1 << shiftId);
89 });
90}
91
92finline void
93ActiveBitTable::setOff(const unsigned itemId)
94{
95 accessItem(itemId,
96 [&](const unsigned blockId, const unsigned shiftId) {
97 mTable[blockId] &= ~((uint64_t)0x1 << shiftId);
98 });
99}

101finline bool
102ActiveBitTable::get(const unsigned itemId) const
103{
104 bool rt = true;
105 accessItem(itemId,
106 [&](const unsigned blockId, const unsigned shiftId) {
107 rt = (mTable[blockId] & ((uint64_t)0x1 << shiftId))? true: false;
108 });
109 return rt;
110}

112finline unsigned
113ActiveBitTable::getActiveTotalBlock() const
114{
115 unsigned total = 0;
116 for (size_t blockId = 0; blockId < mTable.size(); ++blockId) {
117 if (mTable[blockId]) total++;
118 }
119 return total;
120}

124class ActiveBitTables
125//
126// Create hierarchical multi-level active bit table(s) using mip-mapped style idea
127// in order to minimize encoded data size.
128//
129{
130public:
131 // We have multiple choices to encode data depend on the data pattern.
132 // This ActiveBitTables::DumpMode will be combined with RunLenBitTable::DumpMode
133 // and converted as PackActiveTiles's dumpMode (See PackActiveTiles::enqTileMaskBlock()
134 enum class DumpMode : unsigned char {
135 SKIP_DUMP = 0x0, // We don't need to dump because all tiles are active
136 FULL_DUMP = 0x1, // Dump all tiles by tileId (for debug use only)
137 FULL_DELTA_DUMP = 0x2, // Dump all tiles by delta tileId
138 TABLE_DUMP = 0x3, // Dump by hierachical tables
139 LEAF_TABLE_DUMP = 0x4 // Only dump leaf tables
140 };
141 static constexpr unsigned char DUMPMODE_MASK = 0x0f;
142
143 using VContainerDeq = rdl2::ValueContainerDeq;
144 using VContainerEnq = rdl2::ValueContainerEnq;
145
146 ActiveBitTables(unsigned totalItems) :
147 mTotalItems(totalItems),
148 mFullActiveTable(false),
149 mTables(calcTablesSize(totalItems)),
150 mDataSize(0)
151 {
152 // mTables memory setup based on totalItems
153 unsigned blockTotal = mTotalItems;
154 for (size_t id = 0; id < mTables.size(); ++id) {
155 mTables[id].reset(new ActiveBitTable(blockTotal));
156 blockTotal = mTables[id]->getTotalBlock();
157 }
158 }
159
160 void reset() { mTables[0]->reset(); }
161
162 void setOn(const unsigned itemId) { mTables[0]->setOn(itemId); }
163 void setOff(const unsigned itemId) { mTables[0]->setOff(itemId); }
164
165 // You need to call before use of enqTblDump().
166 // Update hierarchical multi-level table condition if needed internally and return
167 // proper DumpMode for encoding (= minimize data size).
168 DumpMode finalize();
169
170 unsigned getTotalBlock() const { return mTables[0]->getTotalBlock(); }
171 uint64_t getBlock(const unsigned blockId) const { return mTables[0]->getBlock(blockId); }

189 // Only using mTables[0] data. This template works without finalize() call.
190 template <typename F>
191 void crawlActiveTblItem(F activeItemFunc) const
192 {
193 for (unsigned blockId = 0; blockId < mTables[0]->getTotalBlock(); ++blockId) {
194 uint64_t currBlock = mTables[0]->getBlock(blockId);
195 for (unsigned shift = 0; currBlock && shift < 64; ++shift) {
196 if (currBlock & 0x1) {
197 unsigned itemId = blockId * 64 + shift;
198 activeItemFunc(itemId);
199 }
200 currBlock >>= 1;
201 }
202 }
203 }

220private:
221 using ActiveBitTableUqPtr = std::unique_ptr<ActiveBitTable>;
222
223 //
224 // Table dimension and max tiles (also pixel resolution)
225 //
226 // Dim Level MaxTiles MaxPixels (if square resolution)
227 // 1 0 64 4096 = 64*64
228 // 2 1 4096 262144 = 512*512
229 // 3 2 262144 16777216 = 4096*4096
230 // 4 3 16777216 1073741824 = 32768*32768
231 // 5 4 1073741824 68719476736 = 262144*262144
232 //
233 // Current max ActiveBitTable resolution is u_int (i.e. = 4294967296 - 1)
234 // and it's required level=5 ActiveBitTables.
235 //
236 // 6 5 4294967295 274877906880 = around 524287.99*524287.99
237 //
238 // Actually level=5 can support totally enough resolution of the image.
239 //
240 unsigned mTotalItems;
241 bool mFullActiveTable; // setup by finalizeTables()
242 std::vector<ActiveBitTableUqPtr> mTables;
243
244 size_t mDataSize;
245
246 // This template only works with hierarchical tables (i.e. non leaf only table).
247 // This means we need to call finalize() and return mode should be TABLE_DUMP.
248 // Otherwise internally mTables is not properly updated and this template can not work
249 // properly.
250 template <typename F>
251 void crawlActiveTblBlockUseNextLevel(const unsigned tblId, F activeBlockFunc) const
252 {
253 unsigned nTblId = tblId + 1; // upper level table id
254 MNRY_ASSERT(nTblId < mTables.size());
255
256 const unsigned totalBlock = mTables[nTblId]->getTotalBlock();
257 for (unsigned nBlockId = 0; nBlockId < totalBlock; ++nBlockId) {
258 uint64_t nTblMask = mTables[nTblId]->getBlock(nBlockId);
259 for (unsigned shift = 0; nTblMask && shift < 64; ++shift) {
260 if (nTblMask & 0x1) {
261 unsigned blockId = nBlockId * 64 + shift;
262 activeBlockFunc(blockId);
263 }
264 nTblMask >>= 1;
265 } // shift
266 } // blockId
267 }

281finline unsigned
282ActiveBitTables::calcTablesSize(const unsigned totalItems)
283//
284// Compute mTables size based on totalItems
285//
286{
287 unsigned tableSize = 1;
288 unsigned blockTotal = totalItems;
289 while (((blockTotal = ActiveBitTable::calcBlockTotal(blockTotal)) > 1)) {
290 tableSize++;
291 }
292 return tableSize;
293}

295finline bool
296ActiveBitTables::finalizeTables()
297//
298// setup all activeBitTable hierarchy information based on leaf activebitTable and return
299// all blocks are active (= true) or not (=false)
300//
301{
302 mFullActiveTable = (mTables[0]->getActiveTotalBlock() == mTables[0]->getTotalBlock());
303 if (!mFullActiveTable) {
304 for (size_t tblId = 1; tblId < mTables.size(); ++tblId) {
305 mTables[tblId]->reset();
306 for (size_t itemId = 0; itemId < mTables[tblId - 1]->getTotalBlock(); ++itemId) {
307 if (mTables[tblId - 1]->getBlock(itemId)) {
308 mTables[tblId]->setOn(itemId);
309 }
310 }
311 }
312 }
313 return mFullActiveTable;
314}

316finline size_t
317ActiveBitTables::calcSerializedTileAddrInfoSizeLeafTblDump() const
318//
319// return serialized data size when DumpMode is LEAF_TABLE_DUMP by byte
320//
321{
322 return mTables[0]->getTotalBlock() * sizeof(uint64_t);

325finline size_t
326ActiveBitTables::calcSerializedTileAddrInfoSizeAllTblDump() const
327//
328// return serialized data size when DumpMode is TABLE_DUMP by byte
329//
330{
331 size_t total = 0;
332 total += sizeof(uint64_t); // data size for mTables[mTables.size() - 1] : this is always 1 item
333
334 if (mTables.size() > 1) {
335 // use reverse order of tblId as same as serialized logic (i.e. = enqTblDump())
336 size_t tblId = mTables.size() - 1;
337 do {
338 tblId--;
339 total += (mTables[tblId]->getActiveTotalBlock() * sizeof(uint64_t));
340 } while (tblId > 0);
341 }
342
343 return total;
344}
