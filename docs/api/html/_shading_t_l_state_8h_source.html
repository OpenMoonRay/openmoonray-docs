35 // Hook up primitive attribute offset table for entire shade tree.
36 // Child shaders can access this directly via the TLS bypassing the extra
37 // indirections. Returns true if the attributes offsets pointer was updated.
38 bool getAttributeOffsetsFromRootShader(const scene_rdl2::rdl2::RootShader& rootShader)
39 {
40 // If there is already an attribute table, then we are a nested shader
41 // (e.g. a cutout). In this case, keep using the parents attribute table.
42 if (mAttributeOffsets == nullptr && rootShader.hasExtension()) {
43 const shading::AttributeTable *attrTable =
44 rootShader.get<shading::RootShader>().getPrimitiveAttributeTable();
45 if (attrTable) {
46 mAttributeOffsets = attrTable->getKeyOffsets();
47 return true;
48 }
49 }
50 return false;
51 }
52
53 void clearAttributeOffsets()
54 {
55 mAttributeOffsets = nullptr;
56 }

75/// Convenience function for iterating over all existing shading TLS instances.
76template <typename Body>
77finline void forEachTLS(Body const &body)
78{
79 unsigned numTLS = mcrt_common::getNumTBBThreads();
80 mcrt_common::ThreadLocalState *tlsList = mcrt_common::getTLSList();
81 for (unsigned i = 0; i < numTLS; ++i) {
82 auto shadingTls = tlsList[i].mShadingTls.get();
83 if (shadingTls) {
84 body(shadingTls);
85 }
86 }
87}
