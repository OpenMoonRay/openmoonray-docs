50class PosMapBuilder: public geom::PrimitiveVisitor
51{
52public:
53 PosMapBuilder(unsigned int udim,
54 unsigned int width,
55 unsigned int height,
56 const std::string &stKey,
57 scene_rdl2::math::Vec3fa *posResult,
58 scene_rdl2::math::Vec3f *nrmResult);
59
60 void visitPrimitive(geom::Primitive& p) override { MNRY_ASSERT(0, "visitPrimitive NYI"); }
61 void visitCurves(geom::Curves& c) override { MNRY_ASSERT(0, "visitCurves NYI"); }
62 void visitInstance(geom::Instance& i) override { MNRY_ASSERT(0, "visitInstance NYI"); }
63 void visitPoints(geom::Points& p) override { MNRY_ASSERT(0, "visitPoints NYI"); }
64 void visitPolygonMesh(geom::PolygonMesh &p) override;
65 void visitPrimitiveGroup(geom::PrimitiveGroup &pg) override;
66 void visitSphere(geom::Sphere& s) override { MNRY_ASSERT(0, "visitSphere NYI"); }
67 void visitBox(geom::Box& b) override { MNRY_ASSERT(0, "visitBox NYI"); }
68 void visitSubdivisionMesh(geom::SubdivisionMesh &s) override;
69 void visitTransformedPrimitive(geom::TransformedPrimitive& t) override;
70 void visitVdbVolume(geom::VdbVolume& v) override { MNRY_ASSERT(0, "visitVdbVolume NYI"); }

72private:

81PosMapBuilder::PosMapBuilder(unsigned int width,
82 unsigned int height,
83 unsigned int udim,
84 const std::string &stKey,
85 scene_rdl2::math::Vec3fa *posResult,
86 scene_rdl2::math::Vec3f *nrmResult):
87 mWidth(width),
88 mHeight(height),
89 mUdim(udim),
90 mStKey(stKey.empty()?
91 shading::StandardAttributes::sSurfaceST :
92 shading::TypedAttributeKey<scene_rdl2::math::Vec2f>(stKey)),
93 mPosResult(posResult),
94 mNrmResult(nrmResult)
95{
96 MNRY_ASSERT(posResult);
97 memset(posResult, 0, width * height * sizeof(scene_rdl2::math::Vec3fa));
98 if (nrmResult) {
99 memset(nrmResult, 0, width * height * sizeof(scene_rdl2::math::Vec3f));
100 }

241void
242BakeCamera::updateImpl(const scene_rdl2::math::Mat4d &world2render)
243{
244 const scene_rdl2::rdl2::Camera *rdlCamera = getRdlCamera();
245
246 // width and height of the map we are baking
247 // i think it is likely a setup error if we
248 // are using aperture or region windows at all when baking.
249 const int width = getApertureWindowWidth();
250 const int height = getApertureWindowHeight();
251 mWidth = width;
252 mHeight = height;
253
254 // has the user supplied a normal map?
255 // TODO: handle udim string subst.
256 const std::string filename = rdlCamera->get(mAttrNormalMap);
257 if (filename != mNormalMap.mFilename) {
258 // cleanup previously allocated texture handle
259 mNormalMap.mFilename = "";
260 mNormalMap.mTextureHandle = nullptr;
261 mNormalMap.mTextureSystem = nullptr;
262
263 if (!filename.empty() && needNormals()) {
264 // allocate new handle
265 texture::TextureSampler *textureSampler = texture::getTextureSampler();
266 OIIO::TextureSystem *textureSystem = textureSampler->getTextureSystem();
267 std::string errorString;
268 texture::TextureHandle *textureHandle =
269 textureSampler->getHandle(filename, errorString, textureSystem->get_perthread_info());
270 if (textureHandle) {
271 // all good
272 mNormalMap.mFilename = filename;
273 mNormalMap.mTextureHandle = textureHandle;
274 mNormalMap.mTextureSystem = textureSystem;
275 } else {
276 rdlCamera->error("failed to open normal map file ", filename, " ", errorString);
277 }
278 }
279 }
280
281 // allocate position and optional normal map
282 int posMapWidth = static_cast<int>(floor(width * rdlCamera->get(mAttrMapFactor)));
283 int posMapHeight = static_cast<int>(floor(height * rdlCamera->get(mAttrMapFactor)));
284
285 if (posMapWidth != mPosMapWidth || posMapHeight != mPosMapHeight) {
286 if (mPosMap) {
287 scene_rdl2::util::alignedFreeArray(mPosMap);
288 mPosMap = nullptr;
289 }
290 if (mNrmMap) {
291 scene_rdl2::util::alignedFreeArray(mNrmMap);
292 mNrmMap = nullptr;
293 }
294 // we unconditionally have a pos map
295 mPosMap = scene_rdl2::util::alignedMallocArray<scene_rdl2::math::Vec3fa>(posMapWidth * posMapHeight * 4);
296 if (needNrmMap()) {
297 mNrmMap = scene_rdl2::util::alignedMallocArray<scene_rdl2::math::Vec3f>(posMapWidth * posMapHeight * 3);
298 }
299
300 mPosMapWidth = posMapWidth;
301 mPosMapHeight = posMapHeight;
302 } else {
303 // if our mode has changed, we might need to delete or allocate a norm map
304 if (needNrmMap() && !mNrmMap) {
305 mNrmMap = scene_rdl2::util::alignedMallocArray<scene_rdl2::math::Vec3f>(mPosMapWidth * mPosMapHeight * 3);
306 } else if (!needNrmMap() && mNrmMap) {
307 scene_rdl2::util::alignedFreeArray(mNrmMap);
308 mNrmMap = nullptr;
309 }
310 }
311
312 // Get the geometry of the mesh (not sure what to do if no such geometry is present):
313 scene_rdl2::rdl2::SceneObject *so = rdlCamera->get(mAttrGeometry);
314 if (so && so->isA<scene_rdl2::rdl2::Geometry>()) {
315 auto rdlGeom = so->asA<scene_rdl2::rdl2::Geometry>();
316 MNRY_ASSERT(rdlGeom != nullptr);
317
318 // the decomposition for the matrix we want initially:
319 const scene_rdl2::math::Mat4d geomToWorldOpen = rdlGeom->get(scene_rdl2::rdl2::Node::sNodeXformKey, 0.0);
320 const scene_rdl2::math::Mat4d geomToWorldClose = rdlGeom->get(scene_rdl2::rdl2::Node::sNodeXformKey, 1.0);
321
322 scene_rdl2::math::decompose(scene_rdl2::math::xform<scene_rdl2::math::XformT<scene_rdl2::math::Mat3d>>(geomToWorldOpen),
323 mGeomToWorldOpen);
324 scene_rdl2::math::decompose(scene_rdl2::math::xform<scene_rdl2::math::XformT<scene_rdl2::math::Mat3d>>(geomToWorldClose),
325 mGeomToWorldClose);
326 if (scene_rdl2::math::dot(mGeomToWorldOpen.r, mGeomToWorldClose.r) < 0.0) { mGeomToWorldClose.r *= -1.0; }
327
328 // the matrix to go from render space to geometry space
329 const scene_rdl2::math::Mat4d worldToGeom = geomToWorldOpen.inverse();
330 const scene_rdl2::math::Mat4d renderToWorld = scene_rdl2::math::toDouble(getRender2Camera()) * getCamera2World();
331 mRenderToGeom = renderToWorld * worldToGeom;
332
333 // to go from world space to render space (used later when transforming the outgoing rays)
334 mWorldToRender = getWorld2Camera() * scene_rdl2::math::toDouble(getCamera2Render());
335 }

338bool
339BakeCamera::computeDpdu(const scene_rdl2::math::Vec3f &P, const scene_rdl2::math::Vec2f &uv, scene_rdl2::math::Vec3f &dpdu) const
340{
341 // it might be worth baking this computation into its own map,
342 // both for correctness and efficiency reasons
343 // this isn't really dpdu, but rather a normalized vector in the
344 // dpdu direction.
345
346 const scene_rdl2::math::Vec2f du = scene_rdl2::math::Vec2f(1.f / (mPosMapWidth - 1.f), 0.f);
347 // data to the right?
348
349 const scene_rdl2::math::Vec2f rgtUv = uv + du;
350 bool rgtHasData = false;
351 scene_rdl2::math::Vec3f rgtP;
352 if (rgtUv[0] <= 1.f) {
353 scene_rdl2::math::Vec3f newN; // unused
354 rgtHasData = interpolatePosMap(rgtUv, rgtP, newN);
355 }
356
357 // ok, try to the left
358 const scene_rdl2::math::Vec2f lftUv = uv - du;
359 bool lftHasData = false;
360 scene_rdl2::math::Vec3f lftP;
361 if (lftUv[0] >= 0.f) {
362 scene_rdl2::math::Vec3f newN; // unused
363 lftHasData = interpolatePosMap(lftUv, lftP, newN);
364 }
365
366 if (rgtHasData && lftHasData) {
367 const scene_rdl2::math::Vec3f dp = rgtP - lftP;
368 if (!scene_rdl2::math::isZero(dp.lengthSqr())) {
369 dpdu = normalize(dp);
370 return true;
371 }
372
373 }
374
375 // either one side doesn't have data, or the delta
376 // between left and right is 0.
377 if (rgtHasData) {
378 const scene_rdl2::math::Vec3f dp = rgtP - P;
379 if (!scene_rdl2::math::isZero(dp.lengthSqr())) {
380 dpdu = normalize(dp);
381 return true;
382 }
383 }
384
385 // getting desperate...
386 if (lftHasData) {
387 const scene_rdl2::math::Vec3f dp = P - lftP;
388 if (!scene_rdl2::math::isZero(dp.lengthSqr())) {
389 dpdu = normalize(dp);
390 return true;
391 }
392 }
393
394 // we failed.... no data
395 return false;
396}

398// return true if uv has a valid position in the uv unwrap map.
399// if true, result will contain an interpolated result from the map.
400bool
401BakeCamera::interpolatePosMap(const scene_rdl2::math::Vec2f &uv, scene_rdl2::math::Vec3f &posResult, scene_rdl2::math::Vec3f &nrmResult) const
402{
403 const float xf = scene_rdl2::math::clamp(uv[0]) * (mPosMapWidth - 1);
404 const float yf = scene_rdl2::math::clamp(uv[1]) * (mPosMapHeight - 1);
405 scene_rdl2::math::BBox2i bbox(scene_rdl2::math::Vec2i(floor(xf), floor(yf)),
406 scene_rdl2::math::Vec2i(ceil(xf), ceil(yf)));
407
408 // bi-linear interpolation.
409 // bbox.upper
410 // +---------+
411 // | |
412 // | xf,yf |
413 // | |
414 // +---------+
415 // bbox.lower
416
417 const scene_rdl2::math::Vec3fa *posLowerLeft = mPosMap + (bbox.lower.y * mPosMapWidth + bbox.lower.x);
418 const scene_rdl2::math::Vec3fa *posUpperLeft = mPosMap + (bbox.upper.y * mPosMapWidth + bbox.lower.x);
419 const scene_rdl2::math::Vec3fa *posLowerRight = mPosMap + (bbox.lower.y * mPosMapWidth + bbox.upper.x);
420 const scene_rdl2::math::Vec3fa *posUpperRight = mPosMap + (bbox.upper.y * mPosMapWidth + bbox.upper.x);
421
422 const scene_rdl2::math::Vec3f *nrmLowerLeft = mNrmMap? mNrmMap + (bbox.lower.y * mPosMapWidth + bbox.lower.x) : nullptr;
423 const scene_rdl2::math::Vec3f *nrmUpperLeft = mNrmMap? mNrmMap + (bbox.upper.y * mPosMapWidth + bbox.lower.x) : nullptr;
424 const scene_rdl2::math::Vec3f *nrmLowerRight = mNrmMap? mNrmMap + (bbox.lower.y * mPosMapWidth + bbox.upper.x) : nullptr;
425 const scene_rdl2::math::Vec3f *nrmUpperRight = mNrmMap? mNrmMap + (bbox.upper.y * mPosMapWidth + bbox.upper.x) : nullptr;
426
427 // left side average
428 scene_rdl2::math::Vec3f posLeftSideAvg(0.f, 0.f, 0.f);
429 scene_rdl2::math::Vec3f nrmLeftSideAvg(0.f, 0.f, 0.f);
430 bool hasLeftSide(false);
431 if (posLowerLeft->w != 0.f) {
432 if (posUpperLeft->w != 0.f) {
433 // both have data, interpolate the two
434 const scene_rdl2::math::Vec3f a(posLowerLeft->asVec3f());
435 const scene_rdl2::math::Vec3f b(posUpperLeft->asVec3f());
436 posLeftSideAvg = scene_rdl2::math::lerp(a, b, yf - bbox.lower.y);
437 if (mNrmMap) {
438 const scene_rdl2::math::Vec3f a(*nrmLowerLeft);
439 const scene_rdl2::math::Vec3f b(*nrmUpperLeft);
440 nrmLeftSideAvg = scene_rdl2::math::lerp(a, b, yf - bbox.lower.y);
441 }
442 hasLeftSide = true;
443 } else {
444 // only the lower left has data, if we are closer
445 // to the lower side than the upper side, use the data,
446 // otherwise, use no data for left side
447 if (yf - bbox.lower.y < .5f) {
448 posLeftSideAvg = posLowerLeft->asVec3f();
449 if (mNrmMap) {
450 nrmLeftSideAvg = *nrmLowerLeft;
451 }
452 hasLeftSide = true;
453 }
454 }
455 } else if (posUpperLeft->w != 0.f) {
456 // only the upper left has data, if we are closer
457 // to the upper side than the lower side, use the data,
458 // otherwise, use no data for the left side
459 if (yf - bbox.lower.y >= .5f) {
460 posLeftSideAvg = posUpperLeft->asVec3f();
461 if (mNrmMap) {
462 nrmLeftSideAvg = *nrmUpperLeft;
463 }
464 hasLeftSide = true;
465 }
466 }
467
468 // right side average
469 scene_rdl2::math::Vec3f posRightSideAvg(0.f, 0.f, 0.f);
470 scene_rdl2::math::Vec3f nrmRightSideAvg(0.f, 0.f, 0.f);
471 bool hasRightSide(false);
472 if (posLowerRight->w != 0.f) {
473 if (posUpperRight->w != 0.f) {
474 // both have data, interpolate the two
475 const scene_rdl2::math::Vec3f a(posLowerRight->asVec3f());
476 const scene_rdl2::math::Vec3f b(posUpperRight->asVec3f());
477 posRightSideAvg = scene_rdl2::math::lerp(a, b, yf - bbox.lower.y);
478 if (mNrmMap) {
479 const scene_rdl2::math::Vec3f a(*nrmLowerRight);
480 const scene_rdl2::math::Vec3f b(*nrmUpperRight);
481 nrmRightSideAvg = scene_rdl2::math::lerp(a, b, yf - bbox.lower.y);
482 }
483 hasRightSide = true;
484 } else {
485 // only the lower right has data, if we are closer
486 // to the lower side than the upper side, use the data,
487 // otherwise, use no data for right side
488 if (yf - bbox.lower.y < .5f) {
489 posRightSideAvg = posLowerRight->asVec3f();
490 if (mNrmMap) {
491 nrmRightSideAvg = *nrmLowerRight;
492 }
493 hasRightSide = true;
494 }
495 }
496 } else if (posUpperRight->w != 0.f) {
497 // only the upper right has data, if we are closer
498 // to the upper side than the lower side, use the data,
499 // otherwise, use no data for the right side
500 if (yf - bbox.lower.y >= .5f) {
501 posRightSideAvg = posUpperRight->asVec3f();
502 if (mNrmMap) {
503 nrmRightSideAvg = *nrmUpperRight;
504 }
505 hasRightSide = true;
506 }
507 }
508
509 // now average the left and right side values
510 bool hasData = false;
511 if (hasLeftSide) {
512 if (hasRightSide) {
513 // both have data, interpolate the two
514 posResult = scene_rdl2::math::lerp(posLeftSideAvg, posRightSideAvg, xf - bbox.lower.x);
515 if (mNrmMap) {
516 nrmResult = scene_rdl2::math::lerp(nrmLeftSideAvg, nrmRightSideAvg, xf - bbox.lower.x);
517 }
518 hasData = true;
519 } else {
520 // only the left side has data. if we are closer
521 // to the left side than the right, use the data,
522 // otherwise, use no data
523 if (xf - bbox.lower.x < .5f) {
524 posResult = posLeftSideAvg;
525 if (mNrmMap) {
526 nrmResult = nrmLeftSideAvg;
527 }
528 hasData = true;
529 }
530 }
531 } else if (hasRightSide) {
532 // only the right side has data. if we are closer
533 // to the right side than the left, use the data,
534 // othwerwise, use no data
535 if (xf - bbox.lower.x >= .5f) {
536 posResult = posRightSideAvg;
537 if (mNrmMap) {
538 nrmResult = nrmRightSideAvg;
539 }
540 hasData = true;
541 }
542 }
543
544 // we need to re-normalize our interpolated normal
545 if (hasData && mNrmMap) {
546 nrmResult = nrmResult.normalize();
547 }
548
549 return hasData;
550}

552void
553BakeCamera::createRayImpl(mcrt_common::RayDifferential *dstRay,
554 float x,
555 float y,
556 float time,
557 float lensU,
558 float lensV) const
559{
560 // raster location
561 const scene_rdl2::math::Vec3f Pr(x, y, -1.0f);
562
563 // uv location
564 scene_rdl2::math::Vec2f uv;
565 uv[0] = Pr.x / mWidth;
566 uv[1] = Pr.y / mHeight;
567
568 const scene_rdl2::math::Vec2f Puv = scene_rdl2::math::Vec2f(uv[0], uv[1]);
569
570 // lookup P and optionally N in render space
571 scene_rdl2::math::Vec3f P = scene_rdl2::math::Vec3f(0, 0, 0);
572 scene_rdl2::math::Vec3f N = scene_rdl2::math::Vec3f(0, 0, 0);
573 bool hasData = interpolatePosMap(Puv, P, N);
574
575 // TODO: extend camera API to more explicitly handle an invalid ray case
576 if (!hasData) {
577 *dstRay = mcrt_common::RayDifferential(scene_rdl2::math::Vec3f(0, 0, 0), scene_rdl2::math::Vec3f(0, 0, 0),
578 scene_rdl2::math::Vec3f(0, 0, 0), scene_rdl2::math::Vec3f(0, 0, 0),
579 scene_rdl2::math::Vec3f(0, 0, 0), scene_rdl2::math::Vec3f(0, 0, 0),
580 /* near = */ scene_rdl2::math::sMaxValue,
581 /* far = */ scene_rdl2::math::sMaxValue,
582 /* time = */ 0.f, /* depth = */ 0);
583 return;
584 }
585
586 // Adjacent UVs along X and Y
587 float uvX = scene_rdl2::math::min(uv[0] + 1.0f / mWidth, 1.0f);
588 float uvY = scene_rdl2::math::min(uv[1] + 1.0f / mHeight, 1.0f);
589
590 // Adjacent Points
591 const scene_rdl2::math::Vec2f PuvX = scene_rdl2::math::Vec2f(uvX, uv[1]);
592 const scene_rdl2::math::Vec2f PuvY = scene_rdl2::math::Vec2f(uv[0], uvY);
593 scene_rdl2::math::Vec3f Px, Py, Nx, Ny;
594 bool hasDataX = interpolatePosMap(PuvX, Px, Nx);
595 if (!hasDataX) {
596 Px = P; Nx = N;
597 }
598 bool hasDataY = interpolatePosMap(PuvY, Py, Ny);
599 if (!hasDataY) {
600 Py = P; Ny = N;
601 }
602
603 // sample our normal map if we have one
604 if (needNormals() && haveNormalMap()) {
605 scene_rdl2::math::Vec3f nMap; // sampled value from map
606 OIIO::TextureOpt textureOpt;
607 textureOpt.apply_gamma = false;
608 textureOpt.swrap = OIIO::TextureOpt::Wrap::WrapClamp;
609 textureOpt.twrap = OIIO::TextureOpt::Wrap::WrapClamp;
610 textureOpt.interpmode = OIIO::TextureOpt::InterpMode::InterpBilinear;
611 bool res = mNormalMap.mTextureSystem->texture(mNormalMap.mTextureHandle,
612 mNormalMap.mTextureSystem->get_perthread_info(),
613 textureOpt,
614 /* s = */ Puv.x, /* t = */ 1.f - Puv.y,
615 /* dsdx = */ 0.f, /* dsdy = */ 0.f,
616 /* dtdx = */ 0.f, /* dtdy = */ 0.f,
617 /* nchans = */ 3, &nMap[0]);
618
619 // not sure what to do if this fails
620 MNRY_ASSERT(res);
621 if (!res) {
622 nMap = N;
623 }
624
625 // if our normal map is in render space (likely created
626 // via a normal aov) - then just use the normal map value directly
627 if (getRdlCamera()->get(mAttrNormalMapSpace) == 0) {
628 // render space
629 N = nMap;
630 } else {
631 // the normal map is in tangent space (see
632 // evalNormal in shading/EvalAttribute.h for how these
633 // normals are expected to be encoded). Note that the frame
634 // input is the state normal (which is stored in our Nrm map
635 // and dpds (which we can compute from our posMap).
636 scene_rdl2::math::Vec3f dpdu;
637 bool hasData = computeDpdu(P, Puv, dpdu);
638 if (hasData) {
639 MNRY_ASSERT(scene_rdl2::math::isNormalized(N));
640 MNRY_ASSERT(scene_rdl2::math::isNormalized(dpdu));
641 const scene_rdl2::math::ReferenceFrame frame(N, dpdu);
642 // recenter Nmap [0, 1] -> [-1, 1]
643 nMap = 2.0f * nMap - scene_rdl2::math::Vec3f(1.f, 1.f, 1.f);
644 nMap = frame.localToGlobal(nMap);
645 N = normalize(nMap);
646 } else {
647 // not a lot of good options in this case, we'll construct
648 // a reference frame around N and hope for the best
649 const scene_rdl2::math::ReferenceFrame frame(N);
650 // recenter Nmap [0, 1] -> [-1, 1]
651 nMap = 2.0f * nMap - scene_rdl2::math::Vec3f(1.f, 1.f, 1.f);
652 nMap = frame.localToGlobal(nMap);
653 N = normalize(nMap);
654 }
655 }
656 }
657 // Transform the normal and the point on the geometry if the geometry is moving:
658 xformPoint(time, P, N);
659 xformPoint(time, Px, Nx);
660 xformPoint(time, Py, Ny);
661
662 // ray direction and ray origin depend on mode
663 scene_rdl2::math::Vec3f rayOrg, rayDir;
664 scene_rdl2::math::Vec3f rayOrgX, rayDirX;
665 scene_rdl2::math::Vec3f rayOrgY, rayDirY;
666 float near = getNear();
667 float far = getFar();
668 float bias = getRdlCamera()->get(mAttrBias);
669 if (getRdlCamera()->get(mAttrUseRelativeBias)) {
670 // scale the bias so that the magnitude of bias compared to 1.0
671 // remains constant to the magnitude of the largest component of P
672 // this is similar to how we implement relative floating point comparisons
673 bias *= scene_rdl2::math::max(scene_rdl2::math::abs(P.x), scene_rdl2::math::abs(P.y),
674 scene_rdl2::math::abs(P.z), 1.f);
675 }
676
677 // Camera position in camera space:
678 scene_rdl2::math::Vec3f cameraPos = scene_rdl2::math::Vec3f(0, 0, 0);
679
680 switch (getRdlCamera()->get(mAttrMode)) {
681 case MODE_FROM_CAMERA_TO_SURFACE:
682 // Transform the camera as appropriate:
683 cameraPos = scene_rdl2::math::transformPoint(computeCamera2Render(time), cameraPos);
684
685 fromCameraToSurface(bias, cameraPos, P, N,
686 rayOrg, rayDir);
687 fromCameraToSurface(bias, cameraPos, Px, Nx,
688 rayOrgX, rayDirX);
689 fromCameraToSurface(bias, cameraPos, Py, Ny,
690 rayOrgY, rayDirY);
691
692 // if the bias is positive,
693 // reset the near plane so that it is between
694 // the bias location and P
695 // *-----|-----P----------|
696 // bias near far
697 if (bias > 0) {
698 near = bias / 2.f;
699 }
700 break;
701
702 case MODE_FROM_SURFACE_ALONG_NORMAL:
703 fromSurfaceAlongNormal(bias, P, N,
704 rayOrg, rayDir);
705 fromSurfaceAlongNormal(bias, Px, Nx,
706 rayOrgX, rayDirX);
707 fromSurfaceAlongNormal(bias, Py, Ny,
708 rayOrgY, rayDirY);
709 break;
710
711 case MODE_FROM_SURFACE_ALONG_REFLECTION_VECTOR:
712 // Transform the camera as appropriate:
713 cameraPos = scene_rdl2::math::transformPoint(computeCamera2Render(time), cameraPos);
714 fromSurfaceAlongReflectionVector(bias, cameraPos,
715 P, N,
716 rayOrg, rayDir);
717 fromSurfaceAlongReflectionVector(bias, cameraPos,
718 Px, Nx,
719 rayOrgX, rayDirX);
720 fromSurfaceAlongReflectionVector(bias, cameraPos,
721 Py, Ny,
722 rayOrgY, rayDirY);
723 break;
724
725 case MODE_ABOVE_SURFACE_REVERSE_NORMAL:
726
727 fromAboveSurfaceNormal(bias, P, N,
728 rayOrg, rayDir);
729 fromAboveSurfaceNormal(bias, Px, Nx,
730 rayOrgX, rayDirX);
731 fromAboveSurfaceNormal(bias, Py, Ny,
732 rayOrgY, rayDirY);
733
734 // if the bias is positive,
735 // reset the near plane so that it is between
736 // the bias location and P
737 // *-----|-----P----------|
738 // bias near far
739 if (bias > 0) {
740 near = bias / 2.f;
741 }
742
743 break;
744
745 default:
746 MNRY_ASSERT(0, "unknown bake camera mode\n");
747 }
748
749 // fill out our destination ray
750 // the RayDifferential now includes a time value to help with object motion blurr.
751 *dstRay = mcrt_common::RayDifferential(rayOrg, rayDir,
752 rayOrgX, rayDirX,
753 rayOrgY, rayDirY,
754 near, far,
755 time, /* depth = */ 0);
756
757}

759// Transform the normal and the point on the geometry if the geometry is moving:
760void
761BakeCamera::xformPoint(float time,
762 scene_rdl2::math::Vec3f& P,
763 scene_rdl2::math::Vec3f& N) const
764{
765 // first we transform them to geometry space:
766 const scene_rdl2::math::Vec3d pGeom = scene_rdl2::math::transformPoint(mRenderToGeom, P);
767 const scene_rdl2::math::Vec3d nGeom = scene_rdl2::math::transformVector(mRenderToGeom, N);
768
769 // the transformation to world space given the time provided
770 const scene_rdl2::math::Mat4d geomToWorld = scene_rdl2::math::Mat4d(slerp(mGeomToWorldOpen, mGeomToWorldClose, time).combined());
771 const scene_rdl2::math::Mat4d geomToRender = geomToWorld * mWorldToRender;
772
773 // transform the point and normal from geometry space to render space:
774 P = scene_rdl2::math::transformPoint(geomToRender, pGeom);
775 N = scene_rdl2::math::transformVector(geomToRender, nGeom);
776}

778// Create Rays from surface along reflection vector
779void
780BakeCamera::fromSurfaceAlongReflectionVector(float bias,
781 const scene_rdl2::math::Vec3f& cameraPos,
782 const scene_rdl2::math::Vec3f& P,
783 const scene_rdl2::math::Vec3f& N,
784 scene_rdl2::math::Vec3f& rayOrg,
785 scene_rdl2::math::Vec3f& rayDir) const
786{

788 // note that we incorrectly bake back-facing polygons,
789 // that should be harmless, but we could improve efficiency
790 // by returning invalid rays in that case.
791 shading::computeReflectionDirection(N, /* wo = */ normalize(cameraPos - P), rayDir);
792 MNRY_ASSERT(isNormalized(rayDir));
793
794 // offset ray position just barely above the surface to avoid
795 // self intersection.
796 rayOrg = P + bias * rayDir;
797}

800void
801BakeCamera::fromCameraToSurface(float bias,
802 const scene_rdl2::math::Vec3f& cameraPos,
803 const scene_rdl2::math::Vec3f& P,

813 // intersect our ray

816}
