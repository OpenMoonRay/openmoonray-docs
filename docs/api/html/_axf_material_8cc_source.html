87// AxfMaterial has its own normal map evaluation since
88// AXF normal map textures are stored in [-1,1].
89scene_rdl2::math::Vec3f
90AxfMaterial::evalNormal(const AxfMaterial * const obj,
91 const scene_rdl2::rdl2::AttributeKey<scene_rdl2::rdl2::Vec3f>& valueKey,
92 const float dial,
93 const int normalSpace,
94 moonray::shading::TLState *tls,
95 const moonray::shading::State& state)
96{
97 if (!obj->mIspc.mUseNormal) {
98 // no normal map bound, use state normal
99 return state.getN();
100 }
101
102 // sample the normal map
103 const scene_rdl2::rdl2::SceneObject* mapObject = obj->getBinding(valueKey);
104 if (!mapObject) {
105 // no map binding
106 return state.getN();
107 }
108
109 const scene_rdl2::rdl2::Map* map = mapObject->asA<scene_rdl2::rdl2::Map>();
110 scene_rdl2::math::Color sampledColor;
111 sample(obj, map, tls, state, &sampledColor);
112
113 // Check for numerical issues
114 if (scene_rdl2::math::isBlack(sampledColor)) {
115 return state.getN();
116 }
117
118 scene_rdl2::math::Vec3f result;
119
120 if (normalSpace == 0) { // tangent space, transform from tangent space to render space
121 const scene_rdl2::math::ReferenceFrame frame(state.getN(), normalize(state.getdPds()));
122 result = frame.localToGlobal(scene_rdl2::math::Vec3f(sampledColor.r, sampledColor.g, sampledColor.b));
123 } else { // render space, no transformation needed
124 result = scene_rdl2::math::Vec3f(sampledColor.r, sampledColor.g, sampledColor.b);
125 }
126
127 result = lerp(state.getN(), result, dial);
128
129 // hack the result if it is not physically plausible
130 return normalize(state.adaptNormal(result));
131}

133void
134AxfMaterial::shade(const scene_rdl2::rdl2::Material* self, moonray::shading::TLState *tls,
135 const State& state, BsdfBuilder& bsdfBuilder)
136{
137 const AxfMaterial* me = static_cast<const AxfMaterial*>(self);
138 scene_rdl2::alloc::Arena *arena = moonray::shading::getArena(tls);
139 moonray::shading::Bsdf *bsdf = const_cast<moonray::shading::Bsdf*>(bsdfBuilder.getBsdf());
140
141 const int normalSpace = me->get(attrNormalSpace);
142 const Vec3f N = AxfMaterial::evalNormal(me,
143 attrNormal,
144 scene_rdl2::math::saturate(me->get(attrInputNormalDial)),
145 normalSpace,
146 tls,
147 state);
148 Color specularColor(0.0f);
149 Vec3f anisoDir(0.0f);
150
151 if (!state.isCausticPath() || me->get(attrCastsCaustics)) {
152 specularColor = evalColor(me, attrSpecularColor, tls, state);
153
154 if (!isBlack(specularColor)) {
155 // compute roughness
156 Vec2f roughness = evalVec2f(me, attrSpecularLobe, tls, state);
157 const Vec2f minRoughnessAniso = state.getMinRoughness();
158 const float minRoughness = min(minRoughnessAniso.x, minRoughnessAniso.y);
159 roughness.x = max(roughness.x, minRoughness);
160 roughness.y = max(roughness.y, minRoughness);
161
162 // Use Fresnel?
163 moonray::shading::Fresnel * fresnel = nullptr;
164 if (me->mIspc.mUseFresnel) {
165 const float F0 = scene_rdl2::math::saturate(evalFloat(me, attrFresnel, tls, state));
166 fresnel = arena->allocWithArgs<moonray::shading::SchlickFresnel>(Color(F0), 1.0f);
167 }
168
169 // create specular lobe
170 moonray::shading::BsdfLobe *lobe;
171 if (isEqual(roughness, Vec2f(zero))) {
172 // create mirror reflection lobe
173 lobe = arena->allocWithArgs<moonray::shading::MirrorReflectionBsdfLobe>(N);
174 } else {
175 // create aniso ward duer reflection lobe
176
177 // compute aniso dir in local frame
178 const float anisoRotation = evalFloat(me, attrAnisoRotation, tls, state);
179 float sinAR, cosAR;
180 sincos(anisoRotation, &sinAR, &cosAR);
181 const Vec2f u(1.0f, 0.0f);
182 const Vec2f localDir(u.x * cosAR - u.y * sinAR,
183 u.y * cosAR + u.x * sinAR);
184
185 // compute aniso dir in global frame
186 const Vec3f T = normalize(state.getdPds());
187 const scene_rdl2::math::ReferenceFrame frame(N, T);
188 anisoDir = frame.localToGlobal(
189 normalize(Vec3f(localDir.x, localDir.y, 0.0f)));
190
191 lobe = arena->allocWithArgs<moonray::shading::WardDuerBsdfLobe>(N, anisoDir,
192 roughness.x, roughness.y, false);
193 }
194
195 lobe->setFresnel(fresnel);
196 lobe->setScale(specularColor);
197 bsdf->addLobe(lobe);
198 }
199 }
200
201 // Diffuse
202 const Color diffuseColor = evalColor(me, attrDiffuseColor, tls, state);
203 if (!isBlack(diffuseColor)) {
204 auto lobe = arena->allocWithArgs<moonray::shading::LambertBsdfLobe>(N, diffuseColor, true);
205 bsdf->addLobe(lobe);
206 }
207}
