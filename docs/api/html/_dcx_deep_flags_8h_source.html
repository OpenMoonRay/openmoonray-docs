66// class DeepFlags

68//====================
69//--------------------------------------------------------------------------------
70//
71// Per-deep sample metadata flag-bits class.
72//
73// Stores surface-type flags and the partial subpixel-coverage bin count which
74// indicates the subpixel-coverage weight applied to the deep channel values.
75//
76// .. note::
77// Important backwards compatibility note!
78// Flags now require 32-bit storage. In 2.2.1 the 4 defined flags would fit
79// easily inside the mantissa of a 16-bit half float, but with the addition
80// of the partial subpixel-coverage weight it's prudent to switch to 32-bit
81// floats with 23 bits worth of mantissa space. This way additional bits can
82// be allocated later on without changing the channel format (again.)
83//
84//--------------------------------------------------------------------------------
85

99 static GNU_CONST_DECL uint32_t LINEAR_INTERP = 0x00001; // Linear surface sample interpolation (not volumetric)
100 static GNU_CONST_DECL uint32_t MATTE_OBJECT = 0x00002; // Matte sample that cuts-out (blackens) other samples
101 static GNU_CONST_DECL uint32_t ADDITIVE = 0x00004; // Additive sample which plusses with adjacent additive samples

103 static GNU_CONST_DECL uint32_t FLAG5 = 0x00010; // Placeholder

107 //
108 static GNU_CONST_DECL uint32_t SURFACE_FLAG_BITS = 0x000ff;
109
110
111 //-----------------------------------------------------------------------------------------
112 //!rst:left-align::
113 //
114 // **Partial Subpixel-Coverage Weight**
115 // ``bits 0x0100..0xff00``
116 //
117 // If any of the 8 partial spcoverage bits are enabled the deep sample values include
118 // partial-coverage weighting equal to the binary value of the coverage bits. These bits
119 // work in conjunction with the ADDITIVE flag.
120 //
121 // Partial-coverage count to partial-coverage weight conversion is biased by 1 so that
122 // 0xff00=0.996, 0x8000=0.5, 0x4000=0.25, 0x0000=0.0, etc.
123 // i.e::
124 //
125 // weight = float((bits & PARTIAL_SPCOVERAGE_BITS) >> 8) / maxSpCoverageScale or
126 // weight = float(partialSpCoverageBits()) / 65536.0f or just use
127 // weight = DeepFlags::getSpCoverageWeight()
128 //
129 // Maximum coverage count is 256 (0x10000) indicating complete subpixel coverage and is
130 // logically the same as all bits off. Since both 0x00000 and 0x10000 effectively mean the
131 // same thing we don't require the 9th bit to store the actual value 256. The methods to
132 // retrieve the bin count check for count==0 and ADDITIVE=1 and return maxSpCoverageCount instead.
133 //
134 // So, when accumulating partial-coverage weight counts and maxSpCoverageCount is reached or
135 // exceeded, *clear* the coverage value to 0x00000 and clear the ADDITIVE flags to indicate
136 // full-coverage.
137 //-----------------------------------------------------------------------------------------
138 static GNU_CONST_DECL uint32_t PARTIAL_SPCOVERAGE0 = 0x00100; // Partial sp-coverage weight bit 0
139 static GNU_CONST_DECL uint32_t PARTIAL_SPCOVERAGE1 = 0x00200; // Partial sp-coverage weight bit 1
140 static GNU_CONST_DECL uint32_t PARTIAL_SPCOVERAGE2 = 0x00400; // Partial sp-coverage weight bit 2
141 static GNU_CONST_DECL uint32_t PARTIAL_SPCOVERAGE3 = 0x00800; // Partial sp-coverage weight bit 3
142 static GNU_CONST_DECL uint32_t PARTIAL_SPCOVERAGE4 = 0x01000; // Partial sp-coverage weight bit 4
143 static GNU_CONST_DECL uint32_t PARTIAL_SPCOVERAGE5 = 0x02000; // Partial sp-coverage weight bit 5
144 static GNU_CONST_DECL uint32_t PARTIAL_SPCOVERAGE6 = 0x04000; // Partial sp-coverage weight bit 6
145 static GNU_CONST_DECL uint32_t PARTIAL_SPCOVERAGE7 = 0x08000; // Partial sp-coverage weight bit 7
146 //
147 static GNU_CONST_DECL uint32_t PARTIAL_SPCOVERAGE_BITS = 0x0ff00;
148 //
149 static GNU_CONST_DECL uint32_t minSpCoverageCount = 0;
150 static GNU_CONST_DECL uint32_t maxSpCoverageCount = 256;
151 static GNU_CONST_DECL float maxSpCoverageScale = 256.0f;
152

329inline uint32_t DeepFlags::getSpCoverageCount () const {
330 const uint32_t count = (bits & PARTIAL_SPCOVERAGE_BITS);
331 return (count == 0 && isAdditive())?maxSpCoverageCount:(count >> 8); // translate 0 to max count
332}
333/*static*/ inline float DeepFlags::getSpCoverageWeightForCount (uint32_t count) {
334 return float(count) / maxSpCoverageScale;
335}
336inline float DeepFlags::getSpCoverageWeight () const {
337 return getSpCoverageWeightForCount(getSpCoverageCount());
338}
339//
340inline void DeepFlags::setSpCoverageCount (uint32_t count) {
341 bits = surfaceFlags();
342 if (count > 0 && count < maxSpCoverageCount)
343 bits |= ((count << 8) | ADDITIVE);
344}
345inline void DeepFlags::setSpCoverageWeight (float weight) {
346 setSpCoverageCount(int(weight * maxSpCoverageScale));
347}
348inline void DeepFlags::clearSpCoverageCount () {
349 bits = surfaceFlags();
350 clearSurfaceFlags(ADDITIVE);
351}
352inline uint32_t DeepFlags::addToSpCoverageCount (uint32_t count) {
353 const uint32_t new_count = getSpCoverageCount() + count;
354 if (new_count >= maxSpCoverageCount)
355 clearSpCoverageCount(); // saturated - clear the weight, disable additive
356 else
357 setSpCoverageCount(new_count);
358 return new_count;
359}
