167class ShadowLinkingSetter : public GeometryManagerPrimitiveVisitor
168{
169public:
170 explicit ShadowLinkingSetter(const scene_rdl2::rdl2::Layer* layer,
171 const std::vector<ShadowExclusionMapping>& mappings):
172 mLayer(layer),
173 mShadowExclusionMappings(mappings)
174 {}
175
176 virtual void visitPrimitive(geom::Primitive& p) override
177 {
178 geom::internal::Primitive* pImpl =
179 geom::internal::PrimitivePrivateAccess::getPrimitiveImpl(&p);
180
181 if (pImpl->getType() == geom::internal::Primitive::INSTANCE) {
182 return;
183 }
184
185 geom::internal::NamedPrimitive* np =
186 static_cast<geom::internal::NamedPrimitive*>(pImpl);
187 np->resetShadowLinking();
188
189 // get caster ids from primitive
190 std::unordered_set<int> casterIds;
191 np->getUniqueAssignmentIds(casterIds);
192 for (int casterId : casterIds) {
193 if (casterId < 0) {
194 continue;
195 }
196
197 // find caster id in mappings
198 const ShadowExclusionMapping* mappingPtr = nullptr;
199 for (auto& mapping : mShadowExclusionMappings) {
200 if (mapping.mCasterParts.empty() ||
201 mapping.mCasterParts.find(casterId) != mapping.mCasterParts.end()) {
202 mappingPtr = &mapping;
203 break;
204 }
205 }
206 // TO-DO: should we check that it doesn't occur a 2nd time?
207
208 const scene_rdl2::rdl2::ShadowSet* shadowSet = mLayer->lookupShadowSet(casterId);
209 const scene_rdl2::rdl2::ShadowReceiverSet* shadowReceiverSet = mLayer->lookupShadowReceiverSet(casterId);
210 bool complementShadowReceiverSet = (shadowReceiverSet &&
211 shadowReceiverSet->get(scene_rdl2::rdl2::ShadowReceiverSet::sComplementKey));
212
213 // create shadow linking if needed
214 if (shadowSet || shadowReceiverSet || mappingPtr) {
215 np->createShadowLinking(casterId, complementShadowReceiverSet);
216 }
217
218 // add lights to shadow set
219 if (shadowSet) {
220 const scene_rdl2::rdl2::SceneObjectVector &lights = shadowSet->getLights();
221 for (auto& light : lights) {
222 np->addShadowLinkedLight(casterId, static_cast<const scene_rdl2::rdl2::Light *>(light));
223 }
224 }
225
226 // Add mappings between casters and receivers (via rdla ShadowReceiverSet mechanism)
227 if (shadowReceiverSet) {
228 const scene_rdl2::rdl2::SceneObjectIndexable& receivers = shadowReceiverSet->getGeometries();
229 for (auto& sceneObject : receivers) {
230 scene_rdl2::rdl2::Geometry* geom = sceneObject->asA<scene_rdl2::rdl2::Geometry>();
231 int receiverId = mLayer->getAssignmentId(geom, "");
232 if (receiverId != -1) {
233 np->addShadowLinkedGeom(casterId, receiverId);
234 }
235 }
236 }
237
238 // Add mappings between casters and receivers (via geometry label attributes mechanism)
239 if (mappingPtr) {
240 for (auto& labeledGeoSet : mappingPtr->mLabeledGeoSets) {
241 for (auto& receiverGeom : labeledGeoSet->mGeometries) {
242 int receiverId = mLayer->getAssignmentId(receiverGeom, "");
243 if (receiverId != -1) {
244 np->addShadowLinkedGeom(casterId, receiverId);
245 }
246 }
247 }
248 }
249 }
250 }
251
252private:
253 const scene_rdl2::rdl2::Layer* mLayer;
254 const std::vector<ShadowExclusionMapping>& mShadowExclusionMappings;
255};

435class PrimitiveAttributeChecker : public GeometryManagerPrimitiveVisitor
436{
437public:
438 PrimitiveAttributeChecker(const scene_rdl2::rdl2::Geometry* geometry,
439 const scene_rdl2::rdl2::Layer* layer):
440 mGeometry(geometry), mLayer(layer)
441 {}
442
443 virtual void visitPrimitive(geom::Primitive& p) override
444 {
445 geom::internal::Primitive* pImpl =
446 geom::internal::PrimitivePrivateAccess::getPrimitiveImpl(&p);
447 if (pImpl->getType() != geom::internal::Primitive::INSTANCE) {
448 geom::internal::NamedPrimitive* np =
449 static_cast<geom::internal::NamedPrimitive*>(pImpl);
450 std::unordered_set<int> assignmentIds;
451 np->getUniqueAssignmentIds(assignmentIds);
452 for (int assignmentId : assignmentIds) {
453 if (assignmentId < 0) {
454 continue;
455 }
456 const scene_rdl2::rdl2::Material* s = mLayer->lookupMaterial(assignmentId);
457 if (s == nullptr) {
458 continue;
459 }
460 const shading::AttributeTable* table =
461 s->get<shading::RootShader>().getPrimitiveAttributeTable();
462 for (const auto& key : table->getRequiredAttributes()) {
463 // intentionally omit the check of wireframe related
464 // required attribute since they are filled in during
465 // postIntersect stage instead of render prep stage
466 if (key == shading::StandardAttributes::sPolyVertexType ||
467 key == shading::StandardAttributes::sNumPolyVertices) {
468 continue;
469 }
470 if (!np->hasAttribute(key)) {
471 // geom part missing attribute key shader require
472 mGeometry->error("part \"",
473 mLayer->lookupGeomAndPart(assignmentId).second,
474 "\" fail to provide primitive attribute \"",
475 key.getName(), "\" required by shading network \"",
476 s->getName(), "\"");
477 }
478 }
479 const scene_rdl2::rdl2::Displacement* d = mLayer->lookupDisplacement(assignmentId);
480 if (d == nullptr) {
481 continue;
482 }
483 table =
484 d->get<shading::RootShader>().getPrimitiveAttributeTable();
485 for (const auto& key : table->getRequiredAttributes()) {
486 if (!np->hasAttribute(key)) {
487 // geom part missing attribute key shader require
488 mGeometry->error("part \"",
489 mLayer->lookupGeomAndPart(assignmentId).second,
490 "\" fail to provide primitive attribute \"",
491 key.getName(), "\" required by displacement network \"",
492 d->getName(), "\"");
493 }
494 }
495 }
496 }
497 }
498
499private:
500 const scene_rdl2::rdl2::Geometry* mGeometry;
501 const scene_rdl2::rdl2::Layer* mLayer;

522static size_t
523fillGenerateList(
524 const scene_rdl2::rdl2::Layer::GeometryToRootShadersMap& fullGeometryRootShaders,
525 scene_rdl2::rdl2::Geometry* topLevelGeometry,
526 scene_rdl2::rdl2::Geometry* currentGeometry,
527 std::unordered_set<const scene_rdl2::rdl2::Geometry*>& isReferenced,
528 std::map<size_t, std::unordered_set<scene_rdl2::rdl2::Geometry*>>& toGenerate,
529 scene_rdl2::rdl2::Layer::GeometryToRootShadersMap& toLoadGeometryRootShaders,
530 int level)
531{
532 if (level > 0) {
533 isReferenced.insert(currentGeometry);
534 }
535
536 // Get the geometry the current geometry references, going down one more level
537 const scene_rdl2::rdl2::SceneObjectVector& references =
538 currentGeometry->get(scene_rdl2::rdl2::Geometry::sReferenceGeometries);
539
540 size_t generateOrder = 0;
541 for (const auto& ref : references) {
542 if (!ref->isA<scene_rdl2::rdl2::Geometry>()) {
543 continue;
544 }
545 scene_rdl2::rdl2::Geometry* referencedGeometry = ref->asA<scene_rdl2::rdl2::Geometry>();
546 generateOrder = max(generateOrder,
547 fillGenerateList(fullGeometryRootShaders, topLevelGeometry, referencedGeometry,
548 isReferenced, toGenerate, toLoadGeometryRootShaders, level + 1) + 1);
549 // the top level geometry references other geometries and needs to be aware of
550 // the shading network attribute requests from referenced geometries
551 const auto& refRootShaders = fullGeometryRootShaders.find(referencedGeometry);
552 if (refRootShaders == fullGeometryRootShaders.end()) {
553 continue;
554 }
555 for (scene_rdl2::rdl2::RootShader* s : refRootShaders->second) {
556 // collect up all the shaders we find in the referenced geometries and
557 // and associate them with the top level geometry
558 toLoadGeometryRootShaders[topLevelGeometry].insert(s);
559 }
560 }
561
562 toGenerate[generateOrder].insert(currentGeometry);
563
564 return generateOrder;
565}

567GeometryManager::GM_RESULT
568GeometryManager::loadGeometries(scene_rdl2::rdl2::Layer* layer,
569 const ChangeFlag flag,
570 const Mat4d& world2render,
571 const int currentFrame,
572 const geom::MotionBlurParams& motionBlurParams,
573 const unsigned threadCount,
574 const shading::PerGeometryAttributeKeySet &perGeometryAttributes)
575{
576 // Get all geometries in the Layer.
577 scene_rdl2::rdl2::Layer::GeometryToRootShadersMap fullGeometryRootShaders;
578 layer->getAllGeometryToRootShaders(fullGeometryRootShaders);
579 scene_rdl2::rdl2::Layer::GeometryToRootShadersMap toLoadGeometryRootShaders;
580 if (flag == ChangeFlag::ALL) {
581 toLoadGeometryRootShaders = fullGeometryRootShaders;
582 } else if (flag == ChangeFlag::UPDATE) {
583 // Get geometries with changed shaders which requires generate.
584 layer->getChangedGeometryToRootShaders(toLoadGeometryRootShaders);
585 }
586
587 if (!toLoadGeometryRootShaders.empty()) {
588 // fill out the list of geometry procedural to generate.
589 // the geometry that is referenced by other geometry will be generated
590 // first (this is somewhat a dependency graph ordered evaluation)
591 std::unordered_set<const scene_rdl2::rdl2::Geometry*> isReferenced;
592 std::map<size_t, std::unordered_set<scene_rdl2::rdl2::Geometry*>> toGenerate;
593 for (const auto& pair : toLoadGeometryRootShaders) {
594 scene_rdl2::rdl2::Geometry* geometry = pair.first;
595 // For the first iteration of this recursive function call, we just say
596 // that the top level geometry and the current geometry are the same
597 // which picks up the top level geometry and its shaders first before chasing any
598 // deeper referenced geometry.
599 fillGenerateList(fullGeometryRootShaders, geometry, geometry, isReferenced,
600 toGenerate, toLoadGeometryRootShaders, 0);
601 }
602
603 {
604 size_t totalGeometries = 0;
605 for (const auto &pair : toGenerate) {
606 const auto &geom = pair.second;
607 totalGeometries += geom.size();
608 }
609 if (mOptions.stats.mGeometryManagerExecTracker.startLoadGeometries(totalGeometries) ==
610 GeometryManagerExecTracker::RESULT::CANCELED) {
611 return GM_RESULT::CANCELED;
612 }
613 }
614
615 std::atomic<bool> geoLoadItemCancelCondition(false);
616
617 // now fire up the generate calls in dependency order
618 for (const auto& pair : toGenerate) {
619 const auto& geometries = pair.second;
620 tbb::parallel_for_each(geometries.begin(), geometries.end(),
621 [&](scene_rdl2::rdl2::Geometry* geometry) {
622
623 if (mOptions.stats.mGeometryManagerExecTracker.startLoadGeometriesItem() ==
624 GeometryManagerExecTracker::RESULT::CANCELED) {
625 geoLoadItemCancelCondition = true;
626 return;
627 }
628
629 mOptions.stats.logString("Generating " +
630 geometry->getSceneClass().getName() +
631 "(\"" + geometry->getName() + "\")");
632
633 // Load the procedural.
634 if (!geometry->getProcedural()) {
635 geometry->loadProcedural();
636 }
637 // Prepend geometry --> world.
638 shading::XformSamples geometry2render;
639 Mat4f l2r0 = toFloat(geometry->get(scene_rdl2::rdl2::Node::sNodeXformKey,
640 scene_rdl2::rdl2::TIMESTEP_BEGIN) * world2render);
641 Mat4f l2r1 = toFloat(geometry->get(scene_rdl2::rdl2::Node::sNodeXformKey,
642 scene_rdl2::rdl2::TIMESTEP_END) * world2render);
643 if (scene_rdl2::math::isEqual(l2r0, l2r1) ||
644 !motionBlurParams.isMotionBlurOn()) {
645 geometry2render = {xform<Xform3f>(l2r0)};
646 } else {
647 geometry2render = {xform<Xform3f>(l2r0), xform<Xform3f>(l2r1)};
648 }
649
650 // TODO the getRender2Object is used in shading stage that doesn't
651 // take time factor into account. We should update this function
652 // interface to accept more than one Xform sample when we need to
653 // do time varying shading calculation
654 geometry->setRender2Object(geometry2render[0].inverse());
655
656 shading::AttributeKeySet requestedAttributes;
657 // Find which primitive attributes this procedural needs to load.
658 const auto& rootShaders = toLoadGeometryRootShaders[geometry];
659 for (const scene_rdl2::rdl2::RootShader* const s : rootShaders) {
660 const auto& table = s->
661 get<shading::RootShader>().getPrimitiveAttributeTable();
662 const auto& reqKeys = table->getRequiredAttributes();
663 requestedAttributes.insert(reqKeys.begin(), reqKeys.end());
664 const auto& optKeys = table->getOptionalAttributes();
665 requestedAttributes.insert(optKeys.begin(), optKeys.end());
666 }
667 shading::PerGeometryAttributeKeySet::const_iterator itr =
668 perGeometryAttributes.find(geometry);
669 if (itr != perGeometryAttributes.end()) {
670 if (!itr->second.empty()) {
671 requestedAttributes.insert(itr->second.begin(),
672 itr->second.end());
673 }
674 }
675
676 // Generate geometry for the procedural.
677 GeomGenerateContext generateContext(layer,
678 geometry, std::move(requestedAttributes), currentFrame,
679 threadCount, motionBlurParams);
680 // TODO: We can't handle nested procedurals yet.
681 if (!geometry->getProcedural()->isLeaf()) {
682 geometry->error("Nested procedurals not supported yet.");
683 }
684 geom::Procedural* procedural = geometry->getProcedural();
685 try {
686 procedural->clear();
687 if (isReferenced.find(geometry) != isReferenced.end()) {
688 // generate in local space and promote the
689 // generated primitives in procedural to be
690 // a shared primitive group. this geometry won't show
691 shading::XformSamples localXform({geom::Mat43(scene_rdl2::math::one)});
692 procedural->generate(generateContext, localXform);
693 geom::internal::PrimitivePrivateAccess::transformToReference(
694 procedural);
695 } else {
696 procedural->generate(generateContext, geometry2render);
697 }
698 } catch (const std::exception &e) {
699 geometry->error(e.what());
700 }
701 RdlGeometrySetter rdlGeometrySetter(geometry);
702 procedural->forEachPrimitive(rdlGeometrySetter);
703
704 if (mOptions.stats.mGeometryManagerExecTracker.endLoadGeometriesItem() ==
705 GeometryManagerExecTracker::RESULT::CANCELED) {
706 geoLoadItemCancelCondition = true;
707 return;
708 }
709 });
710 }
711
712 if (geoLoadItemCancelCondition) {
713 mOptions.stats.mGeometryManagerExecTracker.finalizeLoadGeometriesItem(true); // cancel = true
714 return GM_RESULT::CANCELED;
715 } else {
716 mOptions.stats.mGeometryManagerExecTracker.finalizeLoadGeometriesItem(false); // cancel = false
717 }
718
719 for (const auto& gs : toLoadGeometryRootShaders) {
720 scene_rdl2::rdl2::Geometry * const geometry = gs.first;
721 mOptions.stats.logGeneratingProcedurals(
722 geometry->getSceneClass().getName(), geometry->getName());
723 }
724 } else {
725 if (mOptions.stats.mGeometryManagerExecTracker.startLoadGeometries(0) ==
726 GeometryManagerExecTracker::RESULT::CANCELED) {
727 return GM_RESULT::CANCELED;
728 }
729 }
730
731 if (mOptions.stats.mGeometryManagerExecTracker.endLoadGeometries() ==
732 GeometryManagerExecTracker::RESULT::CANCELED) {
733 return GM_RESULT::CANCELED;
734 }
735
736 return GM_RESULT::FINISHED;
737}

739void
740GeometryManager::updateGeometryData(scene_rdl2::rdl2::Layer* layer,
741 scene_rdl2::rdl2::Geometry* geometry,
742 const std::vector<std::string>& meshNames,
743 const VertexBufferArray& vertexBuffers,
744 const Mat4d& world2render,
745 const int currentFrame,
746 const geom::MotionBlurParams& motionBlurParams,
747 const unsigned threadCount)
748{
749 geom::Procedural* procedural = geometry->getProcedural();
750 if (procedural == nullptr) {
751 std::string msg =
752 std::string("Incorrectly update uninitialized geometry ") +
753 geometry->getName();
754 MNRY_ASSERT(false, msg.c_str());
755 return;
756 }
757
758 // Prepend geometry --> world.
759 shading::XformSamples geometry2render;
760 Mat4f l2r0 = toFloat(geometry->get(scene_rdl2::rdl2::Node::sNodeXformKey,
761 scene_rdl2::rdl2::TIMESTEP_BEGIN) * world2render);
762 Mat4f l2r1 = toFloat(geometry->get(scene_rdl2::rdl2::Node::sNodeXformKey,
763 scene_rdl2::rdl2::TIMESTEP_END) * world2render);
764 if (scene_rdl2::math::isEqual(l2r0, l2r1) ||
765 !motionBlurParams.isMotionBlurOn()) {
766 geometry2render = {xform<Xform3f>(l2r0)};
767 } else {
768 geometry2render = {xform<Xform3f>(l2r0), xform<Xform3f>(l2r1)};
769 }
770 // TODO the getRender2Object is used in shading stage that doesn't
771 // take time factor into account. We should update this function
772 // interface to accept more than one Xform sample when we need to
773 // do time varying shading calculation
774 geometry->setRender2Object(geometry2render[0].inverse());
775
776 GeomUpdateContext updateContext(layer,
777 geometry,
778 currentFrame,
779 threadCount,
780 motionBlurParams);
781
782 updateContext.setMeshNames(meshNames);
783 updateContext.setMeshVertexDatas(vertexBuffers);
784
785 // Hand the update off to the geom library to process.
786 // Note: this also re-tessellates the primitives.
787 procedural->update(updateContext, geometry2render);
788
789 // Make sure the accelerator gets updated accordingly for the
790 // GeometrySets that contain this geometry
791 std::for_each(mSceneContext->beginGeometrySet(),
792 mSceneContext->endGeometrySet(),
793 [&](scene_rdl2::rdl2::GeometrySet* geometrySet) {
794 if (geometrySet->contains(geometry)) {
795 mDeformedGeometrySets.insert(geometrySet);
796 }
797 });
798}

800// collect all the SharedPrimitives (referenced by instances) and
801// leaf Primitives that need to be tessellated from input geometrySets
802static void
803collectPrimitives(const scene_rdl2::rdl2::SceneContext::GeometrySetVector& geometrySets,
804 const scene_rdl2::rdl2::Layer::GeometryToRootShadersMap& g2s,
805 geom::SharedPrimitiveSet& sharedPrimitives,
806 geom::InternalPrimitiveList& primitivesToTessellate,
807 geom::InternalPrimitiveList& curvePrimitives)
808{
809 // there is chance that multiple geometrySets contain shared geometries,
810 // we need to make sure all the geometries to loop through tessellation
811 // are not duplicated to avoid race condition
812 std::unordered_set<geom::Procedural*> uniqueProceduralSet;
813 PrimitiveCollector primitiveCollector(sharedPrimitives, primitivesToTessellate, curvePrimitives);
814 for (auto& geometrySet : geometrySets) {
815 // Some GeometrySets (e.g. ShadowReceiverSets) are not intended to be part of the scene, but are just
816 // there to represent collections of things and should not be added to the BVH.
817 if (!geometrySet->includeInBVH()) {
818 continue;
819 }
820 const scene_rdl2::rdl2::SceneObjectIndexable& geometries = geometrySet->getGeometries();
821 for (auto& sceneObject : geometries) {
822 scene_rdl2::rdl2::Geometry* geom = sceneObject->asA<scene_rdl2::rdl2::Geometry>();
823 if (g2s.find(geom) == g2s.end()) {
824 continue;
825 }
826 geom::Procedural * const procedural = geom->getProcedural();
827 if (!procedural) {
828 // All parts in a procedural are unassigned in the layer
829 continue;
830 }
831 // TODO: We can't handle nested procedurals yet.
832 MNRY_ASSERT_REQUIRE(procedural->isLeaf(),
833 "Nested procedurals not supported yet.");
834 if (uniqueProceduralSet.insert(procedural).second) {
835 geom::ProceduralLeaf* const leaf =
836 static_cast<geom::ProceduralLeaf*>(procedural);
837 leaf->forEachPrimitive(primitiveCollector);
838 } else {
839 sceneObject->warn(
840 "This geometry belongs to more than one geometry set. "
841 "This will make the geometry appear in multiple sub BVHs, "
842 "which can be inefficient and cause rendering artifacts.");
843 }
844 }
845 }

848void
849bakeVolumeShaderDensityMap(const scene_rdl2::rdl2::Layer* layer,
850 geom::internal::NamedPrimitive* prim,
851 const geom::MotionBlurParams& motionBlurParams,
852 const geom::internal::VolumeAssignmentTable* volumeAssignmentTable)
853{
854 // Bake volume shader. This creates a vdb grid.
855 if (prim->hasVolumeAssignment(layer) && (
856 prim->getType() == geom::internal::Primitive::POLYMESH ||
857 prim->getType() == geom::internal::Primitive::VDB_VOLUME)) {
858
859 std::unordered_set<int> assignmentIds;
860 prim->getUniqueAssignmentIds(assignmentIds);
861 const scene_rdl2::rdl2::VolumeShader* volumeShader = nullptr;
862 auto it = assignmentIds.begin();
863 int assignmentId;
864 // get first volume shader assignment
865 while (volumeShader == nullptr && it != assignmentIds.end()) {
866 assignmentId = *it++;
867 volumeShader = layer->lookupVolumeShader(assignmentId);
868 }
869
870 scene_rdl2::math::Mat4f primToRender(scene_rdl2::math::one);
871 switch (prim->getType()) {
872 case geom::internal::Primitive::POLYMESH:
873 primToRender = static_cast<geom::internal::Mesh *>(prim)->getTransform();
874 break;
875 case geom::internal::Primitive::VDB_VOLUME:
876 primToRender = static_cast<geom::internal::VdbVolume *>(prim)->getTransform();
877 break;
878 }
879
880 prim->bakeVolumeShaderDensityMap(volumeShader, primToRender, motionBlurParams,
881 volumeAssignmentTable, assignmentId);
882 }

885void
886GeometryManager::bakeGeometry(scene_rdl2::rdl2::Layer* layer,
887 const geom::MotionBlurParams& motionBlurParams,
888 const std::vector<mcrt_common::Frustum>& frustums,
889 const scene_rdl2::math::Mat4d& world2render,
890 std::vector<std::unique_ptr<geom::BakedMesh>>& bakedMeshes,
891 std::vector<std::unique_ptr<geom::BakedCurves>>& bakedCurves)
892{
893 // update changes for BVH
894 scene_rdl2::rdl2::Layer::GeometryToRootShadersMap g2s;
895
896 scene_rdl2::rdl2::SceneContext::GeometrySetVector geometrySets;
897 geometrySets = mSceneContext->getAllGeometrySets();
898 layer->getAllGeometryToRootShaders(g2s);
899
900 geom::SharedPrimitiveSet sharedPrimitives;
901 geom::InternalPrimitiveList primitivesToTessellate;
902 geom::InternalPrimitiveList curvePrimitives;
903 // if a primitive is not generated, we don't want to
904 // tessellate it. Pass in g2s to verify whether a geometry
905 // in geometrySets has been generated
906 collectPrimitives(geometrySets, g2s,
907 sharedPrimitives,
908 primitivesToTessellate,
909 curvePrimitives);
910 // trasform shared primitives to their own local space
911 TransformConcatenator transformConcatenator(motionBlurParams);
912 ReferenceSetter referenceSetter(0.f, true);
913 for (auto& sharedPrimitive : sharedPrimitives) {
914 sharedPrimitive->getPrimitive()->accept(transformConcatenator);
915 // Set adaptive error to 0 on objects that are instanced to disable
916 // adaptive tessellation
917 sharedPrimitive->getPrimitive()->accept(referenceSetter);
918 }
919
920 // tessellate the primitives
921 const bool fastGeomUpdate = mSceneContext->getSceneVariables().get(
922 scene_rdl2::rdl2::SceneVariables::sFastGeomUpdate);
923
924 const bool enableDisplacement = mSceneContext->getSceneVariables().get(
925 scene_rdl2::rdl2::SceneVariables::sEnableDisplacement);
926
927 for (geom::internal::Primitive* prim : primitivesToTessellate) {
928 geom::internal::NamedPrimitive* namedPrim =
929 static_cast<geom::internal::NamedPrimitive*>(prim);
930 }
931
932 tbb::blocked_range<size_t> range(0, primitivesToTessellate.size());
933 tbb::parallel_for(range, [&](const tbb::blocked_range<size_t> &r) {
934 for (size_t i = r.begin(); i < r.end(); ++i) {
935 geom::internal::NamedPrimitive* prim =
936 static_cast<geom::internal::NamedPrimitive*>(
937 primitivesToTessellate[i]);
938 try {
939 geom::internal::TessellationParams params(layer,
940 frustums,
941 world2render,
942 enableDisplacement,
943 fastGeomUpdate,
944 /* isBaking = */ true,
945 mVolumeAssignmentTable.get());
946 prim->tessellate(params);
947
948 // Bake the density map of a volume shader bound to this primitive. This is more
949 // optimal than directly sampling the density map during mcrt. This creates a vdb grid.
950 bakeVolumeShaderDensityMap(layer, prim, motionBlurParams, mVolumeAssignmentTable.get());
951 } catch (const std::exception &e) {
952 prim->getRdlGeometry()->error(e.what());
953 }
954 }
955 });
956
957 const float fps = mSceneContext->getSceneVariables().get(scene_rdl2::rdl2::SceneVariables::sFpsKey);
958 const float currentFrame = mSceneContext->getSceneVariables().get(scene_rdl2::rdl2::SceneVariables::sFrameKey);
959 std::vector<float> motionSteps = motionBlurParams.getMotionSteps();
960
961 // retrieve the tessellated geometry
962 for (geom::internal::Primitive* prim : primitivesToTessellate) {
963 geom::internal::NamedPrimitive* namedPrim =
964 static_cast<geom::internal::NamedPrimitive*>(prim);
965
966 geom::internal::Mesh* mesh = dynamic_cast<geom::internal::Mesh*>(namedPrim);
967 if (mesh) {
968 std::unique_ptr<geom::BakedMesh> bakedMesh = fauxstd::make_unique<geom::BakedMesh>();
969 bakedMesh->mRdlGeometry = mesh->getRdlGeometry();
970 for (size_t m = 0; m < motionSteps.size(); ++m) {
971 bakedMesh->mMotionFrames.push_back((currentFrame + motionSteps[m]) / fps);
972 }
973 mesh->getBakedMesh(*bakedMesh);
974 bakedMeshes.push_back(std::move(bakedMesh));
975 }
976
977 // handle baking other primitive types here
978 }
979
980 // retrieve the curves geometry
981 for (geom::internal::Primitive* prim : curvePrimitives) {
982 geom::internal::NamedPrimitive* namedPrim =
983 static_cast<geom::internal::NamedPrimitive*>(prim);
984
985 geom::internal::Curves* curves = dynamic_cast<geom::internal::Curves*>(namedPrim);
986 if (curves) {
987 std::unique_ptr<geom::BakedCurves> bakedCurve = fauxstd::make_unique<geom::BakedCurves>();
988 bakedCurve->mRdlGeometry = curves->getRdlGeometry();
989 bakedCurve->mLayer = layer;
990 for (size_t m = 0; m < motionSteps.size(); ++m) {
991 bakedCurve->mMotionFrames.push_back((currentFrame + motionSteps[m]) / fps);
992 }
993 curves->getBakedCurves(*bakedCurve);
994 bakedCurves.push_back(std::move(bakedCurve));
995 }
996 }
997}

999GeometryManager::GM_RESULT
1000GeometryManager::finalizeChanges(scene_rdl2::rdl2::Layer* layer,
1001 const geom::MotionBlurParams& motionBlurParams,
1002 const std::vector<mcrt_common::Frustum>& frustums,
1003 const scene_rdl2::math::Mat4d& world2render,
1004 OptimizationTarget accelMode,
1005 bool updateSceneBVH)
1006{
1007 if (mOptions.stats.mGeometryManagerExecTracker.startFinalizeChange() ==
1008 GeometryManagerExecTracker::RESULT::CANCELED) {
1009 return GM_RESULT::CANCELED;
1010 }
1011
1012 if (mChangeStatus == ChangeFlag::NONE) {
1013 mOptions.stats.logString("No need to build, skipping...");
1014 } else {
1015 // update changes for BVH
1016 scene_rdl2::rdl2::Layer::GeometryToRootShadersMap g2s;
1017
1018 scene_rdl2::rdl2::SceneContext::GeometrySetVector geometrySets;
1019 if (mChangeStatus == ChangeFlag::ALL) {
1020 geometrySets = mSceneContext->getAllGeometrySets();
1021 layer->getAllGeometryToRootShaders(g2s);
1022 } else if (mChangeStatus == ChangeFlag::UPDATE){
1023 mSceneContext->getUpdatedOrDeformedGeometrySets(layer, geometrySets);
1024 layer->getChangedGeometryToRootShaders(g2s);
1025 }
1026
1027 geom::SharedPrimitiveSet sharedPrimitives;
1028 geom::InternalPrimitiveList primitivesToTessellate;
1029 geom::InternalPrimitiveList curvePrimitives;
1030 // if a primitive is not (re)generated, we don't want to
1031 // (re)tessellate it. Pass in g2s to verify whether a geometry
1032 // in geometrySets has been (re)generated
1033 collectPrimitives(geometrySets, g2s,
1034 sharedPrimitives,
1035 primitivesToTessellate,
1036 curvePrimitives);
1037 // trasform shared primitives to their own local space
1038 TransformConcatenator transformConcatenator(motionBlurParams);
1039 ReferenceSetter referenceSetter(0.f, true);
1040 for (auto& sharedPrimitive : sharedPrimitives) {
1041 sharedPrimitive->getPrimitive()->accept(transformConcatenator);
1042 // Set adaptive error to 0 on objects that are instanced to disable
1043 // adaptive tessellation
1044 sharedPrimitive->getPrimitive()->accept(referenceSetter);
1045 }
1046
1047 // The following only needs to be done once before we update the BVH
1048 // and before we tessellate geometry that is in the BVH.
1049 // MeshLight geometries are not in the BVH, but do need to be tessellated.
1050 // They do not have layer assignments so they do not need their assignments
1051 // set here.
1052 if (updateSceneBVH) {
1053 // set assignments
1054 for (auto& gsPair : g2s) {
1055 scene_rdl2::rdl2::Geometry* geometry = gsPair.first;
1056 geom::Procedural* procedural = geometry->getProcedural();
1057 if (procedural) {
1058 AssignmentAndXformSetter assignmentSetter(layer);
1059 procedural->forEachPrimitive(assignmentSetter);
1060 }
1061 }
1062
1063 // Build the runtime volume<->assignment lookup table before tessellation
1064 mVolumeAssignmentTable->initLookupTable(layer);
1065 geom::internal::forEachTLS([this](geom::internal::TLState* tls) {
1066 tls->mVolumeRayState.initializeVolumeLookup(
1067 mVolumeAssignmentTable.get());
1068 });
1069 }
1070
1071 // (re)tessellate the primitives
1072 if (tessellate(layer, primitivesToTessellate, frustums, world2render, motionBlurParams) ==
1073 GM_RESULT::CANCELED) {
1074 return GM_RESULT::CANCELED;
1075 }
1076
1077 // Verify whether all the required primitive attributes from
1078 // shading network are provided
1079 for (auto& gsPair : g2s) {
1080 scene_rdl2::rdl2::Geometry* geometry = gsPair.first;
1081 geom::Procedural* procedural = geometry->getProcedural();
1082 if (procedural) {
1083 PrimitiveAttributeChecker primitiveAttributeChecker(
1084 gsPair.first, layer);
1085 procedural->forEachPrimitive(primitiveAttributeChecker);
1086 }
1087 }
1088
1089 if (updateSceneBVH) {
1090 if (mOptions.stats.mGeometryManagerExecTracker.startBVHConstruction() ==
1091 GeometryManagerExecTracker::RESULT::CANCELED) {
1092 return GM_RESULT::CANCELED;
1093 }
1094 updateAccelerator(layer, geometrySets, g2s, accelMode);
1095 if (mOptions.stats.mGeometryManagerExecTracker.endBVHConstruction() ==
1096 GeometryManagerExecTracker::RESULT::CANCELED) {
1097 return GM_RESULT::CANCELED;
1098 }
1099 }
1100 }
1101
1102 if (mOptions.stats.mGeometryManagerExecTracker.endFinalizeChange() ==
1103 GeometryManagerExecTracker::RESULT::CANCELED) {
1104 return GM_RESULT::CANCELED;
1105 }
1106
1107 return GM_RESULT::FINISHED;
1108}

1110void
1111GeometryManager::getEmissiveRegions(const scene_rdl2::rdl2::Layer* layer,
1112 std::vector<geom::internal::EmissiveRegion>& emissiveRegions) const
1113{
1114 // maps assignment id to EmissionDistributions for emissive reference volumes
1115 using SharedEmissionDistributions =
1116 std::unordered_map<int, std::shared_ptr<geom::internal::EmissionDistribution>>;
1117
1118 class SharedEmissionDistributionCollector : public geom::PrimitiveVisitor
1119 {
1120 public:
1121 SharedEmissionDistributionCollector(
1122 SharedEmissionDistributions &emissionDistributions,
1123 int assignmentId,
1124 int visibilityMask,
1125 const scene_rdl2::rdl2::VolumeShader *volumeShader):
1126 mEmissionDistributions(emissionDistributions),
1127 mAssignmentId(assignmentId),
1128 mVisibilityMask(visibilityMask),
1129 mVolumeShader(volumeShader)
1130 {
1131 }
1132
1133 void visitPrimitive(geom::Primitive &p) override
1134 {
1135 geom::internal::Primitive* pImpl =
1136 geom::internal::PrimitivePrivateAccess::getPrimitiveImpl(&p);
1137 constexpr int secondaryVisibilities =
1138 scene_rdl2::rdl2::DIFFUSE_REFLECTION |
1139 scene_rdl2::rdl2::DIFFUSE_TRANSMISSION |
1140 scene_rdl2::rdl2::GLOSSY_REFLECTION |
1141 scene_rdl2::rdl2::GLOSSY_TRANSMISSION |
1142 scene_rdl2::rdl2::MIRROR_REFLECTION |
1143 scene_rdl2::rdl2::MIRROR_TRANSMISSION |
1144 scene_rdl2::rdl2::PHASE_REFLECTION |
1145 scene_rdl2::rdl2::PHASE_TRANSMISSION;
1146 bool canIlluminate =
1147 (secondaryVisibilities & mVisibilityMask) != 0;
1148 if (pImpl->getType() == geom::internal::Primitive::VDB_VOLUME) {
1149 auto pVolume = static_cast<geom::internal::VdbVolume*>(pImpl);
1150 canIlluminate &= pVolume->hasEmissionField();
1151 }
1152 if (canIlluminate) {
1153 mEmissionDistributions[mAssignmentId] =
1154 std::shared_ptr<geom::internal::EmissionDistribution>
1155 (pImpl->computeEmissionDistribution(mVolumeShader));
1156 // the optimization for empty emission grids is
1157 // handled when we create the emissive regions
1158 }
1159 }
1160
1161 void visitPrimitiveGroup(geom::PrimitiveGroup& pg) override
1162 {
1163 pg.forEachPrimitive(*this, /* doParallel = */ false);
1164 }
1165
1166 void visitTransformedPrimitive(geom::TransformedPrimitive& t) override
1167 {
1168 t.getPrimitive()->accept(*this);
1169 }
1170
1171 private:
1172 SharedEmissionDistributions &mEmissionDistributions;
1173 int mAssignmentId;
1174 int mVisibilityMask; // of this geometry
1175 const scene_rdl2::rdl2::VolumeShader *mVolumeShader;
1176 };
1177
1178 // collect the shared emission distributions
1179 SharedEmissionDistributions sharedEmissionDistributions;
1180 for (int assignmentId = 0; assignmentId < layer->getAssignmentCount(); ++assignmentId) {
1181 const scene_rdl2::rdl2::VolumeShader *volumeShader = layer->lookupVolumeShader(assignmentId);
1182 if (volumeShader && volumeShader->getProperties() & scene_rdl2::rdl2::VolumeShader::IS_EMISSIVE) {
1183 const scene_rdl2::rdl2::Geometry *geom = layer->lookupGeomAndPart(assignmentId).first;
1184 geom::Procedural *proc = const_cast<geom::Procedural *>(geom->getProcedural());
1185 if (proc->isLeaf()) {
1186 const geom::ProceduralLeaf *pLeaf = static_cast<geom::ProceduralLeaf *>(proc);
1187 if (pLeaf->isReference()) {
1188 const geom::SharedPrimitive *sharedPrimitive = pLeaf->getReference().get();
1189 SharedEmissionDistributionCollector collector(sharedEmissionDistributions,
1190 assignmentId, geom->getVisibilityMask(), volumeShader);
1191 sharedPrimitive->getPrimitive()->accept(collector);
1192 }
1193 }
1194 }
1195 }
1196
1197 class EmissiveRegionsCollector : public geom::PrimitiveVisitor
1198 {
1199 public:
1200 explicit EmissiveRegionsCollector(
1201 std::vector<geom::internal::EmissiveRegion>& emissiveRegions,
1202 int assignmentId, int volumeId,
1203 const scene_rdl2::rdl2::VolumeShader* volumeShader,
1204 int visibilityMask):
1205 mEmissiveRegions(emissiveRegions), mAssignmentId(assignmentId),
1206 mVolumeId(volumeId), mVolumeShader(volumeShader),
1207 mVisibilityMask(visibilityMask)
1208 {}
1209
1210 virtual void visitPrimitive(
1211 geom::Primitive& p) override
1212 {
1213 geom::internal::Primitive* pImpl =
1214 geom::internal::PrimitivePrivateAccess::getPrimitiveImpl(&p);
1215 if (pImpl->hasAssignment(mAssignmentId)) {
1216 int secondaryVisibilities =
1217 scene_rdl2::rdl2::DIFFUSE_REFLECTION |
1218 scene_rdl2::rdl2::DIFFUSE_TRANSMISSION |
1219 scene_rdl2::rdl2::GLOSSY_REFLECTION |
1220 scene_rdl2::rdl2::GLOSSY_TRANSMISSION |
1221 scene_rdl2::rdl2::MIRROR_REFLECTION |
1222 scene_rdl2::rdl2::MIRROR_TRANSMISSION |
1223 scene_rdl2::rdl2::PHASE_REFLECTION |
1224 scene_rdl2::rdl2::PHASE_TRANSMISSION;
1225 bool canIlluminate =
1226 (secondaryVisibilities & mVisibilityMask) != 0;
1227 if (pImpl->getType() == geom::internal::Primitive::VDB_VOLUME) {
1228 auto pVolume = static_cast<geom::internal::VdbVolume*>(pImpl);
1229 canIlluminate &= pVolume->hasEmissionField();
1230 }
1231
1232 if (canIlluminate) {
1233 geom::internal::EmissiveRegion emissiveRegion(pImpl,
1234 mVolumeShader, mVolumeId, mVisibilityMask);
1235 if (emissiveRegion.size() > 0) {
1236 mEmissiveRegions.emplace_back(std::move(emissiveRegion));
1237 }
1238 }
1239 }
1240 }
1241
1242 virtual void visitPrimitiveGroup(
1243 geom::PrimitiveGroup& pg) override
1244 {
1245 pg.forEachPrimitive(*this, false);
1246 }
1247
1248 virtual void visitTransformedPrimitive(
1249 geom::TransformedPrimitive& t) override
1250 {
1251 t.getPrimitive()->accept(*this);
1252 }
1253
1254 virtual void visitInstance(geom::Instance& i) override
1255 {
1256 // Keep this here so we skip over non-emissive volume instances
1257 }
1258
1259 private:
1260 std::vector<geom::internal::EmissiveRegion>& mEmissiveRegions;
1261 int mAssignmentId;
1262 int mVolumeId;
1263 const scene_rdl2::rdl2::VolumeShader* mVolumeShader;
1264 int mVisibilityMask;
1265 };
1266
1267 emissiveRegions.clear();
1268 for (size_t i = 0; i < mVolumeAssignmentTable->getVolumeCount(); ++i) {
1269 int volumeId = i;
1270 int assignmentId = mVolumeAssignmentTable->getAssignmentId(volumeId);
1271
1272 // do we have a shared emission distribution for this assignment id?
1273 // if so, then this volume id is an instance
1274 SharedEmissionDistributions::const_iterator itr = sharedEmissionDistributions.find(assignmentId);
1275 if (itr != sharedEmissionDistributions.end()) {
1276 std::shared_ptr<geom::internal::EmissionDistribution> emissionDistribution = itr->second;
1277 // "count > 0" handles the edge case where the user submits an emission grid where
1278 // all the values are 0. In that case, it is not worth creating an emissive region.
1279 if (emissionDistribution->count() > 0) {
1280 const int visibilityMask = mVolumeAssignmentTable->getVisibilityMask(volumeId);
1281 geom::internal::EmissionDistribution::Transform transform = emissionDistribution->getTransform();
1282 geom::Mat43 primToRender;
1283 if (mVolumeAssignmentTable->evalInstanceXform(volumeId, /*time = */ 0.f, primToRender)) {
1284 const scene_rdl2::math::Mat4f distToRender[2] = {
1285 transform.getDistToRender()[0] * scene_rdl2::math::Mat4f(primToRender),
1286 transform.getDistToRender()[1] * scene_rdl2::math::Mat4f(primToRender)
1287 };

1289 scene_rdl2::math::XformComponent<scene_rdl2::math::Mat3f> comp;
1290 scene_rdl2::math::decompose(primToRender, comp);
1291 const Vec3f s(comp.s.row0().x, comp.s.row1().y, comp.s.row2().z);
1292 const float invUnitVolume = transform.getInvUnitVolume() / (s.x * s.y * s.z);
1293 transform = geom::internal::EmissionDistribution::Transform(distToRender, invUnitVolume);
1294 }
1295 emissiveRegions.emplace_back(volumeId, visibilityMask, emissionDistribution, transform);
1296 }
1297 continue;
1298 }
1299
1300 // otherwise, this volume is not an instance, create a distribution
1301 const scene_rdl2::rdl2::VolumeShader* volumeShader =
1302 mVolumeAssignmentTable->lookupWithVolumeId(volumeId);
1303 if (volumeShader->getProperties() & scene_rdl2::rdl2::VolumeShader::IS_EMISSIVE) {
1304 // unfortunately right now procedural doesn't offer const
1305 // forEachPrimitive interface so we need to do a const_cast here
1306 scene_rdl2::rdl2::Geometry* geom =
1307 const_cast<scene_rdl2::rdl2::Geometry*>(layer->lookupGeomAndPart(assignmentId).first);
1308 EmissiveRegionsCollector collector(emissiveRegions,
1309 assignmentId, volumeId, volumeShader,
1310 geom->getVisibilityMask());
1311 geom::Procedural* procedural = geom->getProcedural();

1313 // and the number of emissive regions in the scene probably not
1314 // worth the effort to do parallel run
1315 procedural->forEachPrimitive(collector, false);
1316 }

1324GeometryManager::tessellate(scene_rdl2::rdl2::Layer* layer,
1325 geom::InternalPrimitiveList& primitivesToTessellate,
1326 const std::vector<mcrt_common::Frustum>& frustums,
1327 const scene_rdl2::math::Mat4d& world2render,
1328 const geom::MotionBlurParams& motionBlurParams)
1329{
1330 mOptions.stats.logString(
1331 "---------- Tessellating Geometry -------------------------");
1332 size_t statsSize = mOptions.stats.mPerPrimitiveTessellationTime.size();
1333 mOptions.stats.mPerPrimitiveTessellationTime.resize(statsSize +
1334 primitivesToTessellate.size());
1335 // tessellation timer for all primitives
1336 util::AverageDouble previousTessellationTime(
1337 mOptions.stats.mTessellationTime.getCount(),
1338 mOptions.stats.mTessellationTime.getSum());
1339 Timer tessellationTimer(mOptions.stats.mTessellationTime);
1340 tessellationTimer.start();
1341
1342 const bool fastGeomUpdate = mSceneContext->getSceneVariables().get(
1343 scene_rdl2::rdl2::SceneVariables::sFastGeomUpdate);
1344
1345 const bool enableDisplacement = mSceneContext->getSceneVariables().get(
1346 scene_rdl2::rdl2::SceneVariables::sEnableDisplacement);
1347
1348 // This thread id is for the debug log. We tessellate the primitives
1349 // in parallel, so we want to know which primitive is assigned to
1350 // which thread.
1351 gThreadIdCounter = 0;
1352 struct ThreadID {
1353 // When we create a ThreadID, the counter increments and so
1354 // each thread gets a unique human readable id.
1355 ThreadID() : mId(gThreadIdCounter.fetch_and_increment()){}
1356 unsigned mId;
1357 };
1358 typedef tbb::enumerable_thread_specific< ThreadID > EnumerableThreadID;
1359 EnumerableThreadID enumerableThreadID;
1360
1361 if (mOptions.stats.mGeometryManagerExecTracker.startTessellation(primitivesToTessellate.size()) ==
1362 GeometryManagerExecTracker::RESULT::CANCELED) {
1363 return GM_RESULT::CANCELED;
1364 }
1365
1366 std::atomic<bool> tessellationCancelCondition(false);
1367
1368 tbb::blocked_range<size_t> range(0, primitivesToTessellate.size());
1369 tbb::parallel_for(range, [&](const tbb::blocked_range<size_t> &r) {
1370 // When we create this localThreadID, the global atomic gThreadIdCounter is
1371 // incremented. This way, each thread gets a unique id.
1372 EnumerableThreadID::reference localThreadID = enumerableThreadID.local();
1373
1374 for (size_t i = r.begin(); i < r.end(); ++i) {
1375
1376 if (mOptions.stats.mGeometryManagerExecTracker.startTessellationItem() ==
1377 GeometryManagerExecTracker::RESULT::CANCELED) {
1378 tessellationCancelCondition = true;
1379 return;
1380 }
1381
1382 // tessellation timer for each primitive
1383 util::AverageDouble primTessTime;
1384 Timer primTessTimer(primTessTime);
1385 primTessTime.reset();
1386 geom::internal::NamedPrimitive* prim =
1387 static_cast<geom::internal::NamedPrimitive*>(
1388 primitivesToTessellate[i]);
1389
1390 std::stringstream startMsg;
1391 startMsg << "Thread " << localThreadID.mId << "\t: START tessellating "
1392 << prim->getRdlGeometry()->getName() << " " << prim->getName();
1393 mOptions.stats.logDebugString(startMsg.str());
1394 primTessTimer.start();
1395 try {
1396 const geom::internal::TessellationParams tessParams(layer,
1397 frustums,
1398 world2render,
1399 enableDisplacement,
1400 fastGeomUpdate,
1401 /* isBaking = */ false,
1402 mVolumeAssignmentTable.get());
1403 prim->tessellate(tessParams);
1404
1405 // Bake the density map of a volume shader bound to this primitive. This is more
1406 // optimal than directly sampling the density map during mcrt. This creates a vdb grid.
1407 bakeVolumeShaderDensityMap(layer, prim, motionBlurParams, mVolumeAssignmentTable.get());
1408
1409 } catch (const std::exception &e) {
1410 prim->getRdlGeometry()->error(e.what());
1411 }
1412 primTessTimer.stop();
1413 // append tessellation time
1414 mOptions.stats.mPerPrimitiveTessellationTime[statsSize + i] =
1415 std::make_pair(prim, primTessTime.getSum());
1416
1417 std::stringstream finishedMsg;
1418 finishedMsg << "Thread " << localThreadID.mId << "\t: FINISHED tessellating "
1419 << prim->getRdlGeometry()->getName() << " " << prim->getName();
1420 mOptions.stats.logDebugString(finishedMsg.str());
1421
1422 if (mOptions.stats.mGeometryManagerExecTracker.endTessellationItem() ==
1423 GeometryManagerExecTracker::RESULT::CANCELED) {
1424 tessellationCancelCondition = true;
1425 return;
1426 }
1427 }
1428 });
1429 tessellationTimer.stop();
1430 mOptions.stats.mTessellationTime += previousTessellationTime;
1431
1432 // return unused memory from malloc() arena to OS so process memory usage
1433 // stats are accurate
1434 malloc_trim(0);
1435
1436 mOptions.stats.logString("Tessellation finished.");
1437
1438 if (tessellationCancelCondition) {
1439 mOptions.stats.mGeometryManagerExecTracker.finalizeTessellationItem(true); // cancel = true
1440 return GM_RESULT::CANCELED;
1441 } else {
1442 mOptions.stats.mGeometryManagerExecTracker.finalizeTessellationItem(false); // cancel = false
1443 }
1444
1445 if (mOptions.stats.mGeometryManagerExecTracker.endTessellation() ==
1446 GeometryManagerExecTracker::RESULT::CANCELED) {
1447 return GM_RESULT::CANCELED;
1448 }
1449
1450 return GM_RESULT::FINISHED;
1451}

1572 label = strtok_r(NULL, ",", &context);
1573 }

1577void GeometryManager::updateShadowLinkings(const scene_rdl2::rdl2::Layer* layer)
1578{
1579 // Get the geometries from this layer
1580 scene_rdl2::rdl2::Layer::GeometrySet geometries;
1581 layer->getAllGeometries(geometries);
1582
1583 // Generate labeled geometry sets by looking at the shadow receiver label attribute on the geometry objects
1584 std::vector<LabeledGeometrySet> geoSets;
1585 for (const scene_rdl2::rdl2::Geometry* geometry : geometries) {
1586 // Check if geometry has a shadow receiver label
1587 const std::string label = geometry->getShadowReceiverLabel();
1588 if (label != "") {
1589 // It has a label. See if there's already a labeled geometry set with this label
1590 LabeledGeometrySet* pSet = findLabeledGeometrySet(label.c_str(), geoSets);
1591 if (pSet != nullptr) {
1592 // Found the label; add the geom to the corresponding set
1593 pSet->mGeometries.push_back(geometry);
1594 } else {
1595 // No existing labeled geometry set has this label. Create one
1596 geoSets.push_back( {{geometry}, label} );
1597 }
1598 }
1599 }
1600
1601 // Now map shadow casters to receiver sets
1602 for (const scene_rdl2::rdl2::Geometry* geometry : geometries) {
1603
1604 // Construct the mappings described by the shadow_exclusion_mappings geometry string attribute.
1605 std::vector<ShadowExclusionMapping> mappings;
1606 std::string mappingsString = const_cast<std::string&>(geometry->getShadowExclusionMappings());
1607
1608 // Tokenise mappings string into entries each of the form partsString:labelsString where:
1609 // partsString is a comma-separated list of part names, or "*" to map the entire geometry
1610 // labelsString is a comma-separated list of receiver geometry set labels, or "*" for all set labels
1611 char* context;
1612 char* entryString = strtok_r(const_cast<char*>(mappingsString.c_str()), " \t\n", &context);
1613 while (entryString) {
1614 // Split the entry string into part names and receiver set labels
1615 int pos = std::string(entryString).find(':');
1616 if (pos != std::string::npos) {
1617 entryString[pos] = '\0';
1618 char* partsString = entryString;
1619 char* labelsString = &entryString[pos+1];
1620
1621 // Map listed parts to listed receiver sets
1622 ShadowExclusionMapping mapping;
1623 parsePartNames(partsString, mapping, geometry, layer);
1624 if (parseSetLabels(labelsString, mapping, geoSets)) {
1625 mappings.push_back(mapping);
1626 }
1627 } else {
1628 geometry->warn(
1629 "Missing colon in shadow_exclusion_mappings entry.");
1630 }
1631 entryString = strtok_r(NULL, " \t\n", &context);
1632 }
1633
1634 // Set up the shadow linking info
1635 const geom::Procedural* procedural = geometry->getProcedural();
1636 if (procedural) {
1637 ShadowLinkingSetter shadowLinkingSetter(layer, mappings);
1638 const_cast<geom::Procedural *>(procedural)->forEachPrimitive(shadowLinkingSetter);
1639 }
1640 }
1641}
