33/* Statistics collects stats during rendering.
34 * It has an array of counters, where each counter
35 * is a different statistic. It is not thread safe, so
36 * this should be used only in a per thread object.
37 */
38class Statistics
39{
40public:
41 /// Constructor
42 Statistics() {
43 reset();
44 }
45
46 /// Reset the stats
47 void reset() {
48 for (unsigned i = 0; i < NUM_STATS_COUNTERS; ++i) {
49 mCounters[i] = 0;
50 }
51 }
52
53 Statistics &operator += (Statistics const &rhs) {
54 for (unsigned i = 0; i < NUM_STATS_COUNTERS; ++i) {
55 mCounters[i] += rhs.mCounters[i];
56 }
57 return *this;
58 }
59
60 uint64_t getCounter(unsigned counter) const
61 {
62 MNRY_ASSERT(counter < NUM_STATS_COUNTERS);
63 return mCounters[counter];
64 }
65
66 void incCounter(unsigned counter)
67 {
68 MNRY_ASSERT(counter < NUM_STATS_COUNTERS);
69 ++mCounters[counter];
70 }
71
72private:
73 uint64_t mCounters[NUM_STATS_COUNTERS];
74};
