48void
49UsdPrimvarReader_point::update()
50{
51 if (hasChanged(attrVarName)) {
52 mOptionalAttributes.clear();
53 const std::string& varname = get(attrVarName);
54 // We handle these special case "varname" values:
55 // "P"
56 // "surface_P"
57 if (varname == "P") {
58 mIspc.mAttributeMapType = ispc::PRIMVAR_MAP_TYPE_P;
59 } else if (varname == "surface_P") {
60 mIspc.mAttributeMapType = ispc::PRIMVAR_MAP_TYPE_SURFACE_P;
61 moonray::shading::TypedAttributeKey<Vec3f> attributeKey("surface_P");
62 mIspc.mPrimitiveAttributeIndex = attributeKey;
63 mOptionalAttributes.push_back(attributeKey);
64 } else {
65 // Not a special case so create key from the
66 // "varname" value
67 mIspc.mAttributeMapType = ispc::PRIMVAR_MAP_TYPE_PRIMITIVE_ATTRIBUTE;
68 moonray::shading::TypedAttributeKey<Vec3f> attributeKey(varname);
69 mIspc.mPrimitiveAttributeIndex = attributeKey;
70 mOptionalAttributes.push_back(attributeKey);
71 }
72 ::moonshine::primvar::createLogEvent("vec3f",
73 varname,
74 mIspc.mMissingAttributeEvent,
75 mLogEventRegistry);
76 }
77}

79void
80UsdPrimvarReader_point::sample(const scene_rdl2::rdl2::Map *self, moonray::shading::TLState *tls,
81 const moonray::shading::State &state, Color *sample)
82{
83 const UsdPrimvarReader_point* me = static_cast<const UsdPrimvarReader_point*>(self);
84
85 const int attributeMapType = me->mIspc.mAttributeMapType;
86 if (attributeMapType == ispc::PRIMVAR_MAP_TYPE_P) {
87 const Vec3f& p = state.getP();
88 sample->r = p.x;
89 sample->g = p.y;
90 sample->b = p.z;
91 } else {
92 int key = me->mIspc.mPrimitiveAttributeIndex;
93 if (state.isProvided(key)) {
94 const Vec3f& p = state.getAttribute(moonray::shading::TypedAttributeKey<Vec3f>(key));
95 sample->r = p.x;
96 sample->g = p.y;
97 sample->b = p.z;
98 } else {
99 // the primitive attribute is unavailable, use fallback parameter
100 const Vec3f& p = evalVec3f(me, attrFallback, tls, state);
101 sample->r = p.x;
102 sample->g = p.y;
103 sample->b = p.z;
104 if (me->get(attrWarnWhenUnavailable)) {
105 moonray::shading::logEvent(me, tls, me->mIspc.mMissingAttributeEvent);
106 }
107 }
108 }
109}
