174 // If this mesh is bound to a volume shader with a map binding, and the
175 // mesh exhibits motion blur, we must bake the velocity grid so that the
176 // volume shader can be blurred with the mesh.
177 // We do this by using openvdb's mesh to volume conversion tool. The mesh
178 // is a shell and openvdb fills in the voxel grid bound by that shell.
179 // The grid can expand the shell by an exterior bandwith (the bandwith is
180 // number of voxel thick) and expand into the interior of the grid by an
181 // interior bandwidth. A bandwidth of 1 means only the voxels on the surface
182 // of the mesh are active. We want an interior bandwidth such that the entire
183 // interior voxel grid is active.
184 // @param interiorBandwidth : width of the grid in voxels / 2.

274 template<typename T> void
275 computeDerivatives(shading::TypedAttributeKey<T> key,
276 const T& f1, const T& f2, const T& f3,
277 const std::array<float, 4>& invA,
278 shading::Intersection& intersection) const
279 {
280 T df0 = f2 - f1;
281 T df1 = f3 - f1;
282 T dfds = invA[0] * df0 + invA[1] * df1;
283 T dfdt = invA[2] * df0 + invA[3] * df1;
284 intersection.setdAttributeds(key, dfds);
285 intersection.setdAttributedt(key, dfdt);
286 }

386// solve the 2x2 system Ax = b where A is [st2 - st1, st3 - st1],
387// x is the differential we are looking for and b is [f2 - f1, f3 - f1]
388// we can reuse the A^-1 for arbitrary attriutes derivatives calculation
389finline bool
390computeStInverse(const Vec2f& st1, const Vec2f& st2, const Vec2f& st3,
391 std::array<float, 4>& invA)
392{
393 std::array<float , 4> A = {
394 st2[0] - st1[0], st2[1] - st1[1],
395 st3[0] - st1[0], st3[1] - st1[1]};
396 float detA = A[0] * A[3] - A[1] * A[2];
397 if (!std::isnormal(detA)) {
398 return false;
399 }
400 float invDet = 1.0f / detA;
401 invA = {invDet * A[3], invDet * -A[1],
402 invDet * -A[2], invDet * A[0]};
403 return true;
404}
