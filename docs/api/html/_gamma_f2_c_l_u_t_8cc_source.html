217// static function
218unsigned
219GammaF2CLUT::calcTblId(const float f)
220{
221 union Uni {
222 float f;
223 unsigned u;
224 };
225 const Uni *uni = reinterpret_cast<const Uni *>(&f);
226
227 unsigned expMask = (uni->u >> 23) & 0xff;
228 unsigned manMaskShifted = (uni->u >> 16);
229
230 return calcTblId(expMask, manMaskShifted);
231}

233// static function
234unsigned
235GammaF2CLUT::calcTblId(const unsigned expMask, const unsigned manMaskShifted)
236//
237// This function computes id for lookup table which created by GammaF2CLUT::tblGen()
238//
239{
240 //
241 // -- High speed float value to gamma 2.2 8bit conversion by lookup table --
242 //
243 // Regarding to the computation of gamma correction and quantized to 8bit range by lookup table,
244 // I directory generate lookup table index from input float value's bit pattern.
245 //
246 // First of all, we consider following 4 values (f1 ~ f4).
247 //
248 // -- IEEE singleFloat bit pattern --
249 // s| exp | mantissa expVal | gamma255
250 // f1 = 0.000005077049536 : 0 01101101 01010100101101110000110 => 109 0
251 // f2 = 0.000005077049991 : 0 01101101 01010100101101110000111 => 109 1
252 // f3 = 0.999999940395355 : 0 01111110 11111111111111111111111 => 126 254
253 // f4 = 1.000000000000000 : 0 01111111 00000000000000000000000 => 127 255
254 // ^|<---+-->|<----------+---------->|
255 // | | |
256 // | | +-- mantissaBit
257 // | +- exponentBit
258 // +-- signeBit
259 //
260 // f1 is maxinum single float value which converted to 0 after gamma/8bit conversion.
261 // f2 is minimum single float value which converted to 1.
262 // f3 is maxinum single float value which converted to 254.
263 // f4 is minimum single float value which converted to 255.
264 //
265 // This means all the float value equal or smaller than f1 should be converted to 0. And
266 // all the float value equal or bigger than f4 should be converted to 255. We only need to consider between f2 to f3.
267 // (There is no float value between f1 and f2 also f3 and f4).
268 //
269 // Exponent value range for f2 ~ f3 is 109(=0x6d,=01101101) ~ 126(=0x7e,=01111110).
270 // In order to keep all range from 109 to 126, we need at least 5bit. And 5bit can store value range from 109 to 140.
271 // This can save entire range from f1 to f4 (We only use 109 to 126. not using 127 to 140 actually).
272 //
273 // For mantissa, basically we don't need all 23bit of precision. After several tests, I decided to use
274 // top left 7bit for mantissa and final tableId bit size is 12bit.
275 //
276 // expPart manPart
277 // tableId (12bit) = 1 0 9 8 7 6 5 4 3 2 1 0
278 // |<---+--->|<-----+----->|
279 // | |
280 // | +-- from mantissaBit : 7bit 0~127 : top left 7bit of single float mantissa
281 // |
282 // +-- from exponentBit : 5bit 0~31 : exponent range from 109(=0x60) to 140(=0x8c)
283 // But only use from 109 to 126
284 //
285 // Following is a table index and it's input float value and output gamma255 table value (0~255)
286 //
287 // tableId | expPart | manPart | expMask | manMask | inputF | g255
288 // 0 0 0 109 0x000000 0.0000038147 0
289 // 1 0 1 109 0x010000 0.0000038445 0
290 // 2 0 2 109 0x020000 0.0000038743 0
291 // 3 0 3 109 0x030000 0.0000039041 0
292 // 4 0 4 109 0x040000 0.0000039339 0
293 // 5 0 5 109 0x050000 0.0000039637 0
294 // 6 0 6 109 0x060000 0.0000039935 0
295 // ...
296 // 2298 17 122 126 0x7a0000 0.9765625000 252
297 // 2299 17 123 126 0x7b0000 0.9804687500 252
298 // 2300 17 124 126 0x7c0000 0.9843750000 253
299 // 2301 17 125 126 0x7d0000 0.9882812500 253
300 // 2302 17 126 126 0x7e0000 0.9921875000 254
301 // 2303 17 127 126 0x7f0000 0.9960937500 254
302 // 2304 18 0 127 0x000000 1.0000000000 255
303 //
304 // expPart is a value of exponent part of tableId. expPart = (tableId >> 7) & 0x1f;
305 // manPart is a value of mantissa part of tableId. manPart = tableId & 0x7f;
306 // expMask is input value's exponent bitMask based on expPart. expMask = expPart + 109
307 // manMask is input value's mantissa bitMask based on manPart. manMask = manPart << 16
308 // inputF is reconstruct float value based on tableId. inputF = (expMask << 23) | manMask
309 // g255 is gamma corrected 8bit value of inputF g255 = pow(inputF, 1.0/2.2) * 255
310 //
311 // tableId is 12bit and can support 4096 table size. However, we only use 0 ~ 2304.
312 // (max exponent is 126 and exp(126) w/ man(0x7f0000 = full mantissa bitMask) goes to tblId 2303.
313 // But we need index 2304 as to get max value (i.e. 255). This is why max tbl id is 2304 instead of 2303)
314 //
315 // Following logic is a IEEE single float to tablId direct conversion logic
316 //
317 // int convFloatToTblId(const float f)
318 // {
319 // union Uni {
320 // float f;
321 // unsigned u;
322 // };
323 // const Uni *uni = (const Uni *)&f;
324 // unsigned expMask = (uni->u >> 23) & 0xff; // ... (a)
325 // unsigned manMaskShifted = (uni->u >> 16) & 0x7f; // ... (b)
326 //
327 // unsigned tblId = 0;
328 // if (expMask < 109) {
329 // tblId = 0; // ... (c)
330 // } else if (expMask < 127) {
331 // tblId = (((expMask - 109) & 0x1f) << 7) | manMaskShifted; // ... (d)
332 // } else {
333 // tblId = 2304; // ... (e)
334 // }
335 // return tblId;
336 // }
337 //
338 // (a) : expMask is bitmask pattern of exponent : 8bit
339 // (b) : manMaskShifted is shifted bitmask pattern of mantissa (only consider top left 7bit of mantissa bit)
340 // (c) : if exponent value is less than 109, pick tableId as 0 (i.e. final out value is 0)
341 // (d) : if exponent value is between 109~127, generated tblId is 0~2303
342 // (e) : if exponent value is equal or bigger than 127, tableId is 2304.
343 //
344 // Let's remove branch flow by bitMask operation
345 //
346 /* naive version
347 unsigned tblId = 0;
348 if (expMask < 109) {
349 tblId = 0;
350 } else if (expMask < 127) {
351 tblId = (((expMask - 109) & 0x1f) << 7) | (manMaskShifted & 0x7f);
352 } else {
353 tblId = 2304;
354 }
355 return tblId;
356 */
357
358 int tblId = ((((expMask - 109) & 0x1f) << 7) | (manMaskShifted & 0x7f)); // ... (A)
359
360 int expMaxMask = (static_cast<int>(expMask) - 127) >> 31; // ... (B)
361 int tblId2 = (tblId & expMaxMask) | (2304 & ~expMaxMask); // ... (C)
362
363 int expMinMask = ~((static_cast<int>(expMask) - 109) >> 31); // ... (D)
364 int tblId3 = tblId2 & expMinMask; // ... (E)
365
366 return tblId3;
367
368 //
369 // (A) : calculate base tblId based on exponent range (start 109 & 5bit) and left 7 bit of mantissa
370 // (B) : mask for exponent max (127) : expMaxMask = (expMask >= 127)? 0x0: 0xffffffff
371 // (C) : tblId (exponent 109~) is now clipped to 0 ~ 2304.
372 // (D) : mask for exponent min (109) : expMinMask = (expMask < 109)? 0x0: 0xffffffff
373 // (E) : tblId3 (all exponent) is now clipped to 0 ~ 2304. This is result tblId
374 //
375}

377// static function
378unsigned
379GammaF2CLUT::calcTblId2(const unsigned manMaskShifted)
380//
381// This is a optimized version of GammaF2CLUT::calcTblId().
382// Performance is slightly faster but potentially still we need 2 if judgements.
383// Basic idea of this enhancement was suggestion by Mike Day. Thanks Mike!
384// (See detail : http://github.anim.dreamworks.com/arras/moonray/pull/224)
385// Toshi (Feb/28/2018)
386//
387{
388 //
389 // Basic logic and idea is same as GammaF2CLUT::calcTblId().
390 // We processed expPart and manPart separately on GammaF2CLUT::calcTblId() but
391 // we can process together because these are 12 consecutive bits.
392 //
393 // This is a sudo code to compute tblId
394 //
395 // unsigned u = uni->u >> 16
396 // u = clamp(u, 0x36aa, 0x3f80) ... (a)
397 // int tblId = u - 0x3680 ... (b)
398 //
399 // Current LUT start from float value = (expPart=109, manPart=0).
400 // 16bit right shifted value of this float is 00110110 10000000 = 0x3680.
401 // This is why (b) subtracts 0x3680
402 //
403 // (a) needs to clamp by 0x36aa and 0x3f80 because.
404 // 0x36aa is a 16bit right shifted value of f1 value (00110110 10101010 = 0x36aa)
405 // 0x3f80 is a 16bit right shifted value of f4 value (00111111 10000000 = 0x3f80)
406 //
407 // Actually first 43 entries of LUT which generated by GammaF2CLUT::calcTblId() is ZERO
408 // So we can reduce the table size a bit and accessed by following sudo code.
409 //
410 // unsigned u = uni->u >> 16
411 // u = clamp(u, 0x36aa, 0x3f80)
412 // int tblId = u - 0x36aa ... (c)
413 //
414 // We changed offset of LUT and now we should substruct 0x36aa at (c).
415 //
416 // Following code is for reduce LUT size version (i.e. You should remove first 42 entries
417 // from LUT which generated by GammaF2CLUT::tblGen() to use this function).
418 //
419 // -- Performance --
420 // Based on my test and compare with GammaF2CLUT::calcTblId(), this calcTblid2() is slightly faster.
421 // untile w/ gamma correction + 8bit quantize test
422 // calcTblId() : 9.5~10.0 ms range
423 // calcTblId2() : 8.5~9.3 ms range
424 // Speed itself is totally depend on test data itself but I could find calcTblId2() is relatively faster.
425 // Toshi (Feb/28/2018)
426 //
427
428 int tblId = manMaskShifted - 0x36aa;
429 if (tblId < 0) {
430 tblId = 0;
431 } else if (tblId > 2262) {
432 tblId = 2262;
433 }
434
435 //
436 // This is a convert if branch by bit mask operation but somehow this code is slower than
437 // naive c++ if coding. (compiler is very smart :-)
438 //
439 /*
440 int tblId1 = manMaskShifted - 0x36aa;
441
442 int minMask = ~(tblId1 >> 31);
443 int tblId2 = tblId1 & minMask;
444
445 int maxMask = (tblId1 - 2262) >> 31;
446 int tblId = (tblId2 & maxMask) | (2262 & ~maxMask);
447 */
448
449 return tblId;
450}

452// static function
453unsigned
454GammaF2CLUT::calcTblId3(const unsigned manMaskShifted)
455//
456// This id computation function is based on different ideas and
457// used special LUT which generated by GammaF2CLUT::tblGen15bit()
458// Main idea of this id computation and LUT is reduce branch at id computation stage.
459// This is trade off of table size and id computation complexity. Now GammaF2CLUT::tblGen15bit()
460// creates 32768 bytes (=32KByte) table. However id computation does not include
461// any branch operation and just use 16bit shifted value as tblId.
462// Toshi (Mar/01/2018)
463//
464{
465 // -- Performance --
466 // Compare performance between 2 of them.
467 // untile w/ gamma correction + 8bit quantize test
468 // calcTblId() : 9.5~10.0 ms range
469 // calcTblId2() : 8.5~9.3 ms range
470 // calcTblId3() : 7.0~7.2 ms range
471 // Speed itself is totally depend on test data itself but I could find calcTblId3() is fastest.
472 // Toshi (Mar/01/2018)
473 //
474 return manMaskShifted; // uni->u >> 16
475}

559// static function
560std::string
561GammaF2CLUT::tblGen15bit()
562{
563 std::ostringstream ostr;
564
565 int max = 32768;
566
567 ostr << "unsigned char f2g255[" << max << "] = {\n"; // You may change array name if you want
568
569 GammaF2CLUT gLUT;
570 bool topST = true;
571 for (int tblId = 0; tblId < max; ++tblId) {
572 if (topST) {
573 ostr << " /* tblId:" << std::setw(5) << tblId << " */ ";
574 }
575
576 int exponent = (tblId >> 7) & 0xff;
577 int mantissa = (tblId & 0x7f) << 16;
578 gLUT.set(0, exponent, mantissa); // re-construct float value based on exponent and mantissa
579
580 int g255;
581 if (tblId > 32640) {
582 // tblId > 32640 : nan
583 g255 = 0; // we can not compute value -> just set 0
584 } else if (tblId == 32640) {
585 // tblId = 32640 : inf
586 g255 = 255;
587 } else if (tblId >= 16256) {
588 // getF() is more than 1.0
589 g255 = 255;
590 } else {
591 g255 = static_cast<int>(pow(gLUT.getF(), 1.0/2.2) * 255);
592 if (g255 > 255) g255 = 255; // just in case
593 }
594 ostr << std::setw(3) << g255;
595
596 if (tblId != (max - 1)) ostr << ',';
597
598 topST = false;
599 if (!((tblId + 1) % 16)) {
600 ostr << '\n';
601 topST = true;
602 }
603 }
604 ostr << "};";
605 return ostr.str();
606}

629double
630GammaF2CLUT::calcExponentVal() const
631{

637double
