18//
19// --- How to implement your own debug console by using DebugConsoleDriver ---
20//
21// Step-1) Create your consoleDriver class derived from DebugConsoleDriver class
22// example
23// moonray::rndr::RenderContextConsoleDriver
24// mcrt_dataio::ClientReceiverConsoleDriver
25//
26// Step-2) Override parserConfigure() function to configure commands
27// Default DebugConsoleDriver does not have any command configuration.
28// You need to configure your own command into parser object.
29// Example would be found in the following implementations
30// moonray/rndr::RenderContextConsoleDriver::parserConfigure()
31// mcrt_dataio::ClientReceiverConsoleDriver::parserConfigure()
32//
33// Step-3) Call DebugConsoleDriver::initialize() with proper port value.
34// This method boots console thread and open socket port for incoming telnet-connection.
35// Example would be found in the following implementations
36// moonray::rndr::RenderContextConsoleDriver::init()
37// mcrt_dataio::ClientReceiverFb::consoleEnable()
38//
39
40namespace scene_rdl2 {
41namespace grid_util {
42
43class DebugConsoleDriver
44//
45// This class provides debug console features to the process and we can connect process
46// via telnet connection and execute several different command line commands by hand.
47// This functionality is a big help to test the internal functionality from the outside process.
48// (For example, this is heavily used for interactive debugging command-line control especially
49// arras multi-machine configurations.)
50//
51// This class boots an independent thread in order to charge a debug console operation inside
52// the initialize(). If we don't have any incoming socket connection, this debug console threads
53// almost always asleep and minimizes the CPU overhead. This debug console thread is automatically
54// shut down inside the destructor.
55//
56{
57public:
58 enum class ThreadState : int { INIT, IDLE, BUSY, DONE };
59
60 DebugConsoleDriver() :
61 mThreadState(ThreadState::INIT),
62 mThreadShutdown(false),
63 mTlSvrPortNum(0)
64 {}
65 virtual ~DebugConsoleDriver();
66
67 void initialize(unsigned short port); // If you set port as 0, kernel find available port for you.
68 int getPort() const { return mTlSvrPortNum; } // return opened port num. 0 is error and failed to open
69
70 // Non-copyable
71 DebugConsoleDriver &operator =(const DebugConsoleDriver &) = delete;
72 DebugConsoleDriver(const DebugConsoleDriver &) = delete;
73
74 Parser & getRootParser() { return mParser; }
75
76 void showString(const std::string &msg); // msg goes to TlSvr if available
77
78private:
79 static void threadMain(DebugConsoleDriver *driver);
80
81 virtual void parserConfigure(Parser &) {} // You should implement this for adding your command to the parser object
82
83 //------------------------------
84
85 std::thread mThread;
86 std::atomic<ThreadState> mThreadState;
87 std::atomic<bool> mThreadShutdown;
88
89 mutable std::mutex mMutexBoot;
90 std::condition_variable mCvBoot; // using at boot threadMain sequence
91
92 TlSvr mTlSvr; // for telnet server operation
93 int mTlSvrPortNum; // 0 is error
94
95 //------------------------------
96
97 Parser mParser; // root parser object : all command definitions for the incoming command line.
98};
99
100} // namespace grid_util
101} // namespace scene_rdl2
