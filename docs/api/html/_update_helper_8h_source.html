18/**
19 * A helper class for scene objects updating.
20 *
21 * Updating of all the objects in the scene is a two-stage process starts
22 * in applyUpdates() function in SceneContext.cc
23 *
24 * 1. Walk through the object directed acyclic graphs (DAG) serially in
25 * depth first order to decide which objects need to be updated and
26 * decide the order of the updates. The order of updates is maintained
27 * in a graph-depth-based data structure. If there are multiple paths
28 * reaching to the same object, the deepest level depth is recorded.
29 *
30 * Check updatePrep() function in SceneObject.h for more details
31 *
32 * 2. Call update() on all objects which need update level by level from
33 * the deepest level to shallow ones. For the objects which have the same
34 * depth, update() will be called in parallel.
35 *
36 * 3. Leaves in DAG are the nodes which do not have any dependencies. Leaves
37 * are treated seperately here. All leaves can be updated in parallel before
38 * other nodes in DAG.
39 *
40 * Definition of depth of a level
41 * -2 : not found, hasn't been recorded
42 * -1 : leaf
43 * >=0: depth in DAG, root(starting point) is 0
44 *
45 * An example
46 *
47 * A
48 * / \
49 * B \
50 * / \
51 * C D
52 * \ / \
53 * \ / F
54 * \ /
55 * E
56 * /
57 * G
58 *
59 * leaves depth -1: G, F
60 * depth 0: A
61 * depth 1: B, D
62 * depth 2: C
63 * depth 3: E, notice here even from A->D->E we can get depth(E) = 2, we need
64 * to record the deepest depth
65 *
66 */
67

166void UpdateHelper::insert(SceneObject* const &obj, int depth) {
167 MNRY_ASSERT(depth >= 0, "dag depth starts from 0");
168
169 int recordedDepth = getDepth(obj);
170 MNRY_ASSERT(recordedDepth != -1, "this object has been inserted as a leaf");
171
172 if (recordedDepth >= depth) {
173 return;
174 } else {
175 // this object has been recorded before
176 if (recordedDepth >= 0) { // !=-2
177 mDagLevels[recordedDepth].erase(obj);
178 }
179 if (mDagLevels.size() <= (size_t) depth) {
180 mDagLevels.resize(depth+1);
181 }
182 mDagLevels[depth].insert(obj);
183 mDepthMap[obj] = depth;
184 }
185}

187void UpdateHelper::insertLeaf(SceneObject* const &obj) {
188 int recordedDepth = getDepth(obj);
189 // a leaf needs to be either not recorded or recorded as leaf before
190 MNRY_ASSERT(recordedDepth < 0, "conflict when inserting leaf");
191
192 // this node has been recorded as leaf
193 if (recordedDepth == -1) {
194 return;
195 } else {
196 mDagLeaves.push_back(obj);
197 mDepthMap[obj] = -1;
198 }
199}
