94 // insert an object to mDagLevels. if this object has already been inserted
95 // before, we compare the current depth and the depth recorded before. if

105 // get depth of a certain object. return -1 if object is a leaf, return -2
106 // if object hasn't been recorded before, otherwise return the depth in DAG
107 // starting from 0
108 int getDepth (SceneObject* const &obj) const

110 auto it = mDepthMap.find(obj);
111 return (it == mDepthMap.end())?
112 -2 : (*it).second;

166void UpdateHelper::insert(SceneObject* const &obj, int depth) {
167 MNRY_ASSERT(depth >= 0, "dag depth starts from 0");
168
169 int recordedDepth = getDepth(obj);
170 MNRY_ASSERT(recordedDepth != -1, "this object has been inserted as a leaf");
171
172 if (recordedDepth >= depth) {
173 return;
174 } else {
175 // this object has been recorded before
176 if (recordedDepth >= 0) { // !=-2
177 mDagLevels[recordedDepth].erase(obj);
178 }
179 if (mDagLevels.size() <= (size_t) depth) {
180 mDagLevels.resize(depth+1);
181 }
182 mDagLevels[depth].insert(obj);
183 mDepthMap[obj] = depth;
184 }
185}

187void UpdateHelper::insertLeaf(SceneObject* const &obj) {
188 int recordedDepth = getDepth(obj);
189 // a leaf needs to be either not recorded or recorded as leaf before
190 MNRY_ASSERT(recordedDepth < 0, "conflict when inserting leaf");
191
192 // this node has been recorded as leaf
193 if (recordedDepth == -1) {
194 return;
195 } else {
196 mDagLeaves.push_back(obj);
197 mDepthMap[obj] = -1;
198 }
199}
