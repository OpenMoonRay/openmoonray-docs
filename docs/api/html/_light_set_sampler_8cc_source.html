42LightSetSampler::LightSetSampler(scene_rdl2::alloc::Arena *arena, const LightSet &lightSet, const shading::Bsdf &bsdf,
43 const Vec3f &p, int maxSamplesPerLight) :
44 mLightSet(lightSet),
45 mMaxSamplesPerLight(maxSamplesPerLight),
46 mSampleCount(0),
47 mInvSampleCount(0),
48 mLightSampleCount(0),
49 mInvLightSampleCount(0),
50 mBsdf(&bsdf)
51{
52 const int lightCount = mLightSet.getLightCount();
53
54 mLightSampleCount = computeLightSampleCount(mMaxSamplesPerLight);
55 mInvLightSampleCount = rcp(float(mLightSampleCount));
56
57 mSampleCount = lightCount * mLightSampleCount;
58 mInvSampleCount = (mSampleCount > 0 ? rcp(float(mSampleCount)) : 0);
59}

69void
70LightSetSampler::sampleIntersectAndEval(mcrt_common::ThreadLocalState* tls,
71 const Light* light, const LightFilterList* lightFilterList,
72 const Vec3f &P, const Vec3f *N, const LightFilterRandomValues& filterR, float time,
73 const Vec3f& r, LightSample &sample, float rayDirFootprint) const
74{
75 // Draw a sample from light
76 LightIntersection isect;
77 if (!light->sample(P, N, time, r, sample.wi, isect, rayDirFootprint)) {
78 sample.setInvalid();
79 return;
80 }
81 sample.distance = isect.distance;
82
83 // Evaluate light sample
84 sample.Li = light->eval(tls, sample.wi, P, filterR, time, isect, false, lightFilterList, rayDirFootprint,
85 &sample.pdf);
86 if (isSampleInvalid(sample.Li, sample.pdf)) {
87 sample.setInvalid();
88 }
89}
