38// want it parallelized or not.
39template <typename T, typename FUNC>
40finline void
41simpleLoop(bool parallel, T start, T end, const FUNC &func)
42{
43 if (parallel) {
44 tbb::parallel_for (start, end, [&](unsigned iter) {
45 func(iter);
46 });
47 } else {
48 for (unsigned iter = start; iter != end; ++iter) {
49 func(iter);
50 }

246// custom functor.
247template<typename DEST_PIXEL_TYPE, typename SRC_PIXEL_TYPE, typename BODY>
248finline void
249processPixelBuffer(PixelBuffer<DEST_PIXEL_TYPE>& destBuffer, const PixelBuffer<SRC_PIXEL_TYPE>& srcBuffer,
250 BODY const &body, bool parallel)
251{
252 unsigned w = srcBuffer.getWidth();
253 unsigned h = srcBuffer.getHeight();
254
255 destBuffer.init(w, h);
256
257 simpleLoop(parallel, 0u, h, [&](unsigned y) {
258
259 DEST_PIXEL_TYPE *dst = destBuffer.getRow(y);
260 const SRC_PIXEL_TYPE *src = srcBuffer.getRow(y);
261
262 for (unsigned x = 0; x < w; ++x) {
263 body(dst, *src, x, y);
264 ++dst;
265 ++src;
266 }
267 });
268}

299void
300computeNormalizedScaleAndOffset(Vec3f &offset, float &scale, const SRC_BUFFER_TYPE &srcBuffer)
301{
302 offset = Vec3f(0.f);
303 scale = 1.f;
304
305 Vec3f min = Vec3f(std::numeric_limits<float>::max());
306 Vec3f max = Vec3f(std::numeric_limits<float>::min());
307
308 const unsigned int w = srcBuffer.getWidth();
309 const unsigned int h = srcBuffer.getHeight();
310 for (unsigned int y = 0; y < h; ++y) {
311 for (unsigned int x = 0; x < w; ++x) {
312 const auto &p = srcBuffer.getPixel(x, y);
313 if (math::isfinite(p[0])) {
314 if (p[0] < min[0]) min[0] = p[0];
315 if (p[0] > max[0]) max[0] = p[0];
316 }
317 if (math::isfinite(p[1])) {
318 if (p[1] < min[1]) min[1] = p[1];
319 if (p[1] > max[1]) max[1] = p[1];
320 }
321 if (math::isfinite(p[2])) {
322 if (p[2] < min[2]) min[2] = p[2];
323 if (p[2] > max[2]) max[2] = p[2];
324 }
325 }
326 }
327 const Vec3f diff = max - min;
328 const float maxDiff = std::max(diff[0], std::max(diff[1], diff[2]));
329 if (maxDiff > MIN_NORMALIZED_DISTANCE) {
330 scale = 1.f / maxDiff;
331 }
332 offset = -min;
333}
