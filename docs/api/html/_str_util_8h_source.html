66inline
67std::string
68addIndent(const std::string &str, const int indentTotal = 1)
69//
70// str should not end by '\n'.
71// It works without crash if you use string which terminated by '\n' but not recommended.
72// Last '\n' (new line) control should be done by caller function level.
73// General concensus of using addIndent() requires input string which is not terminated by '\n' (new line).
74//
75{
76 std::string indentStr(indentTotal * 2, ' ');
77
78 int totalNewLine = 0;
79 for (size_t i = 0; i < str.size(); ++i) {
80 if (str[i] == '\n') ++totalNewLine;
81 }
82
83 std::string newStr;
84 newStr.resize(str.size() + (totalNewLine + 1) * indentStr.size());
85 newStr = indentStr;
86 for (size_t i = 0; i < str.size(); ++i) {
87 newStr.push_back(str[i]);
88 if (str[i] == '\n') newStr += indentStr;
89 }
90 return newStr;
91}

103inline
104std::string
105byteStr(const size_t numByte)
106{
107 std::ostringstream ostr;
108 if (numByte < (size_t)1024) {
109 ostr << numByte << " Byte";
110 } else if (numByte < (size_t)1024 * (size_t)1024) {
111 float f = (float)numByte / 1024.0f;
112 ostr << std::setw(3) << std::fixed << std::setprecision(2) << f << " KByte";
113 } else if (numByte < (size_t)1024 * (size_t)1024 * (size_t)1024) {
114 float f = (float)numByte / (1024.0f * 1024.0f);
115 ostr << std::setw(3) << std::fixed << std::setprecision(2) << f << " MByte";
116 } else {
117 float f = (float)numByte / (1024.0f * 1024.0f * 1024.0f);
118 ostr << std::setw(3) << std::fixed << std::setprecision(2) << f << " GByte";
119 }
120 return ostr.str();
121}

123inline
124std::string
125secStr(const float sec)
126{
127 float roundedSec;
128 std::ostringstream ostr;
129
130 // In order to displaying 1.000 sec instead of 1000.00 ms, we use rounded logic
131 roundedSec = std::round(sec * 100000.0f) / 100000.0f;
132 if (roundedSec < 1.0f) {
133 float ms = roundedSec * 1000.0f;
134 ostr << std::setw(6) << std::fixed << std::setprecision(2) << ms << " ms";
135 return ostr.str();
136 }
137
138 // Without using a rounded sec value, we get 1 min 60.000 sec when the original sec is 119.9996f
139 // With using a rounded sec value, the result would be 2 min 0.000 sec
140 roundedSec = std::round(sec * 1000.0f) / 1000.0f;
141 if (roundedSec < 60.0f) {
142 ostr << std::setw(6) << std::fixed << std::setprecision(3) << roundedSec << " sec";
143 } else {
144 int m = static_cast<int>(roundedSec / 60.0f);
145 float s = roundedSec - static_cast<float>(m) * 60.0f;
146 ostr << m << " min " << std::setw(6) << std::fixed << std::setprecision(3) << s << " sec";
147 }
148 return ostr.str();
149}

166inline
167std::string
168trimBlank(const std::string &str)
169//
170// Remove visually blank char of first and last part of the specified string
171//
172{
173 static const std::string trimCharList = (const char *)" \t\v\r\n";
174
175 std::string result;
176 std::string::size_type left = str.find_first_not_of(trimCharList);
177 if (left != std::string::npos) {
178 std::string::size_type right = str.find_last_not_of(trimCharList);
179 result = str.substr(left, right - left + 1);
180 }
181 return result;
182}

200inline
201std::string
202replaceBlankToSingleSpace(const std::string &str)
203//
204// Replaces blank char to the single space
205//
206{
207 std::string result;
208
209 // first of all, multi spaces (i.e. blank) should be converted to single space
210 for (size_t i = 0; i < str.size(); ++i) {
211 char c = str[i];
212 if (std::isblank(c)) {
213 if (i > 0) {
214 if (!std::isblank(str[i-1])) result.push_back(' ');
215 }
216 } else {
217 result.push_back(c);
218 }
219 }
220
221 if (result.back() == '\n') result.pop_back(); // rm last newline
222 if (result.back() == ' ') result.pop_back(); // rm last space
223 return result;
224}
