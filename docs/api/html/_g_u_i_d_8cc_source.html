116GUID GUID::uuid4()
117{
118 // We're using random_device to fill in all of the randomness of the guid.
119 // There are two concerns with doing this.
120 // A) We're on a system that doesn't have a random device and we're just
121 // generating the same guid for everything. I'm going to assume this
122 // isn't happening. Unfortunately, on some standard libraries, the
123 // random_device::entropy call is not fully implemented, and so
124 // checking it gives us no information.
125 // B) random_device is blocking for entropy, and is therefore slow. In
126 // the gcc libstdc++, it either opens /dev/urandom (which won't
127 // block) or uses the CPU instruction rdrand.
128 // That said, why not just use it for seeding something like mt19937? If
129 // we seed with one 32-bit value, we only have 2^32 possible states in
130 // which to begin. This only gives us 2^32 guids. We only have to generate
131 // 78,000 guids for a 50% chance of a collision. We can augment this with
132 // more calls to random_device, but at what point to we just use it for
133 // everything?
134 std::random_device rd;
135 std::uniform_int_distribution<uint32_t> dist;
136
137 GUID guid;
138 uint32_t* const p = reinterpret_cast<uint32_t*>(&guid.f1);
139
140 for (int i = 0; i < 4; ++i) {
141 p[i] = dist(rd);
142 }
143
144 ///////////////////////////////////////////////////////////////////////////
145 // The following modifications to the guid are required to meet the uuid4
146 // standard (RFC 4122).
147 // https://en.wikipedia.org/wiki/Globally_unique_identifier#Algorithm
148 ///////////////////////////////////////////////////////////////////////////
149
150 // Make the most significant bits in field 3 "0100".
151 guid.f3 &= htobe16(0x0fffu);
152 guid.f3 |= htobe16(0x4000u);
153
154 // Make the most significant bits in field 4 "10".
155 // Even though the last field is treated as raw bytes, we have to be
156 // consistent in treating our internal representation. It's ultimately
157 // output as if big-endian (lowest address byte is output first).
158 guid.f4 &= htobe64(0x3fffffffffffffffull);
159 guid.f4 |= htobe64(0x8000000000000000ull);
160
161 return guid;
162}

164GUID::GUID(const std::string& in)
165{
166 if (!GUIDdetail::validUUIDString(in)) {
167 throw GUIDException("Invalid UUID string");
168 }
169
170 std::istringstream iss(in);
171 iss >> std::hex >> f1;
172 iss.ignore(); // '-'
173 iss >> std::hex >> f2;
174 iss.ignore(); // '-'
175 iss >> std::hex >> f3;
176 iss.ignore(); // '-'
177
178 char tmp[3] = { 0 };
179 for (int i = 0; i < 8; ++i) {
180 if (i == 2) {
181 iss.ignore(); // '-'
182 }
183 // Read two ascii characters (which make up one 8-bit byte).
184 tmp[0] = iss.get();
185 tmp[1] = iss.get();
186
187 // Convert the hex ascii to numerical data.
188 a4[i] = std::strtol(tmp, nullptr, 16);
189 }
190
191 // Convert the first three fields from big-endian. We don't do the same for
192 // the last field, since we read it a byte-at-a-time and the big-endian
193 // order is implicit.
194 f1 = be32toh(f1);
195 f2 = be16toh(f2);
196 f3 = be16toh(f3);
197}

199GUID GUID::littleEndian(uint8_t i0, uint8_t i1, uint8_t i2, uint8_t i3,
200 uint8_t i4, uint8_t i5, uint8_t i6, uint8_t i7,
201 uint8_t i8, uint8_t i9, uint8_t iA, uint8_t iB,
202 uint8_t iC, uint8_t iD, uint8_t iE, uint8_t iF)
203{
204 GUID guid;
205 // The first three fields need to be ultimately treated as big-endian. We'll
206 // fill the lowest-address bytes as big-endian.
207 guid.a1[0x0] = i3;
208 guid.a1[0x1] = i2;
209 guid.a1[0x2] = i1;
210 guid.a1[0x3] = i0;
211
212 guid.a2[0x0] = i5;
213 guid.a2[0x1] = i4;
214
215 guid.a3[0x0] = i7;
216 guid.a3[0x1] = i6;
217
218 // The last field is treated as raw bytes.
219 guid.a4[0x0] = i8;
220 guid.a4[0x1] = i9;
221
222 guid.a4[0x2] = iA;
223 guid.a4[0x3] = iB;
224 guid.a4[0x4] = iC;
225 guid.a4[0x5] = iD;
226 guid.a4[0x6] = iE;
227 guid.a4[0x7] = iF;
228
229 return guid;
230}

232std::string GUID::asString() const
233{
234 std::ostringstream oss;
235 oss.fill('0');
236 oss.setf(std::ios_base::right, std::ios::adjustfield);
237
238 oss << std::hex << std::setw(8) << htobe32(f1);
239 oss << '-';
240 oss << std::hex << std::setw(4) << htobe16(f2);
241 oss << '-';
242 oss << std::hex << std::setw(4) << htobe16(f3);
243 oss << '-';
244
245 for (int i = 0; i < 8; ++i) {
246 if (i == 2) {
247 oss << '-';
248 }
249 // Since we're outputting a byte at a time, there's no big-endian
250 // conversion here -- it's implicit.
251 oss << std::hex << std::setw(2) << static_cast<int>(a4[i]);
252 }
253
254 return oss.str();
255}
