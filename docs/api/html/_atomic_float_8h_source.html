37// In compare_exchange overloads where only one memory order is given, we have
38// to decide on the other. These are what is laid out by the standard.
39constexpr std::memory_order compare_exchange_duo(std::memory_order in) noexcept
40{
41 constexpr int mapping[6] = {
42 /* std::memory_order_relaxed -> */ static_cast<int>(std::memory_order_relaxed),
43 /* std::memory_order_consume -> */ static_cast<int>(std::memory_order_consume),
44 /* std::memory_order_acquire -> */ static_cast<int>(std::memory_order_acquire),
45 /* std::memory_order_release -> */ static_cast<int>(std::memory_order_relaxed),
46 /* std::memory_order_acq_rel -> */ static_cast<int>(std::memory_order_acquire),
47 /* std::memory_order_seq_cst -> */ static_cast<int>(std::memory_order_seq_cst)
48
49 };
50 return static_cast<std::memory_order>(mapping[static_cast<int>(in)]);
51}
