295// This generates a random point in N-dimensional space within distance [r, 2r]
296// from "basis".
297//
298// Here, 'o' is the basis point, and the filled area is where we're generating
299// points. 2D case represented, 'cause, stuff.
300//
301// xxxxxxx
302// xx.......xx
303// x...........x
304// x.............x
305// x...............x
306// x......xxxxx......x
307// x.....x x.....x
308// x.....x x.....x
309// x....x x....x
310// x....x x....x
311// x....x o x....x
312// x....x x....x
313// x....x x....x
314// x.....x x.....x
315// x.....x x.....x
316// x......xxxxx......x
317// x...............x
318// x.............x
319// x...........x
320// xx.......xx
321// xxxxxxx
322//
323// This can be done more efficiently in 2D space, but it uses rejection
324// sampling because it was easy (and fast enough) in ND space.
325template <unsigned D, typename RNGType>
326inline NPoint<D> generateRandomPoint(const NPoint<D>& basis, float r, RNGType& rng)
327{
328 std::uniform_real_distribution<float> dist(-2.0f*r, 2.0f*r);
329
330 // Rejection sampling.
331 // Loop until we get a valid sample.
332 for ( ; ; ) {
333 NPoint<D> p;
334 for (unsigned i = 0; i < D; ++i) {
335 const float rand = dist(rng);
336 p[i] = basis[i] + rand;
337
338 // These should only execute once at most (in theory), but there
339 // are some numerical precision issues.
340 while (p[i] < 0.0f) {
341 p[i] += 1.0f;
342 }
343 while (p[i] >= 1.0f) {
344 p[i] -= 1.0f;
345 }
346
347 assert(p[i] >= 0.0f);
348 assert(p[i] < 1.0f);
349 }
350 const float rSquared = r*r;
351 const float r2Squared = 2.0f*r * 2.0f*r;
352 const float wd2 = wrappedDistanceSquared(p, basis);
353
354 if (wd2 < r2Squared && wd2 > rSquared) {
355 return p;
356 }
357 }
358}

428inline NPoint<2> fromConcentricDisk(const NPoint<2>& p)
429{
430 float r;
431 float phi;
432 cartesianToPolar(p[0], p[1], r, phi);
433
434 if (phi < -pi/4.0f) {
435 // in range [-pi/4,7pi/4]
436 phi += 2.0f * pi;
437 }
438
439 float a;
440 float b;
441
442 if (phi < pi/4.0f) {
443 a = r;
444 b = phi * a / (pi/4.0f);
445 } else if (phi < 3.0f*pi/4.0f) {
446 b = r;
447 a = -(phi - pi/2.0f) * b / (pi/4.0f);
448 } else if (phi < 5.0f*pi/4.0f) {
449 a = -r;
450 b = (phi - pi)*a / (pi/4.0f);
451 } else {
452 b = -r;
453 a = -(phi - 3.0f*pi/2.0f) * b / (pi/4.0f);
454 }
455
456 const float x = (a + 1.0f) / 2.0f;
457 const float y = (b + 1.0f) / 2.0f;
458
459 return NPoint<2>({x, y});
460}

462inline NPoint<2> toConcentricDisk(const NPoint<2>& p)
463{
464
465 float a = 2.0f*p[0] - 1.0f;
466 float b = 2.0f*p[1] - 1.0f;
467
468 if (a == 0.0f && b == 0.0f) {
469 return NPoint<2>({0.0f, 0.0f});
470 }
471
472 float phi;
473 float r;
474
475 if (std::abs(a) > std::abs(b)) {
476 r = a;
477 phi = (pi/4.0f)*(b/a);
478 } else {
479 r = b;
480 phi = (pi/2.0f) - (pi/4.0f)*(a/b);
481 }
482
483 float x;
484 float y;
485 polarToCartesian(r, phi, x, y);
486 return NPoint<2>({x, y});
487}

489template <typename F>
490F wrapped1DDist(F a, F b)
491{
492 const auto d = std::abs(a - b);
493 if (d < static_cast<F>(0.5)) {
494 return d;
495 } else {
496 return static_cast<F>(1.0) - std::max(a, b) + std::min(a, b);
497 }

500template <typename P>
501auto wrappedDistanceSquared(const P& a, const P& b)
502{
503 typename P::InternalType d2 = 0.0;

505 const auto d = wrapped1DDist(a[i], b[i]);
506 d2 += d*d;

547class MinkowskiWrappedDistanceSquared
548{
549 float mP;
550
551public:
552 explicit MinkowskiWrappedDistanceSquared(unsigned count) :
553 // An exponent of 2 is Euclidian distance. We start with a value that
554 // favors diagonal distance, and move to Euclidian over time.
555 mP(std::min(2.0f, 0.5f + count/8.0f))
556 {
557 }
558
559 float operator()(const NPoint<2>& a, const NPoint<2>& b) const
560 {

562 for (unsigned i = 0; i < 2; ++i) {
563 const float d = wrapped1DDist(a[i], b[i]);
