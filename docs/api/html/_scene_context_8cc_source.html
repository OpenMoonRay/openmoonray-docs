62void
63verifyMatchingSceneClass(const std::string& className, const SceneObject* obj)
64{
65 MNRY_ASSERT(obj);
66 const std::string& existingClassName = obj->getSceneClass().getName();
67 if (className != existingClassName) {
68 throw except::TypeError(util::buildString("Cannot create new"
69 " SceneObject of SceneClass '", className, "' because '",
70 obj->getName(), "' of SceneClass '", existingClassName,
71 "' already exists."));
72 }
73}

122SceneContext::~SceneContext()
123{
124 // Delete all scene objects.
125 for (SceneObjectMap::iterator objIter = mSceneObjects.begin();
126 objIter != mSceneObjects.end(); ++objIter) {
127 // Call any on-delete callbacks first
128 for (auto cb : mDeleteCallbacks) {
129 cb(objIter->second);
130 }
131
132 Geometry *pGeom = dynamic_cast<Geometry *>(objIter->second);
133 if (pGeom)
134 pGeom->unloadProcedural();
135
136 delete objIter->second;
137 objIter->second = nullptr;
138 }
139
140 // Delete all scene classes.
141 for (SceneClassMap::iterator classIter = mSceneClasses.begin();
142 classIter != mSceneClasses.end(); ++classIter) {
143 delete classIter->second;
144 classIter->second = nullptr;
145 }
146}

187const rdl2::Camera*
188SceneContext::getPrimaryCamera() const
189{
190 // Prevent possible race condition with mCameras.push_back() in createSceneObject().
191 tbb::mutex::scoped_lock lock(mCreateSceneObjectMutex);
192
193 if (mCameras.size() == 0) {
194 return nullptr;
195 }
196
197 // The primary camera is the first element in the output cameras vector
198 if (mSceneVariables->get(SceneVariables::sCamera)) {
199 // Use the camera specified by the scene variable
200 return mSceneVariables->get(SceneVariables::sCamera)->asA<Camera>();
201 } else {
202 // Else just the first camera created in the context
203 return mCameras[0];
204 }
205}

207std::vector<const rdl2::Camera*>
208SceneContext::getCameras() const
209{
210 // Prevent possible race condition with mCameras.push_back() in createSceneObject().
211 tbb::mutex::scoped_lock lock(mCreateSceneObjectMutex);
212
213 std::vector<const rdl2::Camera*> cameras;
214
215 if (mCameras.size() == 0) {
216 return cameras;
217 }
218
219 // The primary camera is the first element in the output cameras vector
220 if (mSceneVariables->get(SceneVariables::sCamera)) {
221 // Use the camera specified by the scene variable
222 cameras.push_back(mSceneVariables->get(SceneVariables::sCamera)->asA<Camera>());
223 } else {
224 // Else just the first camera created in the context
225 cameras.push_back(mCameras[0]);
226 }
227
228 // Populate the rest of the cameras, skipping the primary
229 for (size_t i = 0; i < mCameras.size(); i++) {
230 if (cameras[0] != mCameras[i]) {
231 cameras.push_back(mCameras[i]);
232 }
233 }
234
235 return cameras;
236}

238std::vector<const rdl2::Camera*>
239SceneContext::getActiveCameras(void) const
240{
241 // Prevent possible race condition with mCameras.push_back() in createSceneObject().
242 tbb::mutex::scoped_lock lock(mCreateSceneObjectMutex);
243
244 std::vector<const rdl2::Camera*> cameras;
245
246 if (mCameras.size() == 0) {
247 return cameras;
248 }
249
250 // The primary camera is the first element in the output cameras vector
251 if (mSceneVariables->get(SceneVariables::sCamera)) {
252 // Use the camera specified by the scene variable
253 cameras.push_back(mSceneVariables->get(SceneVariables::sCamera)->asA<Camera>());
254 } else {
255 // Else just the first camera created in the context
256 cameras.push_back(mCameras[0]);
257 }
258
259 // Populate the rest of the cameras, skipping the primary
260 const rdl2::SceneContext::RenderOutputVector &outputs = getAllRenderOutputs();
261 for (size_t i = 0; i < mCameras.size(); i++) {
262 if (cameras[0] != mCameras[i]) {
263 // Check if this camera is referenced by a render output, skip those
264 // that aren't.
265 for (size_t j = 0; j < outputs.size(); j++) {
266 if (outputs[j]->getCamera() == mCameras[i]) {
267 cameras.push_back(mCameras[i]);
268 break;
269 }
270 }
271 }
272 }
273
274 return cameras;
275}

277SceneClass*
278SceneContext::createSceneClass(const std::string& className)
279{
280 if (className.empty()) {
281 throw except::ValueError("Cannot create a SceneClass with an empty class name.");
282 }
283
284 // First, do a quick check for existence. If the class already exists,
285 // multiple readers can do this simultaneously.
286
287 { // Begin reader lock scope.
288 SceneClassMap::const_accessor reader;
289
290 if (mSceneClasses.find(reader, className)) {
291 // Found it.
292 return reader->second;
293 }
294 } // End reader lock scope.
295
296 // WARNING: THIS CODE IS CRITICAL TO THREAD SAFETY!
297 //
298 // It's tempting to think we don't need to check for existence again after
299 // this point, but remember that a read lock is not exclusive. That means
300 // if a class does not exist, multiple threads could make it here and try
301 // to create the same class at the same time.
302 //
303 // Acquiring a write lock here will give us exclusive access, which we can
304 // then use to check for existence again. We then have both exclusive
305 // access to the container AND we know whether it exists. If it does exist
306 // in our exclusive test, that means another thread beat us to the
307 // exclusive writer lock, created the SceneClass, and we should just return
308 // what we found.
309 //
310 // It's also important to remember that this exclusive lock is how we are
311 // able to guarantee thread safety to DSO declare() function. That is only
312 // guaranteed because we acquire this exclusive lock before calling it.
313
314 // Writer lock in scope until the end of the function.
315 SceneClassMap::accessor writer;
316
317 // If insert returns true, the item is new.
318 if (mSceneClasses.insert(writer, className)) {
319 // This tbb writer lock is per bucket and not per container, so anything
320 // outside the mSceneClasses container needs to be thread safe by its own
321 // means not depending on the writer lock
322
323 std::unique_ptr<SceneClass> sc;
324
325 // It really does not exist yet. Let's try to create it.
326 try {
327 std::string dsoPath = getDsoPath();
328 if (mProxyModeEnabled) {
329 sc.reset(new SceneClass(this, className,
330 ObjectFactory::createProxyFactory(className, dsoPath)));
331 } else {
332 sc.reset(new SceneClass(this, className,
333 ObjectFactory::createDsoFactory(className, dsoPath)));
334 }
335 sc->declare();
336 sc->setComplete();
337 } catch (...) {
338 // Something went wrong when creating the SceneClass. Roll back
339 // our insertion by erasing the key.
340 mSceneClasses.erase(writer);
341
342 // Rethrow whatever caused the problem.
343 throw;
344 }
345
346 // If no exceptions were thrown, we should have a valid SceneClass and
347 // it should be safe to go ahead with the insert.
348 MNRY_ASSERT(sc, "SceneClass should never be invalid prior to insertion.");
349 writer->second = sc.release();
350 }
351
352 // Regardless of whether we created the SceneClass just now because it was
353 // a new entry in the container, or it wasn't new because another thread
354 // beat us to the writer lock, we want to return it.
355 return writer->second;
356}

358SceneObject*
359SceneContext::createSceneObject(const std::string& className,
360 const std::string& objectName)
361{
362 if (className.empty()) {
363 throw except::ValueError("Cannot create a SceneObject with an empty class name.");
364 } else if (objectName.empty()) {
365 throw except::ValueError("Cannot create a SceneObject with an empty object name.");
366 }
367
368 // Enforce the singleton-ness of the __SceneVariables__ object.
369 if (className == "SceneVariables" && objectName != "__SceneVariables__") {
370 return mSceneVariables;
371 }
372
373 // Do a quick check for existence. If the class already exists, multiple
374 // readers can do this simultaneously.
375
376 { // Begin reader lock scope.
377 SceneObjectMap::const_accessor reader;
378
379 if (mSceneObjects.find(reader, objectName)) {
380 // Found it.
381 verifyMatchingSceneClass(className, reader->second);
382 return reader->second;
383 }
384 } // End reader lock scope.
385
386 // WARNING: THIS CODE IS CRITICAL TO THREAD SAFETY!
387 //
388 // It's tempting to think we don't need to check for existence again after
389 // this point, but remember that a read lock is not exclusive. That means
390 // if the object does not exist, multiple threads could make it here and
391 // try to create the same object at the same time.
392 //
393 // Acquiring a write lock here will give us exclusive access, which we can
394 // then use to check for existence again. We then have both exclusive
395 // access to the container AND we know whether it exists. If it does exist
396 // in our exclusive test, that means another thread beat us to the
397 // exclusive writer lock, created the SceneObject, and we should just return
398 // what we found.
399 //
400 // It's also important to remember that this exclusive lock is how we are
401 // able to guarantee thread safety to DSO create() function. That is only
402 // guaranteed because we acquire this exclusive lock before calling it.
403
404 // Get (or create, if necessary) the SceneClass first.
405 SceneClass* sc = createSceneClass(className);
406
407 // Writer lock in scope until the end of the function.
408 SceneObjectMap::accessor writer;
409
410 // If insert returns true, the item is new.
411 if (mSceneObjects.insert(writer, objectName)) {
412 SceneObject* obj = nullptr;
413
414 // It really does not exist yet. Let's try to create it.
415 try {
416 // The createObject() function should not leak if it fails.
417 obj = sc->createObject(objectName);
418 } catch (...) {
419 // Something went wrong when creating the SceneObject. Roll back
420 // our insertion by erasing the key.
421 mSceneObjects.erase(writer);
422
423 // Rethrow whatever caused the problem.
424 throw;
425 }
426
427 // If no exceptions were thrown, we should have a valid SceneObject and
428 // it should be safe to go ahead with the insert.
429 MNRY_ASSERT(obj, "SceneObject should never be invalid prior to insertion.");
430 writer->second = obj;
431
432 // The containers that are below are not thread safe versions and are not protected
433 // by the mSceneObjects write lock since tbb locks per bucket and not per container.
434 // mCreateSceneObjectMutex is a SceneContext class mutex that protects non thread safe
435 // write access to them and also to the camera. Any other non thread safe code in this
436 // method should also be protected by it. Write and read access at the same time is NOT allowed
437 // and will not since it has performance implications for the read only calls of the renderer.
438 // The api exposes the containers and iterators but they should never be used even for reading
439 // if other thread is performing modifications to them by calling this function.
440
441 // Do any type-specific setup.
442 if (obj->isA<Geometry>()) {
443 tbb::mutex::scoped_lock lock(mCreateSceneObjectMutex);
444 mGeometries.push_back(obj->asA<Geometry>());
445 } else if (obj->isA<GeometrySet>()) {
446 tbb::mutex::scoped_lock lock(mCreateSceneObjectMutex);
447 mGeometrySets.push_back(obj->asA<GeometrySet>());
448 } else if (obj->isA<Camera>()) {
449 tbb::mutex::scoped_lock lock(mCreateSceneObjectMutex);
450 mCameras.push_back(obj->asA<Camera>());
451 } else if (obj->isA<RenderOutput>()) {
452 tbb::mutex::scoped_lock lock(mCreateSceneObjectMutex);
453 mRenderOutputs.push_back(obj->asA<RenderOutput>());
454 }
455
456 // Call any on-creation callbacks
457 for (auto cb : mCreateCallbacks) {
458 cb(obj);
459 }
460 } else {
461 // We didn't win the insertion race, so verify that the SceneClass
462 // matches.
463 verifyMatchingSceneClass(className, writer->second);
464 }
465
466 // Regardless of whether we created the SceneObject just now because it was
467 // a new entry in the container, or it wasn't new because another thread
468 // beat us to the writer lock, we want to return it.
469 return writer->second;
470}

472void
473SceneContext::applyUpdates(Layer * const layer)
474{
475 // Now that the scene variables and the camera are available, we can update the
476 // coefficients in the scene context that hold information about the shutter interval and
477 // motion steps.
478 // Note that the camera's shutter times are ignored if motion blur is disabled (either in
479 // the scene vars or in the camera itself), in order to remain consistent with the way
480 // geometry objects behave.
481
482 const Camera* primaryCamera = getPrimaryCamera();
483
484 float shutterOpen = 0.0f;
485 float shutterClose = 0.0f;
486 if (primaryCamera && mSceneVariables->get(SceneVariables::sEnableMotionBlur)) {
487 shutterOpen = primaryCamera->get(Camera::sMbShutterOpenKey);
488 shutterClose = primaryCamera->get(Camera::sMbShutterCloseKey);
489 }
490 computeTimeRescalingCoeffs(shutterOpen, shutterClose, mSceneVariables->get(SceneVariables::sMotionSteps));
491
492 // cache primitive attributes contained in the shader network of all materials.
493 // This must be done before any updates to SceneObjects.
494 if (layer) {
495 Layer::MaterialSet materials;
496 layer->getAllMaterials(materials);
497 for (const Material* m : materials) {
498 if (m) {
499 m->cacheShaderGraphPrimAttributes();
500 }
501 }
502 }
503
504 // SceneVariables need to be updated first
505 mSceneVariables->updatePrep(mSceneObjectUpdateGraph, 0);
506
507 // Cameras needs to be updated second
508 for (size_t i = 0; i < mCameras.size(); i++) {
509 mCameras[i]->updatePrep(mSceneObjectUpdateGraph, 0);
510 }
511
512 // Flag shaders that are in the update graph, as we will need to re-build the associated attribute tables.
513 // Also flag the associated geometry, since a change in the attribute table might require a geometry update. Also
514 // in this step, we will flag the LightSets, LightFilterSets, ShadowSets, and ShadowReceiverSets that will need
515 // updating in the preFrame() function.
516 if (layer) {
517 layer->updatePrepAssignments(mSceneObjectUpdateGraph, 0);
518 }
519
520 for (GeometrySet * geometrySet : mGeometrySets) {
521 // This is an optimization to avoid calling a full blown updatePrep(),
522 // which would require an unnecessary full loop over geometries again.
523 geometrySet->updatePrepFast(mSceneObjectUpdateGraph, 0);
524 }
525
526 // need to update display filters
527 for (auto& kv : mSceneObjects) {
528 SceneObject* so = kv.second;
529 if(so->isA<DisplayFilter>()) {
530 so->updatePrep(mSceneObjectUpdateGraph, 0);
531 }
532 }
533
534 // Update all leaves
535 int s = mSceneObjectUpdateGraph.size();
536 if (s == 0){
537 Logger::info("There is no leaf scene object need to be updated");
538 } else if (s == 1) {
539 Logger::info("Updating 1 leaf scene object...");
540 } else {
541 Logger::info("Updating ", s, " leaf scene objects...");
542 }
543
544 tbb::parallel_for_each(mSceneObjectUpdateGraph.cbegin(), mSceneObjectUpdateGraph.cend(),
545 [&] (SceneObject* const obj)
546 {
547 obj->debug("Updating");
548 obj->update();
549 });
550
551 // Update the objects from bottom up
552 for (int i = mSceneObjectUpdateGraph.getMaxDepth()-1; i >= 0; --i) {
553 int s = mSceneObjectUpdateGraph.size(i);
554 if (s == 0){
555 Logger::info("There is no scene object need to be updated at level ", i);
556 } else if (s == 1) {
557 Logger::info("Updating 1 scene object at level ", i, "...");
558 } else {
559 Logger::info("Updating ", s, " scene objects at level ", i, "...");
560 }
561
562 tbb::parallel_for_each(mSceneObjectUpdateGraph.cbegin(i), mSceneObjectUpdateGraph.cend(i),
563 [&] (SceneObject* const obj)
564 {
565 obj->debug("Updating");
566 obj->update();
567 });
568 }
569
570 // Changes in a shader's requested primitive attributes require updating
571 // the geometry.
572 // Also, changes in the volumeShader requires updating the geometry because
573 // volume shaders are baked into the geometry.
574 if (layer) {
575 const auto& changedOrDeformedGeometries = layer->getChangedOrDeformedGeometries();
576 // If there are no changed or deformed geometries, do nothing.
577 if (changedOrDeformedGeometries.empty()) return;
578
579 for (auto& geomPair : changedOrDeformedGeometries) {
580 Geometry* geom = geomPair.first;
581 int index = geomPair.second;
582
583 // If a shader requests new primitive attributes from the geometry, then
584 // we need to update the geometry.
585 const Material * material = layer->lookupMaterial(index);
586 if (material && material->haveShaderGraphPrimAttributesChanged()) {
587 geom->requestUpdate();
588 }
589 // If a volume shader needs updating, then we need to update the geometry
590 const VolumeShader * volumeShader = layer->lookupVolumeShader(index);
591 if (volumeShader && volumeShader->updateBakeRequired()) {
592 geom->requestUpdate();
593 }
594 }
595 }
596}

598void
599SceneContext::applyUpdatesToMeshLightLayer(Layer * const layer)
600{
601 // Be sure to call applyUpdates before this function -- it will flag the scene
602 // objects that need updating, while this function handles the layer updates
603
604 if (!layer) {
605 return;
606 }
607
608 // cache primitive attributes contained in the shader network of all materials.
609 // This must be done before any updates to SceneObjects.
610 Layer::MaterialSet materials;
611 layer->getAllMaterials(materials);
612 for (const Material* m : materials) {
613 if (m) {
614 m->cacheShaderGraphPrimAttributes();
615 }
616 }
617
618 // Flag shaders that are in the update graph, as we will need to re-build the associated attribute tables.
619 // Also flag the associated geometry, since a change in the attribute table might require a geometry update. Also
620 // in this step, we will flag the LightSets, LightFilterSets, ShadowSets, and ShadowReceiverSets that will need
621 // updating in the preFrame() function.
622 layer->updatePrepAssignments(mSceneObjectUpdateGraph, 0);
623
624 // Changes in a shader's requested primitive attributes require updating
625 // the geometry.
626 // Also, changes in the volumeShader requires updating the geometry because
627 // volume shaders are baked into the geometry.
628
629 const auto& changedOrDeformedGeometries = layer->getChangedOrDeformedGeometries();
630 // If there are no changed or deformed geometries, do nothing.
631 if (changedOrDeformedGeometries.empty()) return;
632
633 for (auto& geomPair : changedOrDeformedGeometries) {
634 Geometry* geom = geomPair.first;
635 int index = geomPair.second;
636
637 // If a shader requests new primitive attributes from the geometry, then
638 // we need to update the geometry.
639 const Material * const material = layer->lookupMaterial(index);
640 if (material && material->haveShaderGraphPrimAttributesChanged()) {
641 geom->requestUpdate();
642 }
643 }
644}

646std::unordered_map<std::string, size_t>
647SceneContext::getDsoCounts() const
648{
649 std::unordered_map<std::string, size_t> dsoCounts;
650 for (int i = mSceneObjectUpdateGraph.getMaxDepth()-1; i >= 0; --i) {
651 for (auto it = mSceneObjectUpdateGraph.cbegin(i); it != mSceneObjectUpdateGraph.cend(i); ++it) {
652 const SceneObject* obj = *it;
653 if (obj->isA<Material>() ||
654 obj->isA<Map>() ||
655 obj->isA<NormalMap>() ||
656 obj->isA<VolumeShader>() ||
657 obj->isA<DisplayFilter>() ||
658 obj->isA<Geometry>() ||
659 obj->isA<Displacement>() ||
660 obj->isA<LightFilter>() ||
661 obj->isA<Light>()) {
662
663 const std::string& className = obj->getSceneClass().getName();
664 if (dsoCounts.find(className) == dsoCounts.end()) {
665 dsoCounts.insert(std::make_pair(className, 1));
666 } else {
667 ++dsoCounts[className];
668 }
669 }
670 }
671 }
672 return dsoCounts;
673}

683void
684SceneContext::getUpdatedOrDeformedGeometrySets(Layer * const layer,
685 GeometrySetVector & updatedSets)
686{
687 const Layer::GeometryIndexMap & updatedOrDeformedGeometries =
688 layer->getChangedOrDeformedGeometries();
689 for (GeometrySet * const set : mGeometrySets) {
690 if (set->updatePrepApplied()) {
691 updatedSets.push_back(set);
692 continue;
693 }
694 for (SceneObject * const object : set->getGeometries()) {
695 if (updatedOrDeformedGeometries.count(object->asA<Geometry>()) != 0) {
696 updatedSets.push_back(set);
697 break;
698 }
699 }
700 }

712void
713SceneContext::loadAllSceneClasses()
714{
715 std::string remaining(getDsoPath());
716 while (!remaining.empty()) {
717 // Grab the next path entry.
718 std::size_t colonPos = remaining.find_first_of(':');
719 std::string directory = remaining.substr(0, colonPos);
720
721 // Grab the contents of each directory.
722 DIR* directoryPtr = opendir(directory.c_str());
723 if (directoryPtr != nullptr) {
724 struct dirent* entryPtr;
725 while ((entryPtr = readdir(directoryPtr))) {
726 std::string fileName(entryPtr->d_name);
727
728 // If the file is a valid DSO, then create a SceneClass from it.

734 className = fileName.substr(0, fileName.size() - 9);

743 // the declare() function, just move on to the next

745 }
746 }
747 }
748 }

754 } else {

760void
761SceneContext::computeTimeRescalingCoeffs(float shutterOpen, float shutterClose, const std::vector<float> &motionSteps)
762{
763 // See declaration of TimeRescalingCoeffs in Types.h for details.
764
765 tbb::mutex::scoped_lock lock(mTimeRescalingCoeffsMutex);
766
767 MNRY_ASSERT_REQUIRE(motionSteps.size() >= 1 && motionSteps.size() <= 2);
768 if (motionSteps.size() == 1 || motionSteps[0] == motionSteps[1]) {
769 // Handle extraordinary case where we don't have 2 distinct motion steps
770 // (denominator of the coefficients would be 0).
771 mTimeRescalingCoeffs.mScale = 0.0f;
772 mTimeRescalingCoeffs.mOffset = 0.0f;
773 } else {
774 float oneOverDenom = 1.0f / (motionSteps[1] - motionSteps[0]);
775 mTimeRescalingCoeffs.mScale = (shutterClose - shutterOpen ) * oneOverDenom;
776 mTimeRescalingCoeffs.mOffset = (shutterOpen - motionSteps[0]) * oneOverDenom;
777 }
778}

785
786SceneContext::GeometrySetVector SceneContext::getGeometrySetsForLayer(const Layer* layer)
787{
788 GeometrySetVector ret;
789
790 // We have to iterate over the geometry pointers stored in the GeometrySet
791 // to see if the geometry is in the layer. This is in a separate function so
792 // that we can break out of our outer loop if we find it, because we only
793 // want to add the geometry set once.
794 auto geomIter = [&ret, layer] (GeometrySet* p) {
795 for (const auto& sceneobj : p->getGeometries()) {
796 const auto geom = sceneobj->asA<Geometry>();
797 if (geom && layer->contains(geom)) {
798 ret.push_back(p);
799 return;
800 }
801 }
802 };
803
804 for (const auto gs : mGeometrySets) {
805 geomIter(gs);
806 }
807
808 return ret;
809}
