37void
38readSceneFromFile(const std::string& filePath, SceneContext& context)
39{
40 // Grab the file extension and convert it to lower case.
41 auto ext = util::lowerCaseExtension(filePath);
42 if (ext.empty()) {
43 throw except::RuntimeError(util::buildString(
44 "File '", filePath, "' has no extension."
45 " Cannot determine file type."));
46 }
47
48 if (ext == "rdla") {
49 AsciiReader reader(context);
50 reader.fromFile(filePath);
51 } else if (ext == "rdlb") {
52 BinaryReader reader(context);
53 reader.fromFile(filePath);
54 } else {
55 throw except::RuntimeError(util::buildString(
56 "File '", filePath, "' has an unknown extension."
57 " Cannot determine file type."));
58 }
59}

61void
62writeSceneToFile(const SceneContext& context, const std::string& filePath,
63 bool deltaEncoding, bool skipDefaults, size_t elemsPerLine)
64{
65 // Grab the file extension and convert it to lower case.
66 auto ext = util::lowerCaseExtension(filePath);
67
68 if (ext == "rdla") {
69 AsciiWriter writer(context);
70 writer.setDeltaEncoding(deltaEncoding);
71 writer.setSkipDefaults(skipDefaults);
72 writer.setElementsPerLine(elemsPerLine);
73 writer.toFile(filePath);
74 } else if (ext == "rdlb") {
75 BinaryWriter writer(context);
76 writer.setTransientEncoding(false);
77 writer.setDeltaEncoding(deltaEncoding);
78 writer.setSkipDefaults(skipDefaults);
79 writer.toFile(filePath);
80 } else if (ext.empty()) {
81 AsciiWriter awriter(context);
82 awriter.setSkipDefaults(skipDefaults);
83 awriter.setDeltaEncoding(deltaEncoding);
84 awriter.setElementsPerLine(elemsPerLine);
85 awriter.setMaxVectorSize(SPLIT_VEC_SIZE);
86 awriter.toFile(filePath + ".rdla");
87 BinaryWriter bwriter(context);
88 bwriter.setSkipDefaults(skipDefaults);
89 bwriter.setTransientEncoding(false);
90 bwriter.setDeltaEncoding(deltaEncoding);
91 bwriter.setSplitMode(SPLIT_VEC_SIZE + 1);
92 bwriter.toFile(filePath + ".rdlb");
93 } else {
94 throw except::RuntimeError(util::buildString(
95 "File '", filePath, "' has an unknown extension."
96 " Cannot determine file type."));
97 }
98}

100size_t
101vectorSize(const SceneObject& so, const Attribute& attr)
102{
103 switch (attr.getType()) {
104 case TYPE_BOOL_VECTOR: return so.get(AttributeKey<BoolVector>(attr)).size();
105 case TYPE_INT_VECTOR: return so.get(AttributeKey<IntVector>(attr)).size();
106 case TYPE_LONG_VECTOR: return so.get(AttributeKey<LongVector>(attr)).size();
107 case TYPE_FLOAT_VECTOR: return so.get(AttributeKey<FloatVector>(attr)).size();
108 case TYPE_DOUBLE_VECTOR: return so.get(AttributeKey<DoubleVector>(attr)).size();
109 case TYPE_STRING_VECTOR: return so.get(AttributeKey<StringVector>(attr)).size();
110 case TYPE_RGB_VECTOR: return so.get(AttributeKey<RgbVector>(attr)).size();
111 case TYPE_RGBA_VECTOR: return so.get(AttributeKey<RgbaVector>(attr)).size();
112 case TYPE_VEC2F_VECTOR: return so.get(AttributeKey<Vec2fVector>(attr)).size();
113 case TYPE_VEC2D_VECTOR: return so.get(AttributeKey<Vec2dVector>(attr)).size();
114 case TYPE_VEC3F_VECTOR: return so.get(AttributeKey<Vec3fVector>(attr)).size();
115 case TYPE_VEC3D_VECTOR: return so.get(AttributeKey<Vec3dVector>(attr)).size();
116 case TYPE_VEC4F_VECTOR: return so.get(AttributeKey<Vec4fVector>(attr)).size();
117 case TYPE_VEC4D_VECTOR: return so.get(AttributeKey<Vec4dVector>(attr)).size();
118 case TYPE_MAT4F_VECTOR: return so.get(AttributeKey<Mat4fVector>(attr)).size();
119 case TYPE_MAT4D_VECTOR: return so.get(AttributeKey<Mat4dVector>(attr)).size();
120 case TYPE_SCENE_OBJECT_VECTOR: return so.get(AttributeKey<SceneObjectVector>(attr)).size();
121 case TYPE_SCENE_OBJECT_INDEXABLE: return so.get(AttributeKey<SceneObjectIndexable>(attr)).size();
122 default: // the rest are non-vectors
123 return 1;
124 }

140std::string
141replacePoundWithSampleNumber(const std::string& path, float sampleNum)
142{
143 std::string s(path);
144 std::string numStr = util::buildString(sampleNum);
145 std::size_t numStrLength = numStr.size();
146
147 std::size_t pos = 0;
148 while (true) {
149 pos = s.find_first_of('#', pos);
150 if (pos == std::string::npos) break;
151 s.replace(pos, 1, numStr);
152 pos += numStrLength;
153 }
154
155 return s;
156}

158FloatVector
159uniqueSampleNumberRange(const SceneVariables& sceneVars)
160{
161 const auto& motionSteps = sceneVars.get(SceneVariables::sMotionSteps);
162 MNRY_ASSERT_REQUIRE(!motionSteps.empty());
163
164 // Use a frame range if it's set.
165 auto minFrame = sceneVars.get(SceneVariables::sMinFrameKey);
166 auto maxFrame = sceneVars.get(SceneVariables::sMaxFrameKey);
167
168 // Use the current frame if it isn't.
169 if (minFrame == 0.0f && maxFrame == 0.0f) {
170 minFrame = sceneVars.get(SceneVariables::sFrameKey);
171 maxFrame = minFrame;
172 }
173
174 std::set<float> sampleSet;
175 for (auto frame = minFrame; frame <= maxFrame; frame += 1.0f) {
176 for (auto iter = motionSteps.begin(); iter != motionSteps.end(); ++iter) {
177 sampleSet.insert(computeSampleNumber(frame, *iter));
178 }
179 }
180
181 return FloatVector(sampleSet.begin(), sampleSet.end());
182}
