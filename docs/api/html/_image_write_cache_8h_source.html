667ImageWriteCache::setupEnqMode()

680inline void
681ImageWriteCache::setupDeqMode()
682{
683 if (mMode == Mode::ENQ) {
684 // We need to finalize CacheEnqueue when previous condition is ENQ
685 enq()->finalize();
686 for (int yBlockId = 0; yBlockId < mYBlockTotal; ++yBlockId) {
687 enqFullBuff(yBlockId)->finalize();
688 enqHalfBuff(yBlockId)->finalize();
689 }
690 }
691
692 // Re-construct CacheDequeue
693 new(&mCacheQueue.mCDeq) std::unique_ptr<scene_rdl2::cache::CacheDequeue>;
694 mCacheQueue.mCDeq.reset(new scene_rdl2::cache::CacheDequeue(mData.data(), mData.size()));
695
696 for (int yBlockId = 0; yBlockId < mYBlockTotal; ++yBlockId) {
697 new(&mCacheQueueFullBuffArray[yBlockId].mCDeq) std::unique_ptr<scene_rdl2::cache::CacheDequeue>;
698 mCacheQueueFullBuffArray[yBlockId].mCDeq.reset
