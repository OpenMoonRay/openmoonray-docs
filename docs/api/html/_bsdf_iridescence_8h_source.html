33scene_rdl2::math::Color
34computeIridescenceColor(float nDotV, const scene_rdl2::math::Color& primaryColor, const scene_rdl2::math::Color& secondaryColor,
35 bool flipHue, float thickness, float exponent)
36{
37 using scene_rdl2::math::rgbToHue;
38 using scene_rdl2::math::hueToRgb;
39 // 1 - nDotV, to reverse the color spectrum between primaryColor and secondaryColor
40 // so that secondaryColor appears at grazing angles and primaryColor appears straight on.
41 float index = scene_rdl2::math::pow(scene_rdl2::math::abs(1.f - nDotV), exponent) * thickness;
42 index = scene_rdl2::math::fmod(index, 1.f);
43
44 scene_rdl2::math::Color iridescence;
45 if (scene_rdl2::math::isEqual(primaryColor, secondaryColor)) {
46 // If colors are equal, interpolate around the hue spectrum back to the same color.
47 if (flipHue) {
48 // Nothing to interpolate in the counter-clockwise direction.
49 iridescence = primaryColor;
50 } else {
51 // Interpolate clockwise around the whole hue wheel back to the same color.
52 const float huePrimary = scene_rdl2::math::rgbToHue(primaryColor);
53 // Remap range to [hue value, hue value + 1] so that the primary color appears
54 // at near normal and glancing angles and the rest of the hues are interpolated between.
55 const float clampedIndex = fitInRange(index, 0.f, 1.f, huePrimary, huePrimary + 1.f);
56 iridescence = scene_rdl2::math::hueToRgb(clampedIndex);
57 }
58 } else {
59 // If colors are not equal, interpolate a range around the hue wheel and lerp from
60 // the secondaryColor back to the primaryColor.
61 float huePrimary = scene_rdl2::math::rgbToHue(primaryColor);
62 float hueSecondary = rgbToHue(secondaryColor);
63 // If the hue is not flipped but the secondary color is greater than the primary,
64 // we need to wrap around the secondary color to maintain clockwise interpolation around the wheel.
65 // If the hue is flipped but primary is less than secondary, we need to wrap around
66 // the primary color so that it interpolates in reverse, counter-clockwise.
67 if (hueSecondary < huePrimary && !flipHue) {
68 hueSecondary += 1.f;
69 } else if (huePrimary < hueSecondary && flipHue) {
70 huePrimary += 1.f;
71 }
72
73 // Lerp back to the primary color from the secondary color using part of the
74 // nDotV index. The hue spectrum is normally divided into 6 sections or 1/6ths (0.166667),
75 // but since the hue range is limited, these sections become proportionally larger.
76 // The lerpBackDistance is computed by inverting this value and adding 1 to add another "section,"
77 // and then inverting it again to make the lerp distance 1/x, for x = number of sections.
78 const float lerpBackDistance = 1.f / (scene_rdl2::math::abs(hueSecondary - huePrimary) / 0.166667 + 1.f);
79 const float spectrumDistance = 1.f - lerpBackDistance;
80 const float clampedIndex = fitInRange(index, 0.f, spectrumDistance, huePrimary, hueSecondary);
81 if (index < spectrumDistance) {
82 iridescence = hueToRgb(clampedIndex);
83 } else {
84 iridescence = scene_rdl2::math::lerp(hueToRgb(hueSecondary), hueToRgb(huePrimary), (index - spectrumDistance)/lerpBackDistance );
85 }
86 }
87
88 return iridescence;
89}

91finline bool
92computeAverageOfHalfVectorCosines(const scene_rdl2::math::Vec3f& wo, const scene_rdl2::math::Vec3f& wi, float& avgCos)
93{
94 scene_rdl2::math::Vec3f m;
95 if (!computeNormalizedHalfVector(wo, wi, m)) {
96 return false;
97 }
98 const float cosMO = dot(m, wo);
99 const float cosMI = dot(m, wi);
100
101 // There is a slight difference between cosMO (hDotV) and cosMI (hDotL) which causes the unit test
102 // to fail. Averaging the two circumvents this issue and results in identical results
103 // as using just cosMO.
104 avgCos = (cosMO + cosMI) * 0.5f;
105 return true;
106}

145 // BsdfLobe API
146 finline scene_rdl2::math::Color eval(const BsdfSlice &slice,
147 const scene_rdl2::math::Vec3f &wi,
148 float *pdf = NULL) const override
149 {
150 // Mirror lobe will not be evaluated!
151 if (mChildLobe->getType() & MIRROR) {
152 if (pdf) *pdf = 0.0f;
153 return scene_rdl2::math::sBlack;
154 }
155
156 // Glossy lobe
157 // Evaluate the underlying bsdf
158 const scene_rdl2::math::Color lobeResult = mChildLobe->eval(slice, wi, pdf);
159
160 // use half vector dot products for iridescence computation
161 float avgCos;
162 if (!computeAverageOfHalfVectorCosines(slice.getWo(), wi, avgCos)) {
163 return scene_rdl2::math::sBlack;
164 }
165 return tintWithIridescence(avgCos, lobeResult) * computeScaleAndFresnel(avgCos);
166 }
167 finline scene_rdl2::math::Color sample(const BsdfSlice &slice, float r1, float r2,
168 scene_rdl2::math::Vec3f &wi, float &pdf) const override
169 {
170 // Sample the child spec lobe
171 const scene_rdl2::math::Color res = mChildLobe->sample(slice, r1, r2, wi, pdf);
172
173 // Mirror
174 // *Optimization* : using NdotV for iridescence color computation
175 if (mChildLobe->getType() & MIRROR) {
176 const float cosThetaNV = dot(mN, slice.getWo());
177 return tintWithIridescence(cosThetaNV, res) * computeScaleAndFresnel(cosThetaNV);
178 }
179
180 // Glossy - use half vector dot products for iridescence computation
181 float avgCos;
182 if (!computeAverageOfHalfVectorCosines(slice.getWo(), wi, avgCos)) {
183 return scene_rdl2::math::sBlack;
184 }
185 return tintWithIridescence(avgCos, res) * computeScaleAndFresnel(avgCos);
186 }

222private:
223 scene_rdl2::math::Color tintWithIridescence(
224 const float cosThetaHV, const scene_rdl2::math::Color& lobeColor) const
225 {
226 // Early exit
227 if (isBlack(lobeColor)) {
228 return scene_rdl2::math::sBlack;
229 }
230
231 scene_rdl2::math::Color iridescence;
232 if (mColorControl == ispc::SHADING_IRIDESCENCE_COLOR_USE_HUE_INTERPOLATION) {
233 // Creates a view-dependent color by indexing into the hue spectrum between
234 // the given primary and secondary color.
235 iridescence = computeIridescenceColor(
236 cosThetaHV, mPrimary, mSecondary, mFlipHue, mThickness, mExponent);
237 } else { // SHADING_IRIDESCENCE_COLOR_USE_RAMP
238 // Creates a view-dependent color by indexing into the user provided ramp.
239 float index = scene_rdl2::math::pow(scene_rdl2::math::abs(1.f - cosThetaHV), mExponent) * mThickness;
240 index = scene_rdl2::math::fmod(index, 1.f);
241 iridescence = mRampControl.eval1D(index);
242 }
243
244 const float att = scene_rdl2::math::saturate(
245 scene_rdl2::math::lerp(mIridescenceAt90, mIridescenceAt0, cosThetaHV));
246 iridescence = scene_rdl2::math::lerp(
247 scene_rdl2::math::sWhite, iridescence, att * mIridescence);
248
249 // Multiply lobe result with iridescence approximation.
250 return lobeColor * iridescence;
251 }
252
253private:
254 BsdfLobe* mChildLobe;
255 scene_rdl2::math::Vec3f mN;
256 float mIridescence;
257 ispc::SHADING_IridescenceColorMode mColorControl;
258 scene_rdl2::math::Color mPrimary;
259 scene_rdl2::math::Color mSecondary;
260 bool mFlipHue;
261 ColorRampControl mRampControl;
262 float mThickness;
263 float mExponent;
264 float mIridescenceAt0;
265 float mIridescenceAt90;
266};
