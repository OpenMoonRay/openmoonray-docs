48void
49UsdPrimvarReader_int::update()
50{
51 if (hasChanged(attrVarName)) {
52 mOptionalAttributes.clear();
53 const std::string& varname = get(attrVarName);
54 if (varname == "id") {
55 mIspc.mAttributeMapType = ispc::PRIMVAR_MAP_TYPE_ID;
56 mIspc.mPrimitiveAttributeIndex = moonray::shading::StandardAttributes::sId;
57 mOptionalAttributes.push_back(mIspc.mPrimitiveAttributeIndex);
58 } else {
59 mIspc.mAttributeMapType = ispc::PRIMVAR_MAP_TYPE_PRIMITIVE_ATTRIBUTE;
60 moonray::shading::TypedAttributeKey<int> attributeKey(varname);
61 mIspc.mPrimitiveAttributeIndex = attributeKey;
62 mOptionalAttributes.push_back(attributeKey);
63 }
64 ::moonshine::primvar::createLogEvent("int",
65 varname,
66 mIspc.mMissingAttributeEvent,
67 mLogEventRegistry);
68 }
69}

71void
72UsdPrimvarReader_int::sample(const scene_rdl2::rdl2::Map *self, moonray::shading::TLState *tls,
73 const moonray::shading::State &state, Color *sample)
74{
75 const UsdPrimvarReader_int* me = static_cast<const UsdPrimvarReader_int*>(self);
76
77 int key = me->mIspc.mPrimitiveAttributeIndex;
78 if (state.isProvided(key)) {
79 const float& v = static_cast<float>(state.getAttribute(moonray::shading::TypedAttributeKey<int>(key)));
80 sample->r = v;
81 sample->g = v;
82 sample->b = v;
83 } else {
84 // the primitive attribute is unavailable, use fallback parameter
85 const float& v = static_cast<float>(me->get(attrFallback));
86 sample->r = v;
87 sample->g = v;
88 sample->b = v;
89 if (me->get(attrWarnWhenUnavailable)) {
90 moonray::shading::logEvent(me, tls, me->mIspc.mMissingAttributeEvent);
91 }
92 }
93}
