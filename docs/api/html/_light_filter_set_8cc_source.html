44void
45LightFilterSet::add(LightFilter* lightfilter)
46{
47 if (!mUpdateActive) {
48 std::stringstream errMsg;
49 errMsg << "LightFilter '" << lightfilter->getName() << "' can only be added to"
50 " LightFilterSet '" << mName << "' between beginUpdate() and endUpdate()"
51 " calls.";
52 throw except::RuntimeError(errMsg.str());
53 }
54
55 // Retrieve a mutable reference to the lights attribute.
56 SceneObjectVector& lightfilters = getMutable(sLightFiltersKey);
57
58 // Binary search for the insertion point.
59 SceneObjectVector::iterator insertPoint = lowerBoundByName(lightfilters.begin(),
60 lightfilters.end(),
61 lightfilter);
62
63 // Is the lightfilter at the insertion point the same? If so, it's already in the
64 // set.
65 if (insertPoint != lightfilters.end() && *insertPoint == lightfilter) {
66 return;
67 }
68
69 // Otherwise, do the insert.
70 lightfilters.insert(insertPoint, lightfilter);
71
72 // Manually turn on the set flag and dirty flag since we didn't go through
73 // the set() method.
74 mAttributeUpdateMask.set(sLightFiltersKey.mIndex, true);
75 mAttributeSetMask.set(sLightFiltersKey.mIndex, true);
76 mDirty = true;
77}

79void
80LightFilterSet::remove(LightFilter* lightfilter)
81{
82 if (!mUpdateActive) {
83 std::stringstream errMsg;
84 errMsg << "LightFilter '" << lightfilter->getName() << "' can only be removed from"
85 " LightFilterSet '" << mName << "' between beginUpdate() and endUpdate()"
86 " calls.";
87 throw except::RuntimeError(errMsg.str());
88 }
89
90 // Retrieve a mutable reference to the lights attribute.
91 SceneObjectVector& lightfilters = getMutable(sLightFiltersKey);
92
93 // Binary search for the lightfilter.
94 SceneObjectVector::iterator removePoint = lowerBoundByName(lightfilters.begin(),
95 lightfilters.end(),
96 lightfilter);
97
98 // If found, remove it.
99 if (removePoint != lightfilters.end() && *removePoint == lightfilter) {
100 lightfilters.erase(removePoint);
101
102 // Manually turn on the set flag and dirty flag since we didn't go
103 // through the set() method.
104 mAttributeUpdateMask.set(sLightFiltersKey.mIndex, true);
105 mAttributeSetMask.set(sLightFiltersKey.mIndex, true);
106 mDirty = true;
107 }

110bool
111LightFilterSet::contains(const LightFilter* lightfilter) const
112{
113 // Retrieve a const reference to the lights attribute.
114 const SceneObjectVector& lightfilters = get(sLightFiltersKey);
115
116 // Binary search for the lightfilter.
117 SceneObjectVector::const_iterator iter = lowerBoundByName(lightfilters.begin(),
118 lightfilters.end(),
119 lightfilter);
120
121 return iter != lightfilters.end() && *iter == lightfilter;
122}
