17void
18accumLightAovs(pbr::TLState* pbrTls, const BundledOcclRay& occlRay, const FrameState& fs,
19 int numItems, const scene_rdl2::math::Color& matchMultiplier,
20 const scene_rdl2::math::Color* nonMatchMultiplier,
21 int flags)
22{
23 if (!fs.mLightAovs->hasEntries()) {
24 return;
25 }
26
27 for (int i = 0; i < numItems; ++i) {
28 const BundledOcclRayData *b =
29 static_cast<BundledOcclRayData *>(pbrTls->getListItem(occlRay.mDataPtrHandle, i));
30 if (b->mFlags & BundledOcclRayDataFlags::LPE) {
31 if (!(b->mLpeStateId)) break; // no more valid items in list
32
33 MNRY_ASSERT(!fs.mAovSchema->empty());
34 int lpeStateId = b->mLpeStateId;
35 MNRY_ASSERT(lpeStateId > 0);
36 if (lpeStateId > 0) {
37
38 scene_rdl2::math::Color nonMatchValue;
39 if (nonMatchMultiplier) {
40 nonMatchValue = b->mLpeRadiance * (*nonMatchMultiplier);
41 }
42 const scene_rdl2::math::Color *pNonMatchValue = nonMatchMultiplier ? &nonMatchValue : nullptr;
43
44 aovAccumLightAovsBundled(pbrTls, *fs.mAovSchema, b->mCameraId,
45 *fs.mLightAovs, b->mLpeRadiance * matchMultiplier,
46 pNonMatchValue, flags, lpeStateId, occlRay.mPixel, occlRay.mDeepDataHandle,
47 pbr::getFilm(occlRay.mTilePassAndFilm));
48 }
49 }
50 }
51}

53void
54accumVisibilityAovs(pbr::TLState* pbrTls, const BundledOcclRay& occlRay,
55 const FrameState& fs, const int numItems, float value)
56{
57 if (!fs.mLightAovs->hasVisibilityEntries()) {
58 return;
59 }
60
61 for (int i = 0; i < numItems; ++i) {
62 const BundledOcclRayData *b =
63 static_cast<BundledOcclRayData *>(pbrTls->getListItem(occlRay.mDataPtrHandle, i));
64 if ((b->mFlags & BundledOcclRayDataFlags::LPE) &&
65 (b->mFlags & BundledOcclRayDataFlags::LIGHT_SAMPLE)) {
66 if (!(b->mLpeStateId)) break; // no more valid items in list
67
68 MNRY_ASSERT(!fs.mAovSchema->empty());
69 const int lpeStateId = b->mLpeStateId;
70 MNRY_ASSERT(lpeStateId > 0);
71 if (lpeStateId > 0) {
72 if (aovAccumVisibilityAovsBundled(pbrTls, *fs.mAovSchema, b->mCameraId, *fs.mLightAovs,
73 scene_rdl2::math::Vec2f(value, 1.0f), lpeStateId, occlRay.mPixel, occlRay.mDeepDataHandle,
74 pbr::getFilm(occlRay.mTilePassAndFilm))) {
75
76 // we only need to add to the visibility buffer once per shadow ray
77 break;
78 }
79 }
80 }
81 }

98void
99fillBundledRadiance(pbr::TLState* pbrTls, BundledRadiance* dst, const BundledOcclRay& occlRay)
100{
101 const BundledOcclRayData *b = static_cast<BundledOcclRayData *>(
102 pbrTls->getListItem(occlRay.mDataPtrHandle, 0));
103
104 scene_rdl2::math::Color radiance = occlRay.mRadiance;
105
106
107 dst->mRadiance = RenderColor(radiance.r,
108 radiance.g,
109 radiance.b,
110 0.f);
111
112 dst->mPathPixelWeight = 0.f;
113 dst->mPixel = occlRay.mPixel;
114 dst->mSubPixelIndex = occlRay.mSubpixelIndex;
115 dst->mDeepDataHandle = pbrTls->acquireDeepData(occlRay.mDeepDataHandle);
116 // Cryptomatte only records pixel coverage information as seen from the camera; i.e. only primary rays add to
117 // the cryptomatte buffer, so occlusion rays don't need to acquire a handle to the cryptomatte data. (This
118 // differs from deep data, which records radiance information, and hence must acquire the deep data handle even
119 // for secondary rays.)
120 dst->mCryptomatteDataHandle = nullHandle;
121 dst->mTilePassAndFilm = occlRay.mTilePassAndFilm;
122 dst->mCameraId = b->mCameraId;
123}

125scene_rdl2::math::Color
126getTransmittance(pbr::TLState* pbrTls, const BundledOcclRay& occlRay)
127{
128 // is light uniform?
129 const BundledOcclRayData *b = static_cast<BundledOcclRayData *>(
130 pbrTls->getListItem(occlRay.mDataPtrHandle, 0));
131 mcrt_common::Ray trRay(occlRay.mOrigin, occlRay.mDir, occlRay.mMinT, occlRay.mMaxT, occlRay.mTime,
132 occlRay.mDepth + 1);
133
134 return pbrTls->mFs->mIntegrator->transmittance(pbrTls, trRay, occlRay.mPixel,
135 occlRay.mSubpixelIndex, occlRay.mSequenceID, b->mLight);
136}
