24void
25TestBase::shade()
26{
27 mcrt_common::ThreadLocalState *topLevelTls = mcrt_common::getFrameUpdateTLS();
28 shading::TLState *tls = MNRY_VERIFY(topLevelTls->mShadingTls.get());
29 scene_rdl2::alloc::Arena *arena = MNRY_VERIFY(tls)->mArena;
30 SCOPED_MEM(arena);
31
32 // setup a simple rdl scene consisting of a TestBase material
33 scene_rdl2::rdl2::SceneContext ctx;
34 ctx.setDsoPath(ctx.getDsoPath() + ":dso/material");
35 scene_rdl2::rdl2::Material *material =
36 ctx.createSceneObject("BaseMaterial", "/base")->asA<scene_rdl2::rdl2::Material>();
37 material->applyUpdates();
38
39 // all defaults, nothing varying
40 shading::Statev *statev = arena->alloc<shading::Statev>(CACHE_LINE_SIZE);
41 memset(statev, 0, sizeof(shading::Statev));
42 for (unsigned int i = 0; i < VLEN; ++i) {
43 statev->mN.x[i] = 1.f; // must have a unit normal
44 statev->mNg.x[i] = 1.f;
45 statev->mWo.x[i] = 1.f;
46 statev->mdPds.y[1] = 1.f;
47 }
48 Bsdfv *bsdfv = arena->alloc<Bsdfv>();
49
50 // Alloc an array of BsdfBuilderv
51 BsdfBuilderv *builderv = tls->mArena->allocArray<BsdfBuilderv>(1, CACHE_LINE_SIZE);
52
53 // helper to allow calling ispc function via func ptr below
54 typedef void (__cdecl * BsdfBuilderArrayInitFuncv)
55 (shading::TLState *tls,
56 unsigned numStatev,
57 const shading::Statev* state,
58 scene_rdl2::rdl2::Bsdfv* bsdfv,
59 shading::BsdfBuilderv* bsdfBuilderv,
60 SIMD_MASK_TYPE implicitMask);
61
62 // Initialize the BsdfBuilderv's (and the Bsdfv's) by calling ispc
63 // function via func ptr. This approach allows casting from c++ to
64 // ispc types:
65 // * shading::TLState -> ispc::ShadingTLState
66 // * shading::State -> ispc::State
67 BsdfBuilderArrayInitFuncv initBsdfBuilderArray = (BsdfBuilderArrayInitFuncv)ispc::getBsdfBuilderInitFunc();
68 initBsdfBuilderArray(tls, 1, statev, reinterpret_cast<scene_rdl2::rdl2::Bsdfv*>(bsdfv), builderv, scene_rdl2::util::sAllOnMask);
69
70 material->shadev(tls, 1, reinterpret_cast<const scene_rdl2::rdl2::Statev *>(statev), reinterpret_cast<scene_rdl2::rdl2::BsdfBuilderv *>(builderv));
71 CPPUNIT_ASSERT(bsdfv->mNumLobes == 2);
72 // specular
73 CPPUNIT_ASSERT(bsdfv->mLobes[0]->mName == BsdfLobeName::BSDF_LOBE_COOK_TORRANCE);
74 int allmask = 0;
75 for (unsigned int i = 0; i < VLEN; ++i) {
76 allmask |= (1 << i);
77 }
78 CPPUNIT_ASSERT(bsdfv->mLobes[0]->mMask == allmask);
79 CPPUNIT_ASSERT(bsdfv->mLobes[0]->mFresnel != NULL);
80 CPPUNIT_ASSERT(bsdfv->mLobes[0]->mFresnel->mType == FresnelType::FRESNEL_TYPE_SCHLICK_FRESNEL);
81 const SchlickFresnelv *specFresnel = reinterpret_cast<const SchlickFresnelv *>(bsdfv->mLobes[0]->mFresnel);
82 CPPUNIT_ASSERT(specFresnel->mMask == allmask);
83 for (unsigned int i = 0; i < VLEN; ++i) {
84 CPPUNIT_ASSERT(specFresnel->mSpec.r[i] == 0.1f &&
85 specFresnel->mSpec.g[i] == 0.1f &&
86 specFresnel->mSpec.b[i] == 0.1f);
87 CPPUNIT_ASSERT(specFresnel->mFactor[i] == 1.0f);
88 }
89 // directional diffuse is black
90 // translucency is black
91 // transmission is black
92 // diffuse
93 CPPUNIT_ASSERT(bsdfv->mLobes[1]->mName == BsdfLobeName::BSDF_LOBE_LAMBERT);
94 CPPUNIT_ASSERT(bsdfv->mLobes[1]->mMask == allmask);
95 CPPUNIT_ASSERT(bsdfv->mLobes[1]->mFresnel != NULL);
96 CPPUNIT_ASSERT(bsdfv->mLobes[1]->mFresnel->mType ==
97 FresnelType::FRESNEL_TYPE_ONE_MINUS_ROUGH_SCHLICK_FRESNEL);
98 const OneMinusRoughSchlickFresnelv *omrsf =
99 reinterpret_cast<const OneMinusRoughSchlickFresnelv *>(bsdfv->mLobes[1]->mFresnel);
100 CPPUNIT_ASSERT(omrsf->mMask == allmask);
101 for (unsigned int i = 0; i < VLEN; ++i) {
102 CPPUNIT_ASSERT(omrsf->mSpecRoughness[i] == 0.09f);
103 }
104 CPPUNIT_ASSERT(omrsf->mTopFresnel == specFresnel);
105 // self-emission
106 for (unsigned int i = 0; i < VLEN; ++i) {
107 CPPUNIT_ASSERT(bsdfv->mSelfEmission.r[i] == 0.0f &&
108 bsdfv->mSelfEmission.g[i] == 0.0f &&
109 bsdfv->mSelfEmission.b[i] == 0.0f);
110 }
111}
