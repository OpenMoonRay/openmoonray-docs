100void
101TestBssrdf::testDistribution(const shading::Bssrdf &bssrdf, int sampleCount, char *suffix)
102{
103 FloatArray u, v;
104 generateSamples(sampleCount, u, v);
105
106 FloatArray x, red, grn, blu;
107 x.resize(sampleCount);
108 red.resize(sampleCount);
109 grn.resize(sampleCount);
110 blu.resize(sampleCount);
111
112 for (int i=0; i < sampleCount; i++) {
113 Vec3f xi;
114 float r;
115 /* float pdf = */ bssrdf.sampleLocal(u[i], v[i], xi, r);
116 u[i] = xi[0];
117 v[i] = xi[1];
118
119 Color rd = bssrdf.eval(r);
120 x[i] = r;
121 red[i] = rd[0];
122 grn[i] = rd[1];
123 blu[i] = rd[2];
124 }
125
126 std::string prefix = std::string("mat/testBssrdfDistribution-") + suffix;
127 printDistribution(u, v, x, red, grn, blu, prefix);

153void
154TestBssrdf::testPdfIntegral(const shading::Bssrdf &bssrdf, int sampleCount)
155{
156 sampleCount *= 100;
157 FloatArray u, v;
158 generateSamples(sampleCount, u, v);
159
160 float maxRadius = bssrdf.getMaxRadius();
161
162 Color integralUniform = sBlack;
163 float pdf = 1.0f / (4.0f * maxRadius * maxRadius);
164 for (int i=0; i < sampleCount; i++) {
165 Vec3f xi;
166 xi[0] = (u[i] - 0.5f) * 2.0f * maxRadius;
167 xi[1] = (v[i] - 0.5f) * 2.0f * maxRadius;
168 xi[2] = 0.0f;
169
170 float r = xi.length();
171
172 integralUniform += Color(bssrdf.pdfLocal(r));
173 }
174 integralUniform /= sampleCount * pdf;
175
176 printIntegralUniform(integralUniform);
177}

229void
230TestBssrdf::testIntegral(const shading::Bssrdf &bssrdf, int sampleCount)
231{
232 uint32_t scramble[2];
233 FloatArray u, v;
234 generateSamples(sampleCount, scramble, u, v);
235
236 // Compute diffuse reflectance using importance sampling
237 float avgSampledRadius = 0.0f;
238 float maxSampledRadius = 0.0f;
239 Color integralImp = sBlack;
240 for (int i=0; i < sampleCount; i++) {
241 Vec3f xi;
242 float r;
243 float pdf = bssrdf.sampleLocal(u[i], v[i], xi, r);
244
245 avgSampledRadius += r;
246 integralImp += bssrdf.eval(r) / pdf;
247 maxSampledRadius = max(r, maxSampledRadius);
248 }
249 avgSampledRadius /= sampleCount;
250 integralImp /= sampleCount;
251
252
253 // Compute diffuse reflectance using uniform sampling
254 sampleCount *= 10;
255 generate2DHaltonSequence(sampleCount, scramble, u, v);
256
257 float maxRadius = bssrdf.getMaxRadius();
258
259 Color integralUniform = sBlack;
260 float pdf = 1.0f / (4.0f * maxRadius * maxRadius);
261 for (int i=0; i < sampleCount; i++) {
262 Vec3f xi;
263 xi[0] = (u[i] - 0.5f) * 2.0f * maxRadius;
264 xi[1] = (v[i] - 0.5f) * 2.0f * maxRadius;
265 xi[2] = 0.0f;
266
267 float r = xi.length();
268
269 integralUniform += bssrdf.eval(r);
270 }
271 integralUniform /= sampleCount * pdf;
272
273 printIntegralResults(integralImp, integralUniform, bssrdf.eval(maxRadius),
274 avgSampledRadius, maxSampledRadius, maxRadius);
275
