275template <typename T, typename Iterator>
276Iterator
277partitionAndSortByName(Iterator begin, Iterator end)
278{
279 // Partition the objects of type T to the front of the range.
280 auto rest = std::partition(begin, end, [](const SceneObject* obj) {
281 return obj->isA<T>();
282 });
283
284 // Sort the leading partition by object name.
285 std::sort(begin, rest, [](const SceneObject* a, const SceneObject* b) {
286 return a->getName() < b->getName();
287 });
288
289 // Return an iterator to start of the unpartitioned and unsorted objects.
290 return rest;
291}

293// Sort the geometries and parts be names when writing out trace sets and layers
294std::vector<size_t>
295sortGeometriesAndParts(const SceneObjectIndexable& geometries,
296 const StringVector& parts)
297{
298 std::vector<std::size_t> order(geometries.size());
299 std::iota(order.begin(), order.end(), 0u);
300 std::sort(order.begin(), order.end(), [&geometries, &parts](std::size_t a, std::size_t b) -> bool {
301 // Order first by geometry name. If geometry names are the same, order
302 // by part name.
303 const std::string& geomNameA = geometries[a]->getName();
304 const std::string& geomNameB = geometries[b]->getName();
305 return (geomNameA != geomNameB) ? geomNameA < geomNameB : parts[a] < parts[b];
306 });
307 return order;
308}

312template <typename T, typename F>
313std::string
314AsciiWriter::vectorToString(const SceneObject* so, const Attribute* attr,
315 AttributeTimestep timestep, F predicate) const
316{
317 const T& vec = so->get(AttributeKey<T>(*attr), timestep);
318 std::ostringstream s;
319 s << '{';
320 bool first = true;
321 size_t elemsThisLine = 0;
322 for (auto iter = vec.begin(); iter != vec.end(); ++iter) {
323 if (first) {
324 first = false;
325 } else {
326 s << ",";
327 }
328 if (mElemsPerLine > 0 && elemsThisLine == mElemsPerLine) {
329 s << "\n" << mIndent << " ";
330 elemsThisLine = 0;
331 } else {
332 s << " ";
333 }
334 s << predicate(*iter);
335 ++elemsThisLine;
336 }
337 s << '}';
338 return s.str();
339}

347bool
348AsciiWriter::skipAttributeValue(const SceneObject* so, const Attribute* attr) const
349{
350 if (mDeltaEncoding && !so->mAttributeSetMask.test(attr->mIndex)) {
351 return true;
352 }
353 if (mSkipDefaults && !mDeltaEncoding && so->isDefaultAndUnbound(*attr)) {
354 return true;
355 }
356 if (vectorSize(*so,*attr) > mMaxVectorSize) {
357 return true;
358 }
359 return false;

362std::vector<const SceneObject*>
363AsciiWriter::generateWriteOrder() const
364{
365 std::vector<const SceneObject*> order;
366
367 // Write all the other objects. Skip the SceneVariables, we already put
368 // them first.
369 for (auto iter = mContext.beginSceneObject();
370 iter != mContext.endSceneObject(); ++iter) {
371 if (skipSceneObject(iter->second)) continue;
372
373 // Skip the SceneVariables, they're handled separately.
374 if (iter->second->getName() == "__SceneVariables__") continue;
375
376 order.push_back(iter->second);
377 }
378
379 // For now, we order objects by a simple heuristic which tends to put
380 // dependencies first (maps before materials, materials before layers, etc.)
381 // These aren't actual creation order dependencies (the "create or return
382 // existing" semantics of createSceneObject() are working for us here), but
383 // rather dependencies for using Lua variables to refer to SceneObjects as
384 // opposed to using their long-form references (MmGeometry("teapot")).
385 auto rest = partitionAndSortByName<Map>(order.begin(), order.end());
386 rest = partitionAndSortByName<Joint>(rest, order.end());
387 rest = partitionAndSortByName<Geometry>(rest, order.end());
388 rest = partitionAndSortByName<GeometrySet>(rest, order.end());
389 rest = partitionAndSortByName<EnvMap>(rest, order.end());
390 rest = partitionAndSortByName<TraceSet>(rest, order.end());
391 rest = partitionAndSortByName<Material>(rest, order.end());
392 rest = partitionAndSortByName<Light>(rest, order.end());
393 rest = partitionAndSortByName<LightFilterSet>(rest, order.end());
394 rest = partitionAndSortByName<LightSet>(rest, order.end());
395 rest = partitionAndSortByName<ShadowSet>(rest, order.end());
396 rest = partitionAndSortByName<Layer>(rest, order.end());
397 rest = partitionAndSortByName<Camera>(rest, order.end());
398 rest = partitionAndSortByName<Metadata>(rest, order.end());
399
400 return order;
401}

432std::string
433AsciiWriter::boundValueToString(const SceneObject* so, const Attribute* attr) const
434{
435 std::ostringstream s;
436
437 bool isBinding = false;
438 const SceneObject* boundObject = nullptr;
439 try {
440 boundObject = fetchBinding(so, attr);
441 isBinding = true;
442 } catch (const except::TypeError&) {
443 isBinding = false;
444 }
445
446 if (isBinding) {
447 s << "bind(" << sceneObjectRef(boundObject);
448 }
449
450 if ((boundObject || isBinding) && !skipAttributeValue(so, attr)) {
451 s << ", ";
452 }
453
454 if (!skipAttributeValue(so, attr)) {
455 s << blurredValueToString(so, attr);
456 }
457
458 if (isBinding) {
459 s << ')';
460 }
461
462 return s.str();
463}

600void
601AsciiWriter::writeSceneObject(std::ostream& out, const SceneObject* so) const
602{
603 const SceneClass& sc = so->getSceneClass();
604 for (auto iter = sc.beginAttributes(); iter != sc.endAttributes(); ++iter) {
605 const Attribute* attr = *iter;
606
607 // Only skip the attribute if it doesn't have a binding.
608 try {
609 if (!fetchBinding(so, attr) && skipAttributeValue(so, attr)) {
610 continue;
611 }
612 } catch (const except::TypeError&) {
613 if (skipAttributeValue(so, attr)) {
614 continue;
615 }
616 }
617
618 out << mIndent << "[\"" << attr->getName() << "\"] = " <<
619 boundValueToString(so, attr) << ",\n";
620 }
621}

623void
624AsciiWriter::writeTraceSet(std::ostream& out, const TraceSet* traceSet) const
625{
626 const auto& geometries = traceSet->get(TraceSet::sGeometriesKey);
627 const auto& parts = traceSet->get(TraceSet::sPartsKey);
628
629 // Sort the trace set assignments by geometry name, then by part name.
630 const std::vector<std::size_t>& order = sortGeometriesAndParts(geometries, parts);
631
632 // Write out each assignment in the trace set.
633 for (std::size_t i = 0; i < order.size(); ++i) {
634 std::size_t index = order[i];
635 out << mIndent << '{' << sceneObjectRef(geometries[index]) << ", \"" <<
636 parts[index] << "\"},\n";
637 }
638}

640void
641AsciiWriter::writeLayer(std::ostream& out, const Layer* layer) const
642{
643 const auto& geometries = layer->get(Layer::sGeometriesKey);
644 const auto& parts = layer->get(Layer::sPartsKey);
645 const auto& displacements = layer->get(Layer::sDisplacementsKey);
646 const auto& volumeShaders = layer->get(Layer::sVolumeShadersKey);
647 const auto& surfaceShaders = layer->get(Layer::sSurfaceShadersKey);
648 const auto& lightSets = layer->get(Layer::sLightSetsKey);
649 const auto& lightFilterSets = layer->get(Layer::sLightFilterSetsKey);
650 const auto& shadowSets = layer->get(Layer::sShadowSetsKey);
651 const auto& shadowReceiverSets = layer->get(Layer::sShadowReceiverSetsKey);
652
653 // Sort the layer assignments by geometry name, then by part name.
654 const std::vector<std::size_t>& order = sortGeometriesAndParts(geometries, parts);
655
656 // Write out each assignment in the layer.
657 for (std::size_t i = 0; i < order.size(); ++i) {
658 std::size_t index = order[i];
659 out << mIndent << '{' << sceneObjectRef(geometries[index]) << ", \"" <<
660 parts[index] << "\", " <<
661 sceneObjectRef(surfaceShaders[index]) << ", " <<
662 sceneObjectRef(lightSets[index]) << ", " <<
663 sceneObjectRef(displacements[index]) << ", " <<
664 sceneObjectRef(volumeShaders[index]) << ", " <<
665 sceneObjectRef(lightFilterSets[index]) << ", " <<
666 sceneObjectRef(shadowSets[index]) << ", " <<
667 sceneObjectRef(shadowReceiverSets[index]) << "},\n";
668 }
669}

671void
672AsciiWriter::writeMetadata(std::ostream& out, const Metadata* metadata) const
673{
674 const auto& names = metadata->get(Metadata::sNameKey);
675 const auto& types = metadata->get(Metadata::sTypeKey);
676 const auto& values = metadata->get(Metadata::sValueKey);
677
678 // Sort metadata by name.
679 std::vector<std::size_t> order(names.size());
680 std::iota(order.begin(), order.end(), 0u);
681 std::sort(order.begin(), order.end(), [&names](std::size_t a, std::size_t b) -> bool {
682 return names[a] < names[b];
683 });
684
685 // Write out data elements.
686 for (std::size_t i = 0; i < order.size(); ++i) {
687 std::size_t index = order[i];
688 out << mIndent << "{\"" << names[index] << "\", \"" <<
689 types[index] << "\", \"" <<
690 values[index] << "\"},\n";
691 }
692}

717AsciiWriter::toStream(std::ostream& output) const
718{
719 // Write the SceneVariables first.
720 const SceneObject* sceneVars = mContext.getSceneObject("__SceneVariables__");
721 if (!skipSceneObject(sceneVars)) {
722 output << "SceneVariables {\n";
723 writeSceneObject(output, sceneVars);
724 output << "}\n\n";
725 }
726
727 // Order the SceneObjects by the order we intend to write them.
728 auto writeOrder = generateWriteOrder();
729
730 // Write out each object.
731 bool first = true;
732 for (auto iter = writeOrder.begin(); iter != writeOrder.end(); ++iter) {
733 const SceneObject* so = *iter;
734
735 // Write the object header.
736 if (!first) {
737 output << "\n";
738 } else {
739 first = false;
740 }
741 output << sceneObjectRef(so) << " {\n";
742
743 // Write the attributes block, with special cases for sets and layers.
744 if (so->isA<GeometrySet>()) {
745 writeSet(output, so->asA<GeometrySet>()->getGeometries());
746 } else if (so->isA<LightFilterSet>()) {
747 writeSet(output, so->asA<LightFilterSet>()->getLightFilters());
748 } else if (so->isA<ShadowSet>()) {
749 writeSet(output, so->asA<ShadowSet>()->getLights());
750 } else if (so->isA<ShadowReceiverSet>()) {
751 writeSet(output, so->asA<ShadowReceiverSet>()->getGeometries());
752 } else if (so->isA<LightSet>()) {
753 writeSet(output, so->asA<LightSet>()->getLights());
754 } else if (so->isA<Layer>()) {
755 writeLayer(output, so->asA<Layer>());
756 } else if (so->isA<TraceSet>()) {
757 writeTraceSet(output, so->asA<TraceSet>());
758 } else if (so->isA<Metadata>()) {
759 writeMetadata(output, so->asA<Metadata>());
760 } else {
761 writeSceneObject(output, so);
762 }
763
764 // Write the object footer.
765 output << "}\n";
766 }
767}
768
769std::string
770AsciiWriter::toString() const
771{
772 std::ostringstream output;
773 toStream(output);
774 return output.str();
775}
