39void
40TestSnapshotUtil::testHeatMapWeight()
41{
42 int w = sTileReso * 240; // = 1920
43 int h = sTileReso * 135; // = 1080
44 int updatePixTotal = 4096;
45
46 std::vector<double> orgV(w * h);
47 std::vector<float> orgW(w * h);
48 setupBuff(orgV);
49 setupBuff(orgW);
50
51 std::vector<double> dstV(orgV);
52 std::vector<float> dstW(orgW);
53 std::vector<double> srcV(orgV);
54 std::vector<float> srcW(orgW);
55 std::vector<int> updateList;
56 updateBuff(updatePixTotal, w, h,
57 [&](const int offset) { // updatePixelFunc
58 srcV[offset] *= 0.5;
59 srcW[offset] *= 0.5f;
60 },
61 updateList);
62
63#ifdef TIMING_TEST
64 int timingTestLoopMax = 128; // for performance test
65 bool doCompare = true; // for performance test
66#else // else TIMING_TEST
67 int timingTestLoopMax = 1;
68 bool doCompare = false;
69#endif // end !TIMING_TEST
70 snapshotTimingCompare
71 (w, h, timingTestLoopMax, doCompare,
72 [&]() { // resetDataFunc
73 dstV = orgV;
74 dstW = orgW;
75 },
76 [&](int offsetItem) -> uint64_t { // snapshotTileFuncA
77 return fb_util::SnapshotUtil::snapshotTileHeatMapWeight
78 ((uint64_t *)((uintptr_t)(&dstV[0]) + offsetItem * sizeof(double)),
79 (uint32_t *)((uintptr_t)(&dstW[0]) + offsetItem * sizeof(float)),
80 (uint64_t *)((uintptr_t)(&srcV[0]) + offsetItem * sizeof(double)),
81 (uint32_t *)((uintptr_t)(&srcW[0]) + offsetItem * sizeof(float)));
82 },
83 [&](int offsetItem) -> uint64_t { // snapshotTileFuncB
84 return fb_util::SnapshotUtil::snapshotTileHeatMapWeight_SISD
85 ((uint64_t *)((uintptr_t)(&dstV[0]) + offsetItem * sizeof(double)),
86 (uint32_t *)((uintptr_t)(&dstW[0]) + offsetItem * sizeof(float)),
87 (uint64_t *)((uintptr_t)(&srcV[0]) + offsetItem * sizeof(double)),
88 (uint32_t *)((uintptr_t)(&srcW[0]) + offsetItem * sizeof(float)));
89 },
90 [&](std::vector<uint64_t> &pixMaskBuff) -> bool { // verifyFunc
91 return verifyPixMask(updateList, pixMaskBuff) && (dstV == srcV) && (dstW == srcW);
92 });
93}

732void
733TestSnapshotUtil::setupPixMaskBuff(float emptyMaskFraction,
734 float fullMaskFraction,
735 std::vector<uint64_t> &buff) const
736{
737 std::random_device rnd;
738 std::mt19937 mt(rnd());
739 std::uniform_real_distribution<> rand01(0.0f, 1.0f);
740 std::uniform_int_distribution<> randPixTotal(1, 63);
741
742 auto randomPixMask = [&](int totalActivePix) -> uint64_t {
743 if (totalActivePix > 63) totalActivePix = 63;
744
745 std::vector<int> pixIdBuff(64);
746 std::iota(pixIdBuff.begin(), pixIdBuff.end(), 0);
747 std::shuffle(pixIdBuff.begin(), pixIdBuff.end(), mt);
748
749 uint64_t mask = 0x0;
750 for (size_t i = 0; i < (size_t)totalActivePix; ++i) {
751 int currPixId = pixIdBuff[i];
752 mask |= ((uint64_t)0x1 << currPixId);
753 }
754 return mask;
755 };
756
757
758 for (size_t i = 0; i < buff.size(); ++i) {
759 float r0 = rand01(mt);
760
761 uint64_t currMask = 0x0;
762 if (r0 < emptyMaskFraction) {
763 currMask = 0x0; // empty mask
764 } else if (r0 < (emptyMaskFraction + fullMaskFraction)) {
765 currMask = (uint64_t)0xffffffffffffffff; // full mask
766 } else {
767 currMask = randomPixMask(randPixTotal(mt));
768 }
769 buff[i] = currMask;
770 }
771}

773void
774TestSnapshotUtil::updateBuff(const int updateTotal,
775 const int w, // should be tile aligned resolution
776 const int h, // should be tile aligned resolution
777 std::function<void(const int offset)> updatePixelFunc,
778 std::vector<int> &updateList) const
779{
780 int totalTile = w / sTileReso * h / sTileReso;
781
782 std::random_device rnd;
783 std::mt19937 mt(rnd());
784 std::uniform_int_distribution<> randTileId(0, totalTile - 1);
785 std::uniform_int_distribution<> randXY(0, sTileReso - 1);
786
787 auto randomPixel = [&]() -> int {
788 // randomly pick single pixel position
789 int offset = 0;
790 while (1) { // ignore duplicate pixel position
791 int tileId = randTileId(mt); // tileId
792 int xPixId = randXY(mt); // pixel X position inside tile
793 int yPixId = randXY(mt); // pixel Y position inside tile
794
795 offset = tileId * sTileReso * sTileReso + yPixId * sTileReso + xPixId;
796 if (std::find(updateList.begin(), updateList.end(), offset) == updateList.end()) break;
797 }
798 updateList.push_back(offset); // store offset value inside updateList for duplicate check and verify logic
799 return offset;
800 };
801
802 for (int i = 0; i < updateTotal; ++i) {
803 if (updateList.size() >= (size_t)(w * h)) break; // escape if all the pixels are updated
804 updatePixelFunc(randomPixel());
805 }
806
807 std::sort(updateList.begin(), updateList.end()); // sort updateList for verify operation
808}

890bool
891TestSnapshotUtil::verifyPixMask(std::vector<int> &updateList, std::vector<uint64_t> &pixMaskBuff) const
892{
893 auto verifyMask = [](int tileId, const uint64_t &mask, std::vector<int> &updateList, int &total) -> bool {
894 for (int yId = 0; yId < sTileReso; ++yId) {
895 for (int xId = 0; xId < sTileReso; ++xId) {
896 int shift = yId * sTileReso + xId;
897 if (((uint64_t)(mask >> shift) & (uint64_t)0x1) == (uint64_t)0x1) {
898 int offset = tileId * sTileReso * sTileReso + yId * sTileReso + xId;
899 if (updateList[total] != offset) return false;
900 total++;
901 }
902 }
903 }
904 return true;
905 };
906
907 int total = 0;
908 for (size_t tileId = 0; tileId < pixMaskBuff.size(); ++tileId) {
909 if (pixMaskBuff[tileId]) {
910 if (!verifyMask(tileId, pixMaskBuff[tileId], updateList, total)) {
911 return false;
912 }
913 }
914 }
915 return true;
