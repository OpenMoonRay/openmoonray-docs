38// The HitGroupData contains all of the Primitive data needed for Optix to execute the
39// programs on the GPU.
40
41struct HitGroupData
42{
43 // Properties common to all primitives
44 bool mIsSingleSided;
45 bool mIsNormalReversed;
46 bool mVisibleShadow;
47 int *mAssignmentIds;
48 unsigned mNumShadowLinkEntries;
49 int* mShadowLinkAssignmentIds;
50 unsigned long long *mShadowLinkLightIds;

52 // Primitive type-specific properties. This is all unioned together because
53 // we need a fixed-size data structure for the Shader Binding Table entries.
54 // Note that there is no type field telling us the primitive type. We don't
55 // need it because each primitive type is handled by its own programs,
56 // (see program groups), thus it is already known.
57 // Also note that we only store raw pointers below while on the host side we
58 // use GPUBuffer objects. These pointers are obtained from the host-side
59 // GPUBuffer objects.
60 union {
61 struct {
62 GPUXform mL2P;
63 GPUXform mP2L;
64 float mLength;
65 float mHeight;
66 float mWidth;
67 } box;
68 struct {
69 int mMotionSamplesCount;
70 unsigned int mSegmentsPerCurve;
71 const unsigned int* mIndices;
72 int mNumControlPoints;
73 const float4* mControlPoints;
74 } curve;
75 struct {
76 int mMotionSamplesCount;
77 const float4* mPoints;
78 } points;
79 struct {
80 GPUXform mL2P;
81 GPUXform mP2L;
82 float mRadius;
83 float mPhiMax;
84 float mZMin;
85 float mZMax;
86 } sphere;
87 // There is no TriMesh or RoundCurves here because the intersection program that uses
88 // these data is built in to Optix and these data have already been passed in via
89 // the Optix API elsewhere.
90 };
91};
