28const unsigned MAX_VERTS_PER_BLOCK_SHIFT = scene_rdl2::util::countTrailingZeros(MAX_VERTS_PER_BLOCK);

30// hands out the next unused value in sequence if an id hasn't been seen
31// before, otherwise hands out the previous id it was assigned
32template<typename FROM, typename TO>
33struct IdRemapper : public std::map<FROM, TO>
34{
35 TO remapId(FROM id)
36 {
37 auto it = this->find(id);
38 if (it != this->end())
39 return it->second;
40 TO newId = TO(this->size());
41 this->emplace(id, newId);
42 return newId;
43 }
44
45 typedef std::map<FROM, TO> MapType;
46};
47
48finline uint64_t
49fullId(uint32_t vertId, uint16_t threadId)
50{
51 return (uint64_t(threadId) << 32) | uint64_t(vertId);
52}
53
54finline uint64_t
55fullId(DebugRayVertex const &vert)
56{
57 return fullId(vert.mId, vert.mThreadId);
58}
59
60finline uint64_t
61fullParentId(DebugRayVertex const &vert)
62{
63 return fullId(vert.mParentId, vert.mParentThreadId);
64}

117void
118DebugRayRecorder::record()
119{
120 cleanUp();
121
122 mNextId = 0;
123 mEndId = sMaxVertsPerThread;
124
125 if (sMaxVertsPerThread) {
126 VertexBlock *vertBlock = new VertexBlock;
127 vertBlock->reserve(MAX_VERTS_PER_BLOCK);
128 mVertexBlocks.push_back(vertBlock);
129 }
130}

132void
133DebugRayRecorder::stopRecording()
134{
135 mEndId = mNextId;
136}

138void
139DebugRayRecorder::cleanUp()
140{
141 mEndId = 0;
142 mNextId = 0;
143
144 for (auto it = mVertexBlocks.begin(); it != mVertexBlocks.end(); ++it) {
145 delete *it;
146 }
147
148 mVertexBlocks.clear();
149
150 memset(&mDummyVertex, 0, sizeof(mDummyVertex));
151 mDummyVertex.mScreenX = uint16_t(-1);
152}

154DebugRayVertex *
155DebugRayRecorder::startNewRay(Vec3f const &origin, unsigned screenX, unsigned screenY, uint32_t userTags)
156{
157 if (int(screenX) >= sActiveViewport.lower.x && int(screenX) <= sActiveViewport.upper.x &&
158 int(screenY) >= sActiveViewport.lower.y && int(screenY) <= sActiveViewport.upper.y &&
159 mNextId < mEndId) {
160
161 DebugRayVertex *vert = allocRayVertex();
162
163 vert->mHitPoint = origin;
164 vert->mUserTags = userTags;
165 vert->mParentId = NULL_RAY_VERTEX_ID;
166 vert->mDepth = uint8_t(-1);
167 vert->mScreenX = uint16_t(screenX);
168 vert->mScreenY = uint16_t(screenY);
169 vert->mParentThreadId = NULL_THREAD_ID;
170
171 return vert;
172 }
173
174 MNRY_ASSERT(mDummyVertex.mScreenX == uint16_t(-1));
175
176 return &mDummyVertex;
177}
178
179DebugRayVertex *
180DebugRayRecorder::extendRay(DebugRayVertex const *parent)
181{
182 MNRY_ASSERT(parent);
183
184 if (parent->mScreenX != uint16_t(-1) && mNextId < mEndId) {
185
186 // we store depths in 8-bits, check that we don't overflow
187 // the value 0xff is reserved
188 MNRY_ASSERT(parent->mDepth != 254);
189
190 DebugRayVertex *vert = allocRayVertex();
191
192 vert->mParentId = parent->mId;
193 vert->mDepth = parent->mDepth + 1;
194 vert->mScreenX = parent->mScreenX;
195 vert->mScreenY = parent->mScreenY;
196 vert->mParentThreadId = parent->mThreadId;
197
198 return vert;
199 }
200
201 MNRY_ASSERT(mDummyVertex.mScreenX == uint16_t(-1));
202
203 return &mDummyVertex;
204}

231DebugRayVertex *
232DebugRayRecorder::allocRayVertex()
233{
234 // we have checked this is true higher up
235 MNRY_ASSERT(mNextId < mEndId);
236
237 VertexBlock *vertBlock = mVertexBlocks.back();
238 if (vertBlock->size() == MAX_VERTS_PER_BLOCK) {
239 // time to allocate a new list
240 vertBlock = new VertexBlock;
241 vertBlock->reserve(MAX_VERTS_PER_BLOCK);
242 mVertexBlocks.push_back(vertBlock);
243 }
244
245 vertBlock->push_back(DebugRayVertex());
246
247 DebugRayVertex *vert = &vertBlock->back();
248 memset(vert, 0, sizeof(DebugRayVertex));
249
250 // fill in common fields between parent and child vertices
251 vert->mContribution = Vec3f(1.0f, 0.41f, 0.71f); // hot pink
252 vert->mId = mNextId++;
253 vert->mThreadId = mThreadId;
254
255 MNRY_ASSERT(getRay(mNextId - 1) == vert);
256
257 return vert;
258}
259
260DebugRayVertex *
261DebugRayRecorder::getRay(uint32_t id)
262{
263 if (id < mEndId) {
264 size_t listIdx = id >> MAX_VERTS_PER_BLOCK_SHIFT;
265 size_t elemIdx = id & (MAX_VERTS_PER_BLOCK - 1);
266
267 if (listIdx < mVertexBlocks.size() && elemIdx < mVertexBlocks[listIdx]->size()) {
268 return &((*mVertexBlocks[listIdx])[elemIdx]);
269 }
270 }
271 return nullptr;
272}

276bool
277DebugRayBuilder::build(unsigned width, unsigned height, std::vector<DebugRayRecorder *> const &recorders)
278{
279 cleanUp();
280
281 if (recorders.empty() || !width || !height) {
282 return false;
283 }
284
285 mWidth = uint32_t(width);
286 mHeight = uint32_t(height);
287
288 //
289 // build up a rich tree structure from the inputs which we can manipulate
290 //
291
292 // compute total amount of debug vertices
293 size_t numSrcVertices = 0;
294 for (size_t i = 0; i < recorders.size(); ++i) {
295 numSrcVertices += recorders[i]->getNumRayVertices();
296 }
297
298 if (!numSrcVertices) {
299 return false;
300 }
301
302 //
303 // linearize the full 48-bit sparse address space into compact
304 // sequential 32-bit ids and assign all rays to corresponding nodes
305 //
306
307 // add 1 to account for a master root node which we insert into the node tree
308 mNodes.resize(numSrcVertices + 1);
309 mNodes[0].mVertex = nullptr;
310 mNodes[0].mParent = nullptr;
311 mNodes[0].mNumSubnodes = 1; // start at 1 to account for self
312
313 {
314 size_t numNodes = 1;
315 IdRemapper<uint64_t, uint32_t> remapper;
316
317 // reserve 0 as the parent id
318 remapper.remapId(fullId(NULL_RAY_VERTEX_ID, NULL_THREAD_ID));
319
320 for (size_t i = 0; i < recorders.size(); ++i) {
321 DebugRayRecorder *recorder = recorders[i];
322 for (size_t iblock = 0; iblock < recorder->mVertexBlocks.size(); ++iblock) {
323 DebugRayRecorder::VertexBlock &vertBlock = *recorders[i]->mVertexBlocks[iblock];
324 for (size_t ivert = 0; ivert < vertBlock.size(); ++ivert, ++numNodes) {
325 DebugRayVertex *vert = &vertBlock[ivert];
326 MNRY_ASSERT(vert->mScreenX < mWidth && vert->mScreenY < mHeight);
327
328 vert->mParentId = remapper.remapId(fullParentId(*vert));
329 vert->mId = remapper.remapId(fullId(*vert));
330
331 // verify the index of the vert matches up with its location in the
332 // mNodes array
333 MNRY_ASSERT(vert->mId == numNodes);
334
335 Node *node = &mNodes[numNodes];
336 node->mVertex = &vertBlock[ivert];
337 node->mParent = nullptr; // this gets hookup up later
338 node->mNumSubnodes = 1; // start at 1 to account for self
339 }
340 }
341 }
342
343 MNRY_ASSERT(numNodes == mNodes.size());
344
345 // all ids are now sequential! we can release the IdRemapper
346 }
347
348 // hook up all parents and children
349 for (size_t i = 1; i < mNodes.size(); ++i) {
350 Node *node = &mNodes[i];
351 uint32_t parentId = node->mVertex->mParentId;
352
353 node->mParent = &mNodes[parentId];
354 node->mParent->mChildren.insert(node);
355 }
356
357 // verify that there are no orphaned nodes, and we have reasonable values
358 // for hit points
359 MNRY_DURING_ASSERTS
360 (
361 for (size_t i = 0; i < mNodes.size(); ++i) {
362 Node *node = &mNodes[i];
363 MNRY_ASSERT(node->mParent || !node->mChildren.empty());
364 if (node->mVertex) {
365 MNRY_ASSERT(isValidFloat(node->mVertex->mHitPoint.x));
366 MNRY_ASSERT(isValidFloat(node->mVertex->mHitPoint.y));
367 MNRY_ASSERT(isValidFloat(node->mVertex->mHitPoint.z));
368 }
369 }
370 );
371
372 //
373 // At this point, the children in each node are sorted by pixel location.
374 // To get the final ordering which we want to write out, we just need to
375 // do a depth first preorder traversal and write out vertices in that order.
376 //
377 std::vector<uint32_t> ordering;
378 ordering.reserve(mNodes.size());
379 depthFirstPreorderTraversal(&mNodes[0], &mNodes[0], &ordering);
380
381 MNRY_ASSERT(ordering[0] == 0);
382 MNRY_ASSERT(ordering.size() <= mNodes.size());
383 MNRY_ASSERT(mNodes[0].mNumSubnodes == ordering.size());
384
385 if (ordering.size() == 1) {
386 return false;
387 }
388
389 //
390 // copy DebugRays with the final ordering, whilst getting rid of temporary master root node
391 //
392
393 IdRemapper<uint32_t, uint32_t> remapper;
394
395 size_t numSortedVerts = ordering.size() - 1;
396 mSortedVertices.reserve(numSortedVerts);
397
398 // note - skip over master root node so we're back to the original set of inputs
399 for (size_t i = 1; i < ordering.size(); ++i) {
400 Node *node = &mNodes[ordering[i]];
401 DebugRayVertex *vert = node->mVertex;
402
403 MNRY_ASSERT(vert->mId != NULL_RAY_VERTEX_ID && vert->mParentId != NULL_RAY_VERTEX_ID);
404
405 if (vert->mParentId == 0) {
406 MNRY_ASSERT(node->mParent == &mNodes[0]);
407 vert->mParentId = NULL_RAY_VERTEX_ID;
408 } else {
409 vert->mParentId = remapper.remapId(vert->mParentId);
410 }
411
412 vert->mId = remapper.remapId(vert->mId);
413
414 mSortedVertices.push_back(*vert);
415 mSortedVertices.back().mNumSubnodes = node->mNumSubnodes;
416 }
417
418 MNRY_ASSERT(mSortedVertices.size() == numSortedVerts);
419 MNRY_ASSERT(remapper.size() == numSortedVerts);
420
421 //
422 // clean up temp memory
423 //
424 mNodes.clear();
425
426 //
427 // build up table of primary vert entry points
428 //
429 MNRY_DURING_ASSERTS(size_t prevPrimaryRay = 0);
430
431 for (size_t i = 0; i < mSortedVertices.size(); ++i) {
432 DebugRayVertex const &curr = mSortedVertices[i];
433 if (curr.mParentId == NULL_RAY_VERTEX_ID) {
434 mPrimaryRayIndices.push_back(uint32_t(i));
435
436 MNRY_DURING_ASSERTS
437 (
438 if (mPrimaryRayIndices.size() > 1) {
439 DebugRayVertex const &prev = mSortedVertices[prevPrimaryRay];
440
441 // debug validation - check everything is sorted by screen location
442 MNRY_ASSERT(prev.mId < curr.mId);
443 MNRY_ASSERT(prev.mScreenY * mWidth + prev.mScreenX <=
444 curr.mScreenY * mWidth + curr.mScreenX);
445
446 // debug validation - check the numbers of subnodes make sense
447 MNRY_ASSERT(prev.mNumSubnodes == i - prevPrimaryRay);
448 }
449 );
450
451 MNRY_DURING_ASSERTS(prevPrimaryRay = i);
452 }
453 }
454
455 //
456 // transform everything into world space before saving
457 //
458 Mat4f render2world = DebugRayRecorder::sRender2World;
459 if (render2world != Mat4f(one)) {
460
461 for (size_t i = 0; i < mSortedVertices.size(); ++i) {
462
463 // there should be no scaling in the render to world matrix so we can
464 // use it as is to transform normals
465
466 DebugRayVertex &vtx = mSortedVertices[i];
467 vtx.mHitPoint = transformPoint(render2world, vtx.mHitPoint);
468 vtx.mNormal = transform3x3(render2world, vtx.mNormal);
469 vtx.mRayDiffX = transformPoint(render2world, vtx.mRayDiffX);
470 vtx.mRayDiffY = transformPoint(render2world, vtx.mRayDiffY);
471 }
472 }
473
474 return true;
475}

501void
502DebugRayBuilder::depthFirstPreorderTraversal(Node const *rootNode, Node *node, std::vector<uint32_t> *ordering) const
503{
504 // Remove any nodes which have no parent (i.e. dummy root node is the parent)
505 // and no children, these represent rays which startNewRay was called on,
506 // but weren't extended any further. Since these are only single points, don't
507 // bother storing them. We avoid storing them in the final output by not adding
508 // them to the ordering vector.
509 if (node->mParent != rootNode || !node->mChildren.empty()) {
510 ordering->push_back(uint32_t(node - rootNode));
511 } else {
512 node->mNumSubnodes = 0;
513 }
514
515 for (auto it = node->mChildren.begin(); it != node->mChildren.end(); ++it) {
516 Node *child = *it;
517 MNRY_ASSERT(child->mParent == node);
518 depthFirstPreorderTraversal(rootNode, child, ordering);
519 node->mNumSubnodes += child->mNumSubnodes;
520 }
521}

525DebugRayDatabase::Iterator::Iterator(DebugRayDatabase const &db, DebugRayFilter const *filter) :
526 mDb(&db),
527 mNumPrimariesSeen(0),
528 mCurrY(0),
529 mEndY(0),
530 mCurrPrimaryIdx(0),
531 mEndPrimaryIdx(0),
532 mCurrVertex(nullptr),
533 mEndVertex(nullptr),
534 mJumpToVertexLoop(false),
535 mIsDone(false)
536{
537 if (conditionFilter(filter ? *filter : DebugRayFilter(), &mFilter)) {
538 mCurrY = mFilter.mViewport.lower.y;
539 mEndY = mFilter.mViewport.upper.y;
540 mIsDone = !getNextVertex();
541 } else {
542 mIsDone = true;
543 }
544}
545
546bool
547DebugRayDatabase::Iterator::conditionFilter(DebugRayFilter const &input, DebugRayFilter *output) const
548{
549 MNRY_ASSERT(output);
550
551 output->mViewport.lower.x = (input.mViewport.lower.x == -1) ? 0 : min(input.mViewport.lower.x, int(mDb->mWidth) - 1);
552 output->mViewport.lower.y = (input.mViewport.lower.y == -1) ? 0 : min(input.mViewport.lower.y, int(mDb->mHeight) - 1);
553 output->mViewport.upper.x = (input.mViewport.upper.x == -1) ? int(mDb->mWidth) - 1 : min(input.mViewport.upper.x, int(mDb->mWidth) - 1);
554 output->mViewport.upper.y = (input.mViewport.upper.y == -1) ? int(mDb->mHeight) -1 : min(input.mViewport.upper.y, int(mDb->mHeight) - 1);
555
556 output->mStartPrimaryRay = (input.mStartPrimaryRay == uint32_t(-1)) ? 0 : input.mStartPrimaryRay;
557 output->mEndPrimaryRay = (input.mEndPrimaryRay == uint32_t(-1)) ? uint32_t(mDb->mPrimaryRayIndices.size() - 1) : min(input.mEndPrimaryRay, uint32_t(mDb->mPrimaryRayIndices.size() - 1));
558
559 output->mTags = input.mTags;
560 output->mMatchAll = input.mMatchAll ? 1 : 0;
561
562 output->mMinDepth = (input.mMinDepth == uint32_t(-1)) ? 0 : input.mMinDepth;
563 output->mMaxDepth = (input.mMaxDepth == uint32_t(-1)) ? 254 : input.mMaxDepth;
564
565 return true;
566}
567
568bool
569DebugRayDatabase::Iterator::getNextVertex()
570{
571 std::vector<uint32_t>::const_iterator startIt, endIt;
572 uint32_t startScan, endScan, rayIdx;
573 DebugRayVertex const *primaryRay;
574
575 if (mJumpToVertexLoop) {
576 mJumpToVertexLoop = false;
577 goto NextRayVertex;
578 }
579
580 // find first valid vert which passes the filter
581 while (mCurrY <= mEndY) {
582
583 // find a vert greater than or equal to start x
584 startScan = mCurrY * mDb->mWidth + mFilter.mViewport.lower.x;
585 startIt = std::lower_bound(mDb->mScreenPositions.begin(), mDb->mScreenPositions.end(), startScan);
586 if (startIt == mDb->mScreenPositions.end()) {
587 return false;
588 }
589
590 // find a vert less than or equal to end x
591 endScan = mCurrY * mDb->mWidth + mFilter.mViewport.upper.x;
592 endIt = std::upper_bound(mDb->mScreenPositions.begin(), mDb->mScreenPositions.end(), endScan);
593
594 mCurrPrimaryIdx = uint32_t(startIt - mDb->mScreenPositions.begin());
595 mEndPrimaryIdx = uint32_t(endIt - mDb->mScreenPositions.begin());
596
597 while (mCurrPrimaryIdx != mEndPrimaryIdx) {
598
599 // filter by range
600 if (mNumPrimariesSeen < mFilter.mStartPrimaryRay) {
601 goto NextPrimaryRay;
602 }
603
604 if (mNumPrimariesSeen > mFilter.mEndPrimaryRay) {
605 return false;
606 }
607
608 rayIdx = mDb->mPrimaryRayIndices[mCurrPrimaryIdx];
609 primaryRay = &mDb->mVertices[rayIdx];
610
611 mCurrVertex = primaryRay;
612 mEndVertex = mCurrVertex + primaryRay->mNumSubnodes;
613
614 while (mCurrVertex != mEndVertex) {
615
616 // filter by depth
617 if (mCurrVertex->mDepth != uint8_t(-1)) {
618 if (mCurrVertex->mDepth < mFilter.mMinDepth || mCurrVertex->mDepth > mFilter.mMaxDepth) {
619 goto NextRayVertex;
620 }
621 }
622
623 // filter by tags
624 if (mFilter.mTags) {
625 uint32_t maskedTags = mCurrVertex->mUserTags & mFilter.mTags;
626 if ((!mFilter.mMatchAll && !maskedTags) || (mFilter.mMatchAll && (maskedTags != mFilter.mTags))) {
627 goto NextRayVertex;
628 }
629 }
630
631 // bingo, we are now pointing to a vert which survived all of the filters
632 mJumpToVertexLoop = true;
633 return true;
634
635 NextRayVertex:
636 ++mCurrVertex;
637 }
638
639 NextPrimaryRay:
640 ++mCurrPrimaryIdx;
641 ++mNumPrimariesSeen;
642 }
643
644 ++mCurrY;
645 }
646
647 return false;
648}

685#define CHECK_WITH_FAILURE(x) if (!(x)) goto Failure
686
687bool
688DebugRayDatabase::load(char const *fileName)
689{
690 FILE *file = fopen(fileName, "rb");
691 if (!file) {
692 scene_rdl2::logging::Logger::error("[MCRT-RENDER] Unable to open file \"" , fileName , "\".");
693 return false;
694 }
695
696 cleanUp();
697
698 uint32_t version = 0;
699 CHECK_WITH_FAILURE( read(file, &version) );
700 CHECK_WITH_FAILURE( version == DEBUG_RAYS_VERSION );
701
702 CHECK_WITH_FAILURE( read(file, &mWidth) );
703 CHECK_WITH_FAILURE( read(file, &mHeight) );
704
705 uint32_t numVerts;
706 CHECK_WITH_FAILURE( read(file, &numVerts) );
707
708 if (numVerts) {
709 mVertices.resize(numVerts);
710 CHECK_WITH_FAILURE( readRawData(file, &mVertices[0], sizeof(DebugRayVertex) * numVerts) );
711 }
712
713 uint32_t numPrimaryRayIndices;
714 CHECK_WITH_FAILURE( read(file, &numPrimaryRayIndices) );
715
716 if (numPrimaryRayIndices) {
717 mPrimaryRayIndices.resize(numPrimaryRayIndices);
718 CHECK_WITH_FAILURE( readRawData(file, &mPrimaryRayIndices[0], sizeof(uint32_t) * numPrimaryRayIndices) );
719 }
720
721 buildScreenPositions();
722
723 fclose(file);
724
725 MNRY_ASSERT(isValid());
726
727 return true;
728
729Failure:
730 scene_rdl2::logging::Logger::error("[MCRT-RENDER] Failed to load file \"" , fileName , "\".");
731 cleanUp();

736bool
737DebugRayDatabase::save(char const *fileName) const
738{
739 MNRY_ASSERT(isValid());
740
741 FILE *file = fopen(fileName, "wb");
742 if (!file) {
743 scene_rdl2::logging::Logger::error("[MCRT-RENDER] Unable to create file \"" , fileName , "\".");
744 return false;
745 }
746
747 CHECK_WITH_FAILURE( write(file, uint32_t(DEBUG_RAYS_VERSION)) );
748 CHECK_WITH_FAILURE( write(file, mWidth) );
749 CHECK_WITH_FAILURE( write(file, mHeight) );
750 CHECK_WITH_FAILURE( write(file, uint32_t(mVertices.size())) );
751
752 if (!mVertices.empty()) {
753 CHECK_WITH_FAILURE( writeRawData(file, &mVertices[0], sizeof(DebugRayVertex) * mVertices.size()) );
754 }
755
756 CHECK_WITH_FAILURE( write(file, uint32_t(mPrimaryRayIndices.size())) );
757 if (!mPrimaryRayIndices.empty()) {
758 CHECK_WITH_FAILURE( writeRawData(file, &mPrimaryRayIndices[0], sizeof(uint32_t) * mPrimaryRayIndices.size()) );
759 }
760
761 // to allow future expansion of file format
762 CHECK_WITH_FAILURE( write(file, uint32_t(0)) );
763
764 fclose(file);
765 return true;
766
767Failure:
768 scene_rdl2::logging::Logger::error("[MCRT-RENDER] Failed to save file \"" , fileName , "\".");
769 fclose(file);
770 return false;
771}

797void
798DebugRayDatabase::buildScreenPositions()
799{
800 if (mPrimaryRayIndices.empty()) {
801 return;
802 }
803
804 size_t numPrimaryRay = mPrimaryRayIndices.size();
805 MNRY_ASSERT(mVertices.size() >= numPrimaryRay);
806
807 mScreenPositions.resize(numPrimaryRay);
808
809 for (size_t i = 0; i < numPrimaryRay; ++i) {
810 DebugRayVertex const &vert = mVertices[mPrimaryRayIndices[i]];
811 MNRY_ASSERT(vert.mParentId == NULL_RAY_VERTEX_ID);
812
813 mScreenPositions[i] = vert.mScreenY * mWidth + vert.mScreenX;
814 }
815}
