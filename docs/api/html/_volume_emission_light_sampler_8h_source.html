22/// Utility class that wraps multiple samplers used in volume emission
23/// light sample strategy (for easier sampling management)
24class VolumeEmissionLightSampler {
25public:
26 // constructor for hard surface samples
27 VolumeEmissionLightSampler(const Subpixel& sp, const PathVertex& pv,
28 int nLightSamples, int emissionIndex, bool highQuality,
29 unsigned sequenceID):
30 mNonMirrorDepth(pv.nonMirrorDepth)
31 {
32 if (highQuality) {
33 int samplesSoFar = sp.mSubpixelIndex * nLightSamples;
34 SequenceIDIntegrator xyzSid(
35 pv.nonMirrorDepth, sp.mPixel, 0,
36 SequenceType::VolumeEmission,
37 SequenceType::Light,
38 sequenceID, emissionIndex);
39 mSamples.resume(xyzSid, samplesSoFar);
40 } else {
41 int samplesCount = nLightSamples;
42 SequenceIDIntegrator xyzSid(
43 pv.nonMirrorDepth, sp.mPixel, sp.mSubpixelIndex,
44 SequenceType::VolumeEmission,
45 SequenceType::Light,
46 sequenceID, emissionIndex);
47 mSamples.restart(xyzSid, samplesCount);
48 }
49 }
50
51 // constructor for bssrdf subsurface scattering samples
52 VolumeEmissionLightSampler(const Subpixel& sp, const PathVertex& pv,
53 int nLightSamples, int emissionIndex, bool highQuality,
54 int subsurfaceSplitFactor, int subsurfaceIndex,
55 unsigned sssSampleID, bool isLocal):
56 mNonMirrorDepth(pv.nonMirrorDepth)
57 {
58 int localFactor = isLocal ?
59 static_cast<int>(SequenceType::BssrdfLocalLight) :
60 static_cast<int>(SequenceType::BssrdfGlobalLight);
61 if (highQuality) {
62 // subsurfaceIndex range from [0, mBssrdfSamples)
63 // subsurfaceSplitFactor = mBssrdfSamples
64 // Using SSS / SP / L
65 // Note: using SP / SSS / L caused more correlation
66 int samplesSoFar = nLightSamples * (
67 subsurfaceIndex * sp.mPixelSamples + sp.mSubpixelIndex);
68 SequenceIDIntegrator xyzSid(
69 pv.nonMirrorDepth, sp.mPixel, 0,
70 SequenceType::VolumeEmission,
71 SequenceType::Light,
72 localFactor, sssSampleID, emissionIndex);
73 mSamples.resume(xyzSid, samplesSoFar);
74 } else {
75 int samplesCount = nLightSamples;
76 SequenceIDIntegrator xyzSid(
77 pv.nonMirrorDepth, sp.mPixel,
78 sp.mSubpixelIndex * subsurfaceSplitFactor + subsurfaceIndex,
79 SequenceType::VolumeEmission,
80 SequenceType::Light,
81 localFactor, sssSampleID, emissionIndex);
82 mSamples.restart(xyzSid, samplesCount);
83 }
84 }
85
86 // constructor for random walk subsurface scattering samples
87 VolumeEmissionLightSampler(const Subpixel& sp, const PathVertex& pv,
88 int emissionIndex, int subsurfaceSplitFactor, int subsurfaceIndex,
89 unsigned sssSampleID):
90 mNonMirrorDepth(pv.nonMirrorDepth)
91 {
92 SequenceIDIntegrator xyzSid(
93 pv.nonMirrorDepth, sp.mPixel, 0,
94 SequenceType::VolumeEmission,
95 SequenceType::Light,
96 SequenceType::BssrdfLocalLight,
97 sssSampleID, emissionIndex);
98 mSamples.resume(xyzSid, subsurfaceIndex);
99 }
100
101 // constructor for volume scattering samples
102 VolumeEmissionLightSampler(const Subpixel& sp, const PathVertex& pv,
103 int nLightSamples, int nScatterSamples, int emissionIndex,
104 bool highQuality, unsigned sequenceID):
105 mNonMirrorDepth(pv.nonMirrorDepth)
106 {
107 if (highQuality) {
108 int samplesSoFar = sp.mSubpixelIndex *
109 nLightSamples * nScatterSamples;
110 SequenceIDIntegrator xyzSid(
111 pv.nonMirrorDepth, sp.mPixel, 0,
112 SequenceType::VolumeEmission,
113 SequenceType::Light,
114 SequenceType::VolumeScattering,
115 sequenceID, emissionIndex);
116 mSamples.resume(xyzSid, samplesSoFar);
117 } else {
118 int samplesCount = nLightSamples * nScatterSamples;
119 SequenceIDIntegrator xyzSid(
120 pv.nonMirrorDepth, sp.mPixel, sp.mSubpixelIndex,
121 SequenceType::VolumeEmission,
122 SequenceType::Light,
123 SequenceType::VolumeScattering,
124 sequenceID, emissionIndex);
125 mSamples.restart(xyzSid, samplesCount);
126 }
127 }
128
129 // constructor for volume multiple scattering approximation
130 VolumeEmissionLightSampler(const Subpixel& sp, const PathVertex& pv,
131 int nScatterSamples, int emissionIndex, unsigned sequenceID):
132 mNonMirrorDepth(pv.nonMirrorDepth)
133 {
134
135 SequenceIDIntegrator xyzSid(
136 pv.nonMirrorDepth, pv.volumeDepth, sp.mPixel, sp.mSubpixelIndex,
137 SequenceType::VolumeEmission,
138 SequenceType::Light,
139 SequenceType::VolumeScattering,
140 sequenceID, emissionIndex);
141 mSamples.restart(xyzSid, nScatterSamples);
142 }
