51 // BsdfLobe API
52 finline scene_rdl2::math::Color eval(const BsdfSlice &slice, const scene_rdl2::math::Vec3f &wi, float *pdf = NULL) const override
53 {
54 // Note: we assume this lobe has been setup with a OneMinus*Fresnel
55 // We use cosThetaWo to evaluate the fresnel term, as an approximation
56 // of what hDotWi would be for the "over" lobe.
57 const float cosThetaWo = (!getFresnel() ? 1.0f :
58 scene_rdl2::math::max(dot(mN, slice.getWo()), 0.0f));
59
60 const float cosThetaWi = scene_rdl2::math::max(dot(mN, wi), 0.0f);
61
62 // Forward to the under lobe and apply fresnel
63 return (computeScaleAndFresnel(cosThetaWo) *
64 computeTransmission(cosThetaWo, cosThetaWi) *
65 mUnder->eval(slice, wi, pdf));
66
67 }

70 finline scene_rdl2::math::Color sample(const BsdfSlice &slice, float r1, float r2,
71 scene_rdl2::math::Vec3f &wi, float &pdf) const override
72 {
73 // See eval() for info
74 const float cosThetaWo = (!getFresnel() ? 1.0f :
75 scene_rdl2::math::max(dot(mN, slice.getWo()), 0.0f));
76
77 const scene_rdl2::math::Color underResult = mUnder->sample(slice, r1, r2, wi, pdf);
78 const float cosThetaWi = scene_rdl2::math::max(dot(mN, wi), 0.0f);
79
80 // Forward to the under lobe and apply fresnel
81 return (computeScaleAndFresnel(cosThetaWo) *
82 computeTransmission(cosThetaWo, cosThetaWi) *
83 underResult);
84 }

86 finline scene_rdl2::math::Color albedo(const BsdfSlice &slice) const override

90 scene_rdl2::math::max(dot(mN, slice.getWo()), 0.0f));
91
92 // Forward to the under lobe and apply fresnel
93 return computeScaleAndFresnel(cosThetaWo) * mUnder->albedo(slice);

126protected:
127 // Attenuate the under layer based on 'thickness' and 'attenuation color'
128 // and distance traveled through the clearcoat layer
129 scene_rdl2::math::Color computeTransmission(float cosThetaR1,
130 float cosThetaR2) const
131 {
132 if (scene_rdl2::math::isZero(mThickness)) {
133 return scene_rdl2::math::sWhite;
134 }
135
136 float rcp1 = 0.0f, rcp2 = 0.0f;
137 if (!scene_rdl2::math::isZero(cosThetaR1)) rcp1 = scene_rdl2::math::rcp(scene_rdl2::math::abs(cosThetaR1));
138 if (!scene_rdl2::math::isZero(cosThetaR2)) rcp2 = scene_rdl2::math::rcp(scene_rdl2::math::abs(cosThetaR2));
139
140 // length of light path within clearcoat
141 const float distanceTraveled = mThickness * (rcp1 + rcp2);
142
149
150 const scene_rdl2::math::Color T = scene_rdl2::math::Color(
151 scene_rdl2::math::pow(mAttenuationColor.r, distanceTraveled),
152 scene_rdl2::math::pow(mAttenuationColor.g, distanceTraveled),
153 scene_rdl2::math::pow(mAttenuationColor.b, distanceTraveled));
154 return T;
155 }
156
157protected:
158 BsdfLobe *mUnder;
159 scene_rdl2::math::Vec3f mN;
160
161 // Thickness for Attenuation
162 float mThickness;
163 // Attenuation Color
164 scene_rdl2::math::Color mAttenuationColor;
165
166};
