85// DECLARE_HANDLE() Macro:
86// We desire some semblance to type-safety in our ISPC RDL2 queries.
87// There is just one true type (AttributeKeyISPC, see ISPCSupport.h)
88// but we would like to require code to use the appropriately typed
89// key (e.g. float, int, float, float2, etc. - also in ISPCSupport.h)
90//
91// Reluctantly, we employ an old Microsoft trick by which we can
92// have distinctly-typed pointers. The following macro...
93//
94// DECLARE_HANDLE(Color)
95//
96// ...expands into:
97//
98// struct Color__ {
99// int64 unused;
100// };
101//
102// typedef struct Color__* Color;
103//
104// that is, Color__ is (and this is the important part) a *unique type*
105// with not-so-unique storage for a 64 bit pointer that points to
106// (in our usage)
107//
108// 'Color' is typedef'd to be a pointer to that storage
109//
110// Because Color__ is a unique type the compiler cannot auto-cast it
111//
112// It is used in ISPCSupport.h to declare the differt kinds of key types
113// ISPC will use to query parameters from RDL2, e.g.
114//
115// DECLARE_HANDLE(BoolAttributeKeyISPC);
116// DECLARE_HANDLE(FloatAttributeKeyISPC);
117// (etc.)
118
119#ifdef ISPC
120#define DECLARE_HANDLE(name) struct name##ISPC { int64 unused; }; \
121 typedef name##ISPC name;
122#else
123#define DECLARE_HANDLE(name) struct name##ISPC { int64 unused; };
124#endif

127// Macro for creating operators for bitflag enum types. Using these operators
128// can create enum values that aren't explicitly enumerated, but the client
129// interface remains type safe and can still test for the existence of each
130// flag in the usual way.
131#define RDL2_DEFINE_BITFLAG_OPERATORS(bitflag_type) \
132 finline bitflag_type operator~(bitflag_type a) \
133 { \
134 return static_cast<bitflag_type>(~static_cast<int>(a)); \
135 } \
136 \
137 finline bitflag_type operator&(bitflag_type a, bitflag_type b) \
138 { \
139 return static_cast<bitflag_type>(static_cast<int>(a) & static_cast<int>(b)); \
140 } \
141 \
142 finline bitflag_type operator|(bitflag_type a, bitflag_type b) \
143 { \
144 return static_cast<bitflag_type>(static_cast<int>(a) | static_cast<int>(b)); \
145 } \
146 \
147 finline bitflag_type operator^(bitflag_type a, bitflag_type b) \
148 { \
149 return static_cast<bitflag_type>(static_cast<int>(a) ^ static_cast<int>(b)); \
150 } \
151 \
152 finline bitflag_type& operator&=(bitflag_type& a, const bitflag_type& b) \
153 { \
154 a = a & b; \
155 return a; \
156 } \
157 \
158 finline bitflag_type& operator|=(bitflag_type& a, const bitflag_type& b) \
159 { \
160 a = a | b; \
161 return a; \
162 } \
163 \
164 finline bitflag_type& operator^=(bitflag_type& a, const bitflag_type& b) \
165 { \
166 a = a ^ b; \
167 return a; \
168 }
