20// @brief BVHUserData is the user data we pass to embree

23 // @brief The intersection Filter Manager is a container for all intersection

30 /// static filter functions that are registered with embree.
31
32 static void intersectionFilter(const RTCFilterFunctionNArguments* args)
33 {
34 const BVHUserData* userData =
35 (const BVHUserData*)args->geometryUserPtr;
36 const IntersectionFilterManager* fm = userData->mFilterManager;
37 for (auto& filter : fm->mIntersectionFilters) {
38 filter(args);
39 }
40 }
41
42 static void occlusionFilter(const RTCFilterFunctionNArguments* args)
43 {
44 const BVHUserData* userData =
45 (const BVHUserData*)args->geometryUserPtr;
46 const IntersectionFilterManager* fm = userData->mFilterManager;
47 for (auto& filter : fm->mOcclusionFilters) {
48 filter(args);
49 }
50 }
51
52 /// add filter function to list of filter functions
53 void addIntersectionFilter(RTCFilterFunctionN filterFunction)
54 {
55 mIntersectionFilters.push_back(filterFunction);
56 }
57
58 void addOcclusionFilter(RTCFilterFunctionN filterFunction)
59 {
60 mOcclusionFilters.push_back(filterFunction);
61 }
62
63 /// filter function lists
64 std::vector<RTCFilterFunctionN> mIntersectionFilters;
65 std::vector<RTCFilterFunctionN> mOcclusionFilters;
66 };
