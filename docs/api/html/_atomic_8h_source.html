31// In compare_exchange overloads where only one memory order is given, we have
32// to decide on the other. These are what is laid out by the standard.
33constexpr std::memory_order compare_exchange_duo(std::memory_order in) noexcept
34{
35 constexpr std::memory_order mapping[6] = {
36 /* std::memory_order_relaxed -> */ std::memory_order_relaxed,
37 /* std::memory_order_consume -> */ std::memory_order_consume,
38 /* std::memory_order_acquire -> */ std::memory_order_acquire,
39 /* std::memory_order_release -> */ std::memory_order_relaxed,
40 /* std::memory_order_acq_rel -> */ std::memory_order_acquire,
41 /* std::memory_order_seq_cst -> */ std::memory_order_seq_cst
42
43 };
44 return mapping[static_cast<int>(in)];
45}

81inline T
82atomicLoad(const T* v, std::memory_order order = std::memory_order_seq_cst) noexcept
83{
84 MNRY_ASSERT(__atomic_is_lock_free(sizeof(T), v));
85 alignas(T) unsigned char buf[sizeof(T)];
86 auto* const dest = reinterpret_cast<T*>(buf);
87 __atomic_load(v, dest, static_cast<int>(order));
88 return *dest;
89}

141// We want to store the min of the two values (_a_, _b_) in _a_
142template <typename T>
143inline void
144atomicMin(T* a, T b) noexcept
145{
146 alignas(atomicAlignment<T>()) T x = atomicLoad(a, std::memory_order_relaxed);
147 MNRY_ASSERT(__atomic_is_lock_free(sizeof(T), a));
148 do {
149 // If _x_ is less than or equal to _b_, our work is done. If some other thread calls in, the value of _x_ is
150 // only going to get lower, and b is still greater.
151 if (x <= b) {
152 break;
153 }
154 // If we get to the CAS, we know that _b_ is less than _a_ (unless another thread preempted us). Update _a_ to
155 // the value of _b_. If it succeeds, we're done! If it fails, _x_ is updated to the new value of _a_, and we
156 // will continue and check if the new value is less than _b_.
157 } while (!atomicCompareAndSwapWeak(a, x, b, std::memory_order_relaxed));
158}

160// We want to store the max of the two values (_a_, _b_) in _a_
161template <typename T>
162inline void
163atomicMax(T* a, T b) noexcept
164{
165 alignas(atomicAlignment<T>()) T x = atomicLoad(a, std::memory_order_relaxed);
166 MNRY_ASSERT(__atomic_is_lock_free(sizeof(T), a));
167 do {
168 // If _x_ is greater than or equal to _b_, our work is done. If some other thread calls in, the value of x is
169 // only going to get higher, and b is still smaller.
170 if (b <= x) {
171 break;
172 }
173 // If we get to the CAS, we know that _b_ is greater than _a_ (unless another thread preempted us). Update _a_
174 // to the value of _b_. If it succeeds, we're done! If it fails, _x_ is updated to the new value of _a_, and we
175 // will continue and check if the new value is greater than _b_.
176 } while (!atomicCompareAndSwapWeak(a, x, b, std::memory_order_relaxed));
177}

179// We want to store the sum of the two values (_a_, _b_) in _a_
180template <typename T>
181inline void
182atomicAdd(T* a, T b) noexcept
183{
184 alignas(atomicAlignment<T>()) T oldVal = atomicLoad(a, std::memory_order_relaxed);
185 alignas(atomicAlignment<T>()) T newVal = oldVal + b;
186 MNRY_ASSERT(__atomic_is_lock_free(sizeof(T), a));
187 while (!atomicCompareAndSwapWeak(a, oldVal, newVal, std::memory_order_relaxed)) {
188 newVal = oldVal + b;
189 }
190}

192inline void
193atomicAssignIfClosest(float* __restrict val, const float* __restrict newVal)
194{
195 // We don't allow our array of four floats to overlap since we're using restrict.
196 MNRY_ASSERT(val + 4 <= newVal || newVal + 4 <= val);
197
198 // assumes 1) val and newVal point to aligned float[4] types
199 // 2) val[3] and newVal[3] store the depth
200 // 3) These types should produce lock-free atomics
201 struct alignas(atomicAlignment(kAtomicAlignment)) Float4Aligned
202 {
203 float x, y, z, d;
204 };
205 std::atomic<Float4Aligned>* const dest = reinterpret_cast<std::atomic<Float4Aligned>*>(val);
206 const Float4Aligned srcVal = *reinterpret_cast<const Float4Aligned*>(newVal);
207
208 MNRY_ASSERT(reinterpret_cast<uintptr_t>(val) % kAtomicAlignment == 0);
209 MNRY_ASSERT(reinterpret_cast<uintptr_t>(newVal) % kAtomicAlignment == 0);
210 MNRY_ASSERT(std::atomic_is_lock_free(dest));
211
212 // This fails on GCC 9.3, but succeeds on GCC 6.3 and ICC 19.
213 // However, the run-time lock-free check above succeeds.
214 // There are valid reasons why this may be the case: it's just interesting
215 // that later compilers are less strict here.
216 // TODO: C++17 replace with std::atomic::is_always_lock_free
217 //static_assert(__atomic_always_lock_free(sizeof(Float4Aligned), dest), "");
218
219 auto observed = dest->load(std::memory_order_relaxed);
220 do {
221 // If _observed.d_ is less than or equal to _srcVal.d_, our work is done. If some other thread calls in, the
222 // value of _dest.d_ is only going to get lower, so _srcVal.d_ is still greater.
223 if (observed.d <= srcVal.d) {
224 break;
225 }
226 // If we get to the CAS, we know that _srcVal.d_ is less than _dest->d_ (unless another thread preempted us).
227 // Update _dest_ to the value of _srcVal_. If it succeeds, we're done! If it fails, _observed_ is updated to the
228 // new value of _dest_ and we will continue and check if the new value is less than _srcVal_.
229 } while (!dest->compare_exchange_weak(observed, srcVal, std::memory_order_relaxed));
230}
