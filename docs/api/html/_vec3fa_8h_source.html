235 __forceinline Vec3fa cross ( const Vec3fa& a, const Vec3fa& b )
236 {
237 // MoonRay: added simd:: namespace
238 simd::ssef a0 = simd::ssef(a);
239 simd::ssef b0 = simd::shuffle<1,2,0,3>(simd::ssef(b));
240 simd::ssef a1 = simd::shuffle<1,2,0,3>(simd::ssef(a));
241 simd::ssef b1 = simd::ssef(b);
242 return Vec3fa(simd::shuffle<1,2,0,3>(msub(a0,b0,a1*b1)));
243 }
