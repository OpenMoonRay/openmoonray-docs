99po::variables_map
100parseCommandLine(int argc, char* argv[], std::string& dsoPath)
101{
102 po::options_description desc("Allowed options");
103 desc.add_options()
104 ((BO_HELP_S + ",h").c_str(), "show usage statement")
105 (BO_DSO_PATH_S.c_str(), po::value<std::string>(), "Path to RDL2 DSOs")
106 (BO_IN_PATH_S.c_str(), po::value<std::vector<std::string> >(), "RDL2 DSO to convert(s)")
107 (BO_OUT_PATH_S.c_str(), po::value<std::vector<std::string> >(), "Output class file(s)")
108 (BO_BUILT_IN_S.c_str(), "Create class files for built-In classes only")
109 (BO_SPARSED_S.c_str(), "Create separate class files for each RDL2 DSO")
110 (BO_RDL2_VERSION_S.c_str(), po::value<std::string>(), "rdl2 version to embed in output file")
111 (BO_MOONRAY_VERSION_S.c_str(), po::value<std::string>(), "Moonray version to embed in output file")
112 ;
113
114 po::variables_map vm;
115 try {
116 po::store(po::command_line_parser(argc, argv).options(desc).run(), vm);
117 } catch (const po::unknown_option& ) {
118 std::cerr << desc;
119 std::exit(EXIT_FAILURE);
120 }
121 po::notify(vm);
122
123 if (vm.count("help")) {
124 std::cerr << desc;
125 std::exit(EXIT_FAILURE);
126 }
127
128 std::string dsoSearchPath = rdl2::DsoFinder::parseDsoPath(argc, argv);
129 if (!dsoSearchPath.empty()) {
130 dsoPath = dsoSearchPath;
131 }
132
133 return vm;
134}

143void
144handlePossibleEnumJson(Json::Value& root, const rdl2::Attribute& attr)
145{
146 if (!attr.isEnumerable()) {
147 return; // Bail now if this is not an enum
148 }
149
150 Json::Value enumValues;
151
152 std::vector<std::string> lines;
153 for (auto enumIter = attr.beginEnumValues(); enumIter != attr.endEnumValues(); ++enumIter) {
154 enumValues[enumIter->second] = Json::Value(enumIter->first);
155 }
156
157 root["enum"] = enumValues;

160void
161outputAttributeJson(Json::Value& root, const rdl2::Attribute& attr, size_t index)
162{
163 Json::Value attribute;
164
165 attribute["attrType"] = Json::Value(
166 scene_rdl2::rdl2::attributeTypeName(attr.getType()));
167 attribute["order"] = Json::Value(static_cast<unsigned int>(index));
168
169 outputAttrDefaultJson(attribute, attr);
170 handlePossibleEnumJson(attribute, attr);
171
172 if (attr.isBindable())
173 attribute[ATTR_BIND_LABEL] = Json::Value(true);
174
175 if (attr.isFilename())
176 attribute[ATTR_FILENAME_LABEL] = Json::Value(true);
177
178 rdl2::SceneObjectInterface interface = attr.getObjectType();
179 if (interface != rdl2::INTERFACE_GENERIC) {
180 std::string interfaceName = rdl2::interfaceTypeName(interface);
181 attribute[ATTR_INTERFACE_LABEL] = Json::Value(interfaceName);
182 }
183
184 // handle metadata
185
186 if (!attr.metadataEmpty()) {
187
188 Json::Value metadata;
189
190 for (auto iter = attr.beginMetadata(); iter != attr.endMetadata(); ++iter) {
191 metadata[iter->first] = Json::Value(iter->second);
192 }
193
194 attribute["metadata"] = metadata;
195 }
196
197 // handle aliases
198 const std::vector<std::string>& aliases = attr.getAliases();
199 if(!aliases.empty()) {
200 Json::Value alias_list;
201
202 for (const auto& alias : aliases) {
203 alias_list.append(alias);
204 }
205
206 attribute["aliases"] = alias_list;
207 }
208
209 root[attr.getName()] = attribute;
210}

223void
224embedGroupingJson(Json::Value& root, const rdl2::SceneClass& cls)
225{
226 // If the class has no groups just return
227 if (cls.beginGroups() == cls.endGroups()) {
228 return;
229 }
230
231 Json::Value grouping;
232 Json::Value groups;
233 Json::Value order;
234
235 // track which structures we've already covered
236 std::set<std::string> foundStructures;
237 for (auto iter = cls.beginGroups(); iter != cls.endGroups(); ++iter) {
238 const std::string& groupName = *iter;
239 order.append(Json::Value(groupName));
240
241 Json::Value groupAttributes;
242
243 std::vector<const rdl2::Attribute*> attrs = cls.getAttributeGroup(groupName);
244 for (const rdl2::Attribute* attr : attrs) {
245 // default to using rdl2 name
246 std::string nameTmp = attr->getName();
247 // Check for structure type - add single entry, using structure name
248 if (attr->metadataExists(ATTR_STRUCTURE_NAME)) {
249 auto result = foundStructures.insert(attr->getMetadata(ATTR_STRUCTURE_NAME));
250 if (!result.second) {
251 // Already have this. Skip
252 continue;
253 }
254
255 // Use the structure name in the grouping instead.
256 // TODO Note that this won't be kept for things put
257 // in the hidden section. We don't currently expect
258 // this to be an issue.
259 nameTmp = attr->getMetadata(ATTR_STRUCTURE_NAME);
260 }
261
262 groupAttributes.append(Json::Value(nameTmp));
263 }
264
265 groups[groupName] = groupAttributes;
266 }
267
268 grouping["order"] = order;
269 grouping["groups"] = groups;
270 root["grouping"] = grouping;
271}

273void
274writeJson(Json::Value& root, const rdl2::SceneClass& cls)
275{
276 Logger::debug("Writing JSON data for class ", cls.getName(), "...");
277
278 Json::Value objectRoot;
279 rdl2::SceneObjectInterface type = cls.getDeclaredInterface();
280 std::string typeName = rdl2::interfaceTypeName(type);
281 objectRoot["type"] = Json::Value(typeName);
282
283 Json::Value attributes;
284 size_t index = 0;
285 for (auto iter = cls.beginAttributes(); iter != cls.endAttributes(); ++iter)
286 {
287 const rdl2::Attribute& attr = *(*iter);
288 outputAttributeJson(attributes, attr, index++);
289 }
290
291 objectRoot["attributes"] = attributes;
292
293 if (cls.beginGroups() != cls.endGroups()) {
294 embedGroupingJson(objectRoot, cls);
295 }
296
297 // Determine whether this class is part of moonray or moonshine and add
298 // the appropriate version information.
299 std::string srcPath = cls.getSourcePath();
300 if (srcPath.find("moonray") != std::string::npos) {
301 const char* moonrayVersion = getenv("REZ_MOONRAY_VERSION");
302 if (moonrayVersion) {
303 std::string versionString(moonrayVersion);
304 versionString = "moonray-" + versionString;
305 objectRoot["folio"] = versionString;
306 }
307 } else if (srcPath.find("moonshine") != std::string::npos) {
308 const char* moonshineVersion = getenv("REZ_MOONSHINE_VERSION");
309 if (moonshineVersion) {
310 std::string versionString(moonshineVersion);
311 versionString = "moonshine-" + versionString;
312 objectRoot["folio"] = versionString;
313 }
314 }
315
316 root[cls.getName()] = objectRoot;
317
318 Logger::debug("Done writing JSON data for class ", cls.getName());
319
320}

322// Modifies the output file to include the class name, if applicable, and makes sure the file
323// and its parent directory are writable
324void
325setupOutputFile(std::string& outFileName, const std::string& className,
326 const po::variables_map& options, const std::string& extension)
327{
328 bf::path filePath(outFileName);
329
330 // If the path exists and is a directory, add the name and extension
331 if (bf::exists(filePath)) {
332 if (bf::is_directory(outFileName)) {
333 Logger::debug(outFileName, " is a directory");
334 // If we're sparsing, and not working on only a single file, then we can't write to this directory
335 if (!options.count(BO_SPARSED_S) && !options.count(BO_IN_PATH_S)) {
336 throw except::IoError("Output path is a directory");
337 }
338 outFileName = (bf::path(outFileName) / className).string() + extension;
339 filePath = bf::path(outFileName);
340 }
341 // If they specified a directory explicitly, make sure it exists
342 } else if (outFileName.find_last_of(PATH_SEPARATOR) == outFileName.size() - 1) {
343 Logger::debug(outFileName, " is a directory, but does not exist");
344 if (!bf::create_directory(filePath)) {
345 throw except::IoError("Unable to create directory: " + filePath.string());
346 }
347 outFileName = (bf::path(outFileName) / className).string() + extension;
348 filePath = bf::path(outFileName);
349 }
350
351 // Check for writability of the file
352 if (bf::exists(filePath) && access(outFileName.c_str(), W_OK) == -1) {
353 throw except::IoError(std::string(strerror(errno)) + ": " + outFileName);
354 }
355
356 // Check for writability of the directory
357 Logger::debug("filePath.parent_path(): ", filePath.parent_path());
358 struct stat statBuf;
359 stat(filePath.parent_path().string().c_str(), &statBuf);
360 Logger::debug("st_mode & (S_IWUSR): ", (statBuf.st_mode & (S_IWUSR)));
361 if (!(statBuf.st_mode & S_IWUSR)) {
362 throw except::IoError(std::string(strerror(errno)) + ": " + filePath.parent_path().string().c_str() + " is not writable");
363 }
364}

381void
382createFile(const rdl2::SceneClass* scene_class,
383 const po::variables_map& options,
384 const GeneratorData& data, const std::string& outputPath= "")
385{
386 // Get json data
387 Json::Value root;
388
389 writeVersionInfo(root);
390
391 Json::Value sceneObjects;
392 data.mWriteJson(sceneObjects, *scene_class);
393 root["scene_classes"] = sceneObjects;
394
395 // Set default output to stdout
396 std::ostream* out = &std::cout;
397 std::ofstream outfile;
398
399 auto outFileName = outputPath;
400 if (!outFileName.empty()) {
401 setupOutputFile(outFileName, scene_class->getName(), options,
402 data.mExtension);
403 outfile.open(outFileName.c_str(), std::ios_base::out);
404 Logger::debug("Using file: ", outFileName);
405 out = &outfile;
406 }
407
408 Json::StyledStreamWriter writer;
409 writer.write(*out, root);
410
411 if (outfile.is_open()) {
412 outfile.close();
413 }

453void
454createFiles(rdl2::SceneContext& ctx, const po::variables_map& options, const GeneratorData& data)
455{
456 auto classNames = options[BO_IN_PATH_S].as<std::vector<std::string> >();
457 if (options.count(data.mPath)) {
458 const std::string& path = data.mPath;
459 auto outputPaths = options[path].as<std::vector<std::string> >();
460 // And we have a corresponding out path per input path, then create a one-to-one input -> output
461 if (classNames.size() == outputPaths.size()) {
462 for (size_t i = 0; i < classNames.size(); i++) {
463 auto actualClassName = maybeLoadDso(ctx, classNames[i], options);
464 createFile(ctx.getSceneClass(actualClassName), options, data,
465 outputPaths[i]);
466 }
467 // Otherwise, we have multiple inputs, but should only have one output
468 } else {
469 if (outputPaths.size() != 1) {
470 throw except::RuntimeError("Must either specify only a single output path for multiple input files or one output path per input file");
471 } else {
472 auto outputPath = outputPaths[0];
473
474 std::vector<const rdl2::SceneClass*> scene_classes;
475 scene_classes.reserve(classNames.size());
476 for (const std::string& className : classNames) {
477 auto actualClassName = maybeLoadDso(ctx, className, options);
478 scene_classes.push_back(ctx.getSceneClass(actualClassName));
479 }
480
481 createFile(scene_classes, options, data, outputPath);
482 }
483 }
484
485 // No output files specified, stream to stdout
486 } else {
487 for(auto iter = classNames.begin(); iter != classNames.end(); ++iter) {
488 auto actualClassName = maybeLoadDso(ctx, *iter, options);
489 createFile(ctx.getSceneClass(actualClassName), options, data);
490 }
491 }
492}

494void
495createAllFiles(const rdl2::SceneContext& ctx, const po::variables_map& options, const GeneratorData& data)
496{
497 std::string outFileName;
498 if (options.count(data.mPath)) {
499 const std::string& path = data.mPath;
500 outFileName = options[path].as<std::vector<std::string> >()[0];
501 }
502
503 if (options.count(BO_SPARSED_S)) {
504 for (auto i = ctx.beginSceneClass(); i != ctx.endSceneClass(); ++i) {
505 createFile(i->second, options, data, outFileName);
506 }
507 } else {
508 // Get a vector of all scene classes
509 std::vector<const rdl2::SceneClass*> scene_classes;
510 for (auto i = ctx.beginSceneClass(); i != ctx.endSceneClass(); ++i) {
511 scene_classes.push_back(i->second);
512 }
513
514 createFile(scene_classes, options, data, outFileName);
515 }

520int main(int argc, char* argv[])
521{
522 std::string dsoPath;
523 po::variables_map options = parseCommandLine(argc, argv, dsoPath);
524
525 rdl2::SceneContext context; // This loads all built-ins automatically
526
527 if (!dsoPath.empty()) {
528 context.setDsoPath(dsoPath);
529 }
530
531 // Get DSO Path from command line if present
532 if (options.count(BO_DSO_PATH_S)) {
533 context.setDsoPath(options[BO_DSO_PATH_S].as<std::string>());
534 } else { // Otherwise, use the environment
535 const char* rdl2_dso_path_env = getenv("RDL2_DSO_PATH");
536 if (rdl2_dso_path_env != NULL) {
537 context.setDsoPath(rdl2_dso_path_env);
538 }
539 }
540
541 // Use proxy mode. We only need the attribute declarations, and the proxy
542 // DSOs are much, much faster to open.
543 context.setProxyModeEnabled(true);
544
545 // Create the GeneratorData for class and groupings files
546 GeneratorData jsonGeneratorData(BO_OUT_PATH_S, JSON_EXTENSION, writeJson);
547
548 try {
549 // If we've specified one or more input paths
550 if (options.count(BO_IN_PATH_S)) {
551 createFiles(context, options, jsonGeneratorData);
552 // Otherwise, load everything from the DSO path
553 } else {
554 context.loadAllSceneClasses();
555 createAllFiles(context, options, jsonGeneratorData);
556 }
557 } catch (std::exception& e) {
558 std::cerr << "ERROR: " << e.what() << "\n";
559 std::exit(EXIT_FAILURE);
560 }
561
562 return EXIT_SUCCESS;
563}
