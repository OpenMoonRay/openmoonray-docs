81 virtual void visitCurves(geom::Curves& c) override {
82 geom::internal::Primitive* pImpl =
83 geom::internal::PrimitivePrivateAccess::getPrimitiveImpl(&c);
84 MNRY_ASSERT_REQUIRE(pImpl != nullptr);
85 MNRY_ASSERT_REQUIRE(pImpl->getType() == geom::internal::Primitive::CURVES);
86 auto pCurves = static_cast<geom::internal::Curves*>(pImpl);
87 // bind the BVH representation to corresponding Primitive
88 // or update the BVH representation if Primitive got deformed
89 // (real time frame update case)
90 if (mGeometry->isStatic() || !pCurves->isBVHInitialized()) {
91 pCurves->setBVHHandle(createCurvesInBVH(*pCurves,
92 c.getCurvesType(), c.getCurvesSubType(), c.getTessellationRate(), getGeomFlag()));
93 } else {
94 pCurves->updateBVHHandle();
95 }
96 }

181 virtual void visitPrimitiveGroup(geom::PrimitiveGroup& pg) override {
182 // recursively travel down scene graph (we may have primitive group,
183 // instances, regular primitives inside each primitive group)
184
185 // when outer parallel loop involves lock/wait depends on
186 // inner parrel loop, tbb may introduce deadlock due to its
187 // job stealing mechanics. (inner loop done its tasks, stealing outer
188 // loop tasks before unlock mutex). Disable parallel here explicitely
189 // to avoid this issue. For related discussion, see:
190 // https://software.intel.com/en-us/forums/intel-threading-building-blocks/topic/285550
191 bool isParallel = false;
192 pg.forEachPrimitive(*this, isParallel);
193 }

199 virtual void visitInstance(geom::Instance& i) override {
200 const auto& ref = i.getReference();
201 // visit the referenced Primitive if it's not visited yet
202 if (mSharedSceneMap.insert(std::make_pair(ref, false)).second) {
203 RTCScene sharedScene = rtcNewScene(mDevice);
204 rtcSetSceneBuildQuality(sharedScene, mGeometry->isStatic() ?
205 RTC_BUILD_QUALITY_HIGH : RTC_BUILD_QUALITY_LOW);
206 geom::internal::PrimitivePrivateAccess::setBVHScene(*ref,
207 static_cast<void*>(sharedScene));
208 BVHBuilder builder(mLayer, mGeometry, mDevice, sharedScene,
209 mSharedSceneMap, mBVHUserData, mGetAssignments);
210 ref->getPrimitive()->accept(builder);
211 rtcCommitScene(sharedScene);
212 // store if the reference contains volumes or surfaces
213 if (mGetAssignments) {
214 ref->setHasSurfaceAssignment(builder.getHasSurfaceAssignment());
215 ref->setHasVolumeAssignment(builder.getHasVolumeAssignment());
216 }
217 // mark the BVH representation of referenced primitive (group)
218 // has been correctly constructed so that all the instances
219 // reference it can start accessing it
220 mSharedSceneMap[ref] = true;
221 }
222 // wait for the first visited instance to construct the shared scene
223 SharedSceneMap::const_iterator it = mSharedSceneMap.find(ref);
224 while (it == mSharedSceneMap.end() || !it->second) {
225 it = mSharedSceneMap.find(ref);
226 }
227 auto pImpl = geom::internal::PrimitivePrivateAccess::getPrimitiveImpl(&i);
228 MNRY_ASSERT_REQUIRE(pImpl != nullptr);
229 MNRY_ASSERT_REQUIRE(pImpl->getType() == geom::internal::Primitive::INSTANCE);
230 auto pInstance = static_cast<geom::internal::Instance*>(pImpl);
231 // bind the BVH representation to corresponding Primitive
232 // or update the BVH representation if Primitive got deformed
233 // (real time frame update case)
234 if (mGeometry->isStatic() || !pInstance->isBVHInitialized()) {
235 pInstance->setBVHHandle(createInstanceInBVH(*pInstance, getGeomFlag()));
236 } else {
237 pInstance->updateBVHHandle();
238 }
239 }

273 std::unique_ptr<geom::internal::BVHHandle> createPolyMeshInBVH(
274 const geom::internal::Mesh& geomMesh, const RTCBuildQuality flag) {
275
276 geom::internal::Mesh::TessellatedMesh mesh;
277 geomMesh.getTessellatedMesh(mesh);
278
279 RTCGeometry rtcGeom;
280 RTCFormat indexBufferFormat;
281 switch (mesh.mIndexBufferType) {
282 case geom::internal::MeshIndexType::TRIANGLE:
283 {
284 rtcGeom = rtcNewGeometry(mDevice, RTC_GEOMETRY_TYPE_TRIANGLE);
285 indexBufferFormat = RTC_FORMAT_UINT3;
286 break;
287 }
288 case geom::internal::MeshIndexType::QUAD:
289 {
290 rtcGeom = rtcNewGeometry(mDevice, RTC_GEOMETRY_TYPE_QUAD);
291 indexBufferFormat = RTC_FORMAT_UINT4;
292 break;
293 }
294 default:
295 MNRY_ASSERT_REQUIRE(false);
296 break;
297 }
298 size_t mbSteps = mesh.mVertexBufferDesc.size();
299 rtcSetGeometryTimeStepCount(rtcGeom, mbSteps);
300
301 // Set up mesh index buffer
302 rtcSetSharedGeometryBuffer(rtcGeom, RTC_BUFFER_TYPE_INDEX, 0,
303 indexBufferFormat, const_cast<void*>(
304 (const void*)mesh.mIndexBufferDesc.mData),
305 mesh.mIndexBufferDesc.mOffset,
306 mesh.mIndexBufferDesc.mStride,
307 mesh.mFaceCount);
308 // Set up the polygon mesh vertex buffers, one for each motion step
309 for (size_t i = 0; i < mbSteps; i++) {
310 rtcSetSharedGeometryBuffer(rtcGeom, RTC_BUFFER_TYPE_VERTEX, i,
311 RTC_FORMAT_FLOAT3, // xyz
312 const_cast<void*>((const void*)mesh.mVertexBufferDesc[i].mData),
313 mesh.mVertexBufferDesc[i].mOffset,
314 mesh.mVertexBufferDesc[i].mStride,
315 mesh.mVertexCount);
316 }
317
318 rtcSetGeometryMask(rtcGeom, resolveVisibilityMask(geomMesh));
319
320 // set intersection filters
321 IntersectionFilterManager* filterManager =
322 new IntersectionFilterManager();
323 bool hasVolumeAssignment = geomMesh.hasVolumeAssignment(mLayer);
324 // force volume primitive to be two sided for odd-even test
325 if (geomMesh.getIsSingleSided() && !hasVolumeAssignment) {
326 filterManager->addIntersectionFilter(&backFaceCullingFilter);
327 filterManager->addOcclusionFilter(&backFaceCullingFilter);
328 }
329 filterManager->addIntersectionFilter(&bssrdfTraceSetIntersectionFilter);
330 if (hasVolumeAssignment) {
331 filterManager->addIntersectionFilter(&manifoldVolumeIntervalFilter);
332 }
333
334 filterManager->addOcclusionFilter(&skipOcclusionFilter);
335 installFilterCallbacks(rtcGeom, filterManager);
336
337 // set user data
338 geom::internal::BVHUserData* userData =
339 new geom::internal::BVHUserData(mLayer, &geomMesh, filterManager);
340 mBVHUserData.emplace_back(userData);
341 rtcSetGeometryUserData(rtcGeom, (void*)userData);
342 uint32_t geomID = rtcAttachGeometry(mParentScene, rtcGeom);
343 rtcCommitGeometry(rtcGeom);
344 return fauxstd::make_unique<geom::internal::BVHHandle>(
345 mParentScene, geomID);
346 }

348 std::unique_ptr<geom::internal::BVHHandle> createQuadricInBVH(
349 const geom::internal::NamedPrimitive& quadric,
350 const RTCBuildQuality flag) {
351
352 RTCGeometry rtcGeom = rtcNewGeometry(mDevice, RTC_GEOMETRY_TYPE_USER);
353 rtcSetGeometryTimeStepCount(rtcGeom, 1);
354 rtcSetGeometryBuildQuality(rtcGeom, flag);
355 rtcSetGeometryUserPrimitiveCount(rtcGeom,
356 quadric.getSubPrimitiveCount());
357
358 // Set up bounds/intersection/occlusion kernel functions
359 rtcSetGeometryBoundsFunction(rtcGeom,
360 quadric.getBoundsFunction(), nullptr);
361 rtcSetGeometryIntersectFunction(rtcGeom,
362 quadric.getIntersectFunction());
363 rtcSetGeometryOccludedFunction(rtcGeom,
364 quadric.getOccludedFunction());
365
366 rtcSetGeometryMask(rtcGeom, resolveVisibilityMask(quadric));
367
368 // set intersection filter
369 IntersectionFilterManager* filterManager =
370 new IntersectionFilterManager();
371 filterManager->addIntersectionFilter(&bssrdfTraceSetIntersectionFilter);
372 if (quadric.hasVolumeAssignment(mLayer)) {
373 filterManager->addIntersectionFilter(
374 &manifoldVolumeIntervalFilter);
375 }
376 filterManager->addOcclusionFilter(&skipOcclusionFilter);
377 installFilterCallbacks(rtcGeom, filterManager);
378
379 // set user data
380 geom::internal::BVHUserData* userData =
381 new geom::internal::BVHUserData(mLayer, &quadric, filterManager);
382 mBVHUserData.emplace_back(userData);
383 rtcSetGeometryUserData(rtcGeom, (void*)userData);
384 uint32_t geomID = rtcAttachGeometry(mParentScene, rtcGeom);
385 rtcCommitGeometry(rtcGeom);
386 return fauxstd::make_unique<geom::internal::BVHHandle>(
387 mParentScene, geomID);
388 }

390 std::unique_ptr<geom::internal::BVHHandle> createCurvesInBVH(
391 const geom::internal::Curves& geomCurves,
392 const geom::Curves::Type curvesType,
393 const geom::Curves::SubType curvesSubType,
394 const int tessellationRate,
395 const RTCBuildQuality flag) {
396
397 rtcGetDeviceError(mDevice); // clear error code
398
399 geom::internal::Curves::Spans spans;
400 geomCurves.getTessellatedSpans(spans);
401
402 RTCGeometry rtcGeom;
403 if (curvesType == geom::Curves::Type::LINEAR && curvesSubType == geom::Curves::SubType::RAY_FACING) {
404 rtcGeom = rtcNewGeometry(mDevice,
405 RTC_GEOMETRY_TYPE_FLAT_LINEAR_CURVE);
406 } else if (curvesType == geom::Curves::Type::BEZIER && curvesSubType == geom::Curves::SubType::RAY_FACING) {
407 rtcGeom = rtcNewGeometry(mDevice,
408 RTC_GEOMETRY_TYPE_FLAT_BEZIER_CURVE);
409 } else if (curvesType == geom::Curves::Type::BSPLINE && curvesSubType == geom::Curves::SubType::RAY_FACING) {
410 rtcGeom = rtcNewGeometry(mDevice,
411 RTC_GEOMETRY_TYPE_FLAT_BSPLINE_CURVE);
412 } else if (curvesType == geom::Curves::Type::LINEAR && curvesSubType == geom::Curves::SubType::ROUND) {
413 rtcGeom = rtcNewGeometry(mDevice,
414 RTC_GEOMETRY_TYPE_ROUND_LINEAR_CURVE);
415 } else if (curvesType == geom::Curves::Type::BEZIER && curvesSubType == geom::Curves::SubType::ROUND) {
416 rtcGeom = rtcNewGeometry(mDevice,
417 RTC_GEOMETRY_TYPE_ROUND_BEZIER_CURVE);
418 } else if (curvesType == geom::Curves::Type::BSPLINE && curvesSubType == geom::Curves::SubType::ROUND) {
419 rtcGeom = rtcNewGeometry(mDevice,
420 RTC_GEOMETRY_TYPE_ROUND_BSPLINE_CURVE);
421 } else {
422 MNRY_ASSERT_REQUIRE(false);
423 }
424 MNRY_ASSERT_REQUIRE(rtcGeom != NULL);
425
426 size_t mbSteps = spans.mVertexBufferDesc.size();
427 rtcSetGeometryTimeStepCount(rtcGeom, mbSteps);
428 rtcSetGeometryBuildQuality(rtcGeom, flag);
429 // Set up span index buffer
430 MNRY_ASSERT_REQUIRE(spans.mIndexBufferDesc.mData != nullptr);
431 rtcSetSharedGeometryBuffer(rtcGeom, RTC_BUFFER_TYPE_INDEX, 0,
432 RTC_FORMAT_UINT,
433 const_cast<void*>((const void*)spans.mIndexBufferDesc.mData),
434 spans.mIndexBufferDesc.mOffset,
435 spans.mIndexBufferDesc.mStride,
436 spans.mSpanCount);
437 // Set up the control vertex data buffers, one for each motion step
438 for (size_t i = 0; i < mbSteps; i++) {
439 MNRY_ASSERT_REQUIRE(spans.mVertexBufferDesc[i].mData != nullptr);
440 rtcSetSharedGeometryBuffer(rtcGeom, RTC_BUFFER_TYPE_VERTEX, i,
441 RTC_FORMAT_FLOAT4, // xyzr
442 const_cast<void*>((const void*)spans.mVertexBufferDesc[i].mData),
443 spans.mVertexBufferDesc[i].mOffset,
444 spans.mVertexBufferDesc[i].mStride,
445 spans.mVertexCount);
446 }
447
448 rtcSetGeometryMask(rtcGeom, resolveVisibilityMask(geomCurves));
449
450 // set intersection filter
451 IntersectionFilterManager* filterManager =
452 new IntersectionFilterManager();
453 filterManager->addIntersectionFilter(&bssrdfTraceSetIntersectionFilter);
454 filterManager->addOcclusionFilter(&skipOcclusionFilter);
455 installFilterCallbacks(rtcGeom, filterManager);
456
457 rtcSetGeometryTessellationRate(rtcGeom, tessellationRate);
458
459 // set user data
460 geom::internal::BVHUserData* userData =
461 new geom::internal::BVHUserData(mLayer, &geomCurves, filterManager);
462 mBVHUserData.emplace_back(userData);
463 rtcSetGeometryUserData(rtcGeom, (void*)userData);
464 uint32_t geomID = rtcAttachGeometry(mParentScene, rtcGeom);
465 rtcCommitGeometry(rtcGeom);
466
467 // catch any embree errors in this function
468 MNRY_ASSERT_REQUIRE(rtcGetDeviceError(mDevice) == RTC_ERROR_NONE);
469
470 return fauxstd::make_unique<geom::internal::BVHHandle>(
471 mParentScene, geomID);
472 }

474 std::unique_ptr<geom::internal::BVHHandle> createInstanceInBVH(
475 const geom::internal::Instance& instance, const RTCBuildQuality flag) {
476
477 RTCGeometry rtcGeom = rtcNewGeometry(mDevice, RTC_GEOMETRY_TYPE_USER);
478 rtcSetGeometryUserPrimitiveCount(rtcGeom, 1);
479 rtcSetGeometryBuildQuality(rtcGeom, flag);
480 // instancing kernel handle motion blur through matrix decomposition
481 // so we don't need to feed in multiple buffers for motion blur case
482 rtcSetGeometryTimeStepCount(rtcGeom, 1);
483 // Set up bounds/intersection/occlusion kernel functions
484 rtcSetGeometryBoundsFunction(rtcGeom,
485 instance.getBoundsFunction(), nullptr);
486 rtcSetGeometryIntersectFunction(rtcGeom,
487 instance.getIntersectFunction());
488 rtcSetGeometryOccludedFunction(rtcGeom,
489 instance.getOccludedFunction());
490
491 // The visibility mask needs to take volume assignments into account.
492 // We use the visibility flags of the instance geometry, but do the
493 // volume/surface shift based on the reference's assignments.
494 const std::shared_ptr<geom::SharedPrimitive> &ref = instance.getReference();
495 bool hasSurfaceAssignment = ref->getHasSurfaceAssignment();
496 bool hasVolumeAssignment = ref->getHasVolumeAssignment();
497 if (!hasSurfaceAssignment && !hasVolumeAssignment) {
498 // In the case of multi-level instancing we'll need to
499 // recurse through the primitives to check for volume assignments
500 GetAssignments ga(hasVolumeAssignment, hasSurfaceAssignment);
501 ref->getPrimitive()->accept(ga);
502 }
503 unsigned int mask = 0;
504 if (hasSurfaceAssignment) {
505 mask |= mGeometry->getVisibilityMask();
506 }
507 if (hasVolumeAssignment) {
508 mask |= mGeometry->getVisibilityMask() << scene_rdl2::rdl2::sNumVisibilityTypes;
509 }
510 rtcSetGeometryMask(rtcGeom, mask);
511
512 // set user data
513 geom::internal::BVHUserData* userData =
514 new geom::internal::BVHUserData(mLayer, &instance, nullptr);
515 mBVHUserData.emplace_back(userData);
516 rtcSetGeometryUserData(rtcGeom, (void*)userData);
517
518 uint32_t geomID = rtcAttachGeometry(mParentScene, rtcGeom);
519 rtcCommitGeometry(rtcGeom);
520 return fauxstd::make_unique<geom::internal::BVHHandle>(
521 mParentScene, geomID);
522 }
523
524 std::unique_ptr<geom::internal::BVHHandle> createVolumeInBVH(
525 const geom::internal::VdbVolume& geomVolume,
526 const RTCBuildQuality flag) {
527
528 geom::internal::BufferDesc vertexBufferDesc[2];
529 geom::internal::BufferDesc indexBufferDesc;
530 size_t vertexCount, faceCount, mbSteps;
531 geomVolume.getTessellatedMesh(vertexBufferDesc, indexBufferDesc,
532 vertexCount, faceCount, mbSteps);
533
534 RTCGeometry rtcGeom = rtcNewGeometry(mDevice, RTC_GEOMETRY_TYPE_QUAD);
535 // the proxy bounding box for volume is always static,
536 // we use velocity data for volume motion blur
537 rtcSetGeometryTimeStepCount(rtcGeom, mbSteps);
538 rtcSetGeometryBuildQuality(rtcGeom, flag);
539
540 // Set up the quad mesh index buffer
541 rtcSetSharedGeometryBuffer(rtcGeom, RTC_BUFFER_TYPE_INDEX, 0,
542 RTC_FORMAT_UINT4,
543 const_cast<void*>((const void*)indexBufferDesc.mData),
544 indexBufferDesc.mOffset,
545 indexBufferDesc.mStride,
546 faceCount);
547 // Set up the quad mesh vertex buffer, one for each motion step
548 for (size_t i = 0; i < mbSteps; i++) {
549 rtcSetSharedGeometryBuffer(
550 rtcGeom, RTC_BUFFER_TYPE_VERTEX, i, RTC_FORMAT_FLOAT3,
551 const_cast<void*>(static_cast<const void*>(vertexBufferDesc[i].mData)),
552 vertexBufferDesc[i].mOffset,
553 vertexBufferDesc[i].mStride,
554 vertexCount);
555 }
556
557 rtcSetGeometryMask(rtcGeom, resolveVisibilityMask(geomVolume));
558
559 // set intersection filters
560 IntersectionFilterManager* filterManager =
561 new IntersectionFilterManager();
562 if (geomVolume.hasVolumeAssignment(mLayer)) {
563 filterManager->addIntersectionFilter(&vdbVolumeIntervalFilter);
564 }
565 filterManager->addOcclusionFilter(&skipOcclusionFilter);
566 installFilterCallbacks(rtcGeom, filterManager);
567
568 // set user data
569 geom::internal::BVHUserData* userData =
570 new geom::internal::BVHUserData(mLayer, &geomVolume, filterManager);
571 mBVHUserData.emplace_back(userData);
572 rtcSetGeometryUserData(rtcGeom, (void*)userData);
573 uint32_t geomID = rtcAttachGeometry(mParentScene, rtcGeom);
574 rtcCommitGeometry(rtcGeom);
575 return fauxstd::make_unique<geom::internal::BVHHandle>(
576 mParentScene, geomID);
577 }

583
584 unsigned int resolveVisibilityMask(
585 const geom::internal::NamedPrimitive& np) const
586 {
587 unsigned int mask = 0;
588 if (np.hasSurfaceAssignment(mLayer)) {
589 // visibility for surface ray query

665void
666buildBVHBottomUp(const scene_rdl2::rdl2::Layer* layer, scene_rdl2::rdl2::Geometry* geometry,
667 RTCDevice& rtcDevice, RTCScene& rootScene,
668 SharedSceneMap& visitedBVHScene,
669 std::unordered_set<scene_rdl2::rdl2::Geometry*>& visitedGeometry,
670 BVHUserDataList& bvhUserData)
671{
672 geom::Procedural* procedural = geometry->getProcedural();
673 // All parts in a procedural are unassigned in the layer
674 if (!procedural) {
675 return;
676 }
677 // This BVH part for this particular rdl geometry has been constructed
678 if (visitedGeometry.find(geometry) != visitedGeometry.end()) {
679 return;
680 }
681 // Do a bottom up traversal so that referenced BVH got built first
682 const scene_rdl2::rdl2::SceneObjectVector& references =
683 geometry->get(scene_rdl2::rdl2::Geometry::sReferenceGeometries);
684 for (const auto& ref : references) {
685 if (!ref->isA<scene_rdl2::rdl2::Geometry>()) {
686 continue;
687 }
688 scene_rdl2::rdl2::Geometry* referencedGeometry = ref->asA<scene_rdl2::rdl2::Geometry>();
689 buildBVHBottomUp(layer, referencedGeometry, rtcDevice, rootScene,
690 visitedBVHScene, visitedGeometry, bvhUserData);
691 }
692 // We disable the parallel here to solve the non-deterministic
693 // issue for some hair/fur related scenes.
694 // Also, testing shows no speed gain is achieved when
695 // parallelize this for loop.
696 bool doParallel = false;
697 if (procedural->isReference()) {
698 const std::shared_ptr<geom::SharedPrimitive>& ref =
699 procedural->getReference();
700 if (visitedBVHScene.insert(std::make_pair(ref, false)).second) {
701 RTCScene sharedScene = rtcNewScene(rtcDevice);
702 rtcSetSceneBuildQuality(sharedScene, geometry->isStatic()?
703 RTC_BUILD_QUALITY_HIGH : RTC_BUILD_QUALITY_LOW);
704 geom::internal::PrimitivePrivateAccess::setBVHScene(*ref,
705 static_cast<void*>(sharedScene));
706 BVHBuilder builder(layer, geometry, rtcDevice, sharedScene,
707 visitedBVHScene, bvhUserData, /* get assignments = */ true);
708 ref->getPrimitive()->accept(builder);
709 rtcCommitScene(sharedScene);
710 // mark the BVH representation of referenced primitive (group)
711 // has been correctly constructed so that all the instances
712 // reference it can start accessing it
713 visitedBVHScene[ref] = true;
714 }
715 } else {
716 BVHBuilder bvhBuilder(layer, geometry, rtcDevice, rootScene,
717 visitedBVHScene, bvhUserData, /* get assignments = */ false);
718 procedural->forEachPrimitive(bvhBuilder, doParallel);
719 }
720 visitedGeometry.insert(geometry);
721}

723void
724EmbreeAccelerator::build(OptimizationTarget accelMode, ChangeFlag changeFlag,
725 const scene_rdl2::rdl2::Layer *layer,
726 const scene_rdl2::rdl2::SceneContext::GeometrySetVector& geometrySets,
727 const scene_rdl2::rdl2::Layer::GeometryToRootShadersMap* g2s)
728{
729 if (changeFlag == ChangeFlag::ALL) {
730 if (accelMode == OptimizationTarget::HIGH_QUALITY_BVH_BUILD) {
731 rtcSetSceneBuildQuality(mRootScene, RTC_BUILD_QUALITY_HIGH);
732 rtcSetSceneFlags(mRootScene, RTC_SCENE_FLAG_NONE);
733 } else {
734 rtcSetSceneBuildQuality(mRootScene, RTC_BUILD_QUALITY_LOW);
735 rtcSetSceneFlags(mRootScene, RTC_SCENE_FLAG_DYNAMIC);
736 }
737 }
738 scene_rdl2::rec_time::RecTime recTime;
739
740 recTime.start();
741 SharedSceneMap visitedBVHScene;
742 std::unordered_set<scene_rdl2::rdl2::Geometry*> visitedGeometry;
743 for (const auto& geometrySet : geometrySets) {
744 const scene_rdl2::rdl2::SceneObjectIndexable& geometries = geometrySet->getGeometries();
745 for (auto& sceneObject : geometries) {
746 scene_rdl2::rdl2::Geometry* geometry = sceneObject->asA<scene_rdl2::rdl2::Geometry>();
747 if (g2s != nullptr && g2s->find(geometry) == g2s->end()) {
748 continue;
749 }
750 buildBVHBottomUp(layer, geometry, mDevice, mRootScene,
751 visitedBVHScene, visitedGeometry, mBVHUserData);
752 }
753 }
754 mBvhBuildProceduralTime = recTime.end();
755
756 // now build the root scene
757 recTime.start();
758 rtcCommitScene(mRootScene);
759 mRtcCommitTime = recTime.end();
760}

773{
774 MNRY_ASSERT(isValidRay(ray));
775 // carry per ray data that we need to process after rtcIntersect1

777 mcrt_common::IntersectContext context;
778 context.mRayExtension = &ray.ext;

801scene_rdl2::math::BBox3f
802EmbreeAccelerator::getBounds() const
803{
804 RTCBounds bounds;
805 rtcGetSceneBounds(mRootScene, &bounds);
806
807 if (bounds.lower_x > bounds.upper_x ||
808 bounds.lower_y > bounds.upper_y ||
809 bounds.lower_z > bounds.upper_z) {
810 // There is probably no geometry in the scene. Return a default bounding
811 // box centered at the origin.
812 return scene_rdl2::math::BBox3f(scene_rdl2::math::Vec3f(-1.f), scene_rdl2::math::Vec3f(1.f));
813 }
814
815 return scene_rdl2::math::BBox3f(*((scene_rdl2::math::Vec3f *)&bounds.lower_x), *((scene_rdl2::math::Vec3f *)&bounds.upper_x));
816}
