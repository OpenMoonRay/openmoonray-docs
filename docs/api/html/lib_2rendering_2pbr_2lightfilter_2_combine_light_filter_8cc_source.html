64void
65CombineLightFilter::update(const LightFilterMap& lightFilters,
66 const Mat4d& world2Render)
67{
68 if (!mRdlLightFilter) {
69 return;
70 }
71
72 mLightFiltersVec.clear();
73
74 const rdl2::SceneObjectVector& rdlLightFilters =
75 mRdlLightFilter->get<rdl2::SceneObjectVector>(sLightFiltersKey);
76
77 for (rdl2::SceneObject* sceneObject : rdlLightFilters) {
78 const rdl2::LightFilter *rdlLightFilter = sceneObject->asA<rdl2::LightFilter>();
79 if (rdlLightFilter == nullptr) {
80 // not a light filter, ignore
81 continue;
82 }
83 auto search = lightFilters.find(rdlLightFilter);
84 MNRY_ASSERT(search != lightFilters.end());
85 const LightFilter *filter = search->second.get();
86 mLightFiltersVec.push_back(filter);
87 }
88 mLightFilters = mLightFiltersVec.data();
89 mNumLightFilters = mLightFiltersVec.size();
90
91 mMode = mRdlLightFilter->get<rdl2::Int>(sModeKey);
92}

115Color
116CombineLightFilter::eval(const EvalData& data) const
117{
118 Color val;
119
120 switch (mMode) {
121 case MULTIPLY:
122 val = Color(1.f);
123 for (int i = 0; i < mLightFiltersVec.size(); i++) {
124 val *= mLightFiltersVec[i]->eval(data);
125 }
126 break;
127 case MIN:
128 val = Color(1.f);
129 for (int i = 0; i < mLightFiltersVec.size(); i++) {
130 Color filterVal = mLightFiltersVec[i]->eval(data);
131 val.r = std::min(val.r, filterVal.r);
132 val.g = std::min(val.g, filterVal.g);
133 val.b = std::min(val.b, filterVal.b);
134 }
135 break;
136 case MAX:
137 val = Color(0.f);
138 for (int i = 0; i < mLightFiltersVec.size(); i++) {
139 Color filterVal = mLightFiltersVec[i]->eval(data);
140 val.r = std::max(val.r, filterVal.r);
141 val.g = std::max(val.g, filterVal.g);
142 val.b = std::max(val.b, filterVal.b);
143 }
144 break;
145 case ADD:
146 val = Color(0.f);
147 for (int i = 0; i < mLightFiltersVec.size(); i++) {
148 val += mLightFiltersVec[i]->eval(data);
149 }
150 val.r = std::min(val.r, 1.f);
151 val.g = std::min(val.g, 1.f);
152 val.b = std::min(val.b, 1.f);
153 break;
154 case SUBTRACT:
155 val = Color(1.f);
156 for (int i = 0; i < mLightFiltersVec.size(); i++) {
157 if (i == 0) {
158 val = mLightFiltersVec[i]->eval(data);
159 } else {
160 val -= mLightFiltersVec[i]->eval(data);
161 }
162 }
163 val.r = std::max(val.r, 0.f);
164 val.g = std::max(val.g, 0.f);
165 val.b = std::max(val.b, 0.f);
166 break;
167 default:
168 MNRY_ASSERT(false);
169 }
170
171 return val;
172}
