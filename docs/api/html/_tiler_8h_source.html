8// This class abstracts the conversions between linear and the minimal tiling
9// scheme which is implemented here.
10//
11// Terminology used:
12//
13// Linear coordinates: (x, y) coordinates which are in scanline format.
14// For example (0, 0) would refer to the bottom left pixel and (w - 1, h - 1)
15// would refer to the top right pixel. Incrementing x coordinates always go
16// left to right, and incrementing y coordinates always go bottom to top.
17//
18// Tiled coordinates: The memory layout of a buffer is not linear.
19// Tiling is done on 8x8 quads. The 64 pixels of each tile are laid out
20// contiguously in memory in a linear bottom-left to top-right fashion.
21// The tiles themselves also follow a linear pattern the context of the
22// whole image.
23//
24// Tiled offset: A tiled offset refers to the offset from the start of a
25// tiled buffer to a particular pixel in memory.
26//
27// Coarse tiled offset: A tiled offset refers to the offset from the start of
28// a tiled buffer to the start of a particular tile in memory.
29//

55 // Pass in desired (potentially unaligned) width and height.
56 Tiler(unsigned w, unsigned h) :
57 mOriginalW(w),
58 mOriginalH(h),
59 mAlignedW((w + 7) & ~7),
60 mAlignedH((h + 7) & ~7),
61 mNumTiles((mAlignedW * mAlignedH) >> 6)
62 {
63 }

65 unsigned linearCoordsToCoarseTileOffset(unsigned lx, unsigned ly) const
66 {
67 unsigned tileIdx = (ly >> 3) * (mAlignedW >> 3) + (lx >> 3);
68 return tileIdx << 6;
69 }
70
71 unsigned linearCoordsToTiledOffset(unsigned lx, unsigned ly) const
72 {
73 return linearCoordsToCoarseTileOffset(lx, ly) + ((ly & 7) << 3) + (lx & 7);
74 }
75
76 void linearToTiledCoords(unsigned lx, unsigned ly, unsigned *tx, unsigned *ty) const
77 {
78 MNRY_ASSERT(lx < mOriginalW && ly < mOriginalH);
79
80 unsigned tileOfs = linearCoordsToTiledOffset(lx, ly);
81 getTiledCoords(tileOfs, tx, ty);
82
83 MNRY_ASSERT(*tx < mAlignedW && *ty < mAlignedH);
84 }
85
86 bool tiledToLinearCoords(unsigned tx, unsigned ty, unsigned *lx, unsigned *ly) const
87 {
88 MNRY_ASSERT(tx < mAlignedW && ty < mAlignedH);
89
90 unsigned tileOfs = mAlignedW * ty + tx;
91
92 unsigned a = tileOfs >> 6;
93 unsigned b = mAlignedW >> 3;
94
95 unsigned tileY = a / b;
96 unsigned tileX = a % b;
97
98 *lx = (tileX << 3) + (tileOfs & 7);
99 *ly = (tileY << 3) + ((tileOfs & 63) >> 3);
100
101 // Returns false if linear coords are outside of the valid range.
102 return (*lx < mOriginalW && *ly < mOriginalH);
103 }
104
105 unsigned getTiledOffset(unsigned tx, unsigned ty) const
106 {
107 MNRY_ASSERT(tx < mAlignedW && ty < mAlignedH);
108 return mAlignedW * ty + tx;
109 }
110
111 void getTiledCoords(unsigned tileOfs, unsigned *tx, unsigned *ty) const
112 {
113 MNRY_ASSERT(tileOfs < (mAlignedW * mAlignedH));
114
115 *ty = tileOfs / mAlignedW;
116 *tx = tileOfs % mAlignedW;
117
118 MNRY_ASSERT(getTiledOffset(*tx, *ty) == tileOfs);
119 }
120
121 unsigned getLinearOffset(unsigned lx, unsigned ly) const
122 {
123 MNRY_ASSERT(lx < mOriginalW && ly < mOriginalH);
124 return mOriginalW * ly + lx;
125 }
126
127 void getLinearCoords(unsigned linearOfs, unsigned *lx, unsigned *ly) const
128 {
129 MNRY_ASSERT(linearOfs < (mOriginalW * mOriginalH));
130
131 *ly = linearOfs / mOriginalW;
132 *lx = linearOfs % mOriginalW;
133
134 MNRY_ASSERT(getLinearOffset(*lx, *ly) == linearOfs);
135 }
136
137 // Desired user width required. It doesn't have to be tile aligned.
138 unsigned mOriginalW;
139
140 // Desired user height required. It doesn't have to be tile aligned.
141 unsigned mOriginalH;
142
143 // Tile aligned version of mOriginalW, always >= than mOriginalW.
144 unsigned mAlignedW;
145
146 // Tile aligned version of mOriginalH, always >= than mOriginalH.
147 unsigned mAlignedH;
148
149 // Total tiles required to cover the buffer.
150 unsigned mNumTiles;
151};
