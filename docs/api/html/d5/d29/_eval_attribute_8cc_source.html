23math::Vec3f
24evalNormalImpl(const rdl2::Shader* const obj,
25 shading::TLState *tls,
26 const State& state,
27 const math::Vec3f& normalMapValue,
28 const float normalDial,
29 const int normalMapSpace,
30 bool& needsAdapting,
31 float *lengthN = nullptr)
32{
33 const math::Vec3f &N = state.getN();
34
35 // Check for numerical issues
36 if (// Check for nan and infinity
37 (std::isnan(normalMapValue.x) ||
38 std::isnan(normalMapValue.y) ||
39 std::isnan(normalMapValue.z)) ||
40 (!std::isfinite(normalMapValue.x) ||
41 !std::isfinite(normalMapValue.y) ||
42 !std::isfinite(normalMapValue.z)) ||
43 // We require a non-zero length vector
44 math::isZero(normalMapValue)) {
45
46 auto &objTls = obj->getThreadLocalObjectState()[mcrt_common::getThreadIdx(tls)];
47 objTls.mLogs.log(obj->getInvalidNormalMapLogEvent());
48 needsAdapting = false;
49 return N;
50 }
51
52 math::Vec3f result = normalMapValue;
53
54 if (normalMapSpace == 0) { // TBN space
55 // Re-center sampled normal from [0, 1] --> [-1, 1]
56 // z is usually encoded from [0.5, 1.0] so that this re-centering
57 // keeps it positive (in [0, 1])
58 result = 2.0f * result - math::Vec3f(1.0f);
59 }
60
61 // Calculating the normal's length here is necessary for the Toksvig
62 // normal AA strategy used in the Dwa materials since it must then
63 // be normalized to render properly.
64 if (lengthN != nullptr) {
65 *lengthN = result.length();
66 }
67
68 if (normalMapSpace == 0) { // TBN space
69 // We check that the vector be in the upper half plane of tangent space.
70 if (result.z < 0.f) {
71 auto &objTls = obj->getThreadLocalObjectState()[mcrt_common::getThreadIdx(tls)];
72 objTls.mLogs.log(obj->getInvalidNormalMapLogEvent());
73 needsAdapting = false;
74 return N;
75 }
76
77 // Transform from tangent space to shade space
78 const math::ReferenceFrame frame(N, math::normalize(state.getdPds()));
79 result = frame.localToGlobal(result);
80 }
81
82 result.normalize();
83
84 // Linearly interpolate between the surface normal and the mapped
85 // normal if our blend factor is anything other than 1.
86 if (normalDial != 1.0f) {
87 result = math::normalize(N + (result - N) * math::saturate(normalDial));
88 }
89
90 needsAdapting = true;
91 return result;
92}

97math::Vec3f
98evalNormal(const rdl2::Shader* const obj,
99 shading::TLState *tls,
100 const State& state,
101 const math::Vec3f& normalMapValue,
102 const float normalDial,
103 const int normalMapSpace,
104 float *lengthN)
105{
106 bool needsAdapting;
107 math::Vec3f result = evalNormalImpl(obj, tls, state,
108 normalMapValue, normalDial, normalMapSpace,
109 needsAdapting,
110 lengthN);
111 if (needsAdapting) {
112 // hack the result if it is not physically plausible
113 result = state.adaptNormal(result);
114 }
115 return result;
116}

118math::Vec3f
119evalToonNormal(const rdl2::Shader* const obj,
120 shading::TLState *tls,
121 const State& state,
122 const math::Vec3f& normalMapValue,
123 const float normalDial,
124 const int normalMapSpace,
125 float *lengthN)
126{
127 // Intended for use by diffuse lobes from NPR materials only.
128 bool needsAdapting;
129 math::Vec3f result = evalNormalImpl(obj, tls, state,
130 normalMapValue, normalDial, normalMapSpace,
131 needsAdapting);
132 if (needsAdapting) {
133 // hack the result if it is not physically plausible,
134 // unless the material is requesting physically implausible normals.
135
136 // Some NPR materials that want to allow for completely arbitrary shading normals
137 // can request that the integrator does not perform any light culling based on the
138 // normal. In those cases, adaptToonNormal() will check the flag set on the Intersection
139 // and not adapt the normal. This allows the explicit input normals to be unbent
140 // and the black artifacts that would otherwise occur are mitigated because
141 // there is no light culling.
142 result = state.adaptToonNormal(result);
143 }
144 return result;
145}

147/// Same as evalColor but for floats, including an additive term before
148/// the multiplicative term = (mapEvalValue + preAddValue) * attributeValue
149float
150evalFloatWithPreAdd(const rdl2::Shader* const obj,
151 const rdl2::AttributeKey<rdl2::Float>& valueKey,
152 shading::TLState *tls,
153 const State& state,
154 float preAddValue)
155{
156 float result = obj->get(valueKey);
157 if (!math::isZero(result)) {
158 const rdl2::SceneObject* mapObj = obj->getBinding(valueKey);
159 if (mapObj != nullptr) {
160 const rdl2::Map* map = mapObj->asA<rdl2::Map>();
161 math::Color sampleColor;
162 sample(obj, map, tls, state, &sampleColor);
163
164 float avg = (sampleColor.r + sampleColor.g + sampleColor.b) / 3.0f;
165 result *= (avg + preAddValue);
166 }
167 }
168 return result;
169}

172/// Same as evalColor but for Vec2f
173math::Vec2f
174evalVec2f(const rdl2::Shader* const obj,
175 const rdl2::AttributeKey<rdl2::Vec2f>& valueKey,
176 shading::TLState *tls,
177 const State& state)
178{
179 math::Vec2f result = obj->get(valueKey);
180 if (!math::isEqual(result, math::Vec2f(math::zero))) {
181 const rdl2::SceneObject* mapObj = obj->getBinding(valueKey);
182 if (mapObj != nullptr) {
183 const rdl2::Map* map = mapObj->asA<rdl2::Map>();
184 math::Color sampleColor;
185 sample(obj, map, tls, state, &sampleColor);
186
187 result[0] *= sampleColor.r;
188 result[1] *= sampleColor.g;
189 }
190 }
191 return result;
192}

195/// Similar to evalColor for Vec3f. Note that if the attribute is bound
196/// to a map, the attribute value is ignored and is NOT multiplied to the
197/// result of the map. If the attribute is not bound to a map, the attribute
198/// value is returned.
199// TODO: Fix this inconsistency compared to other eval functions
200math::Vec3f
201evalVec3f(const rdl2::Shader* const obj,
202 const rdl2::AttributeKey<rdl2::Vec3f>& valueKey,
203 shading::TLState *tls,
204 const State& state)
205{
206 math::Vec3f result = obj->get(valueKey);
207 const rdl2::SceneObject* mapObj = obj->getBinding(valueKey);
208 if (mapObj != nullptr) {
209 const rdl2::Map* map = mapObj->asA<rdl2::Map>();
210 math::Color input_vec3;
211 sample(obj, map, tls, state, &input_vec3);
212
213 result[0] = input_vec3.r;
214 result[1] = input_vec3.g;
215 result[2] = input_vec3.b;
216 }
217 return result;
218}

221/// Convenience function that evaluates an attribute triplet: color, factor and
222/// boolean toggle. This is a very common case in material shaders (ex:
223/// "diffuse color", "diffuse factor", "show diffuse").
224/// This function will only try to evaluate the color attribute if the boolean
225/// toggle is on and if the factor attribute is non-zero.
226/// This function returns the product of the three.
227math::Color
228evalColorComponent(const rdl2::Shader* const obj,
229 const rdl2::AttributeKey<rdl2::Bool>& showKey,
230 const rdl2::AttributeKey<rdl2::Float>& factorKey,
231 const rdl2::AttributeKey<rdl2::Rgb>& valueKey,
232 shading::TLState *tls,
233 const State& state)
234{
235 math::Color result(0.0f);
236
237 if (obj->get(showKey)) {
238 float factor = obj->get(factorKey);
239 if (!math::isZero(factor)) {
240 result = factor * evalColor(obj, valueKey, tls, state);
241 }
242 }
243 return result;
244}
