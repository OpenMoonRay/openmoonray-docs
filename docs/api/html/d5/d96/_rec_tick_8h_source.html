33inline void
34RecTick::start()
35{
36 uint32_t high, low;
37
38 asm volatile("CPUID\n\t"
39 "RDTSC\n\t"
40 "mov %%edx, %0\n\t"
41 "mov %%eax, %1\n\t": "=r" (high), "=r" (low)::
42 "%rax", "%rbx", "%rcx", "%rdx");
43
44 mStartTick = ((uint64_t(high)) << 32) | low;
45}
46
47inline uint64_t
48RecTick::end()
49{
50 uint32_t high, low;
51
52 asm volatile("RDTSCP\n\t"
53 "mov %%edx, %0\n\t"
54 "mov %%eax, %1\n\t"
55 "CPUID\n\t": "=r" (high), "=r" (low)::
56 "%rax", "%rbx", "%rcx", "%rdx");
57
58 mEndTick = ((uint64_t(high)) << 32) | low;
59
60 return mEndTick - mStartTick;
61}
