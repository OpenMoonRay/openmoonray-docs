68 // crawl all active tile inside ActivePixels
69 template <typename F>
70 static void crawlAllActivePixelsTile(const ActivePixels &activePixels, F tileFunc)
71 {
72 unsigned tileId = 0;
73 for (unsigned yTileId = 0; yTileId < activePixels.getNumTilesY(); ++yTileId) {
74 for (unsigned xTileId = 0; xTileId < activePixels.getNumTilesX(); ++xTileId) {
75 tileFunc(tileId);
76 tileId++;
77 }
78 }

91 // access pixel by pixId inside activePixels
92 template <typename F>
93 static void accessPixel(const ActivePixels &activePixels, const unsigned pixId, F tileFunc)
94 {
95 unsigned pixX = pixId % activePixels.getWidth();
96 unsigned pixY = pixId / activePixels.getWidth();
97 unsigned tileX = pixX / 8;
98 unsigned tileY = pixY / 8;
99 unsigned tileId = tileY * activePixels.getNumTilesX() + tileX;
100 unsigned localX = pixX % 8;
101 unsigned localY = pixY % 8;
102 unsigned shift = localY * 8 + localX;
103 /* useful debug info
104 std::cerr << "pix(" << pixX << ',' << pixY << ") tile(" << tileX << ',' << tileY << ')'
105 << " tileId:" << tileId << " local(" << localX << ',' << localY << ')'
106 << " shift:" << shift << std::endl;
107 */
108 tileFunc(tileId, shift);
109 }
