23 template <typename Proxy_t, typename Base_t>
24 void
25 registerProxyPyBinding(const std::string& basename)
26 {
27 static_assert(std::is_base_of<Base_t, Proxy_t>::value,
28 "In registerProxyPyBinding<Proxy_t, Base_t>() function template, "
29 "Base_t must be an ancestor of Proxy_t.");
30
31 // Function called exactly once, no need for static const string here.
32 const std::string proxyDocstring =
33 "Displacement, CameraProxy, EnvMapProxy, GeometryProxy, LightProxy, MapProxy, "
34 "MaterialProxy, and SceneObjectProxy define proxy classes for objects of each "
35 "customization point in RDL2. \n"
36 "\n"
37 "Effectively these objects will invoke the proper chain of constructors and "
38 "have the same set of attributes as the objects they are standing in for, "
39 "but don't provide the rich interface of those objects. As such, they don't "
40 "drag in any library dependencies. \n"
41 "\n"
42 "This is useful if you want to create objects of those types, but don't want "
43 "to link with or distribute the huge chain of dependencies that your DSOs "
44 "might have. Those are still needed for rendering, but for a content tool which "
45 "just needs to set attribute data those dependencies are overkill. \n"
46 "\n"
47 "Built in classes that come for free with RDL (like the GeometrySet, Layer, "
48 "LightSet, and SceneVariables) never need to be proxied, because they are "
49 "always fully available and have no extra dependencies.";
50
51 const std::string typeName = basename + "Proxy";
52 bp::class_<Proxy_t, std::shared_ptr<Proxy_t>,
53 bp::bases<Base_t>,
54 boost::noncopyable>(
55 typeName.c_str(),
56 proxyDocstring.c_str(),
57 bp::no_init)
58
59 .def(bp::init<const rdl2::SceneClass&, const std::string&>(
60 ( bp::arg("sceneClass"), bp::arg("name") )));
61 }
