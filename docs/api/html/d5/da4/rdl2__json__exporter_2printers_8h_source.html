143// This is a little confusing, but, because pointer is not part of the typeid,
144// I can't do a template specialization that takes a pointer. And I can't pass
145// the object by reference because it might be NULL. So, I'm using function
146// overloading instead.
147Json::Value
148getJsonForValue(const scene_rdl2::rdl2::SceneObject* obj)
149{
150 if (obj) return Json::Value(obj->getName());
151 return Json::Value();
152}

162template <typename T>
163Json::Value
164outputDefaultVectorHelper(const scene_rdl2::rdl2::Attribute& attr)
165{
166 Json::Value arrayValue(Json::arrayValue);
167 auto vec = attr.getDefaultValue<T>();
168 for (const auto& value : vec) {
169 arrayValue.append(getJsonForValue(value));
170 }
171
172 return arrayValue;
173}
