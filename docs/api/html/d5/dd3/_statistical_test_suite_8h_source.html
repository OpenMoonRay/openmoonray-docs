19constexpr double uniformVarianceContinuous(double min, double max) noexcept
20{
21 const double diff = (max - min);
22 return 1.0/12.0 * diff * diff;
23}
24
25constexpr double uniformVarianceDiscrete(std::uint32_t min, std::uint32_t max) noexcept
26{
27 const std::uint32_t n = max - min; // Exclusive max: e.g. [0, 4) is four values
28
29 // return 1.0/12.0 * (n*n - 1);
30 // To avoid overflow (there's a good chance that the range encompasses the
31 // entire representation of the type), we will re-write the above.
32 constexpr double sqrt12 = 3.464101615137754587054892683011744;
33 return (n/sqrt12 - 1.0/sqrt12) * (n/sqrt12 + 1.0/sqrt12);
34}

36template <typename T>
37class UniformCDFContinuous
38{
39public:
40 constexpr UniformCDFContinuous(T a, T b) noexcept : mA(a), mB(b) {}
41 T operator()(T x) const noexcept { return (x - mA) / (mB - mA); }

43private:
44 T mA;
45 T mB;
46};

49class UniformCDFDiscrete
50{
51public:
52 constexpr UniformCDFDiscrete(T a, T b) noexcept : mA(a), mB(b) {}
53 T operator()(T x) const noexcept
54 {
55 const std::uint32_t n = mB - mA; // Exclusive max: e.g. [0, 4) is four values

60 T mA;

64template <template <typename> class LowerCheck, template <typename> class UpperCheck, typename Iterator>
65bool testRange(typename std::iterator_traits<Iterator>::value_type lower,
66 typename std::iterator_traits<Iterator>::value_type upper,
67 Iterator first, Iterator last)
68{
69 using Type = typename std::iterator_traits<Iterator>::value_type;
70
71 LowerCheck<Type> lc;
72 UpperCheck<Type> uc;
73 for ( ; first != last; ++first) {
74 const bool lower_check = lc(*first, lower);
75 const bool upper_check = uc(*first, upper);
76 if (lower_check == false || upper_check == false) {
77 return false;
78 }
79 }
80 return true;
81}

83template <typename FwdIter>
84inline double empiricalDistributionFunction(FwdIter first, FwdIter last, double x)
85{
86 unsigned count = 0;
87 unsigned numLess = 0;
88 for ( ; first != last; ++first) {
89 ++count;
90 if (*first <= x) {
91 ++numLess;
92 }
93 }
94 return static_cast<double>(numLess) / static_cast<double>(count);
95}

97// This test is described in
98// https://www.johndcook.com/Beautiful_Testing_ch10.pdf
99// https://en.wikipedia.org/wiki/Kolmogorov%E2%80%93Smirnov_test
100// https://www.itl.nist.gov/div898/handbook/eda/section3/eda35g.htm
101//
102// The Kolmogorov-Smirnov test compares the empirical distribution with the
103// theoretical distribution. If the distance in values between the two
104// distributions is too great, the test fails.
105//
106// John Cook, in the chapter referenced above, does some hand-waving and states
107// that for a large n (1,000 is large enough), we expect this distance between
108// the empirical and theoretical distributions to be between 0.07089 and 1.5174
109// around 98% of the time.
110template <typename FwdIter, typename CDF>
111bool testKolmogorovSmirnov(FwdIter first, FwdIter last, CDF cdf)
112{
113 const auto n = std::distance(first, last);
114 const auto sqrtn = std::sqrt(static_cast<double>(n));
115
116 auto kPos = std::numeric_limits<double>::lowest();
117 for (auto it = first; it != last; ++it) {
118 const double x = *it;
119 const double Fx = cdf(x);
120 const double Fn = empiricalDistributionFunction(first, last, x);
121 kPos = std::max(kPos, std::abs(Fx - Fn));
122 }
123
124 // Based off of a Kolmogorov-Smirnov table with 95% confidence interval.
125 // This test should fail 5% of the time.
126 return kPos * sqrtn <= 1.36;
127}

129template <typename T>
130bool testMean(const fb_util::RunningStats<T>& stats, T expectedMean, T expectedVariance)
131{
132 // Using the Central Limit Theorem, the mean of our samples should be
133 // normally distributed. The standard deviation of the mean is smaller than
134 // the standard deviation of the individual samples by a factor of
135 // 1/sqrt(n), where n is the number of samples.
136 const T expectedStdDev = std::sqrt(expectedVariance);
137
138 const T scaledSTD = expectedStdDev / std::sqrt(static_cast<T>(stats.numDataValues()));
139
140 // Because it's normally distributed, we should be within two standard
141 // deviations on either side of the mean 95% of the time. This means that
142 // this test is expect to fail 5% of the time!
143
144 const T lower = expectedMean - 2 * scaledSTD;
145 const T upper = expectedMean + 2 * scaledSTD;
146
147 const T sampleMean = stats.mean();
148 const bool validLower = sampleMean >= lower;
149 const bool validUpper = sampleMean <= upper;
150 return validLower && validUpper;
151}

153template <typename T>
154bool testVariance(const fb_util::RunningStats<T>& stats, T expectedVariance)
155{
156 // Much like in testMean, we use the Central Limit Theorem to expect that
157 // our variance over a large number of samples it normally distributed.
158
159 // If n is very large, then S^2 (the sample variance) approximately has a
160 // normal distribution with mean sigma^2 and variance 2*sigma^4 /(n - 1).
161 // (Technically, it's a chi-squared distribution \Chi(n - 1), but as n
162 // approaches infinity, the chi-squared distribution becomes the normal
163 // distribution N(0, 1). For large numbers n, we can approximate it with
164 // the normal distribution).
165 const T scaledVariance = 2 * expectedVariance*expectedVariance /
166 static_cast<T>(stats.numDataValues() - 1);
167
168 const T scaledStdDev = std::sqrt(scaledVariance);
169
170 // Because it's normally distributed, we should be within two standard
171 // deviations on either side of the mean 95% of the time. This means that
172 // this test is expect to file 5% of the time!
173 const T lower = expectedVariance - 2 * scaledStdDev;
174 const T upper = expectedVariance + 2 * scaledStdDev;
175
176 const T sampleVariance = stats.variance();
177 const bool validLower = sampleVariance >= lower;
178 const bool validUpper = sampleVariance <= upper;
179 return validLower && validUpper;
180}

182template <typename Traits, typename Generator>
183bool run_statistical_tests(Generator function, Traits traits = Traits{})
184{
185 using namespace StatisticalTestSuite;
186 using value_type = typename Traits::value_type;
187 using CDF = typename Traits::CDFType;
188
189 fb_util::RunningStats<double> stats;
190 std::vector<value_type> samples;
191
192 constexpr int fullSamples = 1'000;
193 constexpr int statisticalSamples = 1'000'000;
194
195 // https://www.wolframalpha.com/input/?i=binomial+distribution+CDF+n%3D20%2C+p%3D0.05+at+x%3D5
196 // With 20 tests and five allowed failures, where we expect a failure to happen 5% of the time results in the chance
197 // of any one test failing being 1 - 0.999671 = 0.000329 = 0.0329%
198 constexpr int ntests = 20;
199 constexpr int allowedFailures = 5;
200
201 constexpr int errorIndexKS = 0;
202 constexpr int errorIndexMean = 1;
203 constexpr int errorIndexVar = 2;
204 int failures[3] = {0};
205 for (int test = 0; test < ntests; ++test) {
206 samples.clear();
207 stats.clear();
208
209 for (int i = 0; i < std::max(fullSamples, statisticalSamples); ++i) {
210 const auto x = function();
211 if (i < fullSamples) {
212 samples.push_back(x);
213 }
214 if (i < statisticalSamples) {
215 stats.push(x);
216 }
217 }
218
219 const bool rangeResult = testRange<Traits::template LowerBoundCompare,
220 Traits::template UpperBoundCompare>(traits.min, traits.max,
221 samples.cbegin(), samples.cend());
222 if (!rangeResult) { // This should never fail
223 return false;
224 }
225
226 constexpr bool isFloat = std::is_floating_point<value_type>::value;
227 const double expectedMean = (traits.min + traits.max) * 0.5;
228 const double expectedVariance = (isFloat) ? uniformVarianceContinuous(traits.min, traits.max) :
229 uniformVarianceDiscrete(traits.min, traits.max);
230
231 const bool ksResult = testKolmogorovSmirnov(samples.cbegin(), samples.cend(), CDF(traits.min, traits.max));
232 if (!ksResult) {
233 ++failures[errorIndexKS];
234 if (failures[errorIndexKS] > allowedFailures) {
235 return false;
236 }
237 }
238 const bool meanResult = testMean(stats, expectedMean, expectedVariance);
239 if (!meanResult) {
240 ++failures[errorIndexMean];
241 if (failures[errorIndexMean] > allowedFailures) {
242 return false;
243 }
244 }
245 const bool varResult = testVariance(stats, expectedVariance);
246 if (!varResult) {
247 ++failures[errorIndexVar];
248 if (failures[errorIndexVar] > allowedFailures) {
249 return false;
250 }
251 }
252 }
253 return true;
254}
