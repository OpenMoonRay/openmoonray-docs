33/*
34 * This class computes and keeps track of the number of Ticks / Second
35 */
36class TicksPerSecond {
37public:
38 TicksPerSecond():
39 mStartSeconds(0.0),
40 mStartTicks(0),
41 mTicksPerSecond(0.0),
42 mTicksPerSecondComputed(false)
43 {
44 }
45
46 void init()
47 {
48 getTimeSample(&mStartSeconds, &mStartTicks);
49 mTicksPerSecondComputed = false;
50 }
51
52 double getCached() const
53 {
54 if (!mTicksPerSecondComputed) {
55 mTicksPerSecond = getUncached();
56 mTicksPerSecondComputed = true;
57 }
58 return mTicksPerSecond;
59 }
60
61 // This version returns the latest most accurate count at the cost of
62 // recomputing it. Use TicksPerSecond::getCached() to retrieve the fast
63 // cached version.
64 double getUncached() const
65 {
66 // You must call init before calling this function.
67 MNRY_ASSERT(mStartTicks);
68
69 double endSeconds;
70 uint64 endTicks;
71 getTimeSample(&endSeconds, &endTicks);
72
73 MNRY_ASSERT(endSeconds > mStartSeconds);
74
75 return (endTicks - mStartTicks) / (endSeconds - mStartSeconds);
76 }
77
78private:
79 // TODO: For most accurate readings, don't let compiler or CPU reorder our
80 // sampling points.
81 void getTimeSample(double *seconds, uint64 *ticks) const
82 {
83 *seconds = time::getTime();
84 *ticks = getTicks();
85 }
86
87 double mStartSeconds;
88 uint64 mStartTicks;
89 mutable double mTicksPerSecond;
90 mutable bool mTicksPerSecondComputed;
91};

94/*
95 * cpu clock ticks timer that is scope based.
96 * timing starts in the constructor and stops in the destructor.
97 * It updates a given stat object via its operator+=() method.
98 */
99template <typename T>
100class RAIITicker
101{
102public:
103 __forceinline RAIITicker(T &stat) {
104 mStat = &stat;
105 mStart = getTicks();
106 }
107
108 // timer stops and updates upon destruction
109 __forceinline ~RAIITicker() {
110 (*mStat) += (getTicks() - mStart);
111 }
112
113private:
114 T* mStat;
115 int64 mStart;
116};

119/**
120 * This is similar to the RAIITicker but also takes care to decrement
121 * the parentStat's exclusive tick count.
122 *
123 * You would use this ticker when you have parent-task/child-task relationships
124 * in your timings and would like the parent to not include the time spent
125 * in the child.
126 */
127template <typename T>
128class RAIIInclusiveExclusiveTicker
129{
130public:
131 __forceinline RAIIInclusiveExclusiveTicker(
132 util::InclusiveExclusiveAverage<T> &stat,
133 util::InclusiveExclusiveAverage<T> &parentStat)
134 {
135 mStat = &stat;
136 mParentStat = &parentStat;
137 mStart = getTicks();
138 }
139
140 __forceinline ~RAIIInclusiveExclusiveTicker() {
141 int64 elapsed = (getTicks() - mStart);
142 (*mStat) += elapsed;
143 mParentStat->decrementExclusive(elapsed);
144 }
145
146private:
147 util::InclusiveExclusiveAverage<T>* mStat;
148 util::InclusiveExclusiveAverage<T>* mParentStat;
149 int64 mStart;
150};

155/*
156 * cpu clock ticks manual timer that has a classic start / stop / lap interface
157 * the constructor does NOT start the timer
158 * the destructor does NOT stop the timer
159 * starting or stopping a timer that is already started / stopped (respectively)
160 * will trip a MNRY_ASSERT
161 */
162template <typename T>
163class Ticker
164{
165public:
166 __forceinline Ticker(T &stat) {
167 MNRY_DURING_ASSERTS(mStarted = false);
168 mStat = &stat;
169 mStart = getTicks();
170 }
171 __forceinline ~Ticker() {}
172
173 __forceinline void start() {
174 MNRY_ASSERT(mStarted == false);
175 MNRY_DURING_ASSERTS(mStarted = true);
176 mStart = getTicks();
177 }
178
179 __forceinline void stop() {
180 MNRY_ASSERT(mStarted == true);
181 MNRY_DURING_ASSERTS(mStarted = false);
182 (*mStat) += (getTicks() - mStart);
183 }
184
185 __forceinline void lap() {
186 MNRY_ASSERT(mStarted == true);
187 int64 tick = getTicks();
188 (*mStat) += (tick - mStart);
189 mStart = tick;
190 }
191
192private:
193 T* mStat;
194 int64 mStart;
195 MNRY_DURING_ASSERTS(bool mStarted);
196};
197
198typedef Ticker<int64> TickerInt64;
