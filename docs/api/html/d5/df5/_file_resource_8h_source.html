15/** FileResource is base class for different implementations of
16 * file-like objects.
17 *
18 * indexing: some FileResources may represent an indexed set of files.
19 * This could be done by a substitution symbol (e.g. #) or through
20 * a container/directory object. getIndexed() returns a new FileResource*
21 * for the file at a given index value.
22 */
23namespace moonray {
24namespace file_resource {
25
26class FileResource
27{
28public:
29
30 virtual ~FileResource() {}
31
32 /* return a string label for this resource to
33 be reported to a user (e.g. in an error message)
34 */
35 virtual std::string userLabel() const = 0;
36
37 /* Return true if the resource exists
38 **/
39 virtual bool exists() const = 0;
40
41 /* return a stream to read data from this file.
42 the stream should be deleted by the caller.
43 returns NULL if stream cannot be obtained. */
44 virtual std::istream* openIStream() = 0;
45
46 /* return a stream to write data to this file.
47 the stream should be deleted by the caller.
48 returns NULL if stream cannot be obtained. */
49 virtual std::ostream* openOStream() = 0;
50
51 /* return true if this represents an indexed set of
52 files */
53 virtual bool supportsIndexing() const = 0;
54
55 /* return a new FileResource corresponding to a specific
56 index value. returns null pointer if this doesn't support
57 indexing or index is out of range.

59 */
60 virtual FileResource* getIndexed(float indexVal) const = 0;
61
62 virtual std::string fileName() const = 0;
