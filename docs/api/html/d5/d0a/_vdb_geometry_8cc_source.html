43std::string
44getUniqueNameFromGridName(const std::string& gridName, const std::string& name)
45{
46 const openvdb::Name uniqueName = openvdb::io::GridDescriptor::stringAsUniqueName(gridName);
47 if (uniqueName == gridName) {
48 // there is no suffix
49 return name;
50 }
51
52 // ASCII "record separator" character
53 static constexpr char delimiter = 30;
54 if (openvdb::io::GridDescriptor::stringAsUniqueName(name).find(delimiter) != std::string::npos) {
55 // name already has a suffix
56 return name;
57 }
58 int suffix = std::stoi(uniqueName.substr(uniqueName.find(delimiter) + 1, uniqueName.size()));
59
60 return openvdb::io::GridDescriptor::nameAsString(openvdb::io::GridDescriptor::addSuffix(name, suffix));
61}

63std::string
64getIndexedFile(const std::string& vdbFilePath,
65 const scene_rdl2::rdl2::Geometry& rdlGeometry)
66{
67 std::string resolvedPath;
68 const file_resource::FileResource* resource = file_resource::getFileResource(vdbFilePath);
69 if (resource->supportsIndexing()) {
70 const scene_rdl2::rdl2::SceneContext* sceneContext = rdlGeometry.getSceneClass().getSceneContext();
71 int currentFrame = sceneContext->getSceneVariables().get(scene_rdl2::rdl2::SceneVariables::sFrameKey);
72 const file_resource::FileResource* subResource = resource->getIndexed(currentFrame);
73 resolvedPath = subResource->userLabel();
74 } else {
75 resolvedPath = resource->userLabel();
76 }
77
78 return resolvedPath;
79}

83public:
84 OpenVdbProcedural(const State& state) : ProceduralLeaf(state) {}

86 void generate(const GenerateContext& generateContext,
87 const moonray::shading::XformSamples& parent2render)
88 {
89 const scene_rdl2::rdl2::Geometry* rdlGeometry = generateContext.getRdlGeometry();
90 const VdbGeometry* pVdbGeometry =
91 static_cast<const VdbGeometry*>(rdlGeometry);
92
93 const std::string& vdbFilePath = pVdbGeometry->get(attrModel);
94 if (vdbFilePath.empty()) {
95 rdlGeometry->warn("No resource specified for reading");
96 return;
97 }
98 const scene_rdl2::rdl2::Layer* rdlLayer = generateContext.getRdlLayer();
99 int32_t layerAssignmentId;
100 if (!getAssignmentId(rdlLayer, rdlGeometry, "", layerAssignmentId)) {
101 rdlGeometry->error("unassigned primitive.");
102 return;
103 }
104
105 // If the volume is bound to AmorphousVolumeShader, request VdbVolume
106 // to initialize amorphous samplers
107 moonray::shading::TypedAttributeKey<bool> amorphousKey("amorphous_meta_data");
108 moonray::shading::PrimitiveAttributeTable primitiveAttributeTable;
109 if (generateContext.requestAttribute(amorphousKey)) {
110 primitiveAttributeTable.addAttribute(amorphousKey,
111 moonray::shading::RATE_CONSTANT, std::vector<bool>{true});
112 }
113
114 const std::string resolvedVdbFilePath = getIndexedFile(vdbFilePath, *rdlGeometry);
115 const std::string densityGridName = pVdbGeometry->get(attrDensityGrid);
116 const std::string emissionGridName = pVdbGeometry->get(attrEmissionGrid);
117 moonray::geom::VdbVolume::VdbInitData vdbInitData = {
118 resolvedVdbFilePath,
119 densityGridName,
120 emissionGridName,
121 getUniqueNameFromGridName(densityGridName, pVdbGeometry->get(attrVelocityGrid))};
122
123 std::unique_ptr<VdbVolume> primitive = createVdbVolume(
124 vdbInitData,
125 generateContext.getMotionBlurParams(),
126 LayerAssignmentId(layerAssignmentId),
127 std::move(primitiveAttributeTable));
128 if (primitive) {
129 VdbVolume::Interpolation interpolationMode =
130 moonray::geom::VdbVolume::Interpolation(
131 pVdbGeometry->get(attrInterpolation));
132 primitive->setInterpolation(interpolationMode);
133 primitive->setVelocityScale(
134 pVdbGeometry->get(attrVelocityScale));
135 primitive->setVelocitySampleRate(
136 pVdbGeometry->get(attrVelocitySampleRate));
137 primitive->setEmissionSampleRate(
138 pVdbGeometry->get(attrEmissionSampleRate));
139 primitive->setName(densityGridName);
140 addPrimitive(std::move(primitive),
141 generateContext.getMotionBlurParams(), parent2render);
142 }
143 }
