21AttributeTable::AttributeTable(const AttributeKeySet& requiredKeys,
22 const AttributeKeySet& optionalKeys)
23{
24 int maxKey = -1;
25 for (auto k : requiredKeys) {
26 maxKey = math::max(maxKey, (int)k);
27 }
28 for (auto k : optionalKeys) {
29 maxKey = math::max(maxKey, (int)k);
30 }
31 mNumKeys = maxKey+1;
32 mKeyOffset = new int[mNumKeys];
33 for (int i = 0; i < mNumKeys; i++) {
34 mKeyOffset[i] = -1;
35 }
36 // make sure data in allocated buffer aligned
37 // we sort the key by attribute size, and build key->offset table
38 // with size descending order (to avoid wasted padding space)
39 std::map<size_t, std::vector<AttributeKey> > attrSizeMap;
40 for (auto k : requiredKeys) {
41 attrSizeMap[k.getSize()].push_back(k);
42 }
43 for (auto k : optionalKeys) {
44 attrSizeMap[k.getSize()].push_back(k);
45 }
46 size_t totalSize = 0;
47 for (auto rit = attrSizeMap.rbegin(); rit != attrSizeMap.rend(); ++rit) {
48 size_t attrSize = rit->first;
49 for (auto k: rit->second) {
50 mKeyOffset[k] = totalSize;
51 if (k.hasDerivatives()) {
52 // for an attribute f we need to store f, dfds, dfdt
53 totalSize += 3 * attrSize;
54 mDifferentialAttributes.push_back(k);
55 } else {
56 totalSize += attrSize;
57 }
58 }
59 }
60 totalSize = util::alignUp(totalSize, sizeof(float));
61
62 mTotalSize = totalSize;
63 mRequiredAttributes.insert(mRequiredAttributes.begin(),
64 requiredKeys.begin(), requiredKeys.end());
65 mOptionalAttributes.insert(mOptionalAttributes.begin(),
66 optionalKeys.begin(), optionalKeys.end());
67}
