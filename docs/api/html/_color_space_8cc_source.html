17/**
18 * @brief Private helper function for rgbToHsv() and rgbToHsl().
19 * Converts an RGB color value to its corresponding hue.
20 *
21 * r, g, and b are assumed to be [0, 1].
22 * hue is returned between [0, 1].
23 * Hue is determined by warping the hexagonal projection of the
24 * RGB cube into a circle. Hue is the resulting angle of the projected
25 * RGB vector around the origin of the HSV circle.
26 *
27 * @param[in] c RGB color
28 * @param[in] chroma RGB maxChannel - minChannel
29 * @param[in] maxChannelIndex Index of max channel (0,1,2) for (r,g,b) respectively
30 * @return HSL / HSV hue
31 */
32float
33rgbToHue(const Color &c, float chroma, int maxChannelIndex)
34{
35 const float r = c.r; const float g = c.g; const float b = c.b;
36 float h;
37 if (isZero(chroma)) {
38 h = 0.0f;
39 } else {
40 switch (maxChannelIndex) {
41 case 0:
42 // Max channel is red.
43 h = ((g - b) / chroma);
44 break;
45 case 1:
46 // Max channel is green.
47 h = 2.0f + (b - r) / chroma;
48 break;
49 default:
50 // Max channel is blue.
51 h = 4.0f + (r - g) / chroma;
52 break;
53 }
54
55 h *= 60.0f;
56 // Wrap around h values outside [0, 360].
57 h = fmod(h, 360.0f);
58 if (h < 0) {
59 h += 360.0f;
60 }
61
62 // Map h back to [0,1].
63 h /= 360.0f;
64 }
65
66 return h;
67}

69/**
70 * @brief Private helper function for rgbToHsv() and rgbToHsl().
71 * Determines the maximum r, g, b channel and returns an
72 * index to signify which channel was the maximum.
73 *
74 * @param[in] c RGB color
75 * @return RGB maximum value index
76 */
77int
78maxRgbChannel(const Color &c)
79{
80 float maxChannel = c.r;
81 int maxChannelIndex = 0;
82 if (c.g > maxChannel) {
83 maxChannel = c.g;
84 maxChannelIndex = 1;
85 }
86 if (c.b > maxChannel) {
87 maxChannelIndex = 2;
88 }
89 return maxChannelIndex;
90}

95float
96rgbToHue(const Color& c) {
97 const float maxChannel = max(c.r, max(c.g, c.b));
98 const float minChannel = min(c.r, min(c.g, c.b));
99 const float chroma = maxChannel - minChannel;
100 int maxChannelIndex;
101 float h;
102 if (isZero(chroma)) {
103 h = 0.0f;
104 } else {
105 if (isEqual(maxChannel, c.r)) {
106 maxChannelIndex = 0;
107 } else if (isEqual(maxChannel, c.g)) {
108 maxChannelIndex = 1;
109 } else {
110 maxChannelIndex = 2;
111 }
112 h = rgbToHue(c, chroma, maxChannelIndex);
113 }
114
115 return h;
116}

118Color
119rgbToHsv(const Color &rgb)
120{
121 float h, s, v;
122 // v
123 // Value is the maximum of r, g, b channels.
124 // Record the index of the max channel to use in rgbToHue().
125 const int maxChannelIndex = maxRgbChannel(rgb);
126 const float maxChannel = rgb[maxChannelIndex];
127 const float minChannel = min(rgb.r, min(rgb.g, rgb.b));
128 v = maxChannel;
129
130 const float chroma = maxChannel - minChannel;
131
132 // s and h
133 // Saturation is the chroma divided by the maximum chroma.
134 if (!isZero(maxChannel)) {
135 s = chroma / maxChannel;
136 h = rgbToHue(rgb, chroma, maxChannelIndex);
137 } else {
138 s = 0.0f;
139 h = 0.0f;
140 }
141 return Color(h, s, v);
142}

144Color
145rgbToHsl(const Color &rgb)
146{
147 float h, s, l;
148 // l
149 // Lightness is the average of largest and smallest channel.
150 // Record the index of the max channel to use in rgbToHue().
151 const int maxChannelIndex = maxRgbChannel(rgb);
152 const float maxChannel = rgb[maxChannelIndex];
153 const float minChannel = min(rgb.r, min(rgb.g, rgb.b));
154 l = (maxChannel + minChannel) * 0.5f;
155
156 const float chroma = maxChannel - minChannel;
157
158 // s and h
159 if (isZero(chroma)) {
160 h = 0.0f;
161 s = 0.0f;
162 } else {
163 if (l > 0.5) {
164 // If divide by 0.
165 if (isEqual(maxChannel + minChannel, 2.0f)) {
166 // Only occurs if one or more r,g,b > 0.
167 // Set s to theoretical maximum value for s for this divide by 0 case.
168 // (Even though s can be > 0 as max + min approaches 2.0)
169 s = 1.0f;
170 } else {
171 // S = C / (2 - 2L)
172 // Use abs() to prevent s < 0, in case (maxChannel + minChannel > 2)
173 // which only occurs if one or more r,g,b > 0.
174 s = abs(chroma / (2.0f - maxChannel - minChannel));
175 }
176 } else {
177 // S = C / 2L
178 if (maxChannel + minChannel <= 0) {
179 // Only occurs if one or more r,g,b < 0.
180 // Prevents divide by zero if max + min == 0.
181 s = chroma;
182 } else {
183 // Typical scenario, r,g,b > 0.
184 s = chroma / (maxChannel + minChannel);
185 }
186 }
187
188 h = rgbToHue(rgb, chroma, maxChannelIndex);
189 }
190 return Color(h, s, l);
191}

193Color
194hsvToRgb(const Color &hsv)
195{
196 Color c;
197 float h = hsv[0];
198 const float s = hsv[1]; const float v = hsv[2];
199 // Values of s outside [0,1] are meaningless.
200 // But are not clamped in order to maintain reciprocity
201 // with rgbToHsv().
202
203 // HSV color space is hexagonal and divided into 6 sectors.
204 int sector; // 0 - 5
205
206 // No saturation means color is gray.
207 if (isZero(s)) {
208 c.r = c.g = c.b = v;
209 } else {
210 // Wrap around values of h outside [0,1].
211 h = fmod(h, 1.0f);
212 if (h < 0) {
213 h += 1.0f;
214 }
215
216 // Map hue [0,1] -> [0,360).
217 float hue = h * 360.0f;
218 // hue 360 == hue 0.
219 if (isEqual(hue, 360.0f)) {
220 hue = 0.0f;
221 }
222 // Determine which of the 6 sectors hue lies in the HSV circle.
223 hue /= 60.0f;
224 sector = static_cast<int>(hue);
225
226 // Each sector consists of two constant values for RGB: v or p.
227 // And the other component of RGB varies linearly +/- w.r.t. hue: q or t.
228 // H = Hue, S = Saturation, V = Value.
229 // C = V * S
230 // H = H / 60
231 // X = C * (1 - |H mod 2 - 1|)
232 // m = V - C
233
234 // f = H - (int)H
235 // if H mod 2 - 1 < 0
236 // 0 <= H < 1 || 2 <= H < 3 || 4 <= H < 5
237 // H - (int)H == 1 - |H mod 2 - 1|
238 // therefore, X = f * C
239 // else
240 // 1 <= H < 2 || 3 <= H < 4 || 5 <= H < 6
241 // H - (int)H == |H mod 2 - 1|
242 // therefore, X = (1 - f) * C
243 float f = hue - static_cast<float>(sector);
244 // v = m + C = v
245 // p = m
246 float p = v * (1.0f - s); // Constant for given hue.
247 // X = (1 - f) * C
248 // q = m + X
249 float q = v * (1.0f - s * f); // Varies negative linearly w.r.t. hue.
250 // X = f * C
251 // t = m + X
252 float t = v * (1.0f - s * (1.0f - f)); // Varies positive linearly w.r.t. hue.
253 switch (sector) {
254 case 0: c.r = v; c.g = t; c.b = p; break; // 0 <= hue < 1
255 case 1: c.r = q; c.g = v; c.b = p; break; // 1 <= hue < 2
256 case 2: c.r = p; c.g = v; c.b = t; break; // 2 <= hue < 3
257 case 3: c.r = p; c.g = q; c.b = v; break; // 3 <= hue < 4
258 case 4: c.r = t; c.g = p; c.b = v; break; // 4 <= hue < 5
259 case 5: c.r = v; c.g = p; c.b = q; break; // 5 <= hue < 6
260 default: c.r = p; c.g = p; c.b = p; break; // hue undefined
261 }
262 }
263 return c;
264}

266Color
267hslToRgb(const Color &hsl)
268{
269 Color c;
270 float h = hsl[0];
271 const float s = hsl[1]; const float l = hsl[2];
272 // Values of s outside [0,1] are meaningless.
273 // But are not clamped in order to maintain reciprocity
274 // with rgbToHsl().
275
276 // HSL color space is hexagonal and divided into 6 sectors.
277 int sector; // 0 - 5
278
279 // No saturation means color is gray.
280 if (isZero(s)) {
281 c.r = c.g = c.b = l;
282 } else {
283 // Wrap around values of h outside [0,1].
284 h = fmod(h, 1.0f);
285 if (h < 0) {
286 h += 1.0f;
287 }
288
289 // Map hue [0,1] -> [0,360).
290 float hue = h * 360.0f;
291 // hue 360 == hue 0.
292 if (isEqual(hue, 360.0f)) {
293 hue = 0.0f;
294 }
295 // Determine which of the 6 sectors hue lies in the HSL circle.
296 hue /= 60.0f;
297 sector = static_cast<int>(hue);
298
299 // Each sector consists of two constant values for RGB: w or p.
300 // And the other component of RGB varies linearly +/- w.r.t. hue: q or t.
301 // H = Hue, S = Saturation, L = Lightness.
302 // C = (1 - |2L - 1|) * S
303 // H = H / 60
304 // X = C * (1 - |H mod 2 - 1|)
305 // m = L - 0.5 * C
306
307 // f = H - (int)H
308 // if H mod 2 - 1 < 0
309 // 0 <= H < 1 || 2 <= H < 3 || 4 <= H < 5
310 // H - (int)H == 1 - |H mod 2 - 1|
311 // therefore, X = f * C
312 // else
313 // 1 <= H < 2 || 3 <= H < 4 || 5 <= H < 6
314 // H - (int)H == |H mod 2 - 1|
315 // therefore, X = (1 - f) * C
316 float f = hue - static_cast<float>(sector);
317 float p, w, q, t;
318 if (l < 0.5) {
319 // C = 2 * S * L
320 // p = m
321 p = l * (1.0f - s);
322 // w = m + C
323 w = l * (1.0f + s);
324 // X = (1 - f) * C
325 // q = m + X = (0.5 - f) * C + L
326 q = l * (s * (1.0f - 2.0f * f) + 1.0f);
327 // X = f * C
328 // t = m + X = (f - 0.5) * C + L
329 t = l * (s * (2.0f * f - 1.0f) + 1.0f);
330 } else {
331 // C = 2 * S * (1 - L)
332 // p = m
333 p = l * (1.0f + s) - s;
334 // w = m + C
335 w = l * (1.0f - s) + s;
336 // X = (1 - f) * C
337 // q = m + X = (0.5 - f) * C + L
338 q = l * (1.0f + s * (2.0f * f - 1.0f)) + s * (1.0f - 2.0f * f);
339 // X = f * C
340 // t = m + X = (f - 0.5) * C + L
341 t = l * (1.0f + s * (1.0f - 2.0f * f)) + s * (2.0f * f - 1.0f);
342 }
343 switch (sector) {
344 case 0: c.r = w; c.g = t; c.b = p; break; // 0 <= hue < 1
345 case 1: c.r = q; c.g = w; c.b = p; break; // 1 <= hue < 2
346 case 2: c.r = p; c.g = w; c.b = t; break; // 2 <= hue < 3
347 case 3: c.r = p; c.g = q; c.b = w; break; // 3 <= hue < 4
348 case 4: c.r = t; c.g = p; c.b = w; break; // 4 <= hue < 5
349 case 5: c.r = w; c.g = p; c.b = q; break; // 5 <= hue < 6
350 default: c.r = p; c.g = p; c.b = p; break; // hue undefined
351 }
352 }
353 return c;
354}
