16/**
17 * The MinUniqueSuffixMap roughly behaves like a std::map. It consumes
18 * canonicalized, absolute source paths (through insert()), and computes
19 * the minimum unique suffix, which is returned by looking up the original
20 * source path with at().
21 *
22 * The begin() and end() functions return a wrapper around an iterator which
23 * can be advanced with preincrement operator++ and dereferenced with *. When
24 * dereferenced, this iterator wrapper will produce a std::pair of
25 * std::strings, where the first element is the source path and the second
26 * element is the minimum unique suffix destination path.
27 *
28 * The minimum unique suffix is fancy way of saying "what's the shortest
29 * trailing amount of this source path I can get away with and still have all
30 * my paths be unique". We want to shorten the path as much as possible for
31 * our destination file names, but we don't want any collisions for files
32 * that may have the same base name but live in different directories.
33 *
34 * Example:
35 * MinUniqueSuffixMap copyPaths;
36 *
37 * auto toothless = copyPaths.insert("/seq/shot/toothless/skin.mm");
38 * auto hiccup = copyPaths.insert("/seq/shot/hiccup/skin.mm");
39 * auto cove = copyPaths.insert("/seq/shot/envir/cove.mm");
40 *
41 * auto toothyPaths = *toothless;
42 * std::cout << toothyPaths.first; // prints "/seq/shot/toothess/skin.mm"
43 * std::cout << toothyPaths.second; // prints "toothless_skin.mm"
44 *
45 * auto hiccupPaths = *hiccup;
46 * std::cout << hiccupPaths.first; // prints "/seq/shot/hiccup/skin.mm"
47 * std::cout << hiccupPaths.second; // prints "hiccup_skin.mm"
48 *
49 * auto covePaths = *cove;
50 * std::cout << covePaths.first; // prints "/seq/shot/envir/cove.mm"
51 * std::cout << covePaths.second; // prints "cove.mm"
52 */

113 /**
114 * Wraps an internal iterator for traversing all the valid paths stored in
115 * the map. This wrapper can be advanced with preincrement operator++,
116 * compared for equality and inequality with other iterator wrappers, and
117 * be dereferenced with operator*.
118 *
119 * Dereferencing the iterator wrapper returns a std::pair or std::strings,
120 * where the first element is the source path and the second element is the
121 * destination path. Note that operator-> is not defined (due to some
122 * complexities in how it's handled), so this will not work:
123 *
124 * iter->first // compile error
125 *
126 * but this will:
127 *
128 * (*iter).first // ok
129 *
130 * Better yet, just copy the pair out entirely and reference it from there,
131 * because it is built on the fly by traversing the internal data
132 * structures, so it may be expensive to compute over and over.
133 *
134 * auto paths = *iter;
135 * // use paths.first, paths.second...
136 */
137 class IteratorWrapper
138 {
139 public:
140 bool operator==(const IteratorWrapper& b) const { return mIter == b.mIter; }
141 bool operator!=(const IteratorWrapper& b) const { return mIter != b.mIter; }
142 void operator++() { ++mIter; }
