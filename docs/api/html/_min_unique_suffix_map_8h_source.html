74 // Takes the old destination path, pops off the last path component of
75 // its path prefix, and prepends that component to the new destination
76 // path with an underscore. Ex:
77 // Dest Path -> Path Prefix
78 // ------------------------
79 // file.mm -> /some/path/to
80 //
81 // 1) destPath = rotatePathComponent(destPath);
82 //
83 // to_file.mm -> /some/path
84 //
85 // 2) destPath = rotatePathComponent(destPath);
86 //
87 // path_to_file.mm -> /some
88 std::string rotatePathComponent(const std::string& oldDestPath);
89
90 // Rotates the path component, but also steals the guts of this
91 // PathData object and returns a new one, marking this one dead. The
92 // return value is ready to be inserted into the dest -> source map.
93 std::pair<std::string, PathData> relocate(const std::string& oldDestPath);
94
95 // The path prefix of the given destination file. At the start, that's
96 // that's the dirname(), though it may get shortened if the path is
97 // rotated.
98 std::string mPathPrefix;
99
100 // The original source path for this destination file.
101 std::string mSourcePath;
102
103 // Marks an element in the map as dead. Dead elements do not appear in
104 // the output, but still exist in dest -> source map so that they
105 // continue to collide with new destination paths as they are inserted.
106 bool mDead;
