15class RenderPrepStats
16//
17// This class is used to keep renderPrep progress information inside both of progmcrt
18// computation and client process.
19// This condition is updated during renderPrep and updated information is sent to the
20// client via merge computation by InfoCodec.
21// On the client side, all updated renderPrep progress information is stored into this
22// object.
23// We still don't have BVH construction sub-stage progress logic (It's a bit difficult
24// straight forward way due to related to embree) and this would be a future task.
25//
26{
27public:
28 static constexpr int shiftBit = 4;
29 static constexpr unsigned cancelBit = 0x1;
30
31 enum class Stage : unsigned int {
32 NOT_ACTIVE = 0x0, // renderPrep is not active
33
34 RENDER_PREP_START = (0x100 << shiftBit), // renderPrep start
35 RENDER_PREP_START_CANCELED = (0x101 << shiftBit) | cancelBit, // renderPrep start w/ canceled
36
37 //
38 // SceneContext apply update
39 //
40 RENDER_PREP_APPLYUPDATE = (0x200 << shiftBit), // start
41 RENDER_PREP_APPLYUPDATE_CANCELED = (0x201 << shiftBit) | cancelBit, // start canceled
42 RENDER_PREP_APPLYUPDATE_DONE = (0x202 << shiftBit), // done
43 RENDER_PREP_APPLYUPDATE_DONE_CANCELED = (0x203 << shiftBit) | cancelBit, // done canceled
44
45 //
46 // renderPrep loadGeom stage.
47 //
48 // First one (loadGeom 0) is for a regular layer and
49 RENDER_PREP_LOAD_GEOM0 = (0x300 << shiftBit), // loadGeom 0 start
50 RENDER_PREP_LOAD_GEOM0_CANCELED = (0x301 << shiftBit) | cancelBit, // loadGeom 0 start canceled
51 GM_LOADGEO0_START = (0x302 << shiftBit), // Geom 0 start
52 GM_LOADGEO0_START_CANCELED = (0x303 << shiftBit) | cancelBit, // Geom 0 start canceled
53 GM_LOADGEO0_PROCESS = (0x304 << shiftBit), // Geom 0 processing
54 GM_LOADGEO0_DONE = (0x305 << shiftBit), // Geom 0 done
55 GM_LOADGEO0_DONE_CANCELED = (0x306 << shiftBit) | cancelBit, // Geom 0 done canceled
56 RENDER_PREP_LOAD_GEOM0_DONE = (0x307 << shiftBit), // loadGeom 0 done
57 RENDER_PREP_LOAD_GEOM0_DONE_CANCELED = (0x308 << shiftBit) | cancelBit, // loadGeom 0 done canceled
58
59 // Second one (loadGeom 1) is for meshLightLayer.
60 RENDER_PREP_LOAD_GEOM1 = (0x400 << shiftBit), // loadGeom 1 start
61 RENDER_PREP_LOAD_GEOM1_CANCELED = (0x401 << shiftBit) | cancelBit, // loadGeom 1 start canceled
62 GM_LOADGEO1_START = (0x402 << shiftBit), // Geom 1 start
63 GM_LOADGEO1_START_CANCELED = (0x403 << shiftBit) | cancelBit, // Geom 1 start canceled
64 GM_LOADGEO1_PROCESS = (0x404 << shiftBit), // Geom 1 processing
65 GM_LOADGEO1_DONE = (0x405 << shiftBit), // Geom 1 done
66 GM_LOADGEO1_DONE_CANCELED = (0x406 << shiftBit) | cancelBit, // Geom 1 done canceled
67 RENDER_PREP_LOAD_GEOM1_DONE = (0x407 << shiftBit), // loadGeom 1 done
68 RENDER_PREP_LOAD_GEOM1_DONE_CANCELED = (0x408 << shiftBit) | cancelBit, // loadGeom 1 done canceled
69
70 //
71 // renderPrep tessellation/BVH-construction operation.
72 //
73 // First one (tessellation 0 and BVH construction 0) is for a regular layer
74 GM_FINALIZE0_START = (0x500 << shiftBit), // stage-0 start
75 GM_FINALIZE0_START_CANCELED = (0x501 << shiftBit) | cancelBit, // stage-0 start canceled
76 GM_FINALIZE0_TESSELLATION = (0x502 << shiftBit), // tess 0 start
77 GM_FINALIZE0_TESSELLATION_CANCELED = (0x503 << shiftBit) | cancelBit, // tess 0 start canceled
78 GM_FINALIZE0_TESSELLATION_PROCESS = (0x504 << shiftBit), // tess 0 processing
79 GM_FINALIZE0_TESSELLATION_DONE = (0x505 << shiftBit), // tess 0 done
80 GM_FINALIZE0_TESSELLATION_DONE_CANCELED = (0x506 << shiftBit) | cancelBit, // tess 0 done canceled
81 GM_FINALIZE0_BVH = (0x600 << shiftBit), // BVH 0 start
82 GM_FINALIZE0_BVH_CANCELED = (0x601 << shiftBit) | cancelBit, // BVH 0 start canceled
83 GM_FINALIZE0_BVH_DONE = (0x602 << shiftBit), // BVH 0 done
84 GM_FINALIZE0_BVH_DONE_CANCELED = (0x603 << shiftBit) | cancelBit, // BVH 0 done canceled
85 GM_FINALIZE0_DONE = (0x604 << shiftBit), // stage-0 done
86 GM_FINALIZE0_DONE_CANCELED = (0x605 << shiftBit) | cancelBit, // stage-0 done canceled
87
88 // Second one (tessellation 1 and BVH construction 1) is for meshLightLayer.
89 GM_FINALIZE1_START = (0x700 << shiftBit), // stage-1 start
90 GM_FINALIZE1_START_CANCELED = (0x701 << shiftBit) | cancelBit, // stage-1 start canceled
91 GM_FINALIZE1_TESSELLATION = (0x702 << shiftBit), // tess 1 start
92 GM_FINALIZE1_TESSELLATION_CANCELED = (0x703 << shiftBit) | cancelBit, // tess 1 start canceled
93 GM_FINALIZE1_TESSELLATION_PROCESS = (0x704 << shiftBit), // tess 1 processing
94 GM_FINALIZE1_TESSELLATION_DONE = (0x705 << shiftBit), // tess 1 done
95 GM_FINALIZE1_TESSELLATION_DONE_CANCELED = (0x706 << shiftBit) | cancelBit, // tess 1 done canceled
96 GM_FINALIZE1_BVH = (0x800 << shiftBit), // BVH 1 start
97 GM_FINALIZE1_BVH_CANCELED = (0x801 << shiftBit) | cancelBit, // BVH 1 start canceled
98 GM_FINALIZE1_BVH_DONE = (0x802 << shiftBit), // BVH 1 done
99 GM_FINALIZE1_BVH_DONE_CANCELED = (0x803 << shiftBit) | cancelBit, // BVH 1 done w/ canceled
100 GM_FINALIZE1_DONE = (0x804 << shiftBit), // stage-1 done
101 GM_FINALIZE1_DONE_CANCELED = (0x805 << shiftBit) | cancelBit, // stage-1 done canceled
102
103 RENDER_PREP_DONE = (0x900 << shiftBit), // renderPrep done
104 RENDER_PREP_DONE_CANCELED = (0x901 << shiftBit) | cancelBit // renderPrep done w/ canceled
105 };
106
107 RenderPrepStats() :
108 mStage(Stage::NOT_ACTIVE),
109 mLoadGeometriesTotal{0, 0},
110 mLoadGeometriesProcessed{0, 0},
111 mTessellationTotal{0, 0},
112 mTessellationProcessed{0, 0}
113 {}
114 RenderPrepStats(const Stage &stage) :
115 mStage(stage),
116 mLoadGeometriesTotal{0, 0},
117 mLoadGeometriesProcessed{0, 0},
118 mTessellationTotal{0, 0},
119 mTessellationProcessed{0, 0}
120 {}
121
122 void reset();
123
124 Stage &stage() { return mStage; }
125 const Stage &stage() const { return mStage; }
126
127 int &loadGeometriesTotal(int stageId) { return mLoadGeometriesTotal[stageId]; }
128 const int &loadGeometriesTotal(int stageId) const { return mLoadGeometriesTotal[stageId]; }
129
130 int &loadGeometriesProcessed(int stageId) { return mLoadGeometriesProcessed[stageId]; }
131 const int &loadGeometriesProcessed(int stageId) const { return mLoadGeometriesProcessed[stageId]; }
132
133 int &tessellationTotal(int stageId) { return mTessellationTotal[stageId]; }
134 const int &tessellationTotal(int stageId) const { return mTessellationTotal[stageId]; }
135
136 int &tessellationProcessed(int stageId) { return mTessellationProcessed[stageId]; }
137 const int &tessellationProcessed(int stageId) const { return mTessellationProcessed[stageId]; }
138
139 bool isCompleted() const { return (mStage == Stage::RENDER_PREP_DONE); }
140 bool isCanceled() const { return (static_cast<int>(mStage) & cancelBit) == cancelBit; }
141
142 // for renderPrep progress fraction
143 unsigned getTotalSteps() const;
144 unsigned getCurrSteps() const;
145
146 std::string show() const;
147
148 static std::string stageStr(const Stage &stage);
149
150private:
151 bool isStageFinished(const Stage &starge) const;
152
153 //------------------------------
154
155 Stage mStage;
156
157 static constexpr int mStageMax = 2;
158
159 int mLoadGeometriesTotal[mStageMax];
160 int mLoadGeometriesProcessed[mStageMax];
161
162 int mTessellationTotal[mStageMax];
163 int mTessellationProcessed[mStageMax];
164
165 static constexpr int mStageStepsPoints = 5; // for renderPrep progress estimation
166};
