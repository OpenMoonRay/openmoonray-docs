32LatencyItem::show(const std::string &hd, const uint64_t timeBase, const uint32_t prevTime,

51std::string
52LatencyItem::timeStr(const uint64_t &time)
53{
54 time_t tsec = static_cast<time_t>(time / 1000 / 1000);
55 uint64_t usec = static_cast<uint64_t>(time) - static_cast<uint64_t>(tsec) * 1000 * 1000;
56 const struct tm *currTm = localtime(&tsec);
57
58 static const char *wday[] = {"Sun", "Mon", "Tue", "Wed", "Thu", "Fri", "Sat"};
59
60 std::ostringstream ostr;
61 ostr << std::setw(4) << currTm->tm_year + 1900 << '-'
62 << std::setw(2) << std::setfill('0') << currTm->tm_mon + 1 << '-'
63 << std::setw(2) << std::setfill('0') << currTm->tm_mday
64 << std::setw(3) << wday[currTm->tm_wday]
65 << std::setw(2) << std::setfill('0') << currTm->tm_hour << ':'
66 << std::setw(2) << std::setfill('0') << currTm->tm_min << ':'
67 << std::setw(2) << std::setfill('0') << currTm->tm_sec << '.'
68 << std::setw(6) << std::setfill('0') << usec;
69 return ostr.str();
70}

163std::string
164LatencyLog::show(const std::string &hd) const
165{
166 size_t numDigitId = calcNumDigit(mLog.size());
167 size_t numDigitAllTime = 6;
168 {
169 uint64_t allTime = mLog.back().time();
170 float msec = (float)allTime / 1000.0f;
171 numDigitAllTime = calcNumDigit(static_cast<size_t>(msec)) + 3; // plus 3 for .??
172 }
173 size_t numDigitDeltaTime = 5;
174 {
175 size_t maxDeltaTime = 0;
176 for (size_t logId = 1; logId < mLog.size(); ++logId) {
177 uint32_t deltaTime = mLog[logId].time() - mLog[logId - 1].time();
178 if (logId == 1) {
179 maxDeltaTime = deltaTime;
180 } else {
181 if (maxDeltaTime < deltaTime) maxDeltaTime = deltaTime;
182 }
183 }
184 float msec = static_cast<float>(maxDeltaTime) / 1000.0f;
185 numDigitDeltaTime = calcNumDigit(static_cast<size_t>(msec)) + 3; // plus 3 for .??
186 }
187
188 std::ostringstream ostr;
189 ostr << hd << "LatencyLog {\n"
190 << hd << " mName:" << mName << '\n'
191 << hd << " mMachineId:" << mMachineId << '\n'
192 << hd << " mSnapshotId:" << mSnapshotId << '\n'
193 << hd << " mDataSize:" << mDataSize << '\n'
194 << hd << " mTimeBase:" << LatencyItem::timeStr(mTimeBase) << '\n'
195 << hd << " log total:" << mLog.size() << " {\n";
196 for (size_t logId = 0; logId < mLog.size(); ++logId) {
197 uint32_t prevTime = 0;
198 if (logId > 0) {
199 prevTime = mLog[logId - 1].time();
200 }
201 ostr << mLog[logId].show(hd + " " + idStr(logId, numDigitId) + ':', mTimeBase, prevTime,
202 numDigitAllTime, numDigitDeltaTime) + '\n';
203 }
204 ostr << hd << " }\n";
205 ostr << hd << "}";
206 return ostr.str();
207}

225void
226LatencyLogUpstream::decode(VContainerDeq &vContainerDeq)
227{
228 reset();
229
230 while (1) {
231 int machineId;
232 vContainerDeq.deqVLInt(machineId);
233 if (machineId == -1) break; // end marker check
234
235 bool sw;
236 vContainerDeq.deqBool(sw);
237 if (sw) {
238 size_t total;
239 vContainerDeq.deqVLSizeT(total);
240
241 mMachine.emplace_back(total);
242 std::vector<LatencyLog> &currLatencyLog = mMachine.back();
243
244 for (size_t i = 0; i < total; ++i) {
245 size_t dataSize;
246 vContainerDeq.deqVLSizeT(dataSize);
247 const void *data = vContainerDeq.skipByteData(dataSize);
248
249 currLatencyLog[i].decode(data, dataSize); // decode latencyLog
250 }
251 }
252 }
253}

255void
256LatencyLogUpstream::decode(const void *data, const size_t dataSize)
257{
258 VContainerDeq vContainerDeq(data, dataSize);
259 this->decode(vContainerDeq);
260}

262std::string
263LatencyLogUpstream::show(const std::string &hd) const
264{
265 std::ostringstream ostr;
266 ostr << hd << "LatencyLogUpstream {\n";
267 ostr << hd << " machineTotal:" << mMachine.size() << '\n';
268 for (size_t machineId = 0; machineId < mMachine.size(); ++machineId) {
269 ostr << hd << " mId:" << machineId << " logTotal:" << mMachine[machineId].size() << " {\n";
270 for (size_t msgId = 0; msgId < mMachine[machineId].size(); ++msgId) {
271 const LatencyLog &currLatencyLog = mMachine[machineId][msgId];
272 ostr << currLatencyLog.show(hd + " ") << '\n';
273 }
274 ostr << hd << " }\n";
275 }
276 ostr << hd << '}';
277 return ostr.str();
