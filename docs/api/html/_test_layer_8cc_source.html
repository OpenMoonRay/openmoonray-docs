40void
41TestLayer::testSerialize()
42{
43 // Create some geometries and materials.
44 Geometry* teapot1 = mContext->createSceneObject("FakeTeapot", "/seq/shot/teapot1")->asA<Geometry>();
45 Material* material1 = mContext->createSceneObject("FakeMaterial", "/seq/shot/material1")->asA<Material>();
46 Material* material2 = mContext->createSceneObject("FakeMaterial", "/seq/shot/material2")->asA<Material>();
47 Displacement* displacement1 = mContext->createSceneObject("FakeDisplacement", "/seq/shot/displacement1")->asA<Displacement>();
48 VolumeShader* volumeShader1 = mContext->createSceneObject("FakeVolumeShader", "/seq/shot/volumeShader1")->asA<VolumeShader>();
49
50 // Create some lights.
51 Light* key = mContext->createSceneObject("FakeLight", "/seq/shot/key")->asA<Light>();
52 Light* fill = mContext->createSceneObject("FakeLight", "/seq/shot/fill")->asA<Light>();
53
54 // Create a light set.
55 LightSet* lights1 = mContext->createSceneObject("LightSet", "/seq/shot/keyfill")->asA<LightSet>();
56 lights1->beginUpdate();
57 lights1->add(key);
58 lights1->add(fill);
59 lights1->endUpdate();
60
61 // Create a layer.
62 Layer* layer = mContext->createSceneObject("Layer", "/seq/shot/layer")->asA<Layer>();
63
64 // Make some assignments in the layer, verify they get new IDs.
65 layer->beginUpdate();
66 CPPUNIT_ASSERT(layer->assign(teapot1, "lid", material1, lights1, displacement1, volumeShader1) == 0);
67 layer->endUpdate();
68
69 // Serialize
70 BinaryWriter writer(*mContext);
71 std::stringstream ss;
72 writer.toStream(ss);
73
74 // Create a copy of the context
75 SceneContext context;
76 BinaryReader reader(context);
77 reader.fromStream(ss);
78 context.commitAllChanges();
79
80 // Perform an update of the layer
81 mContext->commitAllChanges();
82 layer->beginUpdate();
83 CPPUNIT_ASSERT(layer->assign(teapot1, "lid", material2, lights1, displacement1, volumeShader1) == 0);
84 layer->endUpdate();
85
86 // Serialize with delta encoding on
87 BinaryWriter writer2(*mContext);
88 writer2.setDeltaEncoding(true);
89 std::stringstream ss2;
90 writer2.toStream(ss2);
91
92 // Deserialize to the copy
93 BinaryReader reader2(context);
94 reader2.fromStream(ss2);
95
96 // Get the layer from the copy
97 layer = context.getSceneObject("/seq/shot/layer")->asA<Layer>();
98 teapot1 = context.getSceneObject("/seq/shot/teapot1")->asA<Geometry>();
99 material2 = context.getSceneObject("/seq/shot/material2")->asA<Material>();
100
101 // Read the data and check if the change was applied
102 Layer::GeometryToRootShadersMap gts;
103 layer->getAllGeometryToRootShaders(gts);
104 CPPUNIT_ASSERT(gts.size() == 1);
105 CPPUNIT_ASSERT(gts.count(teapot1) == 1);
106 CPPUNIT_ASSERT(gts[teapot1].count(material2) == 1);
107}

109void
110TestLayer::testAssignAndLookup()
111{
112 // Create some geometries and materials.
113 Geometry* teapot1 = mContext->createSceneObject("FakeTeapot", "/seq/shot/teapot1")->asA<Geometry>();
114 Geometry* teapot2 = mContext->createSceneObject("FakeTeapot", "/seq/shot/teapot2")->asA<Geometry>();
115 Material* material1 = mContext->createSceneObject("FakeMaterial", "/seq/shot/material1")->asA<Material>();
116 Material* material2 = mContext->createSceneObject("FakeMaterial", "/seq/shot/material2")->asA<Material>();
117 Displacement* displacement1 = mContext->createSceneObject("FakeDisplacement", "/seq/shot/displacement1")->asA<Displacement>();
118 Displacement* displacement2 = mContext->createSceneObject("FakeDisplacement", "/seq/shot/displacement2")->asA<Displacement>();
119
120 // Create some lights.
121 Light* key = mContext->createSceneObject("FakeLight", "/seq/shot/key")->asA<Light>();
122 Light* fill = mContext->createSceneObject("FakeLight", "/seq/shot/fill")->asA<Light>();
123 Light* rim = mContext->createSceneObject("FakeLight", "/seq/shot/rim")->asA<Light>();
124
125 // Create some light sets.
126 LightSet* lights1 = mContext->createSceneObject("LightSet", "/seq/shot/keyfill")->asA<LightSet>();
127 lights1->beginUpdate();
128 lights1->add(key);
129 lights1->add(fill);
130 lights1->endUpdate();
131 LightSet* lights2 = mContext->createSceneObject("LightSet", "/seq/shot/fillrim")->asA<LightSet>();
132 lights2->beginUpdate();
133 lights2->add(fill);
134 lights2->add(rim);
135 lights2->endUpdate();
136
137 // Create a layer.
138 Layer* layer = mContext->createSceneObject("Layer", "/seq/shot/layer")->asA<Layer>();
139
140 // Make some assignments in the layer, verify they get new IDs.
141 layer->beginUpdate();
142 CPPUNIT_ASSERT(layer->assign(teapot1, "lid", material1, lights1, displacement1, nullptr) == 0);
143 CPPUNIT_ASSERT(layer->assign(teapot1, "body", material1, lights1, nullptr, nullptr) == 1);
144 layer->endUpdate();
145
146 // Can we look up those assignments?
147 Layer::MaterialLightSetPair idPair0 = layer->lookup(0);
148 CPPUNIT_ASSERT(idPair0.first == material1);
149 CPPUNIT_ASSERT(idPair0.second == lights1);
150 Layer::MaterialLightSetPair tuplePair0 = layer->lookup(teapot1, "lid");
151 CPPUNIT_ASSERT(tuplePair0.first == material1);
152 CPPUNIT_ASSERT(tuplePair0.second == lights1);
153 Layer::MaterialLightSetPair idPair1 = layer->lookup(1);
154 CPPUNIT_ASSERT(idPair1.first == material1);
155 CPPUNIT_ASSERT(idPair1.second == lights1);
156 Layer::MaterialLightSetPair tuplePair1 = layer->lookup(teapot1, "body");
157 CPPUNIT_ASSERT(tuplePair1.first == material1);
158 CPPUNIT_ASSERT(tuplePair1.second == lights1);
159
160 // Bad lookups should throw.
161 CPPUNIT_ASSERT_THROW(layer->lookup(2), except::IndexError);
162 CPPUNIT_ASSERT_THROW(layer->lookup(teapot2, "lid"), except::IndexError);
163 CPPUNIT_ASSERT_THROW(layer->lookup(teapot1, "spout"), except::IndexError);
164
165 // Reassignments should reuse the same assignment ID.
166 layer->beginUpdate();
167 CPPUNIT_ASSERT(layer->assign(teapot1, "lid", material2, lights2, displacement2, nullptr) == 0);
168 layer->endUpdate();
169 Layer::MaterialLightSetPair newIdPair = layer->lookup(0);
170 CPPUNIT_ASSERT(newIdPair.first == material2);
171 CPPUNIT_ASSERT(newIdPair.second == lights2);
172 Layer::MaterialLightSetPair newTuplePair = layer->lookup(teapot1, "lid");
173 CPPUNIT_ASSERT(newTuplePair.first == material2);
174 CPPUNIT_ASSERT(newTuplePair.second == lights2);
175
176 // Same part names on different geometries should be fine.
177 layer->beginUpdate();
178 layer->assign(teapot1, "lid", material1, lights1, displacement1, nullptr);
179 layer->assign(teapot1, "body", material1, lights1, nullptr, nullptr);
180 layer->assign(teapot2, "lid", material2, lights2, displacement2, nullptr);
181 layer->assign(teapot2, "body", material2, lights2, nullptr, nullptr);
182 layer->endUpdate();
183 Layer::MaterialLightSetPair teapot1Lid = layer->lookup(teapot1, "lid");
184 CPPUNIT_ASSERT(teapot1Lid.first == material1);
185 CPPUNIT_ASSERT(teapot1Lid.second == lights1);
186 Layer::MaterialLightSetPair teapot1Body = layer->lookup(teapot1, "body");
187 CPPUNIT_ASSERT(teapot1Body.first == material1);
188 CPPUNIT_ASSERT(teapot1Body.second == lights1);
189 Layer::MaterialLightSetPair teapot2Lid = layer->lookup(teapot2, "lid");
190 CPPUNIT_ASSERT(teapot2Lid.first == material2);
191 CPPUNIT_ASSERT(teapot2Lid.second == lights2);
192 Layer::MaterialLightSetPair teapot2Body = layer->lookup(teapot2, "body");
193 CPPUNIT_ASSERT(teapot2Body.first == material2);
194 CPPUNIT_ASSERT(teapot2Body.second == lights2);
195}

197void
198TestLayer::testDefaultAssignments()
199{
200 Geometry* teapot = mContext->createSceneObject("FakeTeapot", "/seq/shot/teapot")->asA<Geometry>();
201 Material* material1 = mContext->createSceneObject("FakeMaterial", "/seq/shot/material1")->asA<Material>();
202 Material* material2 = mContext->createSceneObject("FakeMaterial", "/seq/shot/material2")->asA<Material>();
203 Displacement* displacement = mContext->createSceneObject("FakeDisplacement", "/seq/shot/displacement")->asA<Displacement>();
204 Light* key = mContext->createSceneObject("FakeLight", "/seq/shot/key")->asA<Light>();
205
206 LightSet* rig = mContext->createSceneObject("LightSet", "/seq/shot/rig")->asA<LightSet>();
207 rig->beginUpdate();
208 rig->add(key);
209 rig->endUpdate();
210
211 // Create a layer.
212 Layer* layer = mContext->createSceneObject("Layer", "/seq/shot/layer")->asA<Layer>();
213
214 // Make some assignments, but not a default one.
215 layer->beginUpdate();
216 layer->assign(teapot, "lid", material1, rig, displacement, nullptr);
217 layer->assign(teapot, "body", material1, rig, nullptr, nullptr);
218 layer->assign(teapot, "spout", material1, rig, displacement, nullptr);
219 layer->endUpdate();
220
221 // Look up a valid and invalid part. Ensure we get "no assignment".
222 CPPUNIT_ASSERT_NO_THROW(
223 Layer::MaterialLightSetPair assignment = layer->lookup(teapot, "body");
224 CPPUNIT_ASSERT(assignment.first == static_cast<RootShader*>(material1));
225 CPPUNIT_ASSERT(assignment.second == rig);
226 );
227 CPPUNIT_ASSERT_THROW(layer->lookup(teapot, "handle"), except::IndexError);
228
229 // Add a default part assignment.
230 layer->beginUpdate();
231 layer->assign(teapot, "", material2, rig, displacement, nullptr);
232 layer->endUpdate();
233
234 // Verify that both lookups succeed.
235 CPPUNIT_ASSERT_NO_THROW(
236 Layer::MaterialLightSetPair assignment = layer->lookup(teapot, "body");
237 CPPUNIT_ASSERT(assignment.first == static_cast<RootShader*>(material1));
238 CPPUNIT_ASSERT(assignment.second == rig);
239 );
240 CPPUNIT_ASSERT_NO_THROW(
241 Layer::MaterialLightSetPair assignment = layer->lookup(teapot, "handle");
242 CPPUNIT_ASSERT(assignment.first == static_cast<RootShader*>(material2));
243 CPPUNIT_ASSERT(assignment.second == rig);
244 );
245}

248TestLayer::testClearLayer()
249{
250 Geometry* teapot = mContext->createSceneObject("FakeTeapot", "/seq/shot/teapot")->asA<Geometry>();
251 Material* material = mContext->createSceneObject("FakeMaterial", "/seq/shot/material")->asA<Material>();
252 Light* key = mContext->createSceneObject("FakeLight", "/seq/shot/key")->asA<Light>();
253 Displacement* displacement = mContext->createSceneObject("FakeDisplacement", "/seq/shot/displacement")->asA<Displacement>();
254
255 LightSet* rig = mContext->createSceneObject("LightSet", "/seq/shot/rig")->asA<LightSet>();
256 rig->beginUpdate();
257 rig->add(key);
258 rig->endUpdate();
259
260 // Create a layer.
261 Layer* layer = mContext->createSceneObject("Layer", "/seq/shot/layer")->asA<Layer>();
262
263 // Make some assignments, verify the assignment IDs.
264 layer->beginUpdate();
265 CPPUNIT_ASSERT(layer->assign(teapot, "lid", material, rig, displacement, nullptr) == 0);
266 CPPUNIT_ASSERT(layer->assign(teapot, "body", material, rig, nullptr, nullptr) == 1);
267 CPPUNIT_ASSERT(layer->assign(teapot, "spout", material, rig, displacement, nullptr) == 2);
268 layer->endUpdate();
269
270 // Clear the layer, verify that assignment IDs reset to 0.
271 layer->beginUpdate();
272 CPPUNIT_ASSERT_NO_THROW(layer->clear());
273 CPPUNIT_ASSERT(layer->assign(teapot, "handle", material, rig, displacement, nullptr) == 0);
274 layer->endUpdate();
275
276 // Clearing the layer outside an update should throw.
277 CPPUNIT_ASSERT_THROW(layer->clear(), except::RuntimeError);
278}
