30///
31/// @class ReferenceFrame ReferenceFrame.h <scene_rdl2/common/math/ReferenceFrame.h>
32/// @brief This class handles the transformation of directions from any global
33/// space (typically render space) to and from the local frame of
34/// reference used for sampling / manipulating spherical directions.
35/// This reference frame is similar as the one defined in PBRT
36/// (see: figure 8.3), except we use a right-handed orthonormal basis.
37///
38class ReferenceFrame
39{
40public:
41 /// Create an identity reference frame, where local space and global space
42 /// are the same
43 finline ReferenceFrame() :
44 mX(1.0f, 0.0f, 0.0f),
45 mY(0.0f, 1.0f, 0.0f),
46 mZ(0.0f, 0.0f, 1.0f) { }
47
48 /// Use this for constructing an isotropic surface bsdf lobe reference frame.
49 /// This reference is defined as:
50 /// X = an arbitrary direction orthogonal to N
51 /// Y = The third leg to form a right-handed orthogonal basis
52 /// Z = N
53 /// This method assumes input vector is normalized
54 finline ReferenceFrame(const Vec3f &N)
55 {
56 MNRY_ASSERT(isNormalized(N));
57
58 // Building an Orthonormal Basis, Revisited
59 // Duff, Tom et. al
60 // Pixar
61 // https://graphics.pixar.com/library/OrthonormalB/paper.pdf
62 const float sign = std::copysign(1.0f, N.z);
63 const float a = -1.0f / (sign + N.z);
64 const float b = N.x * N.y * a;
65 mX = Vec3f(1.0f + sign * N.x * N.x * a, sign * b, -sign * N.x);
66 mY = Vec3f(b, sign + N.y * N.y * a, -N.y);
67 mZ = N;
68 }
69
70 /// Use this for constructing an anisotropic surface bsdf lobe reference frame.
71 /// You can call it with arguments (N, normalize(dPdu)) or
72 /// (N, normalize(anisotropicDirection)).
73 /// This reference is defined as:
74 /// X = T made orthogonal to N and stay in the plane (N, T)
75 /// Y = The third leg to form a right-handed orthogonal basis
76 /// Z = N
77 /// This method assumes both input vectors are normalized
78 finline ReferenceFrame(const Vec3f &N, const Vec3f &T)
79 {
80 MNRY_ASSERT(isNormalized(N));
81 MNRY_ASSERT(isNormalized(T));
82
83 mZ = N;
84 mY = cross(N, T);
85 mY = normalize(mY);
86 mX = cross(mY, mZ);
87 }
88
89 /// Use this for constructing a hair bsdf lobe reference frame. You can call
90 /// it with arguments (I, normalize(dPdu)) or (I, normalize(hairDirection)).
91 /// The argument isHair is unused and is only here to de-ambiguate with the
92 /// constructor above.
93 /// This reference is defined as:
94 /// X = T
95 /// Y = vector orthogonal to plane (wo, T)
96 /// Z = The third leg to form a right-handed orthogonal basis. Z ends up
97 /// in the (wo, T) plane.
98 /// This method assumes both input vectors are normalized
99 finline ReferenceFrame(const Vec3f &wo, const Vec3f &T, bool /*isHair*/):
100 mX(T),
101 mY(normalize(cross(wo, T))),
102 mZ(cross(mX, mY))
103 {
104 MNRY_ASSERT(isNormalized(wo));
105 MNRY_ASSERT(isNormalized(T));
106 }
107
108 /// Utility function to create a ReferenceFrame from the rows of a matrix.
109 /// The matrix is assumed to be orthonormal.
110 finline ReferenceFrame(const Mat4f &m):
111 mX(asVec3(m.row0())),
112 mY(asVec3(m.row1())),
113 mZ(asVec3(m.row2()))
114 {
115 }
116
117 ~ReferenceFrame() { }
118
119
120 /// Accessors for global space vectors that form the orthonormal basis (X, Y, Z).
121 finline const Vec3f &getX() const { return mX; }
122 finline const Vec3f &getY() const { return mY; }
123 finline const Vec3f &getZ() const { return mZ; }
124
125 /// Convention: in all bases, N is mapped onto Z.
126 finline const Vec3f &getN() const { return mZ; }
127
128 /// Convention: T is mapped onto X.
129 finline const Vec3f &getT() const { return mX; }
130
131
132 /// Transform normalized directions to/from the caller's global coordinate
133 /// system (i.e. render space, world space, etc.) to/from the frame of
134 /// reference local coordinate system. If the input direction is normalized,
135 /// the ouput direction is also normalized.
136 finline Vec3f globalToLocal(const Vec3f &dir) const {
137 return Vec3f(dot(mX, dir), dot(mY, dir), dot(mZ, dir));
138 }
139 finline Vec3f localToGlobal(const Vec3f &dir) const {
140 return mX * dir[0] + mY * dir[1] + mZ * dir[2];
141 }
142
143
144 /// TODO: Flip the frame of reference. Rotate 180 about mX or negate all vectors ?
145
146 /// TODO: add method to make a global vector out of theta and phi ?
147 /// see: Util.h: computeLocalSphericalDirection()
148
149
150 /// Static HUD validation
151 static void hudValidation();
152
153 /// Right-handed orthonormal basis vectors, expressed in the caller's
154 /// coordinate system
155 Vec3f mX;
156 Vec3f mY;
157 Vec3f mZ;
158};
