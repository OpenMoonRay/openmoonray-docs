51void
52CryptomatteBuffer::addSampleScalar(unsigned x, unsigned y, float sampleId, float weight)
53{
54 PixelEntry &pixelEntry = mPixelEntries[y * mWidth + x];
55
56 // Iterate over fragments stored at current pixel and see if we can merge the sample in to any of them
57 for (Fragment &fragment : pixelEntry.mFragments) {
58 if (fragment.mId == sampleId) {
59 fragment.mCoverage += weight;
60 return;
61 }
62 }
63
64 // No match, so add a new fragment.
65 pixelEntry.mFragments.push_back(Fragment(sampleId, weight));

99void
100CryptomatteBuffer::finalize(const scene_rdl2::fb_util::PixelBuffer<unsigned>& samplesCount)
101{
102 if (mFinalized) {
103 return;
104 }
105
106 // Sort fragments in each pixel and compute final coverage values
107 for (size_t py = 0; py < mHeight; py++) {
108 for (size_t px = 0; px < mWidth; px++) {
109 // Get number of samples in this pixel
110 const unsigned numSamples = samplesCount.getPixel(px, py);
111 if (numSamples > 0) {
112 // Sort fragments by decreasing coverage
113 PixelEntry &pixelEntry = mPixelEntries[py * mWidth + px];
114 // We want ties to be broken deterministically so that we don't get false negatives when running Rats
115 // test when the fragments are added to the pixel entry in different orders.
116 pixelEntry.mFragments.sort([](const Fragment &f0, const Fragment &f1) {
117 if (f0.mCoverage != f1.mCoverage) {
118 return f0.mCoverage > f1.mCoverage;
119 }
120 return f0.mId > f1.mId;
121 });
122
123 // Normalize coverages so that they sum to 1 over the pixel
124 float recipNumSamples = 1.0f / static_cast<float>(numSamples);
125 for (Fragment &fragment : pixelEntry.mFragments) {
126 fragment.mCoverage *= recipNumSamples;
127 }
128 }
129 }
130 }
131
132 mFinalized = true;
133}

135// This function simply undoes the effect of finalize(), so the coverage values will be ready for further
136// accumulation following a checkpoint
137void
138CryptomatteBuffer::unfinalize(const scene_rdl2::fb_util::PixelBuffer<unsigned>& samplesCount)
139{
140 if (!mFinalized) {
141 return;
142 }
143
144 for (size_t py = 0; py < mHeight; py++) {
145 for (size_t px = 0; px < mWidth; px++) {
146 const unsigned numSamples = samplesCount.getPixel(px, py);
147 if (numSamples > 0) {
148 PixelEntry &pixelEntry = mPixelEntries[py * mWidth + px];
149 float numSamplesFloat = static_cast<float>(numSamples);
150 for (Fragment &fragment : pixelEntry.mFragments) {
151 // Convert back from a fraction to a count. The round() function is needed here
152 // because the product might not be exactly an integer due to truncation error.
153 fragment.mCoverage = std::round(fragment.mCoverage * numSamplesFloat);
154 }
155 }
156 }
157 }
158
159 mFinalized = false;
160}

162void
163CryptomatteBuffer::outputFragments(unsigned x, unsigned y, int numLayers, float *dest) const
164{
165 // Clamp numLayers so the output string will contain 2 digits 00-99 (per Cryptomatte spec)
166 if (numLayers < 1) numLayers = 1;
167 if (numLayers > 100) numLayers = 100;
168
169 // Output 2 fragments per layer: one pair goes to the R,G channels, the other to the B,A channels.
170 const PixelEntry &pixelEntry = mPixelEntries[y * mWidth + x];
171 int rank = 0;
172 for (const Fragment &fragment : pixelEntry.mFragments) {
173 *dest++ = fragment.mId;
174 *dest++ = fragment.mCoverage;
175 if (++rank >= 2 * numLayers) {
176 break;
177 }
178 }
179
180 // Supplement requested layers with zeros
181 for ( ; rank < 2 * numLayers; rank++)
182 {
183 *dest++ = 0.f;
184 *dest++ = 0.f;
185 }
186}
