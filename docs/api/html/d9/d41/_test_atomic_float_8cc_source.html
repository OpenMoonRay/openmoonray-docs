120template <typename TestClass>
121bool atomicFloatTest()
122{
123 constexpr int numThreads = 20;
124 TestClass tester;
125
126 std::vector<std::thread> threads;
127 threads.reserve(numThreads);
128 for (int i = 0; i < numThreads; ++i) {
129 threads.emplace_back(std::ref(tester));
130 }
131
132 for (auto& t : threads) {
133 t.join();
134 }
135
