40void
41Localizer::localize(const std::string& inFile, const std::string& outFile)
42{
43 // Write test the output file.
44 if (!util::writeTest(outFile, true)) {
45 throw except::IoError(util::buildString("Can't write output file '",
46 outFile, "'."));
47 }
48
49 // Load the input file.
50 rdl2::SceneContext context;
51 context.setProxyModeEnabled(true);
52 if (!mDsoPath.empty()) { // Override dso path from command line
53 context.setDsoPath(mDsoPath);
54 }
55 rdl2::readSceneFromFile(inFile, context);
56
57 // Construct the source prefix based on the the path prefix of the input
58 // RDL file.
59 auto srcPrefix = util::simplifyPath(util::absolutePath(util::splitPath(inFile).first));
60
61 // The asset files should derive their destination paths from the output
62 // file.
63 auto destPrefix = util::simplifyPath(util::absolutePath(util::splitPath(outFile).first));
64 if (!destPrefix.empty()) {
65 destPrefix += '/';
66 }
67
68 // Compute the list of motion sample numbers directly from the SceneVars.
69 const auto& sceneVars = context.getSceneVariables();
70 auto motionSampleNums = rdl2::uniqueSampleNumberRange(sceneVars);
71
72 // Scan the SceneClasses for localizable (filename) attributes.
73 LocalizableAttributes localAttrs(context);
74
75 // Walk all the SceneObjects and build the path tree based on which
76 // attributes are localizable (just filenames, for now).
77 PathTree pathTree;
78 for (auto objIter = context.beginSceneObject(); objIter != context.endSceneObject(); ++objIter) {
79 rdl2::SceneObject& obj = *objIter->second;
80 auto range = localAttrs.getLocalizableAttributes(obj.getSceneClass());
81 for (auto attrIter = range.first; attrIter != range.second; ++attrIter) {
82 const rdl2::Attribute& attr = *attrIter->second;
83 std::string attrValue = obj.get<rdl2::String>(attr.getName());
84 if (attrValue.empty()) continue;
85 pathTree.insert(attrValue, &obj, &attr);
86 }
87 }
88
89 // Expand any # characters into full paths based on the motion sample
90 // numbers.
91 pathTree.expandPaths(motionSampleNums);
92
93 // Trim the file path prefix to remove shared leading directories.
94 auto trimmedPrefix = pathTree.trimPrefix();
95
96 // Generate the list of files that need to be copied.
97 auto fileCopies = pathTree.getFileCopies(srcPrefix, destPrefix, trimmedPrefix);
98
99 // Generate the list of attribute updates that need to be done.
100 auto attrUpdates = pathTree.getAttrUpdates(destPrefix, mRelativePaths);
101
102 // Unless we're force overwriting destination files, make sure that none
103 // of them exist.
104 if (!mForceOverwrite) {
105 // The output RDL2 file.
106 if (access(outFile.c_str(), F_OK) == 0) {
107 throw except::IoError(util::buildString("Destination file '",
108 outFile, "' already exists. Use --force to overwrite."));
109 }
110
111 // The asset files we're going to copy.
112 for (const auto& fileCopy : fileCopies) {
113 if (access(fileCopy.mDestPath.c_str(), F_OK) == 0) {
114 throw except::IoError(util::buildString("Destination file '",
115 fileCopy.mDestPath, "' already exists. (Copying from '",
116 fileCopy.mSrcPath, "'.) Use --force to overwrite."));
117 }
118 }
119 }
120
121 // Write test all the files we're going to copy.
122 for (const auto& fileCopy : fileCopies) {
123 if (!util::writeTest(fileCopy.mDestPath, true)) {
124 throw except::IoError(util::buildString("Can't write file '",
125 fileCopy.mDestPath, "'."));
126 }
127 }
128
129 // Copy the assets. We know the directory exists and is writable because
130 // we did a util::writeTest() at the very beginning of this function.
131 for (const auto& fileCopy : fileCopies) {
132 // copyFile will throw if the source file doesn't exist, in which case
133 // we print out the missing file and continue.
134 try {
135 std::cout << "Copying " << fileCopy.mSrcPath << "\n"
136 " to " << fileCopy.mDestPath << std::endl;
137 util::copyFile(fileCopy.mSrcPath, fileCopy.mDestPath);
138 }
139 catch (const except::IoError &e) {
140 std::cerr << e.what() << std::endl;
141 }
142 }
143
144 // Update the attribute values.
145 for (const auto& update : attrUpdates) {
146 std::cout << "Updating " << update.mSceneObject->getName() << "\n"
147 " attr " << update.mAttribute->getName() << "\n"
148 " to " << update.mValue << std::endl;
149 {
150 rdl2::SceneObject::UpdateGuard guard(update.mSceneObject);
151 update.mSceneObject->set(update.mAttribute->getName(), update.mValue);
152 }
153 }
154
155 // Write the localized output file.
156 std::cout << "Writing " << outFile << std::endl;
157 rdl2::writeSceneToFile(context, outFile);
158}
