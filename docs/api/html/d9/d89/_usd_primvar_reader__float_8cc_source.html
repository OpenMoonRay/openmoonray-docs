48void
49UsdPrimvarReader_float::update()
50{
51 if (hasChanged(attrVarName)) {
52 mOptionalAttributes.clear();
53 const std::string& varname = get(attrVarName);
54 mIspc.mAttributeMapType = ispc::PRIMVAR_MAP_TYPE_PRIMITIVE_ATTRIBUTE;
55 moonray::shading::TypedAttributeKey<float> attributeKey(varname);
56 mIspc.mPrimitiveAttributeIndex = attributeKey;
57 mOptionalAttributes.push_back(attributeKey);
58 ::moonshine::primvar::createLogEvent("float",
59 varname,
60 mIspc.mMissingAttributeEvent,
61 mLogEventRegistry);
62 }

65void
66UsdPrimvarReader_float::sample(const scene_rdl2::rdl2::Map *self, moonray::shading::TLState *tls,
67 const moonray::shading::State &state, Color *sample)
68{
69 const UsdPrimvarReader_float* me = static_cast<const UsdPrimvarReader_float*>(self);
70
71 int key = me->mIspc.mPrimitiveAttributeIndex;
72 if (state.isProvided(key)) {
73 const float& v = state.getAttribute(moonray::shading::TypedAttributeKey<float>(key));
74 sample->r = v;
75 sample->g = v;
76 sample->b = v;
77 } else {
78 // the primitive attribute is unavailable, use fallback parameter
79 const float& v = evalFloat(me, attrFallback, tls, state);
80 sample->r = v;
81 sample->g = v;
82 sample->b = v;
83 if (me->get(attrWarnWhenUnavailable)) {
84 moonray::shading::logEvent(me, tls, me->mIspc.mMissingAttributeEvent);
85 }
86 }
87}
