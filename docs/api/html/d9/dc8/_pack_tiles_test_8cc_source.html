22static bool
23readActivePixelsArray(const std::string &filename,
24 ActivePixelsArray &outActivePixelsArray)
25{
26 std::ifstream fin(filename, std::ios::in|std::ios::binary);
27 if (!fin) {
28 std::cerr << "read open failed. file:" << filename << std::endl;
29 return false;
30 }
31
32 fin.seekg(0, std::ios_base::end);
33 size_t fileSize = fin.tellg();
34 fin.seekg(0, std::ios_base::beg);
35
36 std::string data(fileSize, 0x0);
37 fin.read(&data[0], fileSize);
38 if (!fin) {
39 std::cerr << "read data failed. file:" << filename << std::endl;
40 return false;
41 }
42
43 fin.close();
44
45 //------------------------------
46
47 try {
48 outActivePixelsArray.decode(data);
49 }
50 catch (...) {
51 std::cerr << "decode activePixelsArray failed." << std::endl;
52 return false;
53 }
54
55 return true;
56}

131//
132// EnqTimeMaskBlock ver1+ver2 timing test using already dumped ActivePixelsArray data
133// ver1 : original naive activeTileId + activePixelMask
134// ver2 : PackActiveTiles encoding method
135//
136// Typical method to create this snapshotDeltaDump file is to use debug console command of
137// progmcrt_dispatch.
138// Followings are related progmcrt_dispatch debug console commands.
139// snapshotDeltaRecStart : start snapshotDelta rec
140// snapshotDeltaRecStop : stop snapshotDelta rec
141// snapshotDeltaRecReset : reset and clear previous snapshotDelta rec info and status
142// snapshotDeltaRecDump file : output snapshotDelta rec info to the file. required "stop" first.
143//
144{
145 // Typically, cerr output from this function will be used by gnuplot.
146 // So we output start by # symbol about comment information.
147

149
150 ActivePixelsArray activePixelsArray;
151 if (!readActivePixelsArray(filename, activePixelsArray)) {
152 std::cerr << "read activePixelsArray failed." << std::endl;
153 return;
154 }
155
156 //------------------------------
157
158 std::cerr << "# 1 2 3 4 5 6 7"
159 << " 8 9" << std::endl;
160 std::cerr << "# coarse totalActivePixels ver1Time ver2Time ver1Size ver2Size %"
161 << " ver1PixPosInfoAve ver2PixPosInfoAve" << std::endl;
162
163 for (size_t i = 0; i < activePixelsArray.size(); ++i) {
164 const fb_util::ActivePixels &currActivePixels = activePixelsArray.get(i);
165 bool currCoarsePass = activePixelsArray.getCoarsePass(i);
166
167 std::cerr << currCoarsePass << ' ';
168
169 PackTiles::PrecisionMode precisionMode =
170 ((currCoarsePass)? PackTiles::PrecisionMode::H16: PackTiles::PrecisionMode::F32);
171 PackTiles::timingAndSizeTest(currActivePixels, precisionMode);
172 }
173
174 std::cerr << "#>> PackTilestest.cc replaySnapshotDelta() filename:" << filename << " done" << std::endl;
175}

178void
179PackTilesTest::replaySnapshotDelta_dumpActivePixPos(const std::string &filename,
180 unsigned snapshotId)
181//
182// Dump activePixel position info about particular snapshotId of already dumped ActivePixelsArray data
183//
184// Typical method to create this snapshotDeltaDump file is to use debug console command of
185// progmcrt_dispatch.
186// Followings are related progmcrt_dispatch debug console commands.
187// snapshotDeltaRecStart : start snapshotDelta rec
188// snapshotDeltaRecStop : stop snapshotDelta rec
189// snapshotDeltaRecReset : reset and clear previous snapshotDelta rec info and status
190// snapshotDeltaRecDump file : output snapshotDelta rec info to the file. required "stop" first.
191//
192{
193 // Typically, cerr output from this function will be used by gnuplot.
194 // So we output start by # symbol about comment information.
195
196 std::cerr << "#>> PackTilestest.cc replaySnapshotDelta_dumpActivePixPos()"
197 << " filename:" << filename
198 << " snapshotId:" << snapshotId
199 << " start" << std::endl;
200
201 ActivePixelsArray activePixelsArray;
202 if (!readActivePixelsArray(filename, activePixelsArray)) {
203 std::cerr << "read activePixelsArray failed." << std::endl;
204 return;
205 }
206
207 //------------------------------
208
209 std::cerr << "# totalSnapshotCount:" << activePixelsArray.size() << std::endl;
210
211 snapshotId = std::min(snapshotId, (unsigned)activePixelsArray.size() - 1);
212 const fb_util::ActivePixels &currActivePixels = activePixelsArray.get(snapshotId);
213
214 std::cerr << "# activePixelTotal:" << currActivePixels.getActivePixelTotal() << std::endl;
215
216 unsigned tileXTotal = currActivePixels.getNumTilesX();
217
218 std::cerr << "# 1 2" << std::endl;
219 std::cerr << "# posX posY" << std::endl;
220
221 fb_util::ActivePixels::crawlAllActivePixels
222 (currActivePixels,
223 [&](unsigned currPixOffset) {
224 unsigned tileId = currPixOffset / 64;
225 unsigned tileBaseX = (tileId % tileXTotal) * 8;
226 unsigned tileBaseY = (tileId / tileXTotal) * 8;
227 unsigned pixLocalId = currPixOffset % 64;
228 unsigned pixLocalX = pixLocalId % 8;
229 unsigned pixLocalY = pixLocalId / 8;
230 unsigned pixX = tileBaseX + pixLocalX;
231 unsigned pixY = tileBaseY + pixLocalY;
232
233 std::cerr << pixX << ' ' << pixY << std::endl;
234 });
235
236 std::cerr << "#>> PackTilestest.cc replaySnapshotDelta_dumpActivePixPos()"
237 << " filename:" << filename
238 << " snapshotId:" << snapshotId
239 << " done" << std::endl;
240}
