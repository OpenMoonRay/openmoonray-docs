535 const SceneClass* fakeTeapot = context.createSceneClass("FakeTeapot");
536 AttributeKey<Float> fakenessKey = fakeTeapot->getAttributeKey<Float>("fakeness");
537
538 // New objects, even with no attributes set, should be encoded.
539 SceneObject* teapot = context.createSceneObject("FakeTeapot", "/seq/shot/teapot");
540 BinaryWriter writer1(context);
541 writer1.setDeltaEncoding(true);
542 writer1.toFile("delta1.rdlb");

544 BinaryReader reader1(readContext1);
545 reader1.fromFile("delta1.rdlb");
546 CPPUNIT_ASSERT(readContext1.getSceneObject("/seq/shot/teapot"));
547
548 context.commitAllChanges();
549
550 // Delta encoding with no changes should result in no update.
551 BinaryWriter writer2(context);
552 writer2.setDeltaEncoding(true);
553 writer2.toFile("delta2.rdlb");
554 SceneContext readContext2;
555 BinaryReader reader2(readContext2);
556 reader2.fromFile("delta2.rdlb");
557 CPPUNIT_ASSERT_THROW(
558 readContext2.getSceneObject("/seq/shot/teapot");
559 , except::KeyError);
560
561 // Changing an attribute should result in an update.
562 teapot->beginUpdate();
563 teapot->set(fakenessKey, 99.99f);
564 teapot->endUpdate();
565 BinaryWriter writer3(context);
566 writer3.setDeltaEncoding(true);
567 writer3.toFile("delta3.rdlb");
568 SceneContext readContext3;
569 BinaryReader reader3(readContext3);
570 reader3.fromFile("delta3.rdlb");
571 CPPUNIT_ASSERT_NO_THROW(
572 CPPUNIT_ASSERT_DOUBLES_EQUAL(99.99f,
573 readContext3.getSceneObject("/seq/shot/teapot")->get(fakenessKey), 0.0001f);
574 );
