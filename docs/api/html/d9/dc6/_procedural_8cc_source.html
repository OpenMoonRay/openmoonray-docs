26class PrimitiveMemoryAccumulator : public PrimitiveVisitor
27{
28public:
29 PrimitiveMemoryAccumulator(
30 tbb::atomic<Primitive::size_type>& usage,
31 SharedPrimitiveSet& sharedPrimitives,
32 bool inPrimitiveGroup = false) :
33 mUsage(usage), mSharedPrimitives(sharedPrimitives),
34 mInPrimitiveGroup(inPrimitiveGroup) {}
35
36 virtual void visitPrimitive(Primitive& p) override {
37 // if the primitive is in PrimitiveGroup, the memory usage of this
38 // Primitive has been taken into account
39 if (!mInPrimitiveGroup) {
40 mUsage += p.getMemory();
41 }
42 }
43
44 virtual void visitPrimitiveGroup(PrimitiveGroup& pg) override {
45 if (!mInPrimitiveGroup) {
46 mUsage += pg.getMemory();
47 }
48 // make sure the primitives in this PrimitiveGroup doesn't get added
49 // twice. The reason we recursively travel down is that we need to
50 // collect all SharedPrimitive that get instanced around
51 PrimitiveMemoryAccumulator accumulator(mUsage, mSharedPrimitives, true);
52 pg.forEachPrimitive(accumulator);
53 }
54
55 virtual void visitInstance(Instance& i) override {
56 if (!mInPrimitiveGroup) {
57 mUsage += i.getMemory();
58 }
59 const auto& ref = i.getReference();
60 // visit the referenced Primitive if it's not visited yet
61 if (mSharedPrimitives.insert(ref).second) {
62 PrimitiveMemoryAccumulator accumulator(mUsage, mSharedPrimitives);
63 ref->getPrimitive()->accept(accumulator);
64 }
65 }
66
67private:
68 tbb::atomic<Primitive::size_type>& mUsage;
69 SharedPrimitiveSet& mSharedPrimitives;
70 bool mInPrimitiveGroup;
71};

73class StatisticsAccumulator : public PrimitiveVisitor
74{
75public:
76 StatisticsAccumulator(
77 GeometryStatistics& geometryStatistics,
78 SharedPrimitiveSet& sharedPrimitives) :
79 mGeometryStatistics(geometryStatistics),
80 mSharedPrimitives(sharedPrimitives) {}
81
82 virtual void visitPrimitive(Primitive& p) override {
83 }
84
85 virtual void visitPolygonMesh(PolygonMesh& p) override {
86 mGeometryStatistics.mFaceCount += p.getFaceCount();
87 mGeometryStatistics.mMeshVertexCount += p.getVertexCount();
88 }
89
90 virtual void visitSubdivisionMesh(SubdivisionMesh& s) override {
91 mGeometryStatistics.mFaceCount += s.getSubdivideFaceCount();
92 mGeometryStatistics.mMeshVertexCount += s.getSubdivideVertexCount();
93 }
94
95 virtual void visitCurves(Curves& c) override {
96 mGeometryStatistics.mCurvesCount += c.getCurvesCount();
97 const auto& curvesVertexCount = c.getCurvesVertexCount();
98 mGeometryStatistics.mCVCount += std::accumulate(curvesVertexCount.begin(),
99 curvesVertexCount.end(), 0);
100 }
101
102 virtual void visitTransformedPrimitive(geom::TransformedPrimitive& t) override {
103 t.getPrimitive()->accept(*this);
104 }
105
106 virtual void visitPrimitiveGroup(PrimitiveGroup& pg) override {
107 pg.forEachPrimitive(*this);
108 }
109
110 virtual void visitInstance(Instance& i) override {
111 ++mGeometryStatistics.mInstanceCount;
112 const auto& ref = i.getReference();
113 // visit the referenced Primitive if it's not visited yet
114 if (mSharedPrimitives.insert(ref).second) {
115 ref->getPrimitive()->accept(*this);
116 }
117 }
118
119private:
120 GeometryStatistics& mGeometryStatistics;
121 SharedPrimitiveSet& mSharedPrimitives;
122};

125Procedural::size_type
126Procedural::getMemory()
127{
128 tbb::atomic<Primitive::size_type> usage {0u};
129 SharedPrimitiveSet sharedPrimitives;
130 PrimitiveMemoryAccumulator accumulator(usage, sharedPrimitives);
131 forEachPrimitive(accumulator);
132 return usage;
133}

135GeometryStatistics
136Procedural::getStatistics() const
137{
138 GeometryStatistics geometryStatistics;
139
140 SharedPrimitiveSet sharedPrimitives;
141 StatisticsAccumulator accumulator(geometryStatistics, sharedPrimitives);
142 const_cast<Procedural *>(this)->forEachPrimitive(accumulator);
143
144 return geometryStatistics;
145}
