27 void clear() {
28 // TODO the "delete" part should be removed after we update tbb
29 // and change tbb::concurrent_vector<Primitive*> to
30 // tbb::concurrent_vector<std::unique_ptr<Primitive>>

34 }
35 mChildren.clear();
36 }

38 Primitive::size_type getMemory() const {
39 Primitive::size_type memory = sizeof(PrimitiveGroup::Impl) +
40 mChildren.capacity() * sizeof(Primitive*);
41 for (const auto& child : mChildren) {
42 memory += child->getMemory();
43 }
44 return memory;
45 }

47 tbb::concurrent_vector<Primitive*> mChildren;

100void
101PrimitiveGroup::forEachPrimitive(PrimitiveVisitor& visitor, bool parallel)
102{
103 if (parallel) {
104 tbb::blocked_range<size_t> range(0, mImpl->mChildren.size());
105 tbb::parallel_for(range, [&](const tbb::blocked_range<size_t> &r)
106 {
107 for (size_t i = r.begin(); i < r.end(); ++i) {
108 mImpl->mChildren[i]->accept(visitor);
109 }
110 });
111 } else {
112 for (size_t i = 0; i < mImpl->mChildren.size(); ++i) {
113 mImpl->mChildren[i]->accept(visitor);
114 }
115 }
116}
