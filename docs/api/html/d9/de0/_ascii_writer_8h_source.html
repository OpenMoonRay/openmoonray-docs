114template <typename Container>
115void
116AsciiWriter::writeSet(std::ostream& out, const Container& members) const
117{
118 // Sort the elements of the set by name.
119 std::vector<const SceneObject*> order(members.begin(), members.end());
120 std::sort(order.begin(), order.end(), [](const SceneObject* a, const SceneObject* b) {
121 return a->getName() < b->getName();
122 });
123
124 // Write out each member in the set, in order.
125 for (auto iter = order.begin(); iter != order.end(); ++iter) {
126 // TODO: don't use a full object reference?
127 out << mIndent << sceneObjectRef(*iter) << ",\n";
128 }
129}
