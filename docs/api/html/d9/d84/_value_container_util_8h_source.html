90private:
91 //
92 // When using variable length encoding for unsigned integers, we can achieve smaller memory footprint
93 // if using small value (i.e., close to 0). And memory footprint is getting bigger which related to value is bigger.
94 //
95 // unsigned int (32bit) case, data sizes are as follows.
96 // 0 ~ 127 = 1Byte
97 // 128 ~ 16,383 = 2Byte
98 // 16,384 ~ 2,097,151 = 3Byte
99 // 2,097,152 ~ 268,435,455 = 4Byte
100 // 268,435,456 ~ 4,294,967,295 = 5Byte
101 //
102 // unsigned long (64bit) case, data sizes are as follows.
103 // 0 ~ 127 = 1Byte
104 // 128 ~ 16,383 = 2Byte
105 // 16,384 ~ 2,097,151 = 3Byte
106 // 2,097,152 ~ 268,435,455 = 4Byte
107 // 268,435,456 ~ 34,359,738,367 = 5Byte
108 // 34,359,738,368 ~ 4,398,046,511,103 = 6Byte
109 // 4,398,046,511,104 ~ 562,949,953,421,311 = 7Byte
110 // 562,949,953,421,312 ~ 72,057,594,037,927,935 = 8Byte
111 // 72,057,594,037,927,936 ~ 9,223,372,036,854,775,807 = 9Byte
112 // 9,223,372,036,854,775,808 ~ 18,446,744,073,709,551,615 = 10Byte
113 //
114 // Obviously, smaller number gets small memory footprint. Usually we are using very close to 0 number a lot.
115 // This is a motivation to use variable length coding for integers.
116 //
117 // We need some trick to handle signed numbers. If we simply apply same technics to signed number,
118 // negative small number (like -1) creates very long encoded value due to negative number is converted to
119 // very huge unsigned number. In order to keep same characteristic of unsigned, we converted signed integers
120 // to unsigned by using zig-zag coding logic. This is a very simple idea.
121 //
122 // signed <-> unsisgned
123 // 0 <-> 0
124 // -1 <-> 1
125 // 1 <-> 2
126 // -2 <-> 3
127 // 2 <-> 4
128 // ...
129 // 2,147,483,647 <-> 4,294,967,294
130 // -2,147,483,648 <-> 4,294,967,295
131 //
132 // If we apply zig-zag coding, singed int can be nicely fit into variable length coding logic and
133 // we can achieve small number is small memory characteristic.
134 //
135 static inline unsigned int zigZagEncoding(const int i) { return (i >> 31) ^ (i << 1); }
136 static inline int zigZagDecoding(const unsigned int ui) { return (ui >> 1) ^ -(ui & 1); }
137 static inline unsigned long zigZagEncoding(const long l) { return (l >> 63) ^ (l << 1); }
138 static inline long zigZagDecoding(const unsigned long ul) { return (ul >> 1) ^ -(ul & 1); }
139};

191inline size_t
192ValueContainerUtil::variableLengthEncoding(unsigned int ui, void *outPtr)
193{
194 char *out = static_cast<char *>(outPtr);
195
196 size_t size = 0;
197 while (ui > 0x7f) {
198 *out++ = static_cast<char>((ui & 0x7f) | 0x80);
199 ui >>= 7;
200 size++;
201 }
202 *out = static_cast<char>(ui & 0xff);
203 return ++size;
204}

206// static function
207inline size_t
208ValueContainerUtil::variableLengthDecoding(const void *inPtr, unsigned int &ui)
209{
210 const char *in = static_cast<const char *>(inPtr);
211
212 ui = 0x0;
213 size_t size = 0;
214 int shift = 0;
215 while (1) {
216 size++;
217 ui |= ((*in) & 0x7f) << shift;
218 if (!(*in++ & 0x80)) break;
219 shift += 7;
220 }
221 return size;
222}
