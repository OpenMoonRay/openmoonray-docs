17class MotionBlurParams {
18public:
19 MotionBlurParams() {}
20
21 MotionBlurParams(const std::vector<float>& motionSteps,
22 float shutterOpen, float shutterClose,
23 bool isMotionBlurOn, float fps):
24 mMotionSteps(motionSteps),
25 mShutterOpen(shutterOpen),
26 mShutterClose(shutterClose),
27 mIsMotionBlurOn(isMotionBlurOn),
28 mInvFps(1.f / fps)
29 {
30 // calculate the delta fraction of shutter open/close time in
31 // motionSteps duration. For example: if first motion step is -1
32 // and the last motion step is 1, and shutter open time is -0.75
33 // shutter close time is 0.3
34 // -1 1
35 // --------------------------------------------------------
36 // -0.75 0.3
37 // then shutterOpenDelta = (-0.75 - (-1)) / (1 - (-1)) = 0.125
38 // shutterCloseDelta = ( 0.3 - (-1)) / (1 - (-1)) = 0.65
39 //
40 // this info is used to interpolate vertex data and primitive atttribute
41 // from frame time coordinate to shutter time coordinate
42 // (where shutter open time = 0 and shutter close time = 1)
43
44 if (mMotionSteps.size() <= 1) {
45 // static case
46 mShutterOpenDelta = 0.0f;
47 mShutterCloseDelta = 0.0f;
48 mDt = 0.0f;
49 } else if (mMotionSteps.size() == 2) {
50 // We have an animation, interpolate/extrapolate to shutter open/close
51 float m0 = mMotionSteps[0];
52 float m1 = mMotionSteps[1];
53 if (!scene_rdl2::math::isEqual(m0, m1)) {
54 mShutterOpenDelta = (mShutterOpen - m0) / (m1 - m0);
55 mShutterCloseDelta = (mShutterClose - m0) / (m1 - m0);
56 } else {
57 // it makes no sense to have two same motion steps,
58 // but we should avoid generating nan if it happens
59 mShutterOpenDelta = 0.0f;
60 mShutterCloseDelta = 0.0f;
61 }
62 mT0 = m0 * mInvFps;
63 mT1 = m1 * mInvFps;
64 mDt = mT1 - mT0;
65 } else {
66 MNRY_ASSERT_REQUIRE(false, "only support two time samples for motionblur at this moment");
67 }
68 }
69
70 const std::vector<float>& getMotionSteps() const { return mMotionSteps; }
71
72 float getShutterOpen() const { return mShutterOpen; }
73
74 float getShutterClose() const { return mShutterClose; }
75
76 void getMotionBlurDelta(float& shutterOpenDelta, float& shutterCloseDelta) const
77 {
78 shutterOpenDelta = mShutterOpenDelta;
79 shutterCloseDelta = mShutterCloseDelta;
80 }
81
82 void getMotionStepTimes(float& t0, float& t1) const
83 {
84 t0 = mT0;
85 t1 = mT1;
86 }
87
88 float getDt() const { return mDt; }
89
90 float getInvFps() const { return mInvFps; }
91
92 bool isMotionBlurOn() const { return mIsMotionBlurOn; }
93
94private:
95 std::vector<float> mMotionSteps;
96 float mShutterOpen;
97 float mShutterClose;
98 // delta fraction of shutter open/close time in motionSteps duration
99 float mShutterOpenDelta;
100 float mShutterCloseDelta;
101 bool mIsMotionBlurOn;
102 float mT0, mT1, mDt;
103 float mInvFps;
104};
