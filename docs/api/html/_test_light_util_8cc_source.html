42static void
43testPlaneDistance(const Vec3f &o, const Vec3f &n, float t1, float t2, float l, float eps=epsilon)
44{
45 Plane plane(o, n);
46 Vec3f T1 = cross(n, Vec3f(1.0f, 0.0f, 0.0f));
47 Vec3f T2 = cross(n, Vec3f(0.0f, 0.0f, 1.0f));
48
49 Vec3f p = o + T1 * t1 + T2 * t2 + n * l;
50 float d = plane.getDistance(p);
51 testAssert(isEqual(d, l, eps), "%f != %f", d, l);
52}

84{

97static void
98computeFalloffCurveSamples(const FalloffCurve &fc, float samples[sSampleCount])
99{
100 for (int i=0; i < sSampleCount; i++) {
101 float x = i / float(sSampleCount - 1);
102 samples[i] = fc.eval(x);
103 }

107static void
108testFalloffCurveType(FalloffCurveType type, ispc::FalloffCurveType ispcType)
109{
110 CACHE_ALIGN float samples[sSampleCount];
111 bool equal;
112
113 FalloffCurve fc(type);
114 computeFalloffCurveSamples(fc, samples);
115 printFalloffCurveSamples(type, samples);
116 equal = asCppBool(ispc::testFalloffCurve(fc.asIspc(), samples, sSampleCount));
117 if (!equal) {
118 fprintf(stderr, "Error: ISPC samples are different:\n");
119 printFalloffCurveSamples(type, samples);
120 }
121 CPPUNIT_ASSERT(equal);
122}
