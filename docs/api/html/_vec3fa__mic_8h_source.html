179 __forceinline float reduce_add(const Vec3fa_t& a) {
180 __m512 b = _mm512_mask_blend_ps(0x8,a.m512,_mm512_set1_ps(0.0f));
181 __m512 c = _mm512_add_ps(b, _mm512_swizzle_ps(b, _MM_SWIZ_REG_BADC));
182 __m512 d = _mm512_add_ps(c, _mm512_swizzle_ps(c, _MM_SWIZ_REG_CDAB));
183 return _mm512_cvtss_f32(d);
184 }

186 __forceinline float reduce_mul(const Vec3fa_t& a) {
187 __m512 b = _mm512_mask_blend_ps(0x8,a.m512,_mm512_set1_ps(1.0f));
188 __m512 c = _mm512_mul_ps(b, _mm512_swizzle_ps(b, _MM_SWIZ_REG_BADC));
189 __m512 d = _mm512_mul_ps(c, _mm512_swizzle_ps(c, _MM_SWIZ_REG_CDAB));
190 return _mm512_cvtss_f32(d);
191 }

229 __forceinline Vec3fa_t cross ( const Vec3fa_t& a, const Vec3fa_t& b )
230 {
231 __m512 c = _mm512_mul_ps(b,_mm512_swizzle_ps(a,_MM_SWIZ_REG_DACB));
232 c = _mm512_fmsub_ps(_mm512_swizzle_ps(b,_MM_SWIZ_REG_DACB),a,c);
233 c = _mm512_swizzle_ps(c,_MM_SWIZ_REG_DACB);
234 return c;
