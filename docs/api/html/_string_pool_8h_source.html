39finline const std::string* StringPool::get(const std::string& s)
40{
41 auto it = mStringMap.find(s);
42 if (it != mStringMap.end()) {
43 return it->second;
44 } else {
45 std::string* newString = new std::string(s);
46 std::pair<std::string, std::string*> item(s, newString);
47 std::pair<StringMap::iterator, bool> result = mStringMap.insert(item);
48 if (result.second) {
49 return newString;
50 } else {
51 delete newString;
52 return mStringMap[s];
53 }
54 }
