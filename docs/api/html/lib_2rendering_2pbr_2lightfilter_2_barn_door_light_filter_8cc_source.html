94Xform3f
95BarnDoorLightFilter::getSlerpXformRender2Local(float time) const
96{
97 // The result of applying the inverse of Local2Render to a point p is
98 // (p + trans) * rot * scale, where trans = -position,
99 // rot = render2localRot.l, and scale = render2localScale.
100 const Vec3f trans = (mMb & LIGHTFILTER_MB_TRANSLATION) ?
101 -lerp(mPosition[0], mPosition[1], time) :
102 -mPosition[0];
103 const Mat3f rot = (mMb & LIGHTFILTER_MB_ROTATION) ?
104 // transpose is inverse
105 Mat3f(slerp(mOrientation[0], mOrientation[1], time)).transposed() :
106 mRender2LocalRot0.l;
107 const Mat3f scale = (mMb & LIGHTFILTER_MB_SCALE) ?
108 // The inverse() call below, while unfortunate, is necessary.
109 // Reversing the order of operations (and lerping precomputed inverses)
110 // does not look correct.
111 lerp(mLocal2RenderScale[0], mLocal2RenderScale[1], time).inverse() :
112 mRender2LocalScale0;
113
114 return Xform3f(one, trans) * Xform3f(rot * scale, zero);
115}

117bool
118BarnDoorLightFilter::updateTransforms(const Mat4f &local2Render, int ti)
119{
120 // update transforms at time index (ti)
121 // ti == 0 is at normalized rayTime = 0.f
122 // ti == 1 is at normalized rayTime = 1.f;
123 MNRY_ASSERT(mRdlLightFilter);
124
125 Quaternion3f rot;
126 Mat3f scale;
127 Xform3f local2RenderXform = xform<Xform3f>(local2Render);
128 decompose(local2RenderXform.l, scale, rot);
129
130 mLocal2RenderScale[ti] = scale;
131
132 // Single shutter-time items (for when there is no MB)
133 if (ti == 0) {
134 mRender2Local0 = local2RenderXform.inverse();
135 mRender2LocalScale0 = scale.inverse();
136 mRender2LocalRot0 = Mat3f(rot).transposed();
137 }
138
139 mPosition[ti] = asVec3(local2Render.row3());
140
141 Xform3f local2RenderRot = Xform3f(Mat3f(rot), zero);
142 mOrientation[ti] = Quaternion3f(local2RenderRot.row0(),
143 local2RenderRot.row1(),
144 local2RenderRot.row2());
145
146 // This code relies on update() being called twice: first with ti == 0,
147 // and then with ti == 1.
148 //
149 // After setting the second quaternion (ti==1), check if the angle from the
150 // first quaternion is negative and flip accordingly.
151 if (ti==1 && dot(mOrientation[0], mOrientation[1]) < 0) {
152 mOrientation[1] *= -1.f;
153 }
154
155 return true;
156}

158bool
159BarnDoorLightFilter::updateParamAndTransforms(const Mat4f &local2Render0,
160 const Mat4f &local2Render1)
161{
162 if (!updateTransforms(local2Render0, /* ti = */ 0)) return false;
163
164 // setup mMb
165 mMb = LIGHTFILTER_MB_NONE;
166 const scene_rdl2::rdl2::SceneVariables &vars =
167 mRdlLightFilter->getSceneClass().getSceneContext()->getSceneVariables();
168 const bool mb = vars.get(scene_rdl2::rdl2::SceneVariables::sEnableMotionBlur) &&
169 !isEqual(local2Render0, local2Render1);
170
171 if (mb) {
172 if (!updateTransforms(local2Render1, /* ti = */ 1)) return false;
173
174 // set specific mb bits
175 if (!isEqual(mPosition[0], mPosition[1]))
176 mMb |= LIGHTFILTER_MB_TRANSLATION;
177 if (!isEqual(mOrientation[0], mOrientation[1]))
178 mMb |= LIGHTFILTER_MB_ROTATION;
179 if (!isEqual(mLocal2RenderScale[0], mLocal2RenderScale[1]))
180 mMb |= LIGHTFILTER_MB_SCALE;
181 }
182
183 return true;
184}

186// Similar to CookieLightFilter::computePerspectiveProjectionMatrix()
187math::Mat4f
188BarnDoorLightFilter::getPerspectiveProjectionMatrix()
189{
190 // See Appendix C.5 of the SGI Graphics Library Programming Guide
191 // for format of projection matrix.
192 // http://irix7.com/techpubs/007-1702-020.pdf (pg. 238)
193 // where:
194 // left = bottom = -1
195 // right = top = near = 1
196 // far = inf
197 // and Y and Z columns are negated for barnRotMat.
198 // The Z column (3rd) doesn't matter since the Barn Door only concerns itself
199 // with X and Y values after applying this matrix (see use of mProjector2Screen).
200 return Mat4f(1.f, 0.f, 0.f, 0.f,
201 0.f, -1.f, 0.f, 0.f,
202 0.f, 0.f, 1.f, 1.f,
203 0.f, 0.f, 2.f, 0.f);
204}

226void
227BarnDoorLightFilter::update(const LightFilterMap& /*lightFilters*/,
228 const Mat4d& world2Render)
229{
230 if (!mRdlLightFilter) {
231 return;
232 }
233
234 // Get flap-opening width (mm)
235 const float inputFilmWidth = max(mRdlLightFilter->get(sProjectorWidthKey), 0.00001f);
236
237 // Get flap-opening height (mm)
238 const float inputFilmHeight = max(mRdlLightFilter->get(sProjectorHeightKey), 0.00001f);
239
240 // Get per-edge size adjustments
241 float sizes[4];
242 sizes[BARNDOOR_EDGE_LEFT] = mRdlLightFilter->get<rdl2::Float>(sSizeLeftKey);
243 sizes[BARNDOOR_EDGE_BOTTOM] = mRdlLightFilter->get<rdl2::Float>(sSizeBottomKey);
244 sizes[BARNDOOR_EDGE_RIGHT] = mRdlLightFilter->get<rdl2::Float>(sSizeRightKey);
245 sizes[BARNDOOR_EDGE_TOP] = mRdlLightFilter->get<rdl2::Float>(sSizeTopKey);
246
247 // Determine sizes
248 float effectiveWidth = inputFilmWidth + sizes[BARNDOOR_EDGE_LEFT] + sizes[BARNDOOR_EDGE_RIGHT];
249 float effectiveHeight = inputFilmHeight + sizes[BARNDOOR_EDGE_BOTTOM] + sizes[BARNDOOR_EDGE_TOP];
250 float smallestEdge = min(effectiveWidth, effectiveHeight);
251
252 // Convert radius from relative ([0,1]) to absolute size
253 float relativeRadius = mRdlLightFilter->get(sRadiusKey, 0.f);
254 mRadius = relativeRadius * (smallestEdge * 0.5f);
255
256 // Convert edge from relative ([0,1]) to absolute size
257 float relativeEdge = mRdlLightFilter->get(sEdgeKey, 0.f);
258 mEdge = relativeEdge * smallestEdge;
259
260 // Compute inverse edge scales
261 float edgeScaleLeft = mRdlLightFilter->get<rdl2::Float>(sEdgeScaleLeftKey);
262 float edgeScaleBottom = mRdlLightFilter->get<rdl2::Float>(sEdgeScaleBottomKey);
263 float edgeScaleRight = mRdlLightFilter->get<rdl2::Float>(sEdgeScaleRightKey);
264 float edgeScaleTop = mRdlLightFilter->get<rdl2::Float>(sEdgeScaleTopKey);
265
266 mReciprocalEdgeScales[BARNDOOR_EDGE_LEFT] = rcp(max(1e-9f, mRadius + mEdge * edgeScaleLeft));
267 mReciprocalEdgeScales[BARNDOOR_EDGE_BOTTOM] = rcp(max(1e-9f, mRadius + mEdge * edgeScaleBottom));
268 mReciprocalEdgeScales[BARNDOOR_EDGE_RIGHT] = rcp(max(1e-9f, mRadius + mEdge * edgeScaleRight));
269 mReciprocalEdgeScales[BARNDOOR_EDGE_TOP] = rcp(max(1e-9f, mRadius + mEdge * edgeScaleTop));
270
271 // Get focal length (mm). Will use as distance to projector plane
272 mFocalDist = mRdlLightFilter->get(sProjectorFocalDistKey, 0.f);
273 mFocalDist = max(mFocalDist, 0.1f);
274
275 // Setup mProjector2Screen
276 mProjector2Screen = Mat4f(math::one);
277 switch (mRdlLightFilter->get(sProjectorTypeKey)) {
278 case PERSPECTIVE:
279 mProjector2Screen = getPerspectiveProjectionMatrix();
280 break;
281 case ORTHOGRAPHIC:
282 mProjector2Screen = getOrthoProjectionMatrix();
283 break;
284 default:
285 MNRY_ASSERT(false);
286 }
287
288 // Make radius inscribe box: reduce width and height by 2X radius
289 Vec2f filmSize(effectiveWidth - 2.f * mRadius,
290 effectiveHeight - 2.f * mRadius);
291
292 // Calculate center
293 Vec2f center = Vec2f(sizes[BARNDOOR_EDGE_RIGHT] - sizes[BARNDOOR_EDGE_LEFT ],
294 sizes[BARNDOOR_EDGE_TOP ] - sizes[BARNDOOR_EDGE_BOTTOM]) * 0.5f;
295
296 // Precompute screen-space aperture corners (at virtual focal length of 1)
297 mMinCorner = center - filmSize * 0.5f;
298 mMaxCorner = center + filmSize * 0.5f;
299
300 // A note about rotations
301 //
302 // Most lights (Spot, Disk, Rect, Sphere, Distant) have a 180 degree
303 // rotation about X in their transform, called sRotateX180 which makes the
304 // light spaces' +z direction point -z in render space. That rotation is
305 // baked into the light's xform used by the barn door when
306 // mUseLightXform==true.
307 //
308 // When the barn door is not bound to a light, we apply an equal rotation,
309 // called barnRotMat below, instead. That way the barn door will point in
310 // the same direction regardless of mUseLightXform if the light's and barn
311 // door's node_xforms match.
312 //
313 // Unfortunately that means:
314 //
315 // 1) barnRotMat is applied to barn doors on lights that don't use a
316 // baked rotation, namely Cyl, Mesh, and Env, and the filter will not
317 // point in the same direction as those lights' mDirection. To re-aim
318 // the barn door in those cases, either rotate the light or unbind the
319 // filter and set its node_xform. I matched the bound/unbound barn
320 // door light filter rotations in the RaTS test by rotating the
321 // lights:
322 //
323 // Cyl: rotate(180,1,0,0)
324 // Mesh: rotate(180,1,0,0)
325 // Env: EnvLight::sLocalOrientation.inverse() * rotate(180,1,0,0)
326 // = rotate(-90,0,1,0) * rotate(90,1,0,0) * rotate(180,1,0,0)
327 // = rotate(-90,0,1,0) * rotate(-90,1,0,0)
328 //
329 // The Env rotation was found by solving this eqn:
330 //
331 // sLocalOrientation * EnvLight.xform = barnRotMat * BarnDoor.xform
332 //
333 // for EnvLight.xform:
334 //
335 // EnvLight.xform = sLocalOrientation.inverse() * barnRotMat * BarnDoor.xform
336 //
337 // 2) The projection matrices had to be inverted in Y and Z directions
338 // (as noted in their comments) in order for the filter not to appear
339 // backwards or upside down.
340 //
341
342 // Apply twist rotation about the filter's direction
343 float rotAngle = math::degreesToRadians(mRdlLightFilter->get<rdl2::Float>(sRotationKey));
344 Mat4f twistRotMat = Mat4f::rotate(Vec3f(0, 0, 1), rotAngle);
345 mProjector2Screen = twistRotMat * mProjector2Screen;
346
347 // This should match Light::sRotateX180
348 Mat4d barnRotMat = Mat4d::rotate(Vec3d(1, 0, 0), math::sPi);
349
350 // local space is projector space
351 Mat4d local2World0 = barnRotMat * mRdlLightFilter->get(sProjectorXformKey, 0.f);
352 Mat4d local2World1 = barnRotMat * mRdlLightFilter->get(sProjectorXformKey, 1.f);
353
354 Mat4f local2Render0 = toFloat(local2World0 * world2Render);
355 Mat4f local2Render1 = toFloat(local2World1 * world2Render);
356
357 if (!updateParamAndTransforms(local2Render0, local2Render1)) {
358 return;
359 }
360
361 mMode = mRdlLightFilter->get<rdl2::Int>(sModeKey);
362 mUseLightXform = mRdlLightFilter->get<rdl2::Bool>(sUseLightXformKey);
363 mPreBarnMode = mRdlLightFilter->get<rdl2::Int>(sPreBarnModeKey);
364 mPreBarnDist = mRdlLightFilter->get<rdl2::Float>(sPreBarnDistKey);
365 mDensity = clamp(mRdlLightFilter->get<rdl2::Float>(sDensityKey), 0.f, 1.f);
366 mInvert = mRdlLightFilter->get<rdl2::Bool>(sInvertKey);
367 mColor = mRdlLightFilter->get<rdl2::Rgb>(sColorKey);
368}

370bool
371BarnDoorLightFilter::canIlluminate(const CanIlluminateData& data) const
372{
373
374 // compute barn intersection in projector space
375 Xform3f r2l = mUseLightXform ?
376 data.lightRender2LocalXform : getXformRender2Local(data.time);
377
378 // z distance from projector
379 float shadingDist = transformPoint(r2l, data.shadingPointPosition).z;
380
381 // No illumination behind barn
382 if (shadingDist < 0.f) {
383 return false;
384 }
385
386 if (shadingDist < mPreBarnDist) {
387 // The shading point is behind the projector.
388 switch (mPreBarnMode) {
389 case BLACK:
390 return false;
391 case WHITE:
392 return true;
393 case DEFAULT:
394 default:
395 break; // Do nothing
396 }
397 }
398
399 return true;
400}

404 // Compute a (Gaussian) blur fall off value
405 //
406 // Input: Distance t in the range [0,1]
407 // Return: Fall off value, decreasing from 1 to 0 as t goes from 0 to 1.
408 //
409 // To calculate the blur value we convolve a step function f, where:
410 // f(x) = 1 for x <= 0 (the interior of the BarnDoor portal) and
411 // f(x) = 0 for x > 0 (the exterior of the BarnDoor portal)
412 // with g, a blur kernel chosen to be the Gaussian normal distribution PDF.
413 //
414 // To evaluate the blur value we center the blur kernel
415 // at x=t and integrate f(x)*g(x-t) where they overlap. Because f=1 where they
416 // overlap, this amounts to just integrating g, which is given by its CDF.
417
418 // The CDF of a normal function can be found here:
419 // https://en.wikipedia.org/wiki/Normal_distribution
420 float gaussianCDF(float x) {
421
422 // We choose the variance = sigma^2 = 0.1 to make the PDF approach zero
423 // for |x|>1 so that the blur radius is 1.
424 const float variance = 0.1f;
425
426 return 0.5f * (1.f + erf( x / ( sqrtf(variance) * sqrtf(2.f) ) ) );
427 }
428
429 // Unfortunately, the normal distribution has infinite extent which is
430 // inconvenient because we want the blur to have finite radius.
431 //
432 // Therefore we clip the domain from [-inf,inf] to [-1,1]. That region
433 // is finite and contains 99.8% of the area under the PDF, but not 100%.
434 // As a result the CDF range is about [0.0008, 1-0.0008] instead of [0,1].
435 //
436 // Without compensation this will produce noticeable clipping at the beginning
437 // and end of the blur region. To compensate we expand the CDF slightly.
438 float gaussianBlur(float t) {
439
440 // Remap input from [0, 1] to [1, -1] for input to the CDF
441 // We turn the CDF around so it decreases (instead of increases)
442 // and scale it horizontally to fit our desired domain.
443 t = 1.f - 2.f * t;
444
445 // Evaluate the CDF
446 float val = gaussianCDF(t);
447
448 // Scale and offset the CDF so values at output to expand its range
449 // from ~[0.0008, 0.9992] to [0,1].
450 // (gcc and icc compile minVal to a constant with -O2 or -O3)
451 float minVal = gaussianCDF(-1.f); // 0.000786
452 return val * (1.f + 2.f * minVal) - minVal;
453 }

456Color
457BarnDoorLightFilter::eval(const EvalData& data) const
458{
459 // get render to projector space xform
460 Xform3f r2l = mUseLightXform ?
461 data.lightRender2LocalXform : getXformRender2Local(data.time);
462
463 // shading point in projector space
464 Vec3f pt = transformPoint(r2l, data.shadingPointPosition);
465
466 // z distance from projector
467 float shadingDist = pt.z;
468
469 if (shadingDist < mPreBarnDist) {
470 // The shading point is behind the projector, we don't need to evaluate the
471 // barn door.
472 switch (mPreBarnMode) {
473 case BLACK:
474 return sBlack;
475 case WHITE:
476 return sWhite;
477 case DEFAULT:
478 default:
479 break; // Do nothing
480 }
481 }
482
483 // In analytic mode we use the projection of the shading point directly into screen space.
484
485 // For physical mode we intersect the light ray with the focal plane,
486 // then project that point into screen space.
487 if (mMode == PHYSICAL) {
488
489 // Find intersection of the light ray with the focal plane
490 //
491 // pt = data.shadingPointPosition
492 // wi = direction of light sample from pt
493 // po = projectorPos = (0,0,0)
494 // pd = projectorDir = (0,0,1)
495 // sv = shadingVec = pt - po
496 // f = distance to from projector center to focal plane
497 // X = intersection of wi with focal plane
498 // t = distance along wi from pt to X
499 //
500 // Points along ray from pt in direction wi:
501 // X = pt + t * wi
502 //
503 // Points in the projector focal plane:
504 // dot(X - po, pd) = f
505 //
506 // Substitute ray equation into plane equation and solve for t:
507 // dot(pt + t * wi - po, pd) = f
508 // dot(pt - po + t * wi, pd) = f
509 // dot(sv + t * wi, pd) = f
510 // dot(sv, pd) + dot(t * wi, pd) - f = 0
511 // dot(sv, pd) - f = -dot(t * wi, pd)
512 // dot(sv, pd) - f = t * -dot(wi, pd)
513 // (dot(sv, pd) - f) / -dot(wi, pd) = t
514 // (shadingDist - f) / -wi.z = t
515 // (f - shadingDist) / wi.z = t
516 //
517 // Then plug t into ray equation to find intersection point X
518 Vec3f wi = transformVector(r2l, data.wi); // light direction
519 float t = (mFocalDist - shadingDist) / wi.z;
520 pt = pt + t * wi;
521 MNRY_ASSERT(isEqual(pt.z, mFocalDist)); // true if things went as planned
522 }
523
524 // We need the screen space position of the render space point.
525 Vec3f screenP = transformH(mProjector2Screen, pt); // projection using a homogenous divide
526
527 // Convert to 2D
528 Vec2f shadingPoint(screenP.x, screenP.y);
529
530 // Find point of inner rectangle closest to shading point
531 Vec2f clampedShadingPoint = clamp(shadingPoint, mMinCorner, mMaxCorner);
532
533 // Take the difference
534 Vec2f diffVec = shadingPoint - clampedShadingPoint;
535
536 // Choose the scales for the relevant corner
537 Vec2f reciprocalScales(
538 mReciprocalEdgeScales[diffVec.x < 0.f ? BARNDOOR_EDGE_LEFT : BARNDOOR_EDGE_RIGHT],
539 mReciprocalEdgeScales[diffVec.y < 0.f ? BARNDOOR_EDGE_BOTTOM : BARNDOOR_EDGE_TOP ]);
540
541 // Inversely scale according to ellipse dimensions
542 Vec2f scaledVec = diffVec * reciprocalScales;
543
544 // Compute the 3 distances we need:
545
546 // Distance from the inner box to the first (inner)
547 // rounded-rectangle-with-circular-corners (the edge start).
548 float r0 = mRadius;
549
550 // Distance from the inner box to shading point.
551 float r = length(diffVec);
552
553 // Distance from the inner box to the second (outer)
554 // rounded-rectangle-with-ellipse-corners (the edge end).
555 // The distance to the ellipse is related to r by a ratio. That
556 // ratio can be computed in the frame of a unit circle, since we know the
557 // distance to the circle in that frame. Also the ratio is preserved during
558 // transformation since it's affine. Then we just apply the ratio to r.
559 float r1 = r / length(scaledVec);
560
561 // Compute scale value
562 float scale = 0.f; // Exterior, r >= r1. 0 means no light allowed.
563 if (r <= r0) { // Interior
564 scale = 1.f;
565 } else if (r < r1) { // Edge transition
566 float lerpParam = (r - r0) / (r1 - r0);
567 scale = gaussianBlur(lerpParam);
568 }
569
570 // invert
571 if (mInvert) {
572 scale = 1.f - scale;
573 }
574
575 // apply color
576 Color result = mColor * scale;
577
578 // apply density (lerp)
579 result = math::lerp(sWhite, result, mDensity);
580
581 return result;
582}
