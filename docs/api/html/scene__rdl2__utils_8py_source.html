47def CompilerCheckAndExit(self, expected_compiler=None):

54 print colorize(
55 '* ERROR\n'

57 (self['COMPILER_LABEL'], expected_compiler), 'red')
58 check_call("scons -h --skip_default_compiler_check",shell=True)
59 sys.exit(1)

109def runCmd(cmd, blocking=True, quiet=False, quiet_errors=False):
110 """
111 Abstraction for shelling out to run commands
112
113 if not blocking
114 continue execution in non-zero return code cases
115
116 if quiet:
117 block stdout out
118
119 if quiet_errors:
120 hides stderr
121
122 """
123 p = subprocess.Popen(
124 args=cmd,
125 shell=True,
126 stderr=subprocess.PIPE,
127 stdout=subprocess.PIPE
128 )
129 stdout, stderr = p.communicate()
130
131 if not quiet:
132 if stdout:
133 sys.stdout.write(stdout)
134 if not quiet_errors:
135 if stderr:
136 sys.stderr.write(stderr)
137 else:
138 if blocking:
139 raise Exception('%s failed\n' % cmd)
140
141 return p.returncode, stdout, stderr

143def _localizeAction(target, source, env):
144
145 target = target[0].abspath
146 LDD_CMD = "%s %s" % ("ldd", target)
147 output = runCmd(LDD_CMD, quiet=True, quiet_errors=True, blocking=False)
148 result = output[1]
149 lines = result.split("\n")
150 libs = {}
151 for line in lines:
152 match = re.match(r'\t(.*) => (/(rel|work)/.*) \‍(0x', line)
153 if match:
154 libs[match.group(1)] = match.group(2)
155
156 if libs:
157 install_path = os.path.join(env.Dir(env['INSTALL_DIR']).abspath, 'ext')
158 try:
159 os.makedirs(install_path)
160 except OSError as exc:
161 if exc.errno == errno.EEXIST and os.path.isdir(install_path):
162 pass
163 else:
164 raise
165
166 for lib in libs:
167 RSYNC_CMD = 'rsync --chmod a=rx %s %s/%s' % (libs[lib], install_path , lib)
168 runCmd(RSYNC_CMD, quiet=True, quiet_errors=False, blocking=False)

220def GatherProxies(env):
221 """
222 Gather up the paths to any proxy or json class descriptions files in this
223 package so that they can be copied to the proxy package.
224 """
225
226 proxies = {}
227 for var, pat in proxyLocations.items():
228 proxies[var] = env.Glob(pat)
229 env.AppendUnique(**proxies)

231def CopyProxies(env):
232 """
233 Copy any proxies or json class descriptions previously gathered up and copy
234 them to the correct location in the current package.
235 """
236
237 targets = []
238 for var, pat in proxyLocations.items():
239 t = env.DWAInstallFiles(os.path.dirname(pat), env.get(var, []))
240 targets.extend(t)
241 env.NoCache(t)
242 return targets
