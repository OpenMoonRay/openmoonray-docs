71//=====================

73// class SpMask8
74//
75//=====================
76//-----------------------------------------------------------------------------
77//
78// An 8x8 A-buffer bitmask class which wraps a 64-bit unsigned int.
79// It offers convenience methods for x/y bit access and translating
80// to/from floats for I/O.
81//
82// The 8x8 fixed size (64 grey values) is a compromise between per-sample memory
83// use and providing enough resolution to resolve spatial alignments. A larger
84// size such as 16x16 would consume 8 floats per sample vs. just 2 for 8x8, while
85// still not providing enough resolution to perform precision transform filtering.
86// To improve subpixel resolution the 8x8 mask can be used in conjunction with
87// 8 additional bits in the DeepFlags class representing partial subpixel-coverage
88// weight which allows each sample to represent 16384 grey values rather than just
89// 64.
90//
91// The pattern is spatially interpreted with the lowest bit 0 in the
92// lower-left corner and the highest bit 63 in the upper-right corner
93// and always oriented Y-up. This matches most current applications and
94// renderers (c 2016) - even though OpenEXR's pixel coordinate axis is
95// Y-down:
96// ::
97//
98// -------------------
99// 7 |56 . . . . . . 63| +Y
100// 6 | . . . . . . . . |
101// 5 | . . . . . . . . |
102// 4 | . . . . . . . . |
103// 3 | . . . . . . . . |
104// 2 | . . . . . . . . |
105// 1 | . . . . . . . . |
106// 0 | 0 . . . . . . 7 | -Y
107// -------------------
108// -X 0 1 2 3 4 5 6 7 +X
109//
110//-----------------------------------------------------------------------------
111

349inline int SpMask8::bitsOn () const
350{
351 if (m==allBitsOff || m==allBitsOn)
352 return numBits;
353#if 1
354#define BX_(x) ((x) - (((x)>>1)&0x77777777) \
355 - (((x)>>2)&0x33333333) \
356 - (((x)>>3)&0x11111111))
357#define BITCOUNT(x) (((BX_(x)+(BX_(x)>>4)) & 0x0F0F0F0F) % 255)
358 const uint32_t lo = uint32_t(m);
359 const uint32_t hi = uint32_t(m >> 32);
360 return BITCOUNT(lo) + BITCOUNT(hi);
361#else
362 int numOnBits = 0;
363 SpMask8 sp(1ull);
364 for (uint32_t sp_bin=0; sp_bin < numBits; ++sp_bin, ++sp)
365 if (m & sp)
366 ++numOnBits;
367 return numOnBits;
368#endif
369}
370inline int SpMask8::bitsOff() const { return (numBits - bitsOn()); }
371inline /*static*/ void SpMask8::mapXCoord (int inX, int inW, int& outX, int& outR)
372{
373 if (inW == OPENDCX_INTERNAL_NAMESPACE::SpMask8::width)
374 outX = outR = inX;
375 else

381 }
382}

393 }

395inline float SpMask8::toCoverage() const
396{
397 if (m==allBitsOff || m==allBitsOn)
398 return 1.0f; // Zero mask indicates source doesn't have subpixel masks so weight is also 1
399 const int numOnBits = bitsOn();
400 return (numOnBits == 0)?0.0f:float(numOnBits)/float(numBits);
401}
402inline void SpMask8::printPattern (std::ostream& os,
403 const char* prefix) const
404{
405 for (int sy=(int)(SpMask8::width-1); sy >= 0; --sy)
406 {
407 SpMask8 sp(1ull << (sy << 3));
408 os << prefix;
409 for (int sx=0; sx < (int)SpMask8::width; ++sx, ++sp)
410 if (m & sp) os << "1 "; else os << ". ";
411 os << std::endl;
412 }
413}
