24finline void
25setBsdfLabels(const scene_rdl2::rdl2::Material *material,
26 shading::TLState *tls,
27 int numStatev,
28 const shading::Statev *statev,
29 scene_rdl2::rdl2::Bsdfv *bsdfv,
30 const int parentLobeCount)
31{
32 // The Bsdf object holds several different bits of label data used by Aovs.
33 // This data must be set or transformed after shading, prior to integration.
34 //
35 // --
36 // BsdfLobe::mLabel
37 // --
38 // The shader can place labels on the lobes it outputs. These labels,
39 // stored in mLabel, are simple integers that are only unique to the
40 // shader itself. For example, two different shaders will use
41 // mLabel = "1", "2", "3" etc... to mean completely different labels.
42 // Within the shader's extension object is a mapping from the local label
43 // ids of the shader to the global label ids used by the aov system.
44 // There is a separate mapping for material aovs (lobeLabelIds)
45 // and light path expression aovs (lpeLobeLabelIds). This code transforms
46 // BsdfLobe::mLabel into a value that encodes the global lobeLabelId, and
47 // lpeLobeLabelId for the lobe.
48 //
49 // It is safe to call this function multiple times on the same Bsdf. Once
50 // lobes are transformed, the transform operation is a no-op.
51 //
52 // The BsdfLobe::mLabel is a uniform member, so no per-lane processing
53 // is needed.
54 //
55 // --
56 // Bsdf::mGeomLabelId
57 // --
58 // Geometry object's have a user visible "label" attribute. This label
59 // can appear in material aov expressions. The geometry extension object
60 // contains the integer mapping this label to the global material aov system.
61 //
62 // Since the geometry object can vary (but probably often doesn't) per lane,
63 // the geometry id must be explicitly set for each lane.
64 //
65 // It is safe to set this data multiple times from the same Bsdf as long
66 // as the Intersection remains invariant (i.e. the geometry id per lane does not
67 // change).
68
69
70 if (material->hasExtension()) {
71 const shading::Material &ext = material->get<shading::Material>();
72 Bsdfv *shadingBsdfv = reinterpret_cast<Bsdfv *>(bsdfv);
73 for (int i = 0; i < numStatev; ++i) {
74 const int materialLabelId = ext.getMaterialLabelId(); // material aovs
75 const auto &lobeLabelIds = ext.getLobeLabelIds(); // material aovs
76 const int lpeMaterialLabelId = ext.getLpeMaterialLabelId(); // lpe aovs
77 const auto &lpeLobeLabelIds = ext.getLpeLobeLabelIds(); // lpe aovs
78
79 // uniform labels - we have only a single material per bsdf
80 shadingBsdfv[i].mMaterialLabelId = materialLabelId;
81 shadingBsdfv[i].mLpeMaterialLabelId = lpeMaterialLabelId;
82
83 // potentially have a different geometry per-lane
84 // multiple geometry can potentially share the same material
85 const mcrt_common::ConstAddress64v &geometryObject =
86 reinterpret_cast<const mcrt_common::ConstAddress64v &>(statev[i].mGeometryObject);
87 for (unsigned int lane = 0; lane < VLEN; ++lane) {
88 const scene_rdl2::rdl2::Geometry *geometry =
89 static_cast<const scene_rdl2::rdl2::Geometry *>(geometryObject.get(lane));
90 if (geometry && geometry->hasExtension()) {
91 shadingBsdfv[i].mGeomLabelId[lane] =
92 geometry->get<shading::Geometry>().getGeomLabelId();
93 }
94 }
95
96 // Transform the shader local aov labels to global aov label ids.
97 // We only transform the lobe labels of the current material.
98 // Lobes are added in a top down manner, with parent lobes
99 // added before child lobes. Lobe labels, on the other hand,
100 // Are transformed in a bottom up manner, with child labels
101 // transformed before parent labels.
102 // Therefore we must skip the lobes of the parent material here.
103 // If this material has a child material, its lobe labels are
104 // already transformed. Labels that are already transformed
105 // are ignored in aovEncodeLabels.
106 for (int li = parentLobeCount; li < shadingBsdfv[i].mNumLobes; ++li) {
107 BsdfLobev *lobev = shadingBsdfv[i].mLobes[li];
108 lobev->mLabel = aovEncodeLabels(lobev->mLabel,
109 materialLabelId, lpeMaterialLabelId,
110 lobeLabelIds, lpeLobeLabelIds);
111 }
112 Bssrdfv *bssrdfv = shadingBsdfv[i].mBssrdf;
113 if (bssrdfv) {
114 bssrdfv->mLabel = aovEncodeLabels(bssrdfv->mLabel,
115 materialLabelId, lpeMaterialLabelId,
116 lobeLabelIds, lpeLobeLabelIds);
117 }
118 VolumeSubsurfacev *vsv = shadingBsdfv[i].mVolumeSubsurface;
119 if (vsv) {
120 vsv->mLabel = aovEncodeLabels(vsv->mLabel,
121 materialLabelId, lpeMaterialLabelId,
122 lobeLabelIds, lpeLobeLabelIds);
123 }
124 }
125 }
126}
