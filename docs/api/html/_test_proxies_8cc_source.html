31void
32TestProxies::testProxyDwaBaseLayerable()
33{
34 // This tests the ability of the INTERFACE_DWABASELAYERABLE flag to
35 // work properly in proxy mode (see MOONRAY-2047).
36
37 // Create the context, class, and object in proxy mode.
38 SceneContext context;
39 context.setProxyModeEnabled(true);
40 const SceneClass* sc = context.createSceneClass("LibLadenDwaBaseLayerable");
41 Material *mat = context.createSceneObject("LibLadenDwaBaseLayerable", "/seq/shot/layerMat")->asA<Material>();
42 Material *matA = context.createSceneObject("LibLadenDwaBaseLayerable", "/seq/shot/layerMatA")->asA<Material>();
43
44 // Set an attribute specific to the proxied type.
45 AttributeKey<SceneObject *> key = sc->getAttributeKey<rdl2::SceneObject *>("mat A");
46 mat->beginUpdate();
47 mat->set(key, matA);
48 mat->endUpdate();
49
50 // Write it out.
51 BinaryWriter writer(context);
52 writer.toFile("proxyDwaBaseLayerable.rdlb");
53
54 // Read it back in.
55 SceneContext verifyContext;
56 verifyContext.setProxyModeEnabled(true);
57 BinaryReader reader(verifyContext);
58 reader.fromFile("proxyDwaBaseLayerable.rdlb");
59
60 // Verify the attribute.
61 Material *verifyMat = static_cast<rdl2::Material *>
62 (verifyContext.getSceneObject("/seq/shot/layerMat")->asA<Material>());
63 CPPUNIT_ASSERT(verifyMat);
64 Material *verifyMatA = static_cast<rdl2::Material *>
65 (verifyContext.getSceneObject("/seq/shot/layerMatA")->asA<Material>());
66 CPPUNIT_ASSERT(verifyMatA);
67 sc = verifyContext.getSceneClass("LibLadenDwaBaseLayerable");
68 key = sc->getAttributeKey<rdl2::SceneObject *>("mat A");
69 CPPUNIT_ASSERT(verifyMat->get(key) == verifyMatA);
70}
