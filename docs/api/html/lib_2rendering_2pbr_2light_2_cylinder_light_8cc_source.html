52bool
53CylinderLight::update(const Mat4d& world2render)
54{
55 MNRY_ASSERT(mRdlLight);
56
57 mOn = mRdlLight->get(scene_rdl2::rdl2::Light::sOnKey);
58 if (!mOn) {
59 return false;
60 }
61
62 updateVisibilityFlags();
63 updatePresenceShadows();
64 updateRayTermination();
65 updateTextureFilter();
66 updateMaxShadowDistance();
67
68 const Mat4d l2w0 = mRdlLight->get(scene_rdl2::rdl2::Node::sNodeXformKey, /* rayTime = */ 0.f);
69 const Mat4d l2w1 = mRdlLight->get(scene_rdl2::rdl2::Node::sNodeXformKey, /* rayTime = */ 1.f);
70 mLocalRadius = mRdlLight->get<scene_rdl2::rdl2::Float>(sRadiusKey);
71 mLocalHalfHeight = mRdlLight->get<scene_rdl2::rdl2::Float>(sHeightKey) * 0.5f;
72
73 UPDATE_ATTRS_CLEAR_RADIUS
74
75 const Mat4f local2Render0 = toFloat(l2w0 * world2render);
76 const Mat4f local2Render1 = toFloat(l2w1 * world2render);
77 if (!updateParamAndTransforms(local2Render0, local2Render1, mLocalHalfHeight, 1.0f)) {
78 return false;
79 }
80
81 // Compute render space quantities
82 // TODO: mb scale radiance
83 mActualRadius = xformLocal2RenderScale(mLocalRadius, 0.f);
84 mRcpActualRadius = 1.0f / mActualRadius;
85 mActualRadiusSquared = mActualRadius * mActualRadius;
86 mActualHalfHeight = xformLocal2RenderScale(mLocalHalfHeight, 0.f);
87 mArea = sFourPi * mActualRadius * mActualHalfHeight;
88 mInvArea = 1.0f / mArea;
89
90 // Compute radiance.
91 mRadiance = computeLightRadiance(mRdlLight, scene_rdl2::rdl2::Light::sColorKey,
92 scene_rdl2::rdl2::Light::sIntensityKey, scene_rdl2::rdl2::Light::sExposureKey, sNormalizedKey,
93 sApplySceneScaleKey, mInvArea);
94 if (isBlack(mRadiance)) {
95 mOn = false;
96 return false;
97 }
98
99 if (!updateImageMap(Distribution2D::PLANAR)) {
100 return false;
101 }
102
103 // Sidedness
104 mSidedness = static_cast<LightSidednessType>(mRdlLight->get<rdl2::Int>(sSidednessKey));
105 return true;
106}

149// Computing a tight bounding box for a cylinder light
150// ---------------------------------------------------
151//
152// See the comments for DiskLight::getBounds() for explanation of tight bounds for a disk.
153// Bounding the cross-section of a cylinder works in the same way, with a couple of minor
154// modifications - the axis of rotational symmetry is the local y-axis for a cylinder, where
155// it's the local z-axis for a disk; and also a non-unit mLocalRadius is supported for a
156// cylinder, where a disk light has radius 1 in local space.
157//
158// The method for a disk is augmented by sweeping the disk along the local y-axis so that
159// it covers the interval [-mLocalHalfHeight, mLocalHalfHeight] in the local y-direction.
160// In order to bound the resulting volume in render space, to the half-dimesions for disk
161// bounds we add the vector h*(M10,M11,M12) where h is the half-height and (Mij) represents
162// the local2render matrix.
163
164BBox3f
165CylinderLight::getBounds() const
166{
167 const float r = mLocalRadius;
168 const float h = mLocalHalfHeight;
169 const Vec3f pos = getPosition(0.f);
170 const Vec3f xRow = mLocal2Render[0].l.vx;
171 const Vec3f yRow = mLocal2Render[0].l.vy;
172 const Vec3f zRow = mLocal2Render[0].l.vz;
173 const Vec3f halfDims = r * scene_rdl2::math::sqrt(xRow*xRow + zRow*zRow) + h * scene_rdl2::math::abs(yRow);
174 BBox3f bounds(pos - halfDims, pos + halfDims);
175 if (isMb()) {
176 const Vec3f pos = getPosition(1.f);
177 const Vec3f xRow = mLocal2Render[1].l.vx;
178 const Vec3f yRow = mLocal2Render[1].l.vy;
179 const Vec3f zRow = mLocal2Render[1].l.vz;
180 const Vec3f halfDims = r * scene_rdl2::math::sqrt(xRow*xRow + zRow*zRow) + h * scene_rdl2::math::abs(yRow);
181 bounds.extend(BBox3f(pos - halfDims, pos + halfDims));
182 }
183 return bounds;
184}

187// Ray/cylinder intersection test
188// ------------------------------
189//
190// Define a ray in the usual way:
191// p = origin
192// wi = direction
193//
194// The ray's parametric equation is x = p + t*wi
195//
196// Define a cylinder as follows:
197// x0 = centre (i.e. midpoint along axis between the 2 end planes)
198// e = unit vector along axis (in either direction)
199// r = radius
200// h = half-height, measured along axis from centre to either end plane
201//
202// The implicit equation of the cylinder's quadric surface is
203//
204// |cross(x-x0, e)|^2 = r^2
205//
206// Substituting the ray's parametric equation, we obtain a quadratic in t:
207//
208// |cross(t*wi - dp, e)|^2 = r^2
209//
210// where dp = x0 - p.
211// Now define vectors u = wi x e and v = dp x e. The quadratic in t
212// expands as follows:
213//
214// (u.u)t^2 - 2(u.v)t + (v.v)-r^2 = 0
215//
216// Or
217//
218// a*t^2 - 2b*t + c = 0
219//
220// with a=u.u, b=u.v, c=v.v-r^2, the discriminant is
221//
222// D = b^2-a*c
223//
224// We can reduce the work required to compute D as follows:
225//
226// D = (u.v)^2 - (u.u)((v.v)-r^2)
227// = (u.v)^2 - (u.u)(v.v) + (u.u)r^2
228// = (u.u)r^2 - |u x v|^2
229//
230// Noting that u.u = a, and also that
231//
232// u x v = (wi x e) x (dp x e)
233// = (wi . (dp x e)) e - (e . (dp x e)) wi
234// = - ((wi x e).dp) e
235// = - (u.dp) e
236//
237// we have
238//
239// D = a r^2 - (u.dp)^2
240//
241// We reject rays for which D is negative. Otherwise we proceed to solve the
242// quadratic equation in the usual way:
243//
244// t = (b +/- sqrt(D)) / a
245//
246// where we take the smaller root for regular cylinder lights and the larger for ones with
247// reverse sidedness. The computed intersection point can then be tested for inclusion between
248// the 2 end planes.
249//
250// Considering the cylinder as the intersection of an infinite cylinder with a slab defined
251// by the end planes, trivial rejection can be done in a couple of easy ways.
252//
253// Trivial rejection based on the ray's configuration relative to the slab works by rejecting
254// when the origin is outside the slab and the direction is pointing away from it.
255// i.e reject if (wi.e>0 && dp.e>h) || (wi.e<0 && dp.e<-h)
256//
257// Trivial rejection based on considering the infinite cylinder is similar to the rejection
258// done for SphereLight. It's based on 2 conditions:
259// - whether the ray's origin is inside the cylinder;
260// - whether the ray's direction faces away from the cylinder (strictly, from the plane through
261// the cylinder's axis perpendicular to the line joining the ray's origin to the cylinder axis).
262// These 2 conditions are combined differently depending on the sidedness of the light.
263//
264// The algebra of the rejection tests is simplified by defining the following:
265// A = dp.e, B = wi.e, C = dp.wi
266//
267// These are also used to simplify computing the quadratic coefficients a,b,c:
268// a = u^2 = (wi x e)^2 = 1 - (wi.e)^2 = 1 - B^2
269// b = u.v = (wi x e) . (dp x e) = [wi x e, dp, e] = ((wi x e) x dp) . e = ((wi.dp)e - (e.dp)wi) . e
270// = wi.dp - (e.dp)(wi.e) = C - AB
271// c = v^2 - r^2 = dp^2 e^2 - (dp.e)^2 - r^2 = dp^2 - A^2 - r^2
272//
273// We can use the following conditions to trivially reject certain cases, depending on the light's sidedness:
274// b<0, ray pointing away from cylinder axis (strictly, away from a plane through the axis perpendicular to the
275// plane containing both the axis and the illumination point)
276// c<0, ray origin inside (infinite) cylinder
277// D<0, no real roots, i.e. ray misses the (infinite) cylinder
278
279bool
280CylinderLight::intersect(const Vec3f &p, const Vec3f *n, const Vec3f &wi, float time,
281 float maxDistance, LightIntersection &isect) const
282{
283 const Vec3f e = xformVectorLocal2RenderRot(Vec3f(0.f, 1.f, 0.f), time);
284 const Vec3f dp = getPosition(time) - p;
285
286 // Trivial rejection if ray is outside slab and travelling away from it
287 const float A = dot(dp, e);
288 const float B = dot(wi, e);
289 if (A * sign(B) <= -mActualHalfHeight) {
290 return false;
291 }
292
293 const float C = dot(dp, wi);
294 const float b = C - A*B;
295 const float c = lengthSqr(dp) - A*A - mActualRadiusSquared;
296
297 // Do trivial rejection for the following cases:
298 // regular sidedness: reject if ray is traveling away from axis OR ray origin is INSIDE cylinder
299 // reverse sidedness: reject if ray is traveling away from axis AND ray origin is OUTSIDE cylinder
300 // (two-sided is not supported)
301 // c > 0 when ray origin is outside cylinder
302 // b <= 0 when ray is travelling away from cylinder axis
303 if ((c > 0.0f) ? (b <= 0.0f) : (mSidedness == LightSidednessType::LIGHT_SIDEDNESS_REGULAR)) {
304 return false;
305 }
306
307 // Compute discriminant; reject unless positive
308 float a = 1.0f - B*B;
309 const float D = b * b - a * c;
310 if (D <= 0.0f) {
311 return false;
312 }
313
314 // Compute intersection t-value along ray
315 // Note: a=0 is rejected simply to protect against generating a NaN. It's perfectly fine for 'a' to be very
316 // small and overflow the result to +/-inf becuse the t range test will reject it.
317 const float s = scene_rdl2::math::sqrt(D);
318 const float t = (mSidedness == LightSidednessType::LIGHT_SIDEDNESS_REGULAR) ? (b - s) / a : (b + s) / a;
319 if (a == 0.0f || t < 0.0f || t > maxDistance) {
320 return false;
321 }
322
323 // Compute intersection point relative to cylinder centre; check it's between end planes
324 const float y = t * B - A;
325 if (scene_rdl2::math::abs(y) > mActualHalfHeight) {
326 return false;
327 }
328
329 // Fill in isect members
330 const Vec3f hitPoint = t * wi - dp;
331 // We don't need to worry about 2-sided here because that's not supported for CylinderLight
332 isect.N = ((mSidedness == LightSidednessType::LIGHT_SIDEDNESS_REGULAR) ? mRcpActualRadius : -mRcpActualRadius)
333 * (hitPoint - y * e);
334 isect.uv = mDistribution ? local2uv(xformVectorRender2Local(hitPoint, time)) : scene_rdl2::math::zero;
335 isect.distance = t;
336
337 return true;
338}

341bool
342CylinderLight::sample(const Vec3f &p, const Vec3f *n, float time, const Vec3f& r,
343 Vec3f &wi, LightIntersection &isect, float rayDirFootprint) const
344{
345 MNRY_ASSERT(mOn);
346
347 // Sample the image distribution if any, otherwise sample the cylinder
348 // uv parameterization directly.
349 Vec2f uv;
350 if (mDistribution) {
351 mDistribution->sample(r[0], r[1], 0, &uv, nullptr, mTextureFilter);
352 isect.uv = uv;
353 } else {
354 uv = Vec2f(r[0], r[1]);
355 isect.uv = Vec2f(zero);
356 }
357
358 const Vec3f localHit = uv2local(uv);
359 const Vec3f renderHit = xformPointLocal2Render(localHit, time);
360
361 // Compute wi and d
362 wi = renderHit - p;
363 isect.distance = length(wi);
364 if (isect.distance < sEpsilon) {
365 return false;
366 }
367 wi *= rcp(isect.distance);
368 if (n && dot(*n, wi) < sEpsilon) {
369 return false;
370 }
371
372 const Vec3f localN = Vec3f(localHit.x, 0.0f, localHit.z);
373 const Vec3f renderN = normalize(xformNormalLocal2Render(localN, time));
374 // We don't need to worry about 2-sided here because that's not supported for CylinderLight
375 isect.N = (mSidedness == LightSidednessType::LIGHT_SIDEDNESS_REGULAR) ? renderN : -renderN;
376
377 // Reject if the generated point is on the back-facing surface of the cylinder
378 if (dot(isect.N, wi) > -sEpsilon) {
379 return false;
380 }
381
382 return true;
383}

386Color
387CylinderLight::eval(mcrt_common::ThreadLocalState* tls, const Vec3f &wi, const Vec3f &p, const LightFilterRandomValues& filterR,
388 float time, const LightIntersection &isect, bool fromCamera, const LightFilterList *lightFilterList,
389 float rayDirFootprint, float *pdf) const
390{
391 MNRY_ASSERT(mOn);
392
393 // Point sample the texture
394 // TODO: Use proper filtering with ray differentials and mip-mapping.
395 Color radiance = mRadiance;
396
397 if (mDistribution) {
398 radiance *= mDistribution->eval(isect.uv[0], isect.uv[1], 0, mTextureFilter);
399 }
400
401 if (lightFilterList) {
402 evalLightFilterList(lightFilterList,
403 { tls, &isect, getPosition(time),
404 getDirection(time), p,
405 filterR, time,
406 getXformRender2Local(time, lightFilterList->needsLightXform()),
407 wi
408 },
409 radiance);
410 }
411
412 if (pdf) {
413 *pdf = mInvArea;
414 if (mDistribution) {
415 *pdf *= mDistribution->pdf(isect.uv[0], isect.uv[1], 0, mTextureFilter);
416 }
417 *pdf *= areaToSolidAngleScale(wi, isect.N, isect.distance);
418 }
419
420 return radiance;
421}
