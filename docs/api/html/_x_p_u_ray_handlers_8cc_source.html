111 // Create the BundledRadiance objects as required based on the occlusion

170 // it would already have been filled by the previous call.
171 if (fs.mAovSchema->hasLpePrefixFlags(AovSchema::sLpePrefixUnoccluded)) {
172 accumLightAovs(pbrTls, occlRay, fs, numItems, scene_rdl2::math::sWhite, nullptr,
173 AovSchema::sLpePrefixUnoccluded);
174 }
175 }
176 }
177
178 // LPE
179 // we are responsible for freeing LPE memory
180 if (occlRay.mDataPtrHandle != nullHandle) {
181 pbrTls->freeList(occlRay.mDataPtrHandle);
182 }
183 pbrTls->releaseDeepData(occlRay.mDeepDataHandle);
184 }
185
186 return numRadiancesFilled;

191void
192xpuOcclusionQueryBundleHandlerGPU(mcrt_common::ThreadLocalState *tls,
193 unsigned numRays,
194 BundledOcclRay *rays,
195 const rt::GPUOcclusionRay *gpuRays,
196 tbb::spin_mutex& mutex)
197{
198 pbr::TLState *pbrTls = tls->mPbrTls.get();
199
200 EXCL_ACCUMULATOR_PROFILE(pbrTls, EXCL_ACCUM_OCCL_QUERY_HANDLER);
201
202 scene_rdl2::alloc::Arena *arena = &tls->mArena;
203 SCOPED_MEM(arena);
204
205 BundledRadiance *results = arena->allocArray<BundledRadiance>(numRays, CACHE_LINE_SIZE);
206
207 unsigned numRadiancesFilled =
208 computeXPUOcclusionQueriesOnGPU(tls,
209 numRays,
210 rays,
211 gpuRays,
212 results,
213 mutex);
214
215 MNRY_ASSERT(numRadiancesFilled <= numRays);
216
217 CHECK_CANCELLATION(pbrTls, return);
218
219 pbrTls->addRadianceQueueEntries(numRadiancesFilled, results);
220}

222void
223xpuOcclusionQueryBundleHandlerCPU(mcrt_common::ThreadLocalState *tls,
224 unsigned numRays,
225 const BundledOcclRay *rays,
226 void *userData)
227{
228 pbr::TLState *pbrTls = tls->mPbrTls.get();
229
230 EXCL_ACCUMULATOR_PROFILE(pbrTls, EXCL_ACCUM_OCCL_QUERY_HANDLER);
231
232 scene_rdl2::alloc::Arena *arena = &tls->mArena;
233 SCOPED_MEM(arena);
234
235 RayHandlerFlags handlerFlags = RayHandlerFlags((uint64_t)userData);
236 MNRY_ASSERT(handlerFlags <= RAYHANDLER_FLAGS_MASK);
237
238 BundledRadiance *results = arena->allocArray<BundledRadiance>(numRays, CACHE_LINE_SIZE);
239
240 // We need an array of pointers to entries, not of the entries themselves.
241 BundledOcclRay** entries = arena->allocArray<BundledOcclRay*>(numRays, CACHE_LINE_SIZE);
242 for (int i = 0; i < numRays; i++) {
243 entries[i] = const_cast<BundledOcclRay*>(rays + i);
244 }
245
246 // We just call the regular CPU vector mode code.
247 unsigned numRadiancesFilled = computeOcclusionQueriesBundled(pbrTls,
248 numRays,
249 entries,
250 results,
251 handlerFlags);
252
253 MNRY_ASSERT(numRadiancesFilled <= numRays);
254
255 CHECK_CANCELLATION(pbrTls, return);
256
257 pbrTls->addRadianceQueueEntries(numRadiancesFilled, results);
258}
