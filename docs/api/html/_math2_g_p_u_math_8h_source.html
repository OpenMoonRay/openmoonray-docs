17inline GPUXform
18mat43ToGPUXform(const geom::Mat43& mat)
19{
20 // Note that Mat43 is a 4x3 matrix with the translation as the bottom row,
21 // while GPUXform is a 3x4 matrix with the translation as the last column.
22 // The core 3x3 part of the matrix is transposed.
23 GPUXform xf;
24 xf.m[0][0] = mat.row0().x;
25 xf.m[0][1] = mat.row1().x;
26 xf.m[0][2] = mat.row2().x;
27 xf.m[1][0] = mat.row0().y;
28 xf.m[1][1] = mat.row1().y;
29 xf.m[1][2] = mat.row2().y;
30 xf.m[2][0] = mat.row0().z;
31 xf.m[2][1] = mat.row1().z;
32 xf.m[2][2] = mat.row2().z;
33 // translation part
34 xf.m[0][3] = mat.row3().x;
35 xf.m[1][3] = mat.row3().y;
36 xf.m[2][3] = mat.row3().z;
37
38 return xf;
39}
