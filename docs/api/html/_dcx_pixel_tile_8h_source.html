70//-----------------------------------------------------------------------------
71//
72// Abstract class to manage a rectangular region of pixels.
73//
74// This class is intended for simplifing common pixel-region loops that
75// step from pixel to pixel processing multiple channels simultaneously,
76// vs. using the IlmImfUtil classes which appear organized more for
77// per-plane processing and targeted primarily for texture use.
78//
79//-----------------------------------------------------------------------------

247inline IMATH_NAMESPACE::Box2i PixelTile::flipY(const IMATH_NAMESPACE::Box2i& bbox) const {
248 IMATH_NAMESPACE::Box2i flipped(bbox);
249 if (m_yaxis_up) {
250 flipped.max.y = m_top_reference - bbox.min.y;
251 flipped.min.y = m_top_reference - bbox.max.y;
