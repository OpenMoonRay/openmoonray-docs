97 void initialize()
98 {
99 // MOONRAY-4193 - https://jira.dreamworks.net/browse/MOONRAY-4193
100 // We should not enter this function twice but it is happening
101 // in the AssignmentAndXformSetter in GeometryManager.cc
102 // MNRY_ASSERT(!isInitialized());
103 if (isInitialized()) {
104 return;
105 }
106 if ((mUninitializedData.mSample0 == mUninitializedData.mSample1) ||
107 (mSampleStart == 0.0f && mSampleDelta == 0.0f)) {
108 mState = State::INITIALIZED_STATIC;
109 mStaticData = StaticData(mUninitializedData.mSample0);
110 } else {
111 scene_rdl2::math::XformComponent3f sample0;
112 decompose(mUninitializedData.mSample0, sample0);
113 scene_rdl2::math::XformComponent3f sample1;
114 decompose(mUninitializedData.mSample1, sample1);
115 correctSlerpAngle(sample0, sample1);
116 float t0 = mSampleStart;
117 float t1 = t0 + mSampleDelta;
118 if (scene_rdl2::math::isEqual(t0, t1)) {
119 mState = State::INITIALIZED_STATIC;
120 mStaticData = StaticData(
121 scene_rdl2::math::slerp(sample0, sample1, t0).combined());
122 } else {
123 mState = State::INITIALIZED_MOTION;
124 mMotionData = MotionData(sample0, sample1);
125 }
126 }
127 }
128
129 bool isInitialized() const { return mState != State::UNINITIALIZED; }
130
131 bool isStatic() const { return mState == State::INITIALIZED_STATIC; }
132
133 bool isMotion() const { return mState == State::INITIALIZED_MOTION; }

135private:
136 void correctSlerpAngle(scene_rdl2::math::XformComponent3f& c0,
137 scene_rdl2::math::XformComponent3f& c1) {
138 // avoid flipping rotation by checking the angle between two transforms
139 // (one rotation matrix can be represented by two opposite quaternions)
140 if (scene_rdl2::math::dot(c0.r, c1.r) < 0.0f) {
141 c1.r = -c1.r;
142 }
143 }

152 class UninitializedData {
153 public:
154 UninitializedData(const shading::XformSamples& xformSamples) {
155 size_t xformSamplesCount = xformSamples.size();
156 if (xformSamplesCount == 1) {
157 mSample0 = xformSamples[0];
158 mSample1 = xformSamples[0];
159 } else if (xformSamplesCount == 2) {
160 mSample0 = xformSamples[0];
161 mSample1 = xformSamples[1];
162 } else {
163 MNRY_ASSERT(false,
164 "we only support at most two xform samples now");
165 }
166 }
167
168 void appendXform(const shading::XformSamples& xformSamples) {
169 size_t xformSamplesCount = xformSamples.size();
170 if (xformSamplesCount == 1) {
171 mSample0 = mSample0 * xformSamples[0];
172 mSample1 = mSample1 * xformSamples[0];
173 } else if (xformSamplesCount == 2) {
174 mSample0 = mSample0 * xformSamples[0];
175 mSample1 = mSample1 * xformSamples[1];
176 } else {
177 MNRY_ASSERT(false,
178 "we only support at most two xform samples now");
179 }
180 }
181
182 public:
183 Mat43 mSample0;
184 Mat43 mSample1;
185 };
