45 // has its own issues: namely that we use two values (and go through two states) for generating our double

115 // Get a random int in [0, limit)
116 std::uint32_t getNextUInt(std::uint32_t limit) noexcept
117 {
118 // Several methods were evaluated for this function, all of which use
119 // some sort of rejection sampling to make this unbiased.
120 //
121 // https://crypto.stackexchange.com/a/5709
122 // Where we reject if our random uint is greater than or equal to
123 // floor(max()/limit) * limit. Fast, but uses integer division and
124 // mod.
125 //
126 // https://crypto.stackexchange.com/a/5721
127 // A modification of the above, which pools entropy to reduce the
128 // number of rejections. However, this was slower than above.
129 //
130 // Likewise, trying to use most of the bits of the returned value on
131 // failures ended up being slower than just getting a new random int.
132 //
133 // https://crypto.stackexchange.com/a/7998
134 // This is the same as the first listed, but with more detail. We are
135 // using this method with the smallest range possible for rejection,
136 // which allows us to eliminate integer division and modular
137 // arithmetic, which is slow in SIMD. In scalar code, this is slightly
138 // slower than using the largest range possible with a division and
139 // mod.
140 //
141 MNRY_ASSERT(limit > 0);
142 const std::uint32_t bits = solveForExponent(limit);
143 const std::uint32_t mask = ~((~0u) << bits);
144
145 while (true) {
146 const auto r = getNextUInt() & mask;
147 if (r < limit) {
148 return r;
149 }
150 }
151 }
152
153 result_type operator()() noexcept
154 {
155 return getNextUInt();
156 }

158 // Multi-step advance function (jump-ahead, jump-back)
159 //
160 // The method used here is based on Brown, "Random Number Generation
161 // with Arbitrary Stride", Transactions of the American Nuclear
162 // Society (Nov. 1994). The algorithm is very similar to fast
163 // exponentiation.
164 void advance(std::int64_t delta) noexcept
165 {
166 std::uint64_t curMult = s_mult;
167 std::uint64_t curPlus = mStream;
168 std::uint64_t accMult = 1u;
169 std::uint64_t accPlus = 0u;
170
171 // Even though delta is an unsigned integer, we can pass a signed
172 // integer to go backwards, it just goes "the long way round".
173 auto deltaU = static_cast<uint64_t>(delta);
174
175 while (deltaU > 0) {
176 if (deltaU & 1) {
177 accMult *= curMult;
178 accPlus = accPlus * curMult + curPlus;
179 }
180 curPlus = (curMult + 1) * curPlus;
181 curMult *= curMult;
182 deltaU /= 2;
183 }
184 mState = accMult * mState + accPlus;
185 }

220 // Solve for the smallest u such than 2^u >= v
221 static std::uint32_t solveForExponent(std::uint32_t v) noexcept
222 {
223 const float f = static_cast<float>(2u * v - 1u);
224 const std::uint32_t u = *reinterpret_cast<const std::uint32_t*>(&f);
225 return (u >> 23u) - 0x7Fu;
226 }

229 {
230 MNRY_ASSERT(mStream % 2 == 1);
231 return state * s_mult + mStream;

239 static std::uint32_t output(std::uint64_t input) noexcept
240 {
241 constexpr std::uint8_t bits = 64u;
242 constexpr std::uint8_t xtypebits = 32u;
243 constexpr std::uint8_t sparebits = 32u;
244 constexpr std::uint8_t opbits = 5u;
245 constexpr std::uint8_t mask = (1u << opbits) - 1u;
246 constexpr std::uint8_t topspare = opbits;
247 constexpr std::uint8_t bottomspare = sparebits - topspare;
248 constexpr std::uint8_t xshift = (topspare + xtypebits)/2u;
249 const std::uint8_t rot = (input >> (bits - opbits)) & mask;
250 input ^= input >> xshift;
251 const std::uint32_t amprot = static_cast<std::uint32_t>(rot & mask);
252 const std::uint32_t result = std::uint32_t(input >> bottomspare);
253 return util::rotateRight(result, amprot);
254 }

298// The standard allows explicit full specialization within the namespace. We
299// can do a Bernoulli distribution faster, since we already generate a double
300// in [0, 1) very quickly. In my tests, this was more than three times as fast
301// as not specializing the distribution.
302//
303// The Bernoulli distribution can be used anywhere where you want to generate
304// boolean values with a certain probability. Multiple Bernoulli trials lead
305// toa binomial distribution.
306namespace std {
307template <>
308inline bool bernoulli_distribution::operator()<scene_rdl2::util::Random>(scene_rdl2::util::Random& g)
309{
310 return g.getNextDouble() < this->p();
311}
