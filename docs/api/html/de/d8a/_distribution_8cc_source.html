169float
170GuideDistribution1D::tabulateCdf()
171{
172 // Find sum of weights
173 float sum = 0.0f;
174 for (size_type i = 0; i < mSizeCdf; i++) {
175 MNRY_ASSERT(mCdf[i] >= 0.0f);
176 sum += mCdf[i];
177 }
178
179 // Input function is represented as weights stored in the elements of mCdf.
180 // Accumulate weights to make a homogeneous cdf.
181 // (Regular cdf can be recovered by multiplying by mInvTotalWeight.)
182 if (sum != 0.0f) {
183 float partialSum = 0.0f;
184 for (size_type i=0; i < mSizeCdf-1; i++) {
185 partialSum += mCdf[i];
186 if (partialSum > sum) {
187 // This is unlikely to occur, but it is theoretically possible due to rounding (because of differences
188 // in object code generated for this loop versus the one that computed sum)
189 partialSum = sum;
190 }
191 mCdf[i] = partialSum;
192 }
193 } else {
194 // Cater to edge case of all-zero weights
195 for (size_type i = 0; i < mSizeCdf-1; i++) {
196 mCdf[i] = (i + 1) * mInvSizeCdf;
197 }
198 sum = 1.0f;
199 }
200
201 // Store sum as final table entry. (Note that we don't include this line in the loop that builds the cdf,
202 // becuse that loop actually produces a slightly different final sum due to differences in object code generated.)
203 mCdf[mSizeCdf-1] = sum;
204
205 // Store total weight and its reciprocal
206 mTotalWeight = sum;
207 mInvTotalWeight = 1.0f / sum;
208
209 // Compute guide table
210 size_type i = 0;
211 for (size_type j=0; j < mSizeGuide; ++j) {
212
213 // Find min i such that cdf[i] >= j * mTotalWeight / mSizeGuide
214 const float tau = j * mTotalWeight * mInvSizeGuide;
215 const size_type iBound = mSizeCdf - 1;
216 while (i < iBound && mCdf[i] < tau) {
217 ++i;
218 }
219 mGuide[j] = i;
220 }
221
222 // Set helper values for sampleLinear() function.
223 // The cdf curve is piecewise quadratic, but it has linear segments at the left and right ends.
224 // The values here are for the early-outs in sampleLinear() that deal with these linear cases.
225 // mThresholdLow is the threshold value which the input random variable r must not exceed for it to fall within
226 // the first half-texel, for which a linear segment is used. The gradient of that segment is then stored in
227 // mLinearCoeffLow for easy evaluation in that case.
228 // The values mThresholdHigh and mLinearCoeffHigh are used in a corresponding way for the final half-texel.
229 mThresholdLow = 0.5f * mCdf[0] * mInvTotalWeight;
230 mThresholdHigh = (mSizeCdf > 1) ? 0.5f * (mCdf[mSizeCdf-2] + mCdf[mSizeCdf-1]) * mInvTotalWeight
231 : mThresholdLow;
232 mLinearCoeffLow = (mThresholdLow != 0.0f) ? 0.5f * mInvSizeCdf / mThresholdLow : 0.0f;
233 mLinearCoeffHigh = (mThresholdHigh != 1.0f) ? 0.5f * mInvSizeCdf / (1.0f - mThresholdHigh) : 0.0f;
234
235 // Return the integral of the input function
236 return sum * mInvSizeCdf;
237}

258float
259GuideDistribution1D::pdfLinear(float u) const
260{
261 if (u < 0.0f || u > 1.0f) return 0.0f;
262 if (mSizeCdf == 1) return 1.0f;
263
264 int i;
265 float t;
266
267 // Here we must add 0.5 to the texture coordinate (after scaling it to be in units of texels). This is because
268 // we're evaluating the texture as a piecewise-linear function, with a new linear segment for each texel, but in
269 // particular the segments join at the texel centers, whose positions have a fractional part of 0.5. Thus we want
270 // to advance the index i whenever the fractional part reaches 0.5. (Compare this with the situation for nearest
271 // neighbour textures, where the segments join a the texel boundaries, whose positions have a fractional part of
272 // 0.0.)
273 intAndFrac(u * mSizeCdf + 0.5f, &i, &t);
274
275 float p;
276 if (i == 0) {
277 // Clamp to lower border
278 p = mCdf[0];
279 }
280 else if (i == mSizeCdf) {
281 // Clamp to upper border
282 p = mCdf[mSizeCdf-1] - mCdf[mSizeCdf-2];
283 }
284 else {
285 // Lerp between 2 values
286 float p0 = (i == 1) ? mCdf[0] : mCdf[i-1] - mCdf[i-2];
287 float p1 = mCdf[i] - mCdf[i-1];
288 p = lerp(p0, p1, t);
289 }
290
291 return p * mInvTotalWeight * mSizeCdf;
292}

295// Returns the index of the interval in the discrete distribution where the value r falls.
296// rRemapped represents the fractional distance of r along that interval.
297GuideDistribution1D::size_type
298GuideDistribution1D::sampleDiscrete(float r, float *const pdf, float *const rRemapped) const
299{
300 size_type i = getGuideIndex(r);
301 float rw = r * mTotalWeight; // Scale here because cdf is non-normalized
302 while (mCdf[i] < rw) {
303 ++i;
304 }
305 MNRY_ASSERT(i < mSizeCdf);
306 const float prevCdf = (i > 0) ? mCdf[i - 1] : 0.0f;
307 const float funcVal = mCdf[i] - prevCdf;
308 if (pdf) *pdf = funcVal * mInvTotalWeight;
309 if (rRemapped) *rRemapped = (rw - prevCdf) / funcVal;
310 return i;
311}

314float
315GuideDistribution1D::sampleContinuous(float r, float *pdf, size_type *index) const
316{
317 size_type i = getGuideIndex(r);
318 float rw = r * mTotalWeight; // Scale here because cdf is non-normalized
319 while (mCdf[i] < rw) {
320 ++i;
321 }
322 MNRY_ASSERT(i < mSizeCdf);
323
324 if (index) *index = i;
325
326 const float prevCdf = (i > 0) ? mCdf[i-1] : 0.0f;
327 const float funcVal = mCdf[i] - prevCdf;
328
329 // Lerp inverted cdf value
330 // (Note here that an expression of the form lerp(i,i+1,t) has been replaced with the simpler and more
331 // efficient expression i+t)
332 float invCdf = (float(i) + (rw - prevCdf) / funcVal) * mInvSizeCdf;
333
334 // May reach 1.0 due to float imprecision
335 invCdf = min(invCdf, sOneMinusEpsilon);
336
337 if (pdf) *pdf = funcVal * mInvTotalWeight * mSizeCdf;
338
339 return invCdf;
340}

343// Helper function to solve for the parameter t in [0,1] where a given quadratic curve segment takes on the
344// given value k. (2*k is actually passed in.)
345//
346// The calling code will have looked up 3 consecutive cdf values, c0, c1, c2. We are interested in the curve
347// segment covering the range of values from (c0+c1)/2 to (c1+c2)/2, corresponding to the interval between
348// texel centers.
349//
350// The curve segment is a quadratic Bezier curve with control points (c0+c1)/2, c1, (c1+c2)/2. It is easy to
351// show that this has expression q(t) = a*t^2 + 2b*t + c, with
352// a = ((c2 - c1) - (c1-c0)) / 2
353// b = (c1 - c0) / 2
354// c = (c0 + c1) / 2
355//
356// We wish to solve for t the equation q(t) = k, or Q(t) = A*t^2 + 2*B*t + C = 0, with
357// A = (c2 - c1) - (c1 - c0)
358// B = c1 - c0
359// C = c0 + c1 - 2k
360//
361// This is easy to solve via an application of the quadratic formula. The quadratic has 2 roots and we want to
362// return a particular one - the lower root, if Q(t) is convex upwards (A > 0), and the upper root if it is
363// convex downwards (A < 0). Note that it can also degenerate into a linear equation when all 3 points c0, c1, c2
364// lie in a straight line (A = 0).
365//
366// If A > 0, the upper root must be (-B + sqrt(discrim)) / A
367// And if A < 0, the lower root must be (-B + sqrt(discrim)) / A, i.e. the same expression.
368// Since B >= 0, -B and sqrt(discrim) have opposite signs, so to avoid possible catastrophic cancellation
369// (which will happen if A*C is close to zero), we swith to alternate expression C / (-B - sqrt(discrim))
370// in both cases.
371//
372// Notice that this conveniently caters to the degenerate linear case, too, since if A = 0 the equation
373// reduces to 2*B*t + C = 0 with solution -C/(2B), and since sqrt(discrim) will equal B, this is exactly
374// the solution generated.
375//
376// Note also that the discriminant is evaluated as B^2-AC, rather than the probably more familiar B^2-4AC, and there's
377// a corresponding removal of a factor 2 in the final result (the denominator 2A -> A for the "standard" form, or the
378// numerator 2C -> C for the form used below). This is because the factor 2 has been explicitly pulled out of the
379// linear term of Q(t) = A*t^2 + 2*B*t + C = 0, allowing the various cancellations to ensue.
380
381inline float
382quadraticHelper(float c0, float c1, float c2, float twoK, float &p)
383{
384 float p0 = c1 - c0;
385 float p1 = c2 - c1;
386
387 float A = p1 - p0;
388 float B = p0;
389 float C = c0 + c1 - twoK;
390
391 float discrim = B*B - A*C;
392 float denom = -B - scene_rdl2::math::sqrt(discrim);
393
394 float t = (denom != 0.0f) ? C / denom : 0.0f;
395 p = lerp(p0, p1, t);
396
397 return t;
398}

401// Sampling from a distribution which is a linearly-interpolated 1D texture. The cdf in this case is made up
402// of a tangent-continuous set of quadratic Bezier curves, with linear portions in the first and last half-texel.
403// The reason it is piecewise quadratic is that the cdf is the integral of the pdf, which in this case is a
404// piecewise-linear function. And the reason the cdf is linear over the first and last half-texels is that the pdf
405// is clamped to a constant border value over those intervals.
406// We proceed as in the nearest-neighbour case by finding the texel span which the given r-value intersects, and
407// then computing the instersection.
408
409float
410GuideDistribution1D::sampleLinear(float r, float *pdf) const
411{
412 MNRY_ASSERT(r >= 0.0f && r <= 1.0f);
413
414 // Linear interpolation in 1st half-texel
415 if (r <= mThresholdLow) {
416 return min(r * mLinearCoeffLow, sOneMinusEpsilon);
417 }
418
419 // Linear interpolation in last half-texel
420 if (r >= mThresholdHigh) {
421 return min(1.0f - (1.0f - r) * mLinearCoeffHigh, sOneMinusEpsilon);
422 }
423
424 // Look up in guide table.
425 // The guide table entries are optimal for nearest-neighbour textures, not for linear ones. It would be
426 // possible to add in a second table, but the added memory cost and code support would outweigh the very
427 // slight performance gain, so we just use the existing table.
428 int i = getGuideIndex(r);
429 MNRY_ASSERT(i < mSizeCdf);
430
431 // Early-out in 1st linear case above means we don't need to search starting at i=0
432 i = max(i, 1);
433
434 float twoWr = 2.0f * mTotalWeight * r;
435
436 // Step index until we find the curve segment which r intersects with
437 float c0 = (i >= 2) ? mCdf[i-2] : 0.0f;
438 float c1 = mCdf[i-1];
439 float c2 = mCdf[i];
440 while (c1 + c2 < twoWr) // We use precomputed 2*wr to avoid multiplying c1+c2 here by 0.5.
441 {
442 ++i;
443 MNRY_ASSERT(i < mSizeCdf);
444 c0 = c1;
445 c1 = c2;
446 c2 = mCdf[i];
447 }
448
449 // Solve for intersection with selected curve segment.
450 // The curve segments are necessarily quadratic, because the cdf is the integral of the pdf
451 // which in this case is piecewise linear.
452 float p;
453 float t = quadraticHelper(c0, c1, c2, twoWr, p);
454
455 if (pdf) *pdf = p * mInvTotalWeight * mSizeCdf;
456
457 // Return location which is a fraction t between texel centers i, i+1
458 return min((static_cast<float>(i) + 0.5f + t) * mInvSizeCdf, sOneMinusEpsilon);
459}

496
497void
498Distribution2D::tabulateCdf(Mapping mapping)
499{
500 size_type sizeU = getSizeU();
501 size_type sizeV = getSizeV();
502
503 // Modify weights to respect mappings right before we compute the final CDF.
504 switch (mapping) {
505 case SPHERICAL: {
506 // Note: We are slighly pulling in the extents so we don't get full black for
507 // the top and bottom scanlines
508 float sclV, ofsV;
509 getScaleOffset(-0.5f, float(sizeV) - 0.5f, 0.0f, sPi, &sclV, &ofsV);
510 tbb::parallel_for(tbb::blocked_range<size_type>(0, sizeV, sizeV / sRangeDivider),
511 [&](const tbb::blocked_range<size_type> range) {
512 for (size_type y = range.begin(); y < range.end(); ++y) {
513 // The sinTheta term accounts for the distortion of the image mapping
514 // onto the sphere into the sampling pdf: two pixels of equal brightness
515 // should not be sampled similarly if one is close to the sphere pole
516 // and the other close to the equator (see pbrt section 14.6.5)
517 float sinTheta = scene_rdl2::math::sin(float(y) * sclV + ofsV);
518 for (size_type x = 0; x < sizeU; ++x) {
519 scaleWeight(x, y, max(sinTheta, sEpsilon));
520 }
521 }
522 });
523 break;
524 }
525
526 case HEMISPHERICAL: {
527 // Note: We are slighly pulling in the extents so we don't get full black for
528 // the top and bottom scanlines
529 float sclV, ofsV;
530 getScaleOffset(-0.5f, float(sizeV) - 0.5f, 0.0f, sPi, &sclV, &ofsV);
531 unsigned halfSizeV = sizeV / 2;
532
533 tbb::parallel_for(tbb::blocked_range<size_type>(0, sizeV, sizeV / sRangeDivider),
534 [&](const tbb::blocked_range<size_type> range) {
535 for (size_type y = range.begin(); y < range.end(); ++y) {
536 // The sinTheta term accounts for the distortion of the image mapping
537 // onto the sphere into the sampling pdf: two pixels of equal brightness
538 // should not be sampled similarly if one is close to the sphere pole
539 // and the other close to the equator (see pbrt section 14.6.5)
540 float sinTheta = (y >= halfSizeV) ? scene_rdl2::math::sin(float(y) * sclV + ofsV) : sEpsilon;
541 for (size_type x = 0; x < sizeU; ++x) {
542 scaleWeight(x, y, sinTheta);
543 }
544 }
545 });
546 break;
547 }
548
549 case PLANAR: {
550 // Nothing to do!
551 break;
552 }
553
554 case CIRCULAR: {
555 // Weights outside of the maximal inscribed circle are set to black so
556 // we don't (or rarely) sample them.
557 // note: We are slighly pushing out the extents so we don't get full
558 // black for the edge scanlines.
559 float sclU, ofsU, sclV, ofsV;
560 getScaleOffset(-0.5f, float(sizeU) - 0.5f, -1.0f, 1.0f, &sclU, &ofsU);
561 getScaleOffset(-0.5f, float(sizeV) - 0.5f, -1.0f, 1.0f, &sclV, &ofsV);
562
563 tbb::parallel_for(tbb::blocked_range<size_type>(0, sizeV, sizeV / sRangeDivider),
564 [&](const tbb::blocked_range<size_type> range) {
565 for (size_type y = range.begin(); y < range.end(); ++y) {
566 float t = float(y) * sclV + ofsV;
567 float t2 = t * t;
568 for (size_type x = 0; x < sizeU; ++x) {
569 float s = float(x) * sclU + ofsU;
570 if (s*s + t2 > 1.0f) {
571 setWeight(x, y, sEpsilon);
572 }
573 }
574 }
575 });
576 break;
577 }
578 default: {
579 MNRY_ASSERT(false, "Undefined mapping in Distribution2D::tabulateCdf()");
580 }
581 }
582
583 // Now compute CDF.
584 tbb::parallel_for(tbb::blocked_range<size_type>(0, sizeV, sizeV / sRangeDivider),
585 [&](const tbb::blocked_range<size_type> range) {
586 for (size_type y = range.begin(); y < range.end(); ++y) {
587 float integral = mConditional[y]->tabulateCdf();
588 mMarginal->setWeight(y, integral);
589 }
590 });
591 mMarginal->tabulateCdf();
592}

595float
596Distribution2D::pdfNearest(float u, float v) const
597{
598 size_type vSize = mMarginal->getSize();
599 size_type vIndex = size_type(v * vSize);
600 vIndex = scene_rdl2::math::min(vIndex, vSize - 1);
601
602 return mMarginal->pdfContinuous(v) *
603 mConditional[vIndex]->pdfContinuous(u);
604}

609{
610 if (u < 0.0f || u >= 1.0f || v < 0.0f || v >= 1.0f) return 0.0f;
611
612 float pMarg = mMarginal->pdfContinuous(v);
613
614 int n = mMarginal->getSize();

618 intAndFrac(v * n + 0.5f, &i1, &s);
619 int i0 = i1 - 1;
620
621 i0 = scene_rdl2::math::max(i0, 0);
622 i1 = scene_rdl2::math::min(i1, n-1);
623
624 float pCond0 = mConditional[i0]->pdfContinuous(u);
625 float pCond1 = mConditional[i1]->pdfContinuous(u);
626
627 float pCond = lerp(pCond0, pCond1, s);
628
629 return pMarg * pCond;
630}

646// A helper function that works like GuideDistribution1D::sampleLinear() except that it linearly blends
647// the coefficients of 2 cdfs (presumably 2 adjacent rows of the conditional cdf table) by the parameter s
648float
649sampleLinearBlended(float r, float *pdf, const GuideDistribution1D *cond0, const GuideDistribution1D *cond1, float s)
650{
651 MNRY_ASSERT(r >= 0.0f && r <= 1.0f);
652
653 // Linear interpolation in 1st half-texel
654 if (r <= lerp(cond0->getThresholdLow(), cond1->getThresholdLow(), s)) {
655 return min(r * lerp(cond0->getLinearCoeffLow(), cond1->getLinearCoeffLow(), s), sOneMinusEpsilon);
656 }
657
658 // Linear interpolation in last half-texel
659 if (r >= lerp(cond0->getThresholdHigh(), cond1->getThresholdHigh(), s)) {
660 return min(1.0f - (1.0f - r) * lerp(cond0->getLinearCoeffHigh(), cond1->getLinearCoeffHigh(), s),
661 sOneMinusEpsilon);
662 }
663
664 // Look up in guide tables
665 int i = min(cond0->getGuideIndex(r), cond1->getGuideIndex(r));
666 MNRY_ASSERT(i < cond0->getSize());
667
668 // Early-out in 1st linear case above means we don't need to search starting at i=0
669 i = max(i, 1);
670
671 float w = lerp(cond0->getTotalWeight(), cond1->getTotalWeight(), s);
672 float twoWr = 2.0f * w * r;
673 const float *cdf0 = cond0->getCdf();
674 const float *cdf1 = cond1->getCdf();
675
676 // Step index until we find the curve segment which r intersects with
677 float c0 = (i >= 2) ? lerp(cdf0[i-2], cdf1[i-2], s) : 0.0f;
678 float c1 = lerp(cdf0[i-1], cdf1[i-1], s);
679 float c2 = lerp(cdf0[i], cdf1[i], s);
680 const int n = cond0->getSize();
681 while (c1 + c2 < twoWr) {
682 ++i;
683 if (i >= n) {
684 // When the true sample result is very close to the boundary between 2 table entries, rounding can
685 // cause i to step too high by 1, and if this happens at the top end of the array it will be out of
686 // bounds. If this case is detected we simply fall back on the linear interpolation in the uppermost
687 // half-texel.
688 return min(1.0f - (1.0f - r) * lerp(cond0->getLinearCoeffHigh(), cond1->getLinearCoeffHigh(), s),
689 sOneMinusEpsilon);
690 }
691 c0 = c1;
692 c1 = c2;
693 c2 = lerp(cdf0[i], cdf1[i], s);
694 }
695
696 // Solve for intersection with selected curve segment.
697 // The curve segments are necessarily quadratic, because the cdf is the integral of the pdf
698 // which in this case is piecewise linear.
699 float p;
700 float t = quadraticHelper(c0, c1, c2, twoWr, p);
701
702 if (pdf) *pdf = p * cond0->getInvSize() / w;
703
704 // Return location which is a fraction t between texel centers i, i+1
705 return min((static_cast<float>(i) + 0.5f + t) * cond0->getInvSize(), sOneMinusEpsilon);
706}

709void
710Distribution2D::sampleBilinear(float ru, float rv, scene_rdl2::math::Vec2f *uv, float *pdf) const
711{
712 MNRY_ASSERT(uv != nullptr);
713
714 float u, v, pMarg, pCond;
715
716 v = mMarginal->sampleLinear(rv, &pMarg);
717
718 int n = mMarginal->getSize();
719
720 int i1;
721 float s;
722 intAndFrac(v * n, &i1, &s);
723 int i0 = i1 - 1;

725 i0 = scene_rdl2::math::max(i0, 0);
726 i1 = scene_rdl2::math::min(i1, n-1);
727
728 u = sampleLinearBlended(ru, &pCond, mConditional[i0], mConditional[i1], s);
729
730 uv->x = u;
731 uv->y = v;
732 if (pdf) *pdf = pMarg * pCond;
733}

780void
781ImageDistribution::init(const std::string &mapFilename,
782 Distribution2D::Mapping mapping,
783 const Color& gamma,
784 const Color& contrast,
785 const Color& saturation,
786 const Color& gain,
787 const Color& offset,
788 const Vec3f& temperature,
789 const float rotationAngle,
790 const Vec2f& translation,
791 const Vec2f& coverage,
792 const float repsU,
793 const float repsV,
794 const bool mirrorU,
795 const bool mirrorV,
796 const Color& borderColor)
797{
798 // Get unique_ptr to ImageInput - cleanup will be handled automatically
799 rndr::OiioUtils::ImageInputUqPtr in = rndr::OiioUtils::openFile(mapFilename);
800
801 // Throw exception if 'in' is null
802 if (in == nullptr) {
803 throw scene_rdl2::except::IoError("Image file \"" + mapFilename + "\" does not exist. Using black radiance");
804 }
805
806 // Get ImageSpec
807 OIIO::ImageSpec inputImageSpec = in->spec();
808
809 const int simdWidth = ispc::PBR_getSIMDWidth();
810 float* inputPixelBuffer;
811 std::vector<float *> pixelBuffers;
812
813 // We'll need to check each level dims against those of the level above.
814 // (It doesn't matter what we initialise these values to.)
815 int widthLevelAbove = 0;
816 int heightLevelAbove = 0;
817
818 // Loop over mip levels present in the file
819 for (int mipLevel = 0; ; ++mipLevel) {
820
821 // Seek mip level
822 if ( ! in->seek_subimage(0, mipLevel, inputImageSpec)) {
823 // mip level not found
824 break;
825 }
826
827 const int inputLevelWidth = inputImageSpec.width;
828 const int inputLevelHeight = inputImageSpec.height;
829
830 if (inputLevelWidth < 1 || inputLevelHeight < 1) break;
831
832 // Temp workaround: 1x1 mip levels cause problems.
833 // TODO: solve the problems and remove this workaround!
834 if (inputLevelWidth == 1 && inputLevelHeight == 1) {
835 Logger::warn("1x1 mip level not supported at present");
836 break;
837 }
838
839 const int numPixels = inputLevelWidth * inputLevelHeight;
840
841 // Note: we are not making any assumptions about the presence or absence of padding in the image itself;
842 // we simply require the memory buffer to be a multiple of simdWidth so that simd writes work correctly
843 const int paddedNumPixels = (numPixels + simdWidth - 1) & -simdWidth;
844
845 if (mipLevel == 0) {
846 // Record info for base level
847 mWidth = inputLevelWidth;
848 mHeight = inputLevelHeight;
849
850 // Allocate buffer for base level - all smaller levels can reuse it
851 inputPixelBuffer = scene_rdl2::util::alignedMallocArray<float>(paddedNumPixels * inputImageSpec.nchannels);
852 } else {
853 // For smaller levels, verify the size is what we expect.
854 // In the usual case, the dimensions should be exactly half those of the level above.
855 // However, occasionally the level above has an odd dimension, in which case we should allow the
856 // smaller size to be either rounded down or rounded up.
857 bool widthOK = false;
858 if (inputLevelWidth == widthLevelAbove >> 1) {
859 // Standard case, or odd level above getting rounded down
860 widthOK = true;
861 }
862 if (widthLevelAbove & 1) {
863 if (inputLevelWidth == (widthLevelAbove >> 1) + 1) {
864 // Odd level above getting rounded up
865 widthOK = true;
866 }
867 }
868
869 bool heightOK = false;
870 if (inputLevelHeight == heightLevelAbove >> 1) {
871 // Standard case, or odd level above getting rounded down
872 heightOK = true;
873 }
874 if (heightLevelAbove & 1) {
875 if (inputLevelHeight == (heightLevelAbove >> 1) + 1) {
876 // Odd level above getting rounded up
877 heightOK = true;
878 }
879 }
880
881 if (!widthOK || !heightOK) {
882 Logger::warn("Unexpected mip level dimensions in texture file.");
883 break;
884 }
885 }
886
887 // Use current dims to check dims of next level on the next pass
888 widthLevelAbove = inputLevelWidth;
889 heightLevelAbove = inputLevelHeight;
890
891 // Read image in floating point format
892 bool readOk = in->read_image(OIIO::TypeDesc::FLOAT, inputPixelBuffer);
893
894 // Error if it wasn't read properly
895 if (!readOk) {
896 scene_rdl2::util::alignedFreeArray<float> (inputPixelBuffer);
897 inputPixelBuffer = nullptr;
898 throw scene_rdl2::except::IoError("Cannot read image file: \"" + mapFilename +
899 "\" (" + in->geterror() + ")");
900 }
901
902 // Need to change the input format spec since we asked OIIO to convert it to float
903 inputImageSpec.format = OIIO::TypeDesc::FLOAT;
904
905 // The input image might not have 3 channels; specify 3 channels for the final image spec
906 OIIO::ImageSpec finalImageSpec = inputImageSpec;
907 finalImageSpec.nchannels = 3;
908
909 // Allocate 3-channel pixel buffer to permanently hold the texture data for this mip level
910 float *pixelBuffer = scene_rdl2::util::alignedMallocArray<float>(paddedNumPixels * 3);
911 std::fill(pixelBuffer, pixelBuffer + paddedNumPixels * 3, 0.0f);
912 pixelBuffers.push_back(pixelBuffer);
913
914 // Corresponding OIIO image buffers
915 OIIO::ImageBuf inputImageBuf("input", inputImageSpec, inputPixelBuffer);
916 OIIO::ImageBuf finalImageBuf("final", finalImageSpec, pixelBuffer);
917
918 // Flip the image (our frame buffers are upside down compared to what OIIO is expecting).
919 // Set flipped output to 3 channel 32-bit floating point color.
920 OIIO::ImageBufAlgo::flip(finalImageBuf, inputImageBuf);
921 }
922
923 // Clean up
924 scene_rdl2::util::alignedFreeArray<float> (inputPixelBuffer);
925 inputPixelBuffer = nullptr;
926 in->close();
927
928 // Exit if we failed to find the base mip level
929 mNumMipLevels = pixelBuffers.size();
930 if (mNumMipLevels == 0) {
931 throw scene_rdl2::except::IoError("Cannot seek into image file: \"" + mapFilename +
932 "\" (" + in->geterror() + ")");
933 }
934
935 // Set up array of pointers, one to each mip level's pixel buffer
936 mPixelBuffer = scene_rdl2::util::alignedMallocArray<float *>(mNumMipLevels);
937 memcpy(mPixelBuffer, pixelBuffers.data(), mNumMipLevels * sizeof(float *));
938
939 // We'll also need a corresponding distribution for each mip level
940 mDistribution = scene_rdl2::util::alignedMallocArray<Distribution2D *>(mNumMipLevels);
941
942 // Set flag to say whether there's a non-trivial transformation
943 mIsTransformed = (rotationAngle != 0.0f ||
944 translation != Vec2f(zero) ||
945 coverage != Vec2f(one) ||
946 repsU != 1.0f ||
947 repsV != 1.0f);
948
949 // Compute affine transformation, unless it's the identity.
950 // Note that transforming a texture by a given matrix involves applying the inverse of that matrix
951 // to the texture coordinates. The forward transformation is
952 // rotate(angle) * scale(factor) * translate(vector)
953 // and so the inverse translation would be
954 // translate(-vector) * scale(1/factor) * rotate(-angle)
955 // and this is the matrix we build here. The scale factor is computed as coverage/reps, so we take the reciprocal
956 // of that. Also note that the minus sign is missing from the rotation angle because the clockwise rotation
957 // requested by the artists is already mathematically negative.
958 if (mIsTransformed) {
959 // translation
960 float tx = -translation.x;
961 float ty = -translation.y;
962
963 // scale
964 float sx = repsU / coverage.x;
965 float sy = repsV / coverage.y;
966
967 // rotation
968 float theta = rotationAngle * (sPi / 180.0f);
969 float c = scene_rdl2::math::cos(theta);
970 float s = scene_rdl2::math::sin(theta);
971
972 // mat = trn * scl * rot
973 //
974 // (1 0 0) (sx 0 0) ( c s 0)
975 // = (0 1 0) (0 sy 0) (-s c 0)
976 // (tx ty 1) (0 0 1) ( 0 0 1)
977 //
978 // (1 0 0) ( c*sx s*sx 0)
979 // = (0 1 0) (-s*sy c*sy 0)
980 // (tx ty 1) ( 0 0 1)
981 //
982 // ( c*sx s*sx 0)
983 // = ( -s*sy c*sy 0)
984 // (c*sx*tx-s*sy*ty s*sx*tx+c*sy*ty 1)
985 //
986 float csx = c * sx;
987 float ssx = s * sx;
988 float csy = c * sy;
989 float ssy = s * sy;
990 mTransformation = Mat3f(csx, ssx, 0.0f, -ssy, csy, 0.0f, csx*tx-ssy*ty, ssx*tx+csy*ty, 1.0f);
991 } else {
992 mTransformation = Mat3f(1.0f);
993 }
994
995 // Copy remaining data
996 mBorderColor = borderColor;
997 mRepsU = repsU;
998 mRepsV = repsV;
999 mMirrorU = mirrorU;
1000 mMirrorV = mirrorV;
1001
1002 bool gammaOn = (gamma != sWhite);
1003 bool saturationOn = (saturation != sWhite);
1004 bool contrastOn = (contrast != sWhite);
1005 bool gainOffsetOn = ((gain != sWhite) || (offset != sBlack));
1006 bool temperatureControlOn = (temperature != Vec3f(zero));
1007
1008 bool doColorCorrect = gammaOn || saturationOn || contrastOn || gainOffsetOn || temperatureControlOn;
1009
1010 // Loop over mipmap levels
1011 for (int mipLevel = 0; mipLevel < mNumMipLevels; ++mipLevel) {
1012
1013 // Dimensions of mip level
1014 int mipWidth = mWidth >> mipLevel;
1015 int mipHeight = mHeight >> mipLevel;
1016
1017 // Create a sampling distribution from the luminance of the image
1018 mDistribution[mipLevel] = scene_rdl2::util::alignedMallocCtorArgs<Distribution2D>(DEFAULT_MEMORY_ALIGNMENT,
1019 mipWidth, mipHeight);
1020 const int paddedNumPixels = (mipWidth * mipHeight + simdWidth - 1) & -simdWidth;
1021 const int parallelForRange = paddedNumPixels / simdWidth;
1022
1023 // Color Correct the input image
1024 // if any color correct attribute values are non-default, perform color correction...
1025 if (doColorCorrect) {
1026 tbb::parallel_for(tbb::blocked_range<int>( 0 , parallelForRange,
1027 parallelForRange / 64),
1028 [&](const tbb::blocked_range<int> t) {
1029 ispc::PBR_colorCorrectFormatArray(t.end() - t.begin(),
1030 gammaOn, gamma.r, gamma.g, gamma.b,
1031 gainOffsetOn, gain.r, gain.g, gain.b,
1032 offset.r, offset.g, offset.b,
1033 contrastOn, contrast.r, contrast.g, contrast.b,
1034 saturationOn, saturation.r, saturation.g, saturation.b,
1035 temperatureControlOn, temperature.x, temperature.y, temperature.z,
1036 &(mPixelBuffer[mipLevel][t.begin() * simdWidth * 3]));
1037 });
1038 }
1039
1040 // Update the Distribution with Image's luminance value
1041 if (mIsTransformed) {
1042 // Compute a suitable sampling rate in each direction, based on the the upper-left
1043 // 2x2 portion of the transformation matrix, which we can interpret as derivatives
1044 const Mat3f &m = mTransformation;
1045 float uCount = scene_rdl2::math::ceil(2.0f * max(scene_rdl2::math::abs(m[0][0]), scene_rdl2::math::abs(m[0][1]))); // # super-samples per texel in u direction
1046 float vCount = scene_rdl2::math::ceil(2.0f * max(scene_rdl2::math::abs(m[1][0]), scene_rdl2::math::abs(m[1][1]))); // # super-samples per texel in v direction
1047 float uCountRcp = 1.0f / uCount; // reciprocals, for speed
1048 float vCountRcp = 1.0f / vCount;
1049 float uOffset = 0.5f * (uCount - 1.0f); // offsets to center super-samples around
1050 float vOffset = 0.5f * (vCount - 1.0f); // texel center
1051
1052 // Sample the texture at the determined rate, (uCount x vCount) samples per texel
1053 tbb::parallel_for(tbb::blocked_range<size_type>(0, mipHeight, mipHeight / sRangeDivider),
1054 [&](const tbb::blocked_range<size_type> range) {
1055 for (size_type y = range.begin(); y < range.end(); ++y) {
1056 float v = (static_cast<float>(y) + 0.5f) / mipHeight;
1057 for (size_type x = 0; x < mipWidth; ++x) {
1058 float u = (static_cast<float>(x) + 0.5f) / mipWidth;
1059
1060 // Accumulate samples for current texel
1061 Color color(0.f);
1062 for (float vSub = 0.0f; vSub < vCount; vSub += 1.0f) {
1063 float vAdjusted = v + (vSub - vOffset) * vCountRcp;
1064 for (float uSub = 0.0f; uSub < uCount; uSub += 1.0f) {
1065 float uAdjusted = u + (uSub - uOffset) * uCountRcp;
1066 color += textureLookupTransformed(uAdjusted, vAdjusted, mipLevel,
1067 TEXTURE_FILTER_NEAREST);
1068 }
1069 }
1070
1071 // Take luminance of mean color
1072 color *= uCountRcp * vCountRcp;
1073 float lum = luminance(color);
1074 lum = max(lum, sEpsilon);
1075 mDistribution[mipLevel]->setWeight(x, y, lum);
1076 }
1077 }
1078 });
1079 } else {
1080 // Non-transformed case is simpler (1 sample per texel)
1081 tbb::parallel_for(tbb::blocked_range<size_type>(0, mipHeight, mipHeight / sRangeDivider),
1082 [&](const tbb::blocked_range<size_type> range) {
1083 for (size_type y = range.begin(); y < range.end(); ++y) {
1084 float v = (static_cast<float>(y) + 0.5f) / mipHeight;
1085 for (size_type x = 0; x < mipWidth; ++x) {
1086 float u = (static_cast<float>(x) + 0.5f) / mipWidth;
1087 Color color = textureLookupDirect(u, v, mipLevel, TEXTURE_FILTER_NEAREST);
1088 float lum = luminance(color);
1089 lum = max(lum, sEpsilon);
1090 mDistribution[mipLevel]->setWeight(x, y, lum);
1091 }
1092 }
1093 });
1094 }
1095
1096 // Get ready to sample
1097 mDistribution[mipLevel]->tabulateCdf(mapping);
1098 }
1099}
