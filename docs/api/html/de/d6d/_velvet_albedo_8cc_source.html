18/// Returns the AvgAlbedo over all possible thetas for this roughness
19float
20VelvetAlbedo::avg(float roughness)
21{
22 const float* avgAlbedoTable = reinterpret_cast<const float*>(ispc::VelvetAlbedo_getAvgAlbedoTable());
23 int lowerIndex, upperIndex;
24 getAvgTableIndices(roughness, lowerIndex, upperIndex);
25
26 const float frac = roughness - static_cast<int>(roughness);
27 // linear blending
28 return ((1.f-frac) * avgAlbedoTable[lowerIndex] +
29 frac * avgAlbedoTable[upperIndex]);
30}

32/// Returns the albedo at this theta, roughness
33float
34VelvetAlbedo::at(float cosTheta, float roughness)
35{
36 // cosTheta is divided into 100 buckets and ranges from 0 to pi/2
37 // so the indirection here converts back into a lookup index
38 const float* albedoTable = reinterpret_cast<const float*>(ispc::VelvetAlbedo_getAlbedoTable());
39 int lowerIndex, upperIndex;
40 getAlbedoTableIndices(cosTheta, roughness,
41 lowerIndex, upperIndex);
42
43 const float frac = roughness - static_cast<int>(roughness);
44 // linear blending
45 return ((1.f-frac) * albedoTable[lowerIndex] +
46 frac * albedoTable[upperIndex]);
47}
