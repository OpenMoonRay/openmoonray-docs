68bool
69VolumeShader::isHomogenous() const
70{
71 unsigned int property = getProperties();
72 bool result = true;
73 if (property & rdl2::VolumeShader::IS_EXTINCTIVE) {
74 result &= ((property & rdl2::VolumeShader::HOMOGENOUS_EXTINC) != 0);
75 }
76 if (property & rdl2::VolumeShader::IS_SCATTERING) {
77 result &= ((property & rdl2::VolumeShader::HOMOGENOUS_ALBEDO) != 0);
78 }
79 if (property & rdl2::VolumeShader::IS_EMISSIVE) {
80 result &= ((property & rdl2::VolumeShader::HOMOGENOUS_EMISS) != 0);
81 }
82 return result;
83}
