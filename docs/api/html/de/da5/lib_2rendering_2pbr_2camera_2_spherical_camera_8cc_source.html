28void SphericalCamera::createRayImpl(mcrt_common::RayDifferential* dstRay,
29 float x,
30 float y,
31 float time,
32 float /*lensU*/,
33 float /*lensV*/) const
34{
35 // Compute transforms
36 Mat4f ct2render; // "camera space at ray time" --> render space
37 if (getMotionBlur()) {
38 ct2render = computeCamera2Render(time);
39 } else {
40 time = 0.0f;
41 ct2render = getCamera2Render();
42 }
43
44 const Vec3f cameraOrigin = transformPoint(ct2render, Vec3f(0, 0, 0));
45 *dstRay = mcrt_common::RayDifferential(
46 cameraOrigin, transformVector(ct2render, createDirection(x, y)),
47 cameraOrigin, transformVector(ct2render, createDirection(x + 1.0f, y)),
48 cameraOrigin, transformVector(ct2render, createDirection(x, y + 1.0f)),
49 getNear(), getFar(), time, 0);
50}

52Vec3f SphericalCamera::createDirection(float x, float y) const
53{
54 const float width = getApertureWindowWidth();
55 const float height = getApertureWindowHeight();
56
57 const float theta = sPi * y / height;
58 const float phi = 2.0f * sPi * x / width;
59
60 float sintheta, costheta;
61 float sinphi, cosphi;
62
63 // theta is in [0, pi) (excluding filter importance sampling). Subtract from
64 // pi to reverse image y.
65 sincos(sPi - theta, &sintheta, &costheta);
66 sincos(phi, &sinphi, &cosphi);
67
68 return Vec3f(sintheta * cosphi, costheta, sintheta * sinphi);
69}
