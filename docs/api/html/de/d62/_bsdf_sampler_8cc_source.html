28// Warning: this must return at least one sample for each lobe, otherwise
29// the algorithm is biased (unless the lobe has zero albedo, but in this case
30// the shader should never have added the lobe in the first place).
31static finline int
32computeLobeSampleCount(const shading::BsdfLobe *lobe, const shading::BsdfSlice &slice,
33 int maxLobeSamples, int lobeCount)
34{
35 int sampleCount;
36
37 if (lobe->matchesFlags(shading::BsdfLobe::ALL_MIRROR) || maxLobeSamples == 1) {
38 // We devote only a single sample to delta distributions
39 sampleCount = 1;
40 } else {
41 // Compute lobes' approximated albedo to adjust the number of samples.
42 // Note: since we use the approximated albedo, our sampling decisions might
43 // be slightly off in cases where a lobe crosses the surface (at grazing
44 // angles). In these cases, we may spend slightly more samples than we
45 // should. Also since we don't know yet about the incident lighting, we may
46 // spend fewer samples for a given lobe than we wish for (i.e. a low-albedo
47 // lobe that reflects a very strong spike of light), or too many samples
48 // (i.e. a high-albedo lobe that reflects no light at all).
49 // Note: Use the mirror reflection direction assumption to compute the
50 // albedo approximaion, which scales our lobe sampling decisions
51 // TODO: Factor in heuristics based on roughness, open up to
52 // other heuristics (pass in a functor ?)
53#if 1
54 sampleCount = maxLobeSamples;
55#else
56 // Note: we multiply by the number of lobes, because the number of samples
57 // is adjusted by the albedo, which shouldn't add up to more than one over
58 // all lobes, so this is to keep roughly the same sample count as before.
59 float lumAlbedo = abs(luminance(lobe->albedo(slice)));
60 sampleCount = static_cast<int>(ceil(lumAlbedo * maxLobeSamples * lobeCount));
61 sampleCount = min(sampleCount, maxLobeSamples * lobeCount);
62#endif
63 }
64
65 // We need at least 1 sample per lobe otherwise this algorithm is biased
66 MNRY_ASSERT(maxLobeSamples == 0 || sampleCount > 0);
67
68 return sampleCount;
69}

74BsdfSampler::BsdfSampler(scene_rdl2::alloc::Arena *arena, shading::Bsdf &bsdf, const shading::BsdfSlice &slice,
75 int maxSamplesPerLobe, bool doIndirect,
76 const PathGuide &pathGuide) :
77 mBsdf(bsdf),
78 mSlice(slice),
79 mMaxSamplesPerLobe(maxSamplesPerLobe),
80 mDoIndirect(doIndirect),
81 mLobeCount(mBsdf.getLobeCount()),
82 mSampleCount(0),
83 mPathGuide(pathGuide)
84{
85 // Allocate and initialize lobe arrays
86 mLobeIndex = arena->allocArray<char>(mLobeCount);
87 mLobeSampleCount = arena->allocArray<int>(mLobeCount);
88 mInvLobeSampleCount = arena->allocArray<float>(mLobeCount);
89
90 // Compute sample count per lobe, and keep track of lobes with samples
91 int lKeep = 0;
92 shading::BsdfLobe::Type flags = mSlice.getFlags();
93 for (int l = 0; l < mLobeCount; ++l) {
94 const shading::BsdfLobe *lobe = mBsdf.getLobe(l);
95 if (!lobe->matchesFlags(flags)) {
96 continue;
97 }
98
99 int sampleCount = computeLobeSampleCount(lobe, mSlice, maxSamplesPerLobe, mLobeCount);
100
101 mLobeIndex[lKeep] = l;
102 mLobeSampleCount[lKeep] = sampleCount;
103 mInvLobeSampleCount[lKeep] = rcp(float(sampleCount));
104 mSampleCount += sampleCount;
105
106 lKeep++;
107 }
108 mLobeCount = lKeep;
109
110 // Early return if no lobes match the given flag
111 if (mLobeCount == 0) {
112 MNRY_ASSERT(mSampleCount == 0);
113 return;
114 }
115}

122bool
123BsdfSampler::sample(pbr::TLState *pbrTls, int lobeIndex, const Vec3f &p, float r1,
124 float r2, BsdfSample &sample) const
125{
126 // Get lobe in turn and draw a sample from that lobe
127 const shading::BsdfLobe *lobe = getLobe(lobeIndex);
128 sample.sample = Vec2f(r1, r2);
129
130 // Pdf computation needs to be kept in sync when integrating
131 // light samples (see integrateLightSetSample() in PathIntegratorUtil.cc).
132 // Skip path guiding on mirror lobes, because their
133 // sample direction is already precisely determined.
134 if (mPathGuide.canSample() && !(lobe->getType() & shading::BsdfLobe::MIRROR)) {
135 float bsdfPdf = 0.0f;
136 float pgPdf = 0.0f;
137 const float u = mPathGuide.getPercentage();
138
139 // r1 is serving a dual purpose. It is first used to determine if
140 // we are going to use path guiding at all by comparing it to the
141 // path guiding percentage. We then use this same r1 value as one
142 // of the two inputs to either the path guide or lobe sampler.
143 // Since we need to have all possible inputs to the samplers be between [0, 1),
144 // we remap r1 from [0, u] or (u, 1) to [0, 1).
145 if (r1 > u) {
146 // use bsdf lobe sampling direction
147 MNRY_ASSERT(u < 1.0f);
148 const float r = (r1 - u) / (1.0f - u); // remap r1 into [0, 1)
149 sample.f = lobe->sample(mSlice, r, r2, sample.wi, bsdfPdf);
150 pgPdf = mPathGuide.getPdf(p, sample.wi);
151 } else {
152 // use path guide sampling direction
153 const float r = r1 / u; // remap r1 into [0, 1)
154 sample.wi = mPathGuide.sampleDirection(p, r, r2, &pgPdf);
155 sample.f = lobe->eval(mSlice, sample.wi, &bsdfPdf);
156 }
157 // blending pdf values seems to work well enough in practice, and
158 // allows a potential user percentage control.
159 sample.pdf = u * pgPdf + (1.0 - u) * bsdfPdf;
160 } else {
161 sample.f = lobe->sample(mSlice, r1, r2, sample.wi, sample.pdf);
162 }
163
164 // Check if sample is invalid
165 bool isValid = isSampleValid(sample.f, sample.pdf);
166
167 // Check surface flag
168 // TODO: BUG ? Should this be called passing in the lobe instead ?
169 shading::BsdfLobe::Type flags = mSlice.getSurfaceFlags(mBsdf, sample.wi);
170 isValid &= lobe->matchesFlags(flags);
171
172 // Initialize distance, possibly marking sample as invalid
173 sample.distance = (isValid ? scene_rdl2::math::sMaxValue : BsdfSample::sInvalidDistance);
174
175 if (isValid) {
176 pbrTls->mStatistics.incCounter(STATS_BSDF_SAMPLES);
177 }
178
179 return isValid;
180}
