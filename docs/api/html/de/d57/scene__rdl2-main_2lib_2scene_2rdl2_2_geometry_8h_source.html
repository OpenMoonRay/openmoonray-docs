26// This enum is used in multiple attributes.cc files and we require those files to have a minimum of

153 template <typename Container>
154 bool geometryUpdatePrepSequenceContainer(const Attribute* attribute,
155 UpdateHelper& sceneObjects,
156 int depth)
157 {
158 bool updateRequired = false;
159 const AttributeKey<Container> key(*attribute);
160 const Container& objectVector = get(key);
161 for (SceneObject * const object : objectVector) {
162 if (object) {
163 if (object->isA<Geometry>() && !object->asA<Geometry>()->
164 requiresGeometryUpdate(sceneObjects, depth + 1)) {
165 continue;
166 }
167 updateRequired |= object->updatePrep(sceneObjects, depth);
168 }
169 }
170 mAttributeTreeChanged |= updateRequired;
171 return updateRequired;
172 }
