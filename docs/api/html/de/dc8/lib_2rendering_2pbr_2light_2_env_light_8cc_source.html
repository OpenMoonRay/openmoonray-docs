39// Compute local space dir from uv sphere parameterization. The output vector
40// is defined in a z-up local coordinate system.
41finline Vec3f
42uv2local(const Vec2f &uv)
43{
44 // We are flipping the u coordinate here since we're on the inside of the
45 // sphere. We're flipping v since local z and texture v are in opposite
46 // directions.
47 float phi = (1.0f - uv.x) * sTwoPi;
48 float theta = (1.0f - uv.y) * sPi;
49
50 float sinTheta, cosTheta, sinPhi, cosPhi;
51 sincos(theta, &sinTheta, &cosTheta);
52 sincos(phi, &sinPhi, &cosPhi);
53
54 return Vec3f(sinTheta * cosPhi, sinTheta * sinPhi, cosTheta);
55}

58// Compute uv sphere parameterization from local space direction. The input dir
59// is assumed to be defined in a z-up local coordinate system.
60finline Vec2f
61local2uv(const Vec3f &dir)
62{
63 MNRY_ASSERT(isNormalized(dir));
64
65 float theta = scene_rdl2::math::acos(clamp(dir.z, -1.0f, 1.0f));
66 float phi = scene_rdl2::math::atan2(dir.y, dir.x);
67
68 float u = phi * sOneOverTwoPi;
69 float v = 1.0f - (theta * sOneOverPi);
70
71 // Handle singularities at poles, to keep function consistant when inverted.
72 if (v < 0.0005f || v > 0.9995f) {
73 u = 0.0f;
74 } else if (u < 0.0f) {
75 u += 1.0f;
76 }
77
78 // undo the flip of the u coordinate that we did in uv2local
79 u = 1.0f - u;
80
81 MNRY_ASSERT(finite(u));
82 MNRY_ASSERT(finite(v));
83
84 return Vec2f(u, v);
85}

142bool
143EnvLight::update(const Mat4d& world2render)
144{
145 MNRY_ASSERT(mRdlLight);
146
147 mOn = mRdlLight->get(scene_rdl2::rdl2::Light::sOnKey);
148 if (!mOn) {
149 return false;
150 }
151
152 updateVisibilityFlags();
153 updatePresenceShadows();
154 updateRayTermination();
155 updateTextureFilter();
156 updateMaxShadowDistance();
157
158 const Mat4d l2w0 = mRdlLight->get(scene_rdl2::rdl2::Node::sNodeXformKey, /* rayTime = */ 0.0f);
159 const Mat4d l2w1 = mRdlLight->get(scene_rdl2::rdl2::Node::sNodeXformKey, /* rayTime = */ 1.0f);
160 const Mat4f local2render0 = Mat4f::orthonormalize(sLocalOrientation * toFloat(l2w0 * world2render));
161 const Mat4f local2render1 = Mat4f::orthonormalize(sLocalOrientation * toFloat(l2w1 * world2render));
162 ReferenceFrame frame0(local2render0);
163 ReferenceFrame frame1(local2render1);
164 mPosition[0] = mPosition[1] = zero;
165 mOrientation[0] = normalize(scene_rdl2::math::Quaternion3f(frame0.getX(), frame0.getY(), frame0.getZ()));
166 mOrientation[1] = normalize(scene_rdl2::math::Quaternion3f(frame1.getX(), frame1.getY(), frame1.getZ()));
167 if (dot(mOrientation[0], mOrientation[1]) < 0) {
168 mOrientation[1] *= -1.f;
169 }
170 mFrame = frame0;
171 mDirection = mFrame.getZ();
172
173 // setup mMb
174 mMb = LIGHT_MB_NONE;
175 const scene_rdl2::rdl2::SceneVariables &vars =
176 getRdlLight()->getSceneClass().getSceneContext()->getSceneVariables();
177 if (vars.get(scene_rdl2::rdl2::SceneVariables::sEnableMotionBlur) &&
178 getRdlLight()->get(scene_rdl2::rdl2::Light::sMbKey) &&
179 (!isEqual(frame0.getX(), frame1.getX()) ||
180 !isEqual(frame0.getY(), frame1.getY()) ||
181 !isEqual(frame0.getZ(), frame1.getZ()))) {
182 mMb = LIGHT_MB_ROTATION;
183 }
184
185 // We store solid angle and its inverse in the area members, since this is
186 // also useful for the unittest.
187 // Note: As with the DiskLight, we sample the whole (theta, phi) domain
188 // and let the ImageDistribution be responsible for distributing the
189 // samples over the (hemi)sphere as the Distribution2D::HEMISPHERICAL
190 // will zero-out weights in the lower hemisphere and automatically
191 // renormalize the pdf accordingly.
192 mHemispherical = mRdlLight->get<scene_rdl2::rdl2::Bool>(sSampleUpperHemisphereOnlyKey);
193 mArea = (mHemispherical ? sTwoPi : sFourPi);
194 mInvArea = 1.0f / mArea;
195
196 // for EnvLights, color is assumed to always be a radiance value
197 mRadiance = computeLightRadiance(mRdlLight, scene_rdl2::rdl2::Light::sColorKey,
198 scene_rdl2::rdl2::Light::sIntensityKey, scene_rdl2::rdl2::Light::sExposureKey);
199 if (isBlack(mRadiance)) {
200 mOn = false;
201 return false;
202 }
203
204 // Set here in case we early-out
205 mLog2TexelAngle = scene_rdl2::math::neg_inf;
206
207 if (!updateImageMap(mHemispherical ? Distribution2D::HEMISPHERICAL :
208 Distribution2D::SPHERICAL)) {
209 return false;
210 }
211
212 // Precompute a value for use with ray-footprint-based mip level selection.
213 // We base it on the largest extent (lat or long) of the largest texel, storing
214 // log2() of the result because it accelerates the mip level selection calculation.
215 if (mDistribution) {
216 float texelAngleLongitude = sTwoPi / (float)mDistribution->getWidth();
217 float texelAngleLatitude = sPi / (float)mDistribution->getHeight();
218 float texelAngle = max(texelAngleLongitude, texelAngleLatitude);
219 mLog2TexelAngle = scene_rdl2::math::log2(texelAngle);
220 }
221
222 return true;
223}

244bool
245EnvLight::intersect(const Vec3f &p, const Vec3f *n, const Vec3f &wi, float time,
246 float maxDistance, LightIntersection &isect) const
247{
248 if (mHemispherical && dot(wi, getDirection(time)) < 0.0f) {
249 return false;
250 }
251
252 if (sEnvLightDistance > maxDistance) {
253 return false;
254 }
255
256 isect.N = -wi;
257 isect.distance = sEnvLightDistance;
258 isect.uv = mDistribution ? local2uv(globalToLocal(wi, time)) : zero;
259
260 return true;
261}

263bool
264EnvLight::sample(const Vec3f &p, const Vec3f *n, float time, const Vec3f& r,
265 Vec3f &wi, LightIntersection &isect, float rayDirFootprint) const
266{
267 MNRY_ASSERT(mOn);
268
269 if (mDistribution) {
270 float mipLevel = getMipLevel(rayDirFootprint);
271 mDistribution->sample(r[0], r[1], mipLevel, &isect.uv, nullptr, mTextureFilter);
272
273 // Handle singularities at poles so that sample() and intersect()
274 // will return identical values. if v == 0, then throw away u value
275 // since it's meaningless.
276 if (isect.uv.y < 0.0005f || isect.uv.y > 0.9995f) {
277 isect.uv.x = 0.0f;
278 }
279
280 wi = uv2local(isect.uv);
281 wi = localToGlobal(wi, time);
282 } else if (mHemispherical) {
283 isect.uv = zero;
284 wi = shading::sampleLocalHemisphereUniform(r[0], r[1]);
285 wi = localToGlobal(wi, time);
286 } else {
287 isect.uv = zero;
288 wi = shading::sampleSphereUniform(r[0], r[1]);
289 }
290
291 if (n && dot(*n, wi) < sEpsilon) {
292 return false;
293 }
294
295 isect.N = -wi;
296 isect.distance = sEnvLightDistance;
297
298 return true;
299}

302Color
303EnvLight::eval(mcrt_common::ThreadLocalState* tls, const Vec3f &wi, const Vec3f &p, const LightFilterRandomValues& filterR, float time,
304 const LightIntersection &isect, bool fromCamera, const LightFilterList *lightFilterList, float rayDirFootprint,
305 float *pdf) const
306{
307 MNRY_ASSERT(mOn);
308
309 float mipLevel = getMipLevel(rayDirFootprint);
310
311 Color radiance = mRadiance;
312 if (mDistribution) {
313 radiance *= mDistribution->eval(isect.uv[0], isect.uv[1], mipLevel, mTextureFilter);
314 }
315
316 if (lightFilterList) {
317 evalLightFilterList(lightFilterList,
318 { tls, &isect, getPosition(time),
319 getDirection(time), p,
320 filterR, time,
321 globalToLocalXform(time, lightFilterList->needsLightXform()),
322 wi
323 },
324 radiance);
325 }
326
327 if (pdf) {
328 if (mDistribution) {
329 // We must account for the mapping transformation so we express the
330 // pdf density on the sphere in solid angles (see pbrt section 14.6.5)
331 float sinTheta = scene_rdl2::math::sin((isect.uv[1]) * sPi);
332 sinTheta = max(sinTheta, sEpsilon);
333 *pdf = mDistribution->pdf(isect.uv[0], isect.uv[1], mipLevel, mTextureFilter)
334 / (sTwoPiSqr * sinTheta);
335 MNRY_ASSERT(finite(*pdf));
336 } else {
337 *pdf = mInvArea;
338 }
339 }
340
341 return radiance;
342}
