27unsigned
28RenderPrepStats::getTotalSteps() const
29{
30 //
31 // This totalSteps is used for renderPrep progress fraction and indicates total task steps.
32 // We don't know RenderPrep task step total upfront. Total steps are computed on the fly
33 // based on the information which we can get at each renderPrep stage.
34 //
35 // We have following 7 stages for renderPrep
36 // a) apply update
37 // b) load geometry 0
38 // c) load geometry 1
39 // d) tessellation 0
40 // e) bvh construction 0
41 // f) tessellation 1
42 // g) bvh construction 1
43 //
44 // Each stages have own steps at start (5points) and end (5points). (5points is defined by
45 // mStageStepsPoints and this value was heuristically defined with a couple of testing )
46 // We also consider sub-progress total steps if the stage is b, c, d, and f.
47 // These sub-progress total steps are progressively updated based on the renderPrep computation.
48 // They are initially ZERO.
49 //
50 unsigned total = 7 * 2 * mStageStepsPoints; // we count each stage (=7) 2 times (start and end).
51
52 if (isCompleted()) {
53 // early exit : renderPrep has been completed
54 total += (mLoadGeometriesTotal[0] + mLoadGeometriesTotal[1] +
55 mTessellationTotal[0] + mTessellationTotal[1]);
56 return total;
57 }
58
59 if (!isStageFinished(Stage::GM_LOADGEO0_START)) return total;
60
61 int tessellation0_estimation = mLoadGeometriesTotal[0]; // estimate steps for tessellation0
62 total += mLoadGeometriesTotal[0] + tessellation0_estimation;
63 if (!isStageFinished(Stage::GM_LOADGEO1_START)) return total;
64
65 int tessellation1_estimation = mLoadGeometriesTotal[1]; // estimate steps for tessellation0
66 total += mLoadGeometriesTotal[1] + tessellation1_estimation;
67 if (!isStageFinished(Stage::GM_FINALIZE0_TESSELLATION)) return total;
68
69 total += mTessellationTotal[0] - tessellation0_estimation; // replace estimate by actual steps
70 if (!isStageFinished(Stage::GM_FINALIZE1_TESSELLATION)) return total;
71
72 total += mTessellationTotal[1] - tessellation1_estimation; // replace estimate by actual steps
73 return total;
74}

76unsigned
77RenderPrepStats::getCurrSteps() const
78{
79 unsigned steps = 0;
80
81 if (isCompleted()) {
82 // early exit : renderPrep has been completed
83 steps =
84 7 * 2 * mStageStepsPoints +
85 mLoadGeometriesProcessed[0] + mLoadGeometriesProcessed[1] +
86 mTessellationProcessed[0] + mTessellationProcessed[1];
87 return steps;
88 }
89
90 // a) apply update
91 if (!isStageFinished(Stage::RENDER_PREP_APPLYUPDATE)) return steps;
92 steps += mStageStepsPoints;
93 if (!isStageFinished(Stage::RENDER_PREP_APPLYUPDATE_DONE)) return steps;
94 steps += mStageStepsPoints;
95
96 // b) load geometry 0
97 if (!isStageFinished(Stage::RENDER_PREP_LOAD_GEOM0)) return steps;
98 steps += mStageStepsPoints;
99 if (!isStageFinished(Stage::GM_LOADGEO0_START)) return steps;
100 steps += mLoadGeometriesProcessed[0];
101 if (!isStageFinished(Stage::RENDER_PREP_LOAD_GEOM0_DONE)) return steps;
102 steps += mStageStepsPoints;
103
104 // c) load geometry 1
105 if (!isStageFinished(Stage::RENDER_PREP_LOAD_GEOM1)) return steps;
106 steps += mStageStepsPoints;
107 if (!isStageFinished(Stage::GM_LOADGEO1_START)) return steps;
108 steps += mLoadGeometriesProcessed[1];
109 if (!isStageFinished(Stage::RENDER_PREP_LOAD_GEOM1_DONE)) return steps;
110 steps += mStageStepsPoints;
111
112 // d) tessellation 0
113 if (!isStageFinished(Stage::GM_FINALIZE0_TESSELLATION)) return steps;
114 steps += mStageStepsPoints;
115 if (!isStageFinished(Stage::GM_FINALIZE0_TESSELLATION_PROCESS)) return steps;
116 steps += mTessellationProcessed[0];
117 if (!isStageFinished(Stage::GM_FINALIZE0_TESSELLATION_DONE)) return steps;
118 steps += mStageStepsPoints;
119
120 // e) bvh construction 0
121 if (!isStageFinished(Stage::GM_FINALIZE0_BVH)) return steps;
122 steps += mStageStepsPoints;
123 if (!isStageFinished(Stage::GM_FINALIZE0_BVH_DONE)) return steps;
124 steps += mStageStepsPoints;
125
126 // f) tessellation 1
127 if (!isStageFinished(Stage::GM_FINALIZE1_TESSELLATION)) return steps;
128 steps += mStageStepsPoints;
129 if (!isStageFinished(Stage::GM_FINALIZE1_TESSELLATION_PROCESS)) return steps;
130 steps += mTessellationProcessed[1];
131 if (!isStageFinished(Stage::GM_FINALIZE1_TESSELLATION_DONE)) return steps;
132 steps += mStageStepsPoints;
133
134 // g) bvh construction 1
135 if (!isStageFinished(Stage::GM_FINALIZE1_BVH)) return steps;
136 steps += mStageStepsPoints;
137 if (!isStageFinished(Stage::GM_FINALIZE1_BVH_DONE)) return steps;
138 steps += mStageStepsPoints;
139
140 return steps;
141}

143std::string
144RenderPrepStats::show() const
145{
146 std::ostringstream ostr;
147 ostr << "RenderPrepStats {\n"
148 << " mStage:" << stageStr(mStage) << '\n';
149 if (static_cast<unsigned>(mStage) >= static_cast<unsigned>(Stage::GM_LOADGEO0_START)) {
150 ostr << " loadGeometry stage0 {\n"
151 << " mLoadGeometriesTotal:" << mLoadGeometriesTotal[0] << '\n'
152 << " mLoadGeometriesProcessed:" << mLoadGeometriesProcessed[0] << '\n'
153 << " }\n";
154 }
155 if (static_cast<unsigned>(mStage) >= static_cast<unsigned>(Stage::GM_LOADGEO1_START)) {
156 ostr << " loadGeometry stage1 {\n"
157 << " mLoadGeometriesTotal:" << mLoadGeometriesTotal[1] << '\n'
158 << " mLoadGeometriesProcessed:" << mLoadGeometriesProcessed[1] << '\n'
159 << " }\n";
160 }
161 if (static_cast<unsigned>(mStage) >= static_cast<unsigned>(Stage::GM_FINALIZE0_TESSELLATION)) {
162 ostr << " finalizeChange stage0 {\n"
163 << " mTessellationTotal:" << mTessellationTotal[0] << '\n'
164 << " mTessellationProcessed:" << mTessellationProcessed[0] << '\n'
165 << " }\n";
166 }
167 if (static_cast<unsigned>(mStage) >= static_cast<unsigned>(Stage::GM_FINALIZE1_TESSELLATION)) {
168 ostr << " finalizeChange stage1 {\n"
169 << " mTessellationTotal:" << mTessellationTotal[1] << '\n'
170 << " mTessellationProcessed:" << mTessellationProcessed[1] << '\n'
171 << " }\n";
172 }
173 ostr << "}";
174 return ostr.str();
175}
