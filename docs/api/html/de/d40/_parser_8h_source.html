19//
20// How to use Parser/Arg for your command line parsing logic.
21//
22// --- Abstract and topics ---
23//
24// Parser class provides simple command-line parsing logic which is driven by user-defined command definitions
25// as a pretty easy way.
26//
27// There are several runtime topics
28// - Help usage message is automatically generated based on the user-defined command definitions.
29// - Typical procedure to dump help is call Parser::main(Arg("help")) (See Arg::isHelp())
30// - Comment for command-line starts with '#'
31// - Proper error messages are output when Parser::main() detects error internally.
32//
33// This Parser implementation is heavily used by debugConsole feature for interactive debugging under arras
34// multi-machine configuration.
35// It is pretty easy to add new options and maintain them.
36// It also provides child commands parsing solutions as well.
37// This child commands solution can organize nicely too many commands into a tree structure of commands.
38//
39//
40// --- 3 steps for using Parser ---
41//
42// 1) Construction
43// - construct Parser object.
44//
45// 2) Configure options and arguments
46// - There are 2 types of parser item configuration. a) options and b) arguments.
47// - You can combine options and arguments into the same Parser object.
48//
49// 2-a) options
50// - Option is like "optA <a> <b>".
51// - Option is recognized by option name. In this case, the option name is "optA".
52// - This optA requires 2 option argument <a> and <b>
53// - You can configure many options as you want by multiple calls of opt() API.
54// - opt() exec order does not matter.
55// - Internally, the first letter of option argument '-' would be ignored.
56// - with '-' option name definition like "-optA", both "-optA" and "optA" would be recognized the same.
57// - without '-' option name definition like "optB" both "-optB" and "optB" would be recognized the same.
58// 2-b) arguments
59// - You can configure many arguments as you want by arg() API.
60// - Argument is like "<A> <B>". This is not optional, you have to always set <A> and <B>.
61// - You get an error If you don't set <A> and <B>.
62// - You can configure many arguments as you want by multiple calls of arg() API.
63// - Order of arg() call DOES matter.
64// - If you configure 2 argument items, for example,
65// 1st one is arg("<A> <B>", ...) and 2nd one is arg("<C>", ...).
66// In this case, you have to set 3 arguments like "<0> <1> <2>" at the command-line.
67// - Value <0> and <1> go to <A> and <B>. Value<2> goes to <C>.
68// - You get errors If you don't set 3 arguments.
69//
70// 3) Evaluate command-line
71// - You construct an Arg object based on the command-line information. We have several ways to construct
72// an Arg object. (See Arg.h)
73// - Evaluation of this Arg object is simple. just call Parser::main(Arg &arg)
74//
75//
76// --- message output from parser action function ---
77//
78// In order to output (= printout) messages from parser action function, Arg object has msg() method.
79// Using Arg::msg() is strongly recommended rather than directly use std::{cour,cerr}.
80// Basically Arg::msg() API printouts string data to the std::cerr as default.
81// Actually, you can configure more options to the Arg::msg() action.
82// We can configure the Arg object to send a message to the tlSvr (telnet-server) client if this Arg object
83// is constructed by an incoming command-line string received by tlSvr.
84// We can also set a special message handler to the Arg object as well.
85// How to configure tlSvr and/or message handler is totally up to the Arg object construction code before
86// you call Parser::main(). (unitTest code might be a good example. See unittest/TestParser.cc and find
87// setMessageHandler())
88// In order to easily control where the printout message goes, using Arg::msg() is strongly recommended
89// inside the parser action function.
90// (See also Arg.h comments)
91//
92//
93// --- example ---
94//
95// Example-1 : simple option only configuration - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
96//
97// // construction and set description
98// Parser parser;
99// parser.description("very simple option only definition");
100//
101// // configure 2 options
102// parser.opt("foo", // set option name
103// "<a> <b>", // set 2 option arguments a and b, separated by ' '
104// "option foo", // short message for usage
105// [&](Arg &arg) -> bool { // parser action function
106// int a = arg.as<int>(0); // get 1st argument as int
107// double b = arg.as<double>(1); // get 2nd argument as double
108// arg += 2; // increment internal argument id by 2 for the rest of the commands
109// ... do something here using a and b ...
110// return result; // return parser action function result condision
111// });
112// parser.opt("bar", // set option name
113// "<c>", // set single option argument
114// "option bar", // short message for usage
115// [&](Arg &arg) -> bool { // parser action function
116// float c = arg.as<float>(0); // get 1st argument as float
117// arg++; // increment internal argument id for the rest of the command
118// ... do something here using c ...
119// arg.msg("evaluate option bar\n"); // output message using arg.msg()
120// return result; // return parser action function result condition
121// });
122//
123// parser.main(Arg("help")); // display help usage message of above parser option configuration.
124// // help usage message would be like follows (may change depending on the versions).
125// // Description : very simple option only definition
126// // [Command]
127// // bar <c> : option bar
128// // foo <a> <b> : option foo
129// parser.main(Arg("bar 1.23")); // execute option bar and 1.23 goes to <c> ... (A)
130// parser.main(Arg("-bar 1.23")); // equivalent as (A)
131// parser.main(Arg("bar 7.8 foo 123 4.56")); // eval option bar first w/ c = 7.8 then
132// // eval option foo next w/ a=123, b=4.56
133// parser.main(Arg("baz a b c")); // main() returns false due to unknown option baz
134//
135// Example-2 : option and argument configuration - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
136//
137// // construction and set description
138// Parser parser;
139// parser.description("option and argument definition");
140//
141// // configure 2 options and 2 arguements
142// parser.opt("foo", "<a> <b>", "option foo",
143// [&](Arg &arg) -> bool {
144// ... do something here for option foo ...
145// return result; // return parser action function result condision
146// });
147// parser.opt("bar", "<c>", "option bar",
148// [&](Arg &arg) -> bool {
149// ... do something here for option bar ...
150// return result; // return parser action function result condition
151// });
152// parser.arg("<A> <B>", // set 2 arguments A and B, separated by ' '
153// "2 args", // short message for usage
154// [&](Arg &arg) -> bool { // parser action function
155// unsigned long A = arg.as<unsigned long>(0); // get 1st argument as u_long
156// std::string B = arg(1); // get 2nd argument as std::string
157// arg += 2;
158// ... do something here for this argument sets ...
159// return result; // return parser action function result condition
160// });
161// parser.arg("<C>", // set single argument C
162// "1 arg", // short message for usage
163// [&](Arg &arg) -> bool { // parser action function
164// float C = arg.as<float>(0); // get 1st argument as float
165// arg++; // increment internal argument id for the rest of the command
166// return result; // return parser action function result condition
167// });
168//
169// parser.main(Arg("help")); // display help usage message of above parser option configuration.
170// // help usage message would be like follows (may change depending on the versions).
171// // Description : option and argument definition
172// // [Argument]
173// // <A> <B> : 2 args
174// // <C> : 1 arg
175// // [Command]
176// // bar <c> : option bar
177// // foo <a> <b> : option foo
178// parser.main(Arg("123 abc 4.56")); // <A>=123, <B>="abc", <C>=4.56
179// parser.main(Arg("123 abc")); // main return false due to missing <C>
180// parser.main(Arg("123")); // main return false due to missing <B> and <C>
181// parser.main(Arg("foo a b A B bar c C"); // eval option foo w/ a and b first then
182// // eval argument A and B next then
183// // eval option bar w/ c next then
184// // finally eval argument C.
185//
186// Example-3 : child command - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
187// This is a pretty powerful solution to organize too many options into categorized commands of the tree
188// structure.
189//
190// // construction and set description
191// Parser parse, parserChild;
192// parser.description("child command definition");
193//
194// // configure 1 normal option and 1 child command option for top-level command parsing
195// parser.opt("foo", "<a> <b>", "option foo",
196// [&](Arg &arg) -> bool {
197// ... do something here for option foo ...
198// return result; // return parser action function result condision
199// });
200// parser.opt("baz", // set option name
201// "...command..., // No strict rules to use this string but this indicates the need for
202// // at least 1 option argument and all arguments go to the child command.
203// "command for baz", // short message for usage
204// [&](Arg &arg) -> bool { // parser action function
205// // call child parser object with childArg which contents all the rest of the arguments
206// return parserChild.main(arg.childArg());
207// });
208//
209// // configure child command parser object w/ single option definition
210// parserChild.description("child command");
211// parserChild.opt("bar", "<c>", "option bar",
212// [&](Arg &arg) -> bool { // parser action function
213// // printout argument <c>
214// return arg.msg("bar" + std::to_string((arg++).as<float>(0)) + '\n');
215// });
216//
217// parser.main(Arg("help")); // display help usage message of top level parsing configuration
218// // help usage message would be like follows (may change depending on the versions).
219// // Description : top level command definition
220// // [Command]
221// // baz ...command... : child command baz
222// // foo <a> <b> : option foo
223// parser.main(Arg("baz help")); // display help usage message of baz child command
224// // help usage message would be like follows (may change depending on the versions).
225// // Usage : baz [options]
226// // Description : child command
227// // [Options]
228// // bar <c> : option bar
229// parser.main(Arg("foo a b")); // evaluate top level foo option w/ option argument a and b
230// parser.main(Arg("baz bar 1.23")); // evaluate child option baz w/ option argument 1.23
231//
232//
233// Other Example - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -
234// Other example would be found in the following implementations

245class ParserItem
246//
247// This class defines single command/option action for command-line parsing.
248// Probably, you might need to define multiple commands/options for command-line parsing logic.
249// If so, you need to define multiple ParserItems into Parser object.
250//
251{
252public:
253 using ParserFunc = std::function<bool(Arg &)>;
254
255 enum class ItemType: int {
256 OPT = 0, // Option definition with option-key and total N (>=0) option argument(s).
257 ARG // Argument definition of command-line.
258 };
259
260 ParserItem(ItemType itemType,
261 const std::string &name,
262 const std::string &argMsg,
263 const std::string &shortMsg,
264 const ParserFunc &parserFunc) : // parser action function
265 mItemType(itemType),
266 mName(str_util::trimBlank(name)),
267 mArgMsg(cleanStr(argMsg)),
268 mShortMsg(shortMsg),
269 mParserFunc(parserFunc),
270 mArgCount(0)
271 {
272 mArgCount = computeArgCount();
273 }
274 ~ParserItem() {}
275
276 ItemType getItemType() const { return mItemType; }
277 bool isOpt() const { return (mItemType == ItemType::OPT); }
278 bool isArg() const { return (mItemType == ItemType::ARG); }
279
280 const std::string &name() const { return mName; }
281 const std::string &argMsg() const { return mArgMsg; }
282 const std::string &shortMsg() const { return mShortMsg; }
283
284 size_t getNameLen() const { return mName.size(); }
285 size_t getArgMsgLen() const { return mArgMsg.size(); }
286 size_t getArgCount() const { return mArgCount; }
287
288 // parser command main API
289 bool operator ()(Arg &arg) const { return mParserFunc(arg); }
290
291 std::string show() const;
292
293private:
294 static std::string cleanStr(const std::string &arg) {
295 std::string workStr = str_util::trimBlank(arg);
296 workStr = str_util::replaceBlankToSingleSpace(workStr);
297 return workStr;
298 }
299 size_t computeArgCount() const;
300
301 static std::string showItemType(const ItemType &itemType);
302
303 //------------------------------
304
305 ItemType mItemType;
306
307 std::string mName; // Name of this definition. Only used for ItemType = OPT.
308 std::string mArgMsg; // Argument definition string
309 std::string mShortMsg; // The short explanation for usage
310
311 ParserFunc mParserFunc; // This function is executed when this command/option action is evaluated.
312
313 size_t mArgCount; // Total argument counts based on the mArgMsg definition.
314};

316class Parser
317//
318// This class keeps all command definitions and evaluates the command-line based on them.
319// This Parser is mainly used for grid_util::DebugConsoleDriver's command line parsing implementation.
320// And this is heavily used for interactive debugging command-line control especially arras multi-machine
321// configurations.
322//
323{
324public:
325 using ParserFunc = std::function<bool(Arg &)>;
326
327 Parser() :
328 mErrorUnknownOption(true),
329 mTotalArgCount(0)
330 {}
331 ~Parser() {}
332
333 void description(const std::string &str) { mDescription = str; }
334
335 // true : main() return false when unknown option condition
336 // false : main() return true when unknown option condition
337 void setErrorUnknownOption(bool flag) { mErrorUnknownOption = flag; }
338
339 void reset() { mDescription.clear(); mParserItemTbl.clear(); }
340
341 // Configure option
342 void opt(const std::string &name,
343 const std::string &argMsg,
344 const std::string &shortMsg,
345 ParserFunc &&parserFunc) {
346 mParserItemTbl.emplace_back(ParserItem::ItemType::OPT,
347 name,
348 argMsg,
349 shortMsg,
350 std::move(parserFunc));
351 }
352
353 // Configure argument
354 void arg(const std::string &argMsg,
355 const std::string &shortMsg,
356 ParserFunc &&parserFunc) {
357 mParserItemTbl.emplace_back(ParserItem::ItemType::ARG,
358 "", // not using
359 argMsg,
360 shortMsg,
361 std::move(parserFunc));
362 mTotalArgCount = totalArgCount();
363 }
364
365 // Evaluate argument
366 bool main(Arg &arg) const;
367 bool main(Arg &&arg) const { Arg tmpArg = arg; return main(tmpArg); }
368
369 std::string show() const;
370
371protected:
372 std::string usage(const std::string &comName, const bool sort) const;
373 std::string argListOneLine() const;
374 std::string argListDetail() const;
375 std::string optList(const bool sort) const;
376 bool hasArgument() const { return (itemCount(ParserItem::ItemType::ARG)) ? true : false; }
377 bool hasOptions() const { return (itemCount(ParserItem::ItemType::OPT)) ? true : false; }
378 int itemCount(ParserItem::ItemType itemType) const;
379 int totalArgCount() const;
380 static std::string spaces(const size_t total);
381 std::string showParserItemTbl() const;
382
383 //------------------------------
384
385 std::string mDescription;
386
387 bool mErrorUnknownOption;
388 int mTotalArgCount;
389 std::vector<ParserItem> mParserItemTbl;
390};
