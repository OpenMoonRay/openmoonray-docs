38float
39DiskLight::planeDistance(const Vec3f &p, float time) const
40{
41 Plane pl = isMb() ? Plane(getPosition(time), getDirection(time)) : mRenderPlane;
42
43 float distance = pl.getDistance(p);
44 if (mSidedness == LightSidednessType::LIGHT_SIDEDNESS_REVERSE) {
45 distance = -distance;
46 } else if (mSidedness == LightSidednessType::LIGHT_SIDEDNESS_2_SIDED) {
47 distance = scene_rdl2::math::abs(distance);
48 }
49
50 return distance;
51}

53void
54DiskLight::getSpreadSquare(const Vec3f& renderP, const float renderDistance, const float time,
55 Vec3f& center, float& halfSideLength) const
56{
57 // get center and half length of square region of influence
58 center = xformPointRender2Local(renderP, time);
59 float renderSpaceSpreadLength = renderDistance * mTanSpreadTheta;
60 halfSideLength = xformRender2LocalScale(renderSpaceSpreadLength, time);
61}
62
63// Get rectangular bound of region of overlap between DiskLight and square region of influence.
64// The square is centered at localP and has a side length 2 * localLength.
65// We want to get the center of the rectangular bound and its width and height.
66// Returns true if the region overlaps with the light.
67bool
68getOverlapBounds(const Vec2f& localP, float localLength, Vec2f& center, float& width, float& height)
69{
70 float x0 = localP.x - localLength;
71 float x1 = localP.x + localLength;
72 float y0 = localP.y - localLength;
73 float y1 = localP.y + localLength;
74
75 float xc = math::clamp(0.f, x0, x1);
76 float yc = math::clamp(0.f, y0, y1);
77
78 // square is outside circle. Local radius of DiskLight is 1.
79 if (xc * xc + yc * yc >= 1.f) {
80 return false;
81 }
82
83 float xi = math::sqrt(1 - yc * yc);
84 float yi = math::sqrt(1 - xc * xc);
85 float minX = math::max(-xi, x0);
86 float maxX = math::min( xi, x1);
87 float minY = math::max(-yi, y0);
88 float maxY = math::min( yi, y1);
89
90 center.x = 0.5f * (maxX + minX);
91 center.y = 0.5f * (maxY + minY);
92 width = maxX - minX;
93 height = maxY - minY;
94
95 return true;
96}

98bool lightIsInsideSpread(const Vec3f& localP, const float localLength)
99{
100 // Is DiskLight entirely inside square centered at localP with side length 2 * localLength?
101 return localP.x - localLength < -1.f &&
102 localP.x + localLength > 1.f &&
103 localP.y - localLength < -1.f &&
104 localP.y + localLength > 1.f;
105}

120bool
121DiskLight::update(const Mat4d& world2render)
122{
123 MNRY_ASSERT(mRdlLight);
124
125 mOn = mRdlLight->get(scene_rdl2::rdl2::Light::sOnKey);
126 if (!mOn) {
127 return false;
128 }
129
130 updateVisibilityFlags();
131 updatePresenceShadows();
132 updateRayTermination();
133 updateTextureFilter();
134 updateMaxShadowDistance();
135
136 // Radius gets baked into the the mLocal2Render and other matrices such
137 // that the disk radius is always 1 in local light space.
138 const float radius = mRdlLight->get<scene_rdl2::rdl2::Float>(sRadiusKey);
139
140 // mSpread: 1 = 90 degree cone angle, 0 = 0 degree cone angle (cross section is square).
141 mSpread = math::clamp(mRdlLight->get<scene_rdl2::rdl2::Float>(sSpreadKey), math::sEpsilon, 1.f);
142 mTanSpreadTheta = math::tan(mSpread * math::sHalfPi);
143
144 UPDATE_ATTRS_CLEAR_RADIUS
145
146 const Mat4d l2w0 = mRdlLight->get(scene_rdl2::rdl2::Node::sNodeXformKey, /* rayTime = */ 0.f);
147 const Mat4d l2w1 = mRdlLight->get(scene_rdl2::rdl2::Node::sNodeXformKey, /* rayTime = */ 1.f);
148 Mat4f sclRot = sRotateX180;
149 sclRot[0][0] *= radius;
150 sclRot[1][1] *= radius;
151 sclRot[2][2] *= radius;
152 const Mat4f local2Render0 = sclRot * toFloat(l2w0 * world2render);
153 const Mat4f local2Render1 = sclRot * toFloat(l2w1 * world2render);
154 if (!updateParamAndTransforms(local2Render0, local2Render1, 1.0f, 1.0f)) {
155 return false;
156 }
157
158 // Compute render space quantities.
159 // TODO: mb scale radiance
160 const float renderRadiusSqr = mLocal2RenderScale[0] * mLocal2RenderScale[0];
161 mArea = sPi * renderRadiusSqr;
162 mInvArea = 1.0f / mArea;
163
164 // This term is used to adjust the pdf of sampling a point on the light in
165 // the case where we are using an ImageDistribution to handle the
166 // importance sampling of the image. In that case, the distribution class
167 // is setup with a Distribution2D::CIRCULAR mapping (see below), which
168 // zeroes-out all weights of the distribution outside of the disc inscribed
169 // within the [0, 1)^2 domain. The class also automatically normalizes the
170 // weights, as it should. In sample() we actually sample the square, not
171 // the disk and let the ImageDistribution class handle the rest. The pdf
172 // is therefore = distribution_pdf / square_area
173 mDistributionPdfScale = 1.0f / (4.0f * renderRadiusSqr);
174
175 mRadiance = computeLightRadiance(mRdlLight, scene_rdl2::rdl2::Light::sColorKey,
176 scene_rdl2::rdl2::Light::sIntensityKey, scene_rdl2::rdl2::Light::sExposureKey, sNormalizedKey,
177 sApplySceneScaleKey, mInvArea);
178 if (isBlack(mRadiance)) {
179 mOn = false;
180 return false;
181 }
182
183 // cache plane at rayTime == 0
184 mRenderPlane = Plane(getPosition(0.f), getDirection(0.f));
185
186 if (!updateImageMap(Distribution2D::CIRCULAR)) {
187 return false;
188 }
189
190 // Sidedness
191 mSidedness = static_cast<LightSidednessType>(mRdlLight->get<scene_rdl2::rdl2::Int>(sSidednessKey));
192 return true;
193}

197bool
198DiskLight::canIlluminate(const Vec3f p, const Vec3f *n, float time, float radius,
199 const LightFilterList* lightFilterList) const
200{
201 MNRY_ASSERT(mOn);
202
203 // Cull points which are completely on the backside of the light, taking sidedness into account
204 const float threshold = sEpsilon - radius;
205 if ((mSidedness != LightSidednessType::LIGHT_SIDEDNESS_2_SIDED) && (planeDistance(p, time) < threshold)) {
206 return false;
207 }
208
209 // Cull lights which are completely on the backside of the point.
210 // See image at lib/rendering/pbr/doc/DiskLightDiagram.jpg for a 2D breakdown
211 // of what's going on here.
212 // We make used of the fact that the point normal, light normal, and
213 // closest point on the disk to the plane of the point all lie on the
214 // same plane.
215 if (n) {
216
217 // Transform normal from render space into light space.
218 Vec3f localN = xformNormalRender2LocalRot(*n, time);
219
220 // Project onto local xy plane in light local space. The disk light
221 // itself is a unit circle on the xy plane, so this will collapse
222 // the point onto that unit circle.
223 localN.z = 0.0f;
224
225 // By normalizing the 2D projected point, we are putting it on the
226 // circumference of the disk. If diskP couldn't be normalized
227 // it means that the disk light is parallel to the cull plane of point
228 // p, in which case any point on the disk is as good as any other for the
229 // final plane check, so we just pick (1, 0).
230 Vec2f diskP = asVec2(localN);
231 float rcpLength = rsqrt(dot(diskP, diskP));
232 diskP = (rcpLength < float(one_over_epsilon) ? diskP * rcpLength : Vec2f(1.f, 0.f));
233
234 // diskP will now be the point on the unit circle which is furthest in the
235 // direction of n, transform it back into render space to do the final plane
236 // check.
237
238 Vec3f renderP = xformPointLocal2Render(Vec3f(diskP.x, diskP.y, 0.f), time);
239 bool canIllum = dot(*n, renderP - p) > threshold;
240 if (!canIllum) return false;
241 }
242
243 if (lightFilterList) {
244 return canIlluminateLightFilterList(lightFilterList,
245 { getPosition(time),
246 xformLocal2RenderScale(1.0f, time),
247 p, getXformRender2Local(time, lightFilterList->needsLightXform()),
248 radius, time
249 });
250 }
251
252 return true;
253}

273// Computing a tight bounding box for a disk light
274// -----------------------------------------------
275//
276// In local space, a point (x,y,z) on the perimeter of the disk can be expressed parametrically as
277//
278// x = cos t
279// y = sin t
280// z = 0
281//
282// with t ranging from 0 to 2pi.
283// Note that this is a unit circle, because the disk's radius is baked into the local2render matrix.
284// Suppose the local2render matrix is (Mij) with i=0,1,2,3 and j=0,1,2,3.
285// Then the perimeter point (x,y,z) transforms to (x',y',z') in render space, where
286//
287// x' = M00 cos t + M10 sin t + M30
288// y' = M01 cos t + M11 sin t + M31
289// z' = M02 cos t + M12 sin t + M32
290//
291// Consider just the x' coordinate. Its min and max occur at points where dx'/dt=0. Differentiating the above gives
292//
293// dx'/dt = -M00 sin t + M10 cos t
294//
295// If this equals zero then
296//
297// tan t = M10 / M00
298//
299// Or, using standard trig identities,
300//
301// cos^2 t = M00^2 / (M00^2 + M10^2)
302// sin^2 t = M10^2 / (M00^2 + M10^2)
303//
304// Taking square roots, and ensuring the chosen combination of signs preserves the sign of tan t,
305//
306// (cos t, sin t) = +/- (M00, M10) / sqrt(M00^2 + M10^2)
307//
308// Putting these values into the expression for x' gives
309//
310// x' = M30 +/- (M00^2 + M10^2) / sqrt(M00^2 + M10^2)
311// = M30 +/- sqrt(M00^2 + M10^2)
312//
313// The expressions for the min & max y' and z' are similar, and the 3 together can be expressed compactly
314// in terms of Vec3f operations.
315
316BBox3f
317DiskLight::getBounds() const
318{
319 const Vec3f pos = getPosition(0.f);
320 const Vec3f xRow = xformPointLocal2Render(Vec3f(1., 0., 0.), 0.f);
321 const Vec3f yRow = xformPointLocal2Render(Vec3f(0., 1., 0.), 0.f);
322 const Vec3f halfDims = max(sqrt(xRow*xRow + yRow*yRow), Vec3f(sEpsilon)); // ensure non-degeneracy
323 BBox3f bounds(pos - halfDims, pos + halfDims);
324
325 if (isMb()) {
326 const Vec3f pos = getPosition(1.f);
327 const Vec3f xRow = xformPointLocal2Render(Vec3f(1., 0., 0.), 1.f);
328 const Vec3f yRow = xformPointLocal2Render(Vec3f(0., 1., 0.), 1.f);
329 const Vec3f halfDims = max(sqrt(xRow*xRow + yRow*yRow), Vec3f(sEpsilon)); // ensure non-degeneracy
330 bounds.extend(BBox3f(pos - halfDims, pos + halfDims));
331 }
332
333 return bounds;
334}

336bool
337DiskLight::intersect(const Vec3f &p, const Vec3f *n, const Vec3f &wi, float time,
338 float maxDistance, LightIntersection &isect) const
339{
340 // Get local direction and position
341 Vec3f localWi = xformVectorRender2LocalRot(wi, time);
342 Vec3f localP = xformVectorRender2Local(p - getPosition(time), time);
343
344 // Trivial rejection of pos/dir/sidedness cases
345 if (rejectPosDirSidedness(localP.z, localWi.z)) return false;
346
347 // Reject if intersection with plane is beyond maxDistance
348 float localDistance = -localP.z / localWi.z;
349 float renderDistance = xformLocal2RenderScale(localDistance, time);
350 if (renderDistance > maxDistance) {
351 return false;
352 }
353
354 // Reject if intersection is outside disk
355 Vec3f localHit = localP + localWi * localDistance;
356 if (asVec2(localHit).lengthSqr() > 1.0f) {
357 return false;
358 }
359
360 if (mSpread < 1.f) {
361 // Reject if intersection is outside spread
362 Vec3f absLocalWi = scene_rdl2::math::abs(localWi);
363 if (max(absLocalWi.x, absLocalWi.y) > absLocalWi.z * mTanSpreadTheta) {
364 return false;
365 }
366
367 // compute pdf
368 // get half side length of square spread region
369 float spreadLength = localP.z * mTanSpreadTheta;
370 Vec2f center;
371 float width, height;
372 getOverlapBounds(Vec2f(localP.x, localP.y), spreadLength, center, width, height);
373 // store pdf if not rejected
374 isect.pdf = 1.f / (xformLocal2RenderScale(width, time) * xformLocal2RenderScale(height, time));
375 }
376
377 // Fill in isect members
378 // We will only reach this far if p is on the illuminated side of the light's plane (either side, for 2-sided).
379 // The light's mDirection is its local z-axis, which is only guaranteed to concide with the normal for lights
380 // with regular sidedness. In the general case, the correct normal is obtained by ensuring the ray is travelling
381 // against the normal, i.e. the ray's direction has negative local z-coordinate.
382 Vec3f normal = getDirection(time);
383 isect.N = (localWi.z < 0.0f) ? normal : -normal;
384 isect.uv = (mDistribution ? asVec2(localHit) * mUvScale + mUvOffset
385 : math::zero);
386 isect.distance = renderDistance;
387
388 return true;
389}

391bool
392DiskLight::sample(const Vec3f &p, const Vec3f *n, float time, const Vec3f& r,
393 Vec3f &wi, LightIntersection &isect, float rayDirFootprint) const
394{
395 MNRY_ASSERT(mOn);
396
397 float r1 = r[0];
398 float r2 = r[1];
399
400 // We don't cast light on points behind the light. Also we can't always
401 // rely on canIlluminate() to test for that case, as it's an optional
402 // culling function. canIlluminate() can also be passed a non-zero radius,
403 // defining a p-region partially in front and partially behind the light.
404 // So we have to test for it per sample, here too.
405 float distance = planeDistance(p, time);
406 if ((mSidedness != LightSidednessType::LIGHT_SIDEDNESS_2_SIDED) && (distance < sEpsilon)) {
407 return false;
408 }
409
410 Vec3f localHit;
411
412 // There are 3 possible sampling modes:
413 // 1) If there is an image distribution, sample that.
414 // 2) If there is no image distribution, and the spread parameter is less than 1 and
415 // we are in a good region to sample the spread, then sample that region.
416 // 3) Otherwise, sample the shape of the light.
417 if (mDistribution) {
418 // Sample the image distribution if any
419
420 // This type of technique is bad for for vector processors in general
421 // since some lanes will go idle as the rejection loop still needs to
422 // iterate for other lanes. It should work fine here though because the
423 // distribution has been prebuilt with a low probability of sampling
424 // outside of the inscribed circle by using ImageDistribution::CIRCULAR.
425 // This should ensure we only loop rarely.
426 const unsigned maxTries = 8;
427 unsigned attempts = 0;
428 while(attempts != maxTries) {
429 mDistribution->sample(r1, r2, 0, &isect.uv, nullptr, mTextureFilter);
430 float x = isect.uv.x * 2.0f - 1.0f;
431 float y = isect.uv.y * 2.0f - 1.0f;
432 if (x*x + y*y < 1.0f) {
433 break;
434 }
435
436 // Generate new random numbers based on these ones.
437 // TODO: this is hacky code, do a better job of generating
438 // further random numbers
439 r1 = scene_rdl2::math::fmod(r1 + 0.789f, 1.0f);
440 r2 = scene_rdl2::math::fmod(r2 + 0.331f, 1.0f);
441
442 attempts++;
443 }
444
445 if (attempts == maxTries) {
446 return false;
447 }
448
449 localHit = Vec3f((isect.uv.x - mUvOffset.x) / mUvScale.x,
450 (isect.uv.y - mUvOffset.y) / mUvScale.y,
451 0.0f);
452
453 if (mSpread < 1.f) {
454 // Reject samples that are outside of the spread region.
455 Vec3f localP = xformPointRender2Local(p, time);
456 Vec3f localWi = localHit - localP;
457 Vec3f absLocalWi = scene_rdl2::math::abs(localWi);
458 if (max(absLocalWi.x, absLocalWi.y) > absLocalWi.z * mTanSpreadTheta) {
459 return false;
460 }
461 }
462
463 } else if (mSpread < 1.f) {
464 // Setup parameters of sampling the spread.
465 float spreadLength;
466 Vec3f localP;
467 getSpreadSquare(p, distance, time, localP, spreadLength);
468
469 if (!lightIsInsideSpread(localP, spreadLength)) {
470 // Uniformly sample the rectangular overlapping region between the spread region and the DiskLight.
471 Vec2f center;
472 float width, height;
473 if (!getOverlapBounds(Vec2f(localP.x, localP.y), spreadLength, center, width, height)) {
474 return false;
475 }
476
477 // sample rectangular region of influence
478 localHit = Vec3f((0.5f - r1) * width + center.x,
479 (0.5f - r2) * height + center.y,
480 0.0f);
481
482 // reject samples that are outside light
483 if (asVec2(localHit).lengthSqr() > 1.f) {
484 return false;
485 }
486 isect.uv = zero;
487 // store pdf
488 isect.pdf = 1.f / (xformLocal2RenderScale(width, time) * xformLocal2RenderScale(height, time));
489 } else {
490 // Otherwise uniformly distribute samples on disk area
491 isect.uv = zero;
492 squareSampleToCircle(r1, r2, &localHit.x, &localHit.y);
493 localHit.z = 0.0f;
494 isect.pdf = mInvArea;
495 }
496
497 } else {
498 // Otherwise uniformly distribute samples on disk area
499 isect.uv = zero;
500 squareSampleToCircle(r1, r2, &localHit.x, &localHit.y);
501 localHit.z = 0.0f;
502 }
503
504 Vec3f renderHit = xformPointLocal2Render(localHit, time);
505
506 // Compute wi and d
507 wi = renderHit - p;
508 if (n && dot(*n, wi) < sEpsilon) {
509 return false;
510 }
511 isect.distance = length(wi);
512 if (isect.distance < sEpsilon) {
513 return false;
514 }
515 wi *= rcp(isect.distance);
516
517 // We will only reach this far if p is on the illuminated side of the light's plane (either side, for 2-sided).
518 // The light's mDirection is its local z-axis, which is only guaranteed to concide with the normal for lights
519 // with regular sidedness. The correct normal is obtained by ensuring the ray is travelling against the normal,
520 // i.e. the ray's direction has negative dot product with it.
521 Vec3f normal = getDirection(time);
522 isect.N = (dot(normal, wi) < 0.0f) ? normal : -normal;
523
524 return true;
525}

527Color
528DiskLight::eval(mcrt_common::ThreadLocalState* tls, const Vec3f &wi, const Vec3f &p, const LightFilterRandomValues& filterR, float time,
529 const LightIntersection &isect, bool fromCamera, const LightFilterList *lightFilterList, float rayDirFootprint,
530 float *pdf) const
531{
532 MNRY_ASSERT(mOn);
533
534 // Point sample the texture
535 // TODO: Use proper filtering with ray differentials and mip-mapping.
536 Color radiance = mRadiance;
537
538 if (mSpread < 1.f) {
539 // modify radiance as if it were physically blocked by an eggcrate light filter.
540 Vec3f localWi = xformVectorRender2LocalRot(wi, time);
541 radiance *= math::max(1.f - math::abs(localWi.x / (localWi.z * mTanSpreadTheta)), 0.f) *
542 math::max(1.f - math::abs(localWi.y / (localWi.z * mTanSpreadTheta)), 0.f);
543 }
544
545 if (mDistribution) {
546 radiance *= mDistribution->eval(isect.uv[0], isect.uv[1], 0, mTextureFilter);
547 }
548
549 if (pdf) {
550 if (mDistribution) {
551 // we sampled the image distribution
552 *pdf = mDistribution->pdf(isect.uv[0], isect.uv[1], 0, mTextureFilter) *
553 mDistributionPdfScale;
554 } else if (mSpread < 1.f) {
555 // grab the stored pdf.
556 *pdf = isect.pdf;
557 } else {
558 // we uniformly sampled the disk
559 *pdf = mInvArea;
560 }
561 *pdf *= areaToSolidAngleScale(wi, isect.N, isect.distance);
562 }
563
564 if (lightFilterList) {
565 evalLightFilterList(lightFilterList,
566 { tls, &isect, getPosition(time),
567 getDirection(time), p,
568 filterR, time,
569 getXformRender2Local(time, lightFilterList->needsLightXform()),
570 wi
571 },
572 radiance);
573 }
574
575 return radiance;
576}

578Vec3f
579DiskLight::getEquiAngularPivot(const Vec3f& r, float time) const
580{
581 Vec3f hit;
582 Vec2f uv;
583 float r1 = r[0];
584 float r2 = r[1];
585 // Sample the image distribution if any
586 if (mDistribution) {
587 // This type of technique is bad for for vector processors in general
588 // since some lanes will go idle as the rejection loop still needs to
589 // iterate for other lanes. It should work fine here though because the
590 // distribution has been prebuilt with a low probability of sampling
591 // outside of the inscribed circle by using ImageDistribution::CIRCULAR.
592 // This should ensure we only loop rarely.
593 const unsigned maxTries = 8;
594 unsigned attempts = 0;
595 while(attempts != maxTries) {
596 mDistribution->sample(r1, r2, 0, &uv, nullptr, mTextureFilter);
597 float x = uv.x * 2.0f - 1.0f;
598 float y = uv.y * 2.0f - 1.0f;
599 if (x*x + y*y < 1.0f) {
600 break;
601 }
602 // Generate new random numbers based on these ones.
603 // TODO: this is hacky code, do a better job of generating
604 // further random numbers
605 r1 = scene_rdl2::math::fmod(r1 + 0.789f, 1.0f);
606 r2 = scene_rdl2::math::fmod(r2 + 0.331f, 1.0f);
607 attempts++;
608 }
609 if (attempts == maxTries) {
610 return getPosition(time);
611 }
612 hit = Vec3f((uv[0] - mUvOffset.x) / mUvScale.x,
613 (uv[1] - mUvOffset.y) / mUvScale.y,
614 0.0f);
615 } else {
