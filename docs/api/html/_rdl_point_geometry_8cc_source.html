46moonray::geom::LayerAssignmentId
47createPerPointAssignmentId(const RdlPointGeometry* rdlPointGeometry,
48 const scene_rdl2::rdl2::Layer* rdlLayer,
49 const size_t pointCount) {
50
51 // part_list: list of part names
52 // part_indices: part index for each point in vertex_list
53 const std::vector<scene_rdl2::rdl2::Vec3f> &points = rdlPointGeometry->get(attrPos0);
54 const std::vector<scene_rdl2::rdl2::Int> &partIndices = rdlPointGeometry->get(attrPartIndices);
55 const std::vector<scene_rdl2::rdl2::String> &partList = rdlPointGeometry->get(attrPartList);
56
57 // default assignment id
58 const int defaultAssignmentId = rdlLayer->getAssignmentId(rdlPointGeometry,
59 sDefaultPartName);
60
61 if (partList.empty()) {
62 return moonray::geom::LayerAssignmentId(defaultAssignmentId);
63 }
64
65 // input validation, if we have parts
66 if (partIndices.size() && points.size() != partIndices.size()) {
67 rdlPointGeometry->warn("part_indices size does not match vertex_list size");
68 }
69
70 // loop to assign parts
71 bool warn = false;
72 std::vector<int> pointAssignmentIds(points.size(), defaultAssignmentId);
73 for (size_t i = 0; i < partIndices.size() && i < points.size(); ++i) {
74 const int partIndex = partIndices[i];
75 if (partIndex >= 0 && size_t(partIndex) < partList.size()) {
76 const int layerAssignmentId = rdlLayer->getAssignmentId(
77 rdlPointGeometry, partList[partIndex]);
78 pointAssignmentIds[i] = layerAssignmentId;
79 } else if (!warn) {
80 rdlPointGeometry->warn("part_indices contains values outside of part_list length");
81 warn = true;
82 }
83 }
84
85 return moonray::geom::LayerAssignmentId(std::move(pointAssignmentIds));
86}

108void
109RdlPointProcedural::generate(
110 const GenerateContext &generateContext,
111 const shading::XformSamples &parent2render)
112{
113 const scene_rdl2::rdl2::Geometry *rdlGeometry = generateContext.getRdlGeometry();
114
115 const RdlPointGeometry *rdlPointGeometry =
116 static_cast<const RdlPointGeometry*>(rdlGeometry);
117
118 const scene_rdl2::rdl2::Layer *rdlLayer = generateContext.getRdlLayer();
119
120 shading::PrimitiveAttributeTable primitiveAttributeTable;
121
122 auto& procPosList0 = rdlPointGeometry->get(attrPos0);
123 auto& procPosList1 = rdlPointGeometry->get(attrPos1);
124 auto& procVelList0 = rdlPointGeometry->get(attrVel0);
125 auto& procVelList1 = rdlPointGeometry->get(attrVel1);
126 auto& procAccList0 = rdlPointGeometry->get(attrAcc);
127
128 const size_t vertCount = procPosList0.size();
129
130 bool pos1Valid = sizeCheck(rdlPointGeometry, getName(rdlPointGeometry, attrPos1), procPosList1.size(), vertCount);
131 bool vel0Valid = sizeCheck(rdlPointGeometry, getName(rdlPointGeometry, attrVel0), procVelList0.size(), vertCount);
132 bool vel1Valid = sizeCheck(rdlPointGeometry, getName(rdlPointGeometry, attrVel1), procVelList1.size(), vertCount);
133 bool acc0Valid = sizeCheck(rdlPointGeometry, getName(rdlPointGeometry, attrAcc), procAccList0.size(), vertCount);
134
135 // Fall back on static case if we don't have sufficient data for requested mb type
136 int numPosSamples = 1;
137 int numVelSamples = 0;
138 int numAccSamples = 0;
139
140 bool err = false;
141 scene_rdl2::rdl2::MotionBlurType motionBlurType =
142 static_cast<scene_rdl2::rdl2::MotionBlurType>(rdlPointGeometry->get(attrMotionBlurType));
143 switch (motionBlurType) {
144
145 case scene_rdl2::rdl2::MotionBlurType::STATIC:
146 {
147 break;
148 }
149
150 case scene_rdl2::rdl2::MotionBlurType::VELOCITY:
151 {
152 if (vel0Valid) {
153 numVelSamples = 1;
154 } else {
155 err = true;
156 }
157 break;
158 }
159
160 case scene_rdl2::rdl2::MotionBlurType::FRAME_DELTA:
161 {
162 if (pos1Valid) {
163 numPosSamples = 2;
164 } else {
165 err = true;
166 }
167 break;
168 }
169
170 case scene_rdl2::rdl2::MotionBlurType::ACCELERATION:
171 {
172 if (vel0Valid && acc0Valid) {
173 numVelSamples = 1;
174 numAccSamples = 1;
175 } else {
176 err = true;
177 }
178 break;
179 }
180
181 case scene_rdl2::rdl2::MotionBlurType::HERMITE:
182 {
183 if (pos1Valid && vel0Valid && vel1Valid) {
184 numPosSamples = 2;
185 numVelSamples = 2;
186 } else {
187 err = true;
188 }
189 break;
190 }
191
192 case scene_rdl2::rdl2::MotionBlurType::BEST:
193 {
194 if (pos1Valid && vel0Valid && vel1Valid) {
195 // use Hermite mb type
196 numPosSamples = 2;
197 numVelSamples = 2;
198 } else if (vel0Valid && acc0Valid) {
199 // use acceleration mb type
200 numVelSamples = 1;
201 numAccSamples = 1;
202 } else if (pos1Valid) {
203 // use frame delta mb type
204 numPosSamples = 2;
205 } else if (vel0Valid) {
206 // use velocity mb type
207 numVelSamples = 1;
208 }
209 // else just keep static mb type
210 break;
211 }
212
213 default:
214 {
215 err = true;
216 break;
217 }
218
219 } // end of switch statement
220
221 if (err) {
222 rdlPointGeometry->warn("Insufficient data for requested motion blur type. "
223 "Falling back to static case.");
224 }
225
226 // Copy vertices
227 Points::VertexBuffer vertices(vertCount, numPosSamples);
228 for (size_t i = 0; i < vertCount; i++) {
229 const auto& p = procPosList0[i];
230 vertices(i, 0) = Vec3f(p[0], p[1], p[2]);
231 }
232 if (numPosSamples == 2) {
233 for (size_t i = 0; i < vertCount; i++) {
234 const auto& p = procPosList1[i];
235 vertices(i, 1) = Vec3f(p[0], p[1], p[2]);
236 }
237 }
238
239 // Add velocity data
240 if (numVelSamples > 0) {
241 const float velocityScale = rdlPointGeometry->get(attrVelocityScale);
242 std::vector<std::vector<Vec3f>> velocities;
243 velocities.push_back(std::vector<Vec3f>(procVelList0.begin(), procVelList0.end()));
244 for (Vec3f& velocity : velocities.back()) {
245 velocity *= velocityScale;
246 }
247 if (numVelSamples == 2) {
248 velocities.push_back(std::vector<Vec3f>(procVelList1.begin(), procVelList1.end()));
249 for (Vec3f& velocity : velocities.back()) {
250 velocity *= velocityScale;
251 }
252 }
253 primitiveAttributeTable.addAttribute(shading::StandardAttributes::sVelocity,
254 shading::RATE_VERTEX, std::move(velocities));
255 }
256
257 // Add acceleration data
258 if (numAccSamples > 0) {
259 std::vector<Vec3f> accelerations(procAccList0.begin(), procAccList0.end());
260 primitiveAttributeTable.addAttribute(shading::StandardAttributes::sAcceleration,
261 shading::RATE_VERTEX, std::move(accelerations));
262 }
263
264 // radius buffer
265 const std::vector<float> &rv = rdlPointGeometry->get(attrRadius);
266 Points::RadiusBuffer radius(vertCount);
267 const moonray::geom::RateCounts rates{rdlPointGeometry->get(attrPartList).size(), 0, 0, vertCount, 0};
268 switch (pickRate(rdlPointGeometry, getName(rdlPointGeometry, attrRadius), rv.size(), rates)) {
269 case AttributeRate::RATE_UNKNOWN:
270 radius.assign(vertCount, 0.5f);
271 break;
272 case AttributeRate::RATE_CONSTANT:
273 radius.assign(vertCount, rv[0]);
274 break;
275 case AttributeRate::RATE_PART: {
276 const std::vector<scene_rdl2::rdl2::Int> &partIndices = rdlPointGeometry->get(attrPartIndices);
277 radius.reserve(vertCount);
278 for (size_t i = 0; i < vertCount; ++i)
279 radius.emplace_back(rv[partIndices[i]]);
280 break;}
281 default:
282 radius.assign(rv.begin(), rv.begin()+vertCount);
283 break;
284 }
285
286 // layer assignments
287 LayerAssignmentId layerAssignmentId = createPerPointAssignmentId(
288 rdlPointGeometry, rdlLayer, vertCount);
289
290 // primitive attributes
291 scene_rdl2::rdl2::PrimitiveAttributeFrame primitiveAttributeFrame = static_cast<scene_rdl2::rdl2::PrimitiveAttributeFrame>(
292 rdlPointGeometry->get(attrPrimitiveAttributeFrame));
293 bool useFirstFrame = (primitiveAttributeFrame != scene_rdl2::rdl2::PrimitiveAttributeFrame::SECOND_MOTION_STEP);
294 bool useSecondFrame = (primitiveAttributeFrame != scene_rdl2::rdl2::PrimitiveAttributeFrame::FIRST_MOTION_STEP);
295 moonray::geom::processArbitraryData(rdlPointGeometry, attrPrimitiveAttributes, primitiveAttributeTable,
296 rates, useFirstFrame, useSecondFrame);
297
298 std::unique_ptr<Points> primitive = createPoints(
299 std::move(vertices), std::move(radius), std::move(layerAssignmentId),
300 std::move(primitiveAttributeTable));
301
302 if (primitive) {
303 primitive->setCurvedMotionBlurSampleCount(rdlPointGeometry->get(attrCurvedMotionBlurSampleCount));
304
305 // may need to convert the primitive to instance to handle
306 // rotation motion blur
307 std::unique_ptr<Primitive> p = convertForMotionBlur(
308 generateContext, std::move(primitive),
309 (rdlPointGeometry->get(attrUseRotationMotionBlur) &&
310 parent2render.size() > 1));
311 addPrimitive(std::move(p),
312 generateContext.getMotionBlurParams(), parent2render);
313 }
314}
