20std::string
21MinUniqueSuffixMap::PathData::rotatePathComponent(const std::string& oldDestPath)
22{
23 // Split the previous path prefix to obtain the last path component.
24 auto components = util::splitPath(mPathPrefix);
25
26 // The new destination path prepends the last path component in front of
27 // the old prefix path, replacing slashes with underscores.
28 std::string newDestPath(std::move(components.second));
29 newDestPath.append('_' + oldDestPath);
30
31 // The path prefix is now the prefix of path split.
32 mPathPrefix = std::move(components.first);
33
34 return newDestPath;
35}

37std::pair<std::string, MinUniqueSuffixMap::PathData>
38MinUniqueSuffixMap::PathData::relocate(const std::string& oldDestPath)
39{
40 // Rotate the path to resolve the current conflict.
41 std::string newDestPath(rotatePathComponent(oldDestPath));
42
43 // The relocated PathData can steal our path prefix and source path.
44 PathData newPathData(std::move(mPathPrefix), std::move(mSourcePath));
45
46 // This entry is now dead, and shouldn't be included in the output. We
47 // keep it around so that it still generates collisions, though.
48 mDead = true;
49
50 return std::make_pair(std::move(newDestPath), std::move(newPathData));
51}

72MinUniqueSuffixMap::IteratorWrapper
73MinUniqueSuffixMap::insert(const std::string& sourcePath)
74{
75 // Skip duplicates.
76 auto iter = mSourceToDest.find(sourcePath);
77 if (iter != mSourceToDest.end()) {
78 return IteratorWrapper(*this, iter);
79 }
80
81 // Top level insert into the minimum unique suffix map.
82 auto components = util::splitPath(sourcePath);
83 return IteratorWrapper(*this, insert(std::make_pair(
84 std::move(components.second),
85 PathData(std::move(components.first), sourcePath))));
86}

94MinUniqueSuffixMap::SourceToDestMap::const_iterator
95MinUniqueSuffixMap::insert(std::pair<std::string, PathData> newEntry)
96{
97 // Attempt to insert.
98 auto result = mDestToSource.emplace(newEntry);
99
100 if (result.second) { // Base case for recursion.
101 // Insertion succeeded, update the source -> dest map.
102 auto update = mSourceToDest.emplace(newEntry.second.mSourcePath,
103 result.first);
104 if (!update.second) {
105 // Insert failed (it already exists), so update it instead.
106 update.first->second = result.first;
107 }
108
109 return update.first;
110 }
111
112 // Insertion failed due to a conflict, so we need to resolve it.
113
114 // If the item we collided with is not dead (has not been relocated)...
115 auto& conflictingEntry = result.first->second;
116 if (!conflictingEntry.mDead) {
117 // ...relocate the conflicting entry (recursively).
118 insert(conflictingEntry.relocate(newEntry.first));
119 }
120
121 // Rotate the path of the new entry and (recursively) try again.
122 newEntry.first = newEntry.second.rotatePathComponent(newEntry.first);
123 return insert(newEntry);
124}
