54void
55BinaryReader::fromFile(const std::string& filename)
56{
57 // Create an input file stream.
58 std::ifstream in(filename.c_str(), std::ios::binary);
59 if (!in) {
60 std::stringstream errMsg;

62 " an RDL2 binary reader.";

64 }
65 fromStream(in);

68void
69BinaryReader::fromStream(std::istream& input)
70{
71 // Read the manifest length from the stream and convert to native byte order.
72 uint64_t manifestLen;
73 input.read(reinterpret_cast<char*>(&manifestLen), sizeof(uint64_t));
74 manifestLen = be64toh(manifestLen);
75
76 // Read the payload length from the stream and convert to native byte order.
77 uint64_t payloadLen;
78 input.read(reinterpret_cast<char*>(&payloadLen), sizeof(uint64_t));
79 payloadLen = be64toh(payloadLen);
80
81 // Read the manifest.
82 std::string manifest(manifestLen, '\0');
83 input.read(&(manifest[0]), manifestLen);
84
85 // Read the payload.
86 std::string payload(payloadLen, '\0');
87 input.read(&(payload[0]), payloadLen);
88
89 fromBytes(manifest, payload);
90}

92void
93BinaryReader::fromBytes(const std::string& manifest, const std::string& payload)
94{
95 Slice manifestBytes(manifest);
96 Slice payloadBytes(payload);
97
98 // Read the manifest.
99 RecordInfoVector records;
100 readManifest(manifestBytes, records);
101
102 // Loop over records in the manifest and read each out of the payload.
103 // (This could be parallelized fairly easily if it's a bottleneck.)
104 for (RecordInfoVector::const_iterator iter = records.begin(); iter != records.end(); ++iter) {
105 switch (iter->mType) {
106 case SCENE_OBJECT :
107 {
108 std::stringstream errMsg;
109 errMsg << "SCENE_OBJECT payload type is nolonger supported";
110 throw except::TypeError(errMsg.str());
111 }
112 break;
113
114 case SCENE_OBJECT_2 :
115 readSceneObject(Slice(payloadBytes, iter->mOffset, iter->mSize));
116 break;
117
118 default:
119 {
120 std::stringstream errMsg;
121 errMsg << "Encountered unknown payload type '" << iter->mType <<
122 "' in manifest while parsing RDL2 binary file.";
123 throw except::TypeError(errMsg.str());
124 }
125 break;
126 }
127 }
128}

130void
131BinaryReader::readManifest(Slice bytes, RecordInfoVector& info)
132{
133 const char *ptr = static_cast<const char *>(bytes.getData());
134 ValueContainerDeq vContainerDeq(ptr, bytes.getLength());
135
136 size_t size;
137 vContainerDeq.deqVLSizeT(size);
138 ptrdiff_t offset = 0;
139 for (size_t i = 0; i < size; ++i) {
140 unsigned int typeUInt;
141 size_t sizeObj;
142 vContainerDeq.deqVLUInt(typeUInt);
143 vContainerDeq.deqVLSizeT(sizeObj);
144 info.emplace_back(static_cast<RecordType>(typeUInt), offset, sizeObj);
145 offset += sizeObj;
146 }
147}

149void
150BinaryReader::readSceneObject(Slice bytes)
151{
152 const char *ptr = static_cast<const char *>(bytes.getData());
153 ValueContainerDeq vContainerDeq(ptr, bytes.getLength());
154
155 std::string klassName;
156 std::string objName;
157 vContainerDeq.deqString(klassName);
158 vContainerDeq.deqString(objName);
159
160 SceneObject* sceneObject = nullptr;
161 try {
162 // Create the SceneObject.
163 sceneObject = mContext.createSceneObject(klassName, objName);
164 } catch (except::IoError& e) {
165 // Couldn't load DSO.
166 std::string msg = util::buildString(objName, ": ", e.what());
167 if (mWarningsAsErrors) {
168 throw except::IoError(msg); // Rethrow with more information.
169 } else {
170 logging::Logger::warn(msg);
171 }
172 return;
173 }
174
175 // Unpack the data into the object.
176 unpackSceneObject(vContainerDeq, *sceneObject);
177}

179void
180BinaryReader::unpackLayer(BinaryReaderLayerUnpackStrings &layerStrVectors, Layer &layer) const
181{
182 // The values represent the vectors
183 StringVector &geomKlassName = layerStrVectors.mGeomKlassName;
184 StringVector &geomObjName = layerStrVectors.mGeomObjName;
185 StringVector &partsName = layerStrVectors.mPartName;
186 StringVector &materialKlassName = layerStrVectors.mMaterialKlassName;
187 StringVector &materialObjName = layerStrVectors.mMaterialObjName;
188 StringVector &lightSetKlassName = layerStrVectors.mLightSetKlassName;
189 StringVector &lightSetObjName = layerStrVectors.mLightSetObjName;
190 StringVector &lightFilterSetKlassName = layerStrVectors.mLightFilterSetKlassName;
191 StringVector &lightFilterSetObjName = layerStrVectors.mLightFilterSetObjName;
192 StringVector &shadowSetKlassName = layerStrVectors.mShadowSetKlassName;
193 StringVector &shadowSetObjName = layerStrVectors.mShadowSetObjName;
194 StringVector &shadowReceiverSetKlassName = layerStrVectors.mShadowReceiverSetKlassName;
195 StringVector &shadowReceiverSetObjName = layerStrVectors.mShadowReceiverSetObjName;
196 StringVector &displacementKlassName = layerStrVectors.mDisplacementKlassName;
197 StringVector &displacementObjName = layerStrVectors.mDisplacementObjName;
198 StringVector &volumeShaderKlassName = layerStrVectors.mVolumeShaderKlassName;
199 StringVector &volumeShaderObjName = layerStrVectors.mVolumeShaderObjName;
200
201 for (size_t i = 0; i < geomKlassName.size(); ++i) {
202 // Unpack the geometry
203 SceneObject *geomObj = nullptr;
204 if (!geomKlassName[i].empty() && !geomObjName[i].empty()) {
205 geomObj = mContext.createSceneObject(geomKlassName[i], geomObjName[i]);
206 }
207
208 // Unpack the material, might be null
209 SceneObject* materialObj = nullptr;
210 if (!materialKlassName.empty() && !materialKlassName[i].empty() && !materialObjName[i].empty()) {
211 materialObj = mContext.createSceneObject(materialKlassName[i], materialObjName[i]);
212 }
213
214 // Unpack the lightset
215 SceneObject* lightSetObj = nullptr;
216 if (!lightSetKlassName.empty() && !lightSetKlassName[i].empty() && !lightSetObjName[i].empty()) {
217 lightSetObj = mContext.createSceneObject(lightSetKlassName[i], lightSetObjName[i]);
218 }
219
220 // Unpack the lightfilterset
221 SceneObject* lightFilterSetObj = nullptr;
222 if (!lightFilterSetKlassName.empty() && !lightFilterSetKlassName[i].empty() &&
223 !lightFilterSetObjName[i].empty()) {
224 lightFilterSetObj = mContext.createSceneObject(lightFilterSetKlassName[i], lightFilterSetObjName[i]);
225 }
226
227 // Unpack the shadowset
228 SceneObject* shadowSetObj = nullptr;
229 if (!shadowSetKlassName.empty() && !shadowSetKlassName[i].empty() &&
230 !shadowSetObjName[i].empty()) {
231 shadowSetObj = mContext.createSceneObject(shadowSetKlassName[i], shadowSetObjName[i]);
232 }
233
234 // Unpack the shadowreceiverset
235 SceneObject* shadowReceiverSetObj = nullptr;
236 if (!shadowReceiverSetKlassName.empty() && !shadowReceiverSetKlassName[i].empty()
237 && !shadowReceiverSetObjName[i].empty()) {
238 shadowReceiverSetObj = mContext.createSceneObject(shadowReceiverSetKlassName[i],
239 shadowReceiverSetObjName[i]);
240 }
241
242 // Unpack the displacement, might be null
243 SceneObject* displacementObj = nullptr;
244 if (!displacementKlassName.empty() && !displacementKlassName[i].empty() && !displacementObjName[i].empty()) {
245 displacementObj = mContext.createSceneObject(displacementKlassName[i], displacementObjName[i]);
246 }
247
248 // Unpack the volumeShader, might be null
249 SceneObject* volumeShaderObj = nullptr;
250 if (!volumeShaderKlassName.empty() && !volumeShaderKlassName[i].empty() && !volumeShaderObjName[i].empty()) {
251 volumeShaderObj = mContext.createSceneObject(volumeShaderKlassName[i], volumeShaderObjName[i]);
252 }
253
254 LayerAssignment layerAssignment;
255 layerAssignment.mMaterial = materialObj ? materialObj->asA<Material>() : nullptr;
256 layerAssignment.mLightSet = lightSetObj ? lightSetObj->asA<LightSet>() : nullptr;
257 layerAssignment.mLightFilterSet = lightFilterSetObj ? lightFilterSetObj->asA<LightFilterSet>() : nullptr;
258 layerAssignment.mShadowSet = shadowSetObj ? shadowSetObj->asA<ShadowSet>() : nullptr;
259 layerAssignment.mShadowReceiverSet = shadowReceiverSetObj ? shadowReceiverSetObj->asA<ShadowReceiverSet>()
260 : nullptr;
261 layerAssignment.mDisplacement = displacementObj ? displacementObj->asA<Displacement>() : nullptr;
262 layerAssignment.mVolumeShader = volumeShaderObj ? volumeShaderObj->asA<VolumeShader>() : nullptr;
263 layer.assign(geomObj->asA<Geometry>(), partsName[i], layerAssignment);
264 }

267void
268BinaryReader::unpackSceneObject(ValueContainerDeq &vContainerDeq, SceneObject& sceneObject) const
269{
270 SceneObject::UpdateGuard guard(&sceneObject);
271
272 BinaryReaderLayerUnpackStrings layerStrVectors;
273
274 // Step over each attribute provided.
275 while (1) {
276 ValueContainerUtil::ValueType valueType;
277 vContainerDeq.deqAttributeType(valueType);
278 if (valueType == ValueContainerUtil::ValueType::UNKNOWN) break;
279
280 bool transientEncoding;
281 int attributeId = 0;
282 std::string attributeName;
283 vContainerDeq.deqBool(transientEncoding);
284 if (transientEncoding) {
285 vContainerDeq.deqInt(attributeId);
286 } else {
287 vContainerDeq.deqString(attributeName);
288 }
289
290 int timestep = TIMESTEP_BEGIN;
291 int timeMax = 0;
292 {
293 unsigned char uc;
294 vContainerDeq.deqUChar(uc);
295 timeMax = static_cast<int>(uc);
296 }
297 do {
298 try {
299 if (sceneObject.isA<Layer>()) {
300 const SceneClass& sceneClass = sceneObject.getSceneClass();
301 const std::string &attrName =
302 (transientEncoding)? sceneClass.mAttributes[attributeId]->getName(): attributeName;
303 unpackLayerValue(vContainerDeq, layerStrVectors, valueType, attrName);
304 } else {
305 unpackValue(vContainerDeq, sceneObject, valueType, transientEncoding, attributeId, attributeName);
306 }
307 } catch (except::KeyError& e) {
308 // No attribute with that name.
309 std::string msg = util::buildString(sceneObject.getName(), ": ", e.what());
310 if (mWarningsAsErrors) {
311 throw except::KeyError(msg); // Rethrow with more information.
312 } else {
313 logging::Logger::warn(msg);
314 }
315 } catch (except::TypeError& e) {
316 // Type mismatch.
317 std::string msg = util::buildString(sceneObject.getName(), ": ", e.what());
318 if (mWarningsAsErrors) {
319 throw except::TypeError(msg); // Rethrow with more information.
320 } else {
321 logging::Logger::warn(msg);
322 }
323 } catch (except::IoError& e) {
324 // Couldn't load DSO.
325 std::string msg = util::buildString(sceneObject.getName(), ": ", e.what());
326 if (mWarningsAsErrors) {
327 throw except::IoError(msg); // Rethrow with more information.
328 } else {
329 logging::Logger::warn(msg);
330 }
331 }
332 ++timestep;
333 } while (timestep <= timeMax);
334 } // while (1)
335
336 if (sceneObject.isA<Layer>()) {
337 unpackLayer(layerStrVectors, *(sceneObject.asA<Layer>()));
338 }
339
340 while (1) {
341 bool valueBool;
342 vContainerDeq.deqBool(valueBool);
343 if (!valueBool) break;
344
345 bool transientEncoding;
346 int attributeId = -1;
347 std::string attributeName;
348 vContainerDeq.deqBool(transientEncoding);
349 if (transientEncoding) {
350 vContainerDeq.deqInt(attributeId);
351 } else {
352 vContainerDeq.deqString(attributeName);
353 }
354
355 try {
356 std::string klassName, objName;
357 vContainerDeq.deqString(klassName);
358 vContainerDeq.deqString(objName);
359
360 if (!sceneObject.isA<Layer>()) {
361 SceneObject *targetObject = nullptr;
362 if (!klassName.empty() && !objName.empty()) {
363 targetObject = mContext.createSceneObject(klassName, objName);
364 }
365
366 // Set the binding.
367 size_t index = 0;
368 if (!transientEncoding) {
369 const SceneClass& sceneClass = sceneObject.getSceneClass();
370 const Attribute* attribute = sceneClass.getAttribute(attributeName);
371 index = attribute->mIndex;
372 } else {
373 index = attributeId;
374 }
375
376 sceneObject.mBindings[index] = targetObject;
377 sceneObject.mBindingSetMask.set(index, true);
378 sceneObject.mBindingUpdateMask.set(index, true);
379 sceneObject.mDirty = true;
380 }
381
382 } catch (except::KeyError& e) {
383 // No attribute with that name.
384 std::string msg = util::buildString(sceneObject.getName(), ": ", e.what());
385 if (mWarningsAsErrors) {
386 throw except::KeyError(msg); // Rethrow with more information.
387 } else {
388 Logger::warn(msg);
389 }
390 } catch (except::IoError& e) {
391 // Couldn't load DSO.
392 std::string msg = util::buildString(sceneObject.getName(), ": ", e.what());
393 if (mWarningsAsErrors) {
394 throw except::IoError(msg); // Rethrow with more information.
395 } else {
396 Logger::warn(msg);
397 }
398 }
399 }
400}

402void
403BinaryReader::unpackValue(ValueContainerDeq &vContainerDeq,
404 SceneObject &sceneObject,
405 ValueContainerUtil::ValueType valueType,
406 bool transientEncoding,
407 int attributeId,
408 std::string &attributeName) const
409{
410 int timestepInt;
411 {
412 unsigned char uc;
413 vContainerDeq.deqUChar(uc);
414 timestepInt = static_cast<int>(uc);
415 }
416 AttributeTimestep timestep = static_cast<AttributeTimestep>(static_cast<int>(timestepInt));
417
418 const SceneClass& sceneClass = sceneObject.getSceneClass();
419
420 switch (valueType) {
421 case ValueContainerUtil::ValueType::BOOL : {
422 Bool val; vContainerDeq.deqBool(val);
423 sceneObject.set(keyGen<Bool>(transientEncoding, attributeId, attributeName, sceneClass), val, timestep);
424 } break;
425 case ValueContainerUtil::ValueType::INT : {
426 int val; vContainerDeq.deqInt(val);
427 sceneObject.set(keyGen<Int>(transientEncoding, attributeId, attributeName, sceneClass), val, timestep);
428 } break;
429 case ValueContainerUtil::ValueType::LONG : {
430 long val; vContainerDeq.deqLong(val);
431 sceneObject.set(keyGen<Long>(transientEncoding, attributeId, attributeName, sceneClass), val, timestep);
432 } break;
433 case ValueContainerUtil::ValueType::FLOAT : {
434 Float val; vContainerDeq.deqFloat(val);
435 sceneObject.set(keyGen<Float>(transientEncoding, attributeId, attributeName, sceneClass), val, timestep);
436 } break;
437 case ValueContainerUtil::ValueType::DOUBLE : {
438 Double val; vContainerDeq.deqDouble(val);
439 sceneObject.set(keyGen<Double>(transientEncoding, attributeId, attributeName, sceneClass), val, timestep);
440 } break;
441 case ValueContainerUtil::ValueType::STRING : {
442 String val; vContainerDeq.deqString(val);
443 sceneObject.set(keyGen<String>(transientEncoding, attributeId, attributeName, sceneClass), val, timestep);
444 } break;
445 case ValueContainerUtil::ValueType::RGB : {
446 Rgb val; vContainerDeq.deqRgb(val);
447 sceneObject.set(keyGen<Rgb>(transientEncoding, attributeId, attributeName, sceneClass), val, timestep);
448 } break;
449 case ValueContainerUtil::ValueType::RGBA : {
450 Rgba val; vContainerDeq.deqRgba(val);
451 sceneObject.set(keyGen<Rgba>(transientEncoding, attributeId, attributeName, sceneClass), val, timestep);
452 } break;
453 case ValueContainerUtil::ValueType::VEC2F : {
454 Vec2f val; vContainerDeq.deqVec2f(val);
455 sceneObject.set(keyGen<Vec2f>(transientEncoding, attributeId, attributeName, sceneClass), val, timestep);
456 } break;
457 case ValueContainerUtil::ValueType::VEC2D : {
458 Vec2d val; vContainerDeq.deqVec2d(val);
459 sceneObject.set(keyGen<Vec2d>(transientEncoding, attributeId, attributeName, sceneClass), val, timestep);
460 } break;
461 case ValueContainerUtil::ValueType::VEC3F : {
462 Vec3f val; vContainerDeq.deqVec3f(val);
463 sceneObject.set(keyGen<Vec3f>(transientEncoding, attributeId, attributeName, sceneClass), val, timestep);
464 } break;
465 case ValueContainerUtil::ValueType::VEC3D : {
466 Vec3d val; vContainerDeq.deqVec3d(val);
467 sceneObject.set(keyGen<Vec3d>(transientEncoding, attributeId, attributeName, sceneClass), val, timestep);
468 } break;
469 case ValueContainerUtil::ValueType::VEC4F : {
470 Vec4f val; vContainerDeq.deqVec4f(val);
471 sceneObject.set(keyGen<Vec4f>(transientEncoding, attributeId, attributeName, sceneClass), val, timestep);
472 } break;
473 case ValueContainerUtil::ValueType::VEC4D : {
474 Vec4d val; vContainerDeq.deqVec4d(val);
475 sceneObject.set(keyGen<Vec4d>(transientEncoding, attributeId, attributeName, sceneClass), val, timestep);
476 } break;
477 case ValueContainerUtil::ValueType::MAT4F : {
478 Mat4f val; vContainerDeq.deqMat4f(val);
479 sceneObject.set(keyGen<Mat4f>(transientEncoding, attributeId, attributeName, sceneClass), val, timestep);
480 } break;
481 case ValueContainerUtil::ValueType::MAT4D : {
482 Mat4d val; vContainerDeq.deqMat4d(val);
483 sceneObject.set(keyGen<Mat4d>(transientEncoding, attributeId, attributeName, sceneClass), val, timestep);
484 } break;

486 case ValueContainerUtil::ValueType::SCENE_OBJECT : {
487 std::string klassName, objName;
488 vContainerDeq.deqSceneObject(klassName, objName);
489 SceneObject *targetObject = nullptr;

496 //------------------------------ vector type ------------------------------
497
498 case ValueContainerUtil::ValueType::BOOL_VECTOR : {
499 BoolVector vec; vContainerDeq.deqBoolVector(vec);
500 sceneObject.set(keyGen<BoolVector>(transientEncoding, attributeId, attributeName, sceneClass), vec, timestep);
501 } break;
502 case ValueContainerUtil::ValueType::INT_VECTOR : {
503 IntVector vec; vContainerDeq.deqVLIntVector(vec); // We are using VariableLength version
504 sceneObject.set(keyGen<IntVector>(transientEncoding, attributeId, attributeName, sceneClass), vec, timestep);
505 } break;
506 case ValueContainerUtil::ValueType::LONG_VECTOR : {
507 LongVector vec; vContainerDeq.deqVLLongVector(vec); // We are using VariableLength version
508 sceneObject.set(keyGen<LongVector>(transientEncoding, attributeId, attributeName, sceneClass), vec, timestep);
509 } break;
510 case ValueContainerUtil::ValueType::FLOAT_VECTOR : {
511 FloatVector vec; vContainerDeq.deqFloatVector(vec);
512 sceneObject.set(keyGen<FloatVector>(transientEncoding, attributeId, attributeName, sceneClass), vec, timestep);
513 } break;
514 case ValueContainerUtil::ValueType::DOUBLE_VECTOR : {
515 DoubleVector vec; vContainerDeq.deqDoubleVector(vec);
516 sceneObject.set(keyGen<DoubleVector>(transientEncoding, attributeId, attributeName, sceneClass), vec, timestep);
517 } break;
518 case ValueContainerUtil::ValueType::STRING_VECTOR : {
519 StringVector vec; vContainerDeq.deqStringVector(vec);
520 sceneObject.set(keyGen<StringVector>(transientEncoding, attributeId, attributeName, sceneClass), vec, timestep);
521 } break;
522 case ValueContainerUtil::ValueType::RGB_VECTOR : {
523 RgbVector vec; vContainerDeq.deqRgbVector(vec);
524 sceneObject.set(keyGen<RgbVector>(transientEncoding, attributeId, attributeName, sceneClass), vec, timestep);
525 } break;
526 case ValueContainerUtil::ValueType::RGBA_VECTOR : {
527 RgbaVector vec; vContainerDeq.deqRgbaVector(vec);
528 sceneObject.set(keyGen<RgbaVector>(transientEncoding, attributeId, attributeName, sceneClass), vec, timestep);
529 } break;
530 case ValueContainerUtil::ValueType::VEC2F_VECTOR : {
531 Vec2fVector vec; vContainerDeq.deqVec2fVector(vec);
532 sceneObject.set(keyGen<Vec2fVector>(transientEncoding, attributeId, attributeName, sceneClass), vec, timestep);
533 } break;
534 case ValueContainerUtil::ValueType::VEC2D_VECTOR : {
535 Vec2dVector vec; vContainerDeq.deqVec2dVector(vec);
536 sceneObject.set(keyGen<Vec2dVector>(transientEncoding, attributeId, attributeName, sceneClass), vec, timestep);
537 } break;
538 case ValueContainerUtil::ValueType::VEC3F_VECTOR : {
539 Vec3fVector vec; vContainerDeq.deqVec3fVector(vec);
540 sceneObject.set(keyGen<Vec3fVector>(transientEncoding, attributeId, attributeName, sceneClass), vec, timestep);
541 } break;
542 case ValueContainerUtil::ValueType::VEC3D_VECTOR : {
543 Vec3dVector vec; vContainerDeq.deqVec3dVector(vec);
544 sceneObject.set(keyGen<Vec3dVector>(transientEncoding, attributeId, attributeName, sceneClass), vec, timestep);
545 } break;
546 case ValueContainerUtil::ValueType::VEC4F_VECTOR : {
547 Vec4fVector vec; vContainerDeq.deqVec4fVector(vec);
548 sceneObject.set(keyGen<Vec4fVector>(transientEncoding, attributeId, attributeName, sceneClass), vec, timestep);
549 } break;
550 case ValueContainerUtil::ValueType::VEC4D_VECTOR : {
551 Vec4dVector vec; vContainerDeq.deqVec4dVector(vec);
552 sceneObject.set(keyGen<Vec4dVector>(transientEncoding, attributeId, attributeName, sceneClass), vec, timestep);
553 } break;
554 case ValueContainerUtil::ValueType::MAT4F_VECTOR : {
555 Mat4fVector vec; vContainerDeq.deqMat4fVector(vec);
556 sceneObject.set(keyGen<Mat4fVector>(transientEncoding, attributeId, attributeName, sceneClass), vec, timestep);
557 } break;
558 case ValueContainerUtil::ValueType::MAT4D_VECTOR : {
559 Mat4dVector vec; vContainerDeq.deqMat4dVector(vec);
560 sceneObject.set(keyGen<Mat4dVector>(transientEncoding, attributeId, attributeName, sceneClass), vec, timestep);
561 } break;
562
563 case ValueContainerUtil::ValueType::SCENE_OBJECT_VECTOR : {
564 StringVector klassNameVec, objNameVec; vContainerDeq.deqSceneObjectVector(klassNameVec, objNameVec);
565 size_t size = klassNameVec.size();
566 SceneObjectVector vec;
567 vec.resize(size);
568 for (size_t i = 0; i < size; ++i) {
569 SceneObject *targetObject = nullptr;
570 if (!klassNameVec[i].empty() && !objNameVec[i].empty()) {
571 targetObject = mContext.createSceneObject(klassNameVec[i], objNameVec[i]);
572 }
573 vec[i] = targetObject;
574 }
575
576 // If the SceneObject is a GeometrySet, LightSet, Displacement, VolumeShader, LightFilterSet,
577 // or ShadowSet we need to re-sort the vector since its sorted by pointer. The vector must
578 // be sorted to maintain our uniqueness invariant.
579 if (sceneObject.isA<GeometrySet>() || sceneObject.isA<LightSet>() ||
580 sceneObject.isA<Displacement>() || sceneObject.isA<VolumeShader>() ||
581 sceneObject.isA<LightFilterSet>() || sceneObject.isA<ShadowSet>()) {
582 std::sort(vec.begin(), vec.end());
583 }
584
585 sceneObject.set(keyGen<SceneObjectVector>(transientEncoding, attributeId, attributeName, sceneClass), vec);
586 } break;
587
588 case ValueContainerUtil::ValueType::SCENE_OBJECT_INDEXABLE : {
589 StringVector klassNameVec, objNameVec; vContainerDeq.deqSceneObjectIndexable(klassNameVec, objNameVec);
590 size_t size = klassNameVec.size();
591 SceneObjectVector vec;
592 vec.resize(size);
593 for (size_t i = 0; i < size; ++i) {
594 SceneObject *targetObject = nullptr;
595 if (!klassNameVec[i].empty() && !objNameVec[i].empty()) {
596 targetObject = mContext.createSceneObject(klassNameVec[i], objNameVec[i]);
597 }
598 vec[i] = targetObject;
599 }
600
601 // If the SceneObject is a GeometrySet, LightSet, Displacement, VolumeShader, LightFilterSet,
602 // or ShadowSet we need to re-sort the vector since its sorted by pointer. The vector must
603 // be sorted to maintain our uniqueness invariant.
604 if (sceneObject.isA<GeometrySet>() || sceneObject.isA<LightSet>() ||
605 sceneObject.isA<Displacement>() || sceneObject.isA<VolumeShader>() ||
606 sceneObject.isA<LightFilterSet>() || sceneObject.isA<ShadowSet>()) {
607 std::sort(vec.begin(), vec.end());
608 }
609
610 sceneObject.set(keyGen<SceneObjectIndexable>(transientEncoding, attributeId, attributeName, sceneClass),
611 SceneObjectIndexable(vec.begin(), vec.end()));
612 } break;
613
614 default : {
615 } break;
616 }
617}

619void
620BinaryReader::unpackLayerValue(ValueContainerDeq &vContainerDeq,
621 BinaryReaderLayerUnpackStrings &layerStrVectors,
622 ValueContainerUtil::ValueType valueType,
623 const std::string &attrName) const
624{
625 {
626 // Dequeue the timestep, which we don't need but need to dispose of.
627 unsigned char uc;
628 vContainerDeq.deqUChar(uc);
629 }
630
631 switch (valueType) {
632 case ValueContainerUtil::ValueType::STRING_VECTOR : {
633 if (attrName != "parts") {
634 throw except::RuntimeError("encountered invalid attribute name:" + attrName +
635 " during unpack layer value.");
636 }
637 vContainerDeq.deqStringVector(layerStrVectors.mPartName);
638 } break;
639
640 case ValueContainerUtil::ValueType::SCENE_OBJECT_VECTOR :
641 case ValueContainerUtil::ValueType::SCENE_OBJECT_INDEXABLE : {
642 if (attrName == "geometries") {
643 vContainerDeq.deqSceneObjectVector(layerStrVectors.mGeomKlassName,
644 layerStrVectors.mGeomObjName);
645 } else if (attrName == "surface_shaders") {
646 vContainerDeq.deqSceneObjectVector(layerStrVectors.mMaterialKlassName,
647 layerStrVectors.mMaterialObjName);
648 } else if (attrName == "lightsets") {
649 vContainerDeq.deqSceneObjectVector(layerStrVectors.mLightSetKlassName,
650 layerStrVectors.mLightSetObjName);
651 } else if (attrName == "displacements") {
652 vContainerDeq.deqSceneObjectVector(layerStrVectors.mDisplacementKlassName,
653 layerStrVectors.mDisplacementObjName);
654 } else if (attrName == "volume_shaders") {
655 vContainerDeq.deqSceneObjectVector(layerStrVectors.mVolumeShaderKlassName,
656 layerStrVectors.mVolumeShaderObjName);
657 } else if (attrName == "lightfiltersets") {
658 vContainerDeq.deqSceneObjectVector(layerStrVectors.mLightFilterSetKlassName,
659 layerStrVectors.mLightFilterSetObjName);
660 } else if (attrName == "shadowsets") {
661 vContainerDeq.deqSceneObjectVector(layerStrVectors.mShadowSetKlassName,
662 layerStrVectors.mShadowSetObjName);
663 } else if (attrName == "shadowreceiversets") {
664 vContainerDeq.deqSceneObjectVector(layerStrVectors.mShadowReceiverSetKlassName,
665 layerStrVectors.mShadowReceiverSetObjName);
666 } else {
667 throw except::RuntimeError("encountered invalid attribute name:" + attrName +
668 " during unpack layer value.");
669 }
670 } break;
671
672 default : {
673 } break;
674 }
675}
