36struct ALIGN(16) SortedEntry
37{
38 uint32_t mSortKey; // Material bundled id is stored in here.
39 uint32_t mRsIdx; // Global ray state index.
40 uint32_t mIsectIdx;
41 int mLayerAssignmentId;
42};
43
44MNRY_STATIC_ASSERT(sizeof(SortedEntry) == 16);

91void shadeBundleHandler(mcrt_common::ThreadLocalState *tls, unsigned numEntries,
92 shading::SortedRayState *entries, void *userData)
93{
94 pbr::TLState *pbrTls = tls->mPbrTls.get();
95
96 EXCL_ACCUMULATOR_PROFILE(pbrTls, EXCL_ACCUM_SHADE_HANDLER);
97
98 MNRY_ASSERT(numEntries && entries && userData);
99
100 CHECK_CANCELLATION(pbrTls, return);
101
102 shading::Material *ext = MNRY_VERIFY((shading::Material *)userData);
103
104 // We should hit this case rarely, mainly in the case where russian roulette
105 // isn't culling the number of rays we'd like it to.
106 if (tls->checkForHandlerStackOverflowRisk()) {
107 ext->deferEntriesForLaterProcessing(tls, numEntries, entries);
108 return;
109 }
110
111 scene_rdl2::alloc::Arena *arena = pbrTls->mArena;
112 SCOPED_MEM(arena);
113
114 // Retrieve any shading entries which were previously deferred and process
115 // them now (since we're no longer at risk for a handler stack overflow).
116 // We potentially update the numEntries and entries variables with this call.
117 ext->retrieveDeferredEntries(tls, arena, numEntries, entries);
118
119 // Convert array of SortedRayState objects to RayState pointers in-place.
120 RayState **rayStates = decodeRayStatesInPlace(numEntries, entries);
121 MNRY_ASSERT(isRayStateListValid(pbrTls, numEntries, rayStates));
122
123 const FrameState &fs = *pbrTls->mFs;
124
125 //
126 // Compute final sort order for shading points.
127 //
128 const scene_rdl2::rdl2::Layer *layer = fs.mLayer;
129 const Scene *scene = fs.mScene;
130 const scene_rdl2::rdl2::Material &material = *ext->getOwner().asA<scene_rdl2::rdl2::Material>();
131 RayState *baseRayState = indexToRayState(0);
132 shading::TLState *shadingTls = MNRY_VERIFY(tls->mShadingTls.get());
133
134 // Allocate temp working memory buffers (automatically freed when we exit this function).
135 shading::Intersection *isectMemory = arena->allocArray<shading::Intersection>(numEntries, CACHE_LINE_SIZE);
136 SortedEntry *sortedEntries = arena->allocArray<SortedEntry>(numEntries);
137 BundledRadiance *radiances = arena->allocArray<BundledRadiance>(numEntries, CACHE_LINE_SIZE);
138 BundledRadiance *currRadiance = radiances;
139
140 // Allocate memory to gather raystates so we can bulk free them later in the function.
141 unsigned numRayStatesToFree = 0;
142 RayState **rayStatesToFree = arena->allocArray<RayState *>(numEntries);
143
144 // Aovs.
145 float *aovs = nullptr;
146 unsigned aovNumChannels = 0;
147 if (!fs.mAovSchema->empty()) {
148 // scratch space storage for per-pixel aov packing
149 aovNumChannels = fs.mAovSchema->numChannels();
150 aovs = arena->allocArray<float>(aovNumChannels);
151 fs.mAovSchema->initFloatArray(aovs);
152 }
153
154 // If we have a ray that just came from a presence object, we need to handle the case where the ray now hits
155 // a non presence object for the cryptomatte. If we need to handle that case, we set this variable:
156 bool handlePresenceContinue = false;
157
158 for (unsigned i = 0; i < numEntries; ++i) {
159
160 CHECK_CANCELLATION_IN_LOOP(pbrTls, i, return);
161
162 RayState *rs = rayStates[i];
163 mcrt_common::RayDifferential *ray = &rs->mRay;
164 PathVertex &pv = rs->mPathVertex;
165
166 // Code for rendering lights. Only executed for primary rays since lights
167 // appear in deeper passes already. Do this here so that we can avoid
168 // evaluating shaders for any points that hit a light before they hit a
169 // surface.
170 bool terminate = false;
171 if (ray->getDepth() == 0) {
172
173 MNRY_ASSERT(scene_rdl2::math::isEqual(pv.pathPixelWeight, reduce_avg(pv.pathThroughput)));
174
175 // Entries on this code path have hit a geometric surface. Before attempting
176 // to execute the attached shader, we must first check that there isn't a light
177 // blocking the path to the surface.
178
179 LightIntersection hitLightIsect;
180 int numHits = 0;
181 const Light *hitLight;
182
183 SequenceIDIntegrator sid(rs->mSubpixel.mPixel,
184 rs->mSubpixel.mSubpixelIndex,
185 fs.mInitialSeed[rs->mCameraId]);
186 IntegratorSample1D lightChoiceSamples(sid);
187 hitLight = scene->intersectVisibleLight(*ray, ray->getEnd(), lightChoiceSamples, hitLightIsect, numHits);
188
189 if (hitLight) {
190
191 // Evaluate the radiance on a random visible light in camera
192 // Note: we multiply the radiance contribution by the number of lights hit. This is
193 // because we want to compute the sum of all contributing lights, but we're
194 // stochastically sampling just one.
195
196 LightFilterRandomValues lightFilterR = {
197 scene_rdl2::math::Vec2f(0.f, 0.f),
198 scene_rdl2::math::Vec3f(0.f, 0.f, 0.f)}; // light filters don't apply to camera rays
199 scene_rdl2::math::Color radiance = pv.pathThroughput * hitLight->eval(
200 tls, ray->getDirection(), ray->getOrigin(), lightFilterR, ray->getTime(), hitLightIsect, true,
201 nullptr, ray->getDirFootprint()) * numHits;
202
203 // Volumes
204 // This matches the scalar code, but is potentially a bug. We have intersected a geometry
205 // and computed volume transmittance and radiance relative to the ray to the geometry.
206 // We have now discovered a visible light between the geometry and our viewpoint. It seems
207 // like we should recompute the volume radiance and transmittance at this point, but the scalar
208 // code does not do this, so we don't either. It would be expensive to recompute
209 // the transmittance and radiance along this ray, so as long as we aren't seeing artifacts
210 // as a result of this, it is probably best to continue using this approach.
211 if (rs->mVolHit) {
212 radiance *= (rs->mVolTr * rs->mVolTh);
213 radiance += rs->mVolRad;
214 }
215
216 // alpha depends on light opacity and volumes
217 float alpha = 0.0f;
218 if (hitLight->getIsOpaqueInAlpha()) {
219 // We hit a visible light that is opaque in alpha.
220 // Volumes are irrelevant, the alpha contribution is
221 // the full pixel weight.
222 alpha = rs->mPathVertex.pathPixelWeight;
223 } else if (rs->mVolHit) {
224 // We hit a visible light, but the light is not
225 // opaque in alpha (e.g. a distant or env light).
226 // There is a volume along this ray. The volume
227 // transparency determines the alpha contribution.
228 // This is probably an impossible case to hit because visible lights
229 // that are not opaque in alpha are typically env or distant lights.
230 // Such a light would not come between the camera and scene geometry.
231 alpha = rs->mPathVertex.pathPixelWeight * (1.f - reduceTransparency(rs->mVolTalpha));
232 } else {
233 // We hit a visible light, but the light is not
234 // opaque in alpha (e.g. a distant or env light).
235 // There is no volume along the ray.
236 // The alpha contribution is 0.
237 // This is probably an impossible case to hit because visible lights
238 // that are not opaque in alpha are typically env or distant lights.
239 // Such a light would not come between the camera and scene geometry.
240 // If it did, then the light would act like a cutout.
241 alpha = 0.0f;
242 }
243
244 currRadiance->mRadiance = RenderColor(radiance.r, radiance.g, radiance.b, alpha);
245 currRadiance->mPathPixelWeight = pv.pathPixelWeight;
246 currRadiance->mPixel = rs->mSubpixel.mPixel;
247 currRadiance->mSubPixelIndex = rs->mSubpixel.mSubpixelIndex;
248 currRadiance->mDeepDataHandle = pbrTls->acquireDeepData(rs->mDeepDataHandle);
249 currRadiance->mCryptomatteDataHandle = pbrTls->acquireCryptomatteData(rs->mCryptomatteDataHandle);
250 currRadiance->mTilePassAndFilm = rs->mTilePassAndFilm;
251 currRadiance->mCameraId = rs->mCameraId;
252
253 // To maintain parity with scalar mode, specify that we hit something with an Id of 0 when we hit
254 // a light for cryptomattes:
255 if (rs->mCryptomatteDataHandle != pbr::nullHandle) {
256 pbr::CryptomatteData *cryptomatteData =
257 static_cast<pbr::CryptomatteData*>(pbrTls->getListItem(rs->mCryptomatteDataHandle, 0));
258 cryptomatteData->mHit = 1;
259 cryptomatteData->mId = 0.f;
260 }
261
262 ++currRadiance;
263
264 // LPE
265 if (aovs) {
266 EXCL_ACCUMULATOR_PROFILE(pbrTls, EXCL_ACCUM_AOVS);
267
268 const LightAovs &lightAovs = *fs.mLightAovs;
269
270 // transition
271 int lpeStateId = pv.lpeStateId;
272 lpeStateId = lightAovs.lightEventTransition(pbrTls, lpeStateId, hitLight);
273
274 // accumulate matching aovs.
275 aovAccumLightAovsBundled(pbrTls, *fs.mAovSchema, rs->mCameraId,
276 lightAovs, radiance, nullptr, AovSchema::sLpePrefixNone, lpeStateId,
277 rs->mSubpixel.mPixel,
278 rs->mDeepDataHandle,
279 getFilm(rs->mTilePassAndFilm));
280 }
281
282 rayStatesToFree[numRayStatesToFree++] = rs;
283
284 terminate = true;
285 }
286 }
287
288 // Disable ray by default.
289 // All terminated rays are put at the end of the sorted list.
290 // Anything with a sortkey of 0xffffffff is counted as a terminated
291 // ray and doesn't hit the shader evaluation code path.
292 sortedEntries[i].mSortKey = 0xffffffff;
293
294 if (!terminate) {
295
296 const bool isSubsurfaceAllowed = fs.mIntegrator->isSubsurfaceAllowed(pv.subsurfaceDepth);
297
298 // Initialize intersections.
299 geom::initIntersectionFull(isectMemory[i],
300 tls, *ray, layer,
301 pv.mirrorDepth,
302 pv.glossyDepth,
303 pv.diffuseDepth,
304 isSubsurfaceAllowed,
305 pv.minRoughness,
306 -ray->getDirection());
307 isectMemory[i].setCameraId(rs->mCameraId);
308
309 shading::Intersection *isect = &isectMemory[i];
310
311 // early termination if intersection doesn't provide all the
312 // required primitive attributes shader request
313 if (!isect->hasAllRequiredAttributes()) {
314 scene_rdl2::math::Color radiance = pv.pathThroughput * fs.mFatalColor;
315 currRadiance->mRadiance = RenderColor(
316 radiance.r, radiance.g, radiance.b, pv.pathPixelWeight);
317 currRadiance->mPathPixelWeight = pv.pathPixelWeight;
318 currRadiance->mPixel = rs->mSubpixel.mPixel;
319 currRadiance->mSubPixelIndex = rs->mSubpixel.mSubpixelIndex;
320 currRadiance->mDeepDataHandle = pbrTls->acquireDeepData(rs->mDeepDataHandle);
321 currRadiance->mCryptomatteDataHandle = pbrTls->acquireCryptomatteData(rs->mCryptomatteDataHandle);
322 currRadiance->mTilePassAndFilm = rs->mTilePassAndFilm;
323 currRadiance->mCameraId = rs->mCameraId;
324 ++currRadiance;
325
326 rayStatesToFree[numRayStatesToFree++] = rs;
327 terminate = true;
328 } else {
329 // Populate deep data, first hit only for now
330 if (ray->getDepth() == 0 && rs->mDeepDataHandle != pbr::nullHandle) {
331 pbr::DeepData *deepData = static_cast<pbr::DeepData*>(pbrTls->getListItem(rs->mDeepDataHandle, 0));
332 deepData->mHitDeep = 1;
333 deepData->mSubpixelX = rs->mSubpixel.mSubpixelX;
334 deepData->mSubpixelY = rs->mSubpixel.mSubpixelY;
335 deepData->mRayZ = ray->dir.z;
336 deepData->mDeepT = ray->tfar;
337 deepData->mDeepNormal = ray->Ng;
338
339 // retrieve all of the deep IDs for this intersection from the primitive attrs
340 for (size_t i = 0; i < fs.mIntegrator->getDeepIDAttrIdxs().size(); i++) {
341 shading::TypedAttributeKey<float> deepIDAttrKey((fs.mIntegrator->getDeepIDAttrIdxs())[i]);
342 if (isect->isProvided(deepIDAttrKey)) {
343 deepData->mDeepIDs[i] = isect->getAttribute<float>(deepIDAttrKey);
344 } else {
345 deepData->mDeepIDs[i] = 0.f;
346 }
347 }
348 }
349
350 // Populate cryptomatte data
351 if (ray->getDepth() == 0 && rs->mCryptomatteDataHandle != pbr::nullHandle) {
352 pbr::CryptomatteData *cryptomatteData =
353 static_cast<pbr::CryptomatteData*>(pbrTls->getListItem(rs->mCryptomatteDataHandle, 0));
354
355 // Don't want to double count, so we set mHit to 0 only if it was originally a presence ray.
356 if (cryptomatteData->mPrevPresence) {
357 cryptomatteData->mHit = 0;
358 handlePresenceContinue = true;
359 } else {
360 cryptomatteData->mHit = 1;
361 }
362
363 // Retrieve the first deep id (if present) for this intersection from the primitive attrs.
364 // At the request of production, cryptomatte currently only supports one deep id associated
365 // with each layer, rather than a generalized vector of ids. So here we are only interested
366 // in the first id.
367 if (fs.mIntegrator->getDeepIDAttrIdxs().size() != 0) {
368 shading::TypedAttributeKey<float> deepIDAttrKey(fs.mIntegrator->getDeepIDAttrIdxs()[0]);
369 if (isect->isProvided(deepIDAttrKey)) {
370 cryptomatteData->mId = isect->getAttribute<float>(deepIDAttrKey);
371 } else {
372 cryptomatteData->mId = 0.f;
373 }
374 }
375 }
376
377 // TODO: Apply volume transmittance on the segment
378 // ray origin --> ray isect
379
380 // Transfer the ray to its intersection before we run shaders.
381 // This is needed for texture filtering based on ray differentials.
382 // Also scale the final differentials by a user factor.
383 // This is left until the very end and not baked into the
384 // ray differentials since the factor will typically be > 1,
385 // and would cause the ray differentials to be larger
386 // than necessary.
387 isect->transferAndComputeDerivatives(tls, ray,
388 rs->mSubpixel.mTextureDiffScale);
389
390 sortedEntries[i].mSortKey = isect->computeShadingSortKey();
391
392 sortedEntries[i].mLayerAssignmentId = isect->getLayerAssignmentId();
393
394 // state and prim attr aovs
395 const bool doAovs = ray->getDepth() == 0 && !fs.mAovSchema->empty();
396 if (doAovs) {
397 EXCL_ACCUMULATOR_PROFILE(pbrTls, EXCL_ACCUM_AOVS);
398 fs.mAovSchema->initFloatArray(aovs);
399 aovSetStateVars(pbrTls,
400 *fs.mAovSchema,
401 rs->mCameraId,
402 *isect,
403 rs->mVolumeSurfaceT,
404 *ray,
405 *scene,
406 pv.pathPixelWeight,
407 aovs);
408 aovSetPrimAttrs(pbrTls,
409 *fs.mAovSchema,
410 rs->mCameraId,
411 ext->getAovFlags(),
412 *isect,
413 pv.pathPixelWeight,
414 aovs);
415 aovAddToBundledQueue(pbrTls,
416 *fs.mAovSchema,
417 *isect,
418 rs->mRay,
419 AOV_TYPE_STATE_VAR | AOV_TYPE_PRIM_ATTR,
420 aovs,
421 rs->mSubpixel.mPixel,
422 rs->mDeepDataHandle,
423 pbr::getFilm(rs->mTilePassAndFilm));
424 }
425 }
426 }
427
428 sortedEntries[i].mRsIdx = rs - baseRayState;
429 sortedEntries[i].mIsectIdx = i;
430 }
431
432 // Bulk free raystates.
433 pbrTls->freeRayStates(numRayStatesToFree, rayStatesToFree);
434
435 // Resort inputs based on mip selectors and uv coordinates.
436 sortedEntries = scene_rdl2::util::smartSort32<SortedEntry, 0>(numEntries, sortedEntries, 0xffffffff, arena);
437
438 //
439 // Send terminated and primary ray radiance samples to radiance queue.
440 //
441
442 unsigned numTerminated = currRadiance - radiances;
443
444 if (numTerminated) {
445 pbrTls->addRadianceQueueEntries(numTerminated, radiances);
446 CHECK_CANCELLATION(pbrTls, return);
447 }
448
449 numEntries -= numTerminated;
450
451 if (numEntries == 0) {
452 return;
453 }
454
455 //
456 // Shade all non-terminated entries and send them on to the integrator.
457 //
458
459 // Take fs.mShadingWorkloadChunkSize into account so that we're the blocks
460 // of AOS data we need to convert to SOA are small enough that they stay
461 // in our working caches. This works since we've already sorted the entire
462 // entry list by this point.
463
464 const unsigned shadingWorkloadChunkSize = fs.mShadingWorkloadChunkSize;
465 const unsigned numWorkloadChunkBlocks = fs.mShadingWorkloadChunkSize / VLEN;
466
467 MNRY_ASSERT(scene_rdl2::util::isPowerOfTwo(shadingWorkloadChunkSize));
468 MNRY_ASSERT(shadingWorkloadChunkSize >= VLEN);
469
470 shading::Intersectionv *isectsSOA =
471 arena->allocArray<shading::Intersectionv>(numWorkloadChunkBlocks, CACHE_LINE_SIZE);
472
473 shading::Bsdfv *bsdfv = arena->allocArray<shading::Bsdfv>(numWorkloadChunkBlocks, CACHE_LINE_SIZE);
474
475 RayStatev *rayStatesSOA = arena->allocArray<RayStatev>(numWorkloadChunkBlocks, CACHE_LINE_SIZE);
476
477 unsigned numRemaining = numEntries;
478 const SortedEntry *endEntry = sortedEntries + numEntries;
479
480 float *presences = arena->allocArray<float>(shadingWorkloadChunkSize);
481 bool *continueDeepRays = arena->allocArray<bool>(shadingWorkloadChunkSize);
482
483 // Split total work load into workloads which can be kept within the
484 // cache hierarchy.
485
486 while (numRemaining) {
487
488 CHECK_CANCELLATION(pbrTls, return);
489
490 const uint32_t currLayerAssnId = sortedEntries->mLayerAssignmentId;
491 const LightPtrList *lightList = scene->getLightPtrList(currLayerAssnId);
492 const LightFilterLists *lightFilterLists = scene->getLightFilterLists(currLayerAssnId);
493 const LightAccelerator *lightAcc = scene->getLightAccelerator(currLayerAssnId);
494
495 // Figure out how many contiguous entries we have which use the
496 // same lightset.
497 SortedEntry *currEntry = sortedEntries + 1;
498 while (currEntry != endEntry && currLayerAssnId == currEntry->mLayerAssignmentId) {
499 ++currEntry;
500 }
501
502 // Note is likely that workLoadSize isn't a multiple of VLEN anymore.
503 // We handle this by smearing the last valid entry across the remaining
504 // lanes in the various convertAOS*ToSOA utility functions.
505 //
506 // For shading this suffices and we can ignore masking. It shouldn't
507 // cause any slowdown since the extra lanes shouldn't fetch any memory
508 // that we weren't fetching already.
509 //
510 // The vectorized integrator is passed in the workLoadSize value so
511 // it knows how many "real" entries need processing.
512
513 unsigned workLoadSize = scene_rdl2::math::min(unsigned(currEntry - sortedEntries),
514 shadingWorkloadChunkSize);
515 numRemaining -= workLoadSize;
516
517 unsigned numBlocks = (workLoadSize + VLEN_MASK) / VLEN;
518
519 // Resort ray state pointers based on sort results. This is necessary
520 // so that we're updating the correct raystates objects after integration
521 // is complete.
522 // Also grab the presence values.
523 for (unsigned i = 0; i < workLoadSize; ++i) {
524 RayState *rs = baseRayState + sortedEntries[i].mRsIdx;
525 rs->mAOSIsect = &isectMemory[sortedEntries[i].mIsectIdx];
526 rs->mRayStateIdx = sortedEntries[i].mRsIdx;
527 rayStates[i] = rs;
528 presences[i] = shading::presence(&material, shadingTls,
529 shading::State((shading::Intersection *)(rs->mAOSIsect)));
530 // Some NPR materials that want to allow for completely arbitrary shading normals
531 // can request that the integrator does not perform any light culling based on the
532 // normal. In those cases, we also want to prevent our call to adaptNormal() in the
533 // Intersection when the material evaluates its normal map bindings.
534 shading::Intersection * const isect = static_cast<shading::Intersection*>(rs->mAOSIsect);
535 if (shading::preventLightCulling(&material, shading::State(isect))) {
536 isect->setUseAdaptNormal(false);
537 }
538 }
539
540 // TODO: overlapping dielectrics. This is where most of it would be
541 // implemented, similar to the scalar version in PathIntegrator.cc.
542
543 // Do conversion to from AOS to AOSOA.
544 convertAOSIntersectionsToSOA(pbrTls, workLoadSize, isectMemory, isectsSOA,
545 &sortedEntries[0].mIsectIdx);
546
547 // Reset bsdfv memory.
548 memset(bsdfv, 0, sizeof(shading::Bsdfv) * numBlocks);
549
550 {
551 // Compute shading.
552 int64_t ticks = 0;
553 MCRT_COMMON_CLOCK_OPEN(fs.mRequiresHeatMap? &ticks : nullptr)
554
555 shading::shadev(&material, shadingTls, numBlocks,
556 (const shading::Statev *)isectsSOA,
557 (scene_rdl2::rdl2::Bsdfv *)bsdfv);
558
559 MCRT_COMMON_CLOCK_CLOSE();
560 if (fs.mRequiresHeatMap) {
561 heatMapBundledUpdate(pbrTls, ticks, rayStates, workLoadSize);
562 }
563 }
564
565 // Cutout materials in scalar will return immediately when cutout
566 // Need to allow presence continuation in this case, so flag
567 bool cutout = bsdfv->mEarlyTerminationMask != 0;
568
569 // Evaluate any extra aovs on this material
570 if (!cutout && aovs) {
571 aovAccumExtraAovsBundled(pbrTls, fs, rayStates, presences, isectsSOA, &material, workLoadSize);
572 }
573
574 {
575 // ***** Handle spawning continuation rays for multi-layer deeps
576
577 // Count the number of deep layer rays we need
578 unsigned numDeepLayerRays = 0;
579 for (unsigned i = 0; i < workLoadSize; ++i) {
580 continueDeepRays[i] = false;
581
582 RayState *rs = rayStates[i];
583 mcrt_common::RayDifferential *ray = &rs->mRay;
584
585 // We only care about primary rays here
586 if (ray->getDepth() > 0) {
587 continue;
588 }
589
590 // We must be rendering to the deep buffer
591 if (rs->mDeepDataHandle == pbr::nullHandle) {
592 continue;
593 }
594 pbr::DeepData *deepData = static_cast<pbr::DeepData*>(pbrTls->getListItem(rs->mDeepDataHandle, 0));
595
596 // Check the current layer depth against the maximum
597 if (deepData->mLayer == fs.mIntegrator->getDeepMaxLayers() - 1) {
598 continue;
599 }
600
601 // Use fewer samples for deeper layers
602 // +1 because we are computing the samples in the *next* layer
603 int samplesDivision = 1 << ((deepData->mLayer + 1) * 2); // 1, 4, 16, 64 ...
604 if ((rs->mSubpixel.mSubpixelIndex % samplesDivision) > 0) {
605 continue;
606 }
607
608 continueDeepRays[i] = true;
609 numDeepLayerRays++;
610 }
611
612 // Create the deep layer rays if needed
613 RayState **deepLayerRays = nullptr;
614 if (numDeepLayerRays) {
615 deepLayerRays = pbrTls->allocRayStates(numDeepLayerRays);
616 numDeepLayerRays = 0;
617
618 for (unsigned i = 0; i < workLoadSize; ++i) {\
619 if (!continueDeepRays[i]) {
620 continue;
621 }
622
623 RayState *rs = rayStates[i];
624 pbr::DeepData *deepData = static_cast<pbr::DeepData*>(pbrTls->getListItem(rs->mDeepDataHandle, 0));
625
626 // Setup the deep layer ray
627
628 RayState *deepLayerRay = deepLayerRays[numDeepLayerRays++];
629 float deepLayerBias = fs.mIntegrator->getDeepLayerBias();
630
631 // Copy the raystate, but we will need to reset a few things.
632 // We could create a new ray but then we'd need to call the camera
633 // and a bunch of other logic, and that would be slower.
634 *deepLayerRay = *rs;
635
636 // Move the origin back to the original origin.
637 deepLayerRay->mRay.org -= rs->mRay.dir * rs->mRay.tfar;
638
639 // Set tnear to ignore the currently intersected geometry
640 deepLayerRay->mRay.tnear = rs->mRay.tfar + deepLayerBias;
641
642 // Restore tfar to the original camera ray tfar
643 deepLayerRay->mRay.tfar = rs->mRay.getOrigTfar();
644 deepLayerRay->mRay.setOrigTfar(deepLayerRay->mRay.tfar);
645
646 // Reset all the intersection state, throughput, and depths
647 deepLayerRay->mRay.geomID = -1;
648 deepLayerRay->mRay.primID = -1;
649 deepLayerRay->mRay.instID = -1;
650 deepLayerRay->mPathVertex.pathThroughput = scene_rdl2::math::Color(1.f);
651 deepLayerRay->mPathVertex.pathPixelWeight = 1.f;
652 deepLayerRay->mPathVertex.aovPathPixelWeight = 1.f;
653 deepLayerRay->mPathVertex.pathDistance = 0.f;
654 deepLayerRay->mPathVertex.minRoughness = scene_rdl2::math::Vec2f(0.0f);
655 deepLayerRay->mPathVertex.diffuseDepth = 0;
656 deepLayerRay->mPathVertex.subsurfaceDepth = 0;
657 deepLayerRay->mPathVertex.glossyDepth = 0;
658 deepLayerRay->mPathVertex.mirrorDepth = 0;
659 deepLayerRay->mPathVertex.nonMirrorDepth = 0;
660 deepLayerRay->mPathVertex.presenceDepth = 0;
661 deepLayerRay->mPathVertex.totalPresence = 0.f;
662 deepLayerRay->mPathVertex.hairDepth = 0;
663 deepLayerRay->mPathVertex.volumeDepth = 0;
664 deepLayerRay->mPathVertex.accumOpacity = 0.f;
665
666 // Need a new DeepData for this new ray
667 deepLayerRay->mDeepDataHandle = pbrTls->allocList(sizeof(pbr::DeepData), 1);
668 pbr::DeepData *deepData2 =
669 static_cast<pbr::DeepData*>(pbrTls->getListItem(deepLayerRay->mDeepDataHandle, 0));
670 deepData2->mHitDeep = false;
671 deepData2->mRefCount = 1;
672 deepData2->mLayer = deepData->mLayer + 1;
673 }
674
675 // Trace deep layer rays
676 pbrTls->mPrimaryRayQueue.addEntries(pbrTls->mTopLevelTls,
677 numDeepLayerRays,
678 reinterpret_cast<WrappedRayState *>(deepLayerRays),
679 arena);
680 }
681 }
682
683 {
684 // ***** Presence handling code for regular rays. See scalar code.
685
686 // Count the number of presence rays we need
687 unsigned numPresenceRays = 0;
688 for (unsigned i = 0; i < workLoadSize; ++i) {
689 if (presences[i] < 1.f - scene_rdl2::math::sEpsilon) {
690 numPresenceRays++;
691 }
692 }
693
694 // Create the continuation rays if needed or handle the case of extra presence rays.
695 if (numPresenceRays > 0 || handlePresenceContinue) {
696 // If we just have to handle presence continue, then we don't want to allocate more ray states
697 RayState **presenceRays = numPresenceRays > 0 ? pbrTls->allocRayStates(numPresenceRays) : nullptr;
698 numPresenceRays = 0;
699
700 for (unsigned i = 0; i < workLoadSize; ++i) {
701 RayState *parentRay = rayStates[i];
702 shading::Intersection *isect = &isectMemory[i];
703
704 if (presences[i] < 1.f - scene_rdl2::math::sEpsilon) {
705 float totalPresence = (1.0f - parentRay->mPathVertex.totalPresence) * presences[i];
706 RayState *presenceRay = presenceRays[numPresenceRays++];
707
708 // Compute ray epsilon
709 float pathDistance = parentRay->mPathVertex.pathDistance + parentRay->mRay.getEnd();
710 float rayEpsilon = isect->getEpsilonHint();
711 if (rayEpsilon <= 0.0f) {
712 rayEpsilon = sHitEpsilonStart * std::max(pathDistance, 1.0f);
713 }
714
715 *presenceRay = *parentRay;
716
717 if (totalPresence >= fs.mPresenceThreshold ||
718 parentRay->mPathVertex.presenceDepth >= fs.mMaxPresenceDepth) {
719 // The cleanest way to terminate presence traversal is to make it impossible for the
720 // presence continuation ray to hit any more geometry. This means we assume empty space
721 // past the last presence intersection, which will set the pixel's alpha to the
722 // total accumulated presence so far. This is done by setting the ray's near and far
723 // distances to a large value.
724 // The other option is to assume a solid material past the last presence intersection.
725 // We don't want this because that would set the pixel alpha to 1 when we really want
726 // the alpha to be the total accumulated presence.
727 // Intuitively, you would just return here but that fails to set the path throughput
728 // and alpha correctly. There is not a clean way to explicitly set the pixel alpha,
729 // especially in vector mode where it is not accessible at all from the presence code.
730 // It is also not possible to "return" from the vector code like in a traditional recursive
731 // ray tracer.
732 presenceRay->mRay.tnear = scene_rdl2::math::sMaxValue * 0.5f;
733 presenceRay->mRay.tfar = scene_rdl2::math::sMaxValue;
734 } else {
735 presenceRay->mRay.tnear = rayEpsilon;
736 presenceRay->mRay.tfar = parentRay->mRay.getOrigTfar() - parentRay->mRay.tfar;
737 }
738
739 presenceRay->mRay.setOrigTfar(presenceRay->mRay.tfar);
740 presenceRay->mPathVertex.pathDistance += parentRay->mRay.getEnd();
741 presenceRay->mPathVertex.pathPixelWeight *= (1.f - presences[i]);
742 presenceRay->mPathVertex.aovPathPixelWeight *= (1.f - presences[i]);
743 presenceRay->mPathVertex.pathThroughput *= (1.f - presences[i]);
744 presenceRay->mPathVertex.presenceDepth++;
745 presenceRay->mPathVertex.totalPresence = totalPresence;
746 presenceRay->mRay.geomID = -1;
747 presenceRay->mRay.primID = -1;
748 presenceRay->mRay.instID = -1;
749 presenceRay->mDeepDataHandle = nullHandle;
750 // Make sure to call "acquireCryptomatteData" here. This will increment the reference count of
751 // mCryptomatteDataHandle. Later in this function parentRay gets "deleted" and so
752 // mCryptomatteDataHandle's reference count will get decremented. If we don't call
753 // acquireCryptomatteData the data pointed to by the handle will get released too soon and will
754 // lead to undefined behavior.
755 presenceRay->mCryptomatteDataHandle = pbrTls->acquireCryptomatteData(
756 parentRay->mCryptomatteDataHandle);
757 parentRay->mPathVertex.pathPixelWeight *= presences[i];
758 parentRay->mPathVertex.aovPathPixelWeight *= presences[i];
759 parentRay->mPathVertex.pathThroughput *= presences[i];
760
761 // Add to the cryptomatte if we have a handle to it, and isn't a cutout:
762 if (!cutout && parentRay->mCryptomatteDataHandle != nullHandle) {
763 CryptomatteData *cryptomatteData =
764 static_cast<CryptomatteData*>(pbrTls->getListItem(parentRay->mCryptomatteDataHandle,
765 0));
766
767 unsigned px, py;
768 uint32ToPixelLocation(parentRay->mSubpixel.mPixel, &px, &py);
769
770 cryptomatteData->mCryptomatteBuffer->addSampleVector(px, py, cryptomatteData->mId,
771 parentRay->mPathVertex.pathPixelWeight);
772 cryptomatteData->mPrevPresence = 1;
773 cryptomatteData->mHit = 0;
774 }
775
776 // LPE
777 // Presence is a straight event
778 if (aovs) {
779 EXCL_ACCUMULATOR_PROFILE(pbrTls, EXCL_ACCUM_AOVS);
780 const FrameState &fs = *pbrTls->mFs;
781 const LightAovs &lightAovs = *fs.mLightAovs;
782 // transition
783 int lpeStateId = parentRay->mPathVertex.lpeStateId;
784 lpeStateId = lightAovs.straightEventTransition(pbrTls, lpeStateId);
785 presenceRay->mPathVertex.lpeStateId = lpeStateId;
786 }
787 } else if (!cutout && parentRay->mCryptomatteDataHandle != nullHandle) {
788 // Add to the cryptomatte if we came from a presence ray but we no longer hit such a case:
789 CryptomatteData *cryptomatteData =
790 static_cast<CryptomatteData*>(pbrTls->getListItem(parentRay->mCryptomatteDataHandle,
791 0));
792
793 if (!cryptomatteData->mPrevPresence) {
794 continue;
795 }
796
797 unsigned px, py;
798 uint32ToPixelLocation(parentRay->mSubpixel.mPixel, &px, &py);
799
800 cryptomatteData->mCryptomatteBuffer->addSampleVector(px, py, cryptomatteData->mId,
801 parentRay->mPathVertex.pathPixelWeight);
802 cryptomatteData->mPrevPresence = 0;
803 cryptomatteData->mHit = 0;
804 }
805 }
806
807 // Trace presence continuation rays
808 if (presenceRays) {
809 pbrTls->mIncoherentRayQueue.addEntries(pbrTls->mTopLevelTls, numPresenceRays, presenceRays, arena);
810 }
811 }
812
813 // ***** End Presence handling code for regular rays.
814 }
815
816 // Convert AOS RayState objects into SOA bundles.
817 convertAOSRayStatesToSOA(pbrTls, workLoadSize, baseRayState,
818 rayStatesSOA, &sortedEntries[0].mRsIdx);
819
820 // Send results through to the integrator...
821 fs.mIntegrator->integrateBundledv(pbrTls, shadingTls, workLoadSize, rayStatesSOA,
822 isectsSOA, bsdfv, lightList, lightFilterLists, lightAcc);
823
824 // For hybrid scalar/vectorized rendering, the scalar ray states may still
825 // be accessed during integration, so don't free them until after the
826 // integration phase has completed.
827 pbrTls->freeRayStates(workLoadSize, rayStates);
828
829 // Move on to next block of inputs.
830 sortedEntries += workLoadSize;
831 }
832
833 MNRY_ASSERT(sortedEntries == endEntry);
834
835 shadingTls->clearAttributeOffsets();
836
837 // Increment shader evaluation stats.
838 Statistics &stats = pbrTls->mStatistics;
839 stats.addToCounter(STATS_SHADER_EVALS, numEntries);
840}
