48static void
49runTest(TestBsdfSettings &test, int viewAnglesTheta,
50 int viewAnglesPhy, int sampleCount)
51{
52 //printInfo("Sample count: %d", sampleCount);
53
54 mcrt_common::ThreadLocalState *tls = mcrt_common::getFrameUpdateTLS();
55 pbr::TLState *pbrTls = MNRY_VERIFY(tls->mPbrTls.get());
56 scene_rdl2::alloc::Arena *arena = &tls->mArena;
57
58 {
59 SCOPED_MEM(arena);
60 shading::Bsdf *bsdf = test.bsdfFactory(*arena, test.frame);
61 shading::BsdfSlice slice(test.frame.getN(), test.frame.getN(), true, true,
62 ispc::SHADOW_TERMINATOR_FIX_OFF);
63 const PathGuide pg;
64 BsdfSampler sampler(pbrTls->mArena, *bsdf, slice, test.sMaxSamplesPerLobe,
65 true, pg);
66 sampleCount /= sampler.getSampleCount();
67 }
68
69// printInfo("Sampler sample count: %d", sampler.getSampleCount());
70// printInfo("Adjusted sample count: %d", sampleCount);
71
72 runBsdfTest(test, viewAnglesTheta, viewAnglesPhy, sampleCount);
73}

504TestBsdfv::testStochasticFlakes()
505{
506 scene_rdl2::math::ReferenceFrame frame;
507
508 static const size_t flakesDataSize = 100;
509
510 printInfo("##### TestBsdfv::testStochasticFlakes() ####################");
511 for (int i = 0; i < sRoughnessCount; i++) {
512 printInfo("===== Roughness: %f ====================", sRoughness[i]);
513
514 Vec3Array normals;
515 ColorArray colors;
516 generateGGXNormals(sRoughness[i] * sRoughness[i], flakesDataSize, normals);
517 generateWeightedFlakeColors(flakesDataSize, colors);
518 for (size_t j = 0; j < flakesDataSize; ++j) {
519 normals[j] = frame.localToGlobal(normals[j]);
520 }
521
522 StochasticFlakesBsdfFactory factory(&normals[0], &colors[0], flakesDataSize, 0.1f, sRoughness[i]);
523 TestBsdfSettings test(factory, frame, true, 0.008, 0.03,
524 (sRoughness[i] > 0.1f && sRoughness[i] < 0.4f),
525 (sRoughness[i] > 0.1f && sRoughness[i] < 0.4f),
526 TestBsdfSettings::BSDFV);
527 runTest(test, sViewAnglesTheta, 1, getSampleCount(sRoughness[i]));
528
529 normals.clear();
530 colors.clear();
531 }
