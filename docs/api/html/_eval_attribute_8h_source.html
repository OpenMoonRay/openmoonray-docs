49/// Evaluate an object color attribute given its key. If the attribute is bound
50/// to a map, the map shader is sampled and the result multiplied to the attribute
51/// value. Otherwise only the attribute value is returned.
52finline scene_rdl2::math::Color
53evalColor(const scene_rdl2::rdl2::Shader* const obj,
54 const scene_rdl2::rdl2::AttributeKey<scene_rdl2::rdl2::Rgb>& valueKey,
55 shading::TLState *tls,
56 const State& state)
57{
58 scene_rdl2::math::Color result = obj->get(valueKey);
59 if (!scene_rdl2::math::isBlack(result)) {
60 const scene_rdl2::rdl2::SceneObject* mapObj = obj->getBinding(valueKey);
61 if (mapObj != nullptr) {
62 const scene_rdl2::rdl2::Map* map = mapObj->asA<scene_rdl2::rdl2::Map>();
63 scene_rdl2::math::Color sampleColor;
64 sample(obj, map, tls, state, &sampleColor);
65
66 result *= sampleColor;
67 }
68 }
69 return result;
70}

90/// Evaluate normal mapping, given an object input normal attribute key and the
91/// corresponding input normal dial attribute key. The normal map is assumed to
92/// be expressed in tangent space (frame around the shading normal). Colors in
93/// R and G are re-centered from [0..1] to [-0.5..0.5].
94/// The input normal dial blends the resulting normal between the shading normal
95/// (0.0) and the normal-mapped normal (1.0). If the attribute is not bound to a
96/// map, the shading normal is returned.
97/// NOTE: This function can be safely removed after support for moonshine 5 and 6
98/// has ended
99finline scene_rdl2::math::Vec3f
100evalNormal(const scene_rdl2::rdl2::Shader* const obj,
101 const scene_rdl2::rdl2::AttributeKey<scene_rdl2::rdl2::Vec3f>& valueKey,
102 const scene_rdl2::rdl2::AttributeKey<scene_rdl2::rdl2::Float>& dialKey,
103 const scene_rdl2::rdl2::AttributeKey<scene_rdl2::rdl2::Int>& spaceKey,
104 shading::TLState *tls,
105 const State& state,
106 float& lengthN)
107{
108 const scene_rdl2::math::Vec3f &N = state.getN();
109
110 const scene_rdl2::rdl2::SceneObject* sceneObject = obj->getBinding(valueKey);
111 if (sceneObject == nullptr) {
112 return N;
113 }
114
115 float normalMapDial = obj->get(dialKey);
116 if (scene_rdl2::math::isZero(normalMapDial)) {
117 return N;
118 }
119
120 const scene_rdl2::rdl2::Map* map = sceneObject->asA<scene_rdl2::rdl2::Map>();
121 scene_rdl2::math::Color normalMapColor;
122 sample(obj, map, tls, state, &normalMapColor);
123
124 scene_rdl2::math::Vec3f normalMapValue(normalMapColor.r,
125 normalMapColor.g,
126 normalMapColor.b);
127
128 int normalMapSpace = obj->get(spaceKey);
129 return evalNormal(obj,
130 tls,
131 state,
132 normalMapValue,
133 normalMapDial,
134 normalMapSpace,
135 &lengthN);
136}

152/// Evaluate normal mapping, given a NormalMap attribute key and the
153/// corresponding input normal dial value. The normal map is assumed to
154/// be expressed in render space.
155/// The input normal dial blends the resulting normal between the shading normal
156/// (0.0) and the normal-mapped normal (1.0). If the attribute does not point to
157/// a NormalMap, the shading normal is returned.
158finline scene_rdl2::math::Vec3f
159evalNormal(const scene_rdl2::rdl2::Shader* const obj,
160 const scene_rdl2::rdl2::AttributeKey<scene_rdl2::rdl2::SceneObject*>& valueKey,
161 const float dialValue,
162 shading::TLState *tls,
163 const State& state,
164 float *lengthN = nullptr)
165{
166 const scene_rdl2::rdl2::NormalMap* normalMap = getNormalMap(obj, valueKey, dialValue);
167 if (normalMap == nullptr)
168 {
169 return state.getN();
170 }
171
172 scene_rdl2::math::Vec3f normalMapValue;
173 sampleNormal(obj, normalMap, tls, state, &normalMapValue);
174
175 return evalNormal(obj, tls, state,
176 normalMapValue, dialValue, 1 /* render space */,
177 lengthN);
178}

202/// Same as evalColor but for floats
203finline float
204evalFloat(const scene_rdl2::rdl2::Shader* const obj,
205 const scene_rdl2::rdl2::AttributeKey<scene_rdl2::rdl2::Float>& valueKey,
206 shading::TLState *tls,
207 const State& state)
208{
209 float result = obj->get(valueKey);
210 if (!scene_rdl2::math::isZero(result)) {
211 const scene_rdl2::rdl2::SceneObject* mapObj = obj->getBinding(valueKey);
212 if (mapObj != nullptr) {
213 const scene_rdl2::rdl2::Map* map = mapObj->asA<scene_rdl2::rdl2::Map>();
214 scene_rdl2::math::Color sampleColor;
215 sample(obj, map, tls, state, &sampleColor);
216
217 float avg = (sampleColor.r + sampleColor.g + sampleColor.b) / 3.0f;
218 result *= avg;
219 }
220 }
221 return result;
222}
