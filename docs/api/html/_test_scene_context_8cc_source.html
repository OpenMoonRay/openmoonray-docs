35void
36TestSceneContext::testUpdate()
37{
38 scene_rdl2::rdl2::SceneContext context;
39 const scene_rdl2::rdl2::SceneClass* updateTracker = context.createSceneClass("UpdateTracker");
40 scene_rdl2::rdl2::Map* tracker = context.createSceneObject(
41 "UpdateTracker", "/seq/shot/tracker")->asA<scene_rdl2::rdl2::Map>();
42 scene_rdl2::rdl2::Map* other = context.createSceneObject(
43 "UpdateTracker", "/seq/shot/other")->asA<scene_rdl2::rdl2::Map>();
44 scene_rdl2::rdl2::AttributeKey<scene_rdl2::rdl2::Float> pizzaKey =
45 updateTracker->getAttributeKey<scene_rdl2::rdl2::Float>("pizza");
46 scene_rdl2::rdl2::AttributeKey<scene_rdl2::rdl2::Int> cookieKey =
47 updateTracker->getAttributeKey<scene_rdl2::rdl2::Int>("cookie");
48
49 // Make some updates and commit them.
50 tracker->beginUpdate();
51 tracker->set(pizzaKey, 2.34f);
52 tracker->setBinding(pizzaKey, other);
53 tracker->set(cookieKey, 13);
54 tracker->endUpdate();
55 tracker->commitChanges();
56
57 tracker->beginUpdate();
58 tracker->set(pizzaKey, 3.45f);
59 tracker->set(cookieKey, 14);
60 tracker->endUpdate();
61 tracker->commitChanges();
62
63 tracker->beginUpdate();
64 tracker->set(pizzaKey, 4.56f);
65 tracker->endUpdate();
66 tracker->commitChanges();
67
68 // Verify the update counts.
69 moonray::shading::Intersection intersection;
70 moonray::shading::State state(&intersection);
71 scene_rdl2::math::Color result;
72 tracker->sample(nullptr, state, &result);
73 int timesPizzaUpdated = result.r;
74 int timesPizzaBindingUpdated = result.g;
75 int timesCookieUpdated = result.b;
76 CPPUNIT_ASSERT_EQUAL(3, timesPizzaUpdated);
77 CPPUNIT_ASSERT_EQUAL(1, timesPizzaBindingUpdated);
78 CPPUNIT_ASSERT_EQUAL(2, timesCookieUpdated);
79}
