32// The notation used in this module is based on the paper:
33// "Microfacet Models for Refraction through Rough Surfaces"
34// (Walter et al. - EGSR 2007)
35//
36// The formula for the reflection term (or BRDF) of the BSDF
37// described in the paper is the following (eq. 20 with m == Hr):
38// (Note: except for the Anisotropic Cook-Torrance)
39//
40// F(i, m) * G(i, o, m) * D(m)
41// reflectance(i,o,n) = -----------------------------
42// 4 * |i.n| * |o.n|
43//
44// Where the Fresnel term is provided by the lobe Fresnel closure
45//
46// where:
47// i - incoming (light) direction
48// o - outgoing (view) direction
49// n - surface normal
50// m - half-direction for reflection
51// F - Fresnel term
52// G - shadow-masking function
53

57CookTorranceBsdfLobe::CookTorranceBsdfLobe(const Vec3f &N,
58 float roughness,
59 const Color& favg,
60 const Color& favgInv,
61 float etaI,
62 float etaT,
63 bool coupledWithTransmission) :
64 BsdfLobe(Type(REFLECTION | GLOSSY), DifferentialFlags(0), false,
65 PROPERTY_NORMAL | PROPERTY_ROUGHNESS | PROPERTY_PBR_VALIDITY),
66 mFrame(N),
67 mInputRoughness(roughness),
68 mFavg(favg),
69 mFavgInv(favgInv),
70 mEtaI(etaI),
71 mEtaT(etaT),
72 mCoupledWithTransmission(coupledWithTransmission)
73{
74 // Apply roughness squaring to linearize roughness response
75 // See "Physically-Based Shading at Disney" Siggraph course notes.
76 mRoughness = mInputRoughness * mInputRoughness;
77 if (mRoughness < 0.001f) { // TODO: Smaller thresholds trigger nans
78 mRoughness = 0.001f;
79 }
80
81 mInvRoughness = rcp(mRoughness);
82
83 // Use a directional differential scale that varies with roughness
84 mdDFactor = sdDFactorMin + mRoughness * sdDFactorSlope;
85}

87void
88CookTorranceBsdfLobe::computeBeckmannMicrofacet(float r1, float r2,
89 float cosNO,
90 scene_rdl2::math::Vec3f &m) const
91{
92 // generate a random microfacet normal m (eq. 28, 29)
93 // we take advantage of cos(atan(x)) == 1/sqrt(1+x^2)
94 // and sin(atan(x)) == x/sqrt(1+x^2)
95 const float alpha = widenRoughness(mRoughness, cosNO);
96 const float alpha2 = alpha * alpha;
97 const float tanThetaMSqr = -alpha2 * scene_rdl2::math::log(1.0f - r1);
98 const float tanThetaM = scene_rdl2::math::sqrt(tanThetaMSqr);
99 const float cosThetaM = rsqrt(1.0f + tanThetaMSqr);
100 const float sinThetaM = cosThetaM * tanThetaM;
101 const float phiM = 2.0f * sPi * r2;
102
103 // Compute the half vector
104 m = computeLocalSphericalDirection(cosThetaM, sinThetaM, phiM);
105 m = mFrame.localToGlobal(m);
106}

108Color
109CookTorranceBsdfLobe::eval(const BsdfSlice &slice, const Vec3f &wi,
110 float *pdf) const
111{
112 // Based on the paper:
113 // "Microfacet Models for Refraction through Rough Surfaces"
114 // (Walter et al. - EGSR 2007)
115
116 // Prepare for early exit
117 if (pdf != NULL) {
118 *pdf = 0.0f;
119 }
120
121 float cosNO = dot(mFrame.getN(), slice.getWo());
122 if (cosNO <= sEpsilon) return sBlack;
123 cosNO = min(cosNO, sOneMinusEpsilon);
124
125 float cosNI = dot(mFrame.getN(), wi);
126 if (cosNI <= sEpsilon) return sBlack;
127 cosNI = min(cosNI, sOneMinusEpsilon);
128
129 // get half vector
130 Vec3f m;
131 if (!computeNormalizedHalfVector(slice.getWo(), wi, m)) {
132 return sBlack;
133 }
134
135 const float cosMI = dot(m, wi);
136 if (cosMI <= sEpsilon) return sBlack;
137
138 const float cosThetaM = dot(m, mFrame.getN());
139 if (cosThetaM <= sEpsilon) return sBlack;
140
141 // eq. 25: calculate D(m):
142 const float alpha2 = mRoughness * mRoughness;
143 const float cosThetaM2 = cosThetaM * cosThetaM;
144 const float cosThetaM4 = cosThetaM2 * cosThetaM2;
145 const float minusTanThetaM2 = (cosThetaM2 - 1.0f) * rcp(cosThetaM2);
146 const float D4 = 0.25f * scene_rdl2::math::exp(minusTanThetaM2 * rcp(alpha2)) *
147 rcp(sPi * alpha2 * cosThetaM4);
148
149 // eq. 26, 27: now calculate G1(i,m) and G1(o,m)
150 const float ao = cosNO * mInvRoughness * rsqrt(1.0f - cosNO * cosNO);
151 const float ai = cosNI * mInvRoughness * rsqrt(1.0f - cosNI * cosNI);
152 const float G1o = (ao < 1.6f) ? (3.535f * ao + 2.181f * ao * ao) *
153 rcp(1.0f + 2.276f * ao + 2.577f * ao * ao) : 1.0f;
154 const float G1i = (ai < 1.6f) ? (3.535f * ai + 2.181f * ai * ai) *
155 rcp(1.0f + 2.276f * ai + 2.577f * ai * ai) : 1.0f;
156 const float G = G1o * G1i;
157
158
159 // Compute the Cook-Torrance bsdf
160 Color f = computeScaleAndFresnel(cosMI) * G * D4 * rcp(cosNO) *
161 (slice.getIncludeCosineTerm() ? 1.0f : rcp(cosNI));
162
163 const float w2 = energyCompensationWeight();
164
165 // only add energy if weight > 0 -- if we don't do this, we will add energy without
166 // contributing to the pdf or sampling from the lobe, leading to fireflies
167 // TODO: investigate energy compensation for transmissive lobes -- this seems to be
168 // where the issue is
169 if (w2 > 0.0) {
170 const Color compen = evalEnergyCompensation(ispc::MICROFACET_DISTRIBUTION_BECKMANN,
171 cosNO, cosNI,
172 slice.getIncludeCosineTerm());
173 f += compen;
174 }
175
176 if (pdf != NULL) {
177#if PBR_CT_SAMPLE_WIDENED_ROUGHNESS
178 // Compute pdf of sampling wi using eq. (17) in [1]
179 // pdf(wi) = D(m) * |m.n| / (4 * |o.m|)
180 // We cannot re-use D4 computed above without adjusting for the widened
181 // roughness we use in sample(), otherwise our pdf is biased
182 const float alpha = widenRoughness(mRoughness, cosNO);
183 const float alpha2 = alpha * alpha;
184 const float D4 = 0.25f * scene_rdl2::math::exp(minusTanThetaM2 * rcp(alpha2)) *
185 rcp(sPi * alpha2 * cosThetaM4);
186#endif
187 const float p2 = energyCompensationPDF(ispc::MICROFACET_DISTRIBUTION_BECKMANN,
188 cosNI);
189 const float w1 = (1.0f - w2);
190 const float p1 = D4 * cosThetaM * rcp(cosMI);
191 // One Sample PDF Weight
192 *pdf = (w1 * p1 + w2 * p2);
193 }
194
195 // Soften hard shadow terminator due to shading normals
196 const float Gs = slice.computeShadowTerminatorFix(mFrame.getN(), wi);
197
198 return Gs * f;
199}

201Color
202CookTorranceBsdfLobe::sample(const BsdfSlice &slice, float r1, float r2,
203 Vec3f &wi, float &pdf) const
204{
205 const float cosNO = dot(mFrame.getN(), slice.getWo());
206 if (cosNO <= 0.0f) {
207 wi = Vec3f(zero);
208 pdf = 0.0f;
209 return sBlack;
210 }
211
212 // One Sample between energy compensation and regular
213 // microfacet distribution sampling
214 const float w2 = energyCompensationWeight();
215 if (r1 < w2) {
216 r1 = r1 / w2;
217 sampleEnergyCompensation(ispc::MICROFACET_DISTRIBUTION_BECKMANN,
218 slice.getWo(), r1, r2, wi);
219 } else {
220
221 const float w1 = (1.0f - w2);
222 r1 = (r1 - w2) / w1;
223
224 Vec3f m;
225 computeBeckmannMicrofacet(r1, r2, cosNO, m);
226 // Compute reflection direction about the half vector
227 computeReflectionDirection(m, slice.getWo(), wi);
228 }
229
230 Color brdf = eval(slice, wi, &pdf);
231 return brdf;
232}

327void
328CookTorranceBsdfLobe::differentials(const Vec3f &wo, const Vec3f &wi,
329 float r1, float r2, const Vec3f &dNdx, const Vec3f &dNdy,
330 Vec3f &dDdx, Vec3f &dDdy) const
331{
332 // It's complex to capture the full derivative. Instead we use the
333 // derivative of a mirror reflection about the H vector, and scale the
334 // length of the directional derivative proportionally with roughness.
335 Vec3f H;
336 if (!computeNormalizedHalfVector(wo, wi, H)) {
337 H = mFrame.getN();
338 }
339 computeReflectionDirectionDifferential(wo, wi, H, dNdx, dNdy, dDdx, dDdy);
340 dDdx *= mdDFactor;
341 dDdy *= mdDFactor;
342}

384void
385TransmissionCookTorranceBsdfLobe::differentials(const Vec3f &wo, const Vec3f &wi,
386 float r1, float r2, const Vec3f &dNdx, const Vec3f &dNdy,
387 Vec3f &dDdx, Vec3f &dDdy) const
388{
389 // It's complex to capture the full derivative. Instead we use the
390 // derivative of a mirror refraction about the H vector, and scale the
391 // length of the directional derivative proportionally with roughness.
392 Vec3f H;
393 if (computeNormalizedRefractionHalfVector(mEtaI, wo,
394 mEtaT, wi, H)) {
395 // Make sure microfacet points to the same side of the surface wrt. N and wo
396 H = (mEtaI > mEtaT ? -H : H);
397 } else {
398 H = mFrame.getN();
399 }
400 computeRefractionDirectionDifferential(mNeta, wo, wi, H, dNdx, dNdy,
401 dDdx, dDdy);
402 dDdx *= mdDFactor;
403 dDdy *= mdDFactor;

419Color
420GGXCookTorranceBsdfLobe::eval(const BsdfSlice &slice, const Vec3f &wi,
421 float *pdf, const float cosNO, const float cosNI,
422 const ReferenceFrame& frame) const
423{
424 // Based on the paper:
425 // "Microfacet Models for Refraction through Rough Surfaces"
426 // (Walter et al. - EGSR 2007)
427
428 // Prepare for early exit
429 if (pdf != NULL) {
430 *pdf = 0.0f;
431 }
432
433 if (cosNO <= 0.0f || cosNI <= 0.0f) {
434 return sBlack;
435 }
436
437 // get half vector
438 Vec3f m;
439 if (!computeNormalizedHalfVector(slice.getWo(), wi, m)) {
440 return sBlack;
441 }
442
443 const float cosMI = dot(m, wi);
444 if (cosMI <= 0.0f) return sBlack;
445
446 const float cosThetaM = dot(m, frame.getN());
447 if (cosThetaM <= 0.0f) return sBlack;
448
449 // eq. 33: calculate D(m):
450 const float alpha2 = mRoughness * mRoughness;
451 const float cosThetaM2 = cosThetaM * cosThetaM;
452 const float piCosThetaM4 = sPi * cosThetaM2 * cosThetaM2;
453 const float tanThetaM2 = (1.0f - cosThetaM2) / cosThetaM2;
454 const float tmp = alpha2 + tanThetaM2;
455 const float D4 = 0.25f * alpha2 / (piCosThetaM4 * tmp * tmp);
456
457 // eq. 34: now calculate G1(i,m) and G1(o,m)
458 const float invCosNO = 1.0f / cosNO;
459 const float invCosNI = 1.0f / cosNI;
460 const float G1o = 2.0f / (1.0f + scene_rdl2::math::sqrt(1.0f + alpha2 *
461 (1.0f - cosNO * cosNO) * (invCosNO * invCosNO)));
462 const float G1i = 2.0f / (1.0f + scene_rdl2::math::sqrt(1.0f + alpha2 *
463 (1.0f - cosNI * cosNI) * (invCosNI * invCosNI)));
464 const float G = G1o * G1i;
465
466 // Compute the Cook-Torrance bsdf
467 Color f = computeScaleAndFresnel(cosMI) * G * D4 * invCosNO *
468 (slice.getIncludeCosineTerm() ? 1.0f : invCosNI);
469
470 const float w2 = energyCompensationWeight();
471
472 // only add energy if weight > 0 -- if we don't do this, we will add energy without
473 // contributing to the pdf or sampling from the lobe, leading to fireflies
474 // TODO: investigate energy compensation for transmissive lobes -- this seems to be
475 // where the issue is
476 if (w2 > 0.0) {
477 const Color compen = evalEnergyCompensation(ispc::MICROFACET_DISTRIBUTION_GGX,
478 cosNO, cosNI,
479 slice.getIncludeCosineTerm());
480 f = f + compen;
481 }
482
483 // Compute pdf of sampling wi using eq. (17) in [1]
484 // pdf(wi) = D(m) * |m.n| / (4 * |o.m|)
485 if (pdf != NULL) {
486#if PBR_CT_SAMPLE_WIDENED_ROUGHNESS
487 // We cannot re-use D4 computed above without adjusting for the widened
488 // roughness we use in sample(), otherwise our pdf is biased
489 const float alpha = widenRoughness(mRoughness, cosNO);
490 const float alpha2 = alpha * alpha;
491 const float tmp = alpha2 + tanThetaM2;
492 const float D4 = 0.25f * alpha2 / (piCosThetaM4 * tmp * tmp);
493#endif
494 const float p2 = energyCompensationPDF(ispc::MICROFACET_DISTRIBUTION_GGX,
495 cosNI);
496 const float w1 = (1.0f - w2);
497 const float p1 = D4 * cosThetaM * rcp(cosMI);
498 // One Sample PDF Weight
499 *pdf = (w1 * p1 + w2 * p2);
500 }
501
502 return f;
503}

505Vec3f
506GGXCookTorranceBsdfLobe::sample(const BsdfSlice &slice,
507 float cosNO,
508 float r1, float r2,
509 const ReferenceFrame& frame) const
510{
511 // One Sample between energy compensation and regular
512 // microfacet distribution sampling
513 Vec3f wi;
514 // generate a random microfacet normal m (eq. 35,36):
515 // we take advantage of cos(atan(x)) == 1/sqrt(1+x^2)
516 // and sin(atan(x)) == x/sqrt(1+x^2)
517 const float alpha = widenRoughness(mRoughness, cosNO);
518 const float alpha2 = alpha * alpha;
519 const float tanThetaM2 = alpha2 * r1 / (1.0f - r1);
520 const float cosThetaM = 1.0f / scene_rdl2::math::sqrt(1.0f + tanThetaM2);
521 const float sinThetaM = cosThetaM * scene_rdl2::math::sqrt(tanThetaM2);
522 const float phiM = sTwoPi * r2;
523
524 // Compute the half vector
525 Vec3f m = computeLocalSphericalDirection(cosThetaM, sinThetaM, phiM);
526 m = frame.localToGlobal(m);
527
528 computeReflectionDirection(m, slice.getWo(), wi);
529 return wi;
530}

533GGXCookTorranceBsdfLobe::eval(const BsdfSlice &slice, const Vec3f &wi,
534 float *pdf) const
535{
536 const float cosNO = dot(mFrame.getN(), slice.getWo());
537 const float cosNI = dot(mFrame.getN(), wi);
538
539 // Soften hard shadow terminator due to shading normals
540 // The code is here to avoid applying the softening in the case of glitter
541 const float Gs = slice.computeShadowTerminatorFix(mFrame.getN(), wi);
542
543 return Gs * eval(slice, wi, pdf, cosNO, cosNI, mFrame);

546Color
547GGXCookTorranceBsdfLobe::sample(const BsdfSlice &slice, float r1, float r2,
548 Vec3f &wi, float &pdf) const
549{
550 const float cosNO = dot(mFrame.getN(), slice.getWo());
551 if (cosNO <= 0.0f) {
552 wi = Vec3f(zero);
553 pdf = 0.0f;
554 return sBlack;
555 }
556
557 const float w2 = energyCompensationWeight();
558 if (r1 < w2) {
559 r1 = r1 / w2;
560 sampleEnergyCompensation(ispc::MICROFACET_DISTRIBUTION_GGX,
561 slice.getWo(), r1, r2, wi);
562 } else {
563
564 const float w1 = (1.0f - w2);
565 r1 = (r1 - w2) / w1;
566 wi = sample(slice, cosNO, r1, r2, mFrame);
567 }
568
569 return eval(slice, wi, &pdf);
570}

584Color
585GlitterGGXCookTorranceBsdfLobe::eval(const BsdfSlice &slice, const Vec3f &wi,
586 float *pdf) const
587{
588 // Try evaluation with the flake normal
589 float cosNO = dot(mFlakeNormal, slice.getWo());
590 float cosNI = dot(mFlakeNormal, wi);
591
592 ReferenceFrame frame;
593 // If flake normal is too perturbed, evaluate with shading normal
594 if (cosNO <= 0.0f || cosNI <= 0.0f) {
595 frame = mFrame;
596 cosNO = dot(frame.getN(), slice.getWo());
597 cosNI = dot(frame.getN(), wi);
598 } else {
599 frame = ReferenceFrame(mFlakeNormal);
600 }
601
602 return GGXCookTorranceBsdfLobe::eval(slice, wi, pdf, cosNO, cosNI, frame);
603}

605Color
606GlitterGGXCookTorranceBsdfLobe::sample(const BsdfSlice &slice, float r1, float r2,
607 Vec3f &wi, float &pdf) const
608{
609 // Try sampling with the flake normal
610 bool isUsingFlakeNormal = true;
611 float cosNO = dot(mFlakeNormal, slice.getWo());
612
613 ReferenceFrame frame;
614 // If wo is outside the hemisphere of the flake normal, try sampling with shading normal
615 if (cosNO <= 0.0f) {
616 frame = mFrame;
617 cosNO = dot(frame.getN(), slice.getWo());
618 isUsingFlakeNormal = false;
619 // Unsuccessful even with shading normal
620 if (cosNO <= 0.0f) {
621 wi = Vec3f(zero);
622 pdf = 0.0f;
623 return sBlack;
624 }
625 } else {
626 frame = ReferenceFrame(mFlakeNormal);
627 }
628
629 wi = GGXCookTorranceBsdfLobe::sample(slice, cosNO, r1, r2, frame);
630
631 float cosNI = dot(frame.getN(), wi);
632 // If wi is created with flake normal, check if it is within the hemisphere of shading normal
633 // If not, generate another wi using the shading normal instead
634 if (isUsingFlakeNormal) {
635 const float cosNIShading = dot(mFrame.getN(), wi);
636 if (cosNIShading <= 0.0f) {
637 frame = mFrame;
638 cosNO = dot(frame.getN(), slice.getWo());
639 // Unsuccessful even with shading normal
640 if (cosNO <= 0.0f) {
641 wi = Vec3f(zero);
642 pdf = 0.0f;
643 return sBlack;
644 }
645 wi = GGXCookTorranceBsdfLobe::sample(slice, cosNO, r1, r2, frame);
646 cosNI = dot(frame.getN(), wi);
647 }
648 }
649
650 return GGXCookTorranceBsdfLobe::eval(slice, wi, &pdf, cosNO, cosNI, frame);
651}

685Color
686BerryCookTorranceBsdfLobe::eval(const BsdfSlice &slice, const Vec3f &wi,
687 float *pdf) const
688{
689 // Based on the paper:
690 // "Physically-Based Shading at Disney"
691 // (Burley et al. - Siggraph Course 2012)
692
693 // Prepare for early exit
694 if (pdf != NULL) {
695 *pdf = 0.0f;
696 }
697
698 const float cosNO = dot(mFrame.getN(), slice.getWo());
699 if (cosNO <= 0.0f) return sBlack;
700
701 const float cosNI = dot(mFrame.getN(), wi);
702 if (cosNI <= 0.0f) return sBlack;
703
704 // get half vector
705 Vec3f m;
706 if (!computeNormalizedHalfVector(slice.getWo(), wi, m)) {
707 return sBlack;
708 }
709
710 const float cosMI = dot(m, wi);
711 if (cosMI <= 0.0f) return sBlack;
712
713 const float cosThetaM = dot(m, mFrame.getN());
714 if (cosThetaM <= 0.0f) return sBlack;
715
716 // eq. "D_Berry": calculate D(m) (with m=Hr):
717 const float alpha2 = mRoughness * mRoughness;
718 const float cosThetaM2 = cosThetaM * cosThetaM;
719 const float sinThetaM2 = 1.0f - cosThetaM2;
720 const float d0 = (alpha2 - 1.0f) / (sPi * scene_rdl2::math::log(alpha2));
721 const float d1 = 1.0f / (alpha2 * cosThetaM2 + sinThetaM2);
722 const float D4 = 0.25f * d0 * d1;
723
724 // Per section 5.6 "Specular G details", use GGX shadowing and masking functions
725 // However, we don't seem to use the remapping: alpha2 = (0.5 + mRoughness / 2)^2
726 const float invCosNO = 1.0f / cosNO;
727 const float invCosNI = 1.0f / cosNI;
728 const float G1o = 2.0f / (1.0f + scene_rdl2::math::sqrt(1.0f + alpha2 *
729 (1.0f - cosNO * cosNO) * (invCosNO * invCosNO)));
730 const float G1i = 2.0f / (1.0f + scene_rdl2::math::sqrt(1.0f + alpha2 *
731 (1.0f - cosNI * cosNI) * (invCosNI * invCosNI)));
732 const float G = G1o * G1i;
733
734 // Compute the Cook-Torrance bsdf
735 const Color f = computeScaleAndFresnel(cosMI) * G * D4 * invCosNO *
736 (slice.getIncludeCosineTerm() ? 1.0f : invCosNI);
737
738 // Compute pdf of sampling wi using eq. (17) in [1]
739 // pdf(wi) = D(m) * |m.n| / (4 * |o.m|)
740 if (pdf != NULL) {
741#if PBR_CT_SAMPLE_WIDENED_ROUGHNESS
742 // We cannot re-use D4 computed above without adjusting for the widened
743 // roughness we use in sample(), otherwise our pdf is biased
744 const float alpha = widenRoughness(mRoughness, cosNO);
745 const float alpha2 = alpha * alpha;
746 const float d0 = (alpha2 - 1.0f) / (sPi * scene_rdl2::math::log(alpha2));
747 const float d1 = 1.0f / (alpha2 * cosThetaM2 + sinThetaM2);
748 const float D4 = 0.25f * d0 * d1;
749#endif
750 *pdf = D4 * cosThetaM / cosMI;
751 }
752
753 // Soften hard shadow terminator due to shading normals
754 const float Gs = slice.computeShadowTerminatorFix(mFrame.getN(), wi);
755
756 return Gs * f;
757}

760Color
761BerryCookTorranceBsdfLobe::sample(const BsdfSlice &slice, float r1, float r2, Vec3f &wi,
762 float &pdf) const
763{
764 const float cosNO = dot(mFrame.getN(), slice.getWo());
765 if (cosNO <= 0.0f) {
766 wi = Vec3f(zero);
767 pdf = 0.0f;
768 return sBlack;
769 }
770
771 // generate a random microfacet normal m (eq. 35,36):
772 // using: tan(acos(x)) == sqrt(1-x^2) / x
773 // sin(x) == cos(x) * tan(x)
774 const float alpha = widenRoughness(mRoughness, cosNO);
775 const float alpha2 = alpha * alpha;
776 const float cosThetaM = scene_rdl2::math::sqrt((1.0f - scene_rdl2::math::pow(alpha2, (1.0f - r1))) / (1.0f - alpha2));
777 const float tanThetaM = scene_rdl2::math::sqrt(1.0f - cosThetaM * cosThetaM) / cosThetaM;
778 const float sinThetaM = cosThetaM * tanThetaM;
779 const float phiM = 2.0f * sPi * r2;
780
781 Vec3f m = computeLocalSphericalDirection(cosThetaM, sinThetaM, phiM);
782 // TODO: check and remove: m.normalize();
783 m = mFrame.localToGlobal(m);
784
785 // Compute reflection direction about the half vector
786 computeReflectionDirection(m, slice.getWo(), wi);
787
788 return eval(slice, wi, &pdf);
789}

815Color
816AnisoCookTorranceBsdfLobe::eval(const BsdfSlice &slice, const Vec3f &wi,
817 float *pdf) const
818{
819 // Based on the papers:
820 // [1]: "Notes on Ward BRDF"
821 // Bruce Walter, 2005, http://www.graphics.cornell.edu/~bjw/wardnotes.pdf
822 // [2]: "Microfacet Models for Refraction through Rough Surfaces"
823 // Walter et al., EGSR 2007
824 // [3]: "Understanding the Masking-Shadowing Function in Microfacet-Based BRDFs"
825 // Eric Heitz, Journal of Computer Graphics Techniques, JCGT, 2014
826
827 // Prepare for early exit
828 if (pdf != NULL) {
829 *pdf = 0.0f;
830 }
831
832 const float cosNO = dot(mFrame.getN(), slice.getWo());
833 if (cosNO <= 0.0f) return sBlack;
834
835 const float cosNI = dot(mFrame.getN(), wi);
836 if (cosNI <= 0.0f) return sBlack;
837
838 // get half vector
839 Vec3f m;
840 if (!computeNormalizedHalfVector(slice.getWo(), wi, m)) {
841 return sBlack;
842 }
843
844 const float cosMI = dot(m, wi);
845 if (cosMI <= 0.0f) return sBlack;
846
847 const float cosThetaM = dot(m, mFrame.getN());
848 if (cosThetaM <= 0.0f) return sBlack;
849
850 // Calculate D(m) (with m=Hr), using eq. (82) in [3]. Note that we use the
851 // (exact and cheaper) vector form of the exponential term from eq. (4) in [1].
852 const float dotMX = dot(m, mFrame.getX());
853 const float dotMY = dot(m, mFrame.getY());
854 const float cosThetaMX = dotMX / mRoughness;
855 const float cosThetaMX2 = cosThetaMX * cosThetaMX;
856 const float cosThetaMY = dotMY / mRoughnessV;
857 const float cosThetaMY2 = cosThetaMY * cosThetaMY;
858 const float cosThetaM2 = cosThetaM * cosThetaM;
859 const float D4 = 0.25f * scene_rdl2::math::exp(-(cosThetaMX2 + cosThetaMY2) / cosThetaM2)
860 / (sPi * mRoughness * mRoughnessV * cosThetaM2 * cosThetaM2);
861
862 // Calculate G1(i,m) and G1(o,m) using equations (83) and (80) in [3].
863 // First we simplify the expression for a, also using the exact vector form
864 // (similarly to how [1] derives the exponential term in eq. (4)):
865 // a = (v.n) / sqrt(((v.x)*alphax)^2 + ((v.y)*alphay)^2)
866 const float tmpXO = dot(mFrame.getX(), slice.getWo()) * mRoughness;
867 const float tmpYO = dot(mFrame.getY(), slice.getWo()) * mRoughnessV;
868 const float ao = cosNO / scene_rdl2::math::sqrt(tmpXO * tmpXO + tmpYO * tmpYO);
869 const float tmpXI = dot(mFrame.getX(), wi) * mRoughness;
870 const float tmpYI = dot(mFrame.getY(), wi) * mRoughnessV;
871 const float ai = cosNI / scene_rdl2::math::sqrt(tmpXI * tmpXI + tmpYI * tmpYI);
872 // Use the rational approximation of G1: eq. (27) in [2]
873 const float G1o = (ao < 1.6f) ? (3.535f * ao + 2.181f * ao * ao) /
874 (1.0f + 2.276f * ao + 2.577f * ao * ao) : 1.0f;
875 const float G1i = (ai < 1.6f) ? (3.535f * ai + 2.181f * ai * ai) /
876 (1.0f + 2.276f * ai + 2.577f * ai * ai) : 1.0f;
877 const float G = G1o * G1i;
878
879 // Compute the Cook-Torrance bsdf
880 const Color f = computeScaleAndFresnel(cosMI) * G * D4 / cosNO *
881 (slice.getIncludeCosineTerm() ? 1.0f : 1.0f / cosNI);
882
883 // Compute pdf of sampling wi using eq. (17) in [1]
884 // pdf(wi) = D(m) * |m.n| / (4 * |o.m|)
885 if (pdf != NULL) {
886#if PBR_CT_SAMPLE_WIDENED_ROUGHNESS
887 // We cannot re-use D4 computed above without adjusting for the widened
888 // roughness we use in sample(), otherwise our pdf is biased
889 const float alphaU = widenRoughness(mRoughness, cosNO);
890 const float alphaV = widenRoughness(mRoughnessV, cosNO);
891 const float cosThetaMX = dotMX / alphaU;
892 const float cosThetaMX2 = cosThetaMX * cosThetaMX;
893 const float cosThetaMY = dotMY / alphaV;
894 const float cosThetaMY2 = cosThetaMY * cosThetaMY;
895 const float D4 = 0.25f * scene_rdl2::math::exp(-(cosThetaMX2 + cosThetaMY2) / cosThetaM2)
896 / (sPi * alphaU * alphaV * cosThetaM2 * cosThetaM2);
897#endif
898 *pdf = D4 * cosThetaM / cosMI;
899 }
900
901 // Soften hard shadow terminator due to shading normals
902 const float Gs = slice.computeShadowTerminatorFix(mFrame.getN(), wi);
903
904 return Gs * f;
905}

908Color
909AnisoCookTorranceBsdfLobe::sample(const BsdfSlice &slice, float r1, float r2,
910 Vec3f &wi, float &pdf) const
911{
912 const float cosNO = dot(mFrame.getN(), slice.getWo());
913 if (cosNO <= 0.0f) {
914 wi = Vec3f(zero);
915 pdf = 0.0f;
916 return sBlack;
917 }
918
919 // We use the anisotropic sampling equations (6) and (7) in [1], which apply
920 // directly here, since they are in fact sampling the anisotropic
921 // Beckmann distribution we use here (which the Ward model is also based on).
922
923 // For sampling, use the modified sampling distribution in sect 5.3 in [2]
924 // Reduce maximum weight with widened distribution
925 const float alphaU = widenRoughness(mRoughness, cosNO);
926 const float alphaV = widenRoughness(mRoughnessV, cosNO);
927 const float mInvAlphaUSqrd = 1.0f / (alphaU * alphaU);
928 const float mInvAlphaVSqrd = 1.0f / (alphaV * alphaV);
929
930 // Sample phi and compute sin and cos
931 // This is equivalent (but faster) to using eq. (7) in [1].
932 // Using the identity: cos(x)^2 = 1 / (1 + tan(x)^2)
933 const float phi = sTwoPi * r2;
934 const float cosOrigPhi = alphaU * scene_rdl2::math::cos(phi);
935 const float sinOrigPhi = alphaV * scene_rdl2::math::sin(phi);
936 const float cosSqrdPhi = 1.0f
937 / (sinOrigPhi * sinOrigPhi / (cosOrigPhi * cosOrigPhi) + 1.0f);
938 const float sinSqrdPhi = 1.0f - cosSqrdPhi;
939 const float sinPhiM = (sinOrigPhi < 0.0f ? -scene_rdl2::math::sqrt(sinSqrdPhi) : scene_rdl2::math::sqrt(sinSqrdPhi));
940 const float cosPhiM = (cosOrigPhi < 0.0f ? -scene_rdl2::math::sqrt(cosSqrdPhi) : scene_rdl2::math::sqrt(cosSqrdPhi));
941
942 // Sample theta using eq (6) in [1], also simplifying using the identity above
943 const float denominator = cosSqrdPhi * mInvAlphaUSqrd + sinSqrdPhi * mInvAlphaVSqrd;
944 const float cosThetaMSqrd = denominator /
945 (denominator - scene_rdl2::math::log(max(r1, sEpsilon)));
946 const float cosThetaM = scene_rdl2::math::sqrt(cosThetaMSqrd);
947 const float sinThetaM = scene_rdl2::math::sqrt(1.0f - cosThetaMSqrd);
948 MNRY_ASSERT(cosThetaM > sEpsilon);
949
950 // Compute the half vector
951 Vec3f m = computeLocalSphericalDirection(cosThetaM, sinThetaM, cosPhiM, sinPhiM);
952 m = mFrame.localToGlobal(m);
953
954 // Compute reflection direction about the half vector
955 computeReflectionDirection(m, slice.getWo(), wi);
956
957 return eval(slice, wi, &pdf);
958}
