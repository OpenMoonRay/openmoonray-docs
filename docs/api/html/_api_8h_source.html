64/// @brief helper function that concatenate primitive to render space
65/// transform samples
66/// @param state State object for current Procedural
67/// @param parent2render transform from parent space to render space
68/// that is passed in to each Procedural::generate/update call
69/// @return primitive to render space transform
70finline shading::XformSamples
71computePrim2Render(const State& state, const shading::XformSamples& parent2render)
72{
73 shading::XformSamples prim2render;
74 prim2render.reserve(parent2render.size());
75 for (const auto& p2r : parent2render) {
76 prim2render.push_back(state.getProc2Parent() * p2r);
77 }
78 return prim2render;
79}

81/// @brief helpter function that concatenate two shading::XformSamples
82/// @param xformA shading::XformSamples
83/// @param xformB shading::XformSamples
84/// @throws AssertionError if both xformA and xformB contains more than one
85/// xform sample and the number of sample doesn't match
86/// @return shading::XformSamples that is the concatenation of xformA and xformB
87finline shading::XformSamples
88concatenate(const shading::XformSamples& xformA, const shading::XformSamples& xformB)
89{
90 MNRY_ASSERT_REQUIRE(!xformA.empty() && !xformB.empty());
91 shading::XformSamples result;
92 if (xformA.size() == 1) {
93 for (size_t i = 0; i < xformB.size(); ++i) {
94 result.push_back(xformA[0] * xformB[i]);
95 }
96 } else if (xformB.size() == 1) {
97 for (size_t i = 0; i < xformA.size(); ++i) {
98 result.push_back(xformA[i] * xformB[0]);
99 }
100 } else {
101 MNRY_ASSERT_REQUIRE(xformA.size() == xformB.size());
102 for (size_t i = 0; i < xformA.size(); ++i) {
103 result.push_back(xformA[i] * xformB[i]);
104 }
105 }
106 return result;
107}
