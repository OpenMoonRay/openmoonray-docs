21/// Crawl all internal renderOutput AOV buffers and calls one of the user defined function based on the AOV type
22/// for every AOV buffer. This template is useful if you would like to do something for every AOVs inside
23/// renderOutputDriver.
24/// @param funcNonActive execute function if AOV buffer is non active
25/// @param funcVisVariance execute function if AOV buffer is Visibility Variance type
26/// @param funcVariance execute function if AOV buffer is Variance type
27/// @param funcVisibility execute function if AOV buffer is Visibility type
28/// @param funcAOV execute function if AOV buffer is regular AOV type (and non of above)
29template <typename FUNC_NONACTIVE,
30 typename FUNC_VISVARIANCE,
31 typename FUNC_VARIANCE,
32 typename FUNC_VISIBILITY,
33 typename FUNC_AOV>
34void
35crawlAllRenderOutput(const RenderOutputDriver& outputDriver,
36 FUNC_NONACTIVE funcNonActive,
37 FUNC_VISVARIANCE funcVisVariance,
38 FUNC_VARIANCE funcVariance,
39 FUNC_VISIBILITY funcVisibility,
40 FUNC_AOV funcAOV)
41{
42 for (unsigned int roIdx = 0; roIdx < outputDriver.getNumberOfRenderOutputs(); ++roIdx) {
43 switchAovType(outputDriver, roIdx, funcNonActive, funcVisVariance, funcVariance, funcVisibility, funcAOV);
44 }
45}

47/// Useful template if you need to process something for AOV based on AOV's type.
48/// roIdx is used to specify one of AOV buffer inside RenderOutputDriver and you need to specify
49/// all possible functions to execute based on AOV type.
50/// @param roIdx return of getRenderOutputIndx() this defines current AOV buffer which need to process
51/// @param funcNonActive execute function if this AOV is non active
52/// @param funcVisVariance execute function if this AOV is Visibility Variance type
53/// @param funcVariance execute function if this AOV is Variance type
54/// @param funcVisibility execute function if this AOV is Visibility type
55/// @param funcAOV execute function if this AOV is regular AOV type (and non of above)
56template <typename FUNC_NONACTIVE,
57 typename FUNC_VISVARIANCE,
58 typename FUNC_VARIANCE,
59 typename FUNC_VISIBILITY,
60 typename FUNC_AOV>
61void
62switchAovType(const RenderOutputDriver& outputDriver,
63 const int roIdx,
64 FUNC_NONACTIVE funcNonActive,
65 FUNC_VISVARIANCE funcVisVariance,
66 FUNC_VARIANCE funcVariance,
67 FUNC_VISIBILITY funcVisibility,
68 FUNC_AOV funcAOV)
69//
70// roIdx is return of getRenderOutputIndx().
71//
72{
73 const pbr::AovSchema &schema = outputDriver.getAovSchema();
74
75 const int aovIdx = outputDriver.getAovBuffer(roIdx);
76 if (aovIdx < 0) {
77 // non active AOV
78 funcNonActive(outputDriver.getRenderOutput(roIdx));
79 } else if (schema.isVarianceEntry(aovIdx)) {
80 const int varianceSource = schema.getVarianceToSourceAOV(aovIdx);
81 if (outputDriver.isVisibilityAov(varianceSource)) {
82 // Visibility variance AOV
83 funcVisVariance(aovIdx, varianceSource);
84 } else {
85 // Variance AOV
86 funcVariance(aovIdx);
87 }
88 } else if (outputDriver.isVisibilityAov(roIdx)) {
89 // Visibility AOV
90 funcVisibility(aovIdx);
91 } else {
92 // regular AOV
93 funcAOV(aovIdx);
94 }
95}
