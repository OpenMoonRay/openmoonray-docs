97 //

99 // The shade queue is a shared resource, so threads can insert both primary
100 // and secondary ray hits into any shade queue at anytime. In extreme cases,

102 // in a stack overflow.

192template <typename Body>
193inline void
194forEachShadeQueue(mcrt_common::ThreadLocalState *tls, const Body &body)
195{
196 ShadeQueueList &shadeQueues = Material::getAllShadeQueues();
197 for (auto it = shadeQueues.begin(); it != shadeQueues.end(); ++it) {
198 body(tls, *it);
199 }
200}
