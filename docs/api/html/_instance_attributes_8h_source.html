56InstanceAttributes::InstanceAttributes(
57 PrimitiveAttributeTable&& primitiveAttributeTable)
58{
59 // make sure data in allocated buffer aligned
60 // we sort the key by attribute size, and build key->offset table
61 // with size descending order (to avoid wasted padding space)
62 int maxKey = -1;
63 std::map<size_t, std::vector<AttributeKey>> attrSizeMap;
64 for (const auto& kv : primitiveAttributeTable) {
65 AttributeKey key = kv.first;
66 if (primitiveAttributeTable.getRate(key) == RATE_CONSTANT &&
67 primitiveAttributeTable.getTimeSampleCount(key) == 1) {
68 maxKey = scene_rdl2::math::max(maxKey, (int)key);
69 attrSizeMap[key.getSize()].push_back(key);
70 } else {
71 MNRY_ASSERT(false, "InstanceAttributes only support "
72 "constant rate static primitive attributes now");
73 }
74 }
75 mKeyOffset.resize(maxKey + 1, -1);
76 size_t totalSize = 0;
77 for (auto rit = attrSizeMap.rbegin(); rit != attrSizeMap.rend(); ++rit) {
78 size_t attrSize = rit->first;
79 for (auto k: rit->second) {
80 mKeyOffset[k] = totalSize;
81 totalSize += attrSize;
82 }
83 }
84 mConstants.resize(totalSize);
85 // Fill in data
86 for (const auto& kv : primitiveAttributeTable) {
87 AttributeKey key = kv.first;
88 int offset = mKeyOffset[key];
89 if (mKeyOffset[key] != -1) {
90 const auto& primitiveAttribute = kv.second[0];
91 primitiveAttribute->fetchData(0, &mConstants[offset]);
92 }
93 }
94}
