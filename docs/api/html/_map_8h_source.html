41 const moonray::shading::State& state,
42 math::Color* result) const
43 {
44 MNRY_ASSERT(mSampleFunc != nullptr);
45 mSampleFunc(this, tls, state, result);

48 finline void samplev(moonray::shading::TLState *tls,
49 const rdl2::Statev * statev,
50 rdl2::Colorv* resultv) const
51 {
52 if (mSampleFuncv != nullptr) {
53 mSampleFuncv(this, tls, statev, resultv, util::sAllOnMask);
54 }
55 }
56
57 // Unfortuantely, this member has been made public to allow
58 // for computing its offset into the binary
59 SampleFunc mSampleFunc;
60 SampleFuncv mSampleFuncv;
61 // Save away mSampleFunc when we fatal for future restore
62 SampleFunc mOriginalSampleFunc;
63 SampleFuncv mOriginalSampleFuncv;
64
65 // Not thread safe!
66 virtual void setFataled(bool fataled) {
67 if (fataled) {
68 mOriginalSampleFunc = mSampleFunc;
69 mOriginalSampleFuncv = mSampleFuncv;
70 mSampleFunc = mSceneClass.getSceneContext()->getFatalSampleFunc();
71 mSampleFuncv = nullptr;
72 } else {
73 // If we're no longer fataled and we stored away
74 // a sample func, restore it.
75 if (mOriginalSampleFunc != nullptr) {
76 mSampleFunc = mOriginalSampleFunc;
77 }
78 if (mOriginalSampleFuncv != nullptr) {
79 mSampleFuncv = mOriginalSampleFuncv;
80 }
81 }
82 }
83
84 // Some maps can be used as an extra aov map.

86
87 // If this is a ListMap, this method will return true and
