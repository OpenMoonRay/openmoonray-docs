15// This allocator (which should only be used for testing) keeps track of the
16// size passed in for allocation and deallocation of memory, to make sure it's
17// consistent (most allocators ignore the size on dealloation, making testing
18// difficult).
19template <typename T>
20class SizeVerifyingAllocator
21{
22public:
23 using value_type = T;
24 using MapType = std::map<void*, std::size_t>;
25
26 // We use a shared_ptr because we can construct other types of allocators
27 // from this one, use it, and then discard it, only to do the same thing
28 // later (i.e. rebind). We want to hold all of the pointers from these
29 // various incarnations of the allocator.
30 using MapTypePointer = std::shared_ptr<MapType>;
31
32 using propagate_on_container_copy_assignment = std::true_type;
33 using propagate_on_container_move_assignment = std::true_type;
34 using propagate_on_container_swap = std::true_type;
35 using is_always_equal = std::false_type;
36
37 SizeVerifyingAllocator() :
38 mMap(new MapType)
39 {
40 }
41
42 template <typename U>
43 SizeVerifyingAllocator(const SizeVerifyingAllocator<U>& other) :
44 mMap(other.get_map())
45 {
46 }
47
48 T* allocate(std::size_t n)
49 {
50 T* const p = std::allocator<T>().allocate(n);
51 (*mMap)[p] = n;
52 return p;
53 }
54
55 void deallocate(T *ptr, std::size_t n)
56 {
57 if (ptr) {
58 const auto it = mMap->find(ptr);
59 CPPUNIT_ASSERT(it != mMap->end());
60 CPPUNIT_ASSERT(it->second == n);
61 std::allocator<T>().deallocate(ptr, n);
62 }
63 }
64
65 MapTypePointer get_map() const
66 {
67 return mMap;
68 }
69
70private:
71 MapTypePointer mMap;
72
73 template <typename T1, typename T2>
74 friend bool operator==(const SizeVerifyingAllocator<T1> a, const SizeVerifyingAllocator<T2>&b);
75};
