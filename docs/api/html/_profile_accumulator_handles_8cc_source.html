47// Times passed in are assumed to all be per-thread (aka wallclock) times.
48void
49conditionTimes(std::vector<double> &times)
50{
51 //
52 // This logic is common for both bundled and non-bundled execution.
53 //
54
55 ACC(INT_ACCUM_RENDER_DRIVER_SERIAL) = ACC(ACCUM_ROOT) - ACC(ACCUM_RENDER_DRIVER_PARALLEL);
56
57 double totalExclTime = 0.0;
58 for (unsigned i = 0; i < NUM_EXCLUSIVE_ACC; ++i) {
59 double time = ACC(ExclAccType(i));
60 MNRY_ASSERT_REQUIRE(time >= 0.0);
61 totalExclTime += time;
62 }
63
64 ACC(INT_ACCUM_MISSING_TIME) = std::max(ACC(ACCUM_RENDER_DRIVER_PARALLEL) - totalExclTime, 0.0);
65 ACC(INT_ACCUM_TOTALS) = ACC(ACCUM_ROOT);
66
67 // Add up totals for various categories.
68
69 ACC(INT_ACCUM_TOTAL_EMBREE) = ACC(EXCL_ACCUM_EMBREE_INTERSECTION) +
70 ACC(EXCL_ACCUM_EMBREE_OCCLUSION) +
71 ACC(EXCL_ACCUM_EMBREE_PRESENCE) +
72 ACC(EXCL_ACCUM_EMBREE_VOLUME);
73
74 ACC(INT_ACCUM_TOTAL_INTEGRATION) = ACC(EXCL_ACCUM_INTEGRATION) +
75 ACC(EXCL_ACCUM_SSS_INTEGRATION) +
76 ACC(EXCL_ACCUM_VOL_INTEGRATION) +
77 ACC(EXCL_ACCUM_POST_INTEGRATION);
78
79 ACC(INT_ACCUM_TOTAL_RAY_INTERSECTION) = ACC(INT_ACCUM_TOTAL_EMBREE) +
80 ACC(EXCL_ACCUM_RAY_HANDLER) +
81 ACC(EXCL_ACCUM_OCCL_QUERY_HANDLER) +
82 ACC(EXCL_ACCUM_PRESENCE_QUERY_HANDLER) +
83 ACC(EXCL_ACCUM_RAY_SORT_KEY_GEN);
84
85 ACC(INT_ACCUM_TOTAL_SHADING) = ACC(EXCL_ACCUM_SHADE_HANDLER) +
86 ACC(EXCL_ACCUM_DEFER_SHADE_ENTRIES) +
87 ACC(EXCL_ACCUM_SHADING);
88
89 ACC(INT_ACCUM_TOTAL_SCALAR_TIME) = ACC(INT_ACCUM_TOTALS) - (ACC(INT_ACCUM_TOTAL_EMBREE) + ACC(ACCUM_TOTAL_ISPC));
90}
91
92#endif // #ifdef PROFILE_ACCUMULATORS_ENABLED

226unsigned
227snapshotAccumulators(std::vector<AccumulatorResult> *dstResults,
228 double rcpTickFrequency,
229 double threshold)
230{
231#ifdef PROFILE_ACCUMULATORS_ENABLED
232
233 MNRY_ASSERT(dstResults);
234
235 if (!snapshotRawAccumulators(dstResults, rcpTickFrequency)) {
236 return 0;
237 }
238
239 // Copy time data into each to manipulate array.
240 std::vector<AccumulatorResult> &results = *dstResults;
241 std::vector<double> times;
242 times.resize(results.size());
243 for (size_t i = 0; i < results.size(); ++i) {
244 times[i] = results[i].mTimePerThread;
245 }
246
247 // Process various times recorded into more digestible times.
248 conditionTimes(times);
249
250 // Now convert conditioned times back into AccumulatorResult format.
251 const double threadMultiplier = double(getNumAccumulatorThreads());
252 const double pctMultiplier = 100.0 / ACC(gAccumulatorHandles.mRoot);
253
254 for (size_t i = 0; i < results.size(); ++i) {
255 AccumulatorResult &result = results[i];
256 result.mTotalTime = times[i] * threadMultiplier;
257 result.mTimePerThread = times[i];
258 result.mPercentageOfTotal = times[i] * pctMultiplier;
259 }
260
261
262 // Cull results which fall below the threshold or aren't displayable.
263 auto newEnd = std::remove_if(results.begin(), results.end(), [&](const AccumulatorResult &acc) {
264 return acc.mTotalTime < threshold || ((acc.mFlags & ACCFLAG_DISPLAYABLE) == 0);
265 });
266
267 results.erase(newEnd, results.end());
268
269 // Greater than.
270 std::stable_sort(results.begin(), results.end(), [](const AccumulatorResult &a, const AccumulatorResult &b) -> bool {
271 MNRY_ASSERT(a.mName && b.mName);
272
273 bool bracketA = (a.mName[0] == '[');
274 bool bracketB = (b.mName[0] == '[');
275
276 if (bracketA) {
277 if (bracketB) {
278 return a.mTimePerThread > b.mTimePerThread;
279 } else {
280 return false;
281 }
282 } else {
283 if (bracketB) {
284 return true;
285 } else {
286 // If an entry is marked with the TOTAL flag, then sort it
287 // at the end of the exclusive times, but before any bracketed
288 // times.
289 if (a.mFlags & ACCFLAG_TOTAL) return false;
290 if (b.mFlags & ACCFLAG_TOTAL) return true;
291 return a.mTimePerThread > b.mTimePerThread;
292 }
293 }
294 });
295
296 return (unsigned)results.size();
297
298#else
299
300 return 0;
301
302#endif
303}
