19// Light Stage X class, which provides light order for One Light At a Time (OLAT)
20// data. It also provides light coordinates in "light-stage" space: (o, x, y, z):
21// o: physical center of the light-stage
22// x: points right, y: points up, z: points towards the camera
23class LightStage {
24public:
25 // Singleton
26 finline static const LightStage &singleton() {
27 static const LightStage lightStage;
28 return lightStage;
29 }
30
31 // Return a light index from the light order index from the spiraling OLAT
32 // pattern for separated cross / parallel polarized sequence. The given
33 // lightOrderIndex should be in the range [0..sSeparatedLightCount-1]
34 static const int sSeparatedLightCount = 85;
35 int getSeparatedLightIndex(int lightOrderIndex) const;
36
37 // Return a light index from the light order index from the spiraling OLAT
38 // pattern for combined (non-polarized) sequence. The given
39 // lightOrderIndex should be in the range [0..sCombinedLightCount-1]
40 static const int sCombinedLightCount = 201;
41 int getCombinedLightIndex(int lightOrderIndex) const;
42
43 // Get the position of the light
44 scene_rdl2::math::Vec3f getLightPosition(int index) const;
45
46private:
47 // Singleton
48 LightStage();
49 ~LightStage() {}
50
51 // Copy is disabled
52 LightStage(const LightStage &other);
53 const LightStage &operator=(const LightStage &other);
54
55 void printLightStageLightsInOrder();
56};
