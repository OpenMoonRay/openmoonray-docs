420 vertices(4) = Vec3fa( 2.0f, 2.0f / 3.0f * sqrt3, 2.0f, 0.f);

422 vertices(6) = Vec3fa( 0.0f,-4.0f / 3.0f * sqrt3, -2.0f, 0.f);
423 vertices(7) = Vec3fa( 0.0f,-4.0f / 3.0f * sqrt3, 2.0f, 0.f);

425 indices = {

428 3, 5, 4,
429 0, 2, 6,
430 2, 4, 7, 6,
431 4, 5, 7,

437void TestRenderingPrimAttr::testVertex1()
438{
439 scene_rdl2::rdl2::SceneContext ctx;
440 ctx.setDsoPath(ctx.getDsoPath() +
441 ":dso/geometry/TestGeometry:dso/material/TestMaterial");
442 scene_rdl2::rdl2::Geometry* geom =
443 ctx.createSceneObject("TestGeometry", "geom")->asA<scene_rdl2::rdl2::Geometry>();
444 scene_rdl2::rdl2::Layer* layer =
445 ctx.createSceneObject("Layer", "/seq/shot/layer")->asA<scene_rdl2::rdl2::Layer>();
446 scene_rdl2::rdl2::Material* mtl =
447 ctx.createSceneObject("TestMaterial", "mtl")->asA<scene_rdl2::rdl2::Material>();
448 scene_rdl2::rdl2::LightSet* lgt =
449 ctx.createSceneObject("LightSet", "lgt")->asA<scene_rdl2::rdl2::LightSet>();
450
451 layer->beginUpdate();
452 // LayerAssignmentID(0) below
453 layer->assign(geom, "empty", mtl, lgt, nullptr, nullptr);
454 layer->endUpdate();
455
456 // Testing subdivision mesh vertex rate tessellation
457 SubdivisionMesh::FaceVertexCount faceVertexCount;
458 SubdivisionMesh::VertexBuffer vertices;
459 SubdivisionMesh::IndexBuffer indices;
460 fillTestSubdMeshData(faceVertexCount, vertices, indices);
461
462 // insert a Vec3 primitive atribute that is a duplication of position data
463 PrimitiveAttributeTable table;
464 std::vector<Vec3f> pRefAttr1(vertices.size());
465 std::vector<Vec3f> pRefAttr2(vertices.size());
466 std::vector<Vec3f> pRefAttr3(vertices.size());
467 for (size_t i = 0; i < vertices.size(); ++i) {
468 pRefAttr1[i] = vertices(i);
469 pRefAttr2[i] = vertices(i);
470 pRefAttr3[i] = vertices(i);
471 }
472 TypedAttributeKey<Vec3f> pRef1Key("pRef1");
473 TypedAttributeKey<Vec3f> pRef2Key("pRef2");
474 TypedAttributeKey<Vec3f> pRef3Key("pRef3");
475 table.addAttribute(pRef1Key, RATE_VERTEX, std::move(pRefAttr1));
476 table.addAttribute(pRef2Key, RATE_VERTEX, std::move(pRefAttr2));
477 table.addAttribute(pRef3Key, RATE_VERTEX, std::move(pRefAttr3));
478
479 std::unique_ptr<internal::OpenSubdivMesh> subdMesh =
480 fauxstd::make_unique<internal::OpenSubdivMesh>(
481 SubdivisionMesh::Scheme::CATMULL_CLARK,
482 std::move(faceVertexCount), std::move(indices), std::move(vertices),
483 LayerAssignmentId(0), std::move(table));
484 const scene_rdl2::math::Mat4d world2render;
485 subdMesh->setMeshResolution(5);
486 subdMesh->tessellate({layer, {}, world2render, false, false, false, nullptr});
487
488 internal::Mesh::TessellatedMesh tessellatedMesh;
489 subdMesh->getTessellatedMesh(tessellatedMesh);
490 size_t mbSteps = tessellatedMesh.mVertexBufferDesc.size();
491 CPPUNIT_ASSERT(mbSteps == 1);
492 const float* vertexBuffer = reinterpret_cast<const float*>(
493 tessellatedMesh.mVertexBufferDesc[0].mData);
494 size_t offset = tessellatedMesh.mVertexBufferDesc[0].mOffset;
495 size_t stride = tessellatedMesh.mVertexBufferDesc[0].mStride /
496 sizeof(float);
497 for (size_t v = 0; v < tessellatedMesh.mVertexCount; ++v) {
498 const Vec3f& p = Vec3f(&vertexBuffer[offset + v * stride]);
499 const Vec3f& pRef1 = subdMesh->getAttributes()->getVertex(pRef1Key, v);
500 const Vec3f& pRef2 = subdMesh->getAttributes()->getVertex(pRef2Key, v);
501 const Vec3f& pRef3 = subdMesh->getAttributes()->getVertex(pRef3Key, v);
502 CPPUNIT_ASSERT(scene_rdl2::math::isEqual(p, pRef1));
503 CPPUNIT_ASSERT(scene_rdl2::math::isEqual(p, pRef2));

652void TestRenderingPrimAttr::testTransformAttributes0()
653{
654 // constant and uniform testing
655 PrimitiveAttributeTable table;
656 size_t numFaces= 3;
657 Vec3f pRoot = mRNG.randomVec3f();
658 Vec3f tRoot = mRNG.randomVec3f();
659 std::vector<Vec3f> nFace(numFaces);
660 std::vector<Vec3f> noTransform(numFaces);
661 for (size_t f = 0; f < numFaces; ++f) {
662 nFace[f] = mRNG.randomVec3f();
663 noTransform[f] = mRNG.randomVec3f();
664 }
665 TypedAttributeKey<Vec3f> pKey = TypedAttributeKey<Vec3f>("position");
666 TypedAttributeKey<Vec3f> tKey = TypedAttributeKey<Vec3f>("tangent");
667 TypedAttributeKey<Vec3f> nKey = TypedAttributeKey<Vec3f>("normal");
668 TypedAttributeKey<Vec3f> noTransformKey =
669 TypedAttributeKey<Vec3f>("no_transform");
670 table.addAttribute(pKey, RATE_CONSTANT, {pRoot});
671 table.addAttribute(tKey, RATE_CONSTANT, {tRoot});
672 table.addAttribute(nKey, RATE_UNIFORM, std::vector<Vec3f>{nFace});
673 table.addAttribute(noTransformKey, RATE_UNIFORM,
674 std::vector<Vec3f>{noTransform});
675
676 std::unique_ptr<Attributes> attr(
677 Attributes::interleave(table, 0, numFaces, 0,
678 std::vector<size_t>(), 0));
679
680 Vec3f rotateAxis(normalize(mRNG.randomVec3f()));
681 scene_rdl2::math::Xform3f xform =
682 scene_rdl2::math::Xform3f::translate(mRNG.randomVec3f()) *
683 scene_rdl2::math::Xform3f::rotate(rotateAxis, mRNG.randomFloat()) *
684 scene_rdl2::math::Xform3f::scale(mRNG.randomVec3f());
685
686 attr->transformAttributes({xform}, 0.0f, 0.0f,
687 {{pKey, Vec3Type::POINT},
688 {tKey, Vec3Type::VECTOR},
689 {nKey, Vec3Type::NORMAL}});
690
691 // Originally, this line called transformPoint(). The result should be the same as the one generated by
692 // transformAttributes() just above, because that's calling the same function, transformPoint(). However, that
693 // function is inlined, and generates different code in the 2 places, leading to slightly different rounding
694 // that causes a 1-ulp difference in this unit test. So here we get around that by having it call out to a
695 // function in another module, which simply passes through to transformPoint(). That stops it being inlined here,
696 // and brings the results back to being identical.
697 CPPUNIT_ASSERT(verifyConstant(attr, pKey, transformPointUtil(xform, pRoot)));
698
699 CPPUNIT_ASSERT(verifyConstant(attr, tKey, transformVector(xform, tRoot)));
700 for (size_t f = 0; f < numFaces; ++f) {
701 CPPUNIT_ASSERT(verifyUniform(attr, nKey, f,
702 transformNormal(xform.inverse(), nFace[f])));
703 CPPUNIT_ASSERT(verifyUniform(attr, noTransformKey, f,
704 noTransform[f]));
705 }
