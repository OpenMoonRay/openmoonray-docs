164///
165/// All RaaS related thread local data should go in here. This object is passed
166/// around as an argument to many functions throughout the system. Any data put
167/// in here doesn't need to be locked when written to or read from, and
168/// operations don't need to be atomic (e.g. no need for tbb::atomics).
169///
170class CACHE_ALIGN ThreadLocalState
171{
172public:
173 ThreadLocalState(uint32_t threadIdx, bool allocBundledResources);
174
175 ~ThreadLocalState();
176
177 // Call between frames to reset this TLState.
178 void reset();
179
180 bool checkForHandlerStackOverflowRisk();
181
182 // Always keep this as the very first element in this struct so we can
183 // easily convert from the pointer to the thread index.
184 uint32_t mPad; // Pad so that mThreadIdx is at the same
185 // offset in ThreadLocalState and BaseTLState.
186 uint32_t mHandlerStackDepth; // Used to prevent stack overflows
187 // in vectorized mode.
188 const uint32_t mThreadIdx; // Zero based incrementing counter
189
190 // Primary thread local memory arena.
191 scene_rdl2::alloc::Arena mArena;
192
193 // Arena for pixel data. Persists for the rendering of an entire pixel.
194 // This is only true for batch mode and is not useful for progressive and
195 // realtime modes as they render pixels with multiple passes.
196 scene_rdl2::alloc::Arena mPixelArena;
197
198 // Memory deallocation reponsibilities are given to shared_ptr. It is a
199 // good fit since internally it does some magic so that it doesn't need to
200 // be able to see the concrete destructor of an object in order to be able
201 // to delete it.
202 std::shared_ptr<geom::internal::TLState> mGeomTls;
203 std::shared_ptr<pbr::TLState> mPbrTls;
204 std::shared_ptr<shading::TLState> mShadingTls;
205
206 DISALLOW_COPY_OR_ASSIGNMENT(ThreadLocalState);
207};

240/// Returns the list of render thread ThreadLocalState objects. This is useful

251// This is the number of tbb threads, plus one for the gui thread, plus
252// our overflow pool size. All thread indices will be one less than this

258/// This function exists for cases where we need to perform some raytracing operation
259/// on a GUI thread outside of our tbb threadpool. One example we have currently

261/// The returned TLS can be used whilst rendering without side effects.

286//
287// Convenience function for iterating over existing top level render TLS
288// instances. This doesn't include the GUI TLS.
289//
290// Example of syntax:
291//
292// forEachTLS([&](ThreadLocalState *tls)
293// {
294// tls->doWork();
295// });
296//
297template <typename Body>
298finline void
299forEachTLS(const Body &body)
300{
301 unsigned numTLS = getNumTBBThreads();
302 ThreadLocalState *tlsList = getTLSList();
303 for (unsigned i = 0; i < numTLS; ++i) {
304 body(tlsList + i);
305 }
306}
