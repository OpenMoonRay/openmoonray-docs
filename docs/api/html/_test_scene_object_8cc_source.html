407void
408TestSceneObject::testResetToDefault()
409{
410 SceneObject* obj = mDsoClass->createObject("/seq/shot/pizza");
411
412 // Set some values and verify them.
413 obj->beginUpdate();
414 obj->set(mIntKey, Int(9001));
415 obj->set(mStringKey, String("hello"));
416 obj->set(mLongKey, Long(9002));
417 obj->endUpdate();
418
419 CPPUNIT_ASSERT(obj->get(mIntKey) == Int(9001));
420 CPPUNIT_ASSERT(obj->get(mStringKey) == String("hello"));
421 CPPUNIT_ASSERT(obj->get(mLongKey) == Long(9002));
422
423 // Reset 2 of the 3 to the default.
424 obj->beginUpdate();
425 obj->resetToDefault(mIntKey);
426 obj->resetToDefault("string");
427 obj->endUpdate();
428
429 CPPUNIT_ASSERT(obj->get(mIntKey) == Int(100));
430 CPPUNIT_ASSERT(obj->get(mStringKey) == String("wat"));
431 CPPUNIT_ASSERT(obj->get(mLongKey) == Long(9002));
432
433 // Reset to default must be during an update.
434 CPPUNIT_ASSERT_THROW(obj->resetToDefault(mIntKey), except::RuntimeError);
435 mDsoClass->destroyObject(obj);
436}
