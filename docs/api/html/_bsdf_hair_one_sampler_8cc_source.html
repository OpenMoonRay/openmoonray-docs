24Color
25HairOneSampleLobe::eval(const BsdfSlice &slice,
26 const Vec3f &wi,
27 float *pdf) const
28{
29 // calculate all the relevant hair bsdf parameters
30 HairState hairState(slice.getWo(),
31 wi,
32 mHairDir,
33 mH,
34 mIOR,
35 mSigmaA,
36 mHairRotation * mHairUV.x, // use hair s coord to vary rotation from base to tip
37 mHairNormal);
38
39 if (pdf) *pdf = evalPdf(hairState);
40
41 return getScale() * evalBsdf(hairState);
42}

44Color
45HairOneSampleLobe::evalBsdf(const HairState& hairState,
46 bool includeCosineTerm) const
47{
48 Color bsdf = scene_rdl2::math::sBlack;
49
50 const Color T = hairState.absorptionTerm();
51
52 const Color fresnel = evalHairFresnel(hairState,
53 hairState.cosThetaD());
54
55 const Color oneMinusFresnel = scene_rdl2::math::sWhite - fresnel;
56
57 if (mShowR) {
58 const Color rFresnel = fresnel;
59 bsdf += rFresnel *
60 rLobe.getTint() *
61 rLobe.evalMTerm(hairState) *
62 rLobe.evalNTermWithAbsorption(hairState);
63 }
64
65 if (mShowTT) {
66 const Color ttFresnel = oneMinusFresnel * oneMinusFresnel;
67 bsdf += ttFresnel *
68 ttLobe.getTint() *
69 ttLobe.evalMTerm(hairState) *
70 ttLobe.evalNTermWithAbsorption(hairState);
71 }
72
73 if (mShowTRT) {
74 const Color trtFresnel = fresnel * oneMinusFresnel * oneMinusFresnel;
75 bsdf += trtFresnel *
76 trtLobe.getTint() *
77 trtLobe.evalMTerm(hairState) *
78 trtLobe.evalNTermWithAbsorption(hairState);
79 }
80
81 if (mShowTRRT) {
82 const Color trrt = trrtLobe.compensationFactor(fresnel, oneMinusFresnel, T);
83 bsdf += trrt *
84 trrtLobe.getTint() *
85 trrtLobe.evalMTerm(hairState) *
86 trrtLobe.evalNTermWithAbsorption(hairState);
87 }
88
89 return bsdf;
90}

92//-----------------evaluate phi pdf ------------------------------------
93float
94HairOneSampleLobe::evalPdf(const HairState& hairState) const
95{
96#if PBR_HAIR_USE_UNIFORM_SAMPLING
97 return 1.0f / sFourPi;
98#else
99 // weights and cdfs for R, TRT, TT lobes respectivel
100 float weights[3];
101 float cdf[3];
102 calculateSamplingWeightsAndCDF(hairState,
103 weights,
104 cdf);
105 if (scene_rdl2::math::isZero(cdf[2])) {
106 // uniform sampling
107 // oneOverFourPi
108 return (0.5f * sOneOverTwoPi);
109 }
110
111 return evalPdf(hairState, weights);
112#endif
113}

115float
116HairOneSampleLobe::evalPdf(const HairState& hairState,
117 const float (&weights)[3]) const
118{
119#if PBR_HAIR_USE_UNIFORM_SAMPLING
120 return 1.0f / sFourPi;
121#else
122 float rProb = 0.0f, ttProb = 0.0f, trtProb = 0.0f;
123 if (mShowR)
124 rProb = weights[0] * rLobe.evalThetaPdf(hairState) * rLobe.evalPhiPdf(hairState);
125 if (mShowTRT)
126 trtProb = weights[1] * trtLobe.evalThetaPdf(hairState) * trtLobe.evalPhiPdf(hairState);
127 if (mShowTT)
128 ttProb = weights[2] * ttLobe.evalThetaPdf(hairState) * ttLobe.evalPhiPdf(hairState);
129
130 return max(0.0f, (rProb + trtProb + ttProb));
131#endif
132}

135Color
136HairOneSampleLobe::sample(const BsdfSlice &slice,
137 float r1, float r2,
138 Vec3f &wi,
139 float &pdf) const
140{
141 // If none of the lobes are ON
142 if (!mShowR && !mShowTT && !mShowTRT && !mShowTRRT) {
143 pdf = 0.0f;
144 return sBlack;
145 }
146
147 // calculate all the relevant hair bsdf parameters based on omegaO
148 HairState hairState(slice.getWo(),
149 mHairDir,
150 mH,
151 mIOR,
152 mSigmaA,
153 mHairRotation * mHairUV.x, // use hair s coord to vary rotation from base to tip
154 mHairNormal);
155#if PBR_HAIR_USE_UNIFORM_SAMPLING
156 wi = hairState.localToGlobal(sampleSphereUniform(r1, r2));
157 return eval(slice, wi, &pdf);
158#else
159 float phiI, thetaI;
160
161 const HairBsdfLobe* sampleLobe;
162
163 // weights and cdfs for R, TRT, TT lobes respectivel
164 float weights[3];
165 float cdf[3];
166 calculateSamplingWeightsAndCDF(hairState,
167 weights,
168 cdf);
169
170 if (scene_rdl2::math::isZero(cdf[2])) {
171 // Uniform Sample TRRT
172 wi = sampleSphereUniform(r1, r2);
173 wi = hairState.localToGlobal(wi);
174 return eval(slice, wi, &pdf);
175 }
176
177 // Pick a Lobe to Sample From
178 if (r1 < cdf[0]) {
179 r1 = r1 / weights[0];
180 sampleLobe = &rLobe;
181 } else if (r1 < cdf[1]) {
182 r1 = (r1 - cdf[0]) / weights[1];
183 sampleLobe = &trtLobe;
184 } else {
185 // Clamp to ensure [0,1)
186 r1 = clamp((r1 - cdf[1]) / weights[2], 0.0f, 0.99f);
187 sampleLobe = &ttLobe;
188 }
189
190 sampleLobe->sampleTheta(r1,
191 hairState.thetaO(),
192 thetaI);
193 sampleLobe->samplePhi(r2,
194 hairState.phiO(),
195 phiI); // phiI is in (-pi, pi]
196 MNRY_ASSERT(phiI >= -sPi && phiI <= sPi);
197
198 float sinPhi, cosPhi;
199 sincos(phiI, &sinPhi, &cosPhi);
200 float sinTheta, cosTheta;
201 sincos(thetaI, &sinTheta, &cosTheta);
202
203 // Compute the light direction vector for shading.
204 const float uWgt = sinTheta;
205 const float vWgt = cosTheta * cosPhi;
206 const float wWgt = cosTheta * sinPhi;
207 wi = hairState.localToGlobal(uWgt, vWgt, wWgt);
208
209 hairState.updateAngles(wi,
210 phiI,
211 thetaI);
212
213 pdf = evalPdf(hairState, weights);
214
215 return getScale() * evalBsdf(hairState);
216#endif
217}

219void
220HairOneSampleLobe::calculateSamplingWeightsAndCDF(const HairState& hairState,
221 float (&weights)[3],
222 float (&cdf)[3]) const
223{
224 // We use cosThetaO as an approximation to calculate Fresnel here since
225 // in the sampling stage, we only have access to omega_o. Even though the
226 // fresnel in eval() is computed correctly using cosThetaD (which is equivivalent to cosThetaH in
227 // hair frame), cosThetaO serves as a decent approximation to select a lobe to sample.
228 const Color fresnel = evalHairFresnel(hairState,
229 hairState.cosThetaO());
230
231 const Color oneMinusFresnel = scene_rdl2::math::sWhite - fresnel;
232
233 float sum = 0.0f;
234 weights[0] = weights[1] = weights[2] = 0.0f;
235 if (mShowR) {
236 // clamping the R-weight here reduces fireflies for lighter hair
237 // as r term doesn't get unfairly voted high for the cosThetaO approx above.
238 weights[0] = max(0.2f, scene_rdl2::math::luminance(fresnel * rLobe.absorption(hairState)));
239 sum += weights[0];
240 }
241 if (mShowTRT) {
242 weights[1] = scene_rdl2::math::luminance(fresnel * oneMinusFresnel * oneMinusFresnel * trtLobe.absorption(hairState));
243 sum += weights[1];
244 }
245 if (mShowTT) {
246 weights[2] = scene_rdl2::math::luminance(oneMinusFresnel* oneMinusFresnel * ttLobe.absorption(hairState));
247 sum += weights[2];
248 }
249
250 // Normalize the Weights
251 if (!scene_rdl2::math::isZero(sum)) {
252 for (int i = 0; i < 3; ++i) weights[i] /= sum;
253 }
254 cdf[0] = weights[0];
255 cdf[1] = cdf[0] + weights[1];
256 cdf[2] = cdf[1] + weights[2];
257}
