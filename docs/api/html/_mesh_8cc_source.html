23openvdb::VectorGrid::Ptr
24Mesh::createTriMeshVelocityGrid(float interiorBandwidth,
25 const TessellatedMesh& mesh,
26 size_t motionSample,
27 float invFps)
28{
29 // Convert this mesh to a vdb grid. Inputs are the mesh, the world to index space transform, the exterior bandwidth
30 // -- which is aribitrarily set to 3 because that is the default setting in openvdb --, the interior bandwith --
31 // which is the number of voxels wide / 2, and flags.
32
33 const size_t faceVertexCount = 3;
34 const size_t faceCount = mesh.mFaceCount;
35
36 std::vector<openvdb::Vec3s> vertices(mesh.mVertexCount);
37 std::vector<openvdb::Vec3I> indices(faceCount);
38
39 // Grab vertex buffer of this motion sample.
40 const float* vertexBuffer = reinterpret_cast<const float *>(mesh.mVertexBufferDesc[motionSample].mData) +
41 mesh.mVertexBufferDesc[motionSample].mOffset / sizeof(float);
42 const int* indexBuffer = (const int* )mesh.mIndexBufferDesc.mData;
43
44 // Create openvdb index buffer
45 for (size_t f = 0; f < faceCount; ++f) {
46 indices[f] = openvdb::Vec3I(indexBuffer[f * faceVertexCount],
47 indexBuffer[f * faceVertexCount + 1],
48 indexBuffer[f * faceVertexCount + 2]);
49 }
50
51 // Create openvdb vertex buffer
52 size_t vert = 0;
53 for(size_t v = 0; v < mesh.mVertexCount; ++v) {
54 vertices[v] =
55 openvdb::Vec3s(vertexBuffer[vert], vertexBuffer[vert+1], vertexBuffer[vert+2]);
56 vert += mesh.mVertexBufferDesc[motionSample].mStride / sizeof(float);
57 }
58
59 // Set up openvdb mesh.
60 openvdb::tools::QuadAndTriangleDataAdapter<openvdb::Vec3s, openvdb::Vec3I> meshAdapter(vertices, indices);
61 // The primitiveIndexGrid stores the index of the primitive (face) that is closest to the voxel.
62 openvdb::Int32Grid::Ptr primitiveIndexGrid(new openvdb::Int32Grid(0));
63 openvdb::math::Transform::Ptr transform = openvdb::math::Transform::createLinearTransform();
64 primitiveIndexGrid->setTransform(transform);
65 openvdb::tools::meshToVolume<openvdb::FloatGrid>(meshAdapter,
66 *transform,
67 3.0f,
68 interiorBandwidth,
69 0,
70 primitiveIndexGrid.get());
71
72 openvdb::VectorGrid::Ptr velocityGrid = openvdb::Vec3SGrid::create();
73
74 // If a shared primitive, the velocity grid is baked in local space.
75 const scene_rdl2::math::Mat4f primToRender = getIsReference() ? scene_rdl2::math::Mat4f(scene_rdl2::math::one) : mPrimToRender;
76
77 // Fill in velocity grid
78 openvdb::tools::transformValues(primitiveIndexGrid->cbeginValueOn(), *velocityGrid,
79 [&](const openvdb::Int32Grid::ValueOnCIter& it, openvdb::Vec3SGrid::Accessor& accessor) {
80
81 int faceid = it.getValue();
82
83 const openvdb::math::Coord& coord = it.getCoord();
84 const openvdb::Vec3d voxelCenter(coord.x(), coord.y(), coord.z());
85
86 const openvdb::Vec3d p0(vertices[indices[faceid].x()]);
87 const openvdb::Vec3d p1(vertices[indices[faceid].y()]);
88 const openvdb::Vec3d p2(vertices[indices[faceid].z()]);
89 openvdb::Vec3d uvw;
90 openvdb::math::closestPointOnTriangleToPoint(p0, p1, p2, voxelCenter, uvw);
91
92 // Assume only one velocity buffer at t = 0.
93 const scene_rdl2::math::Vec3f v0 = scene_rdl2::math::transformVector(primToRender, getVelocity(faceid, 0, float(motionSample)));
94 const scene_rdl2::math::Vec3f v1 = scene_rdl2::math::transformVector(primToRender, getVelocity(faceid, 1, float(motionSample)));
95 const scene_rdl2::math::Vec3f v2 = scene_rdl2::math::transformVector(primToRender, getVelocity(faceid, 2, float(motionSample)));
96
97 const scene_rdl2::math::Vec3f v = invFps * (v0 * float(uvw[0]) + v1 * float(uvw[1]) + v2 * float(uvw[2]));
98 accessor.setValue(coord, openvdb::Vec3d(v.x, v.y, v.z));
99
100 });
101 velocityGrid->pruneGrid();
102
103 return velocityGrid;
104}

106openvdb::VectorGrid::Ptr
107Mesh::createQuadMeshVelocityGrid(float interiorBandwidth,
108 const TessellatedMesh& mesh,
109 size_t motionSample,
110 float invFps)
111{
112 // Convert this mesh to a vdb grid. Inputs are the mesh, the world to index space transform, the exterior bandwidth
113 // -- which is aribitrarily set to 3 because that is the default setting in openvdb --, the interior bandwith --
114 // which is the number of voxels wide / 2, and flags.
115
116 const size_t faceVertexCount = 4;
117 const size_t faceCount = mesh.mFaceCount;
118
119 std::vector<openvdb::Vec3s> vertices(mesh.mVertexCount);
120 std::vector<openvdb::Vec4I> indices(faceCount);
121
122 // Grab vertex buffer of this motion sample.
123 const float* vertexBuffer = reinterpret_cast<const float *>(mesh.mVertexBufferDesc[motionSample].mData) +
124 mesh.mVertexBufferDesc[motionSample].mOffset / sizeof(float);
125 const int* indexBuffer = (const int* )mesh.mIndexBufferDesc.mData;
126
127 // Create openvdb index buffer
128 for (size_t f = 0; f < faceCount; ++f) {
129 indices[f] = openvdb::Vec4I(indexBuffer[f * faceVertexCount],
130 indexBuffer[f * faceVertexCount + 1],
131 indexBuffer[f * faceVertexCount + 2],
132 indexBuffer[f * faceVertexCount + 3]);
133 }
134
135 // Create openvdb vertex buffer
136 size_t vert = 0;
137 for(size_t v = 0; v < mesh.mVertexCount; ++v) {
138 vertices[v] =
139 openvdb::Vec3s(vertexBuffer[vert], vertexBuffer[vert+1], vertexBuffer[vert+2]);
140 vert += mesh.mVertexBufferDesc[motionSample].mStride / sizeof(float);
141 }
142
143 // Set up openvdb mesh.
144 openvdb::tools::QuadAndTriangleDataAdapter<openvdb::Vec3s, openvdb::Vec4I> meshAdapter(vertices, indices);
145 // The primitiveIndexGrid stores the index of the primitive (face) that is closest to the voxel.
146 openvdb::Int32Grid::Ptr primitiveIndexGrid(new openvdb::Int32Grid(0));
147 openvdb::math::Transform::Ptr transform = openvdb::math::Transform::createLinearTransform();
148 primitiveIndexGrid->setTransform(transform);
149 openvdb::tools::meshToVolume<openvdb::FloatGrid>(meshAdapter,
150 *transform,
151 3.f,
152 interiorBandwidth,
153 0,
154 primitiveIndexGrid.get());
155
156 openvdb::VectorGrid::Ptr velocityGrid = openvdb::Vec3SGrid::create();
157
158 // If a shared primitive, the velocity grid is baked in local space.
159 // In that case mPrimToRender is the worldToRender matrix used for shading,
160 // which we do not want to apply.
161 const scene_rdl2::math::Mat4f primToRender = getIsReference() ? scene_rdl2::math::Mat4f(scene_rdl2::math::one) : mPrimToRender;
162
163 // Fill in velocity grid
164 openvdb::tools::transformValues(primitiveIndexGrid->cbeginValueOn(), *velocityGrid,
165 [&](const openvdb::Int32Grid::ValueOnCIter& it, openvdb::Vec3SGrid::Accessor& accessor) {
166
167 int faceid = it.getValue();
168
169 const openvdb::math::Coord& coord = it.getCoord();
170 const openvdb::Vec3d voxelCenter(coord.x(), coord.y(), coord.z());
171
172 // first triangle
173 const openvdb::Vec3d p0(vertices[indices[faceid].x()]);
174 const openvdb::Vec3d p1(vertices[indices[faceid].y()]);
175 const openvdb::Vec3d p2(vertices[indices[faceid].z()]);
176 openvdb::Vec3d uvw;
177 const openvdb::Vec3d closestP1 =
178 openvdb::math::closestPointOnTriangleToPoint(p0, p1, p2, voxelCenter, uvw);
179 float dist1 = (voxelCenter - closestP1).lengthSqr();
180 int index0 = 0;
181 int index1 = 1;
182 int index2 = 2;
183
184 // second triangle
185 const openvdb::Vec3d p3(vertices[indices[faceid].w()]);
186 openvdb::Vec3d uvw2;
187 const openvdb::Vec3d closestP2 =
188 openvdb::math::closestPointOnTriangleToPoint(p0, p3, p2, voxelCenter, uvw2);
189 float dist2 = (voxelCenter - closestP2).lengthSqr();
190
191 if (dist2 < dist1) {
192 // use second triangle
193 index1 = 3;
194 uvw = uvw2;
195 }
196
197 // Assume only one velocity buffer at t = 0.
198 const scene_rdl2::math::Vec3f v0 = scene_rdl2::math::transformVector(primToRender, getVelocity(faceid, index0,
199 float(motionSample)));
200 const scene_rdl2::math::Vec3f v1 = scene_rdl2::math::transformVector(primToRender, getVelocity(faceid, index1,
201 float(motionSample)));
202 const scene_rdl2::math::Vec3f v2 = scene_rdl2::math::transformVector(primToRender, getVelocity(faceid, index2,
203 float(motionSample)));
204
205 const scene_rdl2::math::Vec3f v = invFps * (v0 * float(uvw[0]) + v1 * float(uvw[1]) + v2 * float(uvw[2]));
206 accessor.setValue(coord, openvdb::Vec3d(v.x, v.y, v.z));
207 });
208 velocityGrid->pruneGrid();
209
210 return velocityGrid;
211}

213void
214Mesh::createVelocityGrid(const float interiorBandwidth,
215 const MotionBlurParams& motionBlurParams,
216 const std::vector<int>& volumeIds)
217{
218 if(!getAttributes()->isSupported(shading::StandardAttributes::sVelocity)) {
219 // don't bake velocity grid if velocity does not exist
220 return;
221 }
222
223 size_t threadCount = mcrt_common::getMaxNumTLS();
224
225 TessellatedMesh mesh;
226 getTessellatedMesh(mesh);
227
228 // Meshes that are bound to volume shaders and exhibit motion blur should have 2 motion samples. If this is the
229 // case we must bake a velocity grid whose bounds contain the full motion of the mesh. We do this by baking
230 // two velocity grids, one for each motion sample, and then compositing them by taking the max value of
231 // the grids.
232 size_t motionSampleCount = mesh.mVertexBufferDesc.size();
233
234 // The input velcity on the mesh is in units of ditsance / second. We must divide out the seconds to make
235 // it in units of distance / frame.
236 float invFps = motionBlurParams.getInvFps();
237
238 openvdb::VectorGrid::Ptr velocityGrid;
239 if (mesh.mIndexBufferType == MeshIndexType::TRIANGLE) {
240 // first motion sample
241 velocityGrid = createTriMeshVelocityGrid(interiorBandwidth, mesh, 0, invFps);
242 if (motionSampleCount > 1)
243 {
244 // For volumes, there should only be two motion samples because motion blur is created
245 // by velocity.
246 MNRY_ASSERT(motionSampleCount == 2);
247 // second motion sample
248 openvdb::VectorGrid::Ptr secondVelocityGrid = createTriMeshVelocityGrid(interiorBandwidth, mesh,
249 motionSampleCount - 1, invFps);
250 // combine velocity grids by taking max value.
251 openvdb::tools::compMax(*velocityGrid, *secondVelocityGrid);
252 }
253 } else {
254 // first motion sample
255 velocityGrid = createQuadMeshVelocityGrid(interiorBandwidth, mesh, 0, invFps);
256 if (motionSampleCount > 1)
257 {
258 // For volumes, there should only be two motion samples because motion blur is created
259 // by velocity.
260 MNRY_ASSERT(motionSampleCount == 2);
261 // second motion sample
262 openvdb::VectorGrid::Ptr secondVelocityGrid = createQuadMeshVelocityGrid(interiorBandwidth, mesh,
263 motionSampleCount - 1, invFps);
264 // combine velocity grids by taking max value.
265 openvdb::tools::compMax(*velocityGrid, *secondVelocityGrid);
266 }
267 }
268
269 mVdbVelocity->setShutterValues(motionBlurParams.getShutterOpen(),
270 motionBlurParams.getShutterClose() - motionBlurParams.getShutterOpen());
271 mVdbVelocity->setVelocityGrid(velocityGrid, volumeIds);
272}

301void
302Mesh::reverseOrientation(const size_t faceVertexCount,
303 std::vector<uint32_t>& indices,
304 std::unique_ptr<shading::Attributes>& attributes)
305{
306 size_t faceCount = indices.size() / faceVertexCount;
307 size_t indexOffset = 0;
308 for (size_t i = 0; i < faceCount; i++) {
309 std::reverse(indices.begin() + indexOffset,
310 indices.begin() + indexOffset + faceVertexCount);
311 indexOffset += faceVertexCount;
312 }
313
314 attributes->reverseFaceVaryingAttributes();
315}

317void
318Mesh::reverseOrientation(const std::vector<uint32_t>& faceVertexCount,
319 std::vector<uint32_t>& indices,
320 std::unique_ptr<shading::Attributes>& attributes)
321{
322 size_t faceCount = faceVertexCount.size();
323 size_t indexOffset = 0;
324 for (size_t i = 0; i < faceCount; i++) {
325 MNRY_ASSERT(faceVertexCount[i] == 3 || faceVertexCount[i] == 4);
326 std::reverse(indices.begin() + indexOffset, indices.begin() + indexOffset + faceVertexCount[i]);
327 indexOffset += faceVertexCount[i];
328 }
329
330 attributes->reverseFaceVaryingAttributes();
331}

334Vec3f
335Mesh::computeMotion(const VertexBuffer<Vec3fa, InterleavedTraits> &vertBuf,
336 uint32_t id1, uint32_t id2, uint32_t id3,
337 float vertexW1, float vertexW2, float vertexW3,
338 const mcrt_common::Ray &ray) const
339{
340 Vec3f pos0, pos1;
341 const Vec3f *pos1Ptr = nullptr;
342 if (isMotionBlurOn()) {
343 const Vec3f pShutterOpen =
344 vertexW1 * vertBuf(id1, 0) + vertexW2 * vertBuf(id2, 0) + vertexW3 * vertBuf(id3, 0);
345 const Vec3f pShutterClose =
346 vertexW1 * vertBuf(id1, 1) + vertexW2 * vertBuf(id2, 1) + vertexW3 * vertBuf(id3, 1);
347 pos0 = lerp(pShutterOpen, pShutterClose, ray.getTime() - sHalfDt);
348 pos1 = lerp(pShutterOpen, pShutterClose, ray.getTime() + sHalfDt);
349 pos1Ptr = &pos1;
350 } else {
351 pos0 = vertexW1 * vertBuf(id1, 0) + vertexW2 * vertBuf(id2, 0) + vertexW3 * vertBuf(id3, 0);
352 pos1Ptr = nullptr;
353 }
354
355 // Motion vectors only support a single instance level, hence we only care
356 // about ray.instance0.
357 const Instance *instance = (ray.isInstanceHit())?
358 static_cast<const Instance *>(ray.ext.instance0OrLight) : nullptr;
359 return computePrimitiveMotion(pos0, pos1Ptr, ray.getTime(), instance);
360}

364static void
365interpolatePosInTri(const Vec2f &p, const Vec2f &a, const Vec2f &b, const Vec2f &c,
366 const Vec3f &posA, const Vec3f &posB, const Vec3f &posC,
367 const Vec3f *nrmA, const Vec3f *nrmB, const Vec3f *nrmC,
368 Vec3fa *posResult, Vec3f *nrmResult)
369{
370 // p is a pos in st space, a, b, and c are st coordinates of a triangle,
371 // which are probably in ccw order, but this code does not assume that.
372 // we first determine if p is inside the triangle using 3 half-plane
373 // tests. if p is on the inside of all three half planes
374 // defined by the edges of the triangle, then p is inside the triangle.
375 // once we determine that p is inside the triangle, we compute
376 // barycentric coordinates for p and use them to create an interpolated
377 // result. if p is outside the triange, we leave the results untouched.
378
379 // A little drawing for reference
380 //
381 // a
382 // *
383 // - -
384 // - *p -
385 // *- - - - - - - *
386 // b c
387
388 bool isNegAB = dot(Vec2f(a.y - b.y, b.x - a.x), p - (a + b) * 0.5f) < 0.f;
389 bool isNegBC = dot(Vec2f(b.y - c.y, c.x - b.x), p - (b + c) * 0.5f) < 0.f;
390 bool isNegCA = dot(Vec2f(c.y - a.y, a.x - c.x), p - (a + c) * 0.5f) < 0.f;
391
392 if (isNegAB == isNegBC && isNegBC == isNegCA) {
393 // point is inside triangle, compute barycentric coordinates
394 const float areaABC = scene_rdl2::math::abs((c.x - b.x) * (a.y - b.y) - (a.x - b.x) * (c.y - b.y)); // actually 2x
395 if (areaABC > 0) { // else degenerate
396 const float areaPBC = scene_rdl2::math::abs((p.x - b.x) * (c.y - b.y) - (c.x - b.x) * (p.y - b.y));
397 const float areaAPC = scene_rdl2::math::abs((c.x - p.x) * (a.y - p.y) - (a.x - p.x) * (c.y - p.y));
398 const float invAreaABC = 1.f / areaABC;
399 float u = areaPBC * invAreaABC; // weight of a
400 float v = areaAPC * invAreaABC; // weight of b
401 float w = 1.f - u - v; // weight of c
402 MNRY_ASSERT(
403 u >= -1e-5f && u <= 1.0f + 1e-5f &&
404 v >= -1e-5f && v <= 1.0f + 1e-5f &&
405 w >= -1e-5f && w <= 1.0f + 1e-5f);
406 // clamp to avoid float point precision error, the uvw coordinate
407 // should be all within [0, 1] range mathematically
408 u = scene_rdl2::math::clamp(u);
409 v = scene_rdl2::math::clamp(v);
410 w = scene_rdl2::math::clamp(w);
411 // now compute the interpolated position result
412 MNRY_ASSERT(posResult);
413 *posResult = Vec3fa(posA * u + posB * v + posC * w, 0.f);
414 // set alpha to 1.f to indicate that this pixel has
415 // valid position data
416 posResult->w = 1.f;
417
418 // and the optional normal vector result
419 if (nrmResult) {
420 MNRY_ASSERT(nrmA && nrmB && nrmC);
421 *nrmResult = *nrmA * u + *nrmB * v + *nrmC * w;
422 }
423 }
424 }

427void
428Mesh::rasterizeTrianglePos(const scene_rdl2::math::BBox2f &roiST, int width, int height,
429 const Vec2f &a, const Vec2f &b, const Vec2f &c,
430 const Vec3f &posA, const Vec3f &posB, const Vec3f &posC,
431 const Vec3f *nrmA, const Vec3f *nrmB, const Vec3f *nrmC,
432 Vec3fa *posResult, Vec3f *nrmResult)
433{
434 const scene_rdl2::math::BBox2i roiXY(scene_rdl2::math::Vec2i(floor(roiST.lower.x * (width - 1)),
435 floor(roiST.lower.y * (height - 1))),
436 scene_rdl2::math::Vec2i(ceil(roiST.upper.x * (width - 1)),
437 ceil(roiST.upper.y * (height - 1))));
438 for (int y = roiXY.lower.y; y <= roiXY.upper.y; ++y) {
439 const float t = float(y) / (height - 1);
440 for (int x = roiXY.lower.x; x <= roiXY.upper.x; ++x) {
441 const Vec2f st(float(x) / (width - 1), t);
442 MNRY_ASSERT(posResult);
443 Vec3fa *pos = posResult + y * width + x;
444 Vec3f *nrm = nrmResult? nrmResult + y * width + x : nullptr;
445 interpolatePosInTri(st, a, b, c, posA, posB, posC, nrmA, nrmB, nrmC, pos, nrm);
446 }
447 }

450bool
451Mesh::udimxform(int udim, Vec2f &st)
452{
453 const int minS = (udim - 1001) % 10;
454 const int minT = (udim - 1001) / 10;
455 if (st.x >= minS && st.x <= minS + 1 &&
456 st.y >= minT && st.y <= minT + 1) {
457 // accept it
458 st.x -= minS;
459 st.y -= minT;
460 st.x = scene_rdl2::math::clamp(st.x);
461 st.y = scene_rdl2::math::clamp(st.y);
462 return true;
463 }
464 return false;
465}
