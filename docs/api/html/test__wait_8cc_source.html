408template <typename T, bool isEnum>

411template <typename T>
412struct MaxThreadsHelper<T, false /*isEnum*/>

417// We don't want more threads than our enum type can handle, otherwise we overflow.

443template <bool isEnum>
444struct IncrementHelper;
445
446template <>
447struct IncrementHelper<false /*isEnum*/>
448{
449 template <typename T>
450 static T increment(T a) noexcept
451 {
452 return a + one<T>;
453 }

456template <>
457struct IncrementHelper<true /*isEnum*/>
458{
459 template <typename T>
460 static T increment(T a) noexcept
461 {
462 using type = typename std::underlying_type<T>::type;
463 const auto val = to_underlying(a);
464 return T(static_cast<type>(val + type(1)));
465 }

469T increment(T a) noexcept
470{
471 return IncrementHelper<std::is_enum<T>::value>::increment(a);

474template <typename T>
475void atomic_increment(std::atomic<T>& a) noexcept
476{
477 T val = a.load();
478 T next;
479 do {
480 next = increment(val);
481 } while (!a.compare_exchange_weak(val, next));
482}

523template <typename T>
524size_t getNumThreads()
525{
526 const size_t maxThreads = MaxThreads<T>::value;
527 const size_t minThreads = 8u;
528 const size_t hw = std::thread::hardware_concurrency();
529 return std::max(minThreads, std::min(maxThreads, hw));
530}

532template <typename T>
533void doTestNotifyAll()
534{
535 const int numThreads = getNumThreads<T>() - 1u;
536
537 std::atomic<int> launchCount(0);
538 std::atomic<T> a(zero<T>);
539
540 std::vector<std::thread> threads;
541 threads.reserve(numThreads);
542 for (int i = 0; i < numThreads; ++i) {
543 threads.emplace_back(notifyAllThreads<T>, std::ref(a), std::ref(launchCount));
544 }
545
546 while (launchCount < numThreads) {
547 CPPUNIT_ASSERT_EQUAL(zero<T>, a.load());
548 }
549
550 std::this_thread::sleep_for(1s);
551 CPPUNIT_ASSERT_EQUAL(zero<T>, a.load());
552
553 a.store(one<T>);
554 notify_all(a);
555
556 for (auto& t : threads) {
557 t.join();
558 }
559
560 CPPUNIT_ASSERT_EQUAL(T(numThreads + 1), a.load()); // The main thread sets it to 1 to start off with
561}
