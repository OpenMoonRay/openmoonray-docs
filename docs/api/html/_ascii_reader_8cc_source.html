47/**
48 * GENERAL NOTES:
49 *
50 * The Lua C API:
51 * If you're going to make changes here, it's worth spending some time to
52 * familiarize yourself with the Lua C API. It's not very complicated.
53 *
54 * The core idea behind interfacing with Lua is that across every Lua/C
55 * function boundary there is a virtual stack. The caller (Lua, usually)
56 * pushes its arguments onto this stack. These arguments are accessible at
57 * stack indices 1, 2, 3, etc. where 1 is the bottom of the stack and the
58 * first argument.
59 *
60 * The stack is also addressable by "virtual indices", which are negative.
61 * These index the stack from the top, where -1 is the top of the stack,
62 * -2 is the second from the top, etc.
63 *
64 * When Lua C API functions are dealing with arguments, they'll usually be
65 * speaking in positive stack indices. When they push something onto the
66 * stack for use, they'll usually be speaking in negative stack indices.
67 *
68 * Error Handling:
69 * There are a couple layers of error handling here. The top level is
70 * when the AsciiReader loads a Lua chunk and executes it. Any Lua errors
71 * that are triggered will be caught by the AsciiReader and translated to
72 * a throwing an exception. (except::RuntimeError)
73 *
74 * This means that if any errors occur in one of our callbacks (C++ code
75 * called from Lua), we need to signal error conditions by triggering a
76 * Lua error. This is important so that we can get good error messages,
77 * line numbers, and potentially stack traces.
78 *
79 * The gist of this is that any function that will be called from Lua
80 * (the ones wrapped in RDL2_LUA_DEFINE() macros) absolutely *must*
81 * catch any exceptions which are caused by common text file mistakes,
82 * (like TypeErrors or wrong number of function arguments) and convert
83 * them to a Lua error. It's recommended to pass along the exception
84 * message as part of the Lua error message, so that we have the detailed
85 * message along with the line number of the problem.
86 *
87 * This is *not* the case for catastrophic correctness exceptions. For
88 * example, if some internal RDL code experiences a huge problem, it's
89 * perfectly fine for it throw an exception that is *not* caught and
90 * turned into a Lua error. We want that to propagate all the way up the
91 * stack and kill the program.
92 *
93 * SceneObjects:
94 * Handling SceneObjects requires some care. It is possible to get null
95 * SceneObjects in the mix if someone requests the value of a
96 * SceneObject* attribute that isn't set, as it defaults to null. Thus,
97 * you cannot assume that just because you retrieved it using
98 * extractSceneObject() or checkSceneObject() that it's non-null. If you
99 * intend to dereference it, you should check for null as well.
100 */
101
102// Macro for defining both a Lua callback handler and its static dispatch
103// function in one go.
104#define RDL2_LUA_DEFINE(func_name) \
105 int \
106 AsciiReader::func_name##_DISPATCHER(lua_State* state) \
107 { \
108 AsciiReader* instance = AsciiReader::loadInstancePtr(state); \
109 return instance->func_name##_HANDLER(); \
110 } \
111 int \
112 AsciiReader::func_name##_HANDLER()
113
114// Macro for converting a callback function name into the correct function
115// pointer. (Used when registering functions by hand, like you might do with
116// metatables.)
117#define RDL2_LUA_FUNCPTR(func_name) AsciiReader::func_name##_DISPATCHER
118
119// Macro for registering metamethod callbacks to a metatable that is on the top
120// of the stack.
121#define RDL2_LUA_METAMETHOD(metamethod_name, func_name) \
122 lua_pushstring(mLua, metamethod_name); \
123 lua_pushcfunction(mLua, RDL2_LUA_FUNCPTR(func_name)); \
124 lua_settable(mLua, -3)

277void
278AsciiReader::fromString(const std::string& code, const std::string& chunkName)
279{
280 // Evaluate the Lua code. At this point we'll get callbacks from Lua for
281 // anything interesting. (This just does what luaL_dostring() does, we've
282 // just expanded it here because we'd like to control the "name" of the
283 // chunk for nice error messages.)
284 if (luaL_loadbuffer(mLua, code.c_str(), code.size(), chunkName.c_str()) ||
285 lua_pcall(mLua, 0, LUA_MULTRET, 0)) {
286 std::string errorMessage("RDLA Error: ");
287 errorMessage.append(lua_tostring(mLua, -1));
288 throw except::RuntimeError(errorMessage);
289 }

292void
293AsciiReader::storeInstancePtr()
294{
295 // Save a pointer to "this" in the Lua registry. This allows us to look up
296 // the right instance when we get a callback and forward the call
297 // appropriately.
298 lua_pushlightuserdata(mLua,
299 static_cast<void*>(const_cast<char*>(&LUA_REGISTRY_KEY)));
300 lua_pushlightuserdata(mLua, static_cast<void*>(this));
301 lua_settable(mLua, LUA_REGISTRYINDEX);
302}
303
304AsciiReader*
305AsciiReader::loadInstancePtr(lua_State* state)
306{
307 // Grabs the "this" pointer out of the Lua registry. This allows us to
308 // forward callbacks to the right AsciiReader instance.
309 MNRY_ASSERT(state);
310 lua_pushlightuserdata(state,
311 static_cast<void*>(const_cast<char*>(&LUA_REGISTRY_KEY)));
312 lua_gettable(state, LUA_REGISTRYINDEX);
313 AsciiReader* ptr = static_cast<AsciiReader*>(lua_touserdata(state, -1));
314 MNRY_ASSERT(ptr);
315 lua_pop(state, 1);
316 return ptr;
317}

536 // Bound values are tables which hold a SceneObject binding and a base
537 // value. They support conversion to strings.
538 if (luaL_newmetatable(mLua, BOUND_VALUE_METATABLE)) {
539 RDL2_LUA_METAMETHOD("__tostring", boundValueToString);
540 }
541 lua_pop(mLua, 1);
542
543 // Blurred values are tables which hold multiple values at different time
544 // samples. They support conversion to strings.
545 if (luaL_newmetatable(mLua, BLURRED_VALUE_METATABLE)) {
546 RDL2_LUA_METAMETHOD("__tostring", blurredValueToString);
547 }
548 lua_pop(mLua, 1);
549
550 // Undef values are equivalent to 'nil', but support assignment in a metatable
551 // which allows values to be unbound and unref'd in the Ascii format since LUA
552 // doesn't support assignment to 'nil' (It's a delete key operation)
553 if (luaL_newmetatable(mLua, UNDEF_VALUE_METATABLE)) {
554 RDL2_LUA_METAMETHOD("__eq", undefValueEqual);
555 RDL2_LUA_METAMETHOD("__tostring", undefValueToString);
556 }
557 lua_pop(mLua, 1);
558}

560template <typename T, typename... Args>
561T*
562AsciiReader::boxNew(const char* metatable, Args&&... args)
563{
564 // We need to allocate enough space on Lua's heap such that we can properly
565 // align an object of type T somewhere within that allocated space.
566 constexpr std::size_t alignment = alignof(T);
567 static_assert(util::StaticIsPowerOfTwo<alignment>::value,
568 "Alignment must be a power of 2.");
569 const std::size_t alignableSize = sizeof(T) + alignment - 1;
570
571 // Allocate the space on Lua's heap, compute the base address which
572 // satisfies the alignment, and manually construct the object there.
573 void* address = lua_newuserdata(mLua, alignableSize);
574 MNRY_ASSERT(address);
575 T* obj = static_cast<T*>(alloc::align(address, alignment));
576 new (obj) T(std::forward<Args>(args)...);
577
578 luaL_setmetatable(mLua, metatable);
579 return obj;
580}

582template <typename T>
583T*
584AsciiReader::unbox(void* boxPtr)
585{
586 // The object doesn't necessarily live at zero offset from the pointer.
587 // Recompute the alignment based on the base pointer and return a typed
588 // pointer to this aligned address.
589 MNRY_ASSERT(boxPtr);
590 constexpr std::size_t alignment = alignof(T);
591 void* obj = alloc::align(boxPtr, alignment);
592 return static_cast<T*>(obj);
593}

1010// Specialized version for converting from lua_Number to float which handled denormals.
1011template <>
1012float
1013AsciiReader::extractNumeric(int index)
1014{
1015 if (!lua_isnumber(mLua, index)) {
1016 throw except::TypeError(util::buildString(
1017 "number expected, got ", luaL_typename(mLua, index)));
1018 }
1019
1020 lua_Number num = lua_tonumber(mLua, index);
1021 if (num >= 0x1.0p-126 || num <= -0x1.0p-126 || num == 0.0) {
1022 // Normal float
1023 return static_cast<float>(num);
1024 }
1025
1026 // Denormal float - create bit pattern for mantissa.
1027 // A denormal float works very much like an integer in its representation - it has a fixed exponent encoded as all
1028 // zeros, its mantissa ranges from 0x00000001 to 0x007FFFFF, and every 1ulp step in this value represents the same
1029 // distance in float space. So you just need to scale the double up by a suitably large power of 2 such that the
1030 // integer part becomes the desired bit pattern, and convert to int. The necessary exponent is 126+23: 126 to undo
1031 // the float exponent bias, plus 23 since 0x007FFFFF is 2^23-1. The only thing needed beyond this is special
1032 // treatment of the sign bit, because the denormal bit pattern doesn't behave like a two's complement number.
1033 int bits = static_cast<int>(std::round(num * 0x1.0p149));
1034 if (num < 0.0) {
1035 // Mantissa needs to be a positive number. So if the value was negative,
1036 // negate the mantissa and set the sign bit
1037 bits = -bits | 0x80000000;
1038 }
1039 return util::bitCast<float>(bits);
1040}

1705void
1706AsciiReader::setAttribute(SceneObject* so, const Attribute* attr, int valueIndex)
1707{
1708 MNRY_ASSERT(so);
1709 MNRY_ASSERT(attr);
1710
1711 // Are we trying to set a binding?
1712 if (hasMetatable(valueIndex, BOUND_VALUE_METATABLE)) {
1713 // Make sure the attribute is actually bindable.
1714 if (!attr->isBindable()) {
1715 throw except::ValueError(util::buildString("Attribute '",
1716 attr->getName(), "' is not bindable."));
1717 }
1718
1719 try {
1720 // Extract the bound object.
1721 lua_getfield(mLua, valueIndex, "binding");
1722 LuaPopGuard bindingGuard(mLua, 1);
1723
1724
1725 SceneObject* boundObj = extractSceneObject(-1);
1726
1727 // Set the binding.
1728 setBinding(so, attr, boundObj);
1729 } catch (except::TypeError& e) {
1730 // Add useful information and rethrow.
1731 throw except::TypeError(util::buildString("bad binding: ", e.what()));
1732 }
1733
1734 // If we also have a value member, change the valueIndex to the top of
1735 // the Lua stack and continue on with setting the attribute value.
1736 lua_getfield(mLua, valueIndex, "value");
1737 if (lua_isnil(mLua, -1)) return; // No base value, we're done.
1738 valueIndex = -1;
1739 } else {
1740 // if we are not setting the binding, make sure there is no binding
1741 if (attr->isBindable()) {
1742 setBinding(so, attr, nullptr);
1743 }
1744 }
1745
1746 // Are we trying to set a blurred value?
1747 if (hasMetatable(valueIndex, BLURRED_VALUE_METATABLE)) {
1748 // Set the begin value.
1749 lua_rawgeti(mLua, valueIndex, 1);
1750 setValue(so, attr, -1, true, TIMESTEP_BEGIN);
1751 lua_pop(mLua, 1);
1752
1753 // Set the end value.
1754 lua_rawgeti(mLua, valueIndex, 2);
1755 setValue(so, attr, -1, true, TIMESTEP_END);
1756 lua_pop(mLua, 1);
1757 } else {
1758 // No blurred value, just set a single value.
1759 setValue(so, attr, valueIndex);
1760 }

1763template <typename SetT, typename ElemT>
1764int
1765AsciiReader::commonCall(const char* setTypeName, const char* elemTypeName)
1766{
1767 // Verify arguments.
1768 checkArgCount(2, util::buildString(setTypeName, " mass set").c_str());
1769 SceneObject* so = unboxPtr<SceneObject>(lua_touserdata(mLua, 1));
1770 if (!so) {
1771 const char* msg = lua_pushfstring(mLua,
1772 "Cannot set members of a null %s.", setTypeName);
1773 return luaL_argerror(mLua, 1, msg);
1774 }
1775 SetT* set = so->asA<SetT>();
1776 if (!set) {
1777 const char* msg = lua_pushfstring(mLua, "%s expected, got %s",
1778 setTypeName, luaL_typename(mLua, 1));
1779 return luaL_argerror(mLua, 1, msg);
1780 }
1781 luaL_checktype(mLua, 2, LUA_TTABLE);
1782
1783 // Pull each element out of the table and ensure that they're all valid
1784 // ElemT objects.
1785 std::vector<ElemT*> elems;
1786 for (size_t i = 1; i <= lua_rawlen(mLua, 2); ++i) {
1787 lua_rawgeti(mLua, 2, i);
1788 LuaPopGuard popGuard(mLua, 1);
1789 try {
1790 // Make sure it's a non-null SceneObject and also an ElemT.
1791 SceneObject* so = extractSceneObject(-1,
1792 SCENE_OBJECT_METATABLE);
1793 if (!so) {
1794 throw except::TypeError(util::buildString(elemTypeName,
1795 " expected, got null SceneObject"));
1796 }
1797 ElemT* elem = so->asA<ElemT>();
1798 // Add it to the list.
1799 elems.push_back(elem);
1800 } catch (except::TypeError& e) {
1801 // Add useful information and error out.
1802 const char* msg = lua_pushfstring(mLua, "bad element #%d"
1803 " in table (%s)", i, e.what());
1804 return luaL_argerror(mLua, 2, msg);
1805 }
1806 }
1807
1808 // Actually set the contents of the set.
1809 SceneObject::UpdateGuard guard(set);
1810 for (auto iter = elems.begin(); iter != elems.end(); ++iter) {
1811 set->add(*iter);
1812 }
1813
1814 // Return the object itself (allows for chaining).
1815 lua_pushvalue(mLua, 1);
1816 return 1;
1817}
