118void
119DeepTile::setChannels (const ChannelSet& channels,
120 bool force)
121{
122 PixelTile::setChannels(channels, force);
123 // Get the spmask and flag channel assignments:
124 m_spmask_channel[0] = m_spmask_channel[1] = OPENDCX_INTERNAL_NAMESPACE::Chan_Invalid;
125 m_flags_channel = OPENDCX_INTERNAL_NAMESPACE::Chan_Invalid;
126 foreach_channel(z, channels)
127 {
128 if (z == Chan_SpBits1 && m_spmask_channel[0] == OPENDCX_INTERNAL_NAMESPACE::Chan_Invalid)
129 m_spmask_channel[0] = Chan_SpBits1;
130 else if (z == Chan_SpBits2 && m_spmask_channel[1] == OPENDCX_INTERNAL_NAMESPACE::Chan_Invalid)
131 m_spmask_channel[1] = Chan_SpBits2;
132 else if (z == Chan_DeepFlags && m_flags_channel == Chan_Invalid)
133 m_flags_channel = Chan_DeepFlags;
134 }
135}
