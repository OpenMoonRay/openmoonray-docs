164int
165Geometry::getVisibilityMask() const
166{
167 int visibilityMask = NONE_VISIBLE;
168 visibilityMask |= get(sVisibleCamera) ? CAMERA : 0;
169 visibilityMask |= get(sVisibleShadow) ? SHADOW : 0;
170 visibilityMask |= get(sVisibleDiffuseReflection) ? DIFFUSE_REFLECTION : 0;
171 visibilityMask |= get(sVisibleDiffuseTransmission) ? DIFFUSE_TRANSMISSION : 0;
172 visibilityMask |= get(sVisibleGlossyReflection) ? GLOSSY_REFLECTION : 0;
173 visibilityMask |= get(sVisibleGlossyTransmission) ? GLOSSY_TRANSMISSION : 0;
174 visibilityMask |= get(sVisibleMirrorReflection) ? MIRROR_REFLECTION : 0;
175 visibilityMask |= get(sVisibleMirrorTransmission) ? MIRROR_TRANSMISSION : 0;
176 visibilityMask |= get(sVisiblePhase) ? PHASE_REFLECTION | PHASE_TRANSMISSION: 0;
177 return visibilityMask;
178}

180bool
181Geometry::requiresGeometryUpdate(UpdateHelper& sceneObjects, int depth)
182{
183 for (auto it = mSceneClass.beginAttributes();
184 it != mSceneClass.endAttributes(); ++it) {
185 const Attribute* attribute = *it;
186 // some attributes (like "ray_epsilon") updates would not change the geometry
187 if (!attribute->updateRequiresGeomReload()) {
188 continue;
189 }
190 // whether this attribute or its dependency is dirtied
191 bool updateRequired = hasChanged(attribute) || hasBindingChanged(attribute);
192 if (updateRequired) {
193 return true;
194 }
195
196 switch (attribute->getType()) {
197 case TYPE_SCENE_OBJECT:
198 {
199 const AttributeKey<SceneObject*> key(*attribute);
200 SceneObject * const object = get(key);
201 if (object) {
202 if (object->isA<Geometry>() &&
203 !object->asA<Geometry>()->requiresGeometryUpdate(sceneObjects, depth + 1)) {
204 continue;
205 }
206 updateRequired |= object->updatePrep(sceneObjects, depth + 1);
207 }
208 }
209 break;
210
211 case TYPE_SCENE_OBJECT_VECTOR:
212 updateRequired |= geometryUpdatePrepSequenceContainer<SceneObjectVector>(
213 attribute, sceneObjects, depth + 1);
214 break;
215
216 case TYPE_SCENE_OBJECT_INDEXABLE:
217 updateRequired |= geometryUpdatePrepSequenceContainer<SceneObjectIndexable>(
218 attribute, sceneObjects, depth + 1);
219 break;
220
221 default:
222 break;
223 }
224
225 if (updateRequired) {
226 return true;
227 }
228 }
229 // This geometry does not have an attribute change that requires an update.
230 // Change the flag to false.
231 mAttributeTreeChanged = false;
232 return false;
233}
