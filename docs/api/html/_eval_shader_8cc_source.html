33/// Transform the shader local aov labels to global aov label ids
34void
35xformLobeLabels(const scene_rdl2::rdl2::Material &material, Bsdf *bsdf, int parentLobeCount)
36{
37 MNRY_ASSERT(material.hasExtension());
38 if (material.hasExtension()) {
39 const auto &ext = material.get<shading::Material>();
40 const int materialLabelId = ext.getMaterialLabelId(); // material aovs
41 const auto &lobeLabelIds = ext.getLobeLabelIds(); // material aovs
42 const int lpeMaterialLabelId = ext.getLpeMaterialLabelId(); // lpe aovs
43 const auto &lpeLobeLabelIds = ext.getLpeLobeLabelIds(); // lpe aovs
44
45 // Transform the shader local aov labels to global aov label ids.
46 // We only transform the lobe labels of the current material.
47 // Lobes are added in a top down manner, with parent lobes
48 // added before child lobes. Lobe labels, on the other hand,
49 // Are transformed in a bottom up manner, with child labels
50 // transformed before parent labels.
51 // Therefore we must skip the lobes of the parent material here.
52 // If this material has a child material, its lobe labels are
53 // already transformed. Labels that are already transformed
54 // are ignored in aovEncodeLabels.
55 for (int i = parentLobeCount; i < bsdf->getLobeCount(); ++i) {
56 BsdfLobe *lobe = bsdf->getLobe(i);
57 lobe->setLabel(aovEncodeLabels(lobe->getLabel(),
58 materialLabelId, lpeMaterialLabelId,
59 lobeLabelIds, lpeLobeLabelIds));
60 }
61
62 Bssrdf *bssrdf = bsdf->getBssrdf();
63 if (bssrdf) {
64 bssrdf->setLabel(aovEncodeLabels(bssrdf->getLabel(),
65 materialLabelId, lpeMaterialLabelId,
66 lobeLabelIds, lpeLobeLabelIds));
67 }
68
69 VolumeSubsurface *vs = bsdf->getVolumeSubsurface();
70 if (vs) {
71 vs->setLabel(aovEncodeLabels(vs->getLabel(),
72 materialLabelId, lpeMaterialLabelId,
73 lobeLabelIds, lpeLobeLabelIds));
74 }
75 }
76}

78/// Hack in bsdf label ids used for aovs
79void
80setBsdfLabels(const scene_rdl2::rdl2::Material &material, const State &state,
81 Bsdf *bsdf, int parentLobeCount)
82{
83 MNRY_ASSERT(material.hasExtension());
84 if (material.hasExtension()) {
85 const auto &ext = material.get<shading::Material>();
86 const scene_rdl2::rdl2::Geometry *geometry = state.getGeometryObject();
87 int geomLabelId = -1;
88 MNRY_ASSERT(geometry);
89 MNRY_ASSERT(geometry->hasExtension());
90 if (geometry && geometry->hasExtension()) {
91 geomLabelId = geometry->get<shading::Geometry>().getGeomLabelId();
92 }
93 bsdf->setLabelIds(ext.getMaterialLabelId(), ext.getLpeMaterialLabelId(),
94 geomLabelId);
95 }
96 xformLobeLabels(material, bsdf, parentLobeCount);
97}

99/// Evaluate a root material from the rendering application
100// TODO: MOONRAY-3174
101// Ideally, this function would be called by the PathIntegrator any more because it
102// could not be inlined and caused a shading performance regression.
103//
104// This function is currently only called from lib/rendering/pvr/integrator/Picking.cc.
105//
106// All non-public types used here need to be made public or replaced with similar
107// functionality provided by a public API so this function can once again be inlined.
108//
109//
110// problematic types:
111// -----------------------------------------------------------------------------------------------------
112// * EXCL_ACCUMULATOR_PROFILE, EXCL_ACCUM_SHADING (rendering/mcrt_common/ProfileAccumulatorHandles.h)
113// * time::RAIITicker<> (common/time/Ticker.h)
114// * util::InclusiveExclusiveAverage<> (common/mcrt_util/Average.h)
115void
116shade(const scene_rdl2::rdl2::Material *material, shading::TLState *tls,
117 const State &state, BsdfBuilder &bsdfBuilder)
118{
119 MNRY_ASSERT(material);
120
121 EXCL_ACCUMULATOR_PROFILE(tls, EXCL_ACCUM_SHADING);
122
123#ifdef SHADING_BRACKET_TIMING_ENABLED
124 auto threadIndex = tls->mThreadIdx;
125 time::RAIITicker<util::InclusiveExclusiveAverage<int64> > ticker(
126 MNRY_VERIFY(material->getThreadLocalObjectState())[threadIndex].mShaderCallStat);
127#endif
128
129 auto bsdf = const_cast<Bsdf*>(bsdfBuilder.getBsdf());
130 int parentLobeCount = bsdf->getLobeCount();
131 material->shade(tls, state, bsdfBuilder);
132 setBsdfLabels(*material, state, bsdf, parentLobeCount);
133#ifdef SHADING_PRINT_DEBUG_BSDF_INFO_ENABLED
134 bsdf->show(material->getSceneClass().getName(), material->getName(), std::cout);
135#endif
136}

139/// Evaluate a child material from a parent material using BsdfBuilder
140// TODO: MOONRAY-3174
141// Profiling shows that this function would benefit from inlining, but
142// TLState and time::RAIIInclusiveExclusiveTicker are not currentlu
143// public. There are essentially three places where shade() is called:
144// 1) by the integrator (actually, this does not currently call shade() see below)
145// 2) by lib/rendering/pbr/integrator/Picker.cc
146// 3) by certain "parent" materials for shading "child" materials
147//
148// The current state of affairs is as follows:
149// 1) integrator: for performance reasons, shade() is not actually called
150// from the integrator currently (ideally it would be). Until the functions
151// can all be inlined, a copy of the relevant functionality is built directly
152// into PathIntegrator.cc.
153// 2) lib/rendering/pbr/integrator/Picker.cc: The function above is called,
154// since performance doesn't seem critical
155// 3) "parent" materials: They call this function, although it is likely
156// not as performant as it would be were it inlined. Efforts should be
157// made to improve the performance of this function by inlining or
158// other means.
159void
160shade(const scene_rdl2::rdl2::Material *parent, const scene_rdl2::rdl2::Material *material,
161 shading::TLState *tls, const State &state, BsdfBuilder& bsdfBuilder)
162{
163 MNRY_ASSERT(material);
164
165 EXCL_ACCUMULATOR_PROFILE(tls, EXCL_ACCUM_SHADING);
166
167#ifdef SHADING_BRACKET_TIMING_ENABLED
168 auto threadIndex = tls->mThreadIdx;
169 time::RAIIInclusiveExclusiveTicker<int64> ticker(
170 MNRY_VERIFY(material->getThreadLocalObjectState())[threadIndex].mShaderCallStat,
171 MNRY_VERIFY(parent->getThreadLocalObjectState())[threadIndex].mShaderCallStat);
172#endif
173
174 int parentLobeCount = bsdfBuilder.getBsdf()->getLobeCount();
175 material->shade(tls, state, bsdfBuilder);
176 setBsdfLabels(*material, state, const_cast<Bsdf*>(bsdfBuilder.getBsdf()), parentLobeCount);
177}
