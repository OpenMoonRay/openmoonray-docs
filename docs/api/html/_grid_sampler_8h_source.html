93template<typename GridType>
94struct VDBSampler
95{
96 typedef typename GridType::ValueType ValueT;
97 typedef typename GridType::ConstPtr GridConstPtr;
98
99 VDBSampler(): mGrid(nullptr), mIsValid(false), mVolumeIdCount(0u)
100 {
101 }
102
103 void initialize(const GridConstPtr& grid,
104 const std::vector<int>& volumeIds,
105 StatCounters statsCounterType)
106 {
107 mGrid = grid;
108 mIsValid = (mGrid != nullptr) && !mGrid->empty();
109 if (mIsValid) {
110 mVolumeIdCount = volumeIds.size();
111 for (uint32_t samplerId = 0; samplerId < mVolumeIdCount; ++samplerId) {
112 mVolumeIdToSamplerId[volumeIds[samplerId]] = samplerId;
113 }
114 unsigned mSamplerCount = mVolumeIdCount * mcrt_common::getNumTBBThreads();
115 mSamplers.reserve(mSamplerCount);
116 for (size_t i = 0; i < mSamplerCount; ++i) {
117 mSamplers.emplace_back(*mGrid);
118 }
119 }
120
121 mStatsCounterType = statsCounterType;
122 }
123
124 ValueT eval(mcrt_common::ThreadLocalState* tls,
125 uint32_t volumeId,
126 const openvdb::Vec3d& p,
127 Interpolation mode) const
128 {
129 if (mIsValid) {
130 tls->mGeomTls->mStatistics.incCounter(mStatsCounterType);
131 uint32_t threadIdx = tls->mThreadIdx;
132 unsigned samplerIdx = threadIdx * mVolumeIdCount + (mVolumeIdToSamplerId.at(volumeId));
133 switch (mode) {
134 case Interpolation::POINT:
135 return mSamplers[samplerIdx].evalPoint(p);
136 break;
137 case Interpolation::BOX:
138 return mSamplers[samplerIdx].evalBox(p);
139 break;
140 case Interpolation::QUADRATIC:
141 return mSamplers[samplerIdx].evalQuadratic(p);
142 break;
143 default:
144 return mSamplers[samplerIdx].evalPoint(p);
145 break;
146 }
147 }
148
149 return ValueT(0.0f);
150 }
151
152 size_t getMemory() const
153 {
154 // We do not need to call mGrid->memUsage() because the grid itself
155 // is shared with other members of the Primitive class, and we
156 // count those members instead.
157
158 return scene_rdl2::util::getVectorElementsMemory(mSamplers);
159 }
160
161 GridConstPtr mGrid;
162 bool mIsValid;
163 std::vector<GridSampler<GridType>> mSamplers;
164 std::unordered_map<uint32_t, uint32_t> mVolumeIdToSamplerId;
165 unsigned mVolumeIdCount;
166 StatCounters mStatsCounterType;
167};
