28// The AdaptiveRegionTree has no notion of threading. All threading considerations are pushed to this class.
29// The AdaptiveRegions, as the name may imply, contains multiple AdaptiveRegionTrees for portions of screen space.
30//
31// On a high level, this allows one thread to update the AdaptiveRegionTree for a region while the other threads are off
32// rendering other sections of the image. The AdaptiveRegions keeps track of how many tiles have been rendered in
33// an area. Once all of an area's tiles have been visited, the last thread updates the AdaptiveRegionTree. This scheme
34// works best when rendering with the Morton curve tile order, so that other threads aren't waiting on the quad to be
35// updated and continue rendering.
36class AdaptiveRegions
37{
38 // When rendering with a Morton curve, 2x2 regions gives us mostly
39 // contention-free updates without a lot of overhead, until we near the end
40 // of rendering and we spend more time updating than rendering. This is
41 // now greater than that for better load-balancing, and if the tile order
42 // is something like image-center out, having smaller regions may allow
43 // inner regions to be updated while the outer regions continue to render.
44 // It may also make sense to increase this value when doing vectorized
45 // rendering, where frame buffer results aren't as deterministic.
46 static constexpr int sDesiredNRegionsPerDimension = 4;
47 static constexpr int sMaxNRegions = sDesiredNRegionsPerDimension*sDesiredNRegionsPerDimension;
48 static constexpr int sTileWidth = COARSE_TILE_SIZE;
49 static constexpr int sTileHeight = COARSE_TILE_SIZE;
50 static constexpr int sRegionOverlap = COARSE_TILE_SIZE;
51
52 using Regions = OverlappingRegions<sRegionOverlap, sTileWidth, sTileHeight, sDesiredNRegionsPerDimension>;
53 using MutexType = scene_rdl2::util::ReaderWriterMutex;

141inline scene_rdl2::math::BBox2i AdaptiveRegions::getSampleArea(const scene_rdl2::math::BBox2i& tile, mcrt_common::ThreadLocalState* tls) const
142{
143 EXCL_ACCUMULATOR_PROFILE(tls, EXCL_QUERY_ADAPTIVE_TREE);
144 const int idx = mRegions.getRegionIndex(tile);
145
146 // It's possible that another thread is updating the tree, so we need to lock on our read.
147 // Divide by the tile dimensions so that neighboring tiles access the mutex pool with different indices.
148 ReadLock lock(mMutexPool[idx].getMutex(tile.lower[0]/sTileWidth, tile.lower[1]/sTileHeight));
149 return mTrees[idx].getSampleArea(tile);
150}

152inline float AdaptiveRegions::getError() const
153{
154 float error = 0.0f;
155 for (auto& e : mRegionError) {
156 error = scene_rdl2::math::max(e.load(std::memory_order_relaxed), error);
157 }
158 std::atomic_thread_fence(std::memory_order_release);
159 return error;
160}
161
162inline bool AdaptiveRegions::done() const
163{
164 bool done = true;
165 for (auto& e : mDone) {
166 if (e.load(std::memory_order_relaxed) == false) {
167 done = false;
168 break;
169 }
170 }
171 std::atomic_thread_fence(std::memory_order_release);
172 return done;
173}
