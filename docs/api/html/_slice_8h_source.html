23/**
24 * In a nutshell, this is basically a ripoff of the concept of "slices" from
25 * modern languages like Python, Rust, D, Go, etc. It just tracks a pointer and
26 * a length into some buffer, and is only valid while that buffer stays alive.
27 *
28 * Slices are useful for writing zero copy code, specifically when you need to
29 * do several things to a particular buffer, like read it from a network socket,
30 * deserialize it in chunks, hand those chunks to processing code, etc. We
31 * can stuff all the pointer arithemetic and range checks inside a slice so
32 * our buffer munging code is zero copy, type safe, and range checked.
33 *
34 * Slices have their pointer and length baked in at construction time and are
35 * immutable from that point forward. However, they're so cheap to construct
36 * that if you need to move the slice's boundaries, you can just construct a
37 * new slice within the bounds of the original slice. You cannot construct a
38 * slice with boundaries outside of the slice used to construct it (unless
39 * you're constructing a slice from a new buffer or byte string, of course).
40 *
41 * As an example, let's say some network code reads data into a byte string
42 * buffer (stored in a std::string) and you need to chop up that buffer and
43 * hand it off to processing code, without copying the buffer. Here's what that
44 * might look like:
45 *
46 * std::string buffer = getBufferFromNetwork();
47 * Slice slice(buffer);
48 * Slice header(slice, 128);
49 * Slice body(slice, 128, Slice::rest(slice, 128));
50 * processHeader(header);
51 * processBody(body);
52 *
53 * The processing functions can then take the slice and get access to its
54 * data pointer and length with the .data() and .len() member function
55 * respectively.
56 *
57 * Lastly, if you do need to make a fresh copy of a slice, you can copy it from
58 * the source buffer to a new buffer with copyTo() (for void* buffers) or
59 * copy() (for byte strings with std::string):
60 *
61 * std::string headerCopy = header.copy();
62 *
63 * WARNING: It's critical that the source buffer you used to construct the
64 * original slice stays alive as long as these slices are to remain valid.
65 * In addition when constructing a slice from a std::string byte string, DO
66 * NOT pass a mutable reference around anywhere. Because the string can be
67 * changed through the reference, it could cause a reallocation under the hood,
68 * which would make all of your slice pointers invalid.
69 *
70 * Why not use std::valarray? Or boost::range? Well, we're primarily interested
71 * in slicing up byte buffers that already exist. Secondly, we really only care
72 * about continuous subranges of those buffers, so things like strides and
73 * component-wise math operations would make the interface more confusing.
74 */
75class Slice
76{
77public:
78 /**
79 * Constructs a slice over the given data buffer with a length of len
80 * bytes. No range check is done on len (it's impossible). You MUST keep
81 * the buffer alive as long as this slice (and slices created from this
82 * slice) are to remain valid.
83 *
84 * @param data An opaque data buffer.
85 * @param length The length (in bytes) of the slice.
86 */
87 finline Slice(const void* data, std::size_t length);
88
89 /**
90 * Constructs a slice over the given string. You MUST keep this string
91 * alive as long as this slice (and slices created from this slice) are to
92 * remain valid. You must also NOT modify the string data in any way, or
93 * pass a non-const reference of it to any function (because a reallocation
94 * could invalidate the buffer pointer).
95 *
96 * @param bytes An opaque byte string.
97 */
98 finline explicit Slice(const std::string& bytes);
99
100 /**
101 * Copy constructs a slice from another slice. They both share the same
102 * data buffer and length.
103 *
104 * @param source The slice you want to copy.
105 */
106 finline Slice(const Slice& source);
107
108 /**
109 * Construct a slice from another slice, but reduce its length to len.
110 * This length reduction is range checked, and throws except::IndexError
111 * if the length is not shorter than the source.
112 *
113 * @param source The slice you want to copy.
114 * @param length The reduced length (in bytes) of the new slice.
115 *
116 * @throw except::IndexError If len is larger than the source's length.
117 */
118 finline Slice(const Slice& source, std::size_t length);
119
120 /**
121 * Construct a slice from another slice, but start the data buffer at the
122 * given offset into the source slice's buffer and set the length to len.
123 * Both the offset adjustment and length are range checked such that the
124 * new slice must be a valid subrange of the source slice. Throws
125 * except::IndexError if the new range is not a valid subrange.
126 *
127 * @param source The slice you want to copy.
128 * @param offset The offset (in bytes) from the beginning of the source
129 * slice's data buffer.
130 * @param length The length (in bytes) of the new slice.
131 *
132 * @throw except::IndexError If the new slice is not a valid subrange
133 * of the source slice.
134 */
135 finline Slice(const Slice& source, std::ptrdiff_t offset, std::size_t length);
136
137 /**
138 * Gets an opaque pointer to the data buffer at the beginning of this
139 * slice.
140 *
141 * @return An opaque pointer to the data buffer at the beginning of the
142 * slice.
143 */
144 finline const void* getData() const;
145
146 /**
147 * Gets the length of this slice.
148 *
149 * @return The length of the slice.
150 */
151 finline std::size_t getLength() const;
152
153 /**
154 * Copies bytes from the source data buffer in the region of this slice
155 * to the given destination buffer of length len. If len is shorter than
156 * the slice's length, only len bytes will be copied.
157 *
158 * @param data Destination data buffer.
159 * @param len Length of the destination buffer, max bytes to copy.
160 */
161 finline void copyTo(void* data, std::size_t length) const;
162
163 /**
164 * Copies bytes from the source data buffer in the region of this slice
165 * into a byte string. The whole length of the slice will be copied.
166 *
167 * @return The byte string with the copied data.
168 */
169 finline std::string copy() const;
170
171 /**
172 * A convenience function for computing the correct length of "the rest of
173 * the slice" given a new offset. You can use it like this:
174 *
175 * std::string buffer("abcdefg");
176 * Slice whole(buffer);
177 *
178 * // Construct a new slice with offset and length, where its length
179 * // is "the rest of the other slice".
180 * Slice defg(whole, 3, Slice::rest(whole, 3));
181 * // defg is now a slice over "defg"
182 *
183 * @param other The slice we're taking the "rest" of.
184 * @param offset The offset into other that we're starting at.
185 * @return The length, starting at offset, to the end of slice other.
186 */
187 static finline std::size_t rest(const Slice& other, std::ptrdiff_t offset);
188
189private:
190 /**
191 * Copy assignment is not defined, because slices are baked const at
192 * construction time. By assignment time it's too late. Use the copy
193 * constructor instead.
194 */
195 Slice& operator=(const Slice& other);
196
197 const void* mData;
198 const std::size_t mLength;
199};
