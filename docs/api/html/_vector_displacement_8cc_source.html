68void VectorDisplacement::displace(const Displacement *self, moonray::shading::TLState *tls,
69 const State &state, Vec3f *displace)
70{
71 const VectorDisplacement* me = static_cast<const VectorDisplacement*>(self);
72
73 // Evaluate vector map and apply factor
74 Vec3f vector = evalVec3f(me, attrVector, tls, state);
75 vector *= me->get(attrFactor);
76
77 if (me->get(attrSourceSpace) == ispc::SOURCE_SPACE_TANGENT) {
78 // Transform from tangent space to the geometry local space.
79 const Vec3f &N = state.getN();
80 const ReferenceFrame frame(N, normalize(state.getdPds()));
81 if (me->get(attrTangentSpaceStyle) == ispc::TANGENT_SPACE_STYLE_TNB) {
82 // In tangent space, the ReferenceFrame works Z-up whereas
83 // the vector displacement local space is defined as "green"-up.
84 vector = frame.localToGlobal(Vec3f(vector.x, vector.z, vector.y));
85 } else {
86 vector = frame.localToGlobal(Vec3f(vector.x, vector.y, vector.z));
87 }
88 } else { // Object space
89 // Transform from object space to render space
90 vector = me->mXform->transformVector(ispc::SHADING_SPACE_OBJECT,
91 ispc::SHADING_SPACE_RENDER,
92 state,
93 vector);
94 }
95
96 *displace = vector;
97}
