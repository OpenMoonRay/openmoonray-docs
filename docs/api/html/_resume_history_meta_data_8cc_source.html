234std::string
235ResumeHistoryMetaData::toJsonSampleInfo(const pbr::Statistics &pbrStats) const
236{
237 std::string str;
238 switch (mSampleType) {
239 case SAMPLE_TYPE::UNIFORM :
240 str = stringCat(jsonPairStr("samplingType", "UNIFORM"), ",\n",
241 jsonPairUnsigned("minSamples", mMinSamples), ",\n",
242 jsonPairUnsigned("maxSamples", mMaxSamples), ",\n");
243 break;
244 case SAMPLE_TYPE::ADAPTIVE :
245 // acoording to the RenderContext.cc RenderContext::buildFrameState(). We are using 10000x bigger number
246 // for user control of adaptive sampling error target.
247 float targetError = mAdaptiveTargetError * 10000.0f;
248 str = stringCat(jsonPairStr("samplingType", "ADAPTIVE"), ",\n",
249 jsonPairUnsigned("minSamples", mMinSamples), ",\n",
250 jsonPairUnsigned("maxSamples", mMaxSamples), ",\n",
251 jsonPairFloat("adaptiveTargetError", targetError), ",\n");
252 break;
253 }
254 str += toJsonSampleResultStat(pbrStats);
255 return jsonPairObj("sampling", str);
256}

258std::string
259ResumeHistoryMetaData::toJsonSampleResultStat(const pbr::Statistics &pbrStats) const
260{
261 const size_t pixelSamples = pbrStats.getCounter(pbr::STATS_PIXEL_SAMPLES);
262 const size_t lightSamples = pbrStats.getCounter(pbr::STATS_LIGHT_SAMPLES);
263 const size_t bsdfSamples = pbrStats.getCounter(pbr::STATS_BSDF_SAMPLES);
264 const size_t bssrdfSamples = pbrStats.getCounter(pbr::STATS_SSS_SAMPLES);
265 const size_t totalSamples = pixelSamples + lightSamples + bsdfSamples + bssrdfSamples;
266
267 std::string str;
268 str = stringCat(jsonPairSizeT("PixelSamples", pixelSamples), ",\n",
269 jsonPairSizeT("lightSamples", lightSamples), ",\n",
270 jsonPairSizeT("bsdfSamples", bsdfSamples), ",\n",
271 jsonPairSizeT("bssrdfSamples", bssrdfSamples), ",\n",
272 jsonPairSizeT("totalSamples", totalSamples));
273 return jsonPairObj("sampleResult", str);
274}

296std::string
297ResumeHistoryMetaData::toJsonTimingInfo() const
298{
299 auto isActiveStint = [&](size_t id) -> bool {
300 if (id >= mMCRT.size()) return false; // outside MCRT stint range
301 if (id == 0) return true; // always active for 1st stint
302 if (mMCRT[id - 1].mEndTileSamplesId == mMCRT[id].mEndTileSamplesId) {
303 return false; // This is non active empty stint
304 }
305 return true;
306 };
307
308 auto stint = [&](size_t id) -> std::string {
309 return jsonVal(stringCat(jsonPairUnsigned("stint", id), ",\n",
310 jsonPairTimeVal("MCRTStartTime", mMCRT[id].mMCRTStartTime), ",\n",
311 jsonPairTimeVal("MCRTEndTime", mMCRT[id].mMCRTEndTime), ",\n",
312 jsonPairUnsigned("endTileSamplesId", mMCRT[id].mEndTileSamplesId)));
313
314 };
315
316 std::string str = stringCat(jsonPairBool("bgCheckpointWrite", mBgCheckpointWriteMode), ",\n",
317 jsonPairUnsigned("startTileSamplesId", mStartTileSamplesId), ",\n",
318 jsonPairTimeVal("procStartTime", mProcStartTime), ",\n",
319 jsonPairTimeVal("frameStartTime", mFrameStartTime), ",\n",
320 jsonKey("MCRT"), ":[\n");
321 for (size_t id = 0; id < mMCRT.size(); ++id) {
322 if (isActiveStint(id)) { // We skip empty MCRTStint
323 str += scene_rdl2::str_util::addIndent(stint(id));
324 if (id < mMCRT.size() - 1) str += ",\n";
325 else str += '\n';
326 }
327 }
328 str += "]";
329 return jsonPairObj("timingDetail", str);
330}

347void
348ResumeHistoryMetaData::computeTimingSummary(float &renderPrepSec,
349 float &mcrtSec,
350 float &checkpointTotalSecExcludeLast,
351 float &checkpointAveSec) const
352{
353 renderPrepSec = 0.0f;
354 mcrtSec = 0.0f;
355 checkpointTotalSecExcludeLast = 0.0f; // total time of checkpoint output action exclude last output
356
357 if (!mMCRT.size()) return;
358
359 // renderPrep = frameStartTime ~ 1st MCRT startTime
360 renderPrepSec = secInterval(mFrameStartTime, mMCRT[0].mMCRTStartTime);
361 for (unsigned i = 0; i < mMCRT.size(); ++i) {
362 if (i > 0) {
363 checkpointTotalSecExcludeLast += secInterval(mMCRT[i-1].mMCRTEndTime, mMCRT[i].mMCRTStartTime);
364 }
365 mcrtSec += secInterval(mMCRT[i].mMCRTStartTime, mMCRT[i].mMCRTEndTime);
366 }
367
368 if (mMCRT.size() > 1) {
369 checkpointAveSec = checkpointTotalSecExcludeLast / (float)(mMCRT.size() - 1);
370 } else {
371 checkpointAveSec = 0.0f;
372 }
373}

375float
376ResumeHistoryMetaData::secInterval(const struct timeval &start, const struct timeval &end) const
377{
378 return (float)(end.tv_sec - start.tv_sec) + (float)(end.tv_usec - start.tv_usec) / 1000.0f / 1000.0f;
379}
