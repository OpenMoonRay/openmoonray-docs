42int32_t
43TraceSet::assign(Geometry* geometry, const String& partName)
44{
45 if (!mUpdateActive) {
46 std::stringstream errMsg;
47 errMsg << "Can only make assignment ('" << geometry->getName() <<
48 "', '" << partName << "') in TraceSet '" << mName << "' between"
49 " beginUpdate() and endUpdate() calls.";
50 throw except::RuntimeError(errMsg.str());
51 }
52
53 // Get mutable references to the attribute vectors.
54 auto& geometries = getMutable(sGeometriesKey);
55 auto& parts = getMutable(sPartsKey);
56
57 // We have to do a const cast here, because we have a container of
58 // SceneObject*, but equal_range takes a const T&. The compiler does not
59 // like us trying to take a const pointer reference. It's complicated and
60 // annoying.
61 const auto iters = geometries.equal_range(const_cast<Geometry*>(geometry));
62 // If the assignment already exists, just return the existing assignment ID.
63 for (auto it = iters.first; it != iters.second; ++it) {
64 const auto idx = *it;
65 if (parts[idx] == partName) {
66 return idx;
67 }
68 }
69
70 // Assignment doesn't exist yet, so create it.
71 geometries.push_back(geometry);
72 parts.push_back(partName);
73
74 mAttributeUpdateMask.set(sGeometriesKey.mIndex, true);
75 mAttributeUpdateMask.set(sPartsKey.mIndex, true);
76 mAttributeSetMask.set(sGeometriesKey.mIndex, true);
77 mAttributeSetMask.set(sPartsKey.mIndex, true);
78 mDirty = true;
79
80 return geometries.size() - 1;
81}

83TraceSet::GeometryPartPair
84TraceSet::lookupGeomAndPart(int32_t assignmentId) const
85{
86 const auto& geometries = get(sGeometriesKey);
87 const auto& parts = get(sPartsKey);
88
89 // Sanity check.
90 if (assignmentId < 0 || std::size_t(assignmentId) >= geometries.size()) {
91 std::stringstream errMsg;
92 errMsg << "Assignment ID '" << assignmentId << "' on trace set '" <<
93 getName() << "' is out of range (contains " << geometries.size() <<
94 " assignments).";
95 throw except::IndexError(errMsg.str());
96 }
97
98 return GeometryPartPair(geometries[assignmentId]->asA<Geometry>(),
99 parts[assignmentId]);
100}

102int32_t
103TraceSet::getAssignmentId(const Geometry* geometry, const String& partName) const
104{
105 const auto& geometries = get(sGeometriesKey);
106 const auto& parts = get(sPartsKey);
107
108 // Save the default assignment (part name "") if we come across it during
109 // the search.
110 int32_t defaultAssignmentId = -1; // Not found.
111
112 // We have to do a const cast here, because we have a container of
113 // SceneObject*, but equal_range takes a const T&. The compiler does not
114 // like us trying to take a const pointer reference. It's complicated and
115 // annoying.
116 const auto iters = geometries.equal_range(const_cast<Geometry*>(geometry));
117 for (auto it = iters.first; it != iters.second; ++it) {
118 const auto idx = *it;
119 // Pointer compare for geometry uniqueness is ok, since the SceneContext
120 // enforces that we can't create two SceneObjects with the same name.
121 const std::string& part = parts[idx];
122 if (part == partName) {
123 return idx;
124 } else if (part == "") {
125 defaultAssignmentId = idx;
126 }
127 }
128
129 // Return the default assignment.
130 return defaultAssignmentId;
131}
