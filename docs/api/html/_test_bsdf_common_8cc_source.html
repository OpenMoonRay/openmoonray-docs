44// To add a BSDF test, you need the following (as passed into TestBsdfCommon.cc runTest()):
45//
46// * A function (or function object) that runs the test for a single thread.
47// * A type that holds the results from a single thread (Result).
48// * A type used to accumulate the results from various threads (Aggregate).
49// (The Result type and the Aggregate type are often the same type).
50// * An overloaded function accumulate() which takes aggregate type by reference and the result type by const reference.
51// This function will accumulate the results from a completed thread.
52//
53// void accumulate(Aggregate&, const Result&);
54//
55// * A check() function that takes TestBsdfSettings by const reference and the result type by const reference. This
56// function is run once all threads have completed and is where the unit test assertions should be placed.
57//
58// void check(const TestBsdfSettings&, const Result&);
59//
60// TestBsdfCommon.cc's runTest function will set up producer/consumer queues to dispatch the work to threads and
61// accumulate the results from the threads.
62//
63// Each task function will be passed a random seed and a random stream number (for use with our scene_rdl2::util::Random
64// class).

153 // Divide by pdf and/or number of samples to get the MC estimate
154 const auto integralUniform = results.mIntegralUniform / results.mSampleCount;
155 const auto integralImportance = results.mIntegralImportance / results.mSampleCount;
156
157 // Test energy conservation
158 bool gray;
159 if (isEqual(integralImportance[0], integralImportance[1], 1e-4f) &&
160 isEqual(integralImportance[0], integralImportance[2], 1e-4f)) {
161 gray = true;
162 printDebug("eval integral importance = %f",
163 integralImportance[0]);
164 } else {
165 gray = false;
166 printDebug("eval integral importance = (%f, %f, %f)",
167 integralImportance[0], integralImportance[1], integralImportance[2]);
168 }
169
170 // If the integral is zero, the filter is likely a delta function
171 if (isEqual(integralUniform, sBlack)) {
172 printDebug("bsdf is a delta-function");
173 return;

175
176 // Test energy conservation again

178 printDebug("eval integral uniform = %f", integralUniform[0]);

183
184 // Warn about energy conservation
185 if (!isValidEvalIntegral(integralUniform, test.toleranceIntegral * 4.0f)) {
186 printWarning("eval() uniform is not energy preserving");
187 }
188 if (!isValidEvalIntegral(integralImportance, test.toleranceIntegral)) {
189 printWarning("eval() importance is not energy preserving");
190 }

217// This function populates the task queue.
218template <typename Task, typename... Args>
219void runProducer(TaskQueueType<Task>& queueTasks, unsigned totalSamples, Args&&... args)
220{
221 using namespace std::chrono_literals;
222 scene_rdl2::util::Random seedRNG(2285938u, 185283u);
223
224 unsigned tasksToDistribute = numTasks;
225 unsigned samplesLeft = totalSamples;
226 unsigned previousStart = 0;
227 while (tasksToDistribute > 0) {
228 const unsigned samples = samplesLeft/tasksToDistribute;
229 const std::uint32_t seed = seedRNG.getNextUInt();
230 const std::uint32_t stream = tasksToDistribute * 64u; // Allow space for ISPC code to modify lane values.
231 Task task(previousStart, previousStart + samples, seed, stream, std::forward<Args>(args)...);
232 queueTasks.push(std::move(task));
233 --tasksToDistribute;
234 samplesLeft -= samples;
235 previousStart += samples;
236 }
237 MNRY_ASSERT(samplesLeft == 0);
238 MNRY_ASSERT(tasksToDistribute == 0);
239}

241// This function consumes results until toConsume is zero.
242template <typename Task, typename Result>
243void runTasks(std::atomic<unsigned>& toConsume, TaskQueueType<Task>& queueTasks, ResultQueueType<Result>& queueResults)
244{
245 while (true) {
246 unsigned count = toConsume.load();
247 if (count == 0) {
248 return;
249 }
250 while (!toConsume.compare_exchange_weak(count, count - 1)) {
251 if (count == 0) {
252 return;
253 }
254 }
255
256 Task task = queueTasks.pop();
257 Result result = task();
258 queueResults.push(std::move(result));
259 }
260}

262template <typename Aggregate, typename Result>
263Aggregate runConsumer(unsigned toConsume, ResultQueueType<Result>& queueResults)
264{
265 Aggregate aggregate;
266
267 // This is threaded: we can't just rely on the queue being empty.
268 for ( ; toConsume > 0; --toConsume) {
269 accumulate(aggregate, queueResults.pop());
270 }
271 MNRY_ASSERT(queueResults.empty());
272
273 return aggregate;
274}

409{
410#ifndef ONLY_ASSERT_INTEGRALS
411 printInfo("----- testConsistency() -----");
412 testConsistency(test, sampleCount);

416 printInfo("----- Assert integrals -----");

418 float phy = 0.0f;
419 Vec3f wo = shading::computeLocalSphericalDirection(scene_rdl2::math::cos(theta), scene_rdl2::math::sin(theta), phy);
420 wo = test.frame.localToGlobal(wo);
421
422 if (test.assertPdfIntegral) {
423 testPdfIntegral(test, wo, sampleCount * 10, true);
424 }
425 if (test.assertEvalIntegral) {
426 testEvalIntegral(test, wo, sampleCount * 10, true);
427 }
428 }
429
430#ifndef ONLY_ASSERT_INTEGRALS
431 float thetaInc = sHalfPi / viewAnglesTheta;
432 float phyInc = sTwoPi / viewAnglesPhy;
433
434 for (int t=0; t < viewAnglesTheta; t++) {
435 for (int p=0; p < viewAnglesPhy; p++) {
436
437 float theta = t * thetaInc + thetaInc / 2.0f;
438 float phy = p * phyInc;
439 Vec3f wo = shading::computeLocalSphericalDirection(scene_rdl2::math::cos(theta), scene_rdl2::math::sin(theta), phy);
440 wo = test.frame.localToGlobal(wo);
441
442 printInfo("----- thetaWo - %f , phyWo - %f -----",
443 theta / sPi * 180.0f,
444 phy / sPi * 180.0f);
445
446 testPdfIntegral(test, wo, sampleCount);
447 testEvalIntegral(test, wo, sampleCount);
448 }
449 }
450#endif
