57template <typename Integer>
58constexpr Integer roundUpPower2(Integer v) noexcept
59{
60 static_assert(std::is_integral<Integer>::value, "Only works on integers");
61 static_assert(sizeof(Integer) == 8, "Only works on 64-bit integers");
62
63 --v;
64 v |= v >> 1;
65 v |= v >> 2;
66 v |= v >> 4;
67 v |= v >> 8;
68 v |= v >> 16;
69 v |= v >> 32;
70 ++v;
71 return v;
72}

371 explicit AlignedElementArray(std::size_t n)
372 : mSize(0)
373 , mData(static_cast<ElementType*>(doAllocate(n * sizeof(ElementType))))
374 {
375 try {
376 for ( ; mSize < n; ++mSize) {
377 new (getPointer(mSize)) T();
378 }
379 } catch (...) {

532private:
533 [[gnu::alloc_size(1)]] [[gnu::malloc]] static void* doAllocate(std::size_t bytes)
534 {
535 void* mem;
536 if (int error = posix_memalign(std::addressof(mem), kAlignment, bytes)) {
537 if (error == EINVAL) {
538 throw std::invalid_argument{"Bad alignment"};
539 } else {
540 throw std::bad_alloc{};
