31// accumulators are not run during displacement/render_prep
32// but neither is ispc, so the displacement argument is ignored
33intptr_t
34CPP_startOIIOAccumulator(shading::TLState *tls, uint32_t displacement)
35{
36 auto *exclAcc = getExclusiveAccumulators(tls);
37 if (exclAcc && exclAcc->push(EXCL_ACCUM_OIIO)) {
38 return (intptr_t)exclAcc;
39 }
40 return 0;
41}
42
43void
44CPP_stopOIIOAccumulator(intptr_t accumulator)
45{
46 if (accumulator) {
47 auto exclAcc = (mcrt_common::ExclusiveAccumulators *)accumulator;
48 MNRY_ASSERT(exclAcc->isRunning(EXCL_ACCUM_OIIO));
49 exclAcc->pop();
50 }
51}

53OIIO::TextureOpt::Wrap
54getOIIOWrap(WrapType wrapType) {
55 switch (wrapType) {
56 case WrapType::Black:
57 return OIIO::TextureOpt::WrapBlack;
58 case WrapType::Clamp:
59 return OIIO::TextureOpt::WrapClamp;
60 case WrapType::Periodic:
61 return OIIO::TextureOpt::WrapPeriodic;
62 case WrapType::Mirror:
63 return OIIO::TextureOpt::WrapMirror;
64 default:
65 return OIIO::TextureOpt::WrapDefault;
66 }

92 mTextureOpt[TrilinearAnisotropic].mipmode = OIIO::TextureOpt::MipModeAniso;
93 mTextureOpt[TrilinearAnisotropic].interpmode = OIIO::TextureOpt::InterpBilinear;

97 mTextureOpt[TrilinearIsotropic].interpmode = OIIO::TextureOpt::InterpBilinear;

104 mTextureOpt[ClosestMipClosestTexel].mipmode = OIIO::TextureOpt::MipModeOneLevel;

133 bool
134 update(const std::string &filename,
135 bool applyGamma,
136 WrapType wrapS,
137 WrapType wrapT,
138 bool useDefaultColor,
139 const scene_rdl2::math::Color& defaultColor,
140 const scene_rdl2::math::Color& fatalColor,
141 std::string &errorMsg)
142 {
143 init();
144
145 mIspc.mUseDefaultColor = useDefaultColor;
146 mIspc.mDefaultColor.r = defaultColor.r;
147 mIspc.mDefaultColor.g = defaultColor.g;
148 mIspc.mDefaultColor.b = defaultColor.b;
149 mIspc.mFatalColor.r = fatalColor.r;
150 mIspc.mFatalColor.g = fatalColor.g;
151 mIspc.mFatalColor.b = fatalColor.b;
152 mIspc.mIsValid = false;
153
154 mConvertFunc = fauxstd::make_unique<ispc::DWAConvertFunction>();
155 mIspc.mConvertFunction = mConvertFunc.get();
156
157 mTextureHandles.assign(1, nullptr);
158 mIspc.mTextureHandles = reinterpret_cast<intptr_t *>(&mTextureHandles[0]);
159 mActiveSubImages.assign(1, nullptr);
160 mIspc.mActiveSubImages = reinterpret_cast<const ispc::DWATiledSubImage **>(&mActiveSubImages[0]);
161
162 // load and prepare the texture...
163 std::string errorString;
164 texture::TextureSampler *textureSampler = texture::getTextureSampler();
165 texture::TextureHandle *handle = textureSampler->getHandle(filename, errorString);
166 if (handle && checkTextureWindow(textureSampler, handle, filename, errorMsg)) {
167 mTextureHandles[0] = handle;
168 mIspc.mIsValid = true;
169 } else {
170 std::ostringstream os;
171 os << "FATAL: failed to open texture file \"" <<
172 filename << "\" (" << errorString << ")";
173 errorMsg = os.str();
174 mIspc.mOIIOSampleFn = 0;
175 textureSampler->unregisterMapForInvalidation(mShader);
176 mIspc.mIsValid = false;
177 if (mIspc.mUseDefaultColor) {
178 return true;
179 } else {
180 return false;
181 }
182 }
183
184 // This will early out and do minimal work if the
185 // filename and 'this' pointer haven't changed.
186 //
187 // NOTE: This line was giving this exception after a reload:
188 // terminate called after throwing an instance of 'except::KeyError'
189 // what(): No Attribute named 'texture' on SceneClass 'ProjectTriplanarMap'.
190 textureSampler->registerMapForInvalidation(filename, mShader, false);
191
192 for (int i=0; i < QualityCount; ++i) {
193 mTextureOpt[i].swrap = getOIIOWrap(wrapS);
194 mTextureOpt[i].twrap = getOIIOWrap(wrapT);
195 mTextureOpt[i].apply_gamma = applyGamma;
196 mTextureOpt[i].dwa_tiledimage = nullptr;
197 mTextureOpt[i].subimagename.clear();
198 }
199
200 MNRY_ASSERT(mTextureHandles[0]);
201
202 mIspc.mOIIOSampleFn = 0;
203 mIspc.mOIIOSampleADFn = 0;
204 mIspc.mActiveSubImages[0] = nullptr;
205
206 OIIO::TextureSystem *textureSystem = textureSampler->getTextureSystem();
207
208 // retrieve resolution and pixel aspect ratio information. This
209 // info may be needed in certain contexts, see ProjectCameraMap_v2 shader
210 OIIO::ImageSpec spec;
211 OIIO::ustring ufilename(filename.c_str());
212 textureSystem->get_imagespec(ufilename, 0, spec);
213 mWidth = spec.width;
214 mHeight = spec.height;
215 mPixelAspectRatio = spec.get_float_attribute("PixelAspectRatio", 1.0f);
216 mIspc.mWidth = mWidth;
217 mIspc.mHeight = mHeight;
218 mIspc.mPixelAspectRatio = mPixelAspectRatio;
219 mIspc.mTextureOptions = (intptr_t) mTextureOpt;
220
221 if (textureSystem->dwa_condition_options(getTextureHandle(), &mTextureOpt[TrilinearAnisotropic]) &&
222 textureSystem->dwa_condition_options(getTextureHandle(), &mTextureOpt[TrilinearIsotropic]) &&
223 textureSystem->dwa_condition_options(getTextureHandle(), &mTextureOpt[LinearMipClosestTexel]) &&
224 textureSystem->dwa_condition_options(getTextureHandle(), &mTextureOpt[ClosestMipClosestTexel])) {
225
226 mTiledImages.emplace_back(
227 fauxstd::make_unique<texture::TiledImage>()
228 );
229 texture::TiledImage *tiledImage = mTiledImages.back().get();
230
231 OIIO::DwaPrepareResult result = textureSystem->dwa_prepare_texture(
232 *getTextureHandle(),
233 mTextureOpt[0],
234 tiledImage,
235 reinterpret_cast<const DWATiledSubImage **>(&mIspc.mActiveSubImages[0])
236 );
237
238 if (result == OIIO::DwaPrepareResult::DwaSuccess) {
239 // Success!
240 mIspc.mOIIOSampleFn = (intptr_t) textureSystem->dwa_get_vectorized_texture_callback();
241 mIspc.mOIIOSampleADFn = (intptr_t) textureSystem->dwa_get_vectorized_texture_AD_callback();
242
243 for (int i=0; i < QualityCount; ++i) {
244 mTextureOpt[i].dwa_tiledimage = tiledImage;
245 }
246
247 if (mIspc.mActiveSubImages[0]) {
248 dwa_get_conversion_functions(
249 (DWAConvertFunction*)mIspc.mConvertFunction,
250 (reinterpret_cast<const DWATiledSubImage*>(mIspc.mActiveSubImages[0]))->nchannels,
251 (reinterpret_cast<const DWATiledSubImage*>(mIspc.mActiveSubImages[0]))->format,
252 (reinterpret_cast<const DWATiledSubImage*>(mIspc.mActiveSubImages[0]))->apply_gamma
253 );
254 }
255 mIspc.mIsValid = true;
256 } else {
257 errorMsg = processTexturePrepareResult(result);
258 mIspc.mIsValid = false;
259 }
260 }
261
262 if (mIspc.mUseDefaultColor) {
263 return true;
264 } else {
265 return mIspc.mIsValid;
266 }
267 }

269 scene_rdl2::math::Color4
270 sample(shading::TLState *tls,
271 const shading::State& state,
272 scene_rdl2::math::Vec2f st,
273 float *derivatives) const
274 {
275 if (!mIspc.mIsValid && mIspc.mUseDefaultColor) {
276 return scene_rdl2::math::Color4(scene_rdl2::math::asCpp(mIspc.mDefaultColor).r,
277 scene_rdl2::math::asCpp(mIspc.mDefaultColor).g,
278 scene_rdl2::math::asCpp(mIspc.mDefaultColor).b,
279 1.0f);
280 }
281
282 const int index = getTextureOptionIndex(state.isDisplacement(), state);
283
284 const texture::TextureOptions& options = mTextureOpt[index];
285
286 scene_rdl2::math::Color4 result;
287
288 OIIO::TextureSystem *texSys = MNRY_VERIFY(tls->mTextureSystem);
289
290 // dwa_texture *must* be given 4 floats for the result.
291 ALIGN(16) float tmp[4];
292 bool res;
293
294 if (state.isDisplacement()) {
295 // For displacement we are filtering over much larger areas than for
296 // regular shading. This causes artifacts with the dwaproduction texture
297 // system's LinearMipClosestTexel filtering method, which is not good
298 // for blurring a texture significantly. We fall-back on the sony texture
299 // system in this case.
300 // TODO: remove this fallback once dwaproduction supports TrilinearIsotropic
301 res = texSys->texture(
302 const_cast<texture::TextureHandle *>(getTextureHandle()),
303 tls->mOIIOThreadData,
304 options,
305 st[0], st[1],
306 derivatives[0], derivatives[1],
307 derivatives[2], derivatives[3],
308 4, tmp
309 );
310 } else {
311 float mipSelector = computeMipSelector(
312 derivatives[0], derivatives[1],
313 derivatives[2], derivatives[3]
314 );
315
316 // accumulators are not run during displacement/render_prep
317 EXCL_ACCUMULATOR_PROFILE(tls, EXCL_ACCUM_OIIO);
318 res = texSys->dwa_texture(
319 getTextureHandle(),
320 tls->mOIIOThreadData,
321 options,
322 reinterpret_cast<const DWAConvertFunction*>(mIspc.mConvertFunction),
323 &st.x,
324 mipSelector,
325 derivatives,
326 tmp
327 );
328 }
329
330 if (res) {
331 result[0] = tmp[0];
332 result[1] = tmp[1];
333 result[2] = tmp[2];
334 result[3] = tmp[3];
335 } else {
336 logEvent(tls, mIspc.mBasicTextureStaticDataPtr->sErrorSampleFail);
337 result[0] = result[1] = result[2] = result[3] = 0.f;
338 }
339
340 return result;
341 }

459void CPP_oiioTexture(const ispc::BASIC_TEXTURE_Data *tx,
460 shading::TLState *tls,
461 const uint32_t displacement,
462 const int pathType,
463 const float *derivatives,
464 const float *st,
465 float *result)
466{
467 if (!tx->mIsValid && tx->mUseDefaultColor) {
468 result[0] = tx->mDefaultColor.r;
469 result[1] = tx->mDefaultColor.g;
470 result[2] = tx->mDefaultColor.b;
471 result[3] = 1.0f;
472 return;
473 }
474
475 scene_rdl2::rdl2::Shader *shader = reinterpret_cast<scene_rdl2::rdl2::Shader *>(tx->mShader);
476 int threadIndex = mcrt_common::getThreadIdx(tls);
477 texture::TextureHandle *textureHandle = (reinterpret_cast<texture::TextureHandle **>(tx->mTextureHandles))[0];
478 texture::TLState::Perthread *threadInfo = tls->mOIIOThreadData;
479 OIIO::TextureSystem *texSys = MNRY_VERIFY(tls->mTextureSystem);
480 const int index = getTextureOptionIndex(displacement != 0,
481 static_cast<shading::Intersection::PathType>(pathType));
482 texture::TextureOptions *options = (reinterpret_cast<texture::TextureOptions *>(tx->mTextureOptions));
483
484 float s = st[0];
485 float t = st[1];
486 float dsdx = derivatives[0];
487 float dtdx = derivatives[1];
488 float dsdy = derivatives[2];
489 float dtdy = derivatives[3];
490 const int nChannels = 4;
491
492 bool res = texSys->texture(textureHandle,
493 threadInfo,
494 options[index],
495 s, t,
496 dsdx, dsdy, dtdx, dtdy,
497 nChannels,
498 result);
499
500 if (!res) {
501 shader->getThreadLocalObjectState()[threadIndex].mLogs.log(
502 tx->mBasicTextureStaticDataPtr->sErrorSampleFail);
503 result[0] = result[1] = result[2] = result[3] = 0.f;
504 }
505}
