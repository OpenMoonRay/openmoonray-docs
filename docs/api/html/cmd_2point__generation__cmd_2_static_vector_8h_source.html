346 iterator erase(const_iterator first, const_iterator last)
347 {
348 iterator pfirst(first - cbegin() + begin());
349 if (first >= last) {
350 return pfirst;
351 }

353 auto it = std::move(plast, end(), pfirst);
354 for ( ; it != end(); ++it) {
355 it->~T();

496 using AlignedStorage = typename std::aligned_storage<sizeof(T), std::alignment_of<T>::value>::type;
497 union
498 {
