134 // position map is the fundamental structure that maps
135 // uv coordinates on the geomety to 3D camera space positions.
136 // The nrm map maps uv coordinates to normals in camera space.
137 // This map does not take material normal mapping into account, but
138 // does respect displacement and explicitly provided mesh normals.
139 //
140 // Not all uv coordinates exist on a piece of geometry. For any
141 // pixel p, mPosMap[p].w is used to determine if geometry exists
142 // at the corresponding uv coordinate. If non-zero, geometry
143 // exists, if 0, then geometry does not exist. For uv coordinates
144 // where geometry does not exist, an invalid ray is generated, and
145 // the integrator is expected to handle this.
146 scene_rdl2::math::Vec3fa *mPosMap;
147 scene_rdl2::math::Vec3f *mNrmMap;
148
149 // position map width and height - by default, equal to the
150 // dimenions of the map we are creating. but this can be adjusted
151 // with the "map factor" attribute.
152 unsigned int mPosMapWidth;
153 unsigned int mPosMapHeight;
