26/**
27 * The LightSet represents a collection of Lights with no duplicates. It's
28 * used in Part/Material assignments in the Layer, where you can define which
29 * Lights affect a particular Part/Material assignment. That collection of
30 * Lights is the LightSet. LightSets can be reused for many Part/Material
31 * assignments.
32 *
33 * It only has one attribute, named "lights", which is a SceneObjectVector.
34 * We provide convenience functions on the LightSet to add and remove Lights
35 * from the set, as well as check if a Light is contained in the set. Please use these
36 * functions to maintain the light's uniqueness invariant. When setting "lights"
37 * directly, make sure the lights are unique and their order is deterministic.
38 *
39 * You can get the Lights as a const SceneObjectVector& with the getLights() method call.
40 */
41class LightSet : public SceneObject
42{
43public:
44 typedef SceneObject Parent;
45
46 LightSet(const SceneClass& sceneClass, const std::string& name);
47 static SceneObjectInterface declare(SceneClass& sceneClass);
48
49 /// Retrieves the set of unique Lights in this LightSet.
50 finline const SceneObjectVector& getLights() const;
51
52 /**
53 * Adds the given Light to the LightSet, if it is not already a member of
54 * the set. If it is already a member of the set, this does nothing.
55 *
56 * @param light The Light to add to the LightSet.
57 */
58 void add(Light* light);
59
60 /**
61 * Removes the given Light from the LightSet, if it is already a member of
62 * the set. If it is not a member of the set, this does nothing.
63 *
64 * @param light The Light to remove from the LightSet.
65 */
66 void remove(Light* light);
67
68 /**
69 * Returns true if the given Light is a member of the LightSet. There's
70 * no need to call this before calling add() or remove(), as they will
71 * gracefully handle those edge cases.
72 *
73 * @param light The Light to check for membership.
74 * @return True if the light is a member of the LightSet.
75 */
76 bool contains(const Light* light) const;
77
78 /**
79 * Check if any of the following that are modified: Lights
80 * in the LightSet, SceneObject attributes, and the LightSet itself.
81 * Should only be called after all UpdateGuards.
82 *
83 * @return True if the light list changed
84 */
85 bool updatePrepLight(UpdateHelper& sceneObjects, size_t depth);
86
87 /**
88 * Alphabetizes light list by name
89 */
90 void update() override;
91
92 /// Completely empties the LightSet so that it doesn't contain anything.
93 void clear();
94
95protected:
96 static AttributeKey<SceneObjectVector> sLightsKey;
97};
