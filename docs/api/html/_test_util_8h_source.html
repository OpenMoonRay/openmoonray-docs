198finline float
199computeError(float v1, float v2)
200{
201 MNRY_ASSERT(v1 >= 0.0f);
202 MNRY_ASSERT(v2 >= 0.0f);
203
204 float max = scene_rdl2::math::max(v1, v2);
205 if (max > 1.0f) {
206 return scene_rdl2::math::abs(v2 - v1) / max;
207 } else {
208 return scene_rdl2::math::abs(v2 - v1);
209 }
210}

215finline void
216generate2DHaltonSequence(uint32_t size, const uint32_t scramble[2],
217 FloatArray &u, FloatArray &v)
218{
219 PixelScramble ps(scramble[1]);
220 const std::uint32_t base3Permutations[3] = { ps.permuteIndex(0, 3), ps.permuteIndex(1, 3), ps.permuteIndex(2, 3) };
221
222 u.resize(size);
223 v.resize(size);
224
225 for (uint32_t i=0; i < size; i++) {
226 u[i] = vanDerCorputSequence(i, scramble[0]);
227 v[i] = permutedRadicalInverse(i, 3, base3Permutations);
228 }
229}

232finline void
233generate2DQmcSequence(uint32_t size, const uint32_t scramble[2],
234 FloatArray &u, FloatArray &v)
235{
236 u.resize(size);
237 v.resize(size);
238
239 for (uint32_t i=0; i < size; i++) {
240 float uv[2];
241 sampleQmc2D(i, scramble, uv);
242 u[i] = uv[0];
243 v[i] = uv[1];
244 }
245}

269finline int
270savePaMatFile(const std::string &filename, const FloatArray &u, const FloatArray &v)
271{
272 char *name = FX_new_name(filename.c_str(), ".mat");
273 if (FX_write_test(name, true) == NULL) {
274 // #### Error reporting!!!
275 PDI_error("Cannot write to matrix file '%s'\n", name);
276 return -1;
277 }
278 MAT_FILE *fd = MAT_FILE_open(name, "w");
279 if (fd == NULL) {
280 // #### Error reporting!!!
281 MAT_error("Cannot open matrix file '%s'\n", name);
282 return -1;
283 }
284
285 int particleCount = static_cast<int>(u.size());
286
287 // Write 3 attributes at a time
288 int paAttribute[3];
289 MAT *mat[3];
290 mat[0] = MAT_create(1, particleCount);
291 mat[1] = MAT_create(1, particleCount);
292 mat[2] = MAT_create(1, particleCount);
293 if (mat[0] == NULL || mat[1] == NULL || mat[2] == NULL) {
294 MAT_error("Cannot create matrix");
295 return -1;
296 }
297
298
299 // Position
300 paAttribute[0] = PA_NAME_POSITION_X;
301 paAttribute[1] = PA_NAME_POSITION_Y;
302 paAttribute[2] = PA_NAME_POSITION_Z;
303 for (int i=0; i < particleCount; i++) {
304 Vec3d position(u[i], v[i], 0.0);
305 mat[0]->data[i] = position[0];
306 mat[1]->data[i] = position[1];
307 mat[2]->data[i] = position[2];
308 }
309 MAT_set_name(mat[0], PA_get_attribute_name(paAttribute[0]));
310 MAT_set_name(mat[1], PA_get_attribute_name(paAttribute[1]));
311 MAT_set_name(mat[2], PA_get_attribute_name(paAttribute[2]));
312 if (!MAT_write(fd, mat[0]) ||
313 !MAT_write(fd, mat[1]) ||
314 !MAT_write(fd, mat[2])) {
315 MAT_error("Cannot write matrix");
316 return -1;
317 }
318
319
320 MAT_destroy(mat[0]);
321 MAT_destroy(mat[1]);
322 MAT_destroy(mat[2]);
323
324 MAT_FILE_close(fd);
325
326 return particleCount;
327}

338// Test if the lengths of two vectors are within a certain percentage of each other.
339template<typename T> finline bool
340isEqualLength(const T &a, const T &b, float eps = float(scene_rdl2::math::sEpsilon))
341{
342 return scene_rdl2::math::isEqual<float>(a.length(), b.length(), eps);

345// Test if the angle between two vectors is within a set tolerance.
346template<typename T> finline bool
347isEqualDirection(const T &a, const T &b, float toleranceInDegrees)
348{
349 return dot(normalize(a), normalize(b)) > scene_rdl2::math::cos(scene_rdl2::math::deg2rad(toleranceInDegrees));
350}
351
352// Test if uvs match. Uvs are assumed wrapped so for example, 0 == 1 == -1 == 2
353// and so on.
354finline bool
355isEqualWrappedUv(const scene_rdl2::math::Vec2f &a, const scene_rdl2::math::Vec2f &b, float eps =
356 float(scene_rdl2::math::sEpsilon))
357{
358 scene_rdl2::math::Vec2f diff = a - b;
359 float diffU = scene_rdl2::math::fmod(diff.x, 1.f);
360 float diffV = scene_rdl2::math::fmod(diff.y, 1.f);
361
362 return (scene_rdl2::math::isZero(diffU, eps) || scene_rdl2::math::isOne(diffU, eps) ||
363 scene_rdl2::math::isEqual(diffU, -1.f, eps)) &&
364 (scene_rdl2::math::isZero(diffV, eps) || scene_rdl2::math::isOne(diffV, eps) ||
365 scene_rdl2::math::isEqual(diffV, -1.f, eps));
366}

572// Wrapper around tbb::parallel_reduce with common plumbing code for iterating
573// over a unit square domain. By systematically iterating over the domain we can
574// get a more accurate estimate of the integral than just by picking random
575// numbers. Fortunately we don't care about aliasing issues here.
576template<typename T, typename FUNC>
577finline T
578doReductionOverUnitSquare(const T &identity, int samplesPerAxis,
579 int grainSizePerAxis, FUNC func)
580{
581 // turn an x, y coordinate into a uniform distributed number within [0, 1)^2
582 float scl, ofs;
583 scene_rdl2::math::getScaleOffset<float>(0, samplesPerAxis - 1, 0.f, 0.99999f, &scl, &ofs);
584
585 return tbb::parallel_reduce(
586
587 // blocked range
588 tbb::blocked_range2d<unsigned>(0u, samplesPerAxis, grainSizePerAxis,
589 0u, samplesPerAxis, grainSizePerAxis),
590
591 // identity
592 identity,
593
594 // body
595 [&](const tbb::blocked_range2d<unsigned> &range, const T &current) -> T
596 {
597 return func(range, current, scl, ofs);
598 },
599
600 // reduction
601 [](const T &a, const T &b)
602 {
603 return a + b;
604 }
605 );
606}

619
620finline void
621generateGGXNormals(float aRandomness, uint32_t size, Vec3Array& normals)
622{
623 constexpr std::uint32_t seed = 0xdaedfeeb;
624 scene_rdl2::util::Random rand(seed);
625
626 normals.resize(size);
627 for (size_t i = 0; i < size; ++i) {
628 float r1 = rand.getNextFloat();
629 float r2 = rand.getNextFloat();
630
631 float theta = scene_rdl2::math::atan(aRandomness*scene_rdl2::math::sqrt(r1) *
632 scene_rdl2::math::rsqrt(1.0f - r1)); // GGX
633 float phi = 2.0f * scene_rdl2::math::sPi * r2;
634
635 float sinTheta, cosTheta;
636 float sinPhi, cosPhi;
637
638 scene_rdl2::math::sincos(theta, &sinTheta, &cosTheta);
639 scene_rdl2::math::sincos(phi, &sinPhi, &cosPhi);
640
641 scene_rdl2::math::Vec3f wn;
642 wn.x = cosPhi * sinTheta;
643 wn.y = sinPhi * sinTheta;
644 wn.z = cosTheta;
645
646 normals[i] = wn;
647 }
648}
