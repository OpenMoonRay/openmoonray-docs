35static void
36render(const scene_rdl2::rdl2::Material *mat, int testNum,
37 int width = 256, int height = 256,
38 int raysPerPixel = 1,
39 bool primeThePump = false)
40{
41 // allocate results buffer and call the test renderer
42 const int bufferSize = 3 * width * height;
43 uint8_t results[bufferSize];
44
45 double start = time::getTime();
46 uint64 shaderTicks = TestRender::render(mat, results, width, height, raysPerPixel, primeThePump);
47 double end = time::getTime();
48 std::cerr << "render test " << testNum << " render time = "
49 << end - start << '\n';
50 std::cerr << "render test " << testNum << " shader ticks = "
51 << shaderTicks << '\n';
52
53 // write the results as a ppm
54 {
55 std::ofstream ppm;
56 std::stringstream filename;
57 filename << "TestRender_result" << testNum << ".ppm";
58 ppm.open(filename.str().c_str());
59 ppm << "P6\n";
60 ppm << width << " " << height << std::endl;
61 const int maxPixel = 255; // 8 bit gamma corrected color
62 ppm << maxPixel << std::endl;
63 ppm.write((const char *) results, bufferSize * sizeof(int8_t));
64 ppm.close();
65 }
66
67 // compare with the canonical
68 std::ifstream ppm;
69 uint8_t canonical[bufferSize];
70 std::stringstream filename;
71 filename << "ref/TestRender_canonical" << testNum << ".ppm";
72 ppm.open(filename.str().c_str());
73 ppm.getline((char *) canonical, bufferSize); // P6\0
74 ppm.getline((char *) canonical, bufferSize); // width height
75 ppm.getline((char *) canonical, bufferSize); // max white
76 ppm.read((char *) canonical, bufferSize);
77 // allow +/-1 code of tolerance
78 uint8_t *resPtr = results;
79 uint8_t *canPtr = canonical;
80 for (int i = 0; i < bufferSize; ++i) {
81 CPPUNIT_ASSERT(abs(*resPtr - *canPtr) <= 1);
82 resPtr++;
83 canPtr++;
84 }
85}
