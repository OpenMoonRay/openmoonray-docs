30//===--------------------------------------------------------------------------
31// SD-Tree based path guiding
32// See "Practical Path Guiding for Efficient Light-Transport Simulation"
33// Thomas Muller, Markus Gross, Jan Novak
34// Proceedings of EGSR 2017, vol. 36, no.4
35//
36// The paper goes to great lengths describing the optimal number of samples
37// to choose for each pass and the optimal ratio of learning to
38// rendering. This implementation pretty much ignores those apsects
39// of the paper. The implementation just expects render driver code to
40// divide rendering up into a set of passes where the number of samples
41// roughly double at every pass. Learning takes place during every pass and
42// sampling is done based on the learning from the previous pass.
43//===--------------------------------------------------------------------------

96// uniformly maps a 3D direction to a 2D planar position
97// in [0,1]x[0,1].
98static Vec2f
99dirToPos(const Vec3f &dir)
100{
101 const float cosTheta = std::min(std::max(dir.z, -1.0f), 1.0f);
102
103 // computing phi using double precision is needed and intentional
104 float phi = std::atan2(dir.y, dir.x);
105 while (phi < 0) {
106 phi += static_cast<double>(scene_rdl2::math::two_pi);
107 }
108
109 return Vec2f((cosTheta + 1.f) * 0.5f, phi * scene_rdl2::math::sOneOverTwoPi);
110}
111
112// maps a 2D planar position [0, 1]x[0, 1] to a 3D direction
113static Vec3f
114posToDir(const Vec2f &pos)
115{
116 const float cosTheta = 2 * pos.x - 1;
117 const float phi = scene_rdl2::math::sTwoPi * pos.y;
118
119 const float sinTheta = scene_rdl2::math::sqrt(1 - cosTheta * cosTheta);
120 float sinPhi, cosPhi;
121 scene_rdl2::math::sincos(phi, &sinPhi, &cosPhi);
122
123 return Vec3f(sinTheta * cosPhi, sinTheta * sinPhi, cosTheta);
124}

126// returns quadrant number (0, 1, 2, or 3) of position pos, and remaps
127// pos such that it is normalized to that quadrant
128static uint32_t
129getChildIndexAndRemap(Vec2f &pos)
130{
131 MNRY_ASSERT(pos.x >= 0.0f && pos.x <= 1.0f && pos.y >= 0.0f && pos.y <= 1.0f);
132 uint32_t childIndex = 0;
133 if (pos.x < 0.5f) {
134 // map x [0, .5] to [0, 1]
135 pos.x *= 2;
136 } else {
137 // map x [.5, 1] to [0, 1]
138 pos.x = (pos.x - 0.5f) * 2;
139 childIndex |= 1;
140 }
141
142 if (pos.y < 0.5f) {
143 // map y [0, .5] to [0, 1]
144 pos.y *= 2;
145 } else {
146 // map y [.5, 1] to [0, 1]
147 pos.y = (pos.y - 0.5f) * 2;
148 childIndex |= 2;
149 }
150
151 return childIndex;
152}

222void
223DirTree::Tree::recordRadiance(const Vec2f &pos, const Color &radiance)
224{
225 MNRY_ASSERT(scene_rdl2::math::isFinite(radiance));
226 if (!scene_rdl2::math::isFinite(radiance)) return;
227
228 const float luminance = scene_rdl2::math::luminance(radiance);
229 mNumSamples.fetch_add(1, std::memory_order_acq_rel);
230 mSum.fetch_add(luminance, std::memory_order_acq_rel);
231
232 // update nodes
233 uint64_t index = 0;
234 Vec2f curPos = pos;
235 do {
236 Node &node = mNodes[index];
237
238 // which quad does this direction belong to? update the mean
239 // for that quad.
240 uint32_t childIndex = getChildIndexAndRemap(curPos);
241
242 // We're make sure that we're consistent with other accesses to mMean, even though Muller's reference
243 // implementation uses relaxed atomics everywhere.
244 node.mMean[childIndex].fetch_add(luminance, std::memory_order_acq_rel);
245
246 // descend into the quad if it is not a leaf.
247 // by construction, if a quad is a leaf, the child index for that quad is 0
248 index = node.mChildren[childIndex];
249 } while (index != 0);
250}

280void
281DirTree::reset(int maxDepth, float threshold)
282{
283 // This method is called during a passReset() after a call to build(). It
284 // computes the topology of a new mBuild to a refined (or coarser)
285 // version of mSample
286
287 // First, initialize mBuild with a single root node
288 mBuild.mNodes.clear();
289 mBuild.mNodes.resize(1);
290 mBuild.mMaxDepth = 0;
291 mBuild.mNumSamples.store(0, std::memory_order_relaxed); // No dependencies
292 mBuild.mSum.store(0.f, std::memory_order_relaxed); // No dependencies
293
294 // Build up mBuild as a refined version of mSample. mSample is traversed
295 // in depth first order and is refined if enough total energy exists
296 // (based on threshold) in a node and we are within the maxDepth parameter
297 struct StackItem {
298 uint64_t mBuildIndex; // index of node we are building (in mBuild)
299 uint64_t mSourceIndex; // index of node we may subdivide
300 Tree *mSourceTree; // tree that source node belongs to (could be sample or build)
301 int mDepth; // depth of node
302 };
303
304 std::stack<StackItem> todo;
305 todo.push({ 0, 0, &mSample, 1 });
306
307 // total energy in mSample tree, we'll subdivide when any node fraction of this
308 // value exceeds threshold
309 const float nodeThreshold = mSample.mSum.load(std::memory_order_acquire) * threshold;
310
311 while (!todo.empty()) {
312 StackItem sItem = todo.top();
313 todo.pop();
314
315 mBuild.mMaxDepth = std::max(mBuild.mMaxDepth, sItem.mDepth);
316 Tree::Node &sourceNode = sItem.mSourceTree->mNodes[sItem.mSourceIndex];
317
318 for (int i = 0; i < 4; ++i) {
319 // We'll acquire here to sync with any previously run threads.
320 const float sourceNodeMean = sourceNode.mMean[i].load(std::memory_order_acquire);
321 if (sItem.mDepth < maxDepth && sourceNodeMean > nodeThreshold) {
322 // if this quadrant is not a leaf in the source node, push the child
323 // onto the stack. note that the only case where the source node
324 // cannot be a leaf is if the source node comes from the sample tree.
325 if (sourceNode.mChildren[i] != 0) {
326 // not a leaf.
327 MNRY_ASSERT(sItem.mSourceTree == &mSample);
328 todo.push({ mBuild.mNodes.size(), sourceNode.mChildren[i], sItem.mSourceTree, sItem.mDepth + 1 });
329 } else {
330 // is a leaf. could be from mBuild or mSample
331 todo.push({ mBuild.mNodes.size(), mBuild.mNodes.size(), &mBuild, sItem.mDepth + 1 });
332 }
333
334 mBuild.mNodes[sItem.mBuildIndex].mChildren[i] = static_cast<uint32_t>(mBuild.mNodes.size());
335 mBuild.mNodes.emplace_back(); // add node
336 // spread the energy evenly among the new node's quadrants
337 const float newMean = sourceNodeMean * 0.25f;
338 Tree::Node &newNode = mBuild.mNodes.back();
339 for (int j = 0; j < 4; ++j) {
340 // We're creating new nodes: we have no dependencies: relaxed ordering
341 newNode.mMean[j].store(newMean, std::memory_order_relaxed);
342 }
343
344 if (mBuild.mNodes.size() > std::numeric_limits<uint32_t>::max()) {
345 scene_rdl2::Logger::warn("DirTree hit max children count.");
346 todo = std::stack<StackItem>();
347 break;
348 }
349
350 }
351 }
352 }
353
354 // now set all the energy to 0
355 for (Tree::Node &node : mBuild.mNodes) {
356 // No dependencies: relaxed ordering
357 node.mMean[0].store(0.0f, std::memory_order_relaxed);
358 node.mMean[1].store(0.0f, std::memory_order_relaxed);
359 node.mMean[2].store(0.0f, std::memory_order_relaxed);
360 node.mMean[3].store(0.0f, std::memory_order_relaxed);
361 }
362
363 // I experimented with mBuild.mNodes.shrink_to_fit(). But this
364 // actually caused the renderer to use more memory in my test cases.
365}

378float
379DirTree::getPdf(const Vec3f &dir) const
380{
381 // Use the sample tree
382
383 // First check for an empty (or nearly empty) tree
384 // in which case, we'll just use spherical sampling
385 const auto numSamples = mSample.mNumSamples.load(std::memory_order_acquire);
386 const auto sum = mSample.mSum.load(std::memory_order_acquire);
387 if (numSamples == 0 || (sum / (scene_rdl2::math::sFourPi * numSamples)) == 0.0f) {
388 return scene_rdl2::math::sOneOverFourPi;
389 }
390
391 Vec2f pos = dirToPos(dir);
392
393 // recurse into the nodes
394 float pdf = scene_rdl2::math::sOneOverFourPi;
395 uint64_t index = 0; // start at the root
396 do {
397 // which quadrant?
398 const uint32_t i = getChildIndexAndRemap(pos); // 0, 1, 2, or 3
399
400 if (mSample.mNodes[index].mMean[i].load(std::memory_order_acquire) <= 0.f) {
401 return 0; // invalid pdf
402 }
403
404 float total = 0.f;
405 for (uint32_t j = 0; j < 4; ++j) {
406 total += mSample.mNodes[index].mMean[j].load(std::memory_order_acquire);
407 }
408 // Why the 4.0f?
409 // The domain is the unit square. Broken into
410 // four quadrants - each with area of 1/4.
411 // mean[0]/total + mean[1]/total + mean[2]/total + mean[3]/total = 1.
412 // A B C D
413 // Total integral value is A/4 + B/4 + C/4 + D/4 = (A + B + C + D)/4.
414 pdf *= (4.0f * mSample.mNodes[index].mMean[i].load(std::memory_order_acquire) / total);
415
416 index = mSample.mNodes[index].mChildren[i];
417 } while (index != 0); // until we hit a leaf
418
419 return pdf;
420}

422static inline bool
423checkForZero(float val, int index, const Vec2f &rpos, Vec2f &pos)
424{
425 // It is possible that some denominators in sampleDirection can reach exactly zero, even though they should
426 // theoretically be greater than 0. In this case we need to break from the recursion. If this is our first time
427 // through in the recursion (index == 0), then we need to set the quad position to some reasonable default. (we'll
428 // just use some random position). We return true if the recursion needs to be broken, false otherwise.
429 if (val == 0.f) {
430 if (index == 0) {
431 pos = rpos;
432 }
433 return true;
434 }
435
436 return false;
437}

439Vec3f
440DirTree::sampleDirection(float r1, float r2) const
441{
442 // Use the sample tree
443
444 // in some cases we may return just a random direction
445 const Vec2f rpos(r1, r2); // random quad position
446
447 // First check for an empty tree
448 // in which case, we'll just use spherical sampling
449 const auto numSamples = mSample.mNumSamples.load(std::memory_order_acquire);
450 const auto sum = mSample.mSum.load(std::memory_order_acquire);
451 if (numSamples == 0 || (sum / (scene_rdl2::math::sFourPi * numSamples)) == 0.0f) {
452 return posToDir(rpos);
453 }
454
455 // recurse into the sample nodes
456 // Pick quadtree children based on their mean value, compared
457 // to the random sample values.
458 uint64_t index = 0; // start at the root
459 Vec2f pos = Vec2f(0.f, 0.f);
460 Vec2f sample = rpos;
461 float scale = 1.0f; // halved with each recursion
462 do {
463 const Tree::Node &node = mSample.mNodes[index];
464
465 // compute a quadrant, as well as a location in
466 // the quadrant
467 int quadrant = 0; // quadrant in node 0, 1, 2, or 3
468 Vec2f quadrantOrigin(0.f, 0.f); // [0 or .5, 0 or .5]
469
470 const float topLeft = node.mMean[0].load(std::memory_order_acquire);
471 const float topRight = node.mMean[1].load(std::memory_order_acquire);
472 const float botLeft = node.mMean[2].load(std::memory_order_acquire);
473 const float botRight = node.mMean[3].load(std::memory_order_acquire);
474 const float total = topLeft + topRight + botLeft + botRight;
475
476 if (checkForZero(total, index, rpos, pos)) break;
477
478 // first the x-axis, re-normalizing sample
479 // as needed
480 float partial = topLeft + botLeft;
481 float boundary = partial / total;
482
483 if (sample.x < boundary) {
484 if (checkForZero(boundary, index, rpos, pos)) break;
485 sample.x /= boundary;
486 if (checkForZero(partial, index, rpos, pos)) break;
487 boundary = topLeft / partial;
488 } else {
489 partial = total - partial;
490 quadrantOrigin.x = 0.5f;
491 if (checkForZero(1.0f - boundary, index, rpos, pos)) break;
492 sample.x = (sample.x - boundary) / (1.0f - boundary);
493 if (checkForZero(partial, index, rpos, pos)) break;
494 boundary = topRight / partial;
495 quadrant |= 1;
496 }
497
498 // now split the y axis
499 if (sample.y < boundary) {
500 if (checkForZero(boundary, index, rpos, pos)) break;
501 sample.y /= boundary;
502 } else {
503 quadrantOrigin.y = 0.5f;
504 if (checkForZero(1.0f - boundary, index, rpos, pos)) break;
505 sample.y = (sample.y - boundary) / (1.0f - boundary);
506 quadrant |= 2;
507 }
508
509 if (node.mChildren[quadrant] == 0) {
510 // we hit a leaf, we are at the end of the recursion
511 pos += scale * (quadrantOrigin + 0.5f * sample);
512 } else {
513 // add in o and continue the recursion,
514 pos += scale * quadrantOrigin;
515 scale *= 0.5f;
516 }
517
518 index = node.mChildren[quadrant];
519 } while (index != 0); // until we hit a leaf
520
521 return posToDir(pos);
522}

571void
572SpatialTree::refine(uint64_t threshold)
573{
574 // TODO: add support for max memory parameter which will
575 // cause an early return from this method if the predicted
576 // max memory will be exceeded. For now, we work under the
577 // assumption of unlimited memory.
578
579 // do a depth first traversal of the tree, split nodes as we go
580 // if the number of radiance updates to that node exceeds the threshold
581 std::stack<uint64_t> todo;
582 todo.push(0); // root node
583 while (!todo.empty()) {
584 uint64_t index = todo.top();
585 todo.pop();
586
587 // if this is a leaf node, we might need to subdivide it
588 if (mNodes[index].mIsLeaf) {
589 // TODO: we may want to place some sort of limit on the
590 // total number of nodes in the tree. If we exceed that
591 // limit we'll disable splitting
592 if (mNodes[index].mDirTree.getNumSamplesBuild() > threshold) {
593 // add two child nodes - alternate split x/y/z axis
594 // each child inherits a copy of the parent's dirtree
595 // but is assigned half the samples
596 mNodes.resize(mNodes.size() + 2);
597 for (size_t i = mNodes.size() - 2; i < mNodes.size(); ++i) {
598 Node &n = mNodes[i];
599 n.mDirTree = mNodes[index].mDirTree;
600 n.mDirTree.setNumSamplesBuild(n.mDirTree.getNumSamplesBuild() / 2);
601 n.mAxis = (mNodes[index].mAxis + 1) % 3;
602 }
603
604 mNodes[index].mDirTree = {};
605 mNodes[index].mIsLeaf = false;
606 mNodes[index].mChildren[0] = mNodes.size() - 2;
607 mNodes[index].mChildren[1] = mNodes.size() - 1;
608 }
609 }
610
611 if (!mNodes[index].mIsLeaf) {
612 // push its two children onto the stack
613 todo.push(mNodes[index].mChildren[0]);
614 todo.push(mNodes[index].mChildren[1]);
615 }
616 }
617
618 // I experiemented with mNodes.shrink_to_fit(). But this
619 // actually caused the renderer to use more memory in my test cases.
620}

648DirTree *
649SpatialTree::getDirTree(const Vec3f &p)
650{
651 // normalize all position look ups to maximize precision
652 Vec3f np = (p - mBbox.lower) / mBbox.size();
653
654 uint64_t index = 0; // start at root
655 while (!mNodes[index].mIsLeaf) {
656 if (np[mNodes[index].mAxis] < 0.5f) {
657 // descend into the 0 child, expand the split
658 // axis so [0, 0.5] -> [0, 1]
659 np[mNodes[index].mAxis] *= 2.f;
660 index = mNodes[index].mChildren[0];
661 } else {
662 // descend into the 1 child, expand the
663 // split axis so [0.5, 1.0] -> [0, 1]
664 np[mNodes[index].mAxis] = 2.0f * (np[mNodes[index].mAxis] - 0.5f);
665 index = mNodes[index].mChildren[1];
666 }
667 }
668
669 MNRY_ASSERT(index < mNodes.size());

718void
719PathGuide::Impl::startFrame(const BBox3f &bbox, const scene_rdl2::rdl2::SceneVariables &vars)
720{
721 mSpatialTree.reset();
722 mEnable = vars.get(scene_rdl2::rdl2::SceneVariables::sPathGuideEnable);
723 if (!mEnable) return;
724
725 // these could become user settings via rdl scene variables
726 // so far, these defaults seem to work reasonably well
727 mPercentage = 0.5f; // half of bsdf samples should use path guiding
728 mSpatialTreeThreshold = 12000;
729 mDirTreeMaxDepth = 20;
730 mDirTreeThreshold = 0.01f;
731 mMinResetIterations = 2; // minimum reset iterations before sampling can be used
732
733 mResetIterations = 2;
734
735 mSpatialTree.reset(new SpatialTree(bbox));
736 mSpatialTree->refine(mSpatialTreeThreshold);
737 mSpatialTree->resetDirTrees(mDirTreeMaxDepth, mDirTreeThreshold);
738}

740void
741PathGuide::Impl::passReset()
742{
743 // Radiance can be recorded, directions sampled, and pdfs computed
744 // safely on multiple threads based on the use of atomic types.
745 // However, this method must only be called by a single thread and
746 // furthermore no other methods of the path guide may be accessed.
747 // It is the responsibility of the render driver to ensure that
748 // this thread safety requirement is met.
749 //
750 // This method is single threaded. If it becomes a slow bottleneck
751 // we can investigate ways to parallelize it - probably by parallelizing
752 // the resetDirTrees method.
753
754 MNRY_ASSERT(mEnable);
755 mSpatialTree->buildDirTrees();
756 ++mResetIterations;
757 // Split a spatial tree node when the number of samples in the node's dirtree
758 // exceeds this parameter value. We want this value to exponentially increase
759 // with each pass reset. This is the main reason why we work best when each
760 // pass has roughly twice as many samples as the previous.
761 mSpatialTree->refine(std::pow(2, mResetIterations / 2.0f) * mSpatialTreeThreshold);
762 mSpatialTree->resetDirTrees(mDirTreeMaxDepth, mDirTreeThreshold);
763}

773}
