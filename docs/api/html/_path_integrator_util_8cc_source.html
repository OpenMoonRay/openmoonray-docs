44finline void
45integrateBsdfSample(const BsdfSampler &bSampler, int lobeIndex,
46 const LightSetSampler &lSampler, const scene_rdl2::math::Color &pathThroughput,
47 const LightContribution &lCo, BsdfSample &bsmp)
48{
49 // Mark the sample valid by setting the distance
50 // TODO: Isn't it better to not add the sample in the first place ?
51 bsmp.distance = (!lCo.isInvalid ? lCo.distance :
52 (bSampler.getDoIndirect() ? scene_rdl2::math::sMaxValue :
53 BsdfSample::sInvalidDistance));
54 if (bsmp.isInvalid()) {
55 return;
56 }
57
58 const int ni = bSampler.getLobeSampleCount(lobeIndex);
59 const float invNi = bSampler.getInvLobeSampleCount(lobeIndex);
60 const scene_rdl2::math::Color pt = pathThroughput * invNi * bsmp.f * scene_rdl2::math::rcp(bsmp.pdf);
61
62 // Compute direct tentative contribution (omit shadowing)
63 const bool lobeIsMirror = bSampler.getLobe(lobeIndex)->matchesFlags(shading::BsdfLobe::ALL_MIRROR);
64 const int nl = (lCo.isInvalid ? 0 :
65 lSampler.getLightSampleCount());
66 bsmp.tDirect = (lCo.isInvalid ? scene_rdl2::math::sBlack : (lobeIsMirror ?
67 // Bsdf importance sampling
68 lCo.Li * pt :
69 // Multiple importance sampling
70 lCo.Li * pt * powerHeuristic(ni * bsmp.pdf, nl * lCo.pdf)
71 ));
72
73 // Compute indirect tentative contribution (omit Lindirect)
74 bsmp.tIndirect = (bSampler.getDoIndirect() ? pt : scene_rdl2::math::sBlack);
75}

77void
78drawBsdfSamples(pbr::TLState *pbrTls, const BsdfSampler &bSampler, const LightSetSampler &lSampler,
79 const Subpixel &sp, const PathVertex &pv, const scene_rdl2::math::Vec3f& P, const scene_rdl2::math::Vec3f* N,
80 float time, unsigned sequenceID, BsdfSample *bsmp, int clampingDepth,
81 float clampingValue, shading::BsdfLobe::Type indirectFlags, float rayDirFootprint)
82{
83 IntegratorSample2D bsdfSamples;
84 IntegratorSample2D lightFilterSamples;
85 IntegratorSample3D lightFilterSamples3D;
86 const int spIndex = (pv.nonMirrorDepth == 0) ? 0 : sp.mSubpixelIndex;
87
88 LightContribution lCo;
89
90 // Loop over active lobes
91 int s = 0;
92 const int lobeCount = bSampler.getLobeCount();
93 for (int lobeIndex = 0; lobeIndex < lobeCount; ++lobeIndex) {
94 const shading::BsdfLobe *lobe = bSampler.getLobe(lobeIndex);
95 const int lobeSampleCount = bSampler.getLobeSampleCount(lobeIndex);
96
97 // Setup sampler sequence
98 // We want one shared sequence for depth 0
99 SequenceIDIntegrator sid( pv.nonMirrorDepth, // TODO: Should be just depth ?
100 sp.mPixel,
101 lobeIndex,
102 SequenceType::Bsdf,
103 spIndex,
104 sequenceID );
105 int samplesSoFar = 0;
106 if (pv.nonMirrorDepth == 0) {
107 samplesSoFar = sp.mSubpixelIndex * lobeSampleCount; // used here and below
108 bsdfSamples.resume(sid, samplesSoFar);
109 } else {
110 bsdfSamples.restart(sid, lobeSampleCount);
111 }
112
113 const Scene *scene = MNRY_VERIFY(pbrTls->mFs->mScene);
114 bool lightFilterNeedsSamples = scene->lightFilterNeedsSamples();
115
116 if (lightFilterNeedsSamples) {
117 SequenceIDIntegrator sidFilter( pv.nonMirrorDepth, // TODO: Should be just depth ?
118 sp.mPixel,
119 lobeIndex,
120 SequenceType::LightFilter,
121 spIndex,
122 sequenceID );
123 SequenceIDIntegrator sidFilter3D( pv.nonMirrorDepth, // TODO: Should be just depth ?
124 sp.mPixel,
125 lobeIndex,
126 SequenceType::LightFilter3D,
127 spIndex,
128 sequenceID );
129
130 if (pv.nonMirrorDepth == 0) {
131 lightFilterSamples.resume(sidFilter, samplesSoFar);
132 lightFilterSamples3D.resume(sidFilter3D, samplesSoFar);
133 } else {
134 lightFilterSamples.restart(sidFilter, lobeSampleCount);
135 lightFilterSamples3D.restart(sidFilter3D, lobeSampleCount);
136 }
137 }
138
139 // Set up sampler sequence for stochastic choice of intersected light
140 SequenceIDIntegrator lightChoiceSid(pv.nonMirrorDepth,
141 sp.mPixel,
142 lobeIndex,
143 SequenceType::IndexSelection,
144 sp.mSubpixelIndex,
145 sequenceID);
146 IntegratorSample1D lightChoiceSamples(lightChoiceSid);
147
148 // lobeType is used for conditionally including ray termination lights.
149 // Ray termination lights provide a way to cheaply fill in the zeros which result from terminating
150 // ray paths too early. This is done by forcing the occlusion test to fail unconditionally.
151 // A path can only be terminated when the relevant indirectFlags bit for this lobeType is set to false.
152 shading::BsdfLobe::Type lobeType = lobe->getType();
153 bool includeRayTerminationLights = (static_cast<uint32_t>(lobeType) &
154 ~static_cast<uint32_t>(indirectFlags) &
155 static_cast<uint32_t>(shading::BsdfLobe::Type::ALL_LOBES))
156 != 0;
157
158 // lobeMask is used for comparison against a light's visibility flags.
159 // This comparison is made in the LightSampler's intersectAndEval() function inside the loop below.
160 int lobeMask = lobeTypeToRayMask(lobeType);
161
162 // Loop over each lobe's samples
163 for (int i = 0; i < lobeSampleCount; ++i, ++s) {
164
165 // Draw the sample and test validity
166 float bsdfSample[2];
167 bsdfSamples.getSample(bsdfSample, pv.nonMirrorDepth);
168
169 // Initialize to some value so you don't get NaNs if
170 // lightFilterNeedsSamples is false (see MOONRAY-4649)
171 LightFilterRandomValues lightFilterSample = {
172 scene_rdl2::math::Vec2f(0.f, 0.f),
173 scene_rdl2::math::Vec3f(0.f, 0.f, 0.f)};
174 if (lightFilterNeedsSamples) {
175 lightFilterSamples.getSample(&lightFilterSample.r2[0], pv.nonMirrorDepth);
176 lightFilterSamples3D.getSample(&lightFilterSample.r3[0], pv.nonMirrorDepth);
177 }
178
179#if DEBUG_BSDF_SAMPLES
180 INIT_PRIMI_VIEW(pvg, PVPIPE);
181 GroupStackGuard guard(pvg, "BSDF Samples");
182 pvg.setColor(0,1,1);
183
184 if (/*sp.mSubpixelIndex == 3 &&*/ bSampler.getCurrentLobeIndex() == 0 &&
185 pv.nonMirrorDepth == 1) {
186 pvg.point(bsdfSample[0], bsdfSample[1], 0);
187 usleep(100000);
188 }
189#endif
190 bool isValid = bSampler.sample(pbrTls, lobeIndex, P, bsdfSample[0], bsdfSample[1], bsmp[s]);
191 if (!isValid) {
192 continue;
193 }
194
195 // Compute lCo, which is the un-occluded light contribution for a
196 // random physical light in the LightSet in the direction of the sample.
197 // It should set Li and pdf to 0 if there is no light in that
198 // direction, and return the distance to the chosen light if
199 // there is. The distance is set to infinity if there is no light or
200 // to sInfiniteLightDistance if there is an InfiniteAreaLight.
201 lSampler.intersectAndEval(pbrTls->mTopLevelTls, P, N, bsmp[s].wi, lightFilterSample, time, false, includeRayTerminationLights,
202 lightChoiceSamples, pv.nonMirrorDepth, lobeMask, lCo, rayDirFootprint);
203
204 integrateBsdfSample(bSampler, lobeIndex, lSampler, pv.pathThroughput, lCo, bsmp[s]);
205 // save the light, we'll need it (for its labels) when processing lpes
206 bsmp[s].lp.light = lCo.isInvalid ? nullptr : lCo.light;
207
208 // Selective clamp of tDirect with clampingValue
209 if (pv.nonMirrorDepth >= clampingDepth) {
210 bsmp[s].tDirect = smartClamp(bsmp[s].tDirect, clampingValue);
211 }
212 }
213 }

219finline void
220integrateLightSetSample(const LightSetSampler &lSampler,
221 int lightIndex, const BsdfSampler &bSampler,
222 const PathVertex &pv, LightSample &lsmp,
223 int clampingDepth, float clampingValue, const scene_rdl2::math::Vec3f &P)
224{
225 // Mark the sample valid only if we have valid lobe contributions and
226 // initialize contribution for summing
227 bool isInvalid = true;
228 lsmp.t = scene_rdl2::math::sBlack;
229
230 const int ni = lSampler.getLightSampleCount();
231 const float invNi = lSampler.getInvLightSampleCount();
232
233 const scene_rdl2::math::Color factor = pv.pathThroughput * invNi * lsmp.Li * scene_rdl2::math::rcp(lsmp.pdf);
234
235 // Integrate with all the matching lobes
236 const shading::BsdfSlice &slice = bSampler.getBsdfSlice();
237 shading::BsdfLobe::Type flags = slice.getSurfaceFlags(bSampler.getBsdf(), lsmp.wi);
238 const int lobeCount = bSampler.getLobeCount();
239 MNRY_ASSERT(lobeCount > 0);
240
241 // initialize lpe member, setting each lobe entry to a null value
242 for (unsigned k = 0; k < shading::Bsdf::maxLobes; ++k) lsmp.lp.lobe[k] = nullptr;
243
244 for (int k = 0; k < lobeCount; ++k) {
245 const shading::BsdfLobe* const lobe = bSampler.getLobe(k);
246 // TODO: Should we still go through MIS calculations if
247 // !lobe->matchesFlags(flags)
248 if (!lobe->matchesFlags(flags)) {
249 continue;
250 }
251
252 // skip lobe if light is marked as not visible from this lobe
253 int lobeMask = lobeTypeToRayMask(lobe->getType());
254 const Light* light = lSampler.getLight(lightIndex);
255 if (!(lobeMask & light->getVisibilityMask())) {
256 continue;
257 }
258
259 // Evaluate the lobe
260 // TODO: Should we still go through MIS calculations if
261 // isSampleInvalid() because of pdf = 0
262 float pdf;
263 scene_rdl2::math::Color f;
264 // Pdf computation needs to be kept in sync with BsdfSampler::sample()
265 // Skip path guiding on mirror lobes, because their
266 // sample direction is already precisely determined.
267 const PathGuide &pg = bSampler.getPathGuide();
268 if (pg.canSample() && !(lobe->getType() & shading::BsdfLobe::MIRROR)) {
269 const float u = pg.getPercentage();
270 const float pgPdf = pg.getPdf(P, lsmp.wi);
271 f = lobe->eval(slice, lsmp.wi, &pdf);
272 // blending pdf values seems to work well enough in practice, and
273 // allows for a potential user percentage control.
274 pdf = u * pgPdf + (1.0 - u) * pdf;
275 } else {
276 f = lobe->eval(slice, lsmp.wi, &pdf);
277 }
278 if (isSampleInvalid(f, pdf)) {
279 continue;
280 }
281
282 // Direct lighting tentative contribution (omits shadowing)
283 // using multiple importance sampling:
284 const int nk = bSampler.getLobeSampleCount(k);
285 scene_rdl2::math::Color t = factor * f * powerHeuristic(ni * lsmp.pdf, nk * pdf);
286
287 // Selective clamp of t with clampingValue
288 if (pv.nonMirrorDepth >= clampingDepth) {
289 t = smartClamp(t, clampingValue);
290 }
291
292 lsmp.t += t;
293
294 // save off per-lobe t info, we'll need these for LPEs
295 MNRY_ASSERT(k < static_cast<int>(shading::Bsdf::maxLobes));
296 lsmp.lp.t[k] = t;
297 lsmp.lp.lobe[k] = lobe;
298
299 isInvalid = false;
300 }
301
302 if (isInvalid) {
303 lsmp.setInvalid();
304 }
305}

308void
309drawLightSetSamples(pbr::TLState *pbrTls, const LightSetSampler &lSampler, const BsdfSampler &bSampler,
310 const Subpixel &sp, const PathVertex &pv, const scene_rdl2::math::Vec3f &P, const scene_rdl2::math::Vec3f *N, float time,
311 unsigned sequenceID, LightSample *lsmp,
312 int clampingDepth, float clampingValue, float rayDirFootprint)
313{
314 IntegratorSample3D lightSamples;
315 IntegratorSample2D lightFilterSamples;
316 IntegratorSample3D lightFilterSamples3D;
317
318 const Scene *scene = MNRY_VERIFY(pbrTls->mFs->mScene);
319 bool lightFilterNeedsSamples = scene->lightFilterNeedsSamples();
320
321 const int spIndex = (pv.nonMirrorDepth == 0) ? 0 : sp.mSubpixelIndex;
322
323 Statistics &stats = pbrTls->mStatistics;
324
325 // Loop over each light
326 int s = 0;
327 const int lightCount = lSampler.getLightCount();
328 const int lightSampleCount = lSampler.getLightSampleCount();
329 for (int lightIndex = 0; lightIndex < lightCount; ++lightIndex) {
330 const Light *light = lSampler.getLight(lightIndex);
331
332 // Ray termination lights are used in an attempt to cheaply fill in the zeros which result from
333 // terminating ray paths too early. We exclude them from light samples because these samples represent
334 // natural ends to light paths. Exclusion is done by invalidating all the light's samples.
335 if (light->getIsRayTerminator()) {
336 for (int i = 0; i < lightSampleCount; ++i, ++s) {
337 lsmp[s].setInvalid();
338 }
339 continue;
340 }
341
342 const LightFilterList *lightFilterList = lSampler.getLightFilterList(lightIndex);
343
344 const int lightSampleCount = lSampler.getLightSampleCount();
345
346 // Setup sampler sequence
347 // We want one shared sequence for depth 0.
348 const SequenceIDIntegrator sid( pv.nonMirrorDepth,
349 sp.mPixel,
350 light->getHash(),
351 SequenceType::Light,
352 spIndex,
353 sequenceID );
354 int samplesSoFar = 0;
355 if (pv.nonMirrorDepth == 0) {
356 samplesSoFar = sp.mSubpixelIndex * lightSampleCount; // used here and below
357 lightSamples.resume(sid, samplesSoFar);
358 } else {
359 lightSamples.restart(sid, lightSampleCount);
360 }
361
362 if (lightFilterNeedsSamples) {
363 const SequenceIDIntegrator sidFilter( pv.nonMirrorDepth,
364 sp.mPixel,
365 light->getHash(),
366 SequenceType::LightFilter,
367 spIndex,
368 sequenceID );
369 const SequenceIDIntegrator sidFilter3D( pv.nonMirrorDepth,
370 sp.mPixel,
371 light->getHash(),
372 SequenceType::LightFilter3D,
373 spIndex,
374 sequenceID );
375
376 if (pv.nonMirrorDepth == 0) {
377 lightFilterSamples.resume(sidFilter, samplesSoFar);
378 lightFilterSamples3D.resume(sidFilter3D, samplesSoFar);
379
380 } else {
381 lightFilterSamples.restart(sidFilter, lightSampleCount);
382 lightFilterSamples3D.restart(sidFilter3D, lightSampleCount);
383 }
384 }
385
386 // Loop over each light's samples
387 for (int i = 0; i < lightSampleCount; ++i, ++s) {
388 // Draw the sample and test validity
389 scene_rdl2::math::Vec3f lightSample;
390 lightSamples.getSample(&lightSample[0], pv.nonMirrorDepth);
391
392 // Initialize to some value so you don't get NaNs if
393 // lightFilterNeedsSamples is false (see MOONRAY-4649)
394 LightFilterRandomValues lightFilterSample = {
395 scene_rdl2::math::Vec2f(0.f, 0.f),
396 scene_rdl2::math::Vec3f(0.f, 0.f, 0.f)};
397 if (lightFilterNeedsSamples) {
398 lightFilterSamples.getSample(&lightFilterSample.r2[0], pv.nonMirrorDepth);
399 lightFilterSamples3D.getSample(&lightFilterSample.r3[0], pv.nonMirrorDepth);
400 }
401 lSampler.sampleIntersectAndEval(pbrTls->mTopLevelTls,
402 light, lightFilterList,
403 P, N, lightFilterSample, time, lightSample,
404 lsmp[s], rayDirFootprint);
405
406 if (lsmp[s].isInvalid()) {
407 continue;
408 }
409
410 integrateLightSetSample(lSampler, lightIndex, bSampler, pv, lsmp[s],
411 clampingDepth, clampingValue, P);
412
413 stats.incCounter(STATS_LIGHT_SAMPLES);
414 }
415 }
416}

421void
422applyRussianRoulette(const BsdfSampler &bSampler, BsdfSample *bsmp,
423 const Subpixel &sp, const PathVertex &pv, unsigned sequenceID,
424 float threshold, float invThreshold)
425{
426 const int sampleCount = bSampler.getSampleCount();
427
428 const SequenceIDRR sid(sp.mPixel,
429 SequenceType::RussianRouletteBsdf,
430 sp.mSubpixelIndex, sequenceID);
431 IntegratorSample1D rrSamples;
432 rrSamples.resume(sid, pv.nonMirrorDepth * sampleCount);
433
434 // Cull rays from bsdf samples
435 for (int s = 0; s < sampleCount; ++s) {
436 if (bsmp[s].isInvalid()) {
437 continue;
438 }
439
440 const float lumDirect = luminance(bsmp[s].tDirect);
441 const float lumIndirect = luminance(bsmp[s].tIndirect);
442 const float lum = scene_rdl2::math::max(lumDirect, lumIndirect);
443 if (lum < threshold) {
444 // This should always be < 1
445 //
446 // The rcp function (SSE version) produces a NaN when the value is
447 // less than 0x1p-64f (the version I tested, anyway). FLT_EPSILON
448 // is much greater than this, but still probably a good threshold
449 // for our minimum probability.
450 const float continueProbability = std::max(scene_rdl2::math::sEpsilon,
451 lum * invThreshold);
452 float sample[1];
453 rrSamples.getSample(sample, pv.nonMirrorDepth);
454 if (sample[0] > continueProbability) {
455 bsmp[s].setInvalid();
456 } else {
457 const float invContinueProbability = scene_rdl2::math::rcp(continueProbability);
458 bsmp[s].tDirect *= invContinueProbability;
459 bsmp[s].tIndirect *= invContinueProbability;
460 }
461 }
462 }

466void
467applyRussianRoulette(const LightSetSampler &lSampler, LightSample *lsmp,
468 const Subpixel &sp, const PathVertex &pv, unsigned sequenceID,
469 float threshold, float invThreshold)
470{
471 const int sampleCount = lSampler.getSampleCount();
472
473 const SequenceIDRR sid(sp.mPixel,
474 SequenceType::RussianRouletteLight,
475 sp.mSubpixelIndex, sequenceID);
476 IntegratorSample1D rrSamples;
477 rrSamples.resume(sid, pv.nonMirrorDepth * sampleCount);
478
479 // Cull shadow rays from the light samples
480 for (int s = 0; s < sampleCount; ++s) {
481 if (lsmp[s].isInvalid()) {
482 continue;
483 }
484
485 const float lum = luminance(lsmp[s].t);
486 if (lum < threshold) {
487 // This should always be < 1
488 //
489 // The rcp function (SSE version) produces a NaN when the value is
490 // less than 0x1p-64f (the version I tested, anyway). FLT_EPSILON
491 // is much greater than this, but still probably a good threshold
492 // for our minimum probability.
493 const float continueProbability = std::max(scene_rdl2::math::sEpsilon,
494 lum * invThreshold);
495 float sample[1];
496 rrSamples.getSample(sample, pv.nonMirrorDepth);
497 if (sample[0] > continueProbability) {
498 lsmp[s].setInvalid();
499 } else {
500 const float continueProbabilityInv = scene_rdl2::math::rcp(continueProbability);
501 lsmp[s].t *= continueProbabilityInv;
502
503 // adjust per lobe values, if needed (see integrateLightSetSample())
504 for (unsigned int k = 0; k < shading::Bsdf::maxLobes; ++k) {
505 if (lsmp[s].lp.lobe[k]) {
506 lsmp[s].lp.t[k] *= continueProbabilityInv;
507 }
508 }
509 }
510 }
511 }

514void
515accumulateRayPresence(pbr::TLState *pbrTls,
516 const Light* light,
517 const mcrt_common::Ray& shadowRay,
518 float rayEpsilon,
519 int maxDepth,
520 float& totalPresence)
521{
522 // Used for presence shadows in scalar and vectorized mode.
523 // Presence primary visibility is handled elsewhere.
524
525 mcrt_common::ThreadLocalState *topLevelTls = pbrTls->mTopLevelTls;
526 shading::TLState *shadingTls = topLevelTls->mShadingTls.get();
527 const Scene *scene = MNRY_VERIFY(pbrTls->mFs->mScene);
528 const scene_rdl2::rdl2::Light* rdlLight = light->getRdlLight();
529
530 mcrt_common::Ray currentShadowRay = shadowRay;
531 shading::Intersection isect;
532
533 while (scene->intersectPresenceRay(topLevelTls, currentShadowRay, isect)) {
534
535 // handle shadow linking. This primitive may not cast shadow
536 // for certain specified lights in occlusionSkipList
537 geom::internal::BVHUserData* userData = static_cast<geom::internal::BVHUserData*>(currentShadowRay.ext.userData);
538 const geom::internal::NamedPrimitive* primPtr =
539 static_cast<const geom::internal::NamedPrimitive*>(userData->mPrimitive);
540 const geom::internal::ShadowLinking* shadowLinking = primPtr->getShadowLinking(isect.getLayerAssignmentId());
541 if (!shadowLinking || shadowLinking->canCastShadow(rdlLight)) {
542 // Finish setting up the intersection with the minimum needed to evaluate the material's
543 // presence
544 geom::initIntersectionPhase2(isect,
545 topLevelTls,
546 0, // mirrordepth
547 0, // glossydepth
548 0, // diffuseDepth
549 false, // subsurface allowed
550 scene_rdl2::math::Vec2f(0.0f, 0.0f), // minRoughness
551 -currentShadowRay.getDirection());
552 isect.setCameraId(0); // not used for material presence query
553
554 // get the presence value from the material
555 const scene_rdl2::rdl2::Material* material = isect.getMaterial()->asA<scene_rdl2::rdl2::Material>();
556 MNRY_ASSERT(material != nullptr);
557 float presence = shading::presence(material, shadingTls, shading::State(&isect));
558 totalPresence += (1.0f - totalPresence) * presence;
559 }
560
561 if (currentShadowRay.getDepth() < maxDepth && totalPresence < pbrTls->mFs->mPresenceThreshold) {
562 float tnear = rayEpsilon + currentShadowRay.tfar;
563 if (tnear >= shadowRay.tfar) {
564 return;
565 }
566 currentShadowRay = mcrt_common::Ray(currentShadowRay, tnear, shadowRay.tfar);
567 // constructor increments the depth automatically
568 } else {
569 return;
570 }
571 }
572}

594void
595scatterAndScale(const scene_rdl2::math::Vec3f &dNdx,
596 const scene_rdl2::math::Vec3f &dNdy,
597 const shading::BsdfLobe &lobe,
598 const scene_rdl2::math::Vec3f &wo,
599 const scene_rdl2::math::Vec3f &wi,
600 float scale,
601 float r1, float r2,
602 mcrt_common::RayDifferential &rd)
603{
604 MNRY_ASSERT(isNormalized(wi));
605
606#ifdef FORCE_SKIP_RAY_DIFFERENTIALS
607
608 rd.clearDifferentials();
609
610#else
611
612 if (rd.hasDifferentials()) {
613
614 MNRY_ASSERT(isNormalized(rd.getDirX()));
615 MNRY_ASSERT(isNormalized(rd.getDirY()));
616
617 scene_rdl2::math::Vec3f dDdx = rd.getdDdx();
618 scene_rdl2::math::Vec3f dDdy = rd.getdDdy();
619
620 lobe.differentials(wo, wi, r1, r2, dNdx, dNdy,
621 dDdx, dDdy);
622
623 scene_rdl2::math::Vec3f const &origin = rd.getOrigin();
624 rd.setOriginX(origin + (rd.getOriginX() - origin) * scale);
625 rd.setOriginY(origin + (rd.getOriginY() - origin) * scale);
626
627 // Scaling dDdx is more accurate than scaling (mDirX - direction)
628 // because of the normalization that happens when converting from
629 // dDdx to mDirX
630 dDdx *= scale;
631 dDdy *= scale;
632
633 rd.setDirX(normalize(wi + dDdx));
634 rd.setDirY(normalize(wi + dDdy));
635 }
636
637#endif // FORCE_SKIP_RAY_DIFFERENTIALS
638
639 rd.setDirection(wi);
640}

642// Subsurface Integrator Utility Functions
643// Tri-Planar Projection Sampling Probabilities - T, B, N
644// Note that [1] suggests a distribution of [0.5, 0.25, 0.25] whereas I find
645// [0.75, 0.125, 0.125] works better
646// 75% to N & the rest divided equally between T & B
647static std::array<float, 3> sBssrdfAxisProbabilities = { 0.75f, 0.125f, 0.125f };
648
649// Selects an axis for projection
650int
651bssrdfSelectAxisAndRemapSample(const scene_rdl2::math::ReferenceFrame &localF,
652 float &rnd,
653 scene_rdl2::math::Vec3f &directionProj)
654{
655 // We divide the T and B probabilities further between positive and negative options
656 std::array<float, 5> axisProbabilities = {sBssrdfAxisProbabilities[0],
657 sBssrdfAxisProbabilities[1]/2.0f,
658 sBssrdfAxisProbabilities[1]/2.0f,
659 sBssrdfAxisProbabilities[2]/2.0f,
660 sBssrdfAxisProbabilities[2]/2.0f};
661
662 std::array<float, 5> axisCDF;
663 axisCDF[0] = sBssrdfAxisProbabilities[0];
664 for (int i = 1; i < 5; i++) {
665 axisCDF[i] = axisProbabilities[i] + axisCDF[i-1];
666 }
667 MNRY_ASSERT(scene_rdl2::math::isOne(axisCDF[4]));
668
669 int axisIndex;
670 if (rnd < axisCDF[0]) {
671 axisIndex = 0;
672 directionProj = -localF.getN();
673 rnd /= axisProbabilities[0];
674 } else if (rnd < axisCDF[1]) {
675 axisIndex = 1;
676 directionProj = -localF.getX();
677 rnd = (rnd - axisCDF[0]) / axisProbabilities[1];
678 } else if (rnd < axisCDF[2]) {
679 axisIndex = 2;
680 directionProj = localF.getX();
681 rnd = (rnd - axisCDF[1]) / axisProbabilities[2];
682 } else if (rnd < axisCDF[3]) {
683 axisIndex = 3;
684 directionProj = -localF.getY();
685 rnd = (rnd - axisCDF[2]) / axisProbabilities[3];
686 } else {
687 axisIndex = 4;
688 directionProj = localF.getY();
689 rnd = (rnd - axisCDF[3]) / axisProbabilities[4];
690 }
691 return axisIndex;
692}

695// Converts the *local* BSSRDF Offset into *render* space using the
696// appropriate local frame based on the projection axis
697scene_rdl2::math::Vec3f
698bssrdfOffsetLocalToGlobal(const scene_rdl2::math::ReferenceFrame &localF,
699 int axisIndex,
700 const scene_rdl2::math::Vec3f &localOffset)
701{
702 scene_rdl2::math::Vec3f globalOffset;
703
704 if (axisIndex == 0) {
705 // Projecting along N
706 globalOffset = localOffset[0] * localF.getX() + localOffset[1] * localF.getY();
707 } else if (axisIndex == 1 || axisIndex == 2) {
708 // Projecting along X
709 globalOffset = localOffset[0] * localF.getY() + localOffset[1] * localF.getN();
710 } else {
711 // Projecting along Y
712 globalOffset = localOffset[0] * localF.getN() + localOffset[1] * localF.getX();
713 }
714 return globalOffset;
715}

717// Calculates the MIS Weight Using Veachs' One-Sample Method
718float
719bssrdfGetMISAxisWeight(const scene_rdl2::math::ReferenceFrame &localF,
720 const scene_rdl2::math::Vec3f &sampleNormal,
721 const scene_rdl2::math::Vec3f &sampleOffset,
722 const shading::Bssrdf &bssrdf)
723{
724 // Convert the global offset into a Local offset
725 scene_rdl2::math::Vec3f dpLocal = localF.globalToLocal(sampleOffset);
726 // Distance along each projection axis
727 float rN = scene_rdl2::math::sqrt(dpLocal[0] * dpLocal[0] + dpLocal[1] * dpLocal[1]);
728 float rX = scene_rdl2::math::sqrt(dpLocal[1] * dpLocal[1] + dpLocal[2] * dpLocal[2]);
729 float rY = scene_rdl2::math::sqrt(dpLocal[2] * dpLocal[2] + dpLocal[0] * dpLocal[0]);
730
731 // Jacobian for Each Projection Axis
732 float nN = scene_rdl2::math::abs(dot(localF.getN(), sampleNormal));
733 float nX = scene_rdl2::math::abs(dot(localF.getX(), sampleNormal));
734 float nY = scene_rdl2::math::abs(dot(localF.getY(), sampleNormal));
735
736 // PDF for Each Projection Axis
737 float pdfN = bssrdf.pdfLocal(rN);
738 float pdfX = bssrdf.pdfLocal(rX);
739 float pdfY = bssrdf.pdfLocal(rY);
740
741 float misWeight = sBssrdfAxisProbabilities[0] * pdfN * nN;
742 misWeight += sBssrdfAxisProbabilities[1] * pdfX * nX;
743 misWeight += sBssrdfAxisProbabilities[2] * pdfY * nY;
744
745 if (!scene_rdl2::math::isZero(misWeight)) {
746 return 1.0f/misWeight;
747 }
748 else {
749 // throw an error?
750 return 1.0f;
751 }
752}
753
754// Calculates the Area-Compensation Term.
755// We divide the analytically computed diffuse reflectance by the one
756// computed via sampling. This gives us a measure of how much the underlying
757// geometry deviates from the "semi-infinite, planar" assumption for diffusion based BSSRDFs.
758// On a regular planar surface, this compensation term amount to 1.
759scene_rdl2::math::Color
760bssrdfAreaCompensation(const scene_rdl2::math::Color &measuredDiffuseReflectance,
761 const scene_rdl2::math::Color &analyticDiffuseReflectance)
762{
763 scene_rdl2::math::Color areaCompensation = scene_rdl2::math::sWhite;
764 for (int i = 0; i < 3; i++) {
765 // Checking if the measured reflectance was too low, indicating really bad sampling
766 // This could happen at really thin areas or because of incorrect sampling, which could spike the
767 // reflectance results. Hence the check below. 0.01f has been good in my tests so far.
768 if ((measuredDiffuseReflectance[i]-0.01f) > 0) {
769 areaCompensation[i] = analyticDiffuseReflectance[i] / measuredDiffuseReflectance[i];
770 }
771 }
772 return areaCompensation;
773}

778// Shadow Falloff Utility Functions
779enum clearRadiusFalloffType {
780 FALLOFF_LINEAR = 0,
781 FALLOFF_EXPONENTIAL_UP,
782 FALLOFF_EXPONENTIAL_DOWN,
783 FALLOFF_SMOOTH
784};
785
786scene_rdl2::math::Color
787calculateShadowFalloff(const Light *light, float distToLight, const scene_rdl2::math::Color unoccludedColor)
788{
789 float crFalloffMin = light->getClearRadius();
790 float crFalloffMax = light->getClearRadiusFalloffDistance() + light->getClearRadius();
791 float falloffDist = light->getClearRadiusFalloffDistance();
792 int interpolationType = light->getClearRadiusInterpolationType();
793
794 // if distance to light less than clear radius, no shadow
795 if (distToLight <= crFalloffMin) {
796 return unoccludedColor;
797 }
798
799 if (distToLight <= crFalloffMax) {
800 float t = (distToLight - crFalloffMin) / falloffDist;
801 float weight;
802 switch (interpolationType) {
803 case (FALLOFF_LINEAR):
804 weight = t;
805 break;
806 case (FALLOFF_EXPONENTIAL_UP):
807 weight = t*t;
808 break;
809 case (FALLOFF_EXPONENTIAL_DOWN):
810 weight = 1.f - t;
811 weight = 1.f - (weight * weight);
812 break;
813 case (FALLOFF_SMOOTH):
814 weight = t*t * (3 - 2*t);
815 break;
816 default:
817 return scene_rdl2::math::Color(0.f);
818 }
819 scene_rdl2::math::Color leftCol = unoccludedColor;
820 scene_rdl2::math::Color rightCol = scene_rdl2::math::Color(0.f);
821 return leftCol + weight * (rightCol - leftCol);
822 }
823 return scene_rdl2::math::Color(0.f);
824}

840void
841CPP_computeRadianceSubsurface(const PathIntegrator * pathIntegrator,
842 pbr::PbrTLState * pbrTls,
843 const uint32_t * rayStateIndices,
844 const Bssrdfv * bssrdfv,
845 const VolumeSubsurfacev * volumeSubsurfacev,
846 const LightSet * lightSet,
847 const int materialLabelIds,
848 const int lpeMaterialLabelIds,
849 const int * geomLabelIds,
850 const int * doIndirect,
851 const float * rayEpsilon,
852 const float * shadowRayEpsilon,
853 uint32_t * sequenceID,
854 float * results, // VLEN rgb colors in SOA format
855 float * ssAovResults, // VLEN rgb colors in SOA format
856 int32_t lanemask)
857{
858 MNRY_ASSERT(pbrTls->isIntegratorAccumulatorRunning());
859 MNRY_ASSERT(pbrTls->isIspcAccumulatorRunning());
860
861 pbrTls->stopIspcAccumulator();
862
863 scene_rdl2::alloc::Arena *arena = pbrTls->mArena;
864 SCOPED_MEM(arena);
865
866 RayState *baseRayState = indexToRayState(0);
867
868 shading::Bsdf bsdf;
869
870 // This is an interesting case for aovs. We are now dropping into the
871 // depth-first integrator which means that aovs are handled by
872 // continually accumulating results into a single aov array.
873 // Note that the bundled integrator does not handle aovs this way.
874 // Instead it immediately queues results as they are matched.
875 //
876 // We will allocate an aov array, and then queue the results
877 // when we exit back into the breadth-first integrator. We
878 // assume that we are called from the bundled integrator, which
879 // means that all the non-light/visibility aovs have already been handled.
880 // So this means we only need to allocate aovs if we have light
881 // aovs in the scene.
882 float *aovs = nullptr;
883
884 unsigned int aovNumChannels = 0;
885 const AovSchema &aovSchema = *pbrTls->mFs->mAovSchema;
886 const LightAovs &lightAovs = *pbrTls->mFs->mLightAovs;
887 if (!aovSchema.empty() && lightAovs.hasEntries()) {
888 aovNumChannels = aovSchema.numChannels();
889 aovs = arena->allocArray<float>(aovNumChannels);
890 }
891
892 for (unsigned i = 0; i < VLEN; ++i) {
893 // Don't compute subsurface radiance for invalid lanes.
894 if (!isActive(lanemask, i)) {
895 continue;
896 }
897 // Convert from Bssrdf from SOA to AOS.
898 shading::Bssrdf *bssrdf = shading::createBSSRDF(arena, bssrdfv, i);
899 // Convert VolumeSubsurface from SOA to AOS.
900 shading::VolumeSubsurface *volumeSubsurface =
901 shading::createVolumeSubsurface(arena, volumeSubsurfacev, i);
902 if (!bssrdf && !volumeSubsurface) {
903 continue;
904 }
905
906 bsdf.setLabelIds(materialLabelIds, lpeMaterialLabelIds, geomLabelIds[i]);
907 bsdf.setBssrdf(bssrdf);
908 bsdf.setVolumeSubsurface(volumeSubsurface);
909
910 RayState *rs = &baseRayState[rayStateIndices[i]];
911
912 const Subpixel &sp = rs->mSubpixel;
913 PathVertex &pv = rs->mPathVertex;
914 const mcrt_common::RayDifferential &ray = rs->mRay;
915 const shading::Intersection &isect = *rs->mAOSIsect;
916
917 // Setup a slice, which handles selecting the lobe types and setup
918 // evaluations to include the cosine term.
919 // Note: Even though we may not be doing indirect for certain lobe types
920 // (according to indirectFlags), we still want to draw samples according
921 // to all lobes for direct lighting MIS.
922 shading::BsdfSlice slice(isect.getNg(), -ray.getDirection(), true,
923 isect.isEntering(), pbrTls->mFs->mShadowTerminatorFix, shading::BsdfLobe::ALL);
924
925 if (aovs) {
926 aovSchema.initFloatArray(aovs);
927 }
928
929 scene_rdl2::math::Color ssAov = scene_rdl2::math::Color(0);
930 scene_rdl2::math::Color radiance = scene_rdl2::math::Color(0);
931 if (bssrdf) {
932 MNRY_ASSERT(!volumeSubsurface); // else our depth count will be messed up
933 // increment subsurface depth
934 // this is not a double increment.
935 // see note in integrateBundled (PathIntegratorBundled.ispc)
936 pv.subsurfaceDepth += 1;
937 radiance += pathIntegrator->computeRadianceDiffusionSubsurface(
938 pbrTls, bsdf, sp, rs->mCameraId, pv, ray, isect, slice, *bssrdf, *lightSet,
939 doIndirect[i], rayEpsilon[i], shadowRayEpsilon[i], sequenceID[i], ssAov, aovs);
940 }
941 if (volumeSubsurface) {
942 MNRY_ASSERT(!bssrdf); // else our depth count will be messed up
943 // increment subsurface depth
944 // this is not a double increment.
945 // see note in integrateBundled (PathIntegratorBundled.ispc)
946 pv.subsurfaceDepth += 1;
947 radiance += pathIntegrator->computeRadiancePathTraceSubsurface(
948 pbrTls, bsdf, sp, rs->mCameraId, pv, ray, isect, *volumeSubsurface, *lightSet,
949 doIndirect[i], rayEpsilon[i], shadowRayEpsilon[i], sequenceID[i], ssAov, aovs);
950 }
951
952 if (aovs) {
953 // This code only computes light and visibility AOVs, so we only want to pass along those
954 // types of AOVs. But, the aov buffer we created here contains ALL the AOVs.
955 // If we pass all types of AOVs, this leads to problems with things
956 // like depth, because we don't set the depth AOV value here but would then pass
957 // the default zero depth value back to the bundled integrator. This leads to erroneous zero
958 // depth values in the final AOV output.
959 // The light aovs had the stateids checked prior to being accumulated into aovs,
960 // so no need to further check them.
961 aovAddToBundledQueue(pbrTls, aovSchema, isect, ray,
962 AOV_TYPE_LIGHT_AOV | AOV_TYPE_VISIBILITY_AOV, aovs, rs->mSubpixel.mPixel,
963 rs->mDeepDataHandle, getFilm(rs->mTilePassAndFilm));
964 }
965
966 results[i] += radiance.r;
967 results[VLEN + i] += radiance.g;
968 results[VLEN * 2 + i] += radiance.b;
969
970 ssAovResults[i] += ssAov.r;
971 ssAovResults[VLEN + i] += ssAov.g;
972 ssAovResults[VLEN * 2 + i] += ssAov.b;
973 }
974
975 pbrTls->startIspcAccumulator();
976}
