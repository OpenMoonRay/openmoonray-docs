113void
114ProceduralLeaf::addPrimitive(std::unique_ptr<Primitive> p,
115 const MotionBlurParams& motionBlurParams,
116 const shading::XformSamples& parent2render)
117{
118 if (p) {
119 const Mat43& proc2parent = getState().getProc2Parent();
120 shading::XformSamples prim2render;
121 for (size_t i = 0; i < parent2render.size(); ++i) {
122 prim2render.push_back(proc2parent * parent2render[i]);
123 }
124 // transform primitive to rendering space and interpolate it with motion blur info
125 internal::PrimitivePrivateAccess::transformPrimitive(p.get(), motionBlurParams, prim2render);
126 mImpl->mPrimitiveGroup->addPrimitive(std::move(p));
127 }
128}
