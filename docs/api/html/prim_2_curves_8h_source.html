192 virtual BBox3f computeAABB() const override
193 {
194 if (mVertexBuffer.empty()) {
195 return BBox3f(scene_rdl2::math::zero);
196 }
197 float maxRadius = 0.0f;
198 size_t motionSampleCount = getMotionSamplesCount();
199 BBox3f result(mVertexBuffer(0));
200 for (size_t v = 0; v < mVertexBuffer.size(); ++v) {
201 for (size_t t = 0; t < motionSampleCount; ++t) {
202 result.extend(mVertexBuffer(v, t));
203 maxRadius = std::max(maxRadius, mVertexBuffer(v, t).w);
204 }
205 }
206 // padding the bounding box with curves radius
207 result.lower -= Vec3f(maxRadius);
208 result.upper += Vec3f(maxRadius);
209 return result;
210 }
211
212 virtual BBox3f computeAABBAtTimeStep(int timeStep) const override
213 {
214 if (mVertexBuffer.empty()) {
215 return BBox3f(scene_rdl2::math::zero);
216 }
217 float maxRadius = 0.0f;
218 MNRY_ASSERT(timeStep >= 0 && timeStep < static_cast<int>(getMotionSamplesCount()), "timeStep out of range");
219 BBox3f result(scene_rdl2::util::empty);
220 for (size_t v = 0; v < mVertexBuffer.size(); ++v) {
221 result.extend(mVertexBuffer(v, timeStep));
222 maxRadius = std::max(maxRadius, mVertexBuffer(v, timeStep).w);
223 }
224 // padding the bounding box with curves radius
225 result.lower -= Vec3f(maxRadius);
226 result.upper += Vec3f(maxRadius);
227 return result;
228 }
