40finline Vec2f
41local2uv(const Vec3f &local)
42{
43 MNRY_ASSERT(isNormalized(local));
44
45 Vec2f uv;
46 if (scene_rdl2::math::abs(local.y) >= 1.0f) {
47 // Handle singularities at poles, to keep function consistent between
48 // sample and intersect
49 uv.x = 0.0f;
50 uv.y = (local.y > 0.0f) ? 0.0f : 1.0f;
51 } else {
52 // Convert to spherical polar coords with y up.
53 // Note that phi is intentionally off by pi, so that adding 0.5 to
54 // uv.x brings it into the range [0,1] without using a conditional.
55 float theta = scene_rdl2::math::acos(local.y);
56 float phi = scene_rdl2::math::atan2(-local.z, -local.x);
57
58 // Generate (u,v)
59 uv.x = phi * sOneOverTwoPi + 0.5f;
60 uv.y = theta * sOneOverPi;
61 }
62
63 return uv;
64}

66finline Vec3f
67vectorFromPolarSinCosVersion(float sinTheta, float cosTheta, float phi)
68{
69 float sinPhi, cosPhi;
70 sincos(phi, &sinPhi, &cosPhi);
71 return Vec3f(cosPhi * sinTheta, sinPhi * sinTheta, cosTheta);
72}
73
74finline Vec3f
75vectorFromPolarCosVersion(float cosTheta, float phi)
76{
77 float sinTheta = scene_rdl2::math::sqrt(1.0f - cosTheta * cosTheta);
78 return vectorFromPolarSinCosVersion(sinTheta, cosTheta, phi);
79}

97bool
98SphereLight::update(const Mat4d& world2render)
99{
100 MNRY_ASSERT(mRdlLight);
101
102 mOn = mRdlLight->get(scene_rdl2::rdl2::Light::sOnKey);
103 if (!mOn) {
104 return false;
105 }
106
107 updateVisibilityFlags();
108 updatePresenceShadows();
109 updateRayTermination();
110 updateTextureFilter();
111 updateMaxShadowDistance();
112
113 // Radius gets baked into the the mLocal2Render and other matrices such
114 // that the sphere radius is always 1 in local light space.
115 const float radius = mRdlLight->get<scene_rdl2::rdl2::Float>(sRadiusKey);
116
117 UPDATE_ATTRS_CLEAR_RADIUS
118
119 const Mat4d l2w0 = mRdlLight->get(scene_rdl2::rdl2::Node::sNodeXformKey, /* rayTime = */ 0.f);
120 const Mat4d l2w1 = mRdlLight->get(scene_rdl2::rdl2::Node::sNodeXformKey, /* rayTime = */ 1.f);
121 Mat4f sclRot = sRotateX180;
122 sclRot[0][0] *= radius;
123 sclRot[1][1] *= radius;
124 sclRot[2][2] *= radius;
125 const Mat4f local2Render0 = sclRot * toFloat(l2w0 * world2render);
126 const Mat4f local2Render1 = sclRot * toFloat(l2w1 * world2render);
127 if (!updateParamAndTransforms(local2Render0, local2Render1, 1.0f, 1.0f)) {
128 return false;
129 }
130
131 // Compute render space quantities.
132 // TODO: mb scale radiance
133 mRadius = getRadius();
134 mRadiusSqr = mRadius * mRadius;
135 mRcpRadius = 1.0f / mRadius;
136 mArea = sFourPi * mRadiusSqr;
137 mInvArea = 1.0f / mArea;
138
139 // Compute radiance.
140 mRadiance = computeLightRadiance(mRdlLight, scene_rdl2::rdl2::Light::sColorKey,
141 scene_rdl2::rdl2::Light::sIntensityKey, scene_rdl2::rdl2::Light::sExposureKey, sNormalizedKey,
142 sApplySceneScaleKey, mInvArea);
143 if (isBlack(mRadiance)) {
144 mOn = false;
145 return false;
146 }
147
148 if (!updateImageMap(Distribution2D::SPHERICAL)) {
149 return false;
150 }
151
152 // Sidedness
153 mSidedness = static_cast<LightSidednessType>(mRdlLight->get<scene_rdl2::rdl2::Int>(sSidednessKey));
154 return true;
155}

159bool
160SphereLight::canIlluminate(const Vec3f p, const Vec3f *n, float time, float radius,
161 const LightFilterList* lightFilterList) const
162{
163 MNRY_ASSERT(mOn);
164
165 // Reject if sphere is completely on the backside of the point's plane
166 if (n) {
167 Plane pl(p, *n);
168 const float planeDist = pl.getDistance(getPosition(time));
169 // mlee: we are losing a small amount of energy here by adding small value.
170 const float planeThreshold = sEpsilon - mRadius - radius;
171 if (planeDist < planeThreshold) {
172 return false;
173 }
174 }
175
176 // Reject when completely inside an outward-facing sphere light
177 if (mSidedness == LightSidednessType::LIGHT_SIDEDNESS_REGULAR) {
178 // mlee: we are losing a small amount of energy here by adding small value.
179 const float sphereThreshold = mRadius + sEpsilon - radius;
180 const Vec3f dp = p - getPosition(time);
181 if (dp.lengthSqr() < sqr(sphereThreshold)) {
182 return false;
183 }
184 }
185
186 if (lightFilterList) {
187 return canIlluminateLightFilterList(lightFilterList,
188 { getPosition(time), mRadius, p,
189 getXformRender2Local(time, lightFilterList->needsLightXform()),
190 radius, time
191 });
192 }
193
194 return true;
195}

218BBox3f
219SphereLight::getBounds() const
220{
221 const Vec3f pos = getPosition(0.f);
222 const float rad = mRadius;
223 BBox3f bounds(pos - Vec3f(rad), pos + Vec3f(rad));
224
225 if (isMb()) {
226 const Vec3f pos1 = getPosition(1.f);
227 bounds.extend(BBox3f(pos1 - Vec3f(rad), pos1 + Vec3f(rad)));
228 }
229
230 return bounds;
231}

233
234// Ray equation: x = p + t * wi
235// Sphere equation: (x-x0)^2 = r^2, with x0 = sphere centre
236// Intersection: t^2 - 2wi.v*t + v^2-r^2 = 0, with v = x0-p and assuming |wi|=1
237// This is a quadratic of the form a*t^2 - 2b*t + c = 0, with a=1, b=2wi.v, c=v^2-r^2
238// The discriminant D = b^2-c
239// For better precision, we use an algebraically equivalent form for the descriminant, D = r^2 - (v - b*wi)^2
240// We can use the following conditions to trivially reject certain cases, depending on the light's sidedness:
241// b<0, ray pointing away from sphere center
242// c<0, ray origin inside sphere
243// D<0, no real roots, i.e. ray misses the sphere
244
245bool
246SphereLight::intersect(const Vec3f &p, const Vec3f *n, const Vec3f &wi, float time,
247 float maxDistance, LightIntersection &isect) const
248{
249 const Vec3f v = getPosition(time) - p;
250 const float b = dot(wi, v);
251 const float v2 = dot(v, v);
252 const float c = v2 - mRadiusSqr;
253
254 // Do trivial rejection based on these values, according to sphere's sidedness...
255 // regular sidedness: reject if ray is travelling away from sphere center OR ray origin is INSIDE sphere
256 // reverse sidedness: reject if ray is travelling away from sphere center AND ray origin is OUTSIDE sphere
257 // c > 0 when ray origin is outside sphere
258 // b <= 0 when ray is travelling away from sphere center
259 if ((c > 0.0f) ? (b <= 0.0f) : (mSidedness == LightSidednessType::LIGHT_SIDEDNESS_REGULAR)) {
260 return false;
261 }
262
263 // Reject if discriminant is negative (ray misses sphere)
264 const float D = mRadiusSqr - lengthSqr(v - b * wi); // More accurate version of discriminant than b^2-c
265 if (D <= 0.0f) {
266 return false;
267 }
268
269 // Calculate distance to intersection, reject if too big
270 const float s = scene_rdl2::math::sqrt(D);
271 // Choose near or far intersection depending on sidedness
272 const float t = (mSidedness == LightSidednessType::LIGHT_SIDEDNESS_REGULAR) ? b - s : b + s;
273 if (t > maxDistance) {
274 return false;
275 }
276
277 // Fill in isect members
278 const Vec3f normal = (t*wi - v) * mRcpRadius; // Outward-facing normal
279 // We don't need to worry about 2-sided here because that's not supported for SphereLight
280 isect.N = (mSidedness == LightSidednessType::LIGHT_SIDEDNESS_REGULAR) ? normal : -normal;
281 isect.distance = t;
282
283 // Optionally compute uvs for texturing
284 isect.uv = mDistribution ? local2uv(xformVectorRender2LocalRot(normal, time)) : zero;
285
286 // Store values which eval() will later use to compute the pdf.
287 // See sample() function for details about each case.
288 if (c < 0.0f) {
289 // Due to earlier rejection, this code only executes for reverse sidedness
290 isect.data[1] = -1.0f; // Signal uniform sampling over the solid angle
291 if (n) {
292 isect.data[0] = sOneOverTwoPi;
293 } else {
294 isect.data[0] = sOneOverFourPi;
295 }
296 } else {
297 // This is the nastiest part of all! We need to match the pdf of the sampling scheme used in the sample()
298 // function, where the samples were generated on the front-facing visible cap of the sphere and then
299 // projected along the ray to the back surface in the case of reverse sidedness. So here, in the reverse
300 // sidedness case, we must generate the values of t, the normal vector, and the trig expressions as if
301 // the intersection point were on the front surface of the sphere.
302 const float sinThetaMaxSq = c / v2;
303 float cosTheta;
304 if (mSidedness == LightSidednessType::LIGHT_SIDEDNESS_REVERSE) {
305 const float t_front = b - s;
306 const Vec3f normal_front = (t_front * wi - v) * mRcpRadius;
307 cosTheta = -dot(v, normal_front) / scene_rdl2::math::sqrt(v2);
308 isect.data[1] = t_front;
309 } else {
310 cosTheta = -dot(v, normal) / scene_rdl2::math::sqrt(v2);
311 isect.data[1] = t;
312 }
313 isect.data[0] = cosTheta / sinThetaMaxSq;
314 }
315
316 return true;
317}

320bool
321SphereLight::sample(const Vec3f &p, const Vec3f *n, float time, const Vec3f& r,
322 Vec3f &wi, LightIntersection &isect, float rayDirFootprint) const
323{
324 MNRY_ASSERT(mOn);
325
326 const Vec3f v = getPosition(time) - p;
327 const float v2 = dot(v, v);
328 const float c = v2 - mRadiusSqr;
329
330 float t;
331 Vec3f normal;
332
333 // Sampling strategy is different for inside vs outside the sphere
334 if (c < 0.0f) {
335
336 // Inside sphere
337 if (mSidedness == LightSidednessType::LIGHT_SIDEDNESS_REGULAR) {
338 // Trivial rejection when inside an outward-facing sphere light
339 return false;
340 }
341
342 // Signal uniform sampling over solid angle
343 isect.data[1] = -1.0f;
344
345 if (n) {
346 // Use surface normal to sample only the visible hemisphere
347 Vec3f localWi = vectorFromPolarCosVersion(r[0], sTwoPi * r[1]);
348 math::ReferenceFrame frame(*n);
349 wi = frame.localToGlobal(localWi);
350 isect.data[0] = sOneOverTwoPi; // pdf = 1 / (solid angle of hemisphere)
351 } else {
352 // No normal supplied; sample the entire sphere
353 wi = vectorFromPolarCosVersion(2.0f * r[0] - 1.0f, sTwoPi * r[1]);
354 isect.data[0] = sOneOverFourPi; // pdf = 1 / (solid angle of full sphere)
355 }
356
357 // See comments for intersect()
358 const float b = dot(wi, v);
359 const float D = b*b - c; // Precision doesn't need to be improved here because radius is large compared to v
360 const float s = scene_rdl2::math::sqrt(D);
361 t = b + s; // Pick far intersection because we're inside the sphere
362 normal = (t*wi - v) * mRcpRadius; // Outward-facing normal
363
364 } else {
365
366 // See Dutre's "Global Illumination Compendium" section 35, page 20.
367 const float sinThetaMaxSq = c / v2;
368 const float sinThetaSq = r[0] * sinThetaMaxSq;
369 const float sinTheta = scene_rdl2::math::sqrt(sinThetaSq);
370 const float cosTheta = scene_rdl2::math::sqrt(1.0f - sinThetaSq);
371
372 math::ReferenceFrame frame(-v/scene_rdl2::math::sqrt(v2));
373 const Vec3f localNormal = vectorFromPolarSinCosVersion(sinTheta, cosTheta, sTwoPi*r[1]);
374 normal = frame.localToGlobal(localNormal); // Outward-facing normal
375
376 Vec3f rayDir = v + mRadius * normal;
377 const float t2 = lengthSqr(rayDir);
378 t = scene_rdl2::math::sqrt(t2);
379 if (t < sEpsilon) {
380 return false;
381 }
382
383 wi = rayDir / t;
384 if (n && dot(*n, wi) < sEpsilon) {
385 return false;
386 }
387
388 // Store values which eval() will later use to compute the pdf.
389 isect.data[0] = cosTheta / sinThetaMaxSq;
390 isect.data[1] = t;
391
392 // We don't need to worry about 2-sided here because that's not supported for SphereLight
393 if (mSidedness == LightSidednessType::LIGHT_SIDEDNESS_REVERSE) {
394 // This is nasty. We use the direction of the generated sample, but continue the ray to where
395 // it hits the far side of the sphere. It's done this way so the sampling scheme doesn't change,
396 // because otherwise it could yield a different image when sample clamping is active. Although
397 // no prior production shots used SphereLights with reverse sidedness, we want the result to match
398 // that of a SphereLight with regular sidedness when there is no geometry inside the SphereLight.
399 // Also, it is important to adjust the values of t and normal AFTER stashing the values in isect.data
400 // which will be used by eval() to compute the pdf. This is because the pdf comes from a distribution
401 // of points on the front-facing surface of the sphere.
402 const float b = dot(wi, v);
403 const float D = mRadiusSqr - lengthSqr(v - b * wi); // More accurate version of discriminant than b^2-c
404 t = b + scene_rdl2::math::sqrt(max(D, 0.0f)); // Rounding can cause D to go slight negative, hence the max()
405 normal = (t*wi - v) * mRcpRadius; // Outward-facing normal
406 }
407 }
408
409 isect.N = (mSidedness == LightSidednessType::LIGHT_SIDEDNESS_REGULAR) ? normal : -normal;
410 isect.distance = t;
411
412 // Optionally compute uvs for texturing
413 isect.uv = mDistribution ? local2uv(xformVectorRender2LocalRot(normal, time)) : zero;
414
415 return true;
416}

419Color
420SphereLight::eval(mcrt_common::ThreadLocalState* tls, const Vec3f &wi, const Vec3f &p, const LightFilterRandomValues& filterR, float time,
421 const LightIntersection &isect, bool fromCamera, const LightFilterList *lightFilterList, float rayDirFootprint,
422 float *pdf) const
423{
424 MNRY_ASSERT(mOn);
425
426 Color radiance = mRadiance;
427
428 if (mDistribution) {
429 // Note: the distribution doesn't contribute to the pdf, since we ignore it when sampling
430 radiance *= mDistribution->eval(isect.uv[0], isect.uv[1], 0, mTextureFilter);
431 }
432
433 if (lightFilterList) {
434 evalLightFilterList(lightFilterList,
435 { tls, &isect, getPosition(time),
436 getDirection(time), p,
437 filterR, time,
438 getXformRender2Local(time, lightFilterList->needsLightXform()),
439 wi
440 },
441 radiance);
442 }
443
444 // To evaluate the pdf, we need information about which sampling scheme was used (or, in the case of the
445 // intersect() function, which one would have been used for the given geometric config) -
446 // (a) point outside sphere, cosine-weighted sampling over front-facing surface
447 // (b) point inside sphere, uniform sampling over hemisphere above local horizon
448 // (c) point inside sphere, uniform sampling over full sphere
449 // Cases (b) and (c) only apply to reverse sidedness. In case (a), whether normal or reverse sidedness,
450 // we use isect.data[0] and [1] to pass a pair of intermediate values through to this point for computing
451 // the pdf. In cases (b) and (c), we pass the pdf directly in isect.data[0], and use a negative value in
452 // isect.data[1] to signal this.
453 if (pdf) {
454 float t = isect.data[1];
455 if (t >= 0.0f) {
456 // See Dutre's "Global Illumination Compendium" section 35, page 20.
457 float costThetaOverSinThetaSq = isect.data[0];
458 MNRY_ASSERT(scene_rdl2::math::isfinite(costThetaOverSinThetaSq));
459 *pdf = costThetaOverSinThetaSq / (sPi * mRadiusSqr);
460 *pdf *= areaToSolidAngleScale(wi, isect.N, t);
461 MNRY_ASSERT(scene_rdl2::math::isfinite(*pdf));
462 } else {
463 // t<0 is used to signal that isect.data[0] stores the pdf directly
464 *pdf = isect.data[0];
465 }
466 }
467
468 return radiance;
469}
