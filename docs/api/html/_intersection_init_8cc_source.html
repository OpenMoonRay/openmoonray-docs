22void initIntersectionPhase1(shading::Intersection &isect,
23 mcrt_common::ThreadLocalState *tls,
24 const mcrt_common::Ray &ray,
25 const scene_rdl2::rdl2::Layer *pRdlLayer)
26{
27 EXCL_ACCUMULATOR_PROFILE(tls, EXCL_ACCUM_INIT_INTERSECTION);
28
29 isect.init();
30 isect.setP(ray.getOrigin() + ray.getDirection() * ray.getEnd());
31
32 // Invoke the primitive's post-intersection
33 geom::internal::BVHUserData* userData =
34 static_cast<geom::internal::BVHUserData*>(ray.ext.userData);
35 const geom::internal::Primitive* primPtr = userData->mPrimitive;
36
37 primPtr->postIntersect(*tls, pRdlLayer, ray, isect);
38
39 // shadow ray epsilon is the same for all geometry primitive types.
40 isect.setShadowEpsilonHint(primPtr->getRdlGeometry()->getShadowRayEpsilon());
41
42 scene_rdl2::math::Vec3f dNds, dNdt;
43 const bool curvatureComputed =
44 primPtr->computeIntersectCurvature(ray, isect, dNds, dNdt);
45
46 if (curvatureComputed) {
47 isect.setdNds(dNds);
48 isect.setdNdt(dNdt);
49 } else {
50 isect.setdNds(scene_rdl2::math::zero);
51 isect.setdNdt(scene_rdl2::math::zero);
52 }
53
54 // We are done setting/overriding instance attributes, we, can now verify if
55 // all required attributes are provided.
56 isect.validateRequiredAttributes();
57
58 // if the ray hits an instance, we need to apply local to render space
59 // transform attached on ray to intersection
60 if (ray.isInstanceHit()) {
61 // TODO for primitive attributes that need to stay in
62 // render space, we need to also apply correspoinding transform
63 // on them
64
65 // Override shadow ray epsilon from instance if it exists
66 if (internal::hasInstanceAttr(shading::StandardAttributes::sShadowRayEpsilon, ray)) {
67 const float shadowRayEpsilon =
68 internal::getInstanceAttr(shading::StandardAttributes::sShadowRayEpsilon,
69 ray,
70 primPtr->getRdlGeometry()->getShadowRayEpsilon());
71 isect.setShadowEpsilonHint(shadowRayEpsilon);
72 }
73
74 // Only transform dPds, dPdt, and N if they aren't explicitly
75 // set on the instance's primitive attribute table
76 if (!internal::hasInstanceAttr(shading::StandardAttributes::sdPds, ray)) {
77 isect.setdPds(transformVector(ray.ext.l2r, isect.getdPds()));
78 }
79 if (!internal::hasInstanceAttr(shading::StandardAttributes::sdPdt, ray)) {
80 isect.setdPdt(transformVector(ray.ext.l2r, isect.getdPdt()));
81 }
82 const scene_rdl2::math::Xform3f r2l = ray.ext.l2r.inverse();
83 if (!internal::hasInstanceAttr(shading::StandardAttributes::sNormal, ray)) {
84 isect.setNg(normalize(transformNormal(r2l, isect.getNg())));
85 isect.setN(normalize(transformNormal(r2l, isect.getN())));
86 }
87 if (curvatureComputed) {
88 isect.setdNds(transformNormal(r2l, isect.getdNds()));
89 isect.setdNdt(transformNormal(r2l, isect.getdNdt()));
90 }
91 }
92
93 // Are we entering or leaving the volume enclosed by the surface hit ?
94 // TODO: does this affect hair shading in any way ?
95 if (isect.isFlatPoint() || dot(isect.getNg(), ray.getDirection()) < 0.0f) {
96 isect.setIsEntering(true);
97 } else {
98 // Flip both geometric and shading normal towards the "viewer"
99 isect.setN(-isect.getN());
100 isect.setNg(-isect.getNg());
101 isect.setIsEntering(false);
102 }
103}

105void initIntersectionPhase2(shading::Intersection &isect,
106 mcrt_common::ThreadLocalState *tls,
107 int mirrorDepth,
108 int glossyDepth,
109 int diffuseDepth,
110 bool isSubsurfaceAllowed,
111 const scene_rdl2::math::Vec2f &minRoughness,
112 const scene_rdl2::math::Vec3f &wo)
113{
114 EXCL_ACCUMULATOR_PROFILE(tls, EXCL_ACCUM_INIT_INTERSECTION);
115
116 shading::Intersection::PathType type;
117 if (diffuseDepth > 0) {
118 type = shading::Intersection::IndirectDiffuse;
119 } else if (glossyDepth > 0) {
120 type = shading::Intersection::IndirectGlossy;
121 } else if (mirrorDepth > 0) {
122 type = shading::Intersection::IndirectMirror;
123 } else {
124 // TODO: Add light path case
125 type = shading::Intersection::Primary;
126 }
127
128 isect.setPathType(type);
129
130 const bool isCausticPath = diffuseDepth > 0;
131 isect.setIsCausticPath(isCausticPath);
132
133 isect.setIsSubsurfaceAllowed(isSubsurfaceAllowed);
134 isect.setMinRoughness(minRoughness);
135 isect.setWo(wo);
136 isect.setGeomInitialized();
137 isect.setUseAdaptNormal(true);
138}

140void initIntersectionFull(shading::Intersection &isect,
141 mcrt_common::ThreadLocalState *tls,
142 const mcrt_common::Ray &ray,
143 const scene_rdl2::rdl2::Layer *pRdlLayer,
144 int mirrorDepth,

146 int diffuseDepth,
147 bool isSubsurfaceAllowed,
148 const scene_rdl2::math::Vec2f &minRoughness,
149 const scene_rdl2::math::Vec3f &wo)
150{
151 initIntersectionPhase1(isect, tls, ray, pRdlLayer);
152 initIntersectionPhase2(isect,
153 tls,
154 mirrorDepth,
155 glossyDepth,
156 diffuseDepth,
157 isSubsurfaceAllowed,
158 minRoughness, wo);
159
160 MNRY_ASSERT(isect.getGeomInitialized());
161}
