20///
21/// @class
22/// @brief The BSpline class is a CubicSpline primitive with overlapping
23/// spans. Each span shares 3 control points with the subsequent span.
24/// We only handle uniform cubic b-splines.
25///
26/// Here is a visual of the span layout:
27///
28/// *--*--*--* span1
29/// *--*--*--* span2
30
31class BSpline : public CubicSpline
32{
33public:
34 BSpline(Curves::Type type,
35 geom::Curves::SubType subtype,
36 geom::Curves::CurvesVertexCount&& curvesVertexCount,
37 geom::Curves::VertexBuffer&& vertices,
38 LayerAssignmentId&& layerAssignmentId,
39 shading::PrimitiveAttributeTable&& primitiveAttributeTable);
40
41private:
42
43 virtual uint32_t getSpansInChain(const uint32_t chain) const override
44 {
45 return mCurvesVertexCount[chain] - 3;
46 }
47
48 // get weights for each of the 4 control points of a span
49 virtual void evalWeights(const float& t, float& w0, float& w1, float& w2, float& w3) const override
50 {
51 const float s = 1.f - t;
52 const float oneSixth(1.0f/6.0f);
53
54 w0 = oneSixth * s * s * s;
55 w1 = 0.5f * s * (s * t + 1.0f) + oneSixth;
56 w2 = 0.5f * t * (s * t + 1.0f) + oneSixth;
57 w3 = oneSixth * t * t * t;
58 }
59
60 // get weights for the difference of neighboring controls points
61 // to scale (p1 - p0), (p2 - p1), and (p3 - p2)
62 virtual void evalDerivWeights(const float& t, float& w0, float& w1, float& w2) const override
63 {
64 const float s = 1.f - t;
65 w0 = 0.5f * s * s;
66 w1 = s * t + 0.5f;
67 w2 = 0.5f * t * t;
68 }
69
70};
