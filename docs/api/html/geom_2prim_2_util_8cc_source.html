18//analytic solution of SVD based condition number calculation
19float tessCondNumber2x2SVD(
20 float a, float b,
21 float c, float d,
22 float aTolerance
23 )
24{
25 MNRY_ASSERT(aTolerance>0.0, "need a positive tolerance");
26
27 //compute AA^T = AAT_j AAT_l
28 // AAT_l AAT_k
29 float AAT_j = a * a + b * b;
30 float AAT_l = a * c + b * d;
31 float AAT_k = c * c + d * d;
32
33 float jmk = AAT_j - AAT_k;
34 float jpk = AAT_j + AAT_k;
35 float s = jmk * jmk + 4 * AAT_l * AAT_l;
36 s = sqrt(s);
37 float d1 = sqrt(0.5 * (jpk + s));
38
39 float diff=0.5*(jpk - s);
40 float d2;
41 if(diff>0.0) {
42 d2 = sqrt(diff);
43 } else {
44 // mathematically diff is always positive or null
45 // but due to numerical precision, it can be negative
46 d2 = 0.0;
47 }
48
49 if (d1 >= d2 && fabs(d2) > aTolerance) {
50 return d1 / d2;
51 }
52 if (d1 < d2 && fabs(d1) > aTolerance) {
53 return d2 / d1;
54 }
55
56 return gIllConditioned; //failed, almost singular or singular matrix
57}
58
59/*
60 * Convert the u, v partial surface derivatives to derivatives
61 * with respect to the texture coordinates s, t.
62 *
63 * Given: dP/d(u,v) P(u,v) -> [x,y,z]
64 * dT/d(u,v) T(u,v) -> [s,t]
65 *
66 * We want: dP/d(s,t)
67 *
68 * Notice that we want to take P composed with T:
69 *
70 * P(u,v) = P(T^{-1}(T(u,v)))
71 * = P(T^{-1}(s,t))
72 *
73 * What we actually want is the partials if P o T{^-1}
74 * Which turns out to be the column vectors of the Jacobian:
75 *
76 * Jacobian: J_{P o T^{-1}}(s,t)
77 *
78 * Note we also have (from inputs of the function) the following Jacobians:
79 *
80 * J_P(u,v) from dP/d(u,v)
81 * and J_T(u,v) from dT/d(u,v)
82 *
83 * Using the Chain Rule we get:
84 *
85 * J_{P o T^{-1}}(s,t) = J_P(T^{-1}(s,t)) * J_T^{-1}(s,t)
86 *
87 * = J_P(u,v) * J_T^{-1}(s,t)
88 *
89 * Using the Inverse Function thereom:
90 *
91 * J_T^{-1}(s,t) = [J_T(u, v)]^{-1}
92 *
93 * I.E. The Jacobian of the function F^{-1} is simply the matrix inverse of
94 * the Jacobian of the function F.
95 *
96 * With substitution we get:
97 *
98 * J_{P o T^{-1}}(s,t) = J_P(u,v) * [J_T(u, v)]^{-1}
99 *
100 * Thus to get the partials dP/d(s,t), we just have to invert J_T(u,v)
101 * and evaluate the matrix multiplies and take the column vectors of
102 * J_{P o T^{-1}}(s,t).
103 *
104 *
105 */
106bool computePartialsWithRespect2Texture(const Vec3f &dP_du,
107 const Vec3f &dP_dv,
108 const Vec2f &dT_du,
109 const Vec2f &dT_dv,
110 Vec3f &dP_ds,
111 Vec3f &dP_dt)
112{
113
114 float detInv;
115 float det = dT_du[0] * dT_dv[1] - dT_du[1] * dT_dv[0];
116
117 const float tolerance=1.e-12;
118
119 //use condition number to decide if matrix is ill conditioned
120 //note 0 is an estimate of condition number Manny put in
121 //1 is a SVD based, accurate calculation
122 //we are keeping both versions just to be able to switch between them if there is performance concerns
123#if 0
124 const float condnum = condNumber2x2(
125 dT_du[0], dT_du[1], // a, b
126 dT_dv[0], dT_dv[1]); // c, d
127#else
128 const float condnum = tessCondNumber2x2SVD(
129 dT_du[0], dT_du[1], // a, b
130 dT_dv[0], dT_dv[1],
131 tolerance); // c, d
132#endif
133
134 if (condnum >= gIllConditioned || fabs(det) < tolerance) {
135// if(!scene_rdl2::math::isnormal(det)) {
136
137 dP_ds[0] = dP_du[0];
138 dP_ds[1] = dP_du[1];
139 dP_ds[2] = dP_du[2];
140 dP_dt[0] = dP_dv[0];
141 dP_dt[1] = dP_dv[1];
142 dP_dt[2] = dP_dv[2];
143
144 return false;
145 }
146
147 detInv = 1.0 / det;
148
149
150 dP_ds[0] = detInv * (dP_du[0] * dT_dv[1] - dP_dv[0] * dT_du[1]);
151 dP_ds[1] = detInv * (dP_du[1] * dT_dv[1] - dP_dv[1] * dT_du[1]);
152 dP_ds[2] = detInv * (dP_du[2] * dT_dv[1] - dP_dv[2] * dT_du[1]);
153
154 dP_dt[0] = detInv * (-dP_du[0] * dT_dv[0] + dP_dv[0] * dT_du[0]);
155 dP_dt[1] = detInv * (-dP_du[1] * dT_dv[0] + dP_dv[1] * dT_du[0]);
156 dP_dt[2] = detInv * (-dP_du[2] * dT_dv[0] + dP_dv[2] * dT_du[0]);
157
158 return true;
159}

161void writeToObj(const std::string& outputPath,
162 const std::vector<uint32_t>& faceVertexCount,
163 const VertexBuffer<Vec3fa, InterleavedTraits>& vertices,
164 const std::vector<uint32_t>& indices,
165 const shading::Vector<Vec2f>& textureVertices,
166 const std::vector<uint32_t>& textureIndices)
167{
168 std::ofstream filestream;
169 filestream.open(outputPath);
170 filestream << "# vertices" << std::endl;
171 for (size_t v = 0; v < vertices.size(); ++v) {
172 filestream << "v " <<
173 vertices(v).x << " " <<
174 vertices(v).y << " " <<
175 vertices(v).z << std::endl;
176 }
177 filestream << std::endl;
178 filestream << "# texture coordinates" << std::endl;
179 for (size_t v = 0; v < textureVertices.size(); ++v) {
180 filestream << "vt " <<
181 textureVertices[v].x << " " <<
182 textureVertices[v].y << std::endl;
183 }
184 filestream << std::endl;
185 filestream << "# face indices" << std::endl;
186 size_t offset = 0;
187 for (size_t f = 0; f < faceVertexCount.size(); ++f) {
188 uint32_t nFv = faceVertexCount[f];
189 filestream << "f ";
190 for (size_t v = 0; v < nFv; ++v) {
191 filestream <<
192 (indices[offset + v] + 1) << "/" <<
193 (textureIndices[offset + v] + 1) << " ";
194 }
195 filestream << std::endl;
196 offset += nFv;
197 }
198 filestream << std::endl;
199 filestream.close();
200}

202Vec3f
203computePrimitiveMotion(const Vec3f &pos0, const Vec3f *pos1, float rayTime,
204 const Instance *instance)
205{
206 if (!pos1 && !instance) return Vec3f(0.0f); // no motion
207
208 Vec3f p0 = pos0;
209 Vec3f p1 = pos1? *pos1 : p0;
210
211 if (instance) {
212 const MotionTransform &xform = instance->getLocal2Parent();
213 if (xform.isStatic()) {
214 if (!pos1) return Vec3f(0.0f); // no motion
215
216 const Mat43 l2p = xform.getStaticXform();
217 p0 = transformPoint(l2p, p0);
218 p1 = transformPoint(l2p, p1);
219 } else {
220 const Mat43 l2p0 = xform.eval(rayTime - sHalfDt);
221 const Mat43 l2p1 = xform.eval(rayTime + sHalfDt);
222 p0 = transformPoint(l2p0, p0);
223 p1 = transformPoint(l2p1, p1);
224 }
225 }
226
227 const Vec3f motion = (p1 - p0) / (2.0f * sHalfDt);
228 return motion;
229}

231Vec3f transformPoint(const scene_rdl2::math::Mat4f mat[2], const scene_rdl2::math::Vec3f& point, const float time, bool isMotionBlurOn)
232{
233 if (isMotionBlurOn) {
234 scene_rdl2::math::Vec3f point0 = scene_rdl2::math::transformPoint(mat[0], point);
235 scene_rdl2::math::Vec3f point1 = scene_rdl2::math::transformPoint(mat[1], point);
236 return scene_rdl2::math::lerp(point0, point1, time);
237 } else {
238 return scene_rdl2::math::transformPoint(mat[0], point);
239 }
240}
