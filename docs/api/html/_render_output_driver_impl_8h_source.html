27// files contain 1 or more (sub-)images (parts). the
28// case of 1 un-named image per file is allowed, and is typically
29// thought of as the "normal" case. if more than one image
30// is present in a file, all sub-images must be named. a particular
31// image may have only 1 compression scheme. only a single render output
32// object may target a particular file/image/channel tripple.
33
34// an Entry is basically a render output that has been processed
35// to include resolved channel names and aov schema
36struct Entry
37{
38 explicit Entry(const scene_rdl2::rdl2::RenderOutput *renderOutput):
39 mRenderOutput(renderOutput),
40 mAovSchemaId(pbr::AOV_SCHEMA_ID_UNKNOWN),
41 mLpePrefixFlags(pbr::AovSchema::sLpePrefixNone)
42 {
43 }
44
45#ifdef DEBUG_DUMP_ENTRIES_AND_FILES
46 std::string showChannelNames(const std::string &hd) const { // for debug
47 std::ostringstream ostr;
48 ostr << hd << "Entry mChannelNames (total:" << mChannelNames.size() << ") {\n";
49 for (size_t i = 0; i < mChannelNames.size(); ++i) {
50 ostr << hd << " i:" << i << " name:" << mChannelNames[i] << '\n';
51 }
52 ostr << hd << "}";
53 return ostr.str();
54 }
55#endif // end DEBUG_DUMP_ENTRIES_AND_FILES
56
57 const scene_rdl2::rdl2::RenderOutput *mRenderOutput;
58 std::vector<std::string> mChannelNames; // resolved channel names
59 int mAovSchemaId; // aov schema id, if any
60 int mLpePrefixFlags; // lpe prefix flags, if any
61};

73};

75struct File
76{
77 explicit File(const std::string &name) :
78 mName(name),
79 mCheckpointName(""),
80 mCheckpointMultiVersionName(""),
81 mResumeName("")
82 {}
83
84 // realpath() of output file
85 std::string mName;
86
87 // realpath() of checkpoint output file : If empty, should not output checkpoint file.
88 std::string mCheckpointName;
89
90 // realpath() of checkpoint multi-version output file. This is used for checkpoint file
91 // overwrite = off condition. If this value is empty, all non-overwrite checkpoint files
92 // go to same location of mCheckpointName
93 std::string mCheckpointMultiVersionName;
94
95 // realpath() of resume input file : If empty, should not input resume file.
96 std::string mResumeName;
97
98 std::vector<Image> mImages; // images in file
99};
