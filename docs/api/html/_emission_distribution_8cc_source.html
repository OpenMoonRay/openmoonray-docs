17float
18EmissionDistribution::pdf(const Transform &xform, const Vec3f& pRender, const Vec3f& wiRender,
19 float& tEnd, float time) const
20{
21 // we do the DDA traversal in grid space
22 Vec3f p, wi;
23 if (mIsMotionBlurOn) {
24 p = scene_rdl2::math::transformPoint( xform.getRenderToDist(time), pRender);
25 wi = scene_rdl2::math::transformVector(xform.getRenderToDist(time), wiRender);
26 } else {
27 p = scene_rdl2::math::transformPoint( xform.getRenderToDist()[0], pRender);
28 wi = scene_rdl2::math::transformVector(xform.getRenderToDist()[0], wiRender);
29 }
30 float lengthScale = 1.0f / wi.length();
31 wi *= lengthScale;
32
33 // bounding box intersection test
34 float tStart = 0.0f;
35 tEnd = FLT_MAX;
36 for (int axis = 0; axis < 3; ++axis) {
37 float invDir = 1.0f / wi[axis];
38 float tnear = (0 - p[axis]) * invDir;
39 float tfar = (mRes[axis] - p[axis]) * invDir;
40 if (tnear > tfar) {
41 std::swap(tnear, tfar);
42 }
43 tStart = tnear > tStart ? tnear : tStart;
44 tEnd = tfar < tEnd ? tfar : tEnd;
45 // the ray doesn't hit bounding box
46 if (tStart > tEnd) {
47 return 0.0f;
48 }
49 }
50 // the tEnd used for later line integral computation is in render space
51 tEnd *= lengthScale;
52 // the tStart used for grid traversal is in local grid space
53 p = p + tStart * wi;
54 float nextT[3];
55 float deltaT[3];
56 int step[3];
57 int out[3];
58 scene_rdl2::math::Vec3i pos;
59 for (int axis = 0; axis < 3; ++axis) {
60 // the clamping below prevents resolved intersection point falls out of
61 // bounding box due to float point precision issue
62 pos[axis] = scene_rdl2::math::clamp(static_cast<int>(p[axis]), 0, mRes[axis] - 1);
63 if (wi[axis] >= 0) {
64 nextT[axis] = tStart + (pos[axis] + 1 - p[axis]) / wi[axis];
65 deltaT[axis] = 1.0f / wi[axis];
66 step[axis] = 1;
67 out[axis] = mRes[axis];
68 } else {
69 nextT[axis] = tStart + (pos[axis] - p[axis]) / wi[axis];
70 deltaT[axis] = -1.0f / wi[axis];
71 step[axis] = -1;
72 out[axis] = -1;
73 }
74 }
75
76 float tCurrent = tStart;
77 float accumedPdf = 0.0f;
78 while (true) {
79 float pdfD = pdfDiscrete(pos);
80 int stepAxis =
81 ((nextT[1] < nextT[0]) | (nextT[2] < nextT[0])) <<
82 (nextT[2] < nextT[1]);
83 if (pdfD > 0.f) {
84 float t0 = tCurrent * lengthScale;
85 float t1 = nextT[stepAxis] * lengthScale;
86 // the (t1^3 - t0^3) / 3 part stands for integrating 3d space pdf
87 // along the line segment enter/exit grid voxel with Jacobian transform
88 // For detail reference see:
89 // "Line Integration for Rendering Heterogeneous Emissive Volume"
90 // EGSR2017 Florian Simon
91 // equation (15) and Algorithm 2
92 accumedPdf += pdfD * xform.getInvUnitVolume() *
93 (t1 * t1 * t1 - t0 * t0 * t0) / 3.0f;
94 }
95 tCurrent = nextT[stepAxis];
96 pos[stepAxis] += step[stepAxis];
97 // getting out of the bounding box
98 if (pos[stepAxis] == out[stepAxis]) {
99 break;
100 }
101 nextT[stepAxis] += deltaT[stepAxis];
102 }
103 return accumedPdf;
104}

106void
107DenseEmissionDistribution::sample(const Transform &xform, const Vec3f& p, float u1, float u2, float u3,
108 Vec3f& wi, float& pdfWi, float& tEnd, float time) const
109{
110 float pdfIndex;
111 float uxRemapped, uyRemapped, uzRemapped;
112 scene_rdl2::math::Vec3i coord = mDistribution->sampleDiscrete(u1, u2, u3,
113 &pdfIndex, &uxRemapped, &uyRemapped, &uzRemapped);
114 Vec3f pDist(
115 coord[0] + uxRemapped,
116 coord[1] + uyRemapped,
117 coord[2] + uzRemapped);
118 Vec3f pt = mIsMotionBlurOn ? scene_rdl2::math::transformPoint(xform.getDistToRender(time), pDist) :
119 scene_rdl2::math::transformPoint(xform.getDistToRender()[0], pDist);
120 wi = normalize(pt - p);
121 pdfWi = pdf(xform, p, wi, tEnd, time);
122}

145SparseEmissionDistribution::SparseEmissionDistribution(
146 const scene_rdl2::math::Vec3i& res,
147 const scene_rdl2::math::Mat4f distToRender[2],
148 float invUnitVolume,
149 const std::map<uint32_t, float>& histogram):
150 EmissionDistribution(res, distToRender, invUnitVolume)
151{
152 int n = histogram.size();
153 auto it = histogram.begin();
154
155 // reset cdf
156 mCdf.clear();
157 mCdf.reserve(n + 1);
158
159 // reset offsets
160 mOffsets.clear();
161
162 // first entry
163 mOffsets[it->first] = mCdf.size();
164 mCdf.push_back({it->first, 0.f});
165
166 // fill in cdf and offsets
167 for (int i = 1; i < n; ++i, ++it) {
168 const auto nextIt = std::next(it);
169 mOffsets[nextIt->first] = mCdf.size();
170 std::pair<uint32_t, float> cdf = {nextIt->first, mCdf[i-1].second + it->second / n};
171 mCdf.push_back(cdf);
172 }
173
174 // last entry has a dummy offset
175 uint32_t dummyOffset = mCdf[n-1].first + 1;
176 mOffsets[dummyOffset] = mCdf.size();
177 std::pair<uint32_t, float> cdf = {dummyOffset, mCdf[n-1].second + it->second / n};
178 mCdf.push_back(cdf);
179
180 // normalize cdf
181 float finalSum = mCdf[n].second;
182 if (finalSum == 0) {
183 int i = 0;
184 for (auto& cdf : mCdf) {
185 cdf.second = float(i++) / float(n);
186 }
187 } else {
188 for (auto& cdf : mCdf) {
189 cdf.second /= finalSum;
190 }
191 }
192}

194finline Vec3f
195SparseEmissionDistribution::indexTo3DCoord(uint32_t index) const
196{
197 int z = index / (mRes[0] * mRes[1]);
198 index = index % (mRes[0] * mRes[1]);
199 int y = index / mRes[0];
200 int x = index % mRes[0];
201 return Vec3f(x, y, z);
202}

204Vec3f
205SparseEmissionDistribution::sample(const Transform &xform, float u1, float u2, float u3, float time) const
206{
207 // get cdf sample
208 int offset = findInterval(mCdf.size(),
209 [&](int index) {
210 return mCdf[index].second <= u1;
211 });
212 const std::pair<uint32_t, float>& cdf0 = mCdf[offset];
213 const std::pair<uint32_t, float>& cdf1 = mCdf[offset + 1];
214
215 // transform offset into 3d coordinate
216 Vec3f pDist = indexTo3DCoord(cdf0.first);
217 float uxRemapped = (u1 - cdf0.second) / (cdf1.second - cdf0.second);
218 pDist.x += uxRemapped;
219 pDist.y += u2;
220 pDist.z += u3;
221
222 return mIsMotionBlurOn ? scene_rdl2::math::transformPoint(xform.getDistToRender(time), pDist) :
223 scene_rdl2::math::transformPoint(xform.getDistToRender()[0], pDist);
224}

234float
235SparseEmissionDistribution::pdfDiscrete(const scene_rdl2::math::Vec3i& pos) const
236{
237 // Get index into 3D space
238 int offset = pos[0] + pos[1] * mRes[0] + pos[2] * mRes[0] * mRes[1];
239 const auto it = mOffsets.find(offset);
240 if (it == mOffsets.end()) {
241 // this position has no value;
242 return 0.f;
243 } else {
244 // get index into cdf
245 size_t index = it->second;
246 return mCdf[index + 1].second - mCdf[index].second;
247 }
248}
