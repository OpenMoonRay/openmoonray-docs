49 /** \brief 4x4 Column Major Matrix
50 *
51 * Matrix formats can be confusing. When we say <b>column major</b> it's only meaningful when we
52 * are discussing it in certain context, such as transformations. <b>We are not referring to the underlying
53 * storage of the matrix data</b>, which could be in any form (arrays, vectors, etc.). Regardless of row major
54 * or column major, multiplications between matrices do not change (multiply a row by a column).
55 * However, when we define transformations of point, vectors, etc., we need to be clear about the meanings
56 * of the rows and columns of a matrix and their corresponding multiplication with vectors.
57 *
58 * DWA specifies the transformation matrix in column major, so vector transformation is applied using
59 * post-multiply. Normal transformation is done using pre-multiply. These are defined as follows.
60 *
61 * <b>Row vector</b> means vectors are defined as v = [x, y, z, w], also can be considered as a 1 by n matrix, where n = 4.
62 *
63 * Column vector is the row vector transposed, which is a n by 1 matrix.
64 *
65 * Matrix multiply for a row vector can be done by post-multiply, where the matrix is after the row vector:
66 *
67 * u = v * M (equivalent to transpose(M) * transpose(v))
68 *
69 * Or it's possible to do pre-multiply, which implicitly transposes the vector v into a column vector:
70 *
71 * w = M * v (where v is assumed to be a column vector)
72 *
73 * In general, u != w (note the missing transpose for M), because matrix multiplication is not commutative.
74 *
75 * The set of <b>transform*()</b> interface should be preferable over multiplication since their intentions are unambiguous.
76 */

130 /*! compute the determinant of the matrix.
131 *
132 * We assume most of the cases the last column of the matrix will be [0, 0, 0, 1], in which case
133 * we only need to compute one 3x3 determinant.
134 */
135 __forceinline const Scalar det() const {
136 typedef Mat3<Vec3<Scalar> > Mat3f;
137 const Scalar d1 = (vx.w == 0) ? 0 : -vx.w * Mat3f(vy.toVec3(),vz.toVec3(),vw.toVec3()).det();
138 const Scalar d2 = (vy.w == 0) ? 0 : vy.w * Mat3f(vx.toVec3(),vz.toVec3(),vw.toVec3()).det();
139 const Scalar d3 = (vz.w == 0) ? 0 : -vz.w * Mat3f(vx.toVec3(),vy.toVec3(),vw.toVec3()).det();
140 const Scalar d4 = (vw.w == 0) ? 0 : vw.w * Mat3f(vx.toVec3(),vy.toVec3(),vz.toVec3()).det();
141 return d1 + d2 + d3 + d4;
142 }
143
144 /*! compute adjoint matrix */
145 __forceinline const Mat4 adjoint() const {
146 typedef Mat3<Vec3<Scalar> > Mat3f;
147 const Scalar m00 = Mat3f(vy.toVec3(1,2,3),vz.toVec3(1,2,3),vw.toVec3(1,2,3)).det();
148 const Scalar m01 = -Mat3f(vy.toVec3(0,2,3),vz.toVec3(0,2,3),vw.toVec3(0,2,3)).det();
149 const Scalar m02 = Mat3f(vy.toVec3(0,1,3),vz.toVec3(0,1,3),vw.toVec3(0,1,3)).det();
150 const Scalar m03 = -Mat3f(vy.toVec3(0,1,2),vz.toVec3(0,1,2),vw.toVec3(0,1,2)).det();
151 const Scalar m10 = -Mat3f(vx.toVec3(1,2,3),vz.toVec3(1,2,3),vw.toVec3(1,2,3)).det();
152 const Scalar m11 = Mat3f(vx.toVec3(0,2,3),vz.toVec3(0,2,3),vw.toVec3(0,2,3)).det();
153 const Scalar m12 = -Mat3f(vx.toVec3(0,1,3),vz.toVec3(0,1,3),vw.toVec3(0,1,3)).det();
154 const Scalar m13 = Mat3f(vx.toVec3(0,1,2),vz.toVec3(0,1,2),vw.toVec3(0,1,2)).det();
155 const Scalar m20 = Mat3f(vx.toVec3(1,2,3),vy.toVec3(1,2,3),vw.toVec3(1,2,3)).det();
156 const Scalar m21 = -Mat3f(vx.toVec3(0,2,3),vy.toVec3(0,2,3),vw.toVec3(0,2,3)).det();
157 const Scalar m22 = Mat3f(vx.toVec3(0,1,3),vy.toVec3(0,1,3),vw.toVec3(0,1,3)).det();
158 const Scalar m23 = -Mat3f(vx.toVec3(0,1,2),vy.toVec3(0,1,2),vw.toVec3(0,1,2)).det();
159 const Scalar m30 = -Mat3f(vx.toVec3(1,2,3),vy.toVec3(1,2,3),vz.toVec3(1,2,3)).det();
160 const Scalar m31 = Mat3f(vx.toVec3(0,2,3),vy.toVec3(0,2,3),vz.toVec3(0,2,3)).det();
161 const Scalar m32 = -Mat3f(vx.toVec3(0,1,3),vy.toVec3(0,1,3),vz.toVec3(0,1,3)).det();
162 const Scalar m33 = Mat3f(vx.toVec3(0,1,2),vy.toVec3(0,1,2),vz.toVec3(0,1,2)).det();
163
164 // transposed
165 return Mat4(m00,m10,m20,m30,
166 m01,m11,m21,m31,
167 m02,m12,m22,m32,
168 m03,m13,m23,m33);
169 }

255 /*! return matrix for rotation around arbitrary axis
256 * @param u the axis of rotation
257 * @param r the angle of rotation in radian
258 */
259 static __forceinline Mat4 rotate(const Vector& u, const Scalar& r) {
260 Vector v = normalize(u);
261 Scalar s = sin(r);
262 Scalar c = cos(r);
263 Scalar t = 1-c;
264 return Mat4(v.x*v.x*t+c, v.x*v.y*t+v.z*s, v.x*v.z*t-v.y*s, 0,
265 v.y*v.x*t-v.z*s, v.y*v.y*t+c, v.y*v.z*t+v.x*s, 0,
266 v.z*v.x*t+v.y*s, v.z*v.y*t-v.x*s, v.z*v.z*t+c, 0,
267 0, 0, 0, 1);
268 }

270 /*! return an orthonormal version of the supplied matrix */
271 static __forceinline Mat4 orthonormalize(const Mat4 &m) {
272 Mat4 r;
273 asVec3(r.vx) = normalize(cross(asVec3(m.vy), asVec3(m.vz)));
274 asVec3(r.vy) = normalize(cross(asVec3(m.vz), asVec3(r.vx)));
275 asVec3(r.vz) = cross(asVec3(r.vx), asVec3(r.vy));
276 r.vx.w = 0.f;
277 r.vy.w = 0.f;
278 r.vz.w = 0.f;
279 r.vw = m.vw;
280 return r;
281 }

328 /// Slerp between two 4x4 affine transformation matrices
329 ///
330 /// WARNING 1: In addition to interpolation, this function performs matrix decomposition
331 /// and quaternion conversions on the input matrices. You should NOT use this function if
332 /// you need to perform many interpolations on the same pair of matrices. In that that case you
333 /// should decompose the matrices yourself into XformComponent classes and then slerp using
334 /// the XformComponent classes.
335 ///
336 /// WARNING 2: This produces undefined results on non-affine matrices
337 /// (i.e. perspective projection matrices)
338 template<typename T, typename U> __forceinline Mat4<Vec4<T> > slerp(const Mat4<Vec4<T> >& a, const Mat4<Vec4<T> >& b, const U t) {
339 typedef Mat3<Vec3<T> > Mat33T;
340 typedef XformT<Mat33T> Xform33T;
341 typedef XformComponent<Mat33T> XformComponent33T;
342
343 XformComponent33T ca;
344 XformComponent33T cb;
345 decompose(xform<Xform33T>(a), ca);
346 decompose(xform<Xform33T>(b), cb);
347 if (dot(ca.r, cb.r) < 0) { cb.r *= U(-1.0); }
348 return Mat4<Vec4<T> >(slerp(ca, cb, t).combined());
349 }

441 /// Transform a Vec3 by post-multiplication, doing homogeneous divison.
442 template<typename VectorT> __forceinline Vec3<typename VectorT::Scalar> transformH(const Mat4<VectorT>& m, const Vec3<typename VectorT::Scalar>& p) {
443 Vec4<typename VectorT::Scalar> v = transform(m, VectorT(p.x,p.y,p.z,1));
444 if (v.w != typename VectorT::Scalar(0)) return Vec3<typename VectorT::Scalar>(v.x / v.w, v.y / v.w, v.z / v.w);

446 }

448 /// Transforms an AABB to an AABB. This is based on this:
449 /// http://dev.theomader.com/transform-bounding-boxes/
450 __forceinline BBox3f
451 transformBBox(const Mat4<Vec4f>& m, const BBox3f& bb)
452 {
453 const auto xa = asVec3(m.row0() * bb.lower.x);
454 const auto xb = asVec3(m.row0() * bb.upper.x);
455
456 const auto ya = asVec3(m.row1() * bb.lower.y);
457 const auto yb = asVec3(m.row1() * bb.upper.y);
458
459 const auto za = asVec3(m.row2() * bb.lower.z);
460 const auto zb = asVec3(m.row2() * bb.upper.z);
461
462 return BBox3f(
463 min(xa, xb) + min(ya, yb) + min(za, zb) + asVec3(m.row3()),
464 max(xa, xb) + max(ya, yb) + max(za, zb) + asVec3(m.row3())
465 );
466 }
