204void
205TestXformv::testSelect()
206{
207 Xform3f scalarLhs = Xform3f::rotate(Vec3f(1, 0, 0), 20);
208 Xform3fv vecLhs = broadcast(scalarLhs);
209 Xform3f scalarRhs = Xform3f::translate(Vec3f(7, 8, 9));
210 Xform3fv vecRhs = broadcast(scalarRhs);
211 {
212 Mask maskOdd;
213 Mask maskEven;
214 for (size_t i = 0; i < VLEN; ++i) {
215 maskOdd[i] = (i & 1) == 1;
216 maskEven[i] = (i & 1) == 0;
217 }
218 Xform3fv resultOdd = select(maskOdd, vecLhs, vecRhs);
219 Xform3fv resultEven = select(maskEven, vecLhs, vecRhs);
220 for (size_t i = 0; i < VLEN; ++i) {
221 if ((i & 1) == 1) {
222 CPPUNIT_ASSERT(isEqual(getXform(resultOdd, i).l, scalarLhs.l));
223 CPPUNIT_ASSERT(isEqual(getXform(resultOdd, i).p, scalarLhs.p));
224 CPPUNIT_ASSERT(isEqual(getXform(resultEven, i).l, scalarRhs.l));
225 CPPUNIT_ASSERT(isEqual(getXform(resultEven, i).p, scalarRhs.p));
226 } else {
227 CPPUNIT_ASSERT(isEqual(getXform(resultOdd, i).l, scalarRhs.l));
228 CPPUNIT_ASSERT(isEqual(getXform(resultOdd, i).p, scalarRhs.p));
229 CPPUNIT_ASSERT(isEqual(getXform(resultEven, i).l, scalarLhs.l));
230 CPPUNIT_ASSERT(isEqual(getXform(resultEven, i).p, scalarLhs.p));
231 }
232 }
233 }
