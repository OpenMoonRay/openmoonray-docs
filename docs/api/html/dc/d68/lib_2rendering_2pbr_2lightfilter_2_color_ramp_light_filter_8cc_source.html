70void
71ColorRampLightFilter::update(const LightFilterMap& /*lightFilters*/,
72 const Mat4d& world2render)
73{
74 if (!mRdlLightFilter) {
75 return;
76 }
77
78 Mat4d render2world = world2render.inverse();
79 Mat4d local2world0 = mRdlLightFilter->get<rdl2::Mat4d>(sNodeXformKey, 0.f);
80 Mat4d local2world1 = mRdlLightFilter->get<rdl2::Mat4d>(sNodeXformKey, 1.f);
81 Mat4d world2local0 = local2world0.inverse();
82 Mat4d world2local1 = local2world1.inverse();
83 mXform[0] = Mat4f(render2world * world2local0);
84 mXform[1] = Mat4f(render2world * world2local1);
85 mUseXform = mRdlLightFilter->get<rdl2::Bool>(sUseXformKey);
86
87 float beginDistance = mRdlLightFilter->get<rdl2::Float>(sBeginDistanceKey);
88 float endDistance = mRdlLightFilter->get<rdl2::Float>(sEndDistanceKey);
89 if (beginDistance >= endDistance) {
90 mRdlLightFilter->error(
91 "Color ramp light filter begin distance is >= end distance, using defaults");
92 beginDistance = 0.f;
93 endDistance = 1.f;
94 }
95 std::vector<Color> colorsVec = mRdlLightFilter->get<rdl2::RgbVector>(sColorsKey);
96 std::vector<float> distancesVec = mRdlLightFilter->get<rdl2::FloatVector>(sDistancesKey);
97 std::vector<int> interpolationTypesVec = mRdlLightFilter->get<rdl2::IntVector>(sInterpolationTypesKey);
98
99 if (distancesVec.size() != colorsVec.size() || distancesVec.size() != interpolationTypesVec.size()) {
100 mRdlLightFilter->error(
101 "Color ramp light filter distances, colors and interpolation types are different sizes, using defaults");
102 colorsVec = {rdl2::Rgb(1.f, 1.f, 1.f), rdl2::Rgb(0.f, 0.f, 0.f)};
103 distancesVec = {0.f, 1.f};
104 interpolationTypesVec = {1, 1};
105 }
106
107 // remap distances to startDistance to endDistance range
108 float scale = endDistance - beginDistance;
109 for (int i = 0; i < distancesVec.size(); i++) {
110 distancesVec[i] = beginDistance + scale * distancesVec[i];
111 }
112
113 mColorRamp.init(
114 distancesVec.size(),
115 distancesVec.data(),
116 colorsVec.data(),
117 reinterpret_cast<const ispc::RampInterpolatorMode*>(interpolationTypesVec.data()),
118 ispc::COLOR_RAMP_CONTROL_SPACE_RGB);
119
120 mIntensity = mRdlLightFilter->get<rdl2::Float>(sIntensityKey);
121 mDensity = mRdlLightFilter->get<rdl2::Float>(sDensityKey);
122 mMode = mRdlLightFilter->get<rdl2::Int>(sModeKey);
123 mWrapMode = mRdlLightFilter->get<rdl2::Int>(sWrapModeKey);
124}

132Color
133ColorRampLightFilter::eval(const EvalData& data) const
134{
135 float dist;
136 switch (mMode) {
137 case RADIAL:
138 if (mUseXform) {
139 Vec3f xformedPoint0 = transformPoint(mXform[0], data.shadingPointPosition);
140 Vec3f xformedPoint1 = transformPoint(mXform[1], data.shadingPointPosition);
141 dist = lerp(length(xformedPoint0), length(xformedPoint1), data.time);
142 } else {
143 dist = data.isect->distance;
144 }
145 break;
146 case DIRECTIONAL:
147 if (mUseXform) {
148 Vec3f xformedPoint0 = transformPoint(mXform[0], data.shadingPointPosition);
149 Vec3f xformedPoint1 = transformPoint(mXform[1], data.shadingPointPosition);
150 Vec3f xformedPoint = lerp(xformedPoint0, xformedPoint1, data.time);
151 dist = -xformedPoint.z;
152 } else {
153 Plane plane(data.lightPosition, data.lightDirection);
154 dist = plane.getDistance(data.shadingPointPosition);
155 if (dot(data.shadingPointPosition - data.lightPosition, data.lightDirection) < 0.f) {
156 // Restores the sign of dist because plane.getDistance() returns the abs distance to the plane
157 dist *= -1.f;
158 }
159 }
160
161 switch (mWrapMode) {
162 case EXTEND:
163 // Causes f(dist) = f(0) for dist < 0
164 if (dist < 0.f) dist = 0.f;
165 break;
166 case MIRROR:
167 // Causes f(dist) = f(-dist), mirroring about dist = 0
168 dist = fabs(dist);
169 break;
170 default:
171 MNRY_ASSERT(false);
172 }
173
174 break;
175 default:
176 MNRY_ASSERT(false);
177 }
178
179 Color filterVal = mColorRamp.eval1D(dist);
180
181 // Scale the filter color value by the intensity
182 filterVal *= mIntensity;
183
184 // Apply density scaling to allow partial light filtering
185 filterVal = Color(1.f - mDensity) + filterVal * mDensity;
186
187 return filterVal;
188}
