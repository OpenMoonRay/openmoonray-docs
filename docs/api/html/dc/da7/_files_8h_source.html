17/**
18 * RAII guard that wraps a file descriptor. The descriptor will always be
19 * closed, no matter how the scope is exited.
20 *
21 * Example:
22 * {
23 * FileDescriptorGuard guard(open(...));
24 * write(guard.fd, ...);
25 * } // close(guard.fd) is automatically called
26 */
27struct FileDescriptorGuard
28{
29 FileDescriptorGuard(int descriptor) :
30 fd(descriptor)
31 {
32 }
33
34 ~FileDescriptorGuard()
35 {
36 close(fd);
37 }
38
39 const int fd;
40};

42/**
43 * Splits a file path string into its basename and dirname components. Returns
44 * a pair<string, string> where the first element is the dirname and the second
45 * element is the basename.
46 *
47 * This is a bit tricky since basename() and dirname() may or may not modify
48 * their arguments and may or may not return pointers to statically allocated
49 * space.
50 *
51 * Example:
52 * auto components = splitPath("some/file/path.txt");
53 * assert(components.first == "some/file");
54 * assert(components.second == "path.txt");
55 */
56std::pair<std::string, std::string> splitPath(const std::string& filePath);

88/**
89 * Locates a file with the given file name in the given (colon separated)
90 * search path, returning the full path to the file.
91 *
92 * This is done by checking successive elements of the search path and
93 * returning the first full path which matches an existing file.
94 *
95 * Full path means <part of the search path that matched> + '/' + name.
96 *
97 * If the search path is exhausted and the file still cannot be found, an
98 * empty string is returned.
99 *
100 * Example:
101 * auto passwd = findFile("passwd", ".:..:/etc");
102 * // passwd is "/etc/passwd"
103 */
104std::string findFile(const std::string& name, const std::string& searchPath);

106/**
107 * Copies the file with the given source path to the given destination path.
108 * The file copy is done entirely within the kernel, so it's more efficient
109 * than copying the file through userspace buffers.
110 *
111 * Throws an except::IoError if any of the following occur:
112 * 1) The source file cannot be opened for reading.
113 * 2) The destination file cannot be created and opened for writing.
114 * 3) The source file cannot be stat()'d to discover its size.
115 *
116 * Example:
117 * copyFile("/etc/passwd", "/home/bsomers/passwd");
118 */
119void copyFile(const std::string& src, const std::string& dst);
