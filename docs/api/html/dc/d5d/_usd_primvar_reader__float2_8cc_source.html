48void
49UsdPrimvarReader_float2::update()
50{
51 if (hasChanged(attrVarName)) {
52 mOptionalAttributes.clear();
53 const std::string& varname = get(attrVarName);
54 // We handle these special case "varname" values:
55 // "st"
56 // "surface_st"
57 // "closest_surface_st"
58 if (varname == "st") {
59 mIspc.mAttributeMapType = ispc::PRIMVAR_MAP_TYPE_ST;
60 } else if (varname == "surface_st") {
61 mIspc.mAttributeMapType = ispc::PRIMVAR_MAP_TYPE_SURFACE_ST;
62 mIspc.mPrimitiveAttributeIndex = moonray::shading::StandardAttributes::sSurfaceST;
63 mOptionalAttributes.push_back(mIspc.mPrimitiveAttributeIndex);
64 } else if (varname == "closest_surface_uv") {
65 mIspc.mAttributeMapType = ispc::PRIMVAR_MAP_TYPE_CLOSEST_SURFACE_ST;
66 mIspc.mPrimitiveAttributeIndex = moonray::shading::StandardAttributes::sClosestSurfaceST;
67 mOptionalAttributes.push_back(mIspc.mPrimitiveAttributeIndex);
68 } else {
69 // Not a special case so create key from the
70 // "varname" value
71 mIspc.mAttributeMapType = ispc::PRIMVAR_MAP_TYPE_PRIMITIVE_ATTRIBUTE;
72 moonray::shading::TypedAttributeKey<Vec2f> attributeKey(varname);
73 mIspc.mPrimitiveAttributeIndex = attributeKey;
74 mOptionalAttributes.push_back(attributeKey);
75 }
76 ::moonshine::primvar::createLogEvent("vec2f",
77 varname,
78 mIspc.mMissingAttributeEvent,
79 mLogEventRegistry);
80 }
81}

83void
84UsdPrimvarReader_float2::sample(const scene_rdl2::rdl2::Map *self, moonray::shading::TLState *tls,
85 const moonray::shading::State &state, Color *sample)
86{
87 const UsdPrimvarReader_float2* me = static_cast<const UsdPrimvarReader_float2*>(self);
88
89 const int attributeMapType = me->mIspc.mAttributeMapType;
90 if (attributeMapType == ispc::PRIMVAR_MAP_TYPE_ST) {
91 const Vec2f& st = state.getSt();
92 sample->r = st.x;
93 sample->g = st.y;
94 sample->b = 0.0f;
95 } else {
96 int key = me->mIspc.mPrimitiveAttributeIndex;
97 if (state.isProvided(key)) {
98 const Vec2f& v2 = state.getAttribute(moonray::shading::TypedAttributeKey<Vec2f>(key));
99 sample->r = v2.x;
100 sample->g = v2.y;
101 sample->b = 0.0f;
102 } else {
103 // the primitive attribute is unavailable, use fallback parameter
104 const Vec2f& v2 = evalVec2f(me, attrFallback, tls, state);
105 sample->r = v2.x;
106 sample->g = v2.y;
107 sample->b = 0.0f;
108 if (me->get(attrWarnWhenUnavailable)) {
109 moonray::shading::logEvent(me, tls, me->mIspc.mMissingAttributeEvent);
110 }
111 }
112 }
