66//
67// Split a string up based on a delimiter list.
68//
69
70inline
71void
72split (const std::string& src,
73 const char* delimiters,
74 std::vector<std::string>& tokens)
75{
76 const size_t len = src.length();
77 for (size_t index=0; ;)
78 {
79 const size_t a = src.find_first_of(delimiters, index);
80 if (a == std::string::npos)
81 {
82 if (index < len)
83 tokens.push_back(src.substr(index, std::string::npos));
84 break;
85 }
86 if (a > index)
87 tokens.push_back(src.substr(index, a-index));
88 index = a+1;
89 }
90}

93//
94// Strip characters from the delimiter list out of the string.
95//
96
97inline
98void
99strip (std::string& str,
100 const char* delimiters=" \n\t\r")
101{
102 std::string s0(str);
103 std::string s1; s1.reserve(s0.length());
104 std::string* p0 = &s0;
105 std::string* p1 = &s1;
106 for (const char* dp=delimiters; *dp; ++dp)
107 {
108 for (std::string::iterator i=p0->begin(); i != p0->end(); ++i)
109 if (*i != *dp)
110 p1->push_back(*i);
111 p0->clear();
112 std::string* t = p0; p0 = p1; p1 = t; // swap pointers
113 }
114 str = *p0;
115}

118//
119// Split the name into layer & chan, if possible.
120//
121
122void
123splitName (const char* name,
124 std::string& layer,
125 std::string& chan)
126{
127 std::string s(name);
128 strip(s); // Remove any whitespace
129
130 // Can we separate layer & chan strings?
131 size_t a = s.find_last_of('.');
132 if (a > 0 && a != std::string::npos)
133 {
134 layer = s.substr(0, a);
135 chan = s.substr(a+1);
136 return;
137 }
138 // Only chan, no layer:
139 chan = name;
140 layer.clear();
141}

228//
229// Static map of channel-matching strings to a StandardChannel
230//
231static std::map<std::string, StandardChannel*> g_standard_channel_matching_map;
232
233
234//
235// If the kind of channel is one of the predefined ones, return
236// the common position that channel occupies in a layer.
237// i.e.
238// kind=Chan_R -> rgba layer position 0
239// kind=Chan_A -> rgba layer position 3
240//
241// **************************************************************
242// * *
243// * KEEP THIS IN SYNC WITH ChannelDefs.h! *
244// * *
245// **************************************************************
246//
247
248int
249getLayerPositionFromKind (ChannelIdx kind)
250{
251 if (kind <= Chan_Invalid || kind >= Chan_ArbitraryStart)
252 return 0; // no idea
253
254 // Determine channel position in layer from kind index:
255 if (kind <= Chan_A) return kind - Chan_R; // rgba
256 else if (kind <= Chan_AB) return kind - Chan_AR; // opacity
257 else if (kind <= Chan_BY) return kind - Chan_RY; // yuv
258 else if (kind <= Chan_ZBack) return kind - Chan_ZFront; // depth
259 else if (kind <= Chan_DeepFlags) return kind - Chan_SpBits1; // spmask/flags
260 else if (kind <= Chan_ZCutout) return kind - Chan_ACutout; // cutouts
261 else if (kind <= Chan_SpCoverage) return kind - Chan_Visibility; // accumulators
262 else if (kind <= Chan_UvQ) return kind - Chan_UvS; // tex
263 else if (kind <= Chan_ID3) return kind - Chan_ID0; // ids
264
265 return 0;
266}

269//
270// Initializes a map of standard-channel matching strings
271//
272
273struct BuildStandardChannels
274{
275 BuildStandardChannels ()
276 {
277 for (StandardChannel* c=g_standard_channel_table; c->layer_name; ++c)
278 {
279 if (!c->match_list || !c->match_list[0])
280 continue;
281 std::vector<std::string> tokens;
282 split(std::string(c->match_list), ",", tokens);
283 for (size_t i=0; i < tokens.size(); ++i)
284 {
285 strip(tokens[i]); // remove whitespace
286 if (!tokens[i].empty())
287 g_standard_channel_matching_map[tokens[i]] = c;
288 }
289 }
290 }
291};
292static BuildStandardChannels build_standard_channels;

296//
297// Returns the best matching pre-defined channel and layer for the channel
298// name (with no layer prefix,).
299// If there's a match it also sets the channel's default io name and pixeltype.
300//
301
302bool
303matchStandardChannel (const char* layer_name,
304 const char* channel_name,
305 std::string& std_layer_name,
306 std::string& std_chan_name,
307 ChannelIdx& std_channel,
308 std::string& std_io_name,
309 Imf::PixelType& std_io_type)
310{
311 std_channel = Chan_Invalid;
312 std_layer_name.clear();
313 std_io_name.clear();
314
315 if (!channel_name || !channel_name[0])
316 return false;
317 std::string name(channel_name);
318
319 // First test for names that potentially collide like 'Y'&'y' or 'Z'&'z':
320 /* Examples:
321 if (name == "y")
322 {
323 // lower-case y takes precendence:
324 std_channel = Chan_VecY;
325 std_layer_name = "vec";
326 std_chan_name = "y";
327 std_io_name = "y";
328 std_io_type = Imf::FLOAT;
329 return true;
330 }
331 else if (name == "Y")
332 {
333 std_channel = Chan_Y;
334 std_layer_name = "yuv";
335 std_chan_name = "Y";
336 std_io_name = "Y";
337 std_io_type = Imf::HALF;
338 return true;
339 }
340 */
341
342 // No collision, see if the name's in the matching map:
343
344 // To upper-case for matching-map comparison:
345 std::transform(name.begin(), name.end(), name.begin(), ::toupper);
346
347 std::map<std::string, StandardChannel*>::iterator it = g_standard_channel_matching_map.find(name);
348 if (it == g_standard_channel_matching_map.end())
349 return false;
350
351 // Also check layer name if it's not empty:
352 if (layer_name && layer_name[0] && strcasecmp(it->second->layer_name, layer_name)!=0)
353 return false;
354
355 std_layer_name = it->second->layer_name;
356 std_chan_name = it->second->channel_name;
357 std_channel = it->second->ordering_index;
358 std_io_name = it->second->dflt_io_name;
359 std_io_type = it->second->dflt_io_pixel_type;
360
361 return true;
362}

384//
385// Print channel or layer.channel name to output stream.
386// If the ChannelContext is NULL only the ChannelIdx number will be
387// printed for arbitrary channels.
388//
389
390void ChannelSet::print (const char* prefix,
391 std::ostream& os,
392 const ChannelContext* ctx) const
393{
394 if (prefix && prefix[0])
395 os << prefix;
396 if (this->all())
397 {
398 os << "[all]";
399 return;
400 }
401 const size_t n = this->size();
402 os << "[";
403 if (n == 0)
404 os << "none";
405 else if (n == 1)
406 if (ctx)
407 ctx->printChannelFullName(os, this->first());
408 else
409 os << this->first();
410 else
411 {
412 int i = 0;
413 const ChannelSet& set = *this;
414 foreach_channel(z, set)
415 {
416 if (i++ > 0)
417 os << ",";
418 if (ctx)
419 ctx->printChannelFullName(os, z);
420 else
421 os << z;
422 }
423 }
424 os << "]";
425}

697//
698// Add a ChannelAlias to shared lists.
699// Context takes ownership of pointer and deletes it in destructor.
700//
701
702ChannelAlias*
703ChannelContext::addChannelAlias (ChannelAlias* alias)
704{
705 if (!alias)
706 return NULL;
707 for (size_t i=0; i < m_channelalias_list.size(); i++)
708 if (m_channelalias_list[i] == alias)
709 return alias; // ignore duplicates
710
711 // No specfic channel slot requested, assign the next one in the list:
712 if (alias->m_channel == Chan_Invalid)
713 alias->m_channel = ++m_last_assigned;
714
715 const int index = (int)m_channelalias_list.size();
716 m_channelalias_list.push_back(alias);
717
718 if (findChannelAlias(alias->m_channel)==NULL)
719 m_channelalias_channel_map[alias->m_channel] = index;
720
721 // Add name keys for full name '<layer>.<channel>' and the file io name.
722 // Don't overwrite existing assignments:
723 if (findChannelAlias(alias->fullName())==NULL)
724 m_channelalias_name_map[alias->fullName()] = index;
725 if (!alias->m_io_name.empty() && findChannelAlias(alias->m_io_name)==NULL)
726 m_channelalias_name_map[alias->m_io_name] = index;
727
728 return alias;
729}

755//
756// Get or create a channel/alias & possibly a new layer.
757// Return a ChannelAlias or NULL if not able to create it.
758//
759// TODO: This logic is a little confused atm - make sure there's a clear way to
760// repeatedly map the same channel name to the same alias.
761// For example, when a standard channel is matched we only create a single
762// alias using the original chan name which may confuse things if the name
763// gets remapped (ie. spmask.3->deepdcx.flags)
764//
765// We should likely create two aliases, one with the provided name and one
766// with the standard name.
767//
768
769ChannelAlias*
770ChannelContext::getChannelAlias (const char* name)
771{
772 if (!name || !name[0])
773 return NULL; // don't crash!
774
775 // Does alias already exist?
776 ChannelAlias* chan = findChannelAlias(name);
777 if (chan)
778 return chan;
779
780 // Not found, see if name can be split into separate layer/chan strings:
781 std::string layer_name, chan_name;
782 splitName(name, layer_name, chan_name);
783
784 ChannelIdx channel = Chan_Invalid;
