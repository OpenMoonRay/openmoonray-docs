76LogStreamBuf::int_type LogStreamBuf::overflow(int_type ch)
77{
78 if (!traits_type::eq_int_type(ch, traits_type::eof())) {
79 MNRY_ASSERT(std::less_equal<char*>()(pptr(), epptr()));
80 if (doFlush()) {
81 traits_type::assign(*pptr(), traits_type::to_char_type(ch));
82 pbump(1);
83 return traits_type::not_eof(ch);
84 }
85 }
86
87 return traits_type::eof();
88}
89
90bool LogStreamBuf::doFlush()
91{
92 const std::ptrdiff_t n = pptr() - pbase();
93 MNRY_ASSERT(n < static_cast<std::ptrdiff_t>(mBuffer.size()));
94 pbump(static_cast<int>(-n));
95 mBuffer[n] = '\0';
96
97 doLog();
98 return true;
99}

125// Unfortunately, every time we send our output to the logging system, a newline
126// is appended. This class removes the newlines from the input before passing
127// the character off to the buffered stream buffer. If we encounter a newline,
128// immediately flush, hopefully leaving enough room in the buffer for the next
129// newline delimited string.
130template <typename StreamBuf>

138
139 inline int getPriority() const;

141
142private:
143 int_type overflow(int_type c) override;
144 StreamBuf mBuf;
145};

172template <typename StreamBuf>
173typename NewLineLogStreamBuf<StreamBuf>::int_type NewLineLogStreamBuf<StreamBuf>::overflow(int_type ch)
174{
175 if (!traits_type::eq_int_type(ch, traits_type::eof())) {
176 if (ch == '\n') {
177 mBuf.pubsync();
178 } else {
179 return mBuf.sputc(traits_type::to_char_type(ch));
180 }
181 return traits_type::not_eof(ch);
182 }
183
184 return traits_type::eof();
185}

192class LogStream : public std::ostream

194public:

201 void open(scene_rdl2::logging::LogLevel level = scene_rdl2::logging::INFO_LEVEL)
202 {
203 try {
204 mBuf.reset(new NewLineLogStreamBuf<StreamBuf>(level));
205 this->init(mBuf.get());
206 this->clear();
207 } catch (const std::exception&) {
208 this->setstate(std::ios_base::failbit);
209 } catch (...) {
210 this->setstate(std::ios_base::failbit);
211 }
212 }
213
214 bool is_open() const noexcept
215 {
216 return static_cast<bool>(mBuf);
217 }
218
219 scene_rdl2::logging::LogLevel setPriority(scene_rdl2::logging::LogLevel priority)
220 {
221 if (mBuf) {
222 mBuf.setPriority(priority);
223 }
224 return priority;
225 }
226
227private:
228 std::unique_ptr<NewLineLogStreamBuf<StreamBuf>> mBuf;
229};
