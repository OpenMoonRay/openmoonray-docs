21// A utility class for excluding some occlusions. The actual skipping of occlusions occurs in the
22// Embree intersection filter skipOcclusionFilter().
23//
24// The additional occlusion control it provides for the associated geometry primitive is:
25// - whether it will cast a shadow from specific lights
26// - whether it will cast a shadow onto specific receivers
27//
28// This feature is provided purely for artistic control and is obviously not physically correct.
29
30class ShadowLinking
31{
32public:
33 ShadowLinking() {}
34 ShadowLinking(bool complementReceiverSet) : mIsComplemented(complementReceiverSet) {}
35
36 void reset()
37 {
38 mLightSet.clear();
39 mReceiverSet.clear();
40 }
41
42 bool canCastShadow(const scene_rdl2::rdl2::Light* light) const
43 {
44 return mLightSet.find(light) == mLightSet.end();
45 }
46
47 void addLight(const scene_rdl2::rdl2::Light* light)
48 {
49 mLightSet.insert(light);
50 }
51
52 const std::unordered_set<const scene_rdl2::rdl2::Light*>& getLights() const
53 {
54 return mLightSet;
55 }
56
57 bool canReceiveShadow(int receiverId) const
58 {
59 if (mIsComplemented) {
60 return mReceiverSet.find(receiverId) != mReceiverSet.end();
61 } else {
62 return mReceiverSet.find(receiverId) == mReceiverSet.end();
63 }
64 }
65
66 void addReceiver(int receiverID)
67 {
68 mReceiverSet.insert(receiverID);
69 }
70
71private:
72 std::unordered_set<const scene_rdl2::rdl2::Light*> mLightSet;
73 std::unordered_set<int> mReceiverSet;
74 bool mIsComplemented;
75};
