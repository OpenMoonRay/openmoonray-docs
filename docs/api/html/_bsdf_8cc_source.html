98Color
99Bsdf::eval(const BsdfSlice &slice, const Vec3f &wi, float &pdf) const
100{
101 BsdfLobe::Type flags = slice.getSurfaceFlags(*this, wi);
102
103 // Add up all matching lobes' eval()
104 Color f(zero);
105 pdf = 0.0f;
106 int size = mLobeArray.size();
107 for (int l=0; l < size; l++) {
108 BsdfLobe* const lobe = mLobeArray[l];
109
110 // We need to account for lobe pdf, even if the surface flag doesn't match
111 float tmpPdf = 0.0f;
112 Color color = lobe->eval(slice, wi, &tmpPdf);
113 if (lobe->matchesFlags(flags)) {
114 f += color;
115 }
116 pdf += tmpPdf;
117 }
118
119 // Assumes all lobes are equi-probable
120 if (size > 1) {
121 pdf /= size;
122 }
123
124 return f;
125}

130Color
131Bsdf::albedo(const BsdfSlice &slice) const
132{
133 Color result(zero);
134
135 // Add up all matching lobes' albedo()
136 int size = mLobeArray.size();
137 for (int l=0; l < size; l++) {
138 if (mLobeArray[l]->matchesFlags(slice.getFlags())) {
139 result += mLobeArray[l]->albedo(slice);
140 }
141 }
142
143 return result;
144}

146void
147Bsdf::show(const std::string &sceneClass, const std::string &name, std::ostream& os) const
148{
149 const size_t numLobes = mLobeArray.size();
150
151 os << "\n";
152 os << "==========================================================\n";
153 os << "BSDF @ " << this << " for " << sceneClass << ": '" << name << "'\n";
154 os << "==========================================================\n";
155 os << "\n";
156
157 for (size_t i = 0; i < numLobes; ++i) {
158 const BsdfLobe* const lobe = mLobeArray[i];
159 lobe->show(os, "");
160 os << "\n";
161 }
162
163 if (mBssrdf) {
164 mBssrdf->show(os, "");
165 os << "\n";
166 }
167
168 os << "[emission] = "
169 << mSelfEmission.r << " "
170 << mSelfEmission.g << " "
171 << mSelfEmission.b << "\n";
172
173 os << "\n";
174 os << "==========================================================\n";
175}
