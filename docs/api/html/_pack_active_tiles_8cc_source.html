26// PackAcitveTiles:enqTileMaskBlock() returns encode data type by unsigned char and
27// this unsigned char status consists of 2 DumpMode (tileMode and pixMaskMode)

29
30static constexpr unsigned char
31combineMode(const ActiveBitTables::DumpMode tileMode,
32 const RunLenBitTable::DumpMode pixMaskMode)
33{
34 return static_cast<unsigned char>(pixMaskMode) | static_cast<unsigned char>(tileMode);
35}

37static void
38retrieveMode(unsigned char dumpMode,
39 ActiveBitTables::DumpMode &tileMode,
40 RunLenBitTable::DumpMode &pixMaskMode)
41{
42 tileMode = static_cast<ActiveBitTables::DumpMode>(dumpMode & ActiveBitTables::DUMPMODE_MASK);
43 pixMaskMode = static_cast<RunLenBitTable::DumpMode>(dumpMode & RunLenBitTable::DUMPMODE_MASK);
44}

62// static function
63unsigned char
64PackActiveTiles::enqTileMaskBlock(const ActivePixels &activePixels,
65 VContainerEnq &vContainerEnq, // output
66 int64_t *sizeInfo) // output for debug
67//
68// int64_t *sizeInfo :
69// statistical information for tracking detail purpose.
70// If you don't need this information, you should set nullptr.
71// There is no computational overhead if sizeInfo is set to nullptr.
72// sizeInfo[0] : PackActiveTiles encoded data size (we call this as version 2)
73// sizeInfo[1] : Size difference between version2 and version1. sizeInfo[1] = sizeInfo[0] - ver1Size
74// i.e. ver1Size = sizeInfo[0] - sizeInfo[1]
75//
76{
77 //
78 // Encoded data consists of 3 parts
79 // 1) DumpMode
80 // Indicates dump mode of following active tile/pixel position blocks
81 // 2) Active tile position block
82 // Focused on encoding active tile position information.
83 // We only encode active tile position and skip all empty tiles.
84 // There are several different dump mode options and automatically pick best mode.
85 // 3) Active pixel position block
86 // Focused on encoding active pixel position information.
87 // We only encode active pixel position and skip all empty pixels.
88 // There are several different dump mode options and automatically pick best mode.
89 //
90
91 unsigned numActiveTiles = activePixels.getActiveTileTotal();
92 if (numActiveTiles == 0) {
93 //
94 // Empty active tile/pixel information.
95 //
96 vContainerEnq.enqUChar(COMBINED_CONDITION_ALL_SKIP);
97# ifdef DEBUG_MSG_ENQ
98 std::cerr << ">> PackActiveTiles.cc no active tiles" << std::endl;
99# endif // end DEBUG_MSG_ENQ
100 return COMBINED_CONDITION_ALL_SKIP;
101 }
102
103 unsigned numTiles = activePixels.getNumTiles();
104 if (numTiles == numActiveTiles) {
105 //
106 // SKIP_DUMP mode : we don't need to output active tile position information
107 // because all tiles are active.
108 //
109 RunLenBitTable pixMaskInfo(numTiles);
110 crawlAllActivePixelsTile(activePixels, [&](const unsigned tileId) {
111 pixMaskInfo.set(tileId, activePixels.getTileMask(tileId)); // all tiles are active
112 });
113
114 ActiveBitTables::DumpMode tileMode = ActiveBitTables::DumpMode::SKIP_DUMP;
115 RunLenBitTable::DumpMode pixMaskMode = pixMaskInfo.finalize();
116 unsigned char dumpMode = combineMode(tileMode, pixMaskMode);
117
118 vContainerEnq.enqUChar(dumpMode); // enqueue dumpMode first
119
120 // Skip enqueue operation about active tile position info here because we know all tiles are active
121 // and dumpMode = SKIP_DUMP indicate enough this condition.
122
123 enqPixMaskInfo(pixMaskMode, pixMaskInfo, vContainerEnq); // enqueue pixelMask info
124
125 if (sizeInfo) {
126 // Compute data size of format version 1 for debug/analyze purpose
127 size_t ver1Size = numTiles * (sizeof(unsigned int) + sizeof(uint64_t)); // old data size
128 size_t ver2Size = 1 + pixMaskInfo.getDataSize(); // PackActiveTiles encoded data size
129 int deltaSize = (int)ver2Size - (int)ver1Size;
130 // if (deltaSize > 0) std::cerr << "ver2 is BIGGER!!" << std::endl; // useful for test
131
132 sizeInfo[0] = (int64_t)ver2Size; // byte
133 sizeInfo[1] = (int64_t)deltaSize; // byte
134
135# ifdef DEBUG_MSG_ENQ
136 float ratio = (float)ver2Size / (float)ver1Size;
137# ifdef DEBUG_MSG_ENQ_DETAIL_INFO
138 std::cerr << ">> PackActiveTiles.cc " << showDumpMode(dumpMode) << '\n'
139 << " total:" << ver2Size
140 << " (ver1:" << deltaSize << ' '
141 << std::setw(5) << std::fixed << std::setprecision(3) << ratio << ')'
142 << std::endl;
143# else // else DEBUG_MSG_ENQ_DETAIL_INFO
144 std::cerr << ">> PackActiveTiles.cc " << showDumpMode(dumpMode)
145 << " size:" << ver2Size
146 << " %:" << std::setw(5) << std::fixed << std::setprecision(3) << ratio
147 << " diff:" << deltaSize << std::endl;
148# endif // end !DEBUG_MSG_ENQ_DETAIL_INFO
149# endif // end DEBUG_MSG_ENQ
150 }
151
152 return dumpMode;
153 }
154
155 //
156 // setup active tile condition
157 //
158 ActiveBitTables tilesInfo(numTiles);
159 RunLenBitTable pixMaskInfo(numActiveTiles);
160 {
161 unsigned activeTileId = 0;
162 crawlAllActivePixelsTile(activePixels,
163 [&](const unsigned tileId) {
164 if (uint64_t currMask = activePixels.getTileMask(tileId)) {
165 tilesInfo.setOn(tileId);
166 pixMaskInfo.set(activeTileId++, currMask);
167 }
168 });
169 }
170
171 //
172 // pick up proper method to dump tilesInfo and pixMaskInfo
173 //
174 ActiveBitTables::DumpMode tileMode = tilesInfo.finalize();
175 RunLenBitTable::DumpMode pixMaskMode = pixMaskInfo.finalize();
176 unsigned char dumpMode = combineMode(tileMode, pixMaskMode);
177
178 vContainerEnq.enqUChar(dumpMode); // enqueue dumpMode first
179
180 // enqueue operation about active tile position info. We have several options to enqueue.
181 // FULL_DUMP is used under debug purpose only. Because FULL_DELTA_DUMP performance is always
182 // better and there is no reason to use FULL_DUMP.
183 switch (tileMode) {
184 case ActiveBitTables::DumpMode::FULL_DUMP: tilesInfo.enqFullDump(vContainerEnq); break;
185 case ActiveBitTables::DumpMode::FULL_DELTA_DUMP: tilesInfo.enqFullDeltaDump(vContainerEnq); break;
186 case ActiveBitTables::DumpMode::TABLE_DUMP: tilesInfo.enqTblDump(vContainerEnq); break;
187 case ActiveBitTables::DumpMode::LEAF_TABLE_DUMP: tilesInfo.enqTblDump(vContainerEnq); break;
188 default : break; // never happen
189 }
190
191 enqPixMaskInfo(pixMaskMode, pixMaskInfo, vContainerEnq); // enqueue pixelMask info
192
193 if (sizeInfo) {
194 // Compute data size of format version 1 for debug/analyze purpose
195 size_t ver1Size = numActiveTiles * (sizeof(unsigned int) + sizeof(uint64_t)); // old data size
196 size_t fullDump, fullDeltaDump, tblDump;
197 size_t tileInfoSize = tilesInfo.debugGetSizeInfo(fullDump, fullDeltaDump, tblDump);
198 size_t pixMaskInfoSize = pixMaskInfo.getDataSize();
199 size_t ver2Size = 1 + tileInfoSize + pixMaskInfoSize; // PackActiveTiles encoded data size
200 int deltaSize = (int)ver2Size - (int)ver1Size;
201 // if (deltaSize > 0) std::cerr << "ver2 is BIGGER!!" << std::endl; // useful for test
202
203 sizeInfo[0] = (int64_t)ver2Size; // byte
204 sizeInfo[1] = (int64_t)deltaSize; // byte
205
206# ifdef DEBUG_MSG_ENQ
207 float ratio = (float)ver2Size / (float)ver1Size;
208# ifdef DEBUG_MSG_ENQ_DETAIL_INFO
209 std::cerr << ">> PackActiveTiles.cc enqTileMaskBlock() {\n"
210 << " " << showDumpMode(dumpMode) << '\n'
211 << " tileInfo(full:" << fullDump
212 << " delta:" << fullDeltaDump
213 << " tbl:" << tblDump << ")=" << tileInfoSize << '\n'
214 << " pixMaskInfo=" << pixMaskInfoSize << '\n'
215 << " total:" << ver2Size
216 << " (ver1:" << ver1Size << " delta:" << deltaSize << " ratio:"
217 << std::setw(5) << std::fixed << std::setprecision(3) << ratio << ")\n"
218 << '}'
219 << std::endl;
220# else // else DEBUG_MSG_ENQ_DETAIL_INFO
221 std::cerr << ">> PackActiveTiles.cc " << showDumpMode(dumpMode)
222 << " size:" << ver2Size
223 << " %:" << std::setw(5) << std::fixed << std::setprecision(3) << ratio
224 << " diff:" << deltaSize << std::endl;
225# endif // end !DEBUG_MSG_ENQ_DETAIL_INFO
226# endif // end DEBUG_MSG_ENQ
227 }
228
229 return dumpMode;
230}

232// static function
233bool
234PackActiveTiles::deqTileMaskBlock(VContainerDeq &vContainerDeq,
235 const unsigned activeTileTotal,
236 ActivePixels &activePixels)
237{
238 unsigned char dumpMode;
239 vContainerDeq.deqUChar(dumpMode);
240 ActiveBitTables::DumpMode tileMode;
241 RunLenBitTable::DumpMode pixMaskMode;
242 retrieveMode(dumpMode, tileMode, pixMaskMode);
243
244 //
245 // reconstruct tilesInfo from data
246 //
247 ActiveBitTables tilesInfo(activePixels.getNumTiles());
248 switch(tileMode) {
249 case ActiveBitTables::DumpMode::SKIP_DUMP:
250 break;
251 case ActiveBitTables::DumpMode::FULL_DUMP:
252 tilesInfo.deqFullDump(vContainerDeq, activeTileTotal);
253 break;
254 case ActiveBitTables::DumpMode::FULL_DELTA_DUMP:
255 tilesInfo.deqFullDeltaDump(vContainerDeq, activeTileTotal);
256 break;
257 case ActiveBitTables::DumpMode::TABLE_DUMP:
258 tilesInfo.deqTblDump(vContainerDeq, false);
259 break;
260 case ActiveBitTables::DumpMode::LEAF_TABLE_DUMP:
261 tilesInfo.deqTblDump(vContainerDeq, true);
262 break;
263 default:
264 return false; // unknown tileMode -> no data
265 }
266
267 //
268 // reconstruct pixMaskInfo from data
269 //
270 RunLenBitTable pixMaskInfo(activeTileTotal);
271 switch (pixMaskMode) {
272 case RunLenBitTable::DumpMode::SKIP_DUMP: break;
273 case RunLenBitTable::DumpMode::ALLMASK_DUMP: pixMaskInfo.deqAllMask(vContainerDeq); break;
274 case RunLenBitTable::DumpMode::ALLID_DUMP: pixMaskInfo.deqAllId(vContainerDeq); break;
275 case RunLenBitTable::DumpMode::RUNLEN_DUMP: pixMaskInfo.deqRunLen(vContainerDeq); break;
276 default: return false; // unknown pixMaskMode -> no data
277 }
278
279 //
280 // reconstruct activePixels from tilesInfo and pixMaskInfo
281 //
282 if (tileMode == ActiveBitTables::DumpMode::SKIP_DUMP) {
283 if (pixMaskMode == RunLenBitTable::DumpMode::SKIP_DUMP) {
284 return false; // no data
285 } else {
286 crawlAllActivePixelsTile(activePixels, [&](const unsigned tileId) {
287 activePixels.setTileMask(tileId, pixMaskInfo.get(tileId));
288 });
289 }
290 } else {
291 unsigned activeTileId = 0;
292 tilesInfo.crawlActiveTblItem([&](unsigned tileId) {
293 activePixels.setTileMask(tileId, pixMaskInfo.get(activeTileId));
294 activeTileId++;
295 });
296 }
297
298 return true;
299}

301// static function
302void
303PackActiveTiles::randomActivePixels(ActivePixels &activePixels, const unsigned totalActivePixels)
304// for debug
305{
306 unsigned totalPixels = activePixels.getWidth() * activePixels.getHeight();
307
308 std::random_device rnd;
309 std::mt19937 mt(rnd());
310 std::uniform_int_distribution<> randGen(0, totalPixels - 1);
311
312 for (unsigned i = 0; i < totalActivePixels; ++i) {
313 while (1) {
314 unsigned pixId = randGen(mt);
315 if (!getPix(activePixels, pixId)) {
316 setPix(activePixels, pixId);
317 break;
318 }
319 }
320 }
321}

323// static function
324bool
325PackActiveTiles::codecVerify(const ActivePixels &activePixels)
326{
327 std::string data;
328 VContainerEnq vContainerEnq(&data);
329 unsigned char dumpMode = enqTileMaskBlock(activePixels, vContainerEnq);
330 size_t dataSize = vContainerEnq.finalize();
331
332 ActivePixels activePixels2;
333 activePixels2.init(activePixels.getWidth(), activePixels.getHeight());
334 activePixels2.reset();
335
336 unsigned activeTileTotal = activePixels.getActiveTileTotal();
337
338 VContainerDeq vContainerDeq(data.data(), dataSize);
339 deqTileMaskBlock(vContainerDeq, activeTileTotal, activePixels2);
340
341 if (!activePixels.compare(activePixels2)) {
342 std::cerr << "codecVerifyError {\n";
343 std::cerr << " " << showDumpMode(dumpMode) << std::endl;
344 std::cerr << activePixels.show(" in ") << std::endl;
345 std::cerr << activePixels2.show(" out") << std::endl;
346 std::cerr << activePixels.showFullInfo(" ") << std::endl;
347 std::cerr << "}\n";
348 return false;
349 }
350 return true;
351}

369// static function
370bool
371PackActiveTiles::getPix(const ActivePixels &activePixels, const unsigned pixId)
372// for debug
373{
374 if (pixId >= activePixels.getWidth() * activePixels.getHeight()) return false; // just in case
375
376 bool pixStatus = false;;
377 accessPixel(activePixels, pixId,
378 [&](unsigned tileId, unsigned shift) {
379 const uint64_t &currMask = activePixels.getTileMask(tileId);
380 if (currMask & ((uint64_t)0x1 << shift)) pixStatus = true;
381 });
382
383 return pixStatus;
384}

386// static function
387void
388PackActiveTiles::setPix(ActivePixels &activePixels, const unsigned pixId)
389// for debug
390{
391 accessPixel(activePixels, pixId,
392 [&](unsigned tileId, unsigned shift) {
393 activePixels.orOp(tileId, ((uint64_t)0x1 << shift));
394 });
395}
