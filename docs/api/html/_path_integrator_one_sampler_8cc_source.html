32// Debug ray recording:
33// Scalar mode has the seldom used feature "moonray -record_rays"
34// which allows us to create a .mm that can be used to visualize
35// ray origin and directions. We store rays that leave a surface
36// and hit a light. Several RaTS tests make use of this feature.
37// I suspect that this feature will just go away when .mm is finally
38// removed. But for now, we support it.
39static void
40debugRayRecordLightHit(
41 pbr::TLState *pbrTls,
42 const scene_rdl2::math::Vec3f &P,
43 const scene_rdl2::math::Vec3f &wi,
44 float distance,
45 const scene_rdl2::math::Color &Li)
46{
47 // If the intersection distance is closer than the distant light, then
48 // assume the hit wasn't due to a distant or env light.
49 if (DebugRayRecorder::isRecordingEnabled()) {
50 if (distance < sDistantLightDistance) {
51 mcrt_common::Ray debugRay(P, wi, 0.0f);
52 RAYDB_EXTEND_RAY_NO_HIT(pbrTls, debugRay, distance);
53 RAYDB_SET_CONTRIBUTION(pbrTls, Li);
54 RAYDB_ADD_TAGS(pbrTls, TAG_AREALIGHT);
55 } else {
56 mcrt_common::Ray debugRay(P, wi, 0.0f);
57 RAYDB_EXTEND_RAY_NO_HIT(pbrTls, debugRay, 40.0f);
58 RAYDB_SET_CONTRIBUTION(pbrTls, Li);
59 RAYDB_ADD_TAGS(pbrTls, TAG_ENVLIGHT);
60 }
61 }

64// Helper function that abstracts the differences between
65// one lobe and one sample sampling within the context of path guiding.
66// The key abstractions are the bsdf sample and eval functions.
67// When doing one sampling, we use the BsdfOneSamplerObject, when
68// doing one lobe, we already have the lobe and use the lobe's
69// sample and eval functions directly.
70static scene_rdl2::math::Color
71sampleWithPathGuide(const PathGuide &pg,
72 bool isMirror,
73 const scene_rdl2::math::Vec3f &P,
74 float &r0, float &r1, float &r2,
75 std::function<scene_rdl2::math::Color(float, float, float, scene_rdl2::math::Vec3f&, float&)> bsdfSampleFn,
76 std::function<scene_rdl2::math::Color(const scene_rdl2::math::Vec3f &, float&)> bsdfEvalFn,
77 scene_rdl2::math::Vec3f &wi,
78 float &pdfBsdf)
79{
80 scene_rdl2::math::Color f;
81 if (pg.canSample() && pg.getPercentage() > 0 && !isMirror) {
82 float pdfPg = 0.f;
83 const float u = pg.getPercentage();
84 if (r1 > u) {
85 // use bsdf sampling direction
86 r1 = (r1 - u) / (1.0f - u); // remap r1 into [0, 1)
87 f = bsdfSampleFn(r0, r1, r2, wi, pdfBsdf);
88 pdfPg = pg.getPdf(P, wi);
89 } else {
90 // use path guide sampling direction
91 r1 = r1 / u; // remap r1 into [0, 1)
92 wi = pg.sampleDirection(P, r1, r2, &pdfPg);
93 f = bsdfEvalFn(wi, pdfBsdf);
94 }
95 if (isSampleValid(f, pdfBsdf)) {
96 // blending pdf values seems to work well enough in practice, and
97 // allows a potential user percentage control
98 pdfBsdf = u * pdfPg + (1.0 - u) * pdfBsdf;
99 }
100 } else {
101 f = bsdfSampleFn(r0, r1, r2, wi, pdfBsdf);
102 }
103
104 return f;
105}

107// Helper function that applies russian roulette.
108// Determine to either cull c or keep and bump of c's contribution.
109// Returns true if c is kept (and bumped up)
110// Returns false if the sample should be culled
111static bool
112applyRussianRoulette(float r, float threshold, float invThreshold, int nonMirrorDepth, float lum, scene_rdl2::math::Color &c)
113{
114 // The direct comparison against 0.0f is intentional.
115 // A threshold of exactly zero means that russian roulette is off.
116 // But we always want to cull 0 luminance samples.
117 if (lum == 0.0f) {
118 return false; // cull
119 }
120
121 if (nonMirrorDepth > 0 && lum < threshold) {
122 const float continueProbability = scene_rdl2::math::max(scene_rdl2::math::sEpsilon, lum * invThreshold);
123 if (r > continueProbability) {
124 return false; // cull
125 }
126 c *= scene_rdl2::math::rcp(continueProbability); // bump it up
127 }
128
129 return true; // keep it
130}

132scene_rdl2::math::Color
133PathIntegrator::oneSamplerDirectLight(pbr::TLState *pbrTls,
134 const Subpixel &sp, int cameraId, const PathVertex &pv, const mcrt_common::RayDifferential &ray,
135 const shading::Intersection &isect, const Light *light, const LightFilterList *lightFilterList,
136 const scene_rdl2::math::Vec3f *cullingNormal, const BsdfOneSampler &bsdfOneSampler, const scene_rdl2::math::Color &pt,
137 const float r[9], float rayEpsilon, float shadowRayEpsilon, unsigned sequenceID, float *aovs) const
138{
139 // r[0:2] used for light sample, r[3] used for russian roulette, r[4:5] for the light filter
140
141 // Sample light direction
142 LightIntersection lightIsect;
143 scene_rdl2::math::Vec3f wi;
144 const scene_rdl2::math::Vec3f rLight(r[0], r[1], r[2]); // first 3 samples used for light sampling
145 if (!light->sample(isect.getP(), cullingNormal, ray.getTime(), rLight, wi, lightIsect, ray.getDirFootprint())) {
146 return scene_rdl2::math::sBlack;
147 }
148
149 // Evaluate light radiance and pdf
150 float lightPdf = 0.0f;
151 const LightFilterRandomValues rLightFilter = {
152 scene_rdl2::math::Vec2f(r[4], r[5]),
153 scene_rdl2::math::Vec3f(r[6], r[7], r[8])};
154 const scene_rdl2::math::Color Li = light->eval(pbrTls->mTopLevelTls, wi, isect.getP(), rLightFilter, ray.getTime(), lightIsect,
155 /* fromCamera = */ false, lightFilterList, ray.getDirFootprint(), &lightPdf);
156 if (isSampleInvalid(Li, lightPdf)) {
157 return scene_rdl2::math::sBlack;
158 }
159
160 // sample is valid - count it
161 pbrTls->mStatistics.incCounter(STATS_LIGHT_SAMPLES);
162
163 // evaluate the bsdf, we'll need individual lobe contributions for aovs
164 BsdfOneSampler::LobesContribution lobesContribution;
165 float bsdfPdf;
166 const scene_rdl2::math::Color f = bsdfOneSampler.eval(wi, bsdfPdf, &lobesContribution);
167 if (isSampleInvalid(f, bsdfPdf)) {
168 return scene_rdl2::math::sBlack;
169 }
170
171 // if we are path guiding, adjust the bsdf pdf accordingly
172 if (mPathGuide.canSample()) {
173 MNRY_ASSERT(lobesContribution.mMatchedLobeCount > 0);
174 const shading::BsdfLobe *lobe = lobesContribution.mLobes[0];
175 MNRY_ASSERT(lobe);
176 if (!(lobe->getType() & shading::BsdfLobe::MIRROR)) {
177 const float u = mPathGuide.getPercentage();
178 const float pgPdf = mPathGuide.getPdf(isect.getP(), wi);
179 // blending pdf values seems to work well enough in practice, and
180 // allows for a potential user percentage control
181 bsdfPdf = u * pgPdf + (1.0 - u) * bsdfPdf;
182 }
183 }
184
185 const float misWeight = powerHeuristic(lightPdf, bsdfPdf);
186
187 // Compute light contribution, taking path throughput into account
188 scene_rdl2::math::Color lightContribution = pt * misWeight * Li / lightPdf;
189
190 // Apply RR based on luminance
191 const float lum = luminance(f * lightContribution);
192 if (!applyRussianRoulette(r[3], mRussianRouletteThreshold, mInvRussianRouletteThreshold,
193 pv.nonMirrorDepth, lum, lightContribution)) {
194 return scene_rdl2::math::sBlack;
195 }
196
197 // resolve occlusion
198 const float tfar = lightIsect.distance - rayEpsilon;
199 const int rayDepth = ray.getDepth() + 1;
200 // THINK: Would a standard ray, rather than a ray differential work better?
201 // Using a ray differential produces more occlusion (i.e. darker images).
202 // Since the multi-sample integrator uses ray differentials, we do so here
203 // as well. It might be worth changing this behavior in both integrators.
204 mcrt_common::RayDifferential shadowRay(isect.getP(), wi,
205 ray.getOriginX(), wi,
206 ray.getOriginY(), wi,
207 rayEpsilon, tfar, ray.getTime(), rayDepth);
208 float presence = 0.0f;
209
210 const FrameState &fs = *pbrTls->mFs;
211 const bool hasUnoccludedFlag = fs.mAovSchema->hasLpePrefixFlags(AovSchema::sLpePrefixUnoccluded);
212 if (isRayOccluded(pbrTls, light, shadowRay, rayEpsilon, shadowRayEpsilon, presence)) {
213 // Visibility LPE
214 if (aovs) {
215 EXCL_ACCUMULATOR_PROFILE(pbrTls, EXCL_ACCUM_AOVS);
216 const LightAovs &lightAovs = *fs.mLightAovs;
217
218 // If there is no visibility AOV or if we don't have unoccluded flags, then we don't need to bother
219 // with accumulating these values here.
220 if (lightAovs.hasVisibilityEntries() || hasUnoccludedFlag) {
221
222 const AovSchema &aovSchema = *fs.mAovSchema;
223 bool addVisibilityAov = true;
224 for (int lobeIndex = 0; lobeIndex < lobesContribution.mMatchedLobeCount; ++lobeIndex) {
225 int lpeStateId = pv.lpeStateId;
226 lpeStateId = lightAovs.scatterEventTransition(pbrTls, lpeStateId, bsdfOneSampler.getBsdf(),
227 *(lobesContribution.mLobes[lobeIndex]));
228 lpeStateId = lightAovs.lightEventTransition(pbrTls, lpeStateId, light);
229
230 // Add any visibility information:
231 if (addVisibilityAov) {
232 if (lightAovs.hasVisibilityEntries()) {
233 if (aovAccumVisibilityAovs(pbrTls, aovSchema, cameraId, lightAovs,
234 scene_rdl2::math::Vec2f(0.0f, 1.0f), lpeStateId, aovs)) {
235 addVisibilityAov = false; // add visibility aov at most once per shadow ray
236 }
237 }
238 }
239
240 if (hasUnoccludedFlag) {
241 scene_rdl2::math::Color unoccludedLobeContribution = lobesContribution.mFs[lobeIndex] * lightContribution;
242 if (pv.nonMirrorDepth >= mSampleClampingDepth) {
243 unoccludedLobeContribution = smartClamp(unoccludedLobeContribution, sp.mSampleClampingValue);
244 }
245 aovAccumLightAovs(pbrTls, aovSchema, cameraId, lightAovs, unoccludedLobeContribution, nullptr,
246 AovSchema::sLpePrefixUnoccluded, lpeStateId, aovs);
247 }
248 }
249 }
250 }
251
252 return scene_rdl2::math::sBlack;
253 }
254
255 scene_rdl2::math::Color unoccludedLightContribution = lightContribution;
256 mcrt_common::Ray trRay(isect.getP(), wi, scene_rdl2::math::max(rayEpsilon, shadowRayEpsilon), tfar, ray.getTime(), rayDepth);
257 scene_rdl2::math::Color tr = transmittance(pbrTls, trRay, sp.mPixel, sp.mSubpixelIndex, sequenceID, light);
258 lightContribution *= tr * (1.0f - presence);
259
260 // Now add in the lobe contributions
261 scene_rdl2::math::Color LDirect = scene_rdl2::math::sBlack;
262 bool addVisibilityAov = true;
263 for (int lobeIndex = 0; lobeIndex < lobesContribution.mMatchedLobeCount; ++lobeIndex) {
264 // skip lobe if light is marked as not visible from this lobe
265 if (!(lobeTypeToRayMask(lobesContribution.mLobes[lobeIndex]->getType()) & light->getVisibilityMask())) {
266 continue;
267 }
268 scene_rdl2::math::Color lobeContribution = lobesContribution.mFs[lobeIndex] * lightContribution;
269 scene_rdl2::math::Color unoccludedLobeContribution = lobesContribution.mFs[lobeIndex] * unoccludedLightContribution;
270 // sample clamping gah!
271 if (pv.nonMirrorDepth >= mSampleClampingDepth) {
272 lobeContribution = smartClamp(lobeContribution, sp.mSampleClampingValue);
273 unoccludedLobeContribution = smartClamp(unoccludedLobeContribution, sp.mSampleClampingValue);
274 }
275 LDirect += lobeContribution;
276 // LPE
277 if (aovs) {
278 EXCL_ACCUMULATOR_PROFILE(pbrTls, EXCL_ACCUM_AOVS);
279 const FrameState &fs = *pbrTls->mFs;
280 const AovSchema &aovSchema = *fs.mAovSchema;
281 const LightAovs &lightAovs = *fs.mLightAovs;
282 // transition
283 int lpeStateId = pv.lpeStateId;
284 lpeStateId = lightAovs.scatterEventTransition(pbrTls, lpeStateId, bsdfOneSampler.getBsdf(),
285 *(lobesContribution.mLobes[lobeIndex]));
286 lpeStateId = lightAovs.lightEventTransition(pbrTls, lpeStateId, light);
287
288 // Accumulate aovs depending on whether or not the unoccluded flag is set.
289 if (hasUnoccludedFlag) {
290 // If the unoccluded flag is set we have to add unoccluded and occluded (with presence and volumes) separately.
291 aovAccumLightAovs(pbrTls, aovSchema, cameraId, lightAovs, unoccludedLobeContribution,
292 &lobeContribution, AovSchema::sLpePrefixUnoccluded, lpeStateId, aovs);
293 } else {
294 // Otherwise, just add the contribution to all non-pre-occlusion aovs.
295 aovAccumLightAovs(pbrTls, aovSchema, cameraId, lightAovs, lobeContribution, nullptr,
296 AovSchema::sLpePrefixNone, lpeStateId, aovs);
297 }
298
299 // visibilty aov
300 if (addVisibilityAov && lightAovs.hasVisibilityEntries()) {
301 if (aovAccumVisibilityAovs(pbrTls, aovSchema, cameraId, lightAovs,
302 scene_rdl2::math::Vec2f(reduceTransparency(tr) * (1 - presence), 1.0f), lpeStateId, aovs)) {
303 addVisibilityAov = false; // add visibility aov at most once per shadow ray
304 }
305 }
306 }
307 }
308
309 // Debug ray record
310 debugRayRecordLightHit(pbrTls, isect.getP(), wi, lightIsect.distance, Li);
311 return LDirect;
312}

314scene_rdl2::math::Color
315PathIntegrator::oneSamplerDirectBsdf(pbr::TLState *pbrTls,
316 const Subpixel &sp, int cameraId, const PathVertex &pv, const mcrt_common::RayDifferential &ray,
317 const shading::Intersection &isect, const Light *light, const LightFilterList *lightFilterList,
318 const scene_rdl2::math::Vec3f *cullingNormal, const BsdfOneSampler &bsdfOneSampler, const scene_rdl2::math::Color &pt,
319 const float r[9], float rayEpsilon, float shadowRayEpsilon, unsigned sequenceID, float *aovs) const
320{
321 // r[0:2] used for bsdf sample, r[3] used for russian roulette, r[4:5] used for light filter
322
323 // draw a sample and see if it hits the light
324 BsdfOneSampler::LobesContribution lobesContribution;
325 scene_rdl2::math::Vec3f wi;
326 float pdfBsdf;
327 float cr[3] = { r[0], r[1], r[2] }; // need a copy or r since sampleWithPathGuide may modify r[1]
328 const scene_rdl2::math::Color f = sampleWithPathGuide(mPathGuide,
329 bsdfOneSampler.getBsdf().getType() & shading::BsdfLobe::MIRROR,
330 isect.getP(),
331 cr[0], cr[1], cr[2],
332 [&](float r0, float r1, float r2, scene_rdl2::math::Vec3f &dir, float &pdf)
333 {
334 return bsdfOneSampler.sample(r0, r1, r2, dir, pdf, &lobesContribution);
335 },
336 [&](const scene_rdl2::math::Vec3f &dir, float &pdf)
337 {
338 return bsdfOneSampler.eval(dir, pdf, &lobesContribution);
339 },
340 wi,
341 pdfBsdf);
342
343 if (isSampleInvalid(f, pdfBsdf)) {
344 return scene_rdl2::math::sBlack;
345 }
346 LightIntersection lightIsect;
347 if (!light->intersect(isect.getP(), cullingNormal, wi, ray.getTime(), scene_rdl2::math::sMaxValue, lightIsect)) {
348 return scene_rdl2::math::sBlack;
349 }
350
351 // ok we hit the light, add it to the stats, now start the expensive stuff
352 // eventhough this was sampled from the bsdf, we still report it as a "light sample".
353 // For purposes of statistics, "light samples" mean NEE samples.
354 pbrTls->mStatistics.incCounter(STATS_LIGHT_SAMPLES);
355
356 // evaluate the light and get a pdf
357 float pdfLight;
358 const LightFilterRandomValues rLightFilter = {
359 scene_rdl2::math::Vec2f(r[4], r[5]),
360 scene_rdl2::math::Vec3f(r[6], r[7], r[8])};
361 const scene_rdl2::math::Color Li = light->eval(pbrTls->mTopLevelTls, wi, isect.getP(), rLightFilter, ray.getTime(), lightIsect,
362 /* fromCamera = */ false, lightFilterList, ray.getDirFootprint(), &pdfLight);
363 if (isSampleInvalid(Li, pdfLight)) {
364 return scene_rdl2::math::sBlack;
365 }
366
367 // compute the mis weight
368 // mirror lobes require special treatment
369 const float misWeight = (lobesContribution.mMatchedLobeCount == 1 &&
370 lobesContribution.mLobes[0]->matchesFlag(shading::BsdfLobe::MIRROR)) ?
371 1.0 : powerHeuristic(pdfBsdf, pdfLight);
372
373 // total value of this sample
374 scene_rdl2::math::Color lightContribution = pt * misWeight * Li / pdfBsdf;
375 scene_rdl2::math::Color unoccludedLightContribution = lightContribution;
376
377 // apply russian roulette
378 const float lum = luminance(f * lightContribution);
379 if (!applyRussianRoulette(r[3], mRussianRouletteThreshold, mInvRussianRouletteThreshold,
380 pv.nonMirrorDepth, lum, lightContribution)) {
381 return scene_rdl2::math::sBlack;
382 }
383
384 // still here? ok resolve occlusion
385 const float tfar = lightIsect.distance - rayEpsilon;
386 const int rayDepth = ray.getDepth() + 1;
387 // THINK: Would a standard ray, rather than a ray differential work better?
388 // Using a ray differential produces more occlusion (i.e. darker images).
389 // Since the multi-sample integrator uses ray differentials, we do so here
390 // as well. It might be worth changing this behavior in both integrators.
391 mcrt_common::RayDifferential shadowRay(isect.getP(), wi,
392 ray.getOriginX(), wi,
393 ray.getOriginY(), wi,
394 rayEpsilon, tfar, ray.getTime(), rayDepth);
395 float presence = 0.0f;
396
397 if (!isRayOccluded(pbrTls, light, shadowRay, rayEpsilon, shadowRayEpsilon, presence)) {
398 // not occluded, compute volume transmittance, take presence into account
399 // shadowRay can be modified in occlusion query, so make a new one
400 mcrt_common::Ray trRay(isect.getP(), wi, scene_rdl2::math::max(rayEpsilon, shadowRayEpsilon), tfar, ray.getTime(), rayDepth);
401 scene_rdl2::math::Color tr = transmittance(pbrTls, trRay, sp.mPixel, sp.mSubpixelIndex, sequenceID, light);
402 lightContribution *= tr * (1.0f - presence);
403
404 // Now add in the direct lobe contributions
405 scene_rdl2::math::Color LDirect = scene_rdl2::math::sBlack;
406 for (int lobeIndex = 0; lobeIndex < lobesContribution.mMatchedLobeCount; ++lobeIndex) {
407 // skip lobe if light is marked as not visible from this lobe
408 if (!(lobeTypeToRayMask(lobesContribution.mLobes[lobeIndex]->getType()) & light->getVisibilityMask())) {
409 continue;
410 }
411 scene_rdl2::math::Color lobeContribution = lobesContribution.mFs[lobeIndex] * lightContribution;
412 scene_rdl2::math::Color unoccludedLobeContribution = lobesContribution.mFs[lobeIndex] * unoccludedLightContribution;
413 // sample clamping gah!
414 if (pv.nonMirrorDepth >= mSampleClampingDepth) {
415 lobeContribution = smartClamp(lobeContribution, sp.mSampleClampingValue);
416 unoccludedLobeContribution = smartClamp(unoccludedLobeContribution, sp.mSampleClampingValue);
417 }
418 LDirect += lobeContribution;
419 // LPE
420 if (aovs) {
421 EXCL_ACCUMULATOR_PROFILE(pbrTls, EXCL_ACCUM_AOVS);
422 const FrameState &fs = *pbrTls->mFs;
423 const AovSchema &aovSchema = *fs.mAovSchema;
424 const LightAovs &lightAovs = *fs.mLightAovs;
425 // transition
426 int lpeStateId = pv.lpeStateId;
427 lpeStateId = lightAovs.scatterEventTransition(pbrTls, lpeStateId, bsdfOneSampler.getBsdf(),
428 *(lobesContribution.mLobes[lobeIndex]));
429 lpeStateId = lightAovs.lightEventTransition(pbrTls, lpeStateId, light);
430 // accumulate matching aovs
431 if (fs.mAovSchema->hasLpePrefixFlags(AovSchema::sLpePrefixUnoccluded)) {
432 aovAccumLightAovs(pbrTls, aovSchema, cameraId, lightAovs, unoccludedLobeContribution, &lobeContribution,
433 AovSchema::sLpePrefixUnoccluded, lpeStateId, aovs);
434 } else {
435 aovAccumLightAovs(pbrTls, aovSchema, cameraId, lightAovs, lobeContribution, nullptr,
436 AovSchema::sLpePrefixNone, lpeStateId, aovs);
437 }
438 }
439 }
440
441 // Debug ray record
442 // TODO: why do we store sWhite, rather than Li for direct bsdf samples?
443 debugRayRecordLightHit(pbrTls, isect.getP(), wi, lightIsect.distance, scene_rdl2::math::sWhite);
444
445 return LDirect;
446 } else {
447 // Rcord unocclusion aovs if we have any:
448 const FrameState &fs = *pbrTls->mFs;
449 const AovSchema &aovSchema = *fs.mAovSchema;
450 const LightAovs &lightAovs = *fs.mLightAovs;
451 if (aovs && fs.mAovSchema->hasLpePrefixFlags(AovSchema::sLpePrefixUnoccluded)) {
452 for (int lobeIndex = 0; lobeIndex < lobesContribution.mMatchedLobeCount; ++lobeIndex) {
453 // skip lobe if light is marked as not visible from this lobe
454 if (!(lobeTypeToRayMask(lobesContribution.mLobes[lobeIndex]->getType()) & light->getVisibilityMask())) {
455 continue;
456 }
457 scene_rdl2::math::Color unoccludedLobeContribution = lobesContribution.mFs[lobeIndex] * unoccludedLightContribution;
458 // sample clamping gah!
459 if (pv.nonMirrorDepth >= mSampleClampingDepth) {
460 unoccludedLobeContribution = smartClamp(unoccludedLobeContribution, sp.mSampleClampingValue);
461 }
462
463 // LPE
464
465 // transition
466 int lpeStateId = pv.lpeStateId;
467 lpeStateId = lightAovs.scatterEventTransition(pbrTls, lpeStateId, bsdfOneSampler.getBsdf(),
468 *(lobesContribution.mLobes[lobeIndex]));
469 lpeStateId = lightAovs.lightEventTransition(pbrTls, lpeStateId, light);
470 // accumulate matching aovs
471 aovAccumLightAovs(pbrTls, aovSchema, cameraId, lightAovs, unoccludedLobeContribution, nullptr,
472 AovSchema::sLpePrefixUnoccluded, lpeStateId, aovs);
473 }
474 }
475 return scene_rdl2::math::sBlack;
476 }
477}

479scene_rdl2::math::Color
480PathIntegrator::oneSamplerDirect(pbr::TLState *pbrTls,
481 const Subpixel &sp, int cameraId, const PathVertex &pv, const mcrt_common::RayDifferential &ray,
482 const shading::Intersection &isect, const BsdfOneSampler &bsdfOneSampler,
483 const LightSet &activeLightSet, const scene_rdl2::math::Vec3f *cullingNormal, float rayEpsilon, float shadowRayEpsilon,
484 IntegratorSample1D &rrSamples, unsigned sequenceID, float *aovs) const
485{
486 // It might not be clear, but this function's outputs are
487 // radiance (returned), and aovs (added into).
488 scene_rdl2::math::Color radiance = scene_rdl2::math::sBlack;
489
490 const bool highQuality = pv.nonMirrorDepth == 0;
491 const int nDirect = highQuality ? mLightSamples : scene_rdl2::math::min(mLightSamples, 1);
492
493 // TODO: we need a strategy to choose a good light subset
494 // But that task is outside the scope of the initial one-sampler implementation
495 // task. We have a similar need with the multi-sampler integrator.
496 for (int lightIndex = 0; lightIndex < activeLightSet.getLightCount(); ++lightIndex) {
497 const Light *light = activeLightSet.getLight(lightIndex);
498 const LightFilterList *lightFilterList = activeLightSet.getLightFilterList(lightIndex);
499
500 // Skip ray termination lights
501 if (light->getIsRayTerminator()) continue;
502
503 // Set up our sample sequences
504 // We could use IntegratorSample3D sequences, rather than a 2D and 1D.
505 // When I tried this, I found some results in RaTS (such as material/dwalayer/transmission)
506 // were a little bit worse. This is exactly the opposite of the result
507 // I saw on the vector side, where using a 3D sampler improved the results and
508 // eliminated correlated noise.
509 IntegratorSample1D lightSamples1D;
510 IntegratorSample2D lightSamples2D;
511 IntegratorSample2D lightFilterSamples2D;
512 IntegratorSample3D lightFilterSamples3D;
513 IntegratorSample1D lobeSamples;
514 IntegratorSample2D bsdfSamples;
515
516 const Scene *scene = MNRY_VERIFY(pbrTls->mFs->mScene);
517 bool lightFilterNeedsSamples = scene->lightFilterNeedsSamples();
518
519 if (highQuality) {
520 // We want one shared sequence for depth 0
521 const int samplesSoFar = sp.mSubpixelIndex * nDirect;
522 SequenceIDIntegrator lSid(pv.nonMirrorDepth, sp.mPixel, 0,
523 SequenceType::NextEventEstimation,
524 SequenceType::Light, light->getHash(),
525 sequenceID);
526 lightSamples1D.resume(lSid, samplesSoFar);
527 lightSamples2D.resume(lSid, samplesSoFar);
528
529 if (lightFilterNeedsSamples) {
530 SequenceIDIntegrator lFilterSid(pv.nonMirrorDepth, sp.mPixel, 0,
531 SequenceType::NextEventEstimation,
532 SequenceType::LightFilter, light->getHash(),
533 sequenceID);
534 lightFilterSamples2D.resume(lFilterSid, samplesSoFar);
535 SequenceIDIntegrator lFilter3DSid(pv.nonMirrorDepth, sp.mPixel, 0,
536 SequenceType::NextEventEstimation,
537 SequenceType::LightFilter3D, light->getHash(),
538 sequenceID);
539 lightFilterSamples3D.resume(lFilter3DSid, samplesSoFar);
540 }
541
542 SequenceIDIntegrator bSid(pv.nonMirrorDepth, sp.mPixel, 0,
543 SequenceType::NextEventEstimation,
544 SequenceType::Bsdf, light->getHash(),
545 sequenceID);
546 lobeSamples.resume(bSid, samplesSoFar);
547 bsdfSamples.resume(bSid, samplesSoFar);
548 } else {
549 SequenceIDIntegrator lSid(pv.nonMirrorDepth, sp.mPixel, sp.mSubpixelIndex,
550 SequenceType::NextEventEstimation,
551 SequenceType::Light, light->getHash(),
552 sequenceID);
553 lightSamples1D.restart(lSid, nDirect);
554 lightSamples2D.restart(lSid, nDirect);
555
556 if (lightFilterNeedsSamples) {
557 SequenceIDIntegrator lFilterSid(pv.nonMirrorDepth, sp.mPixel, sp.mSubpixelIndex,
558 SequenceType::NextEventEstimation,
559 SequenceType::LightFilter, light->getHash(),
560 sequenceID);
561 lightFilterSamples2D.restart(lFilterSid, nDirect);
562 SequenceIDIntegrator lFilter3DSid(pv.nonMirrorDepth, sp.mPixel, sp.mSubpixelIndex,
563 SequenceType::NextEventEstimation,
564 SequenceType::LightFilter3D, light->getHash(),
565 sequenceID);
566 lightFilterSamples3D.restart(lFilter3DSid, nDirect);
567 }
568
569 SequenceIDIntegrator bSid(pv.nonMirrorDepth, sp.mPixel, sp.mSubpixelIndex,
570 SequenceType::NextEventEstimation,
571 SequenceType::Bsdf, light->getHash(),
572 sequenceID);
573 lobeSamples.restart(bSid, nDirect);
574 bsdfSamples.restart(bSid, nDirect);
575 }
576
577 const scene_rdl2::math::Color pt = pv.pathThroughput / nDirect;
578
579 // Loop over each sample
580 for (int s = 0; s < nDirect; ++s) {
581 // light sampling
582 float r[9];
583 lightSamples2D.getSample(&r[0], pv.nonMirrorDepth);
584 lightSamples1D.getSample(&r[2], pv.nonMirrorDepth);
585 rrSamples.getSample(&r[3], pv.nonMirrorDepth);
586
587 if (lightFilterNeedsSamples) {
588 lightFilterSamples2D.getSample(&r[4], pv.nonMirrorDepth);
589 lightFilterSamples3D.getSample(&r[6], pv.nonMirrorDepth);
590 }
591
592 radiance += oneSamplerDirectLight(pbrTls, sp, cameraId, pv, ray,
593 isect, light, lightFilterList, cullingNormal, bsdfOneSampler, pt, r,
594 rayEpsilon, shadowRayEpsilon, sequenceID, aovs);
595
596 // bsdf sampling
597 lobeSamples.getSample(&r[0], pv.nonMirrorDepth);
598 bsdfSamples.getSample(&r[1], pv.nonMirrorDepth);
599 rrSamples.getSample(&r[3], pv.nonMirrorDepth);
600
601 if (lightFilterNeedsSamples) {
602 lightFilterSamples2D.getSample(&r[4], pv.nonMirrorDepth);
603 lightFilterSamples3D.getSample(&r[6], pv.nonMirrorDepth);
604 }
605
606 radiance += oneSamplerDirectBsdf(pbrTls, sp, cameraId, pv, ray,
607 isect, light, lightFilterList, cullingNormal, bsdfOneSampler, pt, r,
608 rayEpsilon, shadowRayEpsilon, sequenceID, aovs);
609 }
610 }
611
612 return radiance;
613}

615scene_rdl2::math::Color
616PathIntegrator::oneSamplerRayTerminatorLights(pbr::TLState *pbrTls,
617 int cameraId, const PathVertex &pv, const mcrt_common::RayDifferential &ray,
618 const shading::Intersection &isect, const scene_rdl2::math::Vec3f &wi, float pdfBsdf,
619 const shading::BsdfLobe &lobe, const shading::Bsdf &bsdf,
620 const LightSet &activeLightSet, const LightFilterRandomValues& lightFilterR,
621 const scene_rdl2::math::Vec3f *cullingNormal, const scene_rdl2::math::Color &pt,
622 float *aovs) const
623{
624 // output radiance and aovs (added into)
625 scene_rdl2::math::Color radiance = scene_rdl2::math::sBlack;
626
627 // TODO: we need a better way to get just the ray-terminator lights
628 for (int lightIndex = 0; lightIndex < activeLightSet.getLightCount(); ++lightIndex) {
629 const Light *light = activeLightSet.getLight(lightIndex);
630 const LightFilterList *lightFilterList = activeLightSet.getLightFilterList(lightIndex);
631
632 // Skip non ray terminator lights
633 if (!light->getIsRayTerminator()) continue;
634
635 // Does the sample direction hit the light?
636 LightIntersection lightIsect;
637 if (!light->intersect(isect.getP(), cullingNormal, wi, ray.getTime(), scene_rdl2::math::sMaxValue, lightIsect)) {
638 continue;
639 }
640
641 // It does, evaluate it and get a pdf
642 float pdfLight;
643 const scene_rdl2::math::Color Li = light->eval(pbrTls->mTopLevelTls, wi, isect.getP(), lightFilterR, ray.getTime(), lightIsect,
644 /* fromCamera = */ false, lightFilterList, ray.getDirFootprint(), &pdfLight);
645
646 // Compute the mis weight
647 const float misWeight = (lobe.matchesFlag(shading::BsdfLobe::MIRROR)) ?
648 1.0f : powerHeuristic(pdfBsdf, pdfLight);
649
650 const scene_rdl2::math::Color contribution = misWeight * Li;
651 radiance += contribution;
652
653 // LPE
654 if (aovs) {
655 EXCL_ACCUMULATOR_PROFILE(pbrTls, EXCL_ACCUM_AOVS);
656 const FrameState &fs = *pbrTls->mFs;
657 const AovSchema &aovSchema = *fs.mAovSchema;
658 const LightAovs &lightAovs = *fs.mLightAovs;
659 // transition
660 int lpeStateId = pv.lpeStateId;
661 lpeStateId = lightAovs.scatterEventTransition(pbrTls, lpeStateId, bsdf, lobe);
662 lpeStateId = lightAovs.lightEventTransition(pbrTls, lpeStateId, light);
663 // Accumulate matching aovs. Don't have to worry about pre-occlusion aovs.
664 aovAccumLightAovs(pbrTls, aovSchema, cameraId, lightAovs, contribution, nullptr,
665 AovSchema::sLpePrefixNone, lpeStateId, aovs);
666 }
667 }
668
669 return radiance;
670}
671
672scene_rdl2::math::Color
673PathIntegrator::computeRadianceBsdfOneSampler(pbr::TLState *pbrTls,
674 const Subpixel &sp, int cameraId, const PathVertex &pv, const mcrt_common::RayDifferential &ray,
675 const shading::Intersection &isect, const shading::Bsdf &bsdf, const shading::BsdfSlice &slice,
676 bool doIndirect, const shading::BsdfLobe::Type indirectFlags, const scene_rdl2::rdl2::Material *newPriorityList[4],
677 int newPriorityListCount[4], const LightSet &activeLightSet, const scene_rdl2::math::Vec3f *cullingNormal,
678 bool hasRayTerminatorLights, float rayEpsilon, float shadowRayEpsilon, const scene_rdl2::math::Color &ssAov,
679 unsigned &sequenceID, float *aovs) const
680{
681 // It might not be clear, but this function's outputs are
682 // radiance (returned) and aovs (added into).
683 // Additionally, the sequenceID is updated as needed by the sampling framework
684 scene_rdl2::math::Color radiance = scene_rdl2::math::sBlack;
685
686 // we are responsible for filling out material aovs
687 if (aovs) {
688 const FrameState &fs = *pbrTls->mFs;
689 const AovSchema &aovSchema = *fs.mAovSchema;
690 const LightAovs &lightAovs = *fs.mLightAovs;
691 const MaterialAovs &materialAovs = *fs.mMaterialAovs;
692 const Scene &scene = *fs.mScene;
693
694 // Since we don't produce BsdfSample objects like multi-sample,
695 // we can pack the material aovs now. The albedo aov will use the slice
696 // and the lobe albedo methods.
697 aovSetMaterialAovs(pbrTls, aovSchema, cameraId, lightAovs, materialAovs, isect,
698 ray, scene, bsdf, ssAov, &slice, pv.aovPathPixelWeight, pv.lpeStateId, aovs);
699 }
700
701 const BsdfOneSampler bsdfOneSampler(bsdf, slice);
702
703 // Both direct and indirect sampling share russian roulette samples
704 IntegratorSample1D rrSamples(SequenceID(
705 sp.mPixel, sp.mSubpixelIndex, pv.nonMirrorDepth,
706 SequenceType::RussianRouletteBsdf, sequenceID));
707
708 // First do NEE
709 const bool doNEE = mLightSamples != 0;
710 if (doNEE) {
711 radiance += oneSamplerDirect(pbrTls, sp, cameraId, pv, ray, isect, bsdfOneSampler,
712 activeLightSet, cullingNormal, rayEpsilon, shadowRayEpsilon, rrSamples, sequenceID, aovs);
713 }
714
715 // We can exit early if doIndirect is false and we have no ray terminator lights
716 if (!doIndirect && !hasRayTerminatorLights) {
717 return radiance;
718 }
719
720 // Spawn new rays through bsdf sampling
721 // Respect the user bsdf samples up to the first non-mirror bounce.
722 // After that, we spawn only a single bsdf sample. If the bsdf
723 // contains mirror lobes, we ignore bsdf samples.
724 const bool highQuality = pv.nonMirrorDepth == 0 &&
725 !(bsdf.getType() & shading::BsdfLobe::MIRROR);
726 const int nIndirect = highQuality ? mBsdfSamples : scene_rdl2::math::min(mBsdfSamples, 1);
727
728 IntegratorSample1D lobeSamples; // picks the lobe
729 IntegratorSample2D bsdfSamples; // samples bsdf
730 IntegratorSample2D lightFilterSamples; // samples light filter
731 IntegratorSample3D lightFilterSamples3D; // samples light filter
732
733 const Scene *scene = MNRY_VERIFY(pbrTls->mFs->mScene);
734 bool lightFilterNeedsSamples = scene->lightFilterNeedsSamples();
735
736 if (highQuality) {
737 // We want one shared sequence for depth 0
738 int samplesSoFar = sp.mSubpixelIndex * nIndirect;
739 SequenceIDIntegrator bSid(
740 pv.nonMirrorDepth, sp.mPixel, 0,
741 SequenceType::IndirectLighting,
742 SequenceType::Bsdf, sequenceID);
743 lobeSamples.resume(bSid, samplesSoFar);
744 bsdfSamples.resume(bSid, samplesSoFar);
745
746 if (lightFilterNeedsSamples) {
747 SequenceIDIntegrator bFilterSid(
748 pv.nonMirrorDepth, sp.mPixel, 0,
749 SequenceType::IndirectLighting,
750 SequenceType::LightFilter, sequenceID);
751 lightFilterSamples.resume(bFilterSid, samplesSoFar);
752 SequenceIDIntegrator bFilter3DSid(
753 pv.nonMirrorDepth, sp.mPixel, 0,
754 SequenceType::IndirectLighting,
755 SequenceType::LightFilter3D, sequenceID);
756 lightFilterSamples3D.resume(bFilter3DSid, samplesSoFar);
757 }
758 } else {
759 SequenceIDIntegrator bSid(
760 pv.nonMirrorDepth, sp.mPixel, sp.mSubpixelIndex,
761 SequenceType::IndirectLighting,
762 SequenceType::Bsdf, sequenceID);
763 lobeSamples.restart(bSid, nIndirect);
764 bsdfSamples.restart(bSid, nIndirect);
765
766 if (lightFilterNeedsSamples) {
767 SequenceIDIntegrator bFilterSid(
768 pv.nonMirrorDepth, sp.mPixel, sp.mSubpixelIndex,
769 SequenceType::IndirectLighting,
770 SequenceType::LightFilter, sequenceID);
771 lightFilterSamples.restart(bFilterSid, nIndirect);
772 SequenceIDIntegrator bFilter3DSid(
773 pv.nonMirrorDepth, sp.mPixel, sp.mSubpixelIndex,
774 SequenceType::IndirectLighting,
775 SequenceType::LightFilter3D, sequenceID);
776 lightFilterSamples3D.restart(bFilter3DSid, nIndirect);
777 }
778 }
779
780 // loop over samples
781 for (int s = 0; s < nIndirect; ++s) {
782 float r[9]; // r[0] lobe select, r[1:2] bsdf, r[3] russian roulette, r[4:5] light filter, r[6:9] light filter 3D
783 lobeSamples.getSample(&r[0], pv.nonMirrorDepth);
784 bsdfSamples.getSample(&r[1], pv.nonMirrorDepth);
785
786 if (lightFilterNeedsSamples) {
787 lightFilterSamples.getSample(&r[4], pv.nonMirrorDepth);
788 lightFilterSamples3D.getSample(&r[6], pv.nonMirrorDepth);
789 }
790
791 scene_rdl2::math::Vec3f wi;
792 float pdfBsdf;
793 scene_rdl2::math::Color f;
794 // The sampled lobe will be the one we use for all lobe dependent
795 // decisions, such as path depth, aovs, etc...
796 const shading::BsdfLobe *lobe = nullptr;
797 if (mBsdfSamplerStrategy == BSDF_SAMPLER_STRATEGY_ONE_LOBE) {
798 float pdfLobe; // probability of picking this lobe
799 lobe = bsdfOneSampler.sampleLobe(r[0], pdfLobe);
800 if (!lobe || pdfLobe == 0.0f) {
801 continue;
802 }
803 f = sampleWithPathGuide(mPathGuide,
804 lobe->getType() & shading::BsdfLobe::MIRROR,
805 isect.getP(),
806 r[0], r[1], r[2],
807 [&](float /*r0*/, float r1, float r2, scene_rdl2::math::Vec3f &dir, float &pdf)
808 {
809 return lobe->sample(slice, r1, r2, dir, pdf);
810 },
811 [&](const scene_rdl2::math::Vec3f &dir, float &pdf)
812 {
813 return lobe->eval(slice, dir, &pdf);
814 },
815 wi,
816 pdfBsdf);
817 pdfBsdf *= pdfLobe; // include probability of selecting lobe
818
819 // We are responsible for checking if this lobe matches the flags
820 shading::BsdfLobe::Type flags = slice.getSurfaceFlags(bsdf, wi);
821 if (!lobe->matchesFlags(flags)) {
822 f = scene_rdl2::math::sBlack;
823 pdfBsdf = 0.f;
824 }
825 } else {
826 MNRY_ASSERT(mBsdfSamplerStrategy == BSDF_SAMPLER_STRATEGY_ONE_SAMPLE);
827 BsdfOneSampler::LobesContribution lobesContribution;
828 f = sampleWithPathGuide(mPathGuide,
829 bsdf.getType() & shading::BsdfLobe::MIRROR,
830 isect.getP(),
831 r[0], r[1], r[2],
832 [&](float r0, float r1, float r2, scene_rdl2::math::Vec3f &dir, float &pdf)
833 {
834 return bsdfOneSampler.sample(r0, r1, r2, dir, pdf, &lobesContribution);
835 },
836 [&](const scene_rdl2::math::Vec3f &dir, float &pdf)
837 {
838 return bsdfOneSampler.eval(dir, pdf, &lobesContribution);
839 },
840 wi,
841 pdfBsdf);
842 if (!lobesContribution.mMatchedLobeCount) {
843 continue;
844 }
845 // The chosen lobe is always at index 0.
846 lobe = lobesContribution.mLobes[0];
847 }
848
849 if (!isSampleValid(f, pdfBsdf)) {
850 continue;
851 }
852
853 // Compute the total throughput potential for this sample.
854 // Terminate the sample if it's throughput is 0.
855 scene_rdl2::math::Color pt = f / (pdfBsdf * nIndirect);
856 pt *= pv.pathThroughput;
857 float lum = luminance(pt);
858
859 // Apply russian roulette
860 rrSamples.getSample(&r[3], pv.nonMirrorDepth);
861 if (!applyRussianRoulette(r[3], mRussianRouletteThreshold, mInvRussianRouletteThreshold,
862 pv.nonMirrorDepth, lum, pt)) {
863 continue;
864 }
865
866 // Terminate path due to depth limits.
867 // FIXME: we should not need to special case the hair lobe depth check.
868 // Make it work with indirectFlags.
869 const bool hairLobe = lobe->getIsHair();
870 if (!doIndirect || !lobe->matchesFlags(indirectFlags)) {
871 // The lobe flags say the depth limit has been meet. But
872 // if we are a hair lobe, me may need to keep going anyway.
873 if (!hairLobe || (pv.hairDepth >= mMaxHairDepth)) {
874 // apply ray terminator lights to non-hair transmission lobes
875 if (hasRayTerminatorLights && !hairLobe &&
876 lobe->matchesFlags(shading::BsdfLobe::ALL_TRANSMISSION)) {
877 LightFilterRandomValues lightFilterR = {
878 scene_rdl2::math::Vec2f(r[4], r[5]),
879 scene_rdl2::math::Vec3f(r[6], r[7], r[8])};
880 radiance += oneSamplerRayTerminatorLights(pbrTls, cameraId, pv, ray, isect, wi, pdfBsdf,
881 *lobe, bsdf, activeLightSet, lightFilterR, cullingNormal, pt, aovs);
882 }
883 continue; // terminate the path
884 }
885 }
886
887 // At this point, count this sample in the stats
888 Statistics &stats = pbrTls->mStatistics;
889 stats.incCounter(STATS_BSDF_SAMPLES);
890
891 const bool diffuseLobe = lobe->matchesFlags(shading::BsdfLobe::ALL_DIFFUSE);
892 const bool glossyLobe = lobe->matchesFlags(shading::BsdfLobe::ALL_GLOSSY);
893 const bool mirrorLobe = lobe->matchesFlags(shading::BsdfLobe::ALL_MIRROR);
894 const bool transmissionLobe = lobe->matchesFlags(shading::BsdfLobe::ALL_TRANSMISSION);
895
896 // Check transparency threshold
897 float newAccumOpacity;
898 if (mirrorLobe && transmissionLobe) {
899 float lobeTransparency = reduceTransparency(f);
900 newAccumOpacity = pv.accumOpacity + (1 - lobeTransparency) * (1 - pv.accumOpacity);
901 if (newAccumOpacity > mTransparencyThreshold) {
902 continue;
903 }
904 } else {
905 newAccumOpacity = pv.accumOpacity;
906 }
907
908 // Prepare a path vertex
909 PathVertex nextPv;
910 nextPv.pathThroughput = pt;
911 nextPv.pathPixelWeight = 0.0f;
912 // Use previous path pixel weight for aovPathPixelWeight as there's existing logic
913 // in vector mode that sometimes assumes that pv.pathPixelWeight = 0. Thus, we must seperately
914 // keep track of the pathPixelWeight for aovs. See comment in PathIntegratorMultiSampler.ispc::
915 // addIndirectOrDirectVisibleContributionsBundled().
916 nextPv.aovPathPixelWeight = pv.pathPixelWeight;
917 nextPv.pathDistance = pv.pathDistance + ray.getEnd();
918 nextPv.minRoughness = shading::computeMinRoughness(*lobe, mRoughnessClampingFactor, pv.minRoughness);
919 nextPv.diffuseDepth = pv.diffuseDepth + (diffuseLobe ? 1 : 0);
920 nextPv.glossyDepth = pv.glossyDepth + (glossyLobe ? 1 : 0);
921 nextPv.mirrorDepth = pv.mirrorDepth + (mirrorLobe ? 1 : 0);
922 nextPv.nonMirrorDepth = pv.nonMirrorDepth + (mirrorLobe ? 0 : 1);
923 nextPv.hairDepth = pv.hairDepth + (hairLobe ? 1 : 0);
924 nextPv.volumeDepth = pv.volumeDepth + 1;
925 nextPv.presenceDepth = pv.presenceDepth;
926 nextPv.subsurfaceDepth = pv.subsurfaceDepth;
927 nextPv.accumOpacity = newAccumOpacity;
928 nextPv.lobeType = lobe->getType();
929
930 // LPE
931 // TODO: Big issue when doing BSDF_SAMPLER_STRATEGY_ONE_SAMPLE.
932 // We can only set a single scatter transition event,
933 // so we'll use the selected lobe - but f contains the evaluation of the all lobes.
934 // so the aovs will bleed into each other.
935 if (aovs) {
936 const FrameState &fs = *pbrTls->mFs;
937 const LightAovs &lightAovs = *fs.mLightAovs;
938 // transition
939 nextPv.lpeStateId = lightAovs.scatterEventTransition(pbrTls, pv.lpeStateId, bsdf, *lobe);
940 // Accumulate post scatter extra aovs
941 aovAccumPostScatterExtraAovs(pbrTls, fs, pv, cameraId, bsdf, aovs);
942 }
943
944 // We have some self-intersections when rays leave at grazing
945 // angle, so we adjust the rayEpsilon accordingly.
946 const float denom = scene_rdl2::math::abs(dot(isect.getNg(), wi));
947 // isect.getNg() itself or the dot product above can be zero.
948 const float start = scene_rdl2::math::isZero(denom) ? rayEpsilon : rayEpsilon / denom;
949 // For NEE to work properly with de-coupled direct and indirect
950 // samples we must skip samples that would have been counted if sampled
951 // as direct bsdf samples. So that means we need to find the shortest
952 // distance to an active light and set the end of our ray to this distance.
953 // TODO: find a faster way to get the shortest distance to a light
954 // Perhaps the light accelerator could add an intersectNearestLight function.
955 // Note that because we update the ray end with the distance to the
956 // nearest light, we eliminate the possibility of intersecting geometry
957 // that is beyond the nearest light.
958 float end = scene_rdl2::math::sMaxValue;
959 int hitLightIndex = -1;
960 LightIntersection hitLightIsect;
961 if (doNEE) {
962 for (int lightIndex = 0; lightIndex < activeLightSet.getLightCount(); ++lightIndex) {
963 const Light *light = activeLightSet.getLight(lightIndex);
964 // skip ray termination lights
965 if (light->getIsRayTerminator()) continue;
966 if (light->intersect(isect.getP(), cullingNormal, wi, ray.getTime(), end, hitLightIsect)) {
967 MNRY_ASSERT(hitLightIsect.distance <= end);
968 end = hitLightIsect.distance;
969 }
970 }
971 if (end <= start) {
972 continue;
973 }
974 } else {
975 // we are not doing NEE, so we must include a light evaluation if we happen
976 // to hit a light. We'll select a light at random along our ray path
977 // and set our ray end to this distance.
978 int lobeIndex = 0;
979 for (; lobeIndex < bsdf.getLobeCount(); ++lobeIndex) {
980 if (bsdf.getLobe(lobeIndex) == lobe) break;
981 }
982 SequenceIDIntegrator lightChoiceSid(pv.nonMirrorDepth,
983 sp.mPixel,
984 lobeIndex,
985 SequenceType::IndexSelection,
986 sp.mSubpixelIndex,
987 sequenceID);
988 IntegratorSample1D lightChoiceSamples(lightChoiceSid);
989 int numHits;
990 hitLightIndex = activeLightSet.intersect(isect.getP(), cullingNormal, wi,
991 ray.getTime(), end, /* includeRayTerminationLights = */ false, lightChoiceSamples,
992 pv.nonMirrorDepth, lobeTypeToRayMask(lobe->getType()), hitLightIsect, numHits);
993 if (hitLightIndex != -1) {
994 MNRY_ASSERT(hitLightIndex < activeLightSet.getLightCount());
995 end = hitLightIsect.distance;
996 }
997 }
998 // Prepare a RayDifferential
999 mcrt_common::RayDifferential nextRay(ray, start, end);
1000 if (transmissionLobe) {
1001 // copy in the new priority list into the ray
1002 setPriorityList(nextRay, newPriorityList, newPriorityListCount);
1003 }
1004 // Scatter and scale our next ray differential
1005 // We scatter the ray based on the sampled lobe
1006 float scale = (lobe->getDifferentialFlags() & shading::BsdfLobe::IGNORES_INCOMING_DIFFERENTIALS) ?
1007 sp.mPrimaryRayDiffScale : 1.0f;
1008 scatterAndScale(isect, *lobe, -ray.dir, wi, scale, r[1], r[2], nextRay);
1009
1010 // Recurse
1011 scene_rdl2::math::Color radianceIndirect;
1012 float transparencyIndirect;
1013 // the volume attenuation along this ray to the first hit (or infinity)
1014 VolumeTransmittance vtIndirect;
1015 ++sequenceID;
1016 bool hitVolume;
1017 IndirectRadianceType result = computeRadianceRecurse(pbrTls, nextRay, sp, cameraId, nextPv,
1018 lobe, radianceIndirect, transparencyIndirect,
1019 vtIndirect, sequenceID, aovs, /* depth = */ nullptr, /* deepParams = */ nullptr,
1020 /* cryptomatteParams = */ nullptr, /* ignoreVolumes = */ false, hitVolume);
1021 if (result != NONE) {
1022 radiance += radianceIndirect;
1023
1024 // path guiding is trained on indirect radiance contributions.
1025 // it turns out that including any other radiance contributions in the
1026 // training causes the path guide to overwhelmingly favor direct lighting
1027 // directions. this is the exact opposite of what we want. we are trying
1028 // to build a distribution that favors important indirect lighting.
1029 mPathGuide.recordRadiance(nextRay.getOrigin(), nextRay.getDirection(), radianceIndirect);
1030 } else {
1031 // We didn't bounce
1032 // If we are not doing NEE, and our ray hit a light, we must
1033 // add contribution now
1034 if (!doNEE && hitLightIndex != -1) {
1035 const Light *hitLight = activeLightSet.getLight(hitLightIndex);
1036 const LightFilterList *hitLightFilterList = activeLightSet.getLightFilterList(hitLightIndex);
1037 float pdfLight;
1038 const LightFilterRandomValues lightFilterR = {
1039 scene_rdl2::math::Vec2f(r[4], r[5]),
1040 scene_rdl2::math::Vec3f(r[6], r[7], r[8])};
1041 const scene_rdl2::math::Color Li = hitLight->eval(pbrTls->mTopLevelTls, wi, isect.getP(), lightFilterR, ray.getTime(),
1042 hitLightIsect, /* fromCamera = */ false, hitLightFilterList, ray.getDirFootprint(), &pdfLight);
1043 const scene_rdl2::math::Color contribution = Li * pt;
1044 radiance += contribution;
1045 // LPE
1046 if (aovs) {
1047 EXCL_ACCUMULATOR_PROFILE(pbrTls, EXCL_ACCUM_AOVS);
1048 const FrameState &fs = *pbrTls->mFs;
1049 const AovSchema &aovSchema = *fs.mAovSchema;
1050 const LightAovs &lightAovs = *fs.mLightAovs;
1051 // transition
1052 int lpeStateId = nextPv.lpeStateId; // already includes the scatter event
1053 lpeStateId = lightAovs.lightEventTransition(pbrTls, lpeStateId, hitLight);
1054 // Accumulate matching aovs. Don't have to worry about pre-occlusion aovs.
1055 aovAccumLightAovs(pbrTls, aovSchema, cameraId, lightAovs, contribution, nullptr,
1056 AovSchema::sLpePrefixNone, lpeStateId, aovs);
1057 }
1058
1059 // Debug ray record
1060 debugRayRecordLightHit(pbrTls, isect.getP(), wi, hitLightIsect.distance, Li);
1061 }
1062 }
1063 }
1064
1065
1066 return radiance;
1067}
