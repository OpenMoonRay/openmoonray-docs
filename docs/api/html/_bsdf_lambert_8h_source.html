37class LambertBsdfLobe : public BsdfLobe
38{
39public:
40 // Constructor / Destructor
41 LambertBsdfLobe(const scene_rdl2::math::Vec3f &N, const scene_rdl2::math::Color& albedo, bool reflection /* = true */) :
42 BsdfLobe(Type((reflection ? REFLECTION : TRANSMISSION) | DIFFUSE),
43 DifferentialFlags(IGNORES_INCOMING_DIFFERENTIALS), false,
44 PROPERTY_NORMAL | PROPERTY_ROUGHNESS | PROPERTY_PBR_VALIDITY),
45 mFrame(N),
46 mAlbedo(albedo) {}
47
48 // BsdfLobe API
49 finline scene_rdl2::math::Color eval(const BsdfSlice &slice, const scene_rdl2::math::Vec3f &wi, float *pdf = NULL) const override
50 {
51 const float cosThetaWi = scene_rdl2::math::max(dot(mFrame.getN(), wi), 0.0f);
52 if (pdf != NULL) {
53 *pdf = cosThetaWi * scene_rdl2::math::sOneOverPi;
54 }
55
56 // Note: we assume this lobe has been setup with a OneMinus*Fresnel
57 // as we want to use 1 - specular_fresnel. Also notice we use
58 // cosThetaWo to evaluate the fresnel term, as an approximation of what
59 // hDotWi would be for the specular lobe.
60 float cosThetaWo = 1.0f;
61
62 if (getFresnel()) {
63 const scene_rdl2::math::Vec3f N = (matchesFlag(REFLECTION)) ? mFrame.getN() : -mFrame.getN();
64 cosThetaWo = scene_rdl2::math::max(dot(N, slice.getWo()), 0.0f);
65 }
66
67 // Soften hard shadow terminator due to shading normals
68 const float Gs = (matchesFlag(REFLECTION)) ?
69 slice.computeShadowTerminatorFix(mFrame.getN(), wi) :
70 1.0f;
71
72 return Gs * mAlbedo * computeScaleAndFresnel(cosThetaWo) * scene_rdl2::math::sOneOverPi *
73 (slice.getIncludeCosineTerm() ? cosThetaWi : 1.0f);
74 }
75
76
77 finline scene_rdl2::math::Color sample(const BsdfSlice &slice, float r1, float r2,
78 scene_rdl2::math::Vec3f &wi, float &pdf) const override
79 {
80 wi = mFrame.localToGlobal(sampleLocalHemisphereCosine(r1, r2));
81 return eval(slice, wi, &pdf);
82 }
83
84
85 finline scene_rdl2::math::Color albedo(const BsdfSlice &slice) const override
86 {
87 float cosThetaWo = scene_rdl2::math::max(dot(mFrame.getN(), slice.getWo()), 0.0f);
88 return computeScaleAndFresnel(cosThetaWo) * mAlbedo;
89 }
90
91 // no surface curvature required
92 finline void differentials(const scene_rdl2::math::Vec3f &wo, const scene_rdl2::math::Vec3f &wi,
93 float r1, float r2, const scene_rdl2::math::Vec3f &dNdx, const scene_rdl2::math::Vec3f &dNdy,
94 scene_rdl2::math::Vec3f &dDdx, scene_rdl2::math::Vec3f &dDdy) const override
95 {
96 // The hemisphere cosine sampling direction derivative seems like a very
97 // good approximation to the full derivative for diffuse sampling. This
98 // is why we ignore the input differentials.
99
100 // TODO: How can we avoid computing these twice (once here, once in
101 // sample()->sampleLocalHemisphereCosine()) ?
102 localHemisphereCosineDifferentials(r1, r2, dDdx, dDdy);
103
104 // The differentials form rectangles which get long and thin close
105 // to the hemisphere pole and equator. We prefer having ray
106 // differentials that form a square, but preserve the ray footprint.
107 squarifyRectangle(dDdx, dDdy);
108
109 dDdx = mFrame.localToGlobal(dDdx);
110 dDdy = mFrame.localToGlobal(dDdy);
111 }
112
113 bool getProperty(Property property, float *dest) const override
114 {
115 bool handled = true;
116
117 switch (property)
118 {
119 case PROPERTY_ROUGHNESS:
120 *dest = 1.0f;
121 *(dest + 1) = 1.0f;
122 break;
123 case PROPERTY_NORMAL:
124 {
125 const scene_rdl2::math::Vec3f &N = mFrame.getN();
126 *dest = N.x;
127 *(dest + 1) = N.y;
128 *(dest + 2) = N.z;
129 }
130 break;
131 case PROPERTY_PBR_VALIDITY:
132 {
133 scene_rdl2::math::Color res = computeAlbedoPbrValidity(mAlbedo);
134 *dest = res.r;
135 *(dest + 1) = res.g;
136 *(dest + 2) = res.b;
137 }
138 break;
139 case PROPERTY_COLOR:
140 {
141 // Special case handling for lambert color property since albedo isn't included
142 // in mScale anymore.
143 scene_rdl2::math::Color result = mAlbedo * computeScaleAndFresnel(1.f);
144 *dest = result.r;
145 *(dest + 1) = result.g;
146 *(dest + 2) = result.b;
147 }
148 break;
149 default:
150 handled = BsdfLobe::getProperty(property, dest);
151 break;
152 }
153
154 return handled;
155 }

157 void show(std::ostream& os, const std::string& indent) const override
158 {
159 const scene_rdl2::math::Color& scale = getScale();
160 const scene_rdl2::math::Vec3f& N = mFrame.getN();
161 const Fresnel * const fresnel = getFresnel();
162
163 os << indent << "[LambertBsdfLobe] "
164 << (matchesFlag(REFLECTION) ? "(reflection)" : "(transmission)") << "\n";
165 os << indent << " " << "scale: "
166 << scale.r << " " << scale.g << " " << scale.b << "\n";
167 os << indent << " " << "N: "
168 << N.x << " " << N.y << " " << N.z << "\n";
169 os << indent << " " << "albedo: "
170 << mAlbedo.r << " " << mAlbedo.g << " " << mAlbedo.b << "\n";
171 if (fresnel) {
172 fresnel->show(os, indent + " ");
173 }
174 }
175
176protected:
177 scene_rdl2::math::ReferenceFrame mFrame;
178 scene_rdl2::math::Color mAlbedo;
179};
