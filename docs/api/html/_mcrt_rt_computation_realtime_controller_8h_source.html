85 void initIntervalArray() {
86 // Several different test, 5 frame average works reasonable so far.
87 // This size is defines sensitivity of adjustDuration and dynamicFPS value change.
88 static const int size = 5;
89 mIntervalArray.resize(size, 1.0 / mDefaultFps); // take 5 frames average
90 mAdjustArray.resize(size, 0.0);
91 }
92
93 double computeFps(const double cInterval) {
94 return 1.0 / shiftPushAndAverageArray(mIntervalArray, cInterval);
95 }
96
97 double computeAdjustDuration(const double currDuration) {
98 //
99 // Returns recent averaged adjust duration values. This logic works well to remove unpredicted spike
100 // of adjust duration change. In other word, adjust duration should not change sensitibly.
101 // Sensitibity is controlled by number of array (see initIntervalArray()).
102 // Big buffer -> insensitive and flat, Small buffer -> peaky.
103 //
104 return shiftPushAndAverageArray(mAdjustArray, currDuration);
105 }
106
107 double shiftPushAndAverageArray(std::vector<double> &array, double v) {
108 //
109 // We only keep some number of recent values and averaged them.
110 // number is defined by initIntervalArray()
111 //
112 double total = 0.0;
113 for (size_t i = array.size() - 1; i > 0; --i) {
114 array[i] = array[i - 1];
115 total += array[i];
116 }
117 array[0] = v;
118 total += array[0];
119 return total / (double)array.size();
120 }
121};
