282 void transfer(const scene_rdl2::math::Vec3f &hitPoint, const scene_rdl2::math::Vec3f &hitNormal)
283 {
284 org = hitPoint;
285
286 if (hasDifferentials()) {
287 float denomX = dot(hitNormal, mDirX);
288 float denomY = dot(hitNormal, mDirY);
289
290 // TODO: do something more expensive here to avoid denormals?
291 if (denomX != 0.0f && denomY != 0.0f) {
292 float planeD = -dot(hitNormal, hitPoint);
293 float distX = -(dot(mOriginX, hitNormal) + planeD) / denomX;
294 float distY = -(dot(mOriginY, hitNormal) + planeD) / denomY;
295
296 mOriginX += mDirX * distX;
297 mOriginY += mDirY * distY;
298 } else {
299 mFlags.clear(HAS_DIFFERENTIALS);
300 }
301 }
302 }

309 const scene_rdl2::math::Vec3f &getDirX() const { return mDirX; }
310 const scene_rdl2::math::Vec3f &getOriginY() const { return mOriginY; }

317 float getDirFootprint() const
318 {
319 if (!hasDifferentials()) {
320 return scene_rdl2::math::neg_inf;
321 }
322 float lx = scene_rdl2::math::length(getdDdx());
323 float ly = scene_rdl2::math::length(getdDdy());
324 return 0.5f * scene_rdl2::math::log2(lx * ly); // = log2(sqrt(lx * ly))
325 }
326
328 scene_rdl2::math::Vec3f getdPdx() const { return mOriginX - getOrigin(); }
329 scene_rdl2::math::Vec3f getdPdy() const { return mOriginY - getOrigin(); }
330
333 scene_rdl2::math::Vec3f getdDdx() const { return mDirX - getDirection(); }
334 scene_rdl2::math::Vec3f getdDdy() const { return mDirY - getDirection(); }

336 void scaleDifferentials(float scale)
337 {
338 scene_rdl2::math::Vec3f const &origin = getOrigin();
339 scene_rdl2::math::Vec3f const &direction = getDirection();
340
341 mOriginX = origin + (mOriginX - origin) * scale;
342 mOriginY = origin + (mOriginY - origin) * scale;
343 mDirX = normalize(direction + (mDirX - direction) * scale);
344 mDirY = normalize(direction + (mDirY - direction) * scale);
345 }
