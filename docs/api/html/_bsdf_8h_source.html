32// General Convention:
33// - All directions below wo and wi are in the global shading coordinate system
34// of the caller (typically render space, but no assumption is made about this).
35// Internally, a BsdfLobe can use a ReferenceFrame class to
36// transform directions from/to global coordinate system to/from the local
37// reference frame. See the ReferenceFrame for more details.
38// - All directions wo and wi are guaranteed to be normalized and pointing
39// away from the surface.
40// - We follow the convention that the global space N and Ng are flipped
41// together so that Ng is pointing towards the observer direction wo -- i.e.
42// reversed incoming eye ray direction / reversed incoming photon direction
43// in the adjoint case.
44// If no flipping happens, the slice entering flag must be set to true, and
45// must be set to false otherwise. This flag indicates if we are entering or
46// leaving the volume inside an object.
47// - By the above conventions, the passed-in wo is guaranteed to be in the
48// positive geometric hemisphere (wrt. Ng). Part of that hemisphere may be in
49// the negative shading hemisphere (wrt. N) when bump mapping or/and
50// interpolated vertex normals are in effect.
51// - See more specific rules in the API below.
52

70 /// This is used to select bsdfs that contribute to reflection / transmission
71 /// or/and to select bsdfs that are categorized as diffuse / glossy / ideal mirror.
72 /// They are organized such that they can easily be translated into ray masks
73 /// (rdl2::Geometry::VisibilityType) by lobeTypeToRayMask()(PathIntegratorUtil.h).
74 enum Type {
75 NONE = 0,
76
77 // Surface side categories. Most lobes belong to only one category, but
78 // there are some exceptions.
79 REFLECTION = 1 << 0,
80 TRANSMISSION = 1 << 1,
81
82 // Lobe categories. These ARE mutually exclusive. A lobe can only
83 // belong to one of the categories below.
84 // The bit shift assignments are for easy determination of ray masking based on
85 // lobe type.
86 DIFFUSE = 1 << 2,
87 GLOSSY = 1 << 4,
88 MIRROR = 1 << 6,
89
90 // Mixed bags
91 ALL_SURFACE_SIDES = REFLECTION | TRANSMISSION,
92 ALL_LOBES = DIFFUSE | GLOSSY | MIRROR,
93 ALL_REFLECTION = ALL_LOBES | REFLECTION,
94 ALL_TRANSMISSION = ALL_LOBES | TRANSMISSION,
95 ALL_DIFFUSE = ALL_SURFACE_SIDES | DIFFUSE,
96 ALL_GLOSSY = ALL_SURFACE_SIDES | GLOSSY,
97 ALL_MIRROR = ALL_SURFACE_SIDES | MIRROR,
98 ALL = ALL_REFLECTION | ALL_TRANSMISSION
99 };

186 /// Evaluate the bsdf and (optionaly) the probability density of sampling the
187 /// direction wi (density measured wrt. solid angle) for the pair of
188 /// directions. wo is the observer or fixed direction and wi is the
189 /// light / varying direction. This function should behave consistently with
190 /// the sample() method below.
191 /// Rules:
192 /// - The returned bsdf should or should not include the cosine term of the
193 /// rendering equation, according to the slice.getIncludeCosineTerm() setting
194 /// - The bsdf should include the fresnel effect if any
195 /// - The returned pdf should always be positive. Zero should only be
196 /// returned if the sample() method actually has zero probability of
197 /// choosing wi.
198 /// - This function shouldn't need to test if wo and wi are in the same
199 /// hemisphere. If that is not the case, the function should still
200 /// evaluate the bsdf return the probability density of choosing wi (unless
201 /// this is required for numerical stability of the underlying bsdf model).
202 /// - Perfect mirror reflection / transmission Bsdfs should always return 0
203 /// for the bsdf and the pdf.
204 /// THREAD-SAFETY WARNING!!!
205 /// This API is not thread safe because it allows BsdfLobe to cache
206 /// computations that are redundant across lobes inside the Bsdf
207 /// class. This is only an issue if multiple threads need to sample / eval
208 /// the same hair Bsdf (lobe) instance. We only really need to do this in
209 /// our unittests, so there we're careful to use a copy of the bsdf per thread
210 virtual scene_rdl2::math::Color eval(const BsdfSlice &slice, const scene_rdl2::math::Vec3f &wi,
211 float *pdf = nullptr) const = 0;
212
213 /// Sample the bsdf for the given observer (fixed) direction wo.
214 /// This function should sample an incident direction wi and return
215 /// the probability density of choosing this direction given wo (density
216 /// measured wrt. solid angle). The returned pdf and evaluated color must
217 /// match the value returned by eval(slice, wi).
218 /// Rules:
219 /// - This function may return a wi direction that may be anywhere in the
220 /// full spherical domain. It is up to the caller to make sure this
221 /// wi is tested before calling eval() further.
222 /// - This function shouldn't need to test if wo and wi are in the same
223 /// geometric hemisphere. If that is the case, the function should still
224 /// return the probability density of choosing wi.
225 /// - A zero color sample may be returned if that is the result
226 /// of eval(slice, wi).
227 /// - A sample with zero probability may be returned if the function is
228 /// unable to sample given its inputs, in which case wi may be left
229 /// un-initialized. However, the pdf() should behave consistently and still
230 /// integrate to one over the hemisphere, otherwise this will result in
231 /// biased importance sampling.
232 /// - Perfect mirror reflection / transmission Bsdfs should sample the
233 /// mirror direction and return the proper contribution and a
234 /// probability of 1.
235 /// - Do not apply russian-roulette inside sample(), this should be done by
236 /// higher-level code.
237 /// - Bsdfs that rely on sampling half-vectors should make sure that
238 /// dot(wo, H) is less than 90 degrees, so the re-constructed H vector
239 /// in eval() is consistently oriented with the sampled H vector
240 /// THREAD-SAFETY WARNING!!!
241 /// This API is not thread safe because it allows BsdfLobe to cache
242 /// computations that are redundant across lobes inside the Bsdf
243 /// class. This is only an issue if multiple threads need to sample / eval
244 /// the same hair Bsdf (lobe) instance. We only really need to do this in
245 /// our unittests, so there we're careful to use a copy of the bsdf per thread
246 virtual scene_rdl2::math::Color sample(const BsdfSlice &slice, float r1, float r2,
247 scene_rdl2::math::Vec3f &wi, float &pdf) const = 0;
248
249 /// Compute a fast approximation of the albedo. The implementation can
250 /// assume the bsdf lobe is normalized, which means the un-scaled bsdf
251 /// function should always integrate to 1. Therefore the
252 /// implementation can just return the scale color * fresnel. The fresnel
253 /// term can also be computed assuming an ideal mirror reflection direction
254 /// as a further approximation.
255 /// This is only used for driving lobe sampling decisions and should only
256 /// affect the noise in the image, never the actual lighting intensity.
257 virtual scene_rdl2::math::Color albedo(const BsdfSlice &slice) const = 0;
258
259 /// Update the ray direction differentials (dDdx, dDdy), for a given
260 /// scattering event (wo, wi). H is the normalized half-vector.
261 /// The normal derivatives (dNdx, dNdy) are also provided to
262 /// take into account the surface curvature. The differentials are
263 /// normally computed by differentiating the sampling routine recursively
264 /// along the path wrt. the random variables used to sample the path.
265 /// This is sometimes overly complex for "blurry" scattering events and
266 /// a simpler approximation is enough (i.e. taking only the diffuse
267 /// sampling into account for diffuse scattering). The routine should assume
268 /// only one sample is taken over the entire domain and the renderer will
269 /// scale the differentials automatically when path splitting happens.
270 //
271 /// TODO: Implement all BsdfLobes
272 /// TODO: It's unclear at the moment if this math should be part of the
273 /// sample() method. Right now it looks like a lot of redundant math which
274 /// could be avoided, but for the sake of expediency and for an initial test,
275 /// I will stick to this for now.
276 virtual void differentials(const scene_rdl2::math::Vec3f &wo, const scene_rdl2::math::Vec3f &wi,
277 float r1, float r2, const scene_rdl2::math::Vec3f &dNdx, const scene_rdl2::math::Vec3f &dNdy,
278 scene_rdl2::math::Vec3f &dDdx, scene_rdl2::math::Vec3f &dDdy) const = 0;
279
280 // prints out a description of this lobe with the provided indentation
281 // prepended.
282 virtual void show(std::ostream& os, const std::string& indent) const = 0;

297/// Utility functions to compute the minimum roughness to be used during
298/// roughness clamping
299finline scene_rdl2::math::Vec2f
300computeMinRoughness(const BsdfLobe &lobe, float roughnessClampingFactor,
301 const scene_rdl2::math::Vec2f &previousMinRoughness)
302{
303 scene_rdl2::math::Vec2f minRoughness(0.0f);
304
305 if (roughnessClampingFactor > 0.0f) {
306 scene_rdl2::math::Vec2f roughness;
307 if (lobe.getProperty(BsdfLobe::PROPERTY_ROUGHNESS, &roughness[0])) {
308 minRoughness = roughness * roughnessClampingFactor;
309 }
310
311 // Keep smaller of the two min roughnesses in x
312 if (minRoughness.x > minRoughness.y) {
313 std::swap(minRoughness.x, minRoughness.y);
314 }
315 }
316
317 minRoughness = scene_rdl2::math::clamp(minRoughness, previousMinRoughness, scene_rdl2::math::Vec2f(1.0f));
318 return minRoughness;
319}
320
321finline scene_rdl2::math::Vec2f
322computeMinRoughness(float lobeRoughness, float roughnessClampingFactor,
323 const scene_rdl2::math::Vec2f &previousMinRoughness)
324{
325 scene_rdl2::math::Vec2f minRoughness(0.0f);
326
327 if (roughnessClampingFactor > 0.0f) {
328 minRoughness = scene_rdl2::math::Vec2f(lobeRoughness * roughnessClampingFactor);
329 }
330
331 minRoughness = scene_rdl2::math::clamp(minRoughness, previousMinRoughness, scene_rdl2::math::Vec2f(1.0f));
332 return minRoughness;
333}
