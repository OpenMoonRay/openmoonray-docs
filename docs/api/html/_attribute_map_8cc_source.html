50void
51AttributeMap::update()
52{
53 if (hasChanged(attrMapType) ||
54 hasChanged(attrPrimitiveAttributeType) ||
55 hasChanged(attrPrimitiveAttributeName)) {
56 mOptionalAttributes.clear();
57 if (get(attrMapType) == ispc::AttributeMapType::PRIMITIVE_ATTRIBUTE) {
58 int type = get(attrPrimitiveAttributeType);
59 const std::string& name = get(attrPrimitiveAttributeName);
60 if (type == ispc::TYPE_FLOAT) {
61 mData.mPrimitiveAttributeType = ispc::TYPE_FLOAT;
62 moonray::shading::TypedAttributeKey<float> attributeKey(name);
63 mData.mPrimitiveAttributeIndex = attributeKey;
64 mOptionalAttributes.push_back(attributeKey);
65 createLogEvent("float", name);
66 } else if (type == ispc::TYPE_VEC2F) {
67 mData.mPrimitiveAttributeType = ispc::TYPE_VEC2F;
68 moonray::shading::TypedAttributeKey<Vec2f> attributeKey(name);
69 mData.mPrimitiveAttributeIndex = attributeKey;
70 mOptionalAttributes.push_back(attributeKey);
71 createLogEvent("vec2f", name);
72 } else if (type == ispc::TYPE_VEC3F) {
73 mData.mPrimitiveAttributeType = ispc::TYPE_VEC3F;
74 moonray::shading::TypedAttributeKey<Vec3f> attributeKey(name);
75 mData.mPrimitiveAttributeIndex = attributeKey;
76 mOptionalAttributes.push_back(attributeKey);
77 createLogEvent("vec3f", name);
78 } else if (type == ispc::TYPE_RGB) {
79 mData.mPrimitiveAttributeType = ispc::TYPE_RGB;
80 moonray::shading::TypedAttributeKey<Color> attributeKey(name);
81 mData.mPrimitiveAttributeIndex = attributeKey;
82 mOptionalAttributes.push_back(attributeKey);
83 createLogEvent("rgb", name);
84 } else if (type == ispc::TYPE_INT) {
85 mData.mPrimitiveAttributeType = ispc::TYPE_INT;
86 moonray::shading::TypedAttributeKey<int> attributeKey(name);
87 mData.mPrimitiveAttributeIndex = attributeKey;
88 mOptionalAttributes.push_back(attributeKey);
89 createLogEvent("int", name);
90 } else {
91 MNRY_ASSERT(0, "unsupported primitive attribute type");
92 }
93 } else if (get(attrMapType) == ispc::AttributeMapType::SURFACE_P) {
94 mData.mPrimitiveAttributeType = ispc::TYPE_VEC3F;
95 moonray::shading::TypedAttributeKey<Vec3f> attributeKey("surface_P");
96 mData.mPrimitiveAttributeIndex = attributeKey;
97 createLogEvent("vec3f", "surface_P");
98 mOptionalAttributes.push_back(attributeKey);
99 } else if (get(attrMapType) == ispc::AttributeMapType::SURFACE_N) {
100 mData.mPrimitiveAttributeType = ispc::TYPE_VEC3F;
101 moonray::shading::TypedAttributeKey<Vec3f> attributeKey("surface_N");
102 mData.mPrimitiveAttributeIndex = attributeKey;
103 mOptionalAttributes.push_back(attributeKey);
104 createLogEvent("vec3f", "surface_N");
105 } else if (get(attrMapType) == ispc::AttributeMapType::SURFACE_ST) {
106 mData.mPrimitiveAttributeType = ispc::TYPE_VEC2F;
107 mData.mPrimitiveAttributeIndex = moonray::shading::StandardAttributes::sSurfaceST;
108 mOptionalAttributes.push_back(mData.mPrimitiveAttributeIndex);
109 createLogEvent("vec2f", "surface_st");
110 } else if (get(attrMapType) == ispc::AttributeMapType::CLOSEST_SURFACE_ST) {
111 mData.mPrimitiveAttributeType = ispc::TYPE_VEC2F;

153void
154AttributeMap::sample(const scene_rdl2::rdl2::Map* self, moonray::shading::TLState *tls,
155 const moonray::shading::State& state, Color* sample)
156{
157 const AttributeMap* me = static_cast<const AttributeMap*>(self);
158 if ( me->get(attrMapType) == ispc::AttributeMapType::PRIMITIVE_ATTRIBUTE ||
159 me->get(attrMapType) == ispc::AttributeMapType::SURFACE_P ||
160 me->get(attrMapType) == ispc::AttributeMapType::SURFACE_N ||
161 me->get(attrMapType) == ispc::AttributeMapType::SURFACE_ST ||
162 me->get(attrMapType) == ispc::AttributeMapType::CLOSEST_SURFACE_ST ||
163 me->get(attrMapType) == ispc::AttributeMapType::ID ||
164 me->get(attrMapType) == ispc::AttributeMapType::VELOCITY ||
165 me->get(attrMapType) == ispc::AttributeMapType::ACCELERATION ||
166 me->get(attrMapType) == ispc::AttributeMapType::MOTIONVEC) {
167
168 int key = me->mData.mPrimitiveAttributeIndex;
169 int type = me->mData.mPrimitiveAttributeType;
170 if (state.isProvided(key)) {
171 if (type == ispc::TYPE_FLOAT) {
172 float f = state.getAttribute(
173 moonray::shading::TypedAttributeKey<float>(key));
174 *sample = Color(f);
175 } else if (type == ispc::TYPE_VEC2F) {
176 const Vec2f& v2 = state.getAttribute(
177 moonray::shading::TypedAttributeKey<Vec2f>(key));
178 sample->r = v2.x;
179 sample->g = v2.y;
180 sample->b = 0.0f;
181 } else if (type == ispc::TYPE_VEC3F) {
182 const Vec3f& v3 = state.getAttribute(
183 moonray::shading::TypedAttributeKey<Vec3f>(key));
184 sample->r = v3.x;
185 sample->g = v3.y;
186 sample->b = v3.z;
187 } else if (type == ispc::TYPE_RGB) {
188 *sample = state.getAttribute(
189 moonray::shading::TypedAttributeKey<Color>(key));
190 } else if (type == ispc::TYPE_INT) {
191 int i = state.getAttribute(
192 moonray::shading::TypedAttributeKey<int>(key));
193 *sample = Color((float)i);
194 } else {
195 // there is an attribute with the right name, but the
196 // type is unknown/unsupported - so report it as missing
197 // and use the default value
198 *sample = evalColor(me, attrDefaultValue, tls, state);
199 if (me->get(attrWarnWhenUnavailable)) {
200 moonray::shading::logEvent(me, tls, me->mData.mMissingAttributeEvent);
201 }
202 }
203 } else {
204 // the primitive attribute is unavailable
205 *sample = evalColor(me, attrDefaultValue, tls, state);
206 if (me->get(attrWarnWhenUnavailable)) {
207 moonray::shading::logEvent(me, tls, me->mData.mMissingAttributeEvent);
208 }
209 }
210 } else if (me->get(attrMapType) == ispc::AttributeMapType::P) {
211 const Vec3f& p = state.getP();
212 sample->r = p.x;
213 sample->g = p.y;
214 sample->b = p.z;
215 } else if (me->get(attrMapType) == ispc::AttributeMapType::ST) {
216 const Vec2f& st = state.getSt();
217 sample->r = st.x;
218 sample->g = st.y;
219 sample->b = 0.0f;
220 } else if (me->get(attrMapType) == ispc::AttributeMapType::N) {
221 const Vec3f& n = state.getN();
222 sample->r = n.x;
223 sample->g = n.y;
224 sample->b = n.z;
225 } else if (me->get(attrMapType) == ispc::AttributeMapType::NG) {
226 const Vec3f& ng = state.getNg();
227 sample->r = ng.x;
228 sample->g = ng.y;
229 sample->b = ng.z;
230 } else if (me->get(attrMapType) == ispc::AttributeMapType::DPDS) {
231 const Vec3f& dpds = state.getdPds();
232 sample->r = dpds.x;
233 sample->g = dpds.y;
234 sample->b = dpds.z;
235 } else if (me->get(attrMapType) == ispc::AttributeMapType::DPDT) {
236 const Vec3f& dpdt = state.getdPdt();
237 sample->r = dpdt.x;
238 sample->g = dpdt.y;
239 sample->b = dpdt.z;
240 } else if (me->get(attrMapType) == ispc::AttributeMapType::DNDS) {
241 const Vec3f& dnds = state.getdNds();
242 sample->r = dnds.x;
243 sample->g = dnds.y;
244 sample->b = dnds.z;
245 } else if (me->get(attrMapType) == ispc::AttributeMapType::DNDT) {
246 const Vec3f& dndt = state.getdNdt();
247 sample->r = dndt.x;
248 sample->g = dndt.y;
249 sample->b = dndt.z;
250 } else if (me->get(attrMapType) == ispc::AttributeMapType::MAP_COLOR) {
251 *sample = evalColor(me, attrColor, tls, state);
252 } else {
253 MNRY_ASSERT(0, "unsupported primitive attribute type");
254 }
255}
