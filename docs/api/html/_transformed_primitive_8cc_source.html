21struct TransformedPrimitive::Impl
22{
23 explicit Impl(const shading::XformSamples& xform,
24 std::unique_ptr<Primitive> primitive) :
25 mXform(xform), mPrimitive(std::move(primitive)),
26 mHasBeenApplied(false) {
27 if (mXform.empty()) {
28 mXform.push_back(Mat43(scene_rdl2::math::one));
29 }
30 }
31
32 Primitive::size_type getMemory() const {
33 return sizeof(TransformedPrimitive::Impl) +
34 scene_rdl2::util::getVectorElementsMemory(mXform) + mPrimitive->getMemory();
35 }
36
37 shading::XformSamples mXform;
38 std::unique_ptr<Primitive> mPrimitive;
39 // TransformedPrimitive should only apply mXform to mPrimitive once during
40 // render prepare stage, flip this flag once transformPrimitive has been called
41 bool mHasBeenApplied;
42};

80void
81TransformedPrimitive::transformPrimitive(
82 const MotionBlurParams& motionBlurParams,
83 const shading::XformSamples& prim2render)
84{
85 if (!mImpl->mHasBeenApplied) {
86 shading::XformSamples local2render = concatenate(mImpl->mXform, prim2render);
87 internal::PrimitivePrivateAccess::transformPrimitive(
88 mImpl->mPrimitive.get(), motionBlurParams,
89 local2render);
90 mImpl->mHasBeenApplied = true;
91 }
92}
