80// Taken from "A Comparison of Methods for Generating Poisson Disk
81// Distributions" by Lagae and Dutré. This computes the maximum radius of disks
82// in optimal disk (circle) packing (spoiler: it's a hex lattice). This assumes
83// the radius to the edges of other disks, but the Poisson radius is to other
84// points, so it needs to be doubled after the fact.
85template <>
86inline float radiusMax<2>(unsigned ndisks, float volume)
87{
88 static const float kSqrt3 = std::sqrt(3.0f);
89 return std::sqrt(volume / (2.0f * kSqrt3 * static_cast<float>(ndisks)));
90}

92// Korkin A., Zolotarev G.: Sur les formes quadratiques positives.
93// Mathematische Annalen 11 (1877), 242–292
94template <>
95inline float radiusMax<3>(unsigned ndisks, float volume)
96{
97 static const float kSqrt2 = std::sqrt(2.0f);
98 return std::pow(volume / (4.0f * kSqrt2 * static_cast<float>(ndisks)), 1.0f/3.0f);
99}
