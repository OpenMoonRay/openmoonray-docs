194 // Traverse a HardSurfaceSegment linked list and check if the specified
195 // subpixel (mask bit) is blocked by the hard surface at the distance value
196 // t.
197 bool isBlockedByHardSurface(const HardSurfaceSegment *firstSegment,
198 uint64_t maskBit,
199 float t) const
200 {
201 uint64_t mask = 0;
202 const HardSurfaceSegment *current = firstSegment;
203 while (current) {
204 if (current->mTBack < t) {
205 mask |= current->mMask;
206 if (mask & maskBit) {
207 return true;
208 }
209 }
210 current = current->mNext;
211 }
212 return false;
213 }

231 /*
232 * Deep Volumes Overview:
233 *
234 * The PathIntegratorVolume sends both volume segments (VolumeInputSegment)
235 * and radiance/aov samples along the ray (VolumeSample) to the DeepBuffer.
236 * These are stored in a VolumePixelBuffer structure until all of a pixel's segments
237 * and samples are collected. The segments and samples are then merged to
238 * produce a much smaller set of VolumeOutputSegments which are written to the
239 * deep file.
240 *
241 * For each pixel, there is a list of VolumeInputSegments for each pixel sample.
242 * There is only one list of VolumeSamples for the pixel, for all pixel samples.
243 * One list of VolumeOutputSegments is output for each pixel from merging the many
244 * lists of VolumeInputSegments and the single list of VolumeSamples.
245 * The VolumeOutputSegments are a simplified representation
246 * of the actual volume. They are processed to consume minimal space in
247 * the deep file but still produce a compositing result that closely matches
248 * the flat render output from the original volume. There will inevitably
249 * be minor differences due to this simplification.
250 *
251 * The amount of segment merging / deep file size is adjustable so a tradeoff can
252 * be made between deep file size and maintaining the detail of the original
253 * volume. Great care is taken with the math to minimize the visible effects of segment
254 * merging.
255 *
256 * Volume and hard surface data may be mixed within a deep file and these
257 * should interact/composite properly. An example would be a flaming sword
258 * which is a combination of hard surface (the blade) and volume (the fire)
259 * elements.
260 *
261 * A simplifying assumption is made that the coverage mask for a volume segment
262 * in the deep file is either full coverage or no coverage. Volumes tend to
263 * have "soft" edges so the lack of subpixel-accurate coverage information is not
264 * visually noticeable. The code has special treatment of volume segments that
265 * are partially overlapped by hard surfaces.
266 */
267
268 // Segments received from the ray-marching code in the VolumeIntegrator.
269 // No channel data is associated with these. mSigmaT is extinction coefficient
270 // of the volume for the segment, which is used to calculate transmittance with
271 // Beer's law.
272 struct VolumeInputSegment
273 {
274 VolumeInputSegment *mNext;
275 float mTFront;
276 float mTBack;
277 scene_rdl2::math::Color mSigmaT;
278 };
279
280 struct VolumeInputSegmentList
281 {
282 VolumeInputSegment *mFirst;
283 VolumeInputSegment *mLast;
284 unsigned mNumSegments;
285 };
286
287 // One sample of the channel data (radiance + aovs) at a location mT along
288 // the ray inside the volume. mTransmittance gives the transmittance from the
289 // camera to the location mT. The sample may have an arbitrary number of RGB
290 // and aov channels, hence the zero-length array which is a placeholder for
291 // the channel data.
292 struct VolumeSample
293 {
294 VolumeSample *mNext;
295 float mT;
296 scene_rdl2::math::Color mTransmittance;
297 float mChannels[0];
298 };
299
300 // The size of the VolumeSample depends on the number of channels.
301 // Note that we can only dynamically allocate this data structure due to the
302 // unknown size at compile time. This is due to the number of aov channels
303 // which varies per scene context.
304 size_t getVolumeSampleSize() const {
305 return sizeof(VolumeSample) + mNumChannels * sizeof(float);
306 }
307
308 // VolumeInputSegments and VolumeSamples are merged together to produce
309 // VolumeOutputSegments which are output to the deep file.
310 struct VolumeOutputSegment
311 {
312 VolumeOutputSegment *mNext;
313 float mTFront;
314 float mTBack;
315
316 // mTtoZ is a scaling value that converts ray T values to deep Z values.
317 // We work in ray T space right up until the deep file export. This
318 // keeps things consistent between hard surfaces and volumes.
319 float mTtoZ;
320
321 // The combined sigmaT for the segment, created by merging the sigmaTs of
322 // multiple VolumeInputSegments.
323 scene_rdl2::math::Color mSigmaT;
324
325 // VolumeOutputSegments can be partially occluded by hard surfaces.
326 // The amount of occlusion is given by mNumBlockedSamples / mNumPixelSamples.
327 // We need this to compensate for the missing (blocked / occluded) samples.
328 unsigned mNumUnblockedSamples;
329
330 // Again, variable number of channels (beauty + aov) that varies per
331 // scene context.
332 float mChannels[0];
333 };
334 // This works the same as the HardSurfaceSegments. There is one linked list of
335 // VolumeOutputSegments per pixel.
336 mutable std::vector<VolumeOutputSegment *> mVolumeOutputSegments;
337
338 // Similarly to VolumeSample, the size depends on the number of channels
339 // and is only known at run time.
340 size_t getVolumeOutputSegmentSize() const {
341 return sizeof(VolumeOutputSegment) + mNumChannels * sizeof(float);
342 }
343
344 // One of these per thread. Keeps track of the data pertaining to the current
345 // pixel being rendered. Note that this only works with batch render mode and NOT
346 // progressive render mode because batch render mode fully renders a pixel before
347 // moving on to the next one but progressive render mode only partially renders
348 // the pixel and makes multiple passes.
349 // Supporting progressive render mode would require a VolumePixelBuffer for
350 // EACH pixel in the image which would require a prohibitive amount of memory.
351 // With batch render mode we only need one of these per thread as each thread
352 // works on one pixel at a time. The buffer can then be cleared and re-used
353 // for the next pixel.
354 struct VolumePixelBuffer
355 {
356 // x, y coords of the current pixel being worked on by the thread.
357 unsigned mCurrentX;
358 unsigned mCurrentY;
359
360 // Total number of pixel samples for this pixel...
361 unsigned mNumPixelSamples;
362
363 // Max number of pixel samples... we just discard any samples over this limit.
364 static const unsigned mMaxPixelSamples = 1024;
365
366 // For each pixel sample we accumulate the ray.z value and then divide it
367 // by the number of pixel samples to produce an average ray Z value for
368 // the pixel. This is needed to properly handle camera motion blur where the
369 // ray.z value varies significantly between the pixel samples.
370 float mAccumRayZ;
371
372 // One linked list of VolumeInputSegments per pixel sample.
373 // We keep track of the first and last segments in the list, and the
374 // number of segments.
375 VolumeInputSegmentList mInputSegmentLists[mMaxPixelSamples];
376
377 // Which subpixel mask bit does the subpixel overlap?
378 uint64_t mMaskBits[mMaxPixelSamples];
379
380 // Linked list of VolumeSamples for the pixel. Not associated with any
381 // particular pixel sample as these all get merged into the VolumeOutputSegments.
382 // We don't care about the order of these and don't need to keep track of first
383 // + last pointers like the VolumeInputSegments.
384 VolumeSample *mSampleList;
385
386 void clear();
387
388 // Merges the VolumeInputSegments and the VolumeSamples together into a
389 // list of VolumeOutputSegments. Merging/compression is applied based on the
390 // volCompressionRes. The hard surfaces need to be considered to handle
391 // partial occlusion of VolumeOutputSegments by hard surfaces properly.
392 VolumeOutputSegment* mergeSegments(const DeepBuffer& deepBuffer,
393 unsigned samplesPerPixel,
394 unsigned volCompressionRes,
395 const HardSurfaceSegment *hardSegments);
396 };

408/* Deep pixels are independent of each other, so there is no threading hazard
409 * when different threads are writing to different pixels. It is possible
410 * (although uncommon) that multiple threads might write to the same pixel,
411 * so that needs to be protected with a mutex. The simple solution is to
412 * have one mutex per pixel. This would consume a lot of memory, so instead
413 * there is an array of 225 mutexes for the entire image that are shared
414 * between the pixels. Some pixels will share the same mutex, which results
415 * in some extra locking... however testing has shown the overhead is negligible
416 * once there are >= 64 mutexes for ~20 rendering threads. This may need to be
417 * increased as the number of threads increases though.
418 * The mutexes form a 15x15 tile that repeats across the image, and the
419 * getMutexIdx() logic maps a screen coordinate to a [0, 225] tile index.
420 * Note that the tile size is not a multiple of the rendering 8x8 tile size.
421 * This reduces mutex sharing during rendering.
422 */
423 static const int mMutexTileSize = 15;
424 // force mutex to be cache-line aligned for speed
425 struct CACHE_ALIGN AlignedMutex : public tbb::mutex {};
426 AlignedMutex *mPixelMutex;
427 int getMutexIdx(unsigned x, unsigned y) const {
428 return (y % mMutexTileSize) * mMutexTileSize + (x % mMutexTileSize);
429 }
