27// Using the golden ratio for the minimum node size. Because we divide in real-space, a value of 1 probably still
28// ensures that we end up sampling in neighboring tiles to reduce premature convergence issues. Also, being the most
29// irrational number, this may lead to fewer tiling artifacts.
30//static constexpr float sGoldenRatio = 1.61803398874989f;
31static constexpr float sMinNodeSize = 1.27201964951407f; // sqrt(sGoldenRatio)
32static constexpr float sMaxNodeSize = 16.0f;
33static constexpr float sNoSplit = std::numeric_limits<float>::quiet_NaN();
34
35inline bool noSplit(float v)
36{
37 return std::isnan(v);
38}
39
40static float findSplitLocationHelper(float location, float length)
41{
42 if (length * location >= sMinNodeSize) {
43 // Our split is within minimum node size.
44 return location;
45 } else if (length >= sMinNodeSize * 2) {
46 // Our split isn't within minimum node size: move split location.
47 return sMinNodeSize/length;
48 } else {
49 // We can't split and still be within minimum node size.
50 return sNoSplit;
51 }
52}
53
54float AdaptiveRegionTree::findSplitLocation(float location, float length)
55{
56 if (location <= 0.5f) {
57 return findSplitLocationHelper(location, length);
58 } else {
59 return 1.0f - findSplitLocationHelper(1.0f - location, length);
60 }
61}

63int AdaptiveRegionTree::maxNodes(const scene_rdl2::math::BBox2i& bounds)
64{
65 const int w = extents(bounds, 0);
66 const int h = extents(bounds, 1);
67
68 if (w == 0 || h == 0) {
69 return 0;
70 }
71 // Our leaf nodes are non-overlapping. This is the maximum number of leaf nodes we can have.
72 const int nLeafNodes = static_cast<int>(scene_rdl2::math::ceil((w/sMinNodeSize) * (h/sMinNodeSize)));
73
74 // A perfect binary tree has this many nodes given this many leaf nodes.
75 return 2*nLeafNodes - 1;
76}

78/// @return Average error of leaf nodes.
79float AdaptiveRegionTree::update(const scene_rdl2::fb_util::Tiler& tiler,
80 const scene_rdl2::fb_util::RenderBuffer& renderBuf,
81 const scene_rdl2::fb_util::FloatBuffer& numSamplesBuf,
82 const scene_rdl2::fb_util::RenderBuffer& renderBufOdd,
83 float targetError)
84{
85 // On update, we rebuild the entire tree so that it gets rebalanced.
86 mRoot.mChildren = nullptr;
87 mRoot.mDone = false;
88 mDone = false;
89 mNodePool.clear();
90
91 // mRootBounds should be closed-open (inclusive-exclusive).
92 for (auto pit = begin(mRootBounds); pit != end(mRootBounds); ++pit) {
93 const scene_rdl2::math::Vec2i& p = *pit;
94 const scene_rdl2::math::Vec2i pRelative = p - mRootBounds.lower;
95 float pixelError = getPixelError(p[0], p[1], tiler, renderBuf, numSamplesBuf, renderBufOdd);
96 if (std::isinf(pixelError)) {
97 return pixelError;
98 }
99 mPixelErrors(pRelative[0], pRelative[1]) = pixelError;
100 }
101 const float error = updateImpl(mRoot, targetError);
102 mDone = error < targetError;
103 return error;
104}

106/// @return Average error of leaf nodes.
107float AdaptiveRegionTree::updateImpl(Node& node, float targetError)
108{
109 using AdaptiveNS::Axis;
110
111 // This threshold comes from the Dammertz paper.
112 constexpr float sSplitThreshold = 256.0f;
113
114 AdaptiveNS::MemoryPoolRAII<float> accumulatedErrorPoolClearRAII(mAccumulatedErrorPool);
115 AdaptiveNS::PoolAllocator<float> accumlatedErrorAllocator(mAccumulatedErrorPool);
116
117 const auto ext = size(node.mBounds);
118 const float width = ext[0];
119 const float height = ext[1];
120
121 MNRY_ASSERT(volume(node.mBounds) > 0);
122 const Axis axis = (width > height) ? Axis::horizontal : Axis::vertical;
123 const float lengthAlongSplit = std::max(width, height);
124
125 const scene_rdl2::math::BBox2i nodeBounds = roundLarger(node.mBounds);
126
127 // _accumulatedError is the partial sum of the pixel errors and the last element is the sum of all pixel errors
128 // in the vertical or horizontal axis.
129 auto accumulatedError = (axis == Axis::horizontal) ?
130 AdaptiveNS::orientedAccumulatedPixelError<0>(mRootBounds,
131 nodeBounds,
132 mPixelErrors,
133 accumlatedErrorAllocator) :
134 AdaptiveNS::orientedAccumulatedPixelError<1>(mRootBounds,
135 nodeBounds,
136 mPixelErrors,
137 accumlatedErrorAllocator);
138
139 const float vr = volume(node.mBounds);
140 const float error = calculateAreaError(accumulatedError.back(), vr);
141
142 if (error < targetError && lengthAlongSplit < sMaxNodeSize) {
143 node.mDone = true;
144 return error;
145 } else if (error < targetError*sSplitThreshold && lengthAlongSplit > sMinNodeSize*2) { // We can split in half
146 const float offset = findSplitLocation(accumulatedError, lengthAlongSplit);
147 MNRY_ASSERT(!noSplit(offset));
148 if (unlikely(noSplit(offset))) {
149 return error;
150 }
151
152 // This is just a sanity check to make sure that our offset is reasonable. We only look at a very-relaxed
153 // .9 of the node size for precision errors.
154 MNRY_ASSERT(lengthAlongSplit * offset >= sMinNodeSize * 0.9f);
155 MNRY_ASSERT(lengthAlongSplit * (1.0f - offset) >= sMinNodeSize * 0.9f);
156 MNRY_ASSERT(offset >= 0.0f);
157 MNRY_ASSERT(offset <= 1.0f);
158
159 // Now that we've divided the error, we no longer need the value nor the memory. Free up for recursion.
160 // Since accumulatedError is just a container of floats, we don't _strictly_ have to clear it, but let's be
161 // proper so that we don't call the destructor on memory that's no longer ours.
162 accumulatedError.clear();
163 mAccumulatedErrorPool.clear();
164 switch (axis) {
165 case Axis::horizontal:
166 splitHorizontal(mNodePool, node, offset);
167 break;
168 case Axis::vertical:
169 splitVertical(mNodePool, node, offset);
170 break;
171 }
172 MNRY_ASSERT(hasChildren(node));
173 const float errorLeft = updateImpl(node.getLeftChild(), targetError);
174 const float errorRight = updateImpl(node.getRightChild(), targetError);
175 if (node.getLeftChild().mDone && node.getRightChild().mDone) {
176 node.mDone = true;
177 }
178 // Return the mean error of our children.
179 return scene_rdl2::math::max(errorLeft, errorRight);
180 }
181 return error;
182}

184scene_rdl2::math::BBox2i AdaptiveRegionTree::getSampleAreaImpl(const scene_rdl2::math::BBox2f& toCheck, const Node& node) const
185{
186 if (node.mDone) {
187 return scene_rdl2::math::BBox2i{scene_rdl2::util::empty};
188 }
189
190 const scene_rdl2::math::BBox2f isect = intersect(node.mBounds, toCheck);
191 if (isect.empty()) {
192 return scene_rdl2::math::BBox2i{scene_rdl2::util::empty};
193 }
194
195 if (hasChildren(node)) {
196 return merge(getSampleAreaImpl(isect, node.getLeftChild()), getSampleAreaImpl(isect, node.getRightChild()));
197 } else {
198 return roundLarger(isect);
199 }
200}

235bool AdaptiveRegionTree::savePixelErrorsByPPM(const std::string& filename) const
236//
237// For debugging purpose
238// save pixel error valaue into disk by PPM (RGB) format w/ channel value resolution as 0~4095.
239// R : pixel error clipped by 0.0~1.0 range goes to value of 0 ~ 4095.
240// G : pixel is covered by adaptive tree node (4095) or not (0)
241// B : pixel is converged (4095) or not-converged (0)
242//
243{
244 constexpr int valReso = 4096;
245 auto toIntPixError = [&](const float v) -> int {
246 auto clamp = [](const float v) -> float { return ((v < 0.0f)? 0.0f: ((v > 1.0f)? 1.0f: v)); };
247 return (int)(clamp(v) * (float)(valReso - 1) + 0.5f);
248 };
249 auto toIntPixCondition = [&](const bool flag) -> int { return (flag)? 0: valReso - 1; };
250 auto getPixCondition = [&](const int pixCondition, bool& doneFlag, bool& nodeCoveredFlag) {
251 // 3 2 1 0
252 // | |
253 // | +-- pixel covered by Node
254 // +---- Done pixel
255 nodeCoveredFlag = (pixCondition & 0x1)? true: false;
256 doneFlag = (pixCondition & 0x2)? true: false;
257 };
258
259 std::ofstream ofs(filename);
260 if (!ofs) return false; // can not create file.
261
262 scene_rdl2::util::Array2D<int> pixCondition = calcPixelCondition();
263 const scene_rdl2::util::Array2D<float>& pixError = mPixelErrors;
264
265 ofs << "P3\n" << pixError.getWidth() << ' ' << pixError.getHeight() << '\n' << (valReso - 1) << '\n';
266 for (int v = pixError.getHeight() - 1; v >= 0; --v) {
267 for (int u = 0; u < pixError.getWidth(); ++u) {
268 bool nodeCoveredFlag, doneFlag;
269 getPixCondition(pixCondition(u, v), doneFlag, nodeCoveredFlag);
270 ofs << toIntPixError(pixError(u, v)) << ' '
271 << toIntPixCondition(nodeCoveredFlag) << ' '
272 << toIntPixCondition(doneFlag) << ' ';
273 }
274 }
275
276 ofs.close();
277 return true;
278}

284// For debugging purpose
285// Save renderBuf's RGB (clipped 0.0~1.0) to PPM format w/ value resolution as 0~4095.
286//
287{
288 constexpr int valReso = 4096;
289 auto toIntFCol = [&](const float v) -> int {
290 auto clamp = [](const float v) -> float { return ((v < 0.0f)? 0.0f: ((v > 1.0f)? 1.0f: v)); };

297 int xMin = mRootBounds.lower[0];
298 int yMin = mRootBounds.lower[1];
299 int xMax = mRootBounds.upper[0] - 1;
300 int yMax = mRootBounds.upper[1] - 1;
301 int xSize = xMax - xMin + 1;
302 int ySize = yMax - yMin + 1;
303
304 ofs << "P3\n" << xSize << ' ' << ySize << '\n' << (valReso - 1) << '\n';
305 for (int y = yMax; y >= yMin; --y) {
306 for (int x = xMin; x <= xMax; ++x) {
307 unsigned px, py;
308 tiler.linearToTiledCoords((unsigned)x, (unsigned)y, &px, &py);
309 const scene_rdl2::fb_util::RenderColor& c = renderBuf.getPixel(px, py);
310 ofs << toIntFCol(c[0]) << ' ' << toIntFCol(c[1]) << ' ' << toIntFCol(c[2]) << ' ';
311 }
312 }
313
314 ofs.close();
315 return true;

356scene_rdl2::util::Array2D<int> AdaptiveRegionTree::calcPixelCondition() const
357//
358// For debugging purpose
359// Compute pixel condition array based on adaptive tree information
360//
361{
362 auto calcPixConditionVal = [&](const bool donePixel) -> int {
363 // 3 2 1 0
364 // | |
365 // | +-- pixel covered by Node
366 // +---- Done pixel
367 int flag = 0x1;
368 if (donePixel) flag |= 0x2;
369 return flag;
370 };
371 auto fillRegionByVal = [&](scene_rdl2::util::Array2D<int>& array, const scene_rdl2::math::BBox2f& bbox, const int val) {
372 for (int v = (int)bbox.lower[1]; v <= (int)bbox.upper[1]; ++v) {
373 for (int u = (int)bbox.lower[0]; u <= (int)bbox.upper[0]; ++u) {
374 int uu = u - mRootBounds.lower[0];
375 int vv = v - mRootBounds.lower[1];
376 if (!(array(uu, vv) & 0x1)) {
377 array(uu, vv) = val; // very 1st time to set
378 } else {
379 if (array(uu, vv) & 0x2) {
380 array(uu, vv) = val; // only update if this pixel is done-condition
381 }
382 }
383 }
384 }
385 };
386 auto fillRegionByCondition = [&](scene_rdl2::util::Array2D<int>& array, const scene_rdl2::math::BBox2f& bbox, const bool done) {
387 fillRegionByVal(array, bbox, calcPixConditionVal(done));
388 };
389 std::function<void(scene_rdl2::util::Array2D<int>&, const Node&)> fillNode =
390 [&](scene_rdl2::util::Array2D<int>& array, const Node& node) {
391 if (node.mDone) {
392 fillRegionByCondition(array, node.mBounds, true);
393 } else {
394 if (node.mChildren) {
395 fillNode(array, node.getLeftChild());
396 fillNode(array, node.getRightChild());
397 } else {
398 fillRegionByCondition(array, node.mBounds, false);
399 }
400 }
401 };
402
403 scene_rdl2::util::Array2D<int> pixCondition((int)extents(mRoot.mBounds, 0) + 1, (int)extents(mRoot.mBounds, 1) + 1);
404 fillRegionByVal(pixCondition, mRoot.mBounds, 0x0);
405 fillNode(pixCondition, mRoot);
406 return pixCondition;
407}
