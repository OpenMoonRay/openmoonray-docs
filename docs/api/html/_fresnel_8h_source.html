111class SchlickFresnel : public Fresnel
112{
113public:
114 SchlickFresnel(scene_rdl2::math::Color spec, float factor, float neta = 1.0f) :
115 Fresnel(PROPERTY_COLOR | PROPERTY_FACTOR),
116 mSpec(spec),
117 mFactor(factor),
118 mNeta(neta) { }
119
120 SchlickFresnel(scene_rdl2::alloc::Arena *arena, const SchlickFresnelv *fresnelv, int lane)
121 : Fresnel(arena, (const Fresnelv *)fresnelv, lane)
122 {
123 mSpec.r = fresnelv->mSpec.r[lane];
124 mSpec.g = fresnelv->mSpec.g[lane];
125 mSpec.b = fresnelv->mSpec.b[lane];
126 mFactor = fresnelv->mFactor[lane];
127 mNeta = fresnelv->mNeta[lane];
128 }
129
130
131 finline scene_rdl2::math::Color eval(float hDotWi) const override
132 {
133 if (mFactor == 0.0f) {
134 return mSpec * mWeight;
135 }
136
137 // Deal with total internal reflection if neta is > 1.0f
138 float tmp;
139 if (mNeta > 1.0f) {
140 const float rcpNeta = scene_rdl2::math::rcp(mNeta);
141 const float cosSqrCriticalAngle = 1.0f - rcpNeta * rcpNeta;
142 const float cosSqrTheta = hDotWi * hDotWi;
143 if (cosSqrTheta < cosSqrCriticalAngle) {
144 // Total internal reflection!
145 return scene_rdl2::math::sWhite;
146 }
147 // Remap the curve to peak at the critical angle
148 tmp = (1.0f - hDotWi) * scene_rdl2::math::rcp(1.0f - scene_rdl2::math::sqrt(cosSqrCriticalAngle));
149 } else {
150 tmp = 1.0f - hDotWi;
151 }
152
153 float power = tmp * tmp;
154 power = tmp * power * power;
155
156 return (mSpec + mFactor * (scene_rdl2::math::sWhite - mSpec) * power) * mWeight;
157 }
158
159 finline void getProperty(Property prop, float *dest) const override
160 {
161 switch (prop)
162 {
163 case PROPERTY_COLOR:
164 *dest = mSpec.r * mWeight;
165 *(dest + 1) = mSpec.g * mWeight;
166 *(dest + 2) = mSpec.b * mWeight;
167 break;
168 case PROPERTY_FACTOR:
169 *dest = mFactor * mWeight;
170 break;
171 default:
172 MNRY_ASSERT(0 && "unexpected property");
173 }
174 }
175
176 void show(std::ostream& os, const std::string& indent) const override
177 {
178 os << indent << "{SchlickFresnel}" << " " << this << " weight: " << mWeight << "\n";
179 os << indent << " " << "reflectivity at 0: "
180 << mSpec.r << " "
181 << mSpec.g << " "
182 << mSpec.b << "\n";
183 }
184
185private:
186 friend class OneMinusRoughSchlickFresnel;
187
188 SCHLICK_FRESNEL_MEMBERS;
189};

242class OneMinusRoughFresnel : public Fresnel
243{
244public:
245 OneMinusRoughFresnel(const Fresnel *fresnel,
246 float specRoughness) :
247 Fresnel(fresnel->getPropertyFlags() | PROPERTY_ROUGHNESS),
248 mTopFresnel(fresnel),
249 // Base Fresnel Behavior at Normal Incidence
250 mOneMinusFresnelAt0(scene_rdl2::math::sWhite - fresnel->eval(1.0f)),
251 // Apply roughness squaring to linearize roughness response
252 // like all roughness-based lobes do.
253 mSpecRoughness(specRoughness * specRoughness)
254 {
255 // Interpolating non-linearly based on roughness, using:
256 // t = 1 - (1 - roughness)^3
257 const float tmp = 1.0f - mSpecRoughness;
258 mInterpolator = scene_rdl2::math::clamp(1.0f - (tmp * tmp * tmp), 0.0f, 1.0f);
259 }
260
261 OneMinusRoughFresnel(scene_rdl2::alloc::Arena *arena, const OneMinusRoughFresnelv *fresnelv, int lane)
262 : Fresnel(arena, (const Fresnelv *)fresnelv, lane),
263 mTopFresnel(nullptr)
264 {
265 if (fresnelv->mTopFresnel) {
266 mTopFresnel = createFresnel(arena, (const Fresnelv *)fresnelv->mTopFresnel, lane);
267 }
268
269 mOneMinusFresnelAt0.r = fresnelv->mOneMinusFresnelAt0.r[lane];
270 mOneMinusFresnelAt0.g = fresnelv->mOneMinusFresnelAt0.g[lane];
271 mOneMinusFresnelAt0.b = fresnelv->mOneMinusFresnelAt0.b[lane];
272 mSpecRoughness = fresnelv->mSpecRoughness[lane];
273 mInterpolator = fresnelv->mInterpolator[lane];
274 }
275
277 finline scene_rdl2::math::Color eval(float nDotWo) const override
278 {
279 return scene_rdl2::math::lerp(scene_rdl2::math::sWhite - mTopFresnel->eval(nDotWo),
280 mOneMinusFresnelAt0,
281 mInterpolator);
282 }
283
284 finline void getProperty(Property prop, float *dest) const override
285 {
286 switch (prop)
287 {
288 case PROPERTY_ROUGHNESS: {
289 // this is a VEC2F property for compatibility with bsdf lobes
290 const float inputRoughness = scene_rdl2::math::sqrt(mSpecRoughness);
291 *dest = inputRoughness;
292 *(dest + 1) = inputRoughness;
293 break;
294 }
295 default:
296 mTopFresnel->getProperty(prop, dest);
297 // if the property is color, we'll 1 - dest, other
298 // properties will remain as is.
299 if (prop == PROPERTY_COLOR) {
300 *dest = 1.f - *dest;
301 *(dest + 1) = 1.f - *(dest + 1);
302 *(dest + 2) = 1.f - *(dest + 2);
303 }
304 break;
305 }
306 }
307
308 void show(std::ostream& os, const std::string& indent) const override
309 {
310 os << indent << "{OneMinusRoughFresnel}" << " " << this << " weight: " << mWeight << "\n";
311 os << indent << " " << "1 - Fresnel at 0: "
312 << mOneMinusFresnelAt0.r << " "
313 << mOneMinusFresnelAt0.g << " "
314 << mOneMinusFresnelAt0.b << "\n";
315 os << indent << " " << "roughness^2: " << mSpecRoughness << "\n";
316 os << indent << " " << "interpolator: " << mInterpolator << "\n";
317 mTopFresnel->show(os, indent + " ");
318 }

458class DielectricFresnel : public Fresnel
459{
460public:
461 DielectricFresnel(float etaI, float etaT) :
462 Fresnel(PROPERTY_COLOR), mEtaI(etaI), mEtaT(etaT) { }
463
464 DielectricFresnel(scene_rdl2::alloc::Arena *arena, const DielectricFresnelv *fresnelv, int lane)
465 : Fresnel(arena, (const Fresnelv *)fresnelv, lane)
466 {
467 mEtaI = fresnelv->mEtaI[lane];
468 mEtaT = fresnelv->mEtaT[lane];
469 }
470
471 finline scene_rdl2::math::Color eval(float hDotWi) const override
472 {
473 return eval(hDotWi, mEtaI, mEtaT, mWeight);
474 }
475
476 static scene_rdl2::math::Color eval(float hDotWi,
477 float etaI, float etaT,
478 float weight)
479 {
480 // Make sure we don't nan out down below
481 hDotWi = scene_rdl2::math::max(hDotWi, -1.0f);
482 hDotWi = scene_rdl2::math::min(hDotWi, 1.0f);
483
484 // Compute Snell law
485 const float eta = etaI * scene_rdl2::math::rcp(etaT);
486 const float sinThetaTSqr = eta * eta * (1.0f - hDotWi * hDotWi);
487 if (sinThetaTSqr >= 1.0f) {
488 // Total internal reflection
489 return scene_rdl2::math::sWhite * weight;
490
491 } else {
492
493 const float cosThetaT = scene_rdl2::math::sqrt(1.0f - sinThetaTSqr);
494 const float cosThetaI = scene_rdl2::math::abs(hDotWi);
495
496 const float parallel = evalParallel(etaI, etaT, cosThetaI, cosThetaT);
497 const float perp = evalPerp(etaI, etaT, cosThetaI, cosThetaT);
498
499 const float fr = 0.5f * (parallel + perp);
500 return scene_rdl2::math::Color(fr, fr, fr) * weight;
501 }
502 }
503
504 finline scene_rdl2::math::Color computePbrValidity() const override
505 {
506 scene_rdl2::math::Color res = scene_rdl2::math::sBlack;
507
508 if (mEtaT < sPbrValidityDielectricValidLowBegin || mEtaT > sPbrValidityDielectricValidHighEnd) {
509 res = sPbrValidityInvalidColor;
510 } else if (mEtaT > sPbrValidityDielectricValidLowEnd && mEtaT < sPbrValidityDielectricValidHighBegin) {
511 res = sPbrValidityValidColor;
512 } else if (mEtaT > sPbrValidityDielectricValidHighBegin) {
513 const float gradient = (mEtaT - sPbrValidityDielectricValidHighBegin) /
514 (sPbrValidityDielectricValidHighEnd - sPbrValidityDielectricValidHighBegin);
515 res = scene_rdl2::math::lerp(sPbrValidityValidColor, sPbrValidityInvalidColor, gradient);
516 } else { // mEtaT < sPbrValidityDielectricValidLowEnd
517 const float gradient = (mEtaT - sPbrValidityDielectricValidLowBegin) /
518 (sPbrValidityDielectricValidLowEnd - sPbrValidityDielectricValidLowBegin);
519 res = scene_rdl2::math::lerp(sPbrValidityInvalidColor, sPbrValidityValidColor, gradient);
520 }
521 return res;
522 }
523
524 finline void getProperty(Property prop, float *dest) const override
525 {
526 switch (prop)
527 {
528 case PROPERTY_COLOR:
529 *dest = *(dest + 1) = *(dest + 2) = mWeight;
530 break;
531 default:
532 MNRY_ASSERT(0 && "unknown fresnel property");
533 }
534 }
535 void show(std::ostream& os,
536 const std::string& indent) const override
537 {
538 os << indent << "{DielectricFresnel}" << " " << this << " weight: " << mWeight << "\n";
539 os << indent << " " << "etaI: " << mEtaI << "\n";
540 os << indent << " " << "etaT: " << mEtaT << "\n";
541 }
542private:
543 DIELECTRIC_FRESNEL_MEMBERS;
544
545protected:
546 // S Polarized
547 static float evalPerp(float etaI,
548 float etaT,
549 float cosThetaI,
550 float cosThetaT)
551 {
552 const float etaIcosThetaI = etaI * cosThetaI;
553 const float etaTcosThetaT = etaT * cosThetaT;
554 const float perp = ((etaIcosThetaI) - (etaTcosThetaT)) *
555 scene_rdl2::math::rcp((etaIcosThetaI) + (etaTcosThetaT));
556 return perp*perp;
557 }
558
559 // P Polarized
560 static float evalParallel(float etaI,
561 float etaT,
562 float cosThetaI,
563 float cosThetaT)
564 {
565 const float etaTcosThetaI = etaT * cosThetaI;
566 const float etaIcosThetaT = etaI * cosThetaT;
567 const float parallel = ((etaTcosThetaI) - (etaIcosThetaT)) *
568 scene_rdl2::math::rcp((etaTcosThetaI) + (etaIcosThetaT));
569 return parallel*parallel;
570 }

582class LayeredDielectricFresnel : public DielectricFresnel
583{
584public:
585 LayeredDielectricFresnel(float etaI, float etaT, float layerThickness) :
586 DielectricFresnel(etaI, etaT)
587 {
588 // Layers vary in between [0.5, 1.5]
589 // See Table 2
590 mNumLayers = scene_rdl2::math::lerp(0.5f, 1.5f,
591 scene_rdl2::math::saturate(layerThickness));
592 }
593
594 finline Fresnel *copy() const { return new LayeredDielectricFresnel(*this); }
595
596 finline scene_rdl2::math::Color eval(float cosThetaI) const override
597 {
598 // Use the same refractive index for the
599 // parallel and perpendicular components of
600 // the dielectric fresnel equation by default
601 return mWeight*evalFresnel(mEtaI, mEtaT,
602 mEtaT, mEtaT,
603 cosThetaI,
604 mNumLayers);
605 }
606
607 // Layered Fresnel Evaluation with distinct IORs for
608 // perpendicular and parallel components of the dielectric
609 // fresnel equation.
610 // See Eqn (6) & (7) in
611 // "Physically-Accurate Fur Reflectance: Modeling, Measurement and Rendering"
612 static scene_rdl2::math::Color evalFresnel(float etaI,
613 float etaT,
614 float etaPerp,
615 float etaParallel,
616 float cosThetaI,
617 float numLayers)
618 {
619 // Make sure we don't nan out down below
620 cosThetaI = scene_rdl2::math::max(cosThetaI, -1.0f);
621 cosThetaI = scene_rdl2::math::min(cosThetaI, 1.0f);
622
623 // Compute Snell law
624 const float eta = 1.0f * scene_rdl2::math::rcp(etaT);
625 const float sinThetaTSqr = eta * eta * (1.0f - cosThetaI*cosThetaI);
626
627 if (sinThetaTSqr >= 1.0f) {
628 // Total internal reflection
629 return scene_rdl2::math::sWhite;
630 }
631
632 const float cosThetaT = scene_rdl2::math::sqrt(1.0f - sinThetaTSqr);
633 cosThetaI = scene_rdl2::math::abs(cosThetaI);
634
635 // Eqn (6)
636 const float F = evalSingleLayer(etaI,
637 etaPerp,
638 etaParallel,
639 cosThetaI,
640 cosThetaT);
641
642 // Eqn (7)
643 const float n = numLayers * F;
644 const float d = 1.0f + (numLayers - 1.0f) * F;
645
646 const float layeredRefl = scene_rdl2::math::clamp(n/d, 0.0f, 1.0f);
647 return scene_rdl2::math::Color(layeredRefl);
648 }

650 void show(std::ostream& os,
651 const std::string& indent) const override
652 {
653 os << indent << "{LayeredDielectricFresnel}" << " "
654 << this << " weight: " << mWeight << "\n";
655 os << indent << " " << "etaI: " << mEtaI << "\n";
656 os << indent << " " << "etaT: " << mEtaT << "\n";
657 os << indent << " " << "num layers: " << mNumLayers << "\n";
658 }

660private:
661 DIELECTRIC_FRESNEL_MEMBERS;
662
663 float mNumLayers;
664
665 // Reflectance from a single layer
666 // Eqn (6)
667 static float evalSingleLayer(float etaI,
668 float etaPerp,
669 float etaParallel,
670 float cosThetaI,
671 float cosThetaT)
672 {
673 float Fs = evalPerp(etaI, etaPerp, cosThetaI, cosThetaT);
674 float Fp = evalParallel(etaI, etaParallel, cosThetaI, cosThetaT);
675
676 float t1 = Fs + (1.0f - Fs)*(1.0f - Fs) * Fs / (1.0f - Fs*Fs);
677 float t2 = Fp + (1.0f - Fp)*(1.0f - Fp) * Fp / (1.0f - Fp*Fp);
678
679 return (0.5f*t1 + 0.5f*t2);
680 }

687class ConductorFresnel : public Fresnel
688{
689public:
690
691 // Initialize Conductor Fresnel object using real and imaginary components
692 // of a complex IOR
693 ConductorFresnel(const scene_rdl2::math::Color& eta, const scene_rdl2::math::Color& absorption) :
694 Fresnel(PROPERTY_COLOR), mEta(eta),
695 mAbsorption(absorption) {}
696
697 ConductorFresnel(scene_rdl2::alloc::Arena *arena, const ConductorFresnelv *fresnelv, int lane)
698 : Fresnel(arena, (const Fresnelv *)fresnelv, lane)
699 {
700 mEta.r = fresnelv->mEta.r[lane];
701 mEta.g = fresnelv->mEta.g[lane];
702 mEta.b = fresnelv->mEta.b[lane];
703 mAbsorption.r = fresnelv->mAbsorption.r[lane];
704 mAbsorption.g = fresnelv->mAbsorption.g[lane];
705 mAbsorption.b = fresnelv->mAbsorption.b[lane];
706 }
707
708 finline scene_rdl2::math::Color eval(float hDotWi) const override
709 {
710 const float hDotWiSqrd = hDotWi * hDotWi;
711 const scene_rdl2::math::Color factor = mEta * mEta + mAbsorption * mAbsorption;
712 const scene_rdl2::math::Color tmp = factor * hDotWiSqrd;
713 const scene_rdl2::math::Color etaHDotWi = mEta * hDotWi;
714
715 const scene_rdl2::math::Color parallel = (tmp - (2.0f * etaHDotWi) + scene_rdl2::math::Color(1.0f)) /
716 (tmp + (2.0f * etaHDotWi) + scene_rdl2::math::Color(1.0f));
717
718 const scene_rdl2::math::Color perp = (factor - (2.0f * etaHDotWi) + scene_rdl2::math::Color(hDotWiSqrd)) /
719 (factor + (2.0f * etaHDotWi) + scene_rdl2::math::Color(hDotWiSqrd));
720
721 return 0.5f * (parallel + perp) * mWeight;
722 }
723
724 finline scene_rdl2::math::Color computePbrValidity() const override
725 {
726 scene_rdl2::math::Color res = scene_rdl2::math::sBlack;
727 const scene_rdl2::math::Color reflectivity = eval(1.f);
728 const float value = scene_rdl2::math::max(reflectivity.r, reflectivity.g, reflectivity.b);
729
730 if (value < sPbrValidityConductorInvalid || value > 1.0f) {
731 res = sPbrValidityInvalidColor;
732 } else if (value > sPbrValidityConductorValid) {
733 res = sPbrValidityValidColor;
734 } else {
735 const float gradient = (value - sPbrValidityConductorInvalid) /
736 (sPbrValidityConductorValid - sPbrValidityConductorInvalid);
737 res = scene_rdl2::math::lerp(sPbrValidityInvalidColor, sPbrValidityValidColor, gradient);
738 }
739 return res;
740 }
741
742 finline void getProperty(Property prop, float *dest) const override
743 {
744 switch (prop)
745 {
746 case PROPERTY_COLOR:
747 *dest = (1.f - mAbsorption.r) * mWeight;
748 *(dest + 1) = (1.f - mAbsorption.g) * mWeight;
749 *(dest + 2) = (1.f - mAbsorption.b) * mWeight;
750 break;
751 default:
752 MNRY_ASSERT(0 && "unknown fresnel property");
753 }
754 }
755
756 void show(std::ostream& os, const std::string& indent) const override
757 {
758 os << indent << "{ConductorFresnel}" << " " << this << " weight: " << mWeight << "\n";
759 os << indent << " " << "Eta: " << mEta.r << " " << mEta.g << " " << mEta.b << "\n";
760 os << indent << " " << "Absorption: " << mAbsorption.r << " " << mAbsorption.g << " " << mAbsorption.b << "\n";
761 }
762
763private:
764 CONDUCTOR_FRESNEL_MEMBERS;
765};

895// Utility function for diffuse reflectance fresnel used in the dipole model
896finline float
897diffuseFresnelReflectance(float eta)
898{
899 if (eta >= 1.0f) {
900 return -1.4399f / (eta * eta) + 0.7099f / eta + 0.6681f + 0.0636f * eta;
901 } else {
902 float etaSqr = eta * eta;
903 return -0.4399f + 0.7099f / eta - 0.3319f / etaSqr + 0.0636f /
904 (etaSqr * eta);
905 }
906}

908// Average Dielectric Fresnel Reflectance
909// Ref:"Revisiting Physically Based Shading", Kulla'17
910finline float
911averageFresnelReflectance(float neta)
912{
913 MNRY_ASSERT(neta >= 1.0f);
914 float favg =
915 (neta - 1.0f) / (4.08567f + 1.00071f*neta);
916 return favg;
917}
918
919// Average Dielectric Fresnel Reflectance
920// Ref:"Revisiting Physically Based Shading", Kulla'17
921finline float
922averageInvFresnelReflectance(float neta)
923{
924 MNRY_ASSERT(neta < 1.0f);
925 float favgInv = 0.997118f +
926 0.1014f * neta -
927 0.965241f * neta * neta -
928 0.130607f * neta * neta * neta;
929 return favgInv;
930}

932// Average Conductor Fresnel Reflectance
933// Ref:"Revisiting Physically Based Shading", Kulla'17
934finline scene_rdl2::math::Color
935averageFresnelReflectance(const scene_rdl2::math::Color& reflectivity,
936 const scene_rdl2::math::Color& edgeTint)
937{
938 const scene_rdl2::math::Color r2 = reflectivity * reflectivity;
939 const scene_rdl2::math::Color e2 = edgeTint*edgeTint;
940 scene_rdl2::math::Color favg = scene_rdl2::math::Color(0.087237f) + 0.0230685f * edgeTint -
941 0.0864902f * e2 +
942 0.0774594f * e2 * edgeTint +
943 0.782654f * reflectivity -
944 0.136432f * r2 +
945 0.278708f * r2 * reflectivity +
946 0.19744f * edgeTint * reflectivity +
947 0.0360605f * e2 * reflectivity -
948 0.2586f * edgeTint * r2;
949 return favg;
950}
951
952// Average Dielectric Fresnel Reflectance
953// Ref:"Revisiting Physically Based Shading", Kulla'17
954finline void
955averageFresnelReflectance(float neta,
956 float& favg,
957 float& favgInv)
958{
959 if (neta > 1.0f) {
960 favg = averageFresnelReflectance(neta);
961 favgInv = averageInvFresnelReflectance(1.0f/neta);
962 } else if (neta < 1.0f) {
963 favg = averageFresnelReflectance(1.0f/neta);
964 favgInv = averageInvFresnelReflectance(neta);
965 } else {
966 favg = favgInv = 0.0f;
967 }
968}
