49TestHair::TestHair() :
50 mNg(0.0f, 0.0f, 1.0f),
51 mWi(normalize(Vec3f(-0.25f, -0.15f, 0.59f))),
52 mWo(normalize(Vec3f(0.5f, 0.75f, 0.25f)))
53{
54 // setup slices

56 mNg, mWo, true, true, ispc::SHADOW_TERMINATOR_FIX_OFF);

59 ispc::createBsdfSlice(asIspc(mNg), asIspc(mWo), true, true),
60 [](void * p) { ispc::deleteBsdfSlice(p); } );
61
62 // setup HairState
63 static const Vec3f hairDir = normalize(Vec3f(1.0f, 1.0f, 1.0f));
64 static const Color hairColor(0.65f, 0.21f, 0.25f);
65 static const float azimRoughness = 0.22f;

73

75 mWi,
76 hairDir,
77 H,
78 eta,
79 hairSigmaA,
80 hairRotation,
81 hairNormal);
82

84 ispc::createHairState(asIspc(mWo),
85 asIspc(mWi),
86 asIspc(hairDir),

88 eta,
89 asIspc(hairSigmaA),
90 hairRotation,
91 asIspc(hairNormal)),
92 [](void * p) { ispc::deleteHairState(p); } );
93
94 // setup Fresnel
95 mFresnelCpp = fauxstd::make_unique<DielectricFresnel>(1.0f, eta);
96
97 mFresnelIspc = deleted_unique_ptr<void>(
98 ispc::createFresnel(1.0f, eta),
99 [](void * p) { ispc::deleteFresnel(p); } );

106void TestHair::testHairUtil()
107{
108 static const float f = 0.25f;
109 static const Color c(0.25f, 0.55f, 0.75f);
110
111 CPPUNIT_ASSERT(isEqual(Pow<2>(f), ispc::testPow2(f), sTolerance));
112 CPPUNIT_ASSERT(isEqual(Pow<3>(f), ispc::testPow3(f), sTolerance));
113 CPPUNIT_ASSERT(isEqual(Pow<4>(f), ispc::testPow4(f), sTolerance));
114 CPPUNIT_ASSERT(isEqual(Pow<20>(f), ispc::testPow20(f), sTolerance));
115 CPPUNIT_ASSERT(isEqual(Pow<22>(f), ispc::testPow22(f), sTolerance));
116
117 CPPUNIT_ASSERT(isEqual(HairUtil::safeSqrt(f), ispc::testSafeSqrt(f), sTolerance));
118 CPPUNIT_ASSERT(isEqual(scene_rdl2::math::sqr(f), ispc::testSqr(f), sTolerance));
119 CPPUNIT_ASSERT(isEqual(scene_rdl2::math::sinh(f), ispc::testSinh(f), sTolerance));
120
121 CPPUNIT_ASSERT(isEqual(HairUtil::azimuthalVar(f), ispc::testAzimuthalVar(f), sTolerance));
122 CPPUNIT_ASSERT(isEqual(HairUtil::longitudinalVar(f), ispc::testLongitudinalVar(f), sTolerance));
123
124 // test computeAbsorptionCoefficients()
125 {
126 ispc::Color ispcResult;
127 ispc::testComputeAbsorptionCoefficients(asIspc(c), f, ispcResult);
128 CPPUNIT_ASSERT(isEqualColor(HairUtil::computeAbsorptionCoefficients(c, f), ispcResult, sTolerance));
129 }
130
131 CPPUNIT_ASSERT(isEqual(HairUtil::besselIO(f), ispc::testBesselIO(f), sTolerance));
132 CPPUNIT_ASSERT(isEqual(HairUtil::logBesselIO(f), ispc::testLogBesselIO(f), sTolerance));
133
134 // test deonLongitudinalM() func
135 {
136 static const float variance = 0.0123f;
137 static const float thetaI = scene_rdl2::math::sPi / 4.0f;
138 static const float thetaO = scene_rdl2::math::sPi / 6.0f;
139 static const float sinThetaI = scene_rdl2::math::sin(thetaI);
140 static const float cosThetaI = scene_rdl2::math::cos(thetaI);
141 static const float sinThetaO = scene_rdl2::math::sin(thetaO);
142 static const float cosThetaO = scene_rdl2::math::cos(thetaO);
143
144 CPPUNIT_ASSERT(isEqual(
145 HairUtil::deonLongitudinalM(variance, sinThetaI, cosThetaI, sinThetaO, cosThetaO),
146 ispc::testDeonLongitudinalM(variance, sinThetaI, cosThetaI, sinThetaO, cosThetaO),
147 sTolerance));
148 }
149
150 // test logistic functions
151 {
152 static const float x = 0.3425f;
153 static const float s = 0.6345f;
154 static const float a = 0.15f;
155 static const float b = 0.85f;
156
157 CPPUNIT_ASSERT(isEqual(HairUtil::logisticFunction(x, s), ispc::testLogisticFunction(x, s), sTolerance));
158 CPPUNIT_ASSERT(isEqual(HairUtil::logisticCDF(x, s), ispc::testLogisticCDF(x, s), sTolerance));
159 CPPUNIT_ASSERT(isEqual(HairUtil::trimmedLogisticFunction(x, s, a, b), ispc::testTrimmedLogisticFunction(x, s, a, b), sTolerance));
160 CPPUNIT_ASSERT(isEqual(HairUtil::sampleTrimmedLogistic(x, s, a, b), ispc::testSampleTrimmedLogistic(x, s, a, b), sTolerance));
161 }
162
163 // test gaussian/cauchy functions
164 {
165 static const float stddev = 0.1425f;
166 static const float x = 0.6345f;
167 CPPUNIT_ASSERT(isEqual(HairUtil::unitGaussianForShade(stddev, x), ispc::testUnitGaussianForShade(stddev, x), sTolerance));
168 CPPUNIT_ASSERT(isEqual(HairUtil::unitCauchyForShade(stddev, x), ispc::testUnitCauchyForShade(stddev, x), sTolerance));
169 CPPUNIT_ASSERT(isEqual(HairUtil::unitCauchyForSample(stddev, x), ispc::testUnitCauchyForSample(stddev, x), sTolerance));
170 }
171
172 // test compact1By1()
173 {
174 static const uint32_t x = 54321;
175 const uint32_t a = HairUtil::compact1By1(x);
176 const uint32_t b = ispc::testCompact1By1(x);
177 CPPUNIT_ASSERT( a == b );
178 }
179
180 // test demuxFloat()
181 {
182 const Vec2f a = HairUtil::demuxFloat(f);
183 ispc::Vec2f b;
184 ispc::testDemuxFloat(f, b);
185 CPPUNIT_ASSERT(isEqual(a.x, asCpp(b).x, sTolerance));
186 CPPUNIT_ASSERT(isEqual(a.y, asCpp(b).y, sTolerance));
187 }
188}

190void TestHair::testLobe(HairBsdfLobe * lobeCpp, void * lobeIspc)
191{
192 // test members
193 {
194 CPPUNIT_ASSERT(isEqual(lobeCpp->mSinAlpha,
195 ispc::getSinAlpha(lobeIspc),
196 sTolerance));
197 CPPUNIT_ASSERT(isEqual(lobeCpp->mCosAlpha,
198 ispc::getCosAlpha(lobeIspc),
199 sTolerance));
200 CPPUNIT_ASSERT(isEqual(lobeCpp->mLongitudinalRoughness,
201 ispc::getLongitudinalRoughness(lobeIspc),
202 sTolerance));
203 CPPUNIT_ASSERT(isEqual(lobeCpp->mLongitudinalVariance,
204 ispc::getLongitudinalVariance(lobeIspc),
205 sTolerance));
206 }
207 // test evalMTerm()
208 {
209 const float a = lobeCpp->evalMTerm(*mHairStateCpp);
210 const float b = ispc::evalMTerm(lobeIspc,
211 mHairStateIspc.get());
212 CPPUNIT_ASSERT(isEqual(a, b, sTolerance));
213 }
214 // test evalNTermWithAbsorption()
215 {
216 const Color result_a = lobeCpp->evalNTermWithAbsorption(*mHairStateCpp);
217 ispc::Color result_b;
218 ispc::evalNTermWithAbsorption(lobeIspc,
219 mHairStateIspc.get(),
220 result_b);
221 CPPUNIT_ASSERT(isEqualColor(result_a, result_b, sTolerance));
222 }
223 // test evalPhiPdf()
224 {
225 const float a = lobeCpp->evalPhiPdf(*mHairStateCpp);
226 const float b = ispc::evalPhiPdf(lobeIspc,
227 mHairStateIspc.get());
228 CPPUNIT_ASSERT(isEqual(a, b, sTolerance));
229 }
230 // test evalThetaPdf()
231 {
232 // check some internals first
233 const float thetaI_a = mHairStateCpp->thetaI();
234 const float thetaI_b = ispc::getThetaI(mHairStateIspc.get());
235 CPPUNIT_ASSERT(isEqual(thetaI_a, thetaI_b, sTolerance));
236 const float thetaO_a = mHairStateCpp->thetaO();
237 const float thetaO_b = ispc::getThetaO(mHairStateIspc.get());
238 CPPUNIT_ASSERT(isEqual(thetaO_a, thetaO_b, sTolerance));
239
240 const float a = lobeCpp->evalThetaPdf(*mHairStateCpp);
241 const float b = ispc::evalThetaPdf(lobeIspc,
242 mHairStateIspc.get());
243 CPPUNIT_ASSERT(isEqual(a, b, sTolerance));
244 }
245 // test sampleTheta()
246 {
247 // choose a random
248 const float r1 = 0.123f;
249
250 const float thetaO = mHairStateCpp->thetaO();
251 float thetaI_a, thetaI_b;
252 float pdf_a, pdf_b;
253 pdf_a = lobeCpp->sampleTheta(r1, thetaO, thetaI_a);
254 pdf_b = ispc::sampleTheta(lobeIspc, r1, thetaO, thetaI_b);
255 CPPUNIT_ASSERT(isEqual(pdf_a, pdf_b, sTolerance));
256 CPPUNIT_ASSERT(isEqual(thetaI_a, thetaI_b, sTolerance));
257 }
258 // test samplePhi()
259 {
260 // choose a random
261 const float r2 = 0.567;
262 const float phiO = mHairStateCpp->phiO();
263
264 float phiI_a, phiI_b;
265 float pdf_a = lobeCpp->samplePhi(r2,
266 phiO,
267 phiI_a);
268 float pdf_b = ispc::samplePhi(lobeIspc,
269 r2,
270 phiO,
271 phiI_b);
272 CPPUNIT_ASSERT(isEqual(pdf_a, pdf_b, sTolerance));
273 CPPUNIT_ASSERT(isEqual(phiI_a, phiI_b, sTolerance));
274 }
275 // test evalPdf()
276 {
277 const float a = lobeCpp->evalPdf(*mHairStateCpp);
278 const float b = ispc::evalPdf(lobeIspc,
279 mHairStateIspc.get());
280 CPPUNIT_ASSERT(isEqual(a, b, sTolerance));
281 }
282 // test eval()
283 {
284 float pdf_a, pdf_b;
285 const Color result_a = lobeCpp->eval(*mBsdfSliceCpp, mWi, &pdf_a);
286 ispc::Color result_b;
287 ispc::evalLobe(lobeIspc, mBsdfSliceIspc.get(), asIspc(mWi), result_b, pdf_b);
288 CPPUNIT_ASSERT(isEqual(pdf_a, pdf_b, sTolerance));
289 CPPUNIT_ASSERT(isEqualColor(result_a, result_b, sTolerance));
290 }
291 // test sample()
292 {
293 // choose some randoms
294 const float r1 = 0.123f;
295 const float r2 = 0.567f;
296
297 float pdf_a, pdf_b;
298 Vec3f wi_a;
299 ispc::Vec3f wi_b;
300 const Color result_a = lobeCpp->sample(*mBsdfSliceCpp, r1, r2, wi_a, pdf_a);
301 ispc::Color result_b;
302 ispc::sampleLobe(lobeIspc, mBsdfSliceIspc.get(), r1, r2, wi_b, pdf_b, result_b);
303
304 CPPUNIT_ASSERT(isEqualVec3f(wi_a, wi_b, sTolerance));
305 CPPUNIT_ASSERT(isEqual(pdf_a, pdf_b, sTolerance));
306 CPPUNIT_ASSERT(isEqualColor(result_a, result_b, sTolerance));
307 }
308}

479{

505 CPPUNIT_ASSERT(isEqualVec3f(a, b, sTolerance));
