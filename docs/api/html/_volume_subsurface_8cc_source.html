27VolumeSubsurface::VolumeSubsurface(const scene_rdl2::math::Color& trlColor,
28 const scene_rdl2::math::Color& trlRadius,
29 const scene_rdl2::rdl2::Material* material,
30 scene_rdl2::rdl2::EvalNormalFunc evalNormalFn,
31 const scene_rdl2::math::Vec3f& N,
32 bool resolveSelfIntersections):
33 mScatteringColor( scene_rdl2::math::max(0.001f, trlColor[0]),
34 scene_rdl2::math::max(0.001f, trlColor[1]),
35 scene_rdl2::math::max(0.001f, trlColor[2])),
36 mScatteringRadius(scene_rdl2::math::max(0.001f, trlRadius[0] * 0.3333f),
37 scene_rdl2::math::max(0.001f, trlRadius[1] * 0.3333f),
38 scene_rdl2::math::max(0.001f, trlRadius[2] * 0.3333f)),
39 mScale(1.0f),
40 mFresnel(nullptr),
41 mTraceSet(nullptr),
42 mMaterial(material),
43 mEvalNormalFunc(evalNormalFn),
44 mLabel(0),
45 mPropertyFlags(PROPERTY_COLOR | PROPERTY_RADIUS | PROPERTY_PBR_VALIDITY),
46 mN(N),
47 mResolveSelfIntersections(resolveSelfIntersections)
48{
49 // Clamping mScatteringColor to a min, low value similar to what we do in diffusion based BSSRDFs.
50 // Scale the scattering radius by 3
51 // For Random-Walk SSS, we use the Disney equations from
52 // "Practical and Controllable Subsurface Scattering for Production Path Tracing"
53 // to convert the scattering-radius to extinction coefficient (sigmaT).
54 // This differs from our diffusion-based bssrdf in that the scattering-radius
55 // gets used to calculate volume mean free path and not diffuse mean free path.
56 // This makes the same scattering-radius value behave wildly differently between the
57 // different SSS models. In addition, the sigmaT equations are based on
58 // Disney simulations to calculate a scaling factor based on the volume mean free path
59 // set to 1 ("Approximate Reflectance Profiles for Subsurface Scattering").
60 // Now, modeling tells us that we still model in shrekles and it appears Disney models in cm.
61 // 1 shrekle = 3cm
62 // Modeling confirms there are no plans to change this until PipeX.
63 // Via testing, it appears, dividing the scattering-radius by 3 to bring it in cm
64 // gives:
65 // (a) comparable behavior with the diffusion-based SSS and
66 // (b) higher range of values to be used with SSS
67 // (c) an artist-workflow where we set the scattering-radius based on the scale observed via usdview/mm_view etc.
68 // Similar scaling seems to exist in PBRTv3:
69 // https://github.com/mmp/pbrt-v3/blob/9f717d847a807793fa966cf0eaa366852efef167/src/materials/disney.cpp
70 // Line 371
71 // Here the input scattering-radius get scaled by 0.2 with a weird note alluding to
72 // private conversations with Brent Burley and Matt Chiang. Equally Suspect, but only to have some
73 // precedence for scaling the scattering radius.
74
75 // albedo inversion that transform user specified multiple scattering color
76 // and scattering radius to single scattering volume coefficient
77 // sigmaT and albedo
78 // the mapping is a third order exponential-of polynomial curve fitting
79#ifdef USE_IOR14_ALBEDO_INVERSION
80 // with the assumption that transmission IOR is 1.4 from
81 // "The Design and Evolution of Disney's Hyperion Renderer" - Burley et al
82 // 4.4.2 - Path-Traced Skin
83 const scene_rdl2::math::Color& A = trlColor;
84 const scene_rdl2::math::Color& d = trlRadius;
85 mAlbedo = scene_rdl2::math::Color(1.0f) - exp(-11.43f * A + 15.38f * A * A - 13.91f * A * A * A);
86 scene_rdl2::math::Color s = scene_rdl2::math::Color(
87 4.012f) - 15.21f * A + 32.34f * A * A - 34.68f * A * A * A + 13.91f * A * A * A * A;
88 mSigmaT = scene_rdl2::math::Color(1.0f / (d[0] * s[0]), 1.0f / (d[1] * s[1]), 1.0f / (d[2] * s[2]));
89#else
90 // this is the mapping assuming IOR matching in boundary from original talk
91 // (a) "Practical and Controllable Subsurface Scattering for Production Path Tracing"
92 // Matt Jen-Yuan Chiang et al
93 const scene_rdl2::math::Color A = mScatteringColor;
94 const scene_rdl2::math::Color d = mScatteringRadius;
95 // Disney Equation (1) in (a):
96 // mAlbedo = Color(1.0f) - exp(-5.09406f * A + 2.61188f * A * A - 4.31805f * A * A * A);
97 // Our custom equation - see MOONSHINE-1027
98 mAlbedo = scene_rdl2::math::Color(1.0f) - exp(-3.25172*A*A*A + 1.2166*A*A - 4.75914*A);
99
100 // Equation (2) in (a)
101 scene_rdl2::math::Color s = scene_rdl2::math::Color(1.9f) - A + 3.5f * (A - scene_rdl2::math::Color(0.8f)) * (A - scene_rdl2::math::Color(0.8f));
102 // Equation (3) in (a)
103 mSigmaT = scene_rdl2::math::Color(1.0f / (d[0] * s[0]),
104 1.0f / (d[1] * s[1]),
105 1.0f / (d[2] * s[2]));
106
107 // Dwivedi Sampling Parameters
108 // References:
109 // (1) https://cgg.mff.cuni.cz/~jaroslav/papers/2014-zerovar/2014-zerovar-abstract.pdf
110 // (2) https://jo.dreggn.org/home/2016_dwivedi.pdf
111 // (3) https://jo.dreggn.org/home/2016_dwivedi_additional.pdf
112 // Look at the supplemental notes for the formula to
113 // derive V0
114 const scene_rdl2::math::Color omA = scene_rdl2::math::sWhite - mAlbedo;
115 const scene_rdl2::math::Color kappa0 = scene_rdl2::math::sqrt(3.0f*omA) *
116 (scene_rdl2::math::sWhite -
117 omA*(2.0f/5.0f + omA*(12.0f/175.0f + omA*(2.f/125.f + omA*166.f/67375.f))));
118 mDwivediV0 = scene_rdl2::math::rcp(kappa0);
119 mDwivediNormPDF = scene_rdl2::math::log((mDwivediV0 + scene_rdl2::math::sWhite)/
120 (mDwivediV0 - scene_rdl2::math::sWhite));
121 mDwivediNormPDF = scene_rdl2::math::rcp(mDwivediNormPDF);
122
123 // Zero Scatter - when a path exits the surface without scattering inside
124 // Pure Beer's Law Absorption Case:
125 mZeroScatterSigmaT[0] = scene_rdl2::math::isZero(mScatteringRadius[0]) ? sLargeZeroScatterSigmaT :
126 -scene_rdl2::math::log(mScatteringColor[0] - scene_rdl2::math::sEpsilon) / mScatteringRadius[0];
127 mZeroScatterSigmaT[1] = scene_rdl2::math::isZero(mScatteringRadius[1]) ? sLargeZeroScatterSigmaT :
128 -scene_rdl2::math::log(mScatteringColor[1] - scene_rdl2::math::sEpsilon) / mScatteringRadius[1];
129 mZeroScatterSigmaT[2] = scene_rdl2::math::isZero(mScatteringRadius[2]) ? sLargeZeroScatterSigmaT :
130 -scene_rdl2::math::log(mScatteringColor[2] - scene_rdl2::math::sEpsilon) / mScatteringRadius[2];
131
132#endif
133}

143VolumeSubsurface::VolumeSubsurface(scene_rdl2::alloc::Arena *arena,
144 const VolumeSubsurfacev &volumeSubsurfacev,
145 int lane) :
146 mFresnel(nullptr)
147{
148 if (volumeSubsurfacev.mFresnel && ((1 << lane) & volumeSubsurfacev.mFresnelMask)) {
149 setTransmissionFresnel(createFresnel(arena, volumeSubsurfacev.mFresnel, lane));
150 }
151
152 mScatteringColor = colorvToColor(volumeSubsurfacev.mScatteringColor, lane);
153 mScatteringRadius = colorvToColor(volumeSubsurfacev.mScatteringRadius, lane);
154 mSigmaT = colorvToColor(volumeSubsurfacev.mSigmaT, lane);
155 mZeroScatterSigmaT = colorvToColor(volumeSubsurfacev.mZeroScatterSigmaT, lane);
156 mAlbedo = colorvToColor(volumeSubsurfacev.mAlbedo, lane);
157 mDwivediV0 = colorvToColor(volumeSubsurfacev.mDwivediV0, lane);
158 mDwivediNormPDF = colorvToColor(volumeSubsurfacev.mDwivediNormPDF, lane);
159
160 mScale = colorvToColor(volumeSubsurfacev.mScale, lane);
161
162 mLabel = volumeSubsurfacev.mLabel;
163 mPropertyFlags = volumeSubsurfacev.mPropertyFlags;
164 mTraceSet = (const scene_rdl2::rdl2::TraceSet *)volumeSubsurfacev.mTraceSet;
165 mMaterial = (const scene_rdl2::rdl2::Material*)volumeSubsurfacev.mMaterial;
166 mEvalNormalFunc = (scene_rdl2::rdl2::EvalNormalFunc)volumeSubsurfacev.mEvalNormalFn;
167
168 mN = scene_rdl2::math::Vec3f(volumeSubsurfacev.mN.x[lane],
169 volumeSubsurfacev.mN.y[lane],
170 volumeSubsurfacev.mN.z[lane]);
171
172 mResolveSelfIntersections = volumeSubsurfacev.mResolveSelfIntersections;
173}

175bool
176VolumeSubsurface::getProperty(Property property, float *dest) const
177{
178 bool handled = true;
179 switch (property)
180 {
181 case PROPERTY_COLOR:
182 *dest = mScatteringColor[0];
183 *(dest + 1) = mScatteringColor[1];
184 *(dest + 2) = mScatteringColor[2];
185 break;
186 case PROPERTY_RADIUS:
187 *dest = mScatteringRadius[0];
188 *(dest + 1) = mScatteringRadius[1];
189 *(dest + 2) = mScatteringRadius[2];
190 break;
191 case PROPERTY_PBR_VALIDITY:
192 {
193 scene_rdl2::math::Color result = computeAlbedoPbrValidity(mScatteringColor);
194 *dest = result.r;
195 *(dest + 1) = result.g;
196 *(dest + 2) = result.b;
197 }
198 default:
199 handled = false;
200 break;
201 }
202 return handled;
203}
