26void
27RenderContextConsoleDriver::parseConfigure(Parse &parse)
28{
29 parse.description("renderContext console command");
30
31 parse.opt("renderContext", "...command...", "renderContext command",
32 [&](Arg &arg) -> bool {
33 std::lock_guard<std::mutex> lock(mMutexRenderContext);
34 if (!mRenderContext) return arg.msg("renderContext is nullptr\n");
35 return mRenderContext->parse().main(arg.childArg());
36 });
37 parse.opt("invalidateAllTexture", "", "invalidate all texture resources",
38 [&](Arg &arg) -> bool {
39 std::lock_guard<std::mutex> lock(mMutexRenderContext);
40 if (!mRenderContext) return arg.msg("renderContext is nullptr\n");
41
42 bool restart = false;
43 if (mRenderContext->isFrameRendering()) {
44 mRenderContext->stopFrame(); // blocking
45 restart = true;
46 }
47 arg.msg("invalidate start ...");
48 mRenderContext->invalidateAllTextureResources();
49 arg.msg(" done\n");
50 if (restart) {
51 mRenderContext->setForceCallStartFrame();
52 }
53 return true;
54 });
55}
