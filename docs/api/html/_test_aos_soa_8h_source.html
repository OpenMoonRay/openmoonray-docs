17// In bytes. Size must be a multiple of SIMD_MEMORY_ALIGNMENT for the given ISA.
18#define SIZE_OF_AOS_OBJECT 160

20MNRY_STATIC_ASSERT((SIZE_OF_AOS_OBJECT % SIMD_MEMORY_ALIGNMENT) == 0);

25// A single element of AOS data.
26struct AOSData
27{
28 uint32_t mData[SIZE_OF_AOS_OBJECT / sizeof(uint32_t)];
29};
30MNRY_STATIC_ASSERT(sizeof(AOSData) == SIZE_OF_AOS_OBJECT);
31
32// 4 wide SOA block.
33struct ALIGN(SSE_SIMD_MEMORY_ALIGNMENT) SOABlock4
34{
35 enum { LANE_WIDTH = 4 };
36 uint32_t mData[SIZE_OF_AOS_OBJECT / sizeof(uint32_t)][4];
37};
38MNRY_STATIC_ASSERT(sizeof(SOABlock4) == sizeof(AOSData) * 4);
39
40// 8 wide SOA block.
41struct ALIGN(AVX_SIMD_MEMORY_ALIGNMENT) SOABlock8
42{
43 enum { LANE_WIDTH = 8 };
44 uint32_t mData[SIZE_OF_AOS_OBJECT / sizeof(uint32_t)][8];
45};
46MNRY_STATIC_ASSERT(sizeof(SOABlock8) == sizeof(AOSData) * 8);
47
48// 16 wide SOA block.
49struct ALIGN(AVX512_SIMD_MEMORY_ALIGNMENT) SOABlock16
50{
51 enum { LANE_WIDTH = 16 };
52 uint32_t mData[SIZE_OF_AOS_OBJECT / sizeof(uint32_t)][16];
53};
54MNRY_STATIC_ASSERT(sizeof(SOABlock16) == sizeof(AOSData) * 16);
