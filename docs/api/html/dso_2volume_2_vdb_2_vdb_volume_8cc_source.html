107void
108VdbVolume::update()
109{
110 // If map binding does not exist, then property is homogenous.
111 mProp = 0x0;
112 if (getBinding(attrExtinctionGainMult)) {
113 mProp |= IS_EXTINCTIVE;
114 } else if (!isBlack(get(attrExtinctionGainMult))) {
115 mProp |= IS_EXTINCTIVE | HOMOGENOUS_EXTINC;
116 }
117
118 if (getBinding(attrAlbedoMult)) {
119 mProp |= IS_SCATTERING;
120 } else if (!isBlack(get(attrAlbedoMult))) {
121 mProp |= IS_SCATTERING | HOMOGENOUS_ALBEDO;
122 }
123
124 if (getBinding(attrIncandGainMult)) {
125 mProp |= IS_EMISSIVE;
126 } else if (!isBlack(get(attrIncandGainMult))) {
127 mProp |= IS_EMISSIVE | HOMOGENOUS_EMISS;
128 }
129
130 if (get(attrAnisotropy) == 0.0f && getBinding(attrAnisotropy) == nullptr) {
131 mProp |= ISOTROPIC_PHASE;
132 }
133}
