21inline unsigned int f2ui(const float f)
22// Simple cast from float value to unsigned int
23{
24 if (f < 0.0f) return 0; // just in case.
25 return (unsigned int)f; // cast back to unsigned int

28template <typename T, typename F>
29void
30setTiledBuf(OiioReader &reader, T *dstBuf, F pixCopyFunc)
31// Copy all pixels by scanline segment order which is considered tiled memory layout
32// and maximize memory access coherency for both of source and destination in parallel.
33{
34 scene_rdl2::fb_util::Tiler tiler(reader.getWidth(), reader.getHeight());
35 reader.crawlAllTiledScanline([&](const int startX, const int endX, const int pixY, const int nChan, const float *pix) {
36 T *dstPix = dstBuf + tiler.linearCoordsToTiledOffset(startX, pixY);
37 for (int x = startX; x < endX; ++x) {
38 pixCopyFunc(dstPix, pix);
39 dstPix ++;
40 pix += nChan;
41 }
42 });
43}

47bool
48RenderOutputDriver::Impl::revertFilmData(Film &film,
49 unsigned &resumeTileSamples, int &resumeNumConsistentSamples, bool &zeroWeightMask,
50 bool &adaptiveSampling, float adaptiveSampleParam[3])
51//
52// Read all resume files which defined inside renderOutputDriver
53// return 3 values by argument resumeTileSamples and resumeNumConsistentSamples come from file's metadata and
54// zeroWeightMask condition is computed based on read weight AOV.
55// Read data is stored into film and not do any denormalize/zeroWeight operation yet.
56//
57{
58 // reset condition for read file
59 bool errorCondition = false;
60 mResumeTileSamples = -1;
61 mResumeNumConsistentSamples = -1;
62 mResumeAdaptiveSampling = -1;
63 mRevertWeightAOV = false;
64 mRevertBeautyAuxAOV = false;
65 mZeroWeightMask = false;
66
67 std::vector<std::string> resumeFiles;
68
69 // Loop over all files and to read all resume file
70 for (const auto &f: mFiles) {
71 if (!read(f, film)) { // read one resume file
72 std::ostringstream ostr;
73 ostr << "Read file failed and could not revert film data from file. filename:" << f.mResumeName;
74 mErrors.push_back(ostr.str());
75 errorCondition = true;
76 } else {
77 resumeFiles.push_back(f.mResumeName);
78 }
79 }
80 if (!errorCondition) {
81 if (!resumeRenderReadyTest()) {
82 errorCondition = true;
83 }
84 }
85
86 if (!errorCondition) {
87 resumeTileSamples = static_cast<unsigned>(mResumeTileSamples);
88 resumeNumConsistentSamples = mResumeNumConsistentSamples; // if resume file does not include this info. this value is -1
89 zeroWeightMask = mZeroWeightMask;
90 adaptiveSampling = (mResumeAdaptiveSampling == 1)? true: false;
91 adaptiveSampleParam[0] = mResumeAdaptiveParam[0];
92 adaptiveSampleParam[1] = mResumeAdaptiveParam[1];
93 adaptiveSampleParam[2] = mResumeAdaptiveParam[2]; // This is before multiply 10,000 value.
94
95 // log output
96 {
97 std::ostringstream ostr;
98 ostr << "resume render initial information {\n";
99 if (resumeFiles.size() == 1) {
100 ostr << " read resume file : " << resumeFiles[0] << '\n';
101 } else {
102 ostr << " read resume files (total:" << resumeFiles.size() << ") {\n";
103 int iw = std::to_string(resumeFiles.size()).size();
104 for (size_t i = 0; i < resumeFiles.size(); ++i) {
105 ostr << " " << std::setw(iw) << i << ' ' << resumeFiles[i] << '\n';
106 }
107 ostr << " }\n";
108 }
109 ostr << " progressCheckpointTileSamples:" << resumeTileSamples << '\n'
110 << " AovFilterNumConsistentSamples:" << resumeNumConsistentSamples << '\n';
111 if (adaptiveSampling) {
112 // display target adaptive error should be multiplied by 10,000.
113 float targetAdaptiveError = adaptiveSampleParam[2] * 10000.0f;
114 ostr << " sampling:ADAPTIVE\n"
115 << " adaptiveSampleParameters:"
116 << adaptiveSampleParam[0] << ' ' << adaptiveSampleParam[1] << ' ' << targetAdaptiveError << '\n';
117
118 } else {
119 ostr << " sampling:UNIFORM\n";
120 }
121 ostr << "}";
122 scene_rdl2::logging::Logger::info(ostr.str());
123 if (isatty(STDOUT_FILENO)) std::cout << ostr.str() << std::endl;
124 }
125
126 if (adaptiveSampling) {
127 // We have to store resume render initial condition sampleId as a separate buffer.
128 // 64 = tileWidth * tileHeight
129 film.getResumeStartSampleIdBuff().init(film.getWeightBuffer(), mResumeTileSamples / 64);
130 }
131 }
132 runOnResumeScript(!errorCondition, resumeFiles);
133
134 return !errorCondition;
135}

137bool
138RenderOutputDriver::Impl::resumeRenderReadyTest() const
139//
140// Can we continue to do resume render ?
141//
142{
143 auto dumpResumeFileList = [](std::ostringstream &ostr, const std::vector<File> &files) {
144 if (files.size() == 1) {
145 ostr << "specified resume file:" << files[0].mResumeName;
146 } else {
147 ostr << "specified resume files { ";
148 for (size_t i = 0; i < files.size(); ++i) {
149 ostr << files[i].mResumeName;
150 if (i != files.size() - 1) ostr << ", ";
151 }
152 ostr << " }";
153 }
154 };
155
156 std::ostringstream ostr;
157 bool errorCondition = false;
158
159 if (!mRevertWeightAOV) {
160 // We could not revert weight AOV from file. We can not resume render without weight.
161 ostr << "Could not revert weight AOV data from resume file. We need weight AOV data for resume render. ";
162 dumpResumeFileList(ostr, mFiles);
163 mErrors.push_back(ostr.str());
164 errorCondition = true;
165 }
166
167 if (mResumeTileSamples == -1) {
168 ostr.str("");
169 ostr << "Could not find \"progressCheckpointTileSamples\" metadata for resume. "
170 << "Please generate resume file with proper options. ";
171 dumpResumeFileList(ostr, mFiles);
172 mErrors.push_back(ostr.str());
173 errorCondition = true;
174 }
175
176 // mResumeNumConsistentSamples == -1 is not a critical error. We should continue to do resume render
177
178 if (!mRevertBeautyAuxAOV) {
179 ostr.str("");
180 ostr << "Could not revert \"beauty aux\" AOV data from resume file."
181 << " We need \"beauty aux\" AOV data for resume render. ";
182 dumpResumeFileList(ostr, mFiles);
183 mErrors.push_back(ostr.str());
184 errorCondition = true;
185 }
186
187 return !errorCondition;
188}

190bool
191RenderOutputDriver::Impl::read(const File &file, Film &film)
192//
193// Read one resume file and stored into film object as is.
194// No de-normalization and other operations at this moment and just read from file.
195//
196{
197 const std::string &filename = file.mResumeName;
198 if (filename.empty()) return true; // early exit. skip this File data
199
200 //
201 // open file (not read image data yet)
202 //
203 OiioReader reader(filename);
204 if (!reader) {
205 mErrors.push_back("Can't open file");
206 return false;
207 }
208
209 //
210 // get parameters for resume render (read resume render related metadatas)
211 //
212 if (!readResumableParameters(reader)) {
213 return false; // not enough information for resume render -> We can not resume render without them.
214 }
215
216 //
217 // subImage loop
218 //
219 for (size_t imgId = 0; imgId < file.mImages.size(); ++imgId) {
220 // read whole image data from file into memory inside reader here.
221 if (!reader.readData(imgId)) {
222 mErrors.push_back("read data failed.");
223 return false;
224 }
225 // std::cerr << reader.showSpec("") << std::endl; // useful for debug
226
227 // process reader internal memory data to proper way by multi-threaded operation
228 if (!readSubImage(reader, file.mImages[imgId], film)) {
229 return false;
230 }
231 }
232
233 return true;
234}

236bool
237RenderOutputDriver::Impl::readResumableParameters(OiioReader &reader)
238{
239 bool errorCondition = false;
240
241 //
242 // mResumeTileSamples
243 //
244 {
245 //
246 // Read tile sample information for resume rendering setup from resume file.
247 // Checkpoint file already has metadata "progressCheckpointTileSamples" and this number indicates
248 // how many tile samples are done. We need this number to restart redering.
249 //
250 int tileSamples;
251 if (reader.getMetadata("progressCheckpointTileSamples", tileSamples)) {
252 if (mResumeTileSamples == -1) {
253 mResumeTileSamples = tileSamples;
254 } else {
255 if (mResumeTileSamples != tileSamples) {
256 std::ostringstream ostr;
257 ostr << "progressCheckpointTileSamples mismatch between multi resume input files. "
258 << "resumeTileSamples:" << mResumeTileSamples << " != " << tileSamples
259 << " of file:" << reader.filename();
260 mErrors.push_back(ostr.str());
261 errorCondition = true;
262 }
263 }
264 }
265 }
266
267 //
268 // mResumeNumConsistentSamples
269 //
270 {
271 //
272 // Read numConsistentSamples for resume rendering. This number is used for FORCE_CONSISTENT_SAMPLING
273 // Aov filter to limit sampling count.
274 // Regardress of this image's AOV uses this AOV filter or not, just in case this numConsistentSamples
275 // value is saved as metadata. We should read here.
276 int numConsistentSamples;
277 if (reader.getMetadata("AovFilterNumConsistentSamples", numConsistentSamples)) {
278 if (mResumeNumConsistentSamples == -1) {
279 mResumeNumConsistentSamples = numConsistentSamples;
280 } else {
281 if (mResumeNumConsistentSamples != numConsistentSamples) {
282 std::ostringstream ostr;
283 ostr << "AovFilterNumConsistentSamples mismatch between multi resume input files. "
284 << "resumeNumConsistentSamples:" << mResumeNumConsistentSamples << " != " << numConsistentSamples
285 << " of file:" << reader.filename();
286 mErrors.push_back(ostr.str());
287 errorCondition = true;
288 }
289 }
290 }
291 }
292
293 //
294 // mResumeAdaptiveSampling, mResumeAdaptiveParam
295 //
296 {
297 //
298 // Read adaptiveSamplingV1 parameters. This info is used for control adaptive sampling resume rendering.
299 //
300 if (reader.getMetadata("adaptiveSamplingV1", mResumeAdaptiveParam)) {
301 if (mResumeAdaptiveSampling == -1) {
302 mResumeAdaptiveSampling = 1;
303 } else {
304 if (mResumeAdaptiveSampling != 1) {
305 std::ostringstream ostr;
306 ostr << "adaptiveSampleV1 mismatch between multi resume input files. "
307 << "resumeAdaptiveSampling:" << mResumeAdaptiveSampling << " != true"
308 << " of file:" << reader.filename();
309 mErrors.push_back(ostr.str());
310 errorCondition = true;
311 }
312 }
313 }
314 }
315
316 //
317 // resume render history
318 //
319 {
320 std::string resumeHistory;
321 if (reader.getMetadata("resumeHistory", resumeHistory)) {
322 if (mResumeHistory.empty()) {
323 mResumeHistory = std::move(resumeHistory);
324 } else {
325 if (mResumeHistory != resumeHistory) {
326 std::ostringstream ostr;
327 ostr << "resumeHistory information mismatch between multi resume input files."
328 << "resumeHistory:>" << mResumeHistory << "< != >" << resumeHistory
329 << " file:" << reader.filename();
330 mErrors.push_back(ostr.str());
331 errorCondition = true;
332 }
333 }
334 }
335 }
336
337 /* useful debug dump
338 std::cerr << ">> RenderOutputDriver.cc tileSamples:" << mResumeTileSamples << std::endl;
339 std::cerr << ">> RenderOutputDriver.cc numConsistentSamples:" << mResumeNumConsistentSamples << std::endl;
340 std::cerr << ">> RenderOutputDriver.cc"
341 << " mResumeAdaptiveSampling:" << mResumeAdaptiveSampling
342 << " mResumeAdaptiveParam:("
343 << mResumeAdaptiveParam[0] << ' ' << mResumeAdaptiveParam[1] << ' ' << mResumeAdaptiveParam[2]
344 << ")" << std::endl;
345 std::cerr << ">> RenderOutputDriver.cc resumeHistory:>" << mResumeHistory << "<" << std::endl;
346 */
347
348 return !errorCondition;
349}

351bool
352RenderOutputDriver::Impl::readSubImage(OiioReader &reader, const Image &currImage, Film &film)
353{
354 //
355 // Test correctness of subImage by name attribute.
356 //
357 if (!readSubImageNameValidityTest(reader, currImage)) {
358 return false;
359 }
360
361 //
362 // Image's Entry loop : processing each entry in parallel
363 //
364 int roIdxStart = currImage.mStartRoIdx;
365
366 bool errorCondition = false;
367
368#ifdef SINGLE_THREAD_READ
369 size_t taskSize = currImage.mEntries.size();
370#else
371 size_t taskSize = 1;
372#endif
373 tbb::blocked_range<size_t> range(0, currImage.mEntries.size(), taskSize);
374 tbb::parallel_for(range, [&](const tbb::blocked_range<size_t> &r) {
375 for (size_t eId = r.begin(); eId < r.end(); ++eId) {
376 if (!readSubImageOneEntry(reader, roIdxStart + eId, film)) {
377 errorCondition = true;
378 }
379 } // eId
380 });
381
382 return !errorCondition;
383}

385bool
386RenderOutputDriver::Impl::readSubImageNameValidityTest(OiioReader &reader, const Image &currImage) const
387{
388 std::ostringstream ostr;
389
390 std::string subImageName;
391 if (!currImage.mName.empty()) {
392 //
393 // resume read information need to mach with currImage's subImage name.
394 //
395 if (!reader.getMetadata("name", subImageName)) {
396 ostr << "Could not read subImge name attribute which should have (name:" << subImageName << "). "
397 << "This imply subImage order is wrong or improper resume file:" << reader.filename();
398 mErrors.push_back(ostr.str());
399 return false;
400 }
401 if (subImageName != currImage.mName) {
402 ostr << "subImage name is wrong (!=" << subImageName << "). "
403 << "This imply subImage order is wrong or improper resume file:" << reader.filename();
404 mErrors.push_back(ostr.str());
405 return false;
406 }
407 } else {
408 //
409 // currImage does not have subImage name attribute. So resume read information also should not have name
410 //
411 if (reader.getMetadata("name", subImageName)) {
412 ostr << "subImage should not have name attribute but it has (name:" << subImageName << "). "
413 << "This imply subImage order is wrong or improper resume file:" << reader.filename();
414 mErrors.push_back(ostr.str());
415 return false;
416 }
417 }
418 return true;
419}

421bool
422RenderOutputDriver::Impl::readSubImageOneEntry(OiioReader &reader,
423 const int roIdx, // index of RenderOutputDriver::Impl::mEntries[] about currEntry
424 Film &film)
425{
426 //
427 // Properly setup one Entry data from already read resume data into proper destination
428 // of memory inside Film object. Does not consider reverse normalization of weight value yet.
429 // Result of destination is simply same as resume file data at this point.
430 //
431
432 const Entry &currEntry = *(mEntries[roIdx]);
433
434 //
435 // Try to find destination buffer address.
436 //
437 scene_rdl2::fb_util::HeatMapBuffer *heatMapBuffer = nullptr;
438 scene_rdl2::fb_util::FloatBuffer *weightBuffer = nullptr;
439 scene_rdl2::fb_util::VariablePixelBuffer *aovBuffer = nullptr;
440 scene_rdl2::fb_util::RenderBuffer *renderBufferOdd = nullptr;
441 if (!readSubImageSetDestinationBuffer(roIdx,
442 film,
443 &heatMapBuffer, &weightBuffer, &renderBufferOdd, &aovBuffer)) {
444 return true; // we don't need to revert this entry from file (visibility_variance AOV). This is not a error
445 }
446
447 //
448 // Validity check about buffer resolution
449 //
450 if (!readSubImageResoValidityTest(reader, currEntry,
451 heatMapBuffer, weightBuffer, renderBufferOdd, aovBuffer,
452 film.getCryptomatteBuffer())) {
453 return false; // resolution check failed. This resume file is not sutable for resume render
454 }
455
456 //
457 // Setup pixel channel offset table information
458 //
459 std::vector<int> chanOffset;
460 if (!readSubImageSetPixChanOffset(reader, currEntry, chanOffset)) {
461 return false; // Could not find channel inside resume file. So no way to revert data from resume file.
462 }
463
464 //
465 // Loop over all pixels
466 //
467 // This logic is using same idea of RenderOutputDriver::Impl::fillBuffer() (but functionality opposite direction)
468 // Actually destination buffer is tiled memory layout and input read resume data is untiled.
469 // We have to think about memory layout conversion issue here. Main logic of tile <-> untile conversion is done
470 // by inside of "setTiledBuf()" templates.
471 //
472 const scene_rdl2::rdl2::RenderOutput *ro = currEntry.mRenderOutput;
473 switch (ro->getResult()) {
474 case scene_rdl2::rdl2::RenderOutput::RESULT_HEAT_MAP:
475 {
476 int64_t *dstBuf = heatMapBuffer->getData();
477 setTiledBuf(reader, dstBuf, [&](int64_t *dstPix, const float *srcPix) {
478 // We stored seconds value into 32bit single float and already lost some precision.
479 // Simply convert seconds value to nanoseconds as int64_t here.
480 *dstPix = mcrt_common::Clock::nanoseconds((double)srcPix[chanOffset[0]]);
481 });
482 }
483 break;
484 case scene_rdl2::rdl2::RenderOutput::RESULT_WEIGHT:
485 {
486 float *dstBuf = weightBuffer->getData();
487 bool zeroVal = false;
488 setTiledBuf(reader, dstBuf, [&](float *dstPix, const float *srcPix) {
489 *dstPix = srcPix[chanOffset[0]];
490 if (*dstPix == 0.0f) zeroVal = true;
491 });
492 mRevertWeightAOV = true; // This flag indicates weight AOV is properly reverted.
493 if (zeroVal) mZeroWeightMask = true;
494 }
495 break;
496 case scene_rdl2::rdl2::RenderOutput::RESULT_BEAUTY_AUX:
497 {
498 if (renderBufferOdd) { // only read if destination buffer is ready
499 scene_rdl2::fb_util::RenderColor *dstBuf = renderBufferOdd->getData();
500 setTiledBuf(reader, dstBuf, [&](scene_rdl2::fb_util::RenderColor *dstPix, const float *srcPix) {
501 (*dstPix)[0] = srcPix[chanOffset[0]];
502 (*dstPix)[1] = srcPix[chanOffset[1]];
503 (*dstPix)[2] = srcPix[chanOffset[2]];
504 });
505 mRevertBeautyAuxAOV = true; // This flag indicates beauty aux AOV is properly reverted.
506 }
507 }
508 break;
509 case scene_rdl2::rdl2::RenderOutput::RESULT_ALPHA_AUX:
510 {
511 if (renderBufferOdd) {
512 scene_rdl2::fb_util::RenderColor *dstBuf = renderBufferOdd->getData();
513 setTiledBuf(reader, dstBuf, [&](scene_rdl2::fb_util::RenderColor *dstPix, const float *srcPix) {
514 (*dstPix)[3] = srcPix[chanOffset[0]];
515 });
516 }
517 }
518 break;
519
520 case scene_rdl2::rdl2::RenderOutput::RESULT_CRYPTOMATTE:
521 {
522 // Clear the cryptomatte buffer ready for loading resume data
523 pbr::CryptomatteBuffer *cryptomatteBuf = film.getCryptomatteBuffer();
524 cryptomatteBuf->clear();
525 cryptomatteBuf->init(reader.getWidth(), reader.getHeight(), 1);
526
527 scene_rdl2::fb_util::Tiler tiler(reader.getWidth(), reader.getHeight());
528 reader.crawlAllTiledScanline
529 ([&](const int startX, const int endX, const int pixY, const int nChan,
530 const float *pix) {
531 for (int x = startX; x < endX; ++x) {
532 cryptomatteBuf->addFragments(x, pixY, pix, nChan / 2);
533 pix += nChan;
534 }
535 });
536 cryptomatteBuf->setFinalized(true);
537 }
538 break;
539
540 case scene_rdl2::rdl2::RenderOutput::RESULT_BEAUTY:
541 case scene_rdl2::rdl2::RenderOutput::RESULT_ALPHA:
542 case scene_rdl2::rdl2::RenderOutput::RESULT_STATE_VARIABLE:
543 case scene_rdl2::rdl2::RenderOutput::RESULT_DEPTH:
544 case scene_rdl2::rdl2::RenderOutput::RESULT_PRIMITIVE_ATTRIBUTE:
545 case scene_rdl2::rdl2::RenderOutput::RESULT_MATERIAL_AOV:
546 case scene_rdl2::rdl2::RenderOutput::RESULT_WIREFRAME:
547 case scene_rdl2::rdl2::RenderOutput::RESULT_LIGHT_AOV:
548 case scene_rdl2::rdl2::RenderOutput::RESULT_VARIANCE:
549 case scene_rdl2::rdl2::RenderOutput::RESULT_VISIBILITY_AOV:
550 switch (aovBuffer->getFormat()) {
551 case scene_rdl2::fb_util::VariablePixelBuffer::FLOAT:
552 {
553 float *dstBuf = aovBuffer->getFloatBuffer().getData();
554 setTiledBuf(reader, dstBuf, [&](float *dstPix, const float *srcPix) {
555 *dstPix = srcPix[chanOffset[0]];
556 });
557 }
558 break;
559 case scene_rdl2::fb_util::VariablePixelBuffer::FLOAT2:
560 {
561 scene_rdl2::math::Vec2f *dstBuf = aovBuffer->getFloat2Buffer().getData();
562 setTiledBuf(reader, dstBuf, [&](scene_rdl2::math::Vec2f *dstPix, const float *srcPix) {
563 dstPix->x = srcPix[chanOffset[0]];
564 dstPix->y = srcPix[chanOffset[1]];
565 });
566 }
567 break;
568 case scene_rdl2::fb_util::VariablePixelBuffer::FLOAT3:
569 {
570 scene_rdl2::math::Vec3f *dstBuf = aovBuffer->getFloat3Buffer().getData();
571 setTiledBuf(reader, dstBuf, [&](scene_rdl2::math::Vec3f *dstPix, const float *srcPix) {
572 dstPix->x = srcPix[chanOffset[0]];
573 dstPix->y = srcPix[chanOffset[1]];
574 dstPix->z = srcPix[chanOffset[2]];
575 });
576 }
577 break;
578 case scene_rdl2::fb_util::VariablePixelBuffer::FLOAT4:
579 if (ro->getMathFilter() == scene_rdl2::rdl2::RenderOutput::MathFilter::MATH_FILTER_CLOSEST) {
580 // This is a special case and this AOV is using the closest math filter
581 scene_rdl2::math::Vec4f *dstBuf = aovBuffer->getFloat4Buffer().getData();
582 switch (film.getAovNumFloats(getAovBuffer(roIdx))) {
583 case 1 :
584 setTiledBuf(reader, dstBuf, [&](scene_rdl2::math::Vec4f *dstPix, const float *srcPix) {
585 dstPix->x = srcPix[chanOffset[0]];
586 dstPix->y = 0.0f;
587 dstPix->z = 0.0f;
588 dstPix->w = srcPix[chanOffset[1]]; // depth
589 });
590 break;
591 case 2 :
592 setTiledBuf(reader, dstBuf, [&](scene_rdl2::math::Vec4f *dstPix, const float *srcPix) {
593 dstPix->x = srcPix[chanOffset[0]];
594 dstPix->y = srcPix[chanOffset[1]];
595 dstPix->z = 0.0f;
596 dstPix->w = srcPix[chanOffset[2]]; // depth
597 });
598 break;
599 case 3 :
600 setTiledBuf(reader, dstBuf, [&](scene_rdl2::math::Vec4f *dstPix, const float *srcPix) {
601 dstPix->x = srcPix[chanOffset[0]];
602 dstPix->y = srcPix[chanOffset[1]];
603 dstPix->z = srcPix[chanOffset[2]];
604 dstPix->w = srcPix[chanOffset[3]]; // depth
605 });
606 break;
607 default : break; // unexpected aov size -> skip
608 }
609 }
610 break;
611 case scene_rdl2::fb_util::VariablePixelBuffer::RGB_VARIANCE:
612 {
613 scene_rdl2::fb_util::RunningStatsLightWeight<float> *dstBuf =
614 aovBuffer->getRgbVarianceBuffer().getData();
615 setTiledBuf(reader, dstBuf, [&](scene_rdl2::fb_util::RunningStatsLightWeight<float> *dstPix,
616 const float *srcPix) {
617 dstPix->set(f2ui(srcPix[chanOffset[0]]),
618 srcPix[chanOffset[1]],
619 srcPix[chanOffset[2]],
620 srcPix[chanOffset[3]],
621 srcPix[chanOffset[4]]);
622 });
623 }
624 break;
625 case scene_rdl2::fb_util::VariablePixelBuffer::FLOAT_VARIANCE:
626 {
627 scene_rdl2::fb_util::RunningStatsLightWeight<float> *dstBuf =
628 aovBuffer->getFloatVarianceBuffer().getData();
629 setTiledBuf(reader, dstBuf, [&](scene_rdl2::fb_util::RunningStatsLightWeight<float> *dstPix,
630 const float *srcPix) {
631 dstPix->set(f2ui(srcPix[chanOffset[0]]),
632 srcPix[chanOffset[1]],
633 srcPix[chanOffset[2]],
634 srcPix[chanOffset[3]],
635 srcPix[chanOffset[4]]);
636 });
637 }
638 break;
639 case scene_rdl2::fb_util::VariablePixelBuffer::FLOAT2_VARIANCE:
640 {
641 scene_rdl2::fb_util::RunningStatsLightWeight<scene_rdl2::math::Vec2f> *dstBuf =
642 aovBuffer->getFloat2VarianceBuffer().getData();
643 setTiledBuf(reader, dstBuf, [&](scene_rdl2::fb_util::RunningStatsLightWeight<scene_rdl2::math::Vec2f> *dstPix,
644 const float *srcPix) {
645 dstPix->set(f2ui(srcPix[chanOffset[0]]),
646 scene_rdl2::math::Vec2f(srcPix[chanOffset[1]], srcPix[chanOffset[2]]),
647 scene_rdl2::math::Vec2f(srcPix[chanOffset[3]], srcPix[chanOffset[4]]),
648 scene_rdl2::math::Vec2f(srcPix[chanOffset[5]], srcPix[chanOffset[6]]),
649 scene_rdl2::math::Vec2f(srcPix[chanOffset[7]], srcPix[chanOffset[8]]));
650 });
651 }
652 break;
653 case scene_rdl2::fb_util::VariablePixelBuffer::FLOAT3_VARIANCE:
654 {
655 scene_rdl2::fb_util::RunningStatsLightWeight<scene_rdl2::math::Vec3f> *dstBuf =
656 aovBuffer->getFloat3VarianceBuffer().getData();
657 setTiledBuf(reader, dstBuf, [&](scene_rdl2::fb_util::RunningStatsLightWeight<scene_rdl2::math::Vec3f> *dstPix,
658 const float *srcPix) {
659 dstPix->set(f2ui(srcPix[chanOffset[0]]),
660 scene_rdl2::math::Vec3f(srcPix[chanOffset[1]],
661 srcPix[chanOffset[2]],
662 srcPix[chanOffset[3]]),
663 scene_rdl2::math::Vec3f(srcPix[chanOffset[4]],
664 srcPix[chanOffset[5]],
665 srcPix[chanOffset[6]]),
666 scene_rdl2::math::Vec3f(srcPix[chanOffset[7]],
667 srcPix[chanOffset[8]],
668 srcPix[chanOffset[9]]),
669 scene_rdl2::math::Vec3f(srcPix[chanOffset[10]],
670 srcPix[chanOffset[11]],
671 srcPix[chanOffset[12]]));
672 });
673 }
674 break;
675
676 default:
677 MNRY_ASSERT(0 && "unexpected variable pixel buffer format");
678 }
679 break;
680 case scene_rdl2::rdl2::RenderOutput::RESULT_DISPLAY_FILTER:
681 // nothing needs to happen here because display filters are recomputed
682 // from the aovs.
683 break;
684 default:
685 MNRY_ASSERT(0 && "unknown result type");
686 }
687
688 return true;
689}
