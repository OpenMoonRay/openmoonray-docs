40ProcessStats::getBytesRead() const
41{
42 std::string inStr;
43 int64 bytesRead = 0;
44
45 mReadIOMutex.lock();

47 mIOStatFile >> inStr >> bytesRead;
48 //reset file pointer for later reads

55 }
56 mReadIOMutex.unlock();

61int64
62ProcessStats::getProcessMemory() const
63{
64 int64 currentMemoryUsage = 0;
65
66 mMemoryReadMutex.lock();
67 if(mStatmFile.good()) {
68 int64 memorySize = 0;
69 int64 memoryResident = 0;
70 int64 memoryShared = 0;
71 //sometimes the OS is writing to the file
72 try {
73 mStatmFile >> memorySize >> memoryResident >> memoryShared;
74 } catch( std::ios_base::failure f ) {
75 scene_rdl2::logging::Logger::error(f.what());
76 }
77 currentMemoryUsage = memoryResident * sysconf(_SC_PAGE_SIZE);
78 //reset file pointer for later reads
79 mStatmFile.seekg(std::ios_base::beg);
80 } else {
81 // sometimes we lose the file handle so
82 // close it and try to reopen for the next query.
83 mStatmFile.close();
84 mStatmFile.open("/proc/self/statm");
85 }
86 mMemoryReadMutex.unlock();
87 return currentMemoryUsage;
88}
