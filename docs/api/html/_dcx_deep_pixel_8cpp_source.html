316size_t
317DeepPixel::append (const DeepSegment& bs,
318 const Pixelf& bp)
319{
320 const size_t new_segment_index = m_segments.size();
321 const size_t new_pixel_index = m_pixels.size();
322
323 if (m_segments.capacity() < m_segments.size()+1)
324 m_segments.reserve(m_segments.size() + (size_t)int(float(m_segments.size())/1.5f));
325 if (m_pixels.capacity() < m_pixels.size()+1)
326 m_pixels.reserve(m_pixels.size() + (size_t)int(float(m_pixels.size())/1.5f));
327 m_segments.push_back(bs);
328 m_pixels.push_back(bp);
329
330 m_pixels[new_pixel_index].channels = m_channels;
331 m_segments[new_segment_index].index = (int)new_pixel_index;
332
333 invalidateSort(); // force sort to re-run
334
335 return new_segment_index;
336}

378//
379// Combine the segments of two DeepPixels.
380//
381
382void
383DeepPixel::append (const DeepPixel& b)
384{
385 const size_t nCurrent = m_segments.size();
386 // Clear any new channels that are being added by the new set
387 // so we don't end up with junk left in the memory:
388 foreach_channel(z, b.m_channels)
389 {
390 if (!m_channels.contains(z))
391 {
392 for (size_t i=0; i < nCurrent; ++i)
393 m_pixels[i][z] = 0.0f;
394 }
395 }
396 m_channels += b.m_channels; // OR the channel masks together
397 // Copy the segments:
398 const size_t nAdded = b.m_segments.size();
399 m_segments.reserve(m_segments.size() + nAdded);
400 m_pixels.reserve(m_pixels.size() + nAdded);
401 for (size_t i=0; i < nAdded; ++i)
402 {
403 m_segments.push_back(b.m_segments[i]);
404 m_pixels.push_back(b.m_pixels[i]);
405 m_segments[m_segments.size()-1].index = (int)(m_pixels.size()-1);
406 m_pixels[m_pixels.size()-1].channels = m_channels;
407 }
408
409 invalidateSort(); // force sort to re-run
410}

414//
415// Remove a DeepSegment from the segment list, deleting its referenced Pixel.
416// Note that this method will possibly reorder some of the Pixel indices in the
417// DeepSegments, so a previously referenced Pixel index may become invalid and
418// need to be reaquired from its DeepSegment.
419//
420
421void DeepPixel::removeSegment (size_t segment_index)
422{
423 if (segment_index >= m_segments.size())
424 return;
425 const int overwritePixelIndex = m_segments[segment_index].index;
426 const int lastPixelIndex = (int)(m_pixels.size()-1);
427 if (overwritePixelIndex < lastPixelIndex)
428 {
429 // Find the DeepSegment pointing at the last Pixel, then
430 // copy the last Pixel in the list over the one being deleted
431 // and update the index in the changed DeepSegment:
432 const size_t nSegments = m_segments.size();
433 size_t lastPixelSegmentIndex = 0;
434 for (; lastPixelSegmentIndex < nSegments; ++lastPixelSegmentIndex)
435 if (m_segments[lastPixelSegmentIndex].index == lastPixelIndex)
436 break;
437#ifdef DCX_DEBUG_DEEPPIXEL
438 assert(lastPixelSegmentIndex < nSegments); // shouldn't happen...
439#endif
440 m_pixels[overwritePixelIndex] = m_pixels[lastPixelIndex];
441 m_segments[lastPixelSegmentIndex].index = overwritePixelIndex;
442 }
443 // Delete the DeepSegment and the last Pixel:
444 m_segments.erase(m_segments.begin()+segment_index);
445 m_pixels.pop_back();
446#ifdef DCX_DEBUG_DEEPPIXEL
447 assert(m_segments.size() == m_pixels.size());
448#endif
449 invalidateSort(); // force sort to re-run
450}

480//
481// Sort the segments. If the sorted flag is true this returns quickly.
482// This also updates global overlap and coverage flags.
483//
484
485void
486DeepPixel::sort (bool force)
487{
488 if (m_sorted && !force)
489 return;
490 m_sorted = true;
491 m_overlaps = false;
492 const size_t nSegments = m_segments.size();
493 if (nSegments == 0)
494 {
495 m_accum_or_mask = m_accum_and_mask = SpMask8::zeroCoverage;
496 m_accum_or_flags = m_accum_and_flags = DeepFlags::ALL_BITS_OFF;
497 return;
498 }
499
500 // Z-sort the segments:
501 std::sort(m_segments.begin(), m_segments.end());
502
503 const DeepSegment& segment0 = m_segments[0];
504
505 float prev_Zf = segment0.Zf;
506 float prev_Zb = segment0.Zb;
507 // Make sure mixed legacy full-coverage(allBitsOff) and true full-coverage(allBitsOn)
508 // segments are both considered full-coverage:
509 m_accum_or_mask = m_accum_and_mask = (segment0.fullCoverage())?SpMask8::fullCoverage:segment0.spMask();
510 m_accum_or_flags = m_accum_and_flags = segment0.flags();
511
512 // Determine global overlap and coverage status.
513 for (size_t i=1; i < nSegments; ++i)
514 {
515 const DeepSegment& segment = m_segments[i];
516 if (segment.Zf < prev_Zf || segment.Zb < prev_Zf ||
517 segment.Zf < prev_Zb || segment.Zb < prev_Zb)
518 m_overlaps = true;
519
520 const SpMask8 spmask = (segment.fullCoverage())?SpMask8::fullCoverage:segment.spMask();
521 m_accum_or_mask |= spmask;
522 m_accum_and_mask &= spmask;
523 m_accum_or_flags |= segment.flags();
524 m_accum_and_flags &= segment.flags();
525
526 prev_Zf = segment.Zf;
527 prev_Zb = segment.Zb;
528 }
529}

543//
544// Check for overlaps between samples and return true if so.
545// If the spmask arg is not full coverage then determine overlap
546// for the specific subpixel mask.
547//
548
549bool
550DeepPixel::hasOverlaps (const SpMask8& spmask,
551 bool force)
552{
553 const size_t nSegments = m_segments.size();
554 if (nSegments == 0)
555 return false;
556 // Sorting calculates the global overlap:
557 if (!m_sorted || force)
558 this->sort(force);
559 // If full coverage then we can return the global overlap indicator:
560 if (spmask == SpMask8::fullCoverage || allFullCoverage() || isLegacyDeepPixel())
561 return m_overlaps;
562 // Determine the overlap status for the spmask:
563 float prev_Zf = -INFINITYf;
564 float prev_Zb = -INFINITYf;
565 for (size_t i=0; i < nSegments; ++i)
566 {
567 const DeepSegment& segment = m_segments[i];
568 if (segment.maskBitsEnabled(spmask))
569 {
570 if (segment.Zf < prev_Zf || segment.Zb < prev_Zf ||
571 segment.Zf < prev_Zb || segment.Zb < prev_Zb)
572 return true;
573 prev_Zf = segment.Zf;
574 prev_Zb = segment.Zb;
575 }
576 }
577 return false;
578}

645int
646DeepPixel::findNearestMatch (float Zf, float Zb,
647 const SpMask8& match_spmask,
648 DeepFlags match_flags,
649 uint32_t start_index,
650 float depth_threshold)
651{
652 const size_t nSegments = m_segments.size();
653 if (nSegments == 0 || start_index >= nSegments)
654 return -1;
655
656 sort();
657 for (; start_index < nSegments; ++start_index)
658 {
659 const DeepSegment& segment = m_segments[start_index];
660 if ((match_flags!=0 && (segment.flags() & match_flags)==0) ||
661 (match_flags==0 && segment.flags()!=0))
662 continue;
663 else if (match_spmask!=SpMask8::fullCoverage && (match_spmask & segment.spMask())==0)
664 continue;
665 else if (fabsf(segment.Zf - Zf) <= depth_threshold &&
666 fabsf(segment.Zb - Zb) <= depth_threshold)
667 return start_index;
668 }
669
670 return -1;
671}

673int
674DeepPixel::findBestMatch (const DeepSegment& bs,
675 const Pixelf& bp,
676 const ChannelSet& compare_channels,
677 const SpMask8& match_spmask,
678 DeepFlags match_flags,
679 uint32_t start_index,
680 const CombineContext& combine_ctx,
681 float depth_threshold)
682{
683 ChannelSet compare = compare_channels;
684 compare &= bp.channels;
685 compare &= m_channels;
686 while (1) {
687 const int zmatch = findNearestMatch(bs.Zf, bs.Zb, match_spmask, match_flags, start_index, depth_threshold);
688 if (zmatch < 0)
689 return -1; // no z-match
690 // Compare color channels:
691 const Pixelf& ap = m_pixels[m_segments[zmatch].index];
692 bool cmatched = true;
693 foreach_channel(z, compare) {
694 if (fabsf(ap[z] - bp[z]) > combine_ctx.color_threshold) {
695 cmatched = false;
696 break;
697 }
698 }
699 if (cmatched)
700 return zmatch;
701 start_index = zmatch+1;
702 }
703}

711void
712DeepPixel::findNearestMatches (float Zf, float Zb,
713 float depth_threshold,
714 std::vector<int>& matched_segments_list)
715{
716 matched_segments_list.clear();
717 sort();
718 const size_t nSegments = size();
719 for (size_t i=0; i < nSegments; ++i)
720 {
721 const DeepSegment& segment = m_segments[i];
722 // Bail fast if we go outside possible depth range:
723 if ((segment.Zf - depth_threshold) > Zb)
724 return;
725 // If one of the Z's is outside threshold skip it:
726 if (fabsf(segment.Zf - Zf) > depth_threshold || fabsf(segment.Zb - Zb) > depth_threshold)
727 continue;
728 matched_segments_list.push_back((int)i);
729 }
730}

742inline
743int
744findSurfaceMatch (OPENDCX_INTERNAL_NAMESPACE::DeepPixel& dp,
745 const std::vector<int>& segment_indices,
746 const SpMask8& match_spmask,
747 const DeepFlags& match_flags,
748 int match_spbin_count)
749{
750 const size_t nZMatches = segment_indices.size();
751 for (size_t i=0; i < nZMatches; ++i)
752 {
753 const int index = segment_indices[i];
754 if (index < 0 || index >= (int)dp.size())
755 continue;
756 const DeepSegment& segment = dp.getSegment(index);
757 // Compare flags and partial-counts:
758 if (segment.flags() != match_flags ||
759 (match_spbin_count >= 0 && (int)segment.getSpCoverageCount() != match_spbin_count))
760 continue;
761 else if (match_spmask!=SpMask8::fullCoverage && !(match_spmask & segment.spMask()))
762 continue;
763 return (int)i;
764 }
765
766 return -1;
767}
768
769inline
770int
771findBestMatch2 (OPENDCX_INTERNAL_NAMESPACE::DeepPixel& dp,
772 const std::vector<int>& segment_indices,
773 const Pixelf& color,
774 const ChannelSet& compare_channels,
775 const SpMask8& match_spmask,
776 const DeepFlags& match_flags,
777 int match_spbin_count,
778 const DeepPixel::CombineContext& combine_ctx)
779{
780 ChannelSet compare = compare_channels;
781 compare &= dp.channels();
782 const size_t nZMatches = segment_indices.size();
783 for (size_t i=0; i < nZMatches; ++i)
784 {
785 const int index = segment_indices[i];
786 if (index < 0 || index >= (int)dp.size())
787 continue;
788 const DeepSegment& segment = dp.getSegment(index);
789 // Compare flags and partial-counts:
790 if (segment.flags() != match_flags ||
791 (match_spbin_count >= 0 && (int)segment.getSpCoverageCount() != match_spbin_count))
792 continue;
793 else if (match_spmask!=SpMask8::fullCoverage && !(match_spmask & segment.spMask()))
794 continue;
795 // Compare color channels:
796 const Pixelf& ap = dp.getSegmentPixel(segment);
797 bool cmatched = true;
798 foreach_channel(z, compare) {
799 if (fabsf(ap[z] - color[z]) > combine_ctx.color_threshold) {
800 cmatched = false;
801 break;
802 }
803 }
804 if (cmatched)
805 return segment_indices[i];
806 }
807
808 return -1;
809}

898//
899// Add a partial subpixel-coverage segment to the DeepPixel, first searching
900// for matches to combine within a subset list of existing segments.
901//
902// When matches occur the matched segment can be split into two where the original
903// segment has the new segment added to it with the matched bits, and the duplicated
904// segment containing the non-matched bits from the original matched segment.
905//
906
907void
908DeepPixel::appendOrCombineSegment (const std::vector<int>& segment_indices,
909 const DeepSegment& in_segment,
910 const Pixelf& in_pixel,
911 const CombineContext& combine_ctx)
912{
913 const bool in_full_spcoverage = in_segment.hasFullSpCoverage();
914#ifdef DCX_DEBUG_COLLAPSING
915 //std::cout << "xy[" << m_x << " " << m_y << "]" << std::endl;
916 const bool debug_enable = SAMPLINGXY(m_x, m_y);
917 if (debug_enable) {
918 std::cout << " flags=[";
919 in_segment.printFlags(std::cout);
920 std::cout << "] chans=" << in_pixel << std::endl;
921 in_segment.spMask().printPattern(std::cout, " ");
922 }
923#endif
924
925 // First try to match the color channels, flags and partial-weights.
926 // If there's a match we can bail quickly.
927 const int matched_values_index = findBestMatch2(*this,
928 segment_indices,
929 in_pixel,
930 in_pixel.channels,
931 SpMask8::fullCoverage,
932 in_segment.flags(),
933 in_segment.getSpCoverageCount(),
934 combine_ctx);
935 // If there's a values match determine if we ADD or OR the segments:
936 if (matched_values_index >= 0)
937 {
938 //------------------------------------------
939 // -------------- MATCH -----------------
940 //------------------------------------------
941 DeepSegment& matched_segment = m_segments[matched_values_index];
942
943 //--------------------------------------------------------------------------------
944 // Full spcoverage? OR the masks and we're done:
945 //--------------------------------------------------------------------------------
946 if (in_full_spcoverage)
947 {
948 matched_segment.metadata.spmask |= in_segment.spMask();
949#ifdef DCX_DEBUG_COLLAPSING
950 if (debug_enable) {
951 std::cout << " MATCH - combined full segment:" << std::endl;
952 matched_segment.spMask().printPattern(std::cout, " ");
953 }
954#endif
955 return;
956 }
957
958 //--------------------------------------------------------------------------------
959 // Partial spcoverage? The bits that overlap ADD together while the bits
960 // that don't OR together.
961 // If there's no overlapping bits we can OR them and bail.
962 // Overlapping masks case is handled like a no-match:
963 //--------------------------------------------------------------------------------
964
965 const SpMask8 matched_bits = (matched_segment.spMask() & in_segment.spMask());
966 if (matched_bits == SpMask8::zeroCoverage)
967 {
968 // No overlap? OR the masks and we're done:
969 matched_segment.metadata.spmask |= in_segment.spMask();
970#ifdef DCX_DEBUG_COLLAPSING
971 if (debug_enable) {
972 std::cout << " MATCH - combined partial segment:" << std::endl;
973 matched_segment.spMask().printPattern(std::cout, " ");
974 }
975#endif
976 return;
977
978 }
979
980 }
981 else
982 {
983 //------------------------------------------
984 // -------------- NO MATCH -----------------
985 //------------------------------------------
986
987 //--------------------------------------------------------------------------------
988 // Full spcoverage? Add a new opaque segment and bail:
989 //--------------------------------------------------------------------------------
990 if (in_full_spcoverage)
991 {
992#ifdef DCX_DEBUG_COLLAPSING
993 if (debug_enable) std::cout << " NO MATCH - new full segment" << std::endl;
994#endif
995 append(in_segment, in_pixel);
996 return;
997 }
998
999 }
1000
1001 //--------------------------------------------------------------------------------
1002 // No direct values match, but possibly combine PARTIAL spcoverage segments.
1003 //
1004 // (TODO: finish this comment)
1005 //
1006 // There's two scenarios primary:
1007 // 1) Input partial segment's spmask completely or partly matches one of
1008 // the existing partial segments. For the matching bits we add the
1009 // segments together, and for the non-matching bits
1010 //
1011 // 2) Input partial segment's spmask don't coincide at all, match
1012 //--------------------------------------------------------------------------------
1013
1014 const size_t nAllZMatches = segment_indices.size();
1015
1016 // Get the subset of Z matches that are partial and full and can safely combine with this segment:
1017 std::vector<int> partial_segment_indices;
1018 std::vector<int> full_segment_indices;
1019 partial_segment_indices.reserve(nAllZMatches);
1020 full_segment_indices.reserve(nAllZMatches);
1021 for (size_t i=0; i < nAllZMatches; ++i)
1022 {
1023 const int zmatched_index = segment_indices[i];
1024#ifdef DCX_DEBUG_COLLAPSING
1025 assert(zmatched_index >=0 && zmatched_index < (int)m_segments.size()); // shouldn't happen...
1026#else
1027 if (zmatched_index < 0 || zmatched_index >= (int)m_segments.size())
1028 continue;
1029#endif
1030 const DeepSegment& zmatched_segment = m_segments[zmatched_index];
1031
1032 if (zmatched_segment.hasFullSpCoverage())
1033 {
1034 if ((zmatched_segment.flags() | DeepFlags::ADDITIVE) == in_segment.flags())
1035 full_segment_indices.push_back(segment_indices[i]);
1036 continue;
1037 }
1038
1039 if (zmatched_segment.flags() != in_segment.flags() ||
1040 (zmatched_segment.getSpCoverageCount() + in_segment.getSpCoverageCount()) >= DeepFlags::maxSpCoverageCount)
1041 continue;
1042 partial_segment_indices.push_back(segment_indices[i]);
1043 }
1044 const size_t nPartialZMatches = partial_segment_indices.size();
1045
1046 // No partial segments to combine with? Append new segment and bail:
1047 if (nPartialZMatches == 0)
1048 {
1049#ifdef DCX_DEBUG_COLLAPSING
1050 if (debug_enable) std::cout << " NO MATCH - new partial segment" << std::endl;
1051#endif
1052 append(in_segment, in_pixel);
1053 return;
1054 }
1055
1056 // Find partial-coverage segments that have spmask bits that overlap with the new
1057 // segments.
1058 // If there's no overlaps or there's remaining bits that didn't overlap then add
1059 // a new segment:
1060#ifdef DCX_DEBUG_COLLAPSING
1061 if (debug_enable) {
1062 std::cout << " partialZmatches[";
1063 for (size_t i=0; i < nPartialZMatches; ++i)
1064 std::cout << " " << partial_segment_indices[i];
1065 std::cout << " ]";
1066 std::cout << " fullZmatches[";
1067 for (size_t i=0; i < full_segment_indices.size(); ++i)
1068 std::cout << " " << full_segment_indices[i];
1069 std::cout << " ]:" << std::endl;
1070 }
1071#endif
1072
1073 DeepSegment new_segment;
1074 Pixelf new_pixel;
1075
1076 // As matches are made this mask gets reduced:
1077 SpMask8 in_segment_bits = in_segment.spMask();
1078
1079 for (size_t i=0; i < nPartialZMatches; ++i)
1080 {
1081 const int matched_segment_index = partial_segment_indices[i];
1082 DeepSegment& matched_segment = m_segments[matched_segment_index];
1083
1084 const SpMask8 matched_bits = (matched_segment.spMask() & in_segment_bits);
1085 if (matched_bits == SpMask8::zeroCoverage)
1086 continue; // no overlaps
1087
1088 Pixelf& matched_pixel = getSegmentPixel(matched_segment);
1089
1090 const uint32_t bin_count_sum = matched_segment.getSpCoverageCount() + in_segment.getSpCoverageCount();
1091
1092#ifdef DCX_DEBUG_COLLAPSING
1093 assert(bin_count_sum <= DeepFlags::maxSpCoverageCount);
1094 if (debug_enable) {
1095 std::cout << " [ " << matched_segment_index << " ] Z-MATCH:";
1096 std::cout << " in_bin_count=" << in_segment.getSpCoverageCount();
1097 std::cout << ", matched_bin_count=" << matched_segment.getSpCoverageCount();
1098 std::cout << ", bin_count_sum=" << bin_count_sum;
1099 std::cout << std::endl;
1100 //std::cout << " in_segment_bits:" << std::endl;
1101 //in_segment_bits.printPattern(std::cout, " ");
1102 //std::cout << " matched_bits:" << std::endl;
1103 //matched_bits.printPattern(std::cout, " ");
1104 //std::cout << " unmatched_bits:" << std::endl;
1105 //const SpMask8 unmatched_bits = (matched_segment.spMask() & ~in_segment_bits);
1106 //unmatched_bits.printPattern(std::cout, " ");
1107 }
1108#endif
1109
1110 // If all spmask bits match add the partial segments:
1111 if (matched_bits == matched_segment.spMask())
1112 {
1113 new_segment = matched_segment;
1114 new_pixel = matched_pixel;
1115
1116#ifdef DCX_DEBUG_COLLAPSING
1117 if (debug_enable) std::cout << " MATCHED BITS (new_bin_count=" << bin_count_sum << ")";
1118#endif
1119 // If adding the partials together causes alpha to saturate then we
1120 // correct for alpha overshooting 1.0 by attenuating additive contribution
1121 // by the overshoot amount to avoid any brightening artifacts.
1122 // Finally mark the segment as opaque vs. partial:
1123 if (bin_count_sum >= DeepFlags::maxSpCoverageCount)
1124 {
1125 // Add partial segments into opaque segment:
1126 new_segment.clearSpCoverageCount(); // turn off partial coverage
1127 if ((new_pixel[Chan_A] + in_pixel[Chan_A]) >= 1.0f)
1128 new_pixel += in_pixel*((1.0f - new_pixel[Chan_A]) / in_pixel[Chan_A]);
1129 else
1130 new_pixel += in_pixel;
1131
1132 // We can possibly further combine this new opaque segment together
1133 // with the other opaque segments, but this time we would need to delete
1134 // one of the segments:
1135 const int matched_full_index = findBestMatch2(*this,
1136 full_segment_indices,
1137 new_pixel,
1138 new_pixel.channels,
1139 SpMask8::fullCoverage,
1140 new_segment.flags(),
1141 DeepFlags::maxSpCoverageCount,
1142 combine_ctx);
1143 // If match combine them into the new match by simply ORing the spmasks
1144 // then delete the currently matched segment, removing it from the list
1145 // of active matches:
1146 if (matched_full_index >= 0)
1147 {
1148 DeepSegment& combined_segment = getSegment(matched_full_index);
1149 combined_segment.metadata.spmask |= matched_bits;
1150 removeSegment(matched_segment_index);
1151#ifdef DCX_DEBUG_COLLAPSING
1152 if (debug_enable) {
1153 std::cout << " - SECONDARY FULL MATCH: combine_segment=" << matched_full_index;
1154 std::cout << ", remove_segment=" << matched_segment_index << std::endl;
1155 combined_segment.spMask().printPattern(std::cout, " ");
1156 }
1157#endif
1158 }
1159 else
1160 {
1161 // Save combined:
1162 matched_segment = new_segment;
1163 matched_pixel = new_pixel;
1164#ifdef DCX_DEBUG_COLLAPSING
1165 if (debug_enable) std::cout << " -> opaque:" << std::endl;
1166#endif
1167 }
1168
1169 }
1170 else
1171 {
1172 // Add partial segments:
1173 new_segment.setSpCoverageCount(bin_count_sum);
1174 new_pixel += in_pixel;
1175#ifdef DCX_DEBUG_COLLAPSING
1176 if (debug_enable) std::cout << " -> new weight=" << new_segment.getSpCoverageWeight() << ":" << std::endl;
1177#endif
1178
1179 // If the newly added-together weight matches a previous weight we can combine
1180 // yet again, but this time we need to delete one of the segments. This
1181 // can get rid of the first-segment partials that can't be otherwise collapsed:
1182 partial_segment_indices[i] = -1; // so we don't match the same one again
1183 const int matched_partial_index = findBestMatch2(*this,
1184 partial_segment_indices,
1185 new_pixel,
1186 new_pixel.channels,
1187 SpMask8::fullCoverage,
1188 new_segment.flags(),
1189 bin_count_sum,
1190 combine_ctx);
1191 // If matched again, combine them into this new match by ORing the spmasks.
1192 // Then delete the currently matched segment, and removing it from the list
1193 // of active matches:
1194 if (matched_partial_index >= 0)
1195 {
1196#ifdef DCX_DEBUG_COLLAPSING
1197 if (debug_enable) std::cout << " - SECONDARY PARTIAL MATCH=" << matched_partial_index;
1198#endif
1199 DeepSegment& combined_segment = getSegment(matched_partial_index);
1200 combined_segment.metadata.spmask |= matched_bits;
1201 removeSegment(matched_segment_index);
1202 }
1203 else
1204 {
1205 // Restore the matched index if no match and save combined:
1206 partial_segment_indices[i] = matched_segment_index;
1207 matched_segment = new_segment;
1208 matched_pixel = new_pixel;
1209 }
1210
1211#ifdef DCX_DEBUG_COLLAPSING
1212 if (debug_enable) std::cout << " -> partial:" << std::endl;
1213#endif
1214
1215 }
1216
1217 // Turn off matched bits:
1218 in_segment_bits &= ~matched_bits;
1219 }
1220
1221 } // for nPartialZmatches
1222
1223
1224 // Any remaining unmatched bits go to a new partial segment:
1225 if (in_segment_bits != SpMask8::zeroCoverage)
1226 {
1227 new_segment = in_segment;
1228 new_segment.metadata.spmask = in_segment_bits;
1229 append(new_segment, in_pixel);
1230#ifdef DCX_DEBUG_COLLAPSING
1231 if (debug_enable) {
1232 std::cout << " UNMATCHED BITS - new partial segment:" << std::endl;
1233 new_segment.spMask().printPattern(std::cout, " ");
1234 }
1235#endif
1236 }
1237#ifdef DCX_DEBUG_COLLAPSING
1238 if (debug_enable) std::cout << " ---------------------------------------------------------" << std::endl;
1239#endif
1240
1241}

1302//
1303// Log-merge together a subsection from a set of segments.
1304// Merge ALL-LOG no-spmasks samples - calculate total absorption for merged
1305// segments using Foundry-compatible (legacy Nuke) math.
1306//
1307// Segment 'subsegment' contains the Zf-Zb depth range to merge
1308// which must fall within the min/max depth range of the segment
1309// set - no checks are performed to verify this!
1310//
1311
1312void
1313DeepPixel::mergeSectionLegacy (const SegmentEdgeSet& segment_set,
1314 double Zf,
1315 double Zb,
1316 const ChannelSet& channels,
1317 Pixelf& out,
1318 DeepMetadata& merged_metadata)
1319{
1320 Pixelf subsection_color(channels);
1321 const double subsection_thickness = (Zb - Zf);
1322
1323 out.erase(channels);
1324 double absorption_accum = 0.0;
1325 float merged_alpha = 0.0f;
1326
1327 merged_metadata.spmask = SpMask8::fullCoverage;
1328 merged_metadata.flags = DeepFlags::ALL_BITS_OFF;
1329 bool merged_all_matte = true;
1330
1331#ifdef DCX_DEBUG_FLATTENER
1332 const bool debug_enable = SAMPLINGXY(m_x, m_y);
1333#endif
1334
1335 for (SegmentEdgeSet::const_iterator it=segment_set.begin(); it != segment_set.end(); ++it)
1336 {
1337#ifdef DCX_DEBUG_FLATTENER
1338 assert(*it < m_segments.size());
1339#endif
1340 const DeepSegment& segment = m_segments[*it];
1341 const Pixelf& segment_color = getSegmentPixel(segment);
1342
1343 // Get subsection within segment:
1344 const double segment_thickness = (double(segment.Zb) - double(segment.Zf));
1345 const double t = (segment_thickness < EPSILONd)?1.0:(subsection_thickness / segment_thickness);
1346
1347#ifdef DCX_DEBUG_FLATTENER
1348 if (debug_enable) {
1349 std::cout.precision(8);
1350 std::cout << " " << *it << " [" << segment.Zf << " " << segment.Zb;
1351 std::cout << " (" << segment_thickness << " thick)]";
1352 std::cout << "] t=" << t << ", LOG INTERPOLATE (LEGACY)";
1353 }
1354#endif
1355
1356 // Only apply alpha correction if alpha is a grey value:
1357 const float interp_alpha = segment_color[Chan_A];
1358 subsection_color = segment_color;
1359 if (segment.isMatte())
1360 {
1361 // Matte object, blacken color channels except alpha:
1362 const float a = subsection_color[Chan_A];
1363 subsection_color.erase(channels);
1364 subsection_color[Chan_A] = a;
1365#ifdef DCX_DEBUG_FLATTENER
1366 if (debug_enable) std::cout << ", MATTE";
1367#endif
1368
1369 }
1370 else
1371 {
1372 merged_all_matte = false;
1373 subsection_color[Chan_ACutout] = subsection_color[Chan_A];
1374 }
1375
1376 // Composite with final alpha for all samples at this segment:
1377 if (interp_alpha <= 0.0f || interp_alpha >= 1.0f)
1378 {
1379 merged_alpha = merged_alpha*(1.0f - interp_alpha) + interp_alpha;
1380#ifdef DCX_DEBUG_FLATTENER
1381 if (debug_enable) std::cout << ", interp_alpha=" << interp_alpha << std::endl;
1382#endif
1383 }
1384 else
1385 {
1386 const double viz = 1.0 - CLAMP(interp_alpha);
1387 const float subsection_alpha = float(1.0 - pow(viz, t));
1388 const float correction = subsection_alpha / interp_alpha;
1389 foreach_channel(z, channels)
1390 subsection_color[z] *= correction;
1391 subsection_color[Chan_A] = subsection_alpha;
1392
1393 merged_alpha = merged_alpha*(1.0f - subsection_alpha) + subsection_alpha;
1394#ifdef DCX_DEBUG_FLATTENER
1395 if (debug_enable) std::cout << ", interp_alpha=" << interp_alpha << ", subsection_alpha=" << subsection_alpha << std::endl;
1396#endif
1397 }
1398
1399 const float subsection_alpha = subsection_color[Chan_A];
1400 if (subsection_alpha <= 0.0f)
1401 {
1402 // Alpha transparent, skip it
1403
1404 }
1405 else if (subsection_alpha < 1.0f)
1406 {
1407 // Partially-transparent, find the absorbance-weighted average of the
1408 // unpremultiplied subsection color:
1409 const double absorbance = -log(double(1.0f - subsection_alpha));
1410 const float inv_subsection_alpha = 1.0f / subsection_alpha;
1411 foreach_channel(z, channels)
1412 out[z] += float(double(subsection_color[z]*inv_subsection_alpha) * absorbance);
1413 absorption_accum += absorbance;
1414 }
1415 else
1416 {
1417 // Alpha saturated, max absorbance:
1418 if (1/*segment_set.size() == 1*/)
1419 {
1420 // Handle single segment merges:
1421 foreach_channel(z, channels)
1422 out[z] += subsection_color[z];
1423 }
1424 absorption_accum += MAX_ABSORBANCE;
1425 }
1426
1427#ifdef DCX_DEBUG_FLATTENER
1428 if (debug_enable) {
1429 std::cout.precision(6);
1430 std::cout << " subsection_color" << subsection_color << std::endl;
1431 std::cout << " merged_color" << out << std::endl;
1432 }
1433#endif
1434
1435 } // active segments merge loop
1436
1437 // Weight final merged result by the accumulated absorption factor:
1438 if (absorption_accum < EPSILONd)
1439 {
1440 foreach_channel(z, channels)
1441 out[z] = 0.0f;
1442 }
1443 else
1444 {
1445 absorption_accum = 1.0 / absorption_accum;
1446 foreach_channel(z, channels)
1447 out[z] = float(double(out[z]) * absorption_accum)*merged_alpha;
1448 }
1449 out[Chan_A] = merged_alpha;
1450 out[Chan_SpCoverage] = 1.0f; // legacy segments are always have full spcoverage
1451
1452 if (merged_all_matte)
1453 merged_metadata.flags |= DeepFlags::MATTE_OBJECT;
1454
1455#ifdef DCX_DEBUG_FLATTENER
1456 if (debug_enable) std::cout << " final merged_color" << out << std::endl;
1457#endif

1465// Flatten the DeepSegments down into the output Pixel using front-to-back UNDERs.
1466//
1467
1468void
1469DeepPixel::flatten (const ChannelSet& out_channels,
1470 Pixelf& out,
1471 InterpolationMode interpolation)
1472{
1473#ifdef DCX_DEBUG_FLATTENER
1474 const bool debug_enable = SAMPLINGXY(m_x, m_y);
1475 if (debug_enable) {
1476 std::cout << " DeepPixel::flatten() channels=" << out_channels << ", hasOverlaps=" << m_overlaps;
1477 std::cout << ", allFullCoverage=" << allFullCoverage() << ", isLegacy=" << isLegacyDeepPixel();
1478 std::cout << ", ANDmask=" << std::hex << m_accum_and_mask << ", ANDflags=" << m_accum_and_flags << std::dec;
1479 }
1480#endif
1481
1482 // If full coverage or there's no spmasks than we just have to flatten once:
1483 if (allFullCoverage() || isLegacyDeepPixel() || interpolation == INTERP_OFF)
1484 {
1485#ifdef DCX_DEBUG_FLATTENER
1486 if (debug_enable)
1487 std::cout << ": FULL-COVERAGE: interpolation=" << interpolationModeString(interpolation) << std::endl;
1488#endif
1489 if (!hasOverlaps() || interpolation == INTERP_OFF)
1490 flattenNoOverlaps(out_channels, out, SpMask8::fullCoverage); // No overlaps, do a simple linear flatten
1491 else
1492 flattenOverlapping(out_channels, out, SpMask8::fullCoverage, interpolation); // Merge overlapping deep segments
1493 return;
1494 }
1495
1496 //================================================================================
1497 // SUBPIXEL FLATTENING LOOP
1498 // Flatten each subpixel individually where each subpixel may contain a different
1499 // combination of enabled segments.
1500 //================================================================================
1501#ifdef DCX_DEBUG_FLATTENER
1502 if (debug_enable) std::cout << " - SUBPIXEL-LOOP: interpolation=" << interpolationModeString(interpolation) << std::endl;
1503#endif
1504
1505 out.erase(out_channels);
1506 out[Chan_ZFront] = INFINITYf;
1507 out[Chan_ZBack ] = INFINITYf;
1508
1509 Pixelf flattened(out_channels);
1510
1511#ifdef DCX_USE_LARGEST_COVERAGE_Z
1512 // Handle the Z accumulation with coverage weighting - i.e. pick Z from
1513 // the nearest sample with the largest coverage.
1514 // TODO: Finishing this code will likely produce more accurate flattened Z's
1515 std::map<float, int> coverage_map;
1516#endif
1517
1518 // Flatten just the segments that has their sp mask bit enabled for the
1519 // current subpixel:
1520 SpMask8 sp_mask(1ull);
1521 for (int sp_bin=0; sp_bin < SpMask8::numBits; ++sp_bin, ++sp_mask) {
1522 if (!hasOverlaps(sp_mask, false/*force*/))
1523 flattenNoOverlaps(out_channels, flattened, sp_mask); // No overlaps, do a simple linear flatten
1524 else
1525 flattenOverlapping(out_channels, flattened, sp_mask, interpolation); // Merge overlapping deep segments
1526
1527 if (flattened[Chan_A] < EPSILONf)
1528 continue;
1529
1530 // Add the flattened subpixel color to accumulation pixel and handle Z's:
1531 out += flattened;
1532#ifdef DCX_USE_LARGEST_COVERAGE_Z
1533 // Record the ZFronts in a map to count the dominant Zf:
1534 const float Zf = flattened[Chan_ZFront];
1535 if (Zf < INFINITYf)
1536 {
1537 std::map<float, int>::iterator it = coverage_map.find(Zf);
1538 if (it == coverage_map.end())
1539 coverage_map[Zf] = 1;
1540 else
1541 ++it->second;
1542 }
1543#else
1544 // Find nearest Z's:
1545 out[Chan_ZFront] = std::min(out[Chan_ZFront], flattened[Chan_ZFront]);
1546#endif
1547 out[Chan_ZBack] = std::max(out[Chan_ZBack], flattened[Chan_ZBack]);
1548 }
1549 out /= float(SpMask8::width*SpMask8::height);
1550
1551#ifdef DCX_USE_LARGEST_COVERAGE_Z
1552 // Find frontmost Z with largest coverage count:
1553 int max_coverage = 0;
1554 for (std::map<float, int>::iterator it=coverage_map.begin(); it != coverage_map.end(); ++it)
1555 {
1556 if (it->second > max_coverage)
1557 {
1558 out[Chan_ZFront] = std::min(out[Chan_ZFront], it->first);
1559 max_coverage = it->second;
1560 }
1561 }
1562#endif
1563
1564} // DeepPixel::flatten

1786
1787#ifdef DCX_DEBUG_FLATTENER
1788 if (debug_enable) {

1790 std::cout << ", allVolumetric=" << allVolumetric();
1791 std::cout << ", useSpMasks=" << useSpMasks << ", interpolation=" << interpolationModeString(interpolation) << std::endl;

1796 SegmentEdgeList segment_edges;

1800 out[Chan_ZBack] = INFINITYf;
1801#ifdef DCX_DEBUG_FLATTENER
1802 if (debug_enable)

1804#endif

1809 std::sort(segment_edges.begin(), segment_edges.end());

1832 std::cout.precision(8);
1833 std::cout << " " << ((j < 100)?((j < 10)?" ":" "):"") << j;
1834 std::cout << ": segment=" << ((edge.segment < 100)?((edge.segment < 10)?" ":" "):"") << edge.segment;
1835 std::cout << "[" << ((edge.type == DeepSegment::Edge::FRONT)?"front":(edge.type == DeepSegment::Edge::BACK)?" back":" thin");
1836 std::cout << "] depth=" << edge.depth;
1837 std::cout << ", Zf=" << segment.Zf << " Zb=" << segment.Zb;
1838 std::cout << " flags=["; segment.printFlags(std::cout); std::cout << "]";
1839 if (edge.type != DeepSegment::Edge::BACK) {

1841 std::cout << ", color" << pixel;
1842 }
1843 std::cout << std::endl;

1845 std::cout << " ----------------- edges loop -------------------" << std::endl;
1846 std::cout << " ========= "; out.print(std::cout, "OUT", 12/*prec*/, comp_channels_plus_cutouts);
1847 std::cout << ", viz=" << out[Chan_Visibility];
1848 std::cout << " =========" << std::endl;
1849 }
1850#endif
1851 for (uint32_t j=0; j < nEdges; ++j)
1852 {
1853 const DeepSegment::Edge& edge = segment_edges[j];
1854#ifdef DCX_DEBUG_FLATTENER
1855 assert(edge.segment < nSegments);
1856#endif
1857 const DeepSegment& segment0 = m_segments[edge.segment];
1858
1859#ifdef DCX_DEBUG_FLATTENER
1860 if (debug_enable) {
1861 std::cout.precision(8);
1862 std::cout << " e" << j << ": segment=" << edge.segment;
1863 std::cout << ", ";
1864 if (edge.type == DeepSegment::Edge::FRONT)
1865 std::cout << "front - ** start edge **";
1866 else if (edge.type == DeepSegment::Edge::BACK)
1867 std::cout << "back - ** remove edge **";
1868 else
1869 std::cout << "thin - ** UNDER/PLUS immediate **";
1870 if (edge.type != DeepSegment::Edge::BACK) {
1871 std::cout << " flags=["; segment0.printFlags(std::cout);
1872 std::cout << "] depth=" << edge.depth;
1873 std::cout << ", color" << getSegmentPixel(edge.segment);
1874 }
1875 std::cout << std::endl;
1876 }

1882 // So, for each edge add and remove samples as appropriate:
1883 if (edge.type == DeepSegment::Edge::FRONT)
1884 {
1885 if (segment0.isHardSurface())
1886 ++nLinSamples;
1887 else
1888 ++nLogSamples;

1890 // edge is this one's back. If so we don't need to bother
1891 // combining and can composite immediately:
1892 if (active_segments.empty() && j < (nEdges - 1) &&
1893 segment_edges[j+1].segment == edge.segment && segment_edges[j+1].type == DeepSegment::Edge::BACK)
1894 compositeImmediate = true;
1895 else
1896 active_segments.insert(edge.segment); // Add samples on their front edge
1897 }
1898 else if (edge.type == DeepSegment::Edge::BACK)
1899 {
1900 active_segments.erase(edge.segment); // Remove samples on their back edge
1901 if (segment0.isHardSurface())
1902 --nLinSamples;
1903 else
1904 --nLogSamples;
1905 }
1906 else
1907 {
1908 // edge.type == DeepSegment::Edge::THIN
1909 // For samples where front == back, don't bother attempting to merge with
1910 // other active segments since it has no thickness - immediately composite it:
1911 compositeImmediate = true;
1912 }
1913
1914 if (compositeImmediate)
1915 {
1916#ifdef DCX_DEBUG_FLATTENER
1917 if (debug_enable) {
1918 getSegmentPixelWithMetadata(segment0, comp_channels_plus_cutouts, segment_color);
1919 std::cout << " color" << segment_color << " - COMPOSITE IMMEDIATE" << std::endl;
1920 }
1921#endif

1929 }

1938 }

1941 if (active_segments.empty())
1942 continue;
1943
1944 // There is always at least one BACK edge in the list at this point to disable
1945 // each active FRONT edge. In the final iteration in the iterator for-loop,
1946 // active_segments should be empty!
1947#ifdef DCX_DEBUG_FLATTENER
1948 assert((j + 1) < nEdges); // shouldn't heppen...
1949#endif
1950
1951 bool linear_interp0 = (interpolation == INTERP_LIN ||
1952 (interpolation == INTERP_AUTO && segment0.isHardSurface()));
1953
1954 // Get Z distance between this edge and the next edge and build
1955 // a temp segment that spans the subsection:
1956 DeepSegment subsegment(edge.depth, segment_edges[j + 1].depth, segment0.index, segment0.metadata);
1957 const double subsegmentZf = double(subsegment.Zf);
1958 const double subsegmentZb = double(subsegment.Zb);
1959 const double subsegment_thickness = (subsegmentZb - subsegmentZf);
1960
1961 // If segment too thin to interpolate - skip it:
1962 if (!linear_interp0 && subsegment.thickness() < EPSILONf)
1963 continue;
1964
1965
1966#ifdef DCX_DEBUG_FLATTENER
1967 if (debug_enable) {
1968 std::cout << " combine active samples: interpolation=" << interpolationModeString(interpolation);
1969 std::cout << ", nLogSamples=" << nLogSamples << ", nLinSamples=" << nLinSamples;
1970 std::cout << ", [ ";
1971 for (SegmentEdgeSet::const_iterator it=active_segments.begin(); it != active_segments.end(); ++it)
1972 std::cout << *it << ", ";
1973 std::cout.precision(8);
1974 std::cout << "], Zf=" << subsegment.Zf << ", Zb=" << subsegment.Zb << ", thickness=" << subsegment.thickness();
1975 std::cout << std::endl;
1976 }
1977#endif

1979 //---------------------------------------------------------------------------------
1980 // Handle combinations of log and lin segment interpolation types.
1981 //
1982 // If we have all-log active samples and no linear active samples then we can
1983 // use Florian's log merge math, otherwise we need to step through the
1984 // sub-segment range evaluating the samples at each step distance and under them.
1985 //
1986 // TODO: can the all-lin case be done through simple math?
1987 //
1988 //---------------------------------------------------------------------------------
1989
1990
1991#ifdef DCX_DEBUG_FLATTENER
1992 assert(*active_segments.begin() < m_segments.size());
1993#endif
1994
1995 if (interpolation == INTERP_LOG ||
1996 (interpolation == INTERP_AUTO && nLogSamples > 0 && nLinSamples == 0))
1997 {
1998 //=========================================================================================
1999 //
2000 // ALL-LOG samples to merge
2001 //
2002 //=========================================================================================
2003
2004 // Handle legacy vs. spmask log deep samples:
2005 if (legacySamples && !useSpMasks)
2006 {
2007 //---------------------------------------------------------------
2008 // No subpixel masks, use the legacy (old Nuke) log merging math:
2009 //---------------------------------------------------------------
2010 DeepMetadata merged_metadata;
2011 mergeSectionLegacy(active_segments,
2012 subsegment.Zf,
2013 subsegment.Zb,
2014 comp_channels_plus_cutouts,
2015 merged_color,
2016 merged_metadata);
2017
2018 compositeSegmentUnder(out, subsegment, merged_color, comp_channels);
