9// - A tile is the finest granularity we can split up rendering within a single pass.
10// - Tiles are the perfect format for sending data to a merge node since all the pixels
11// for a single tile are contigous in memory. The message where tiles are sent
12// to a merge note should contain a list of tightly packed tiles. This implies
13// that detiling should be done on the merge node (eventually).
14// - No explicit cropping is neccessary since this is handled in the extrapolation passes.

27void
28distributeTiles(std::vector<scene_rdl2::fb_util::Tile>& tiles, unsigned renderNodeIdx,
29 unsigned numRenderNodes)
30{
31 const unsigned numTiles = unsigned(tiles.size());
32 std::vector<scene_rdl2::fb_util::Tile> temp;
33 temp.reserve(numTiles / numRenderNodes + 1u);
34
35 for (unsigned i = renderNodeIdx; i < numTiles; i += numRenderNodes) {
36 temp.push_back(tiles[i]);
37 }
38
39 tiles.swap(temp);
40}

56unsigned
57TileScheduler::generateTiles(scene_rdl2::alloc::Arena *arena,
58 unsigned width,
59 unsigned height,
60 const scene_rdl2::math::Viewport &viewport,
61 unsigned renderNodeIdx,
62 unsigned numRenderNodes,
63 unsigned taskDistribType)
64{
65 MNRY_ASSERT(viewport.mMinX <= viewport.mMaxX && viewport.mMinY <= viewport.mMaxY);
66 MNRY_ASSERT(viewport.mMaxX < (int)width && viewport.mMaxY < (int)height);
67 MNRY_ASSERT(renderNodeIdx < numRenderNodes);
68
69 mRenderNodeIdx = renderNodeIdx;
70 mNumRenderNodes = numRenderNodes;
71
72 mTiles.clear();
73
74 SCOPED_MEM(arena);
75
76 unsigned numTilesX = ((viewport.mMaxX) >> 3) - ((viewport.mMinX) >> 3) + 1;
77 unsigned numTilesY = ((viewport.mMaxY) >> 3) - ((viewport.mMinY) >> 3) + 1;
78 unsigned numTiles = numTilesX * numTilesY;
79 MNRY_ASSERT(numTiles);
80
81 mTileIndices.reset(new uint32_t[numTiles]);
82
83 // under multi-machine context, each mcrt computation uses different random seed
84 generateTileIndices(arena, numTilesX, numTilesY, mTileIndices.get(), uint32_t(mRenderNodeIdx));
85
86#ifdef DEBUG
87
88 // Validate data:
89 uint32_t *tileIndicesCopy = arena->allocArray<uint32_t>(numTiles, CACHE_LINE_SIZE);
90 memcpy(tileIndicesCopy, mTileIndices.get(), numTiles * sizeof(uint32_t));
91 std::sort(tileIndicesCopy, tileIndicesCopy + numTiles);
92 for (unsigned i = 0; i < numTiles; ++i) {
93 MNRY_ASSERT(tileIndicesCopy[i] == i);
94 }
95
96#endif
97
98 // Generate actual tiles from the tileIndices.
99 mTiles.resize(numTiles);
100
101 unsigned baseX = viewport.mMinX & ~0x07;
102 unsigned baseY = viewport.mMinY & ~0x07;
103
104 for (unsigned i = 0; i < numTiles; ++i) {
105
106 unsigned tileIdx = mTileIndices[i];
107 scene_rdl2::fb_util::Tile &tile = mTiles[tileIdx];
108
109 uint32_t x = i % numTilesX;
110 uint32_t y = i / numTilesX;
111
112 tile.mMinX = std::max((x << 3) + baseX, (unsigned)viewport.mMinX);
113 tile.mMaxX = std::min((x << 3) + baseX + 8, (unsigned)viewport.mMaxX + 1);
114
115 tile.mMinY = std::max((y << 3) + baseY, (unsigned)viewport.mMinY);
116 tile.mMaxY = std::min((y << 3) + baseY + 8, (unsigned)viewport.mMaxY + 1);
117
118 MNRY_ASSERT(tile.mMinX < (unsigned)viewport.mMaxX + 1);
119 MNRY_ASSERT(tile.mMaxX > (unsigned)viewport.mMinX);
120 MNRY_ASSERT(tile.mMinY < (unsigned)viewport.mMaxY + 1);
121 MNRY_ASSERT(tile.mMaxY > (unsigned)viewport.mMinY);
122 }
123
124 //
125 // Handle distributed rendering by getting each machine to only render
126 // a subset of the generated tiles.
127 //
128 if (numRenderNodes > 1) {
129 mTaskDistribType = taskDistribType;
130 if (mTaskDistribType == static_cast<unsigned>(scene_rdl2::rdl2::TaskDistributionType::NON_OVERLAPPED_TILE)) {
131 // unorverlapped tile distribution
132 distributeTiles(mTiles, renderNodeIdx, numRenderNodes);
133 }
134 }
135
136 return unsigned(mTiles.size());
137}

139std::unique_ptr<TileScheduler>
140TileScheduler::create(TileScheduler::Type type)
141{
142 std::unique_ptr<TileScheduler> tileScheduler;
143
144 switch (type) {
145 case TOP: tileScheduler.reset(new TopTileScheduler); break;
146 case BOTTOM: tileScheduler.reset(new BottomTileScheduler); break;
147 case LEFT: tileScheduler.reset(new LeftTileScheduler); break;
148 case RIGHT: tileScheduler.reset(new RightTileScheduler); break;
149 case MORTON: tileScheduler.reset(new MortonTileScheduler); break;
150 case RANDOM: tileScheduler.reset(new RandomTileScheduler); break;
151 case SPIRAL_SQUARE: tileScheduler.reset(new SpiralSquareTileScheduler); break;
152 case SPIRAL_RECT: tileScheduler.reset(new SpiralRectTileScheduler); break;
153 default:
154 MNRY_ASSERT(0);
155 }
156
157 return tileScheduler;
158}

172void
173BottomTileScheduler::generateTileIndices(scene_rdl2::alloc::Arena *arena,
174 unsigned numTilesX, unsigned numTilesY,
175 uint32_t *tileIndices, uint32_t seed) const
176{
177 unsigned tileIdx = 0;
178 for (unsigned y = 0; y < numTilesY; ++y) {
179 if (y & 1) {
180 // Odd rows go right to left:
181 for (int x = numTilesX - 1; x >= 0; --x, ++tileIdx) {
182 tileIndices[y * numTilesX + unsigned(x)] = tileIdx;
183 }
184 } else {
185 // Even rows go left to right:
186 for (unsigned x = 0; x < numTilesX; ++x, ++tileIdx) {
187 tileIndices[y * numTilesX + x] = tileIdx;
188 }
189 }
190 }
191
192 MNRY_ASSERT(tileIdx == numTilesX * numTilesY);
193}

226void
227MortonTileScheduler::generateTileIndices(scene_rdl2::alloc::Arena *arena,
228 unsigned numTilesX, unsigned numTilesY,
229 uint32_t *tileIndices, uint32_t seed) const
230{
231 unsigned totalTiles = numTilesX * numTilesY;
232
233 unsigned pow2TilesX = scene_rdl2::util::roundUpToPowerOfTwo(numTilesX);
234 unsigned pow2TilesY = scene_rdl2::util::roundUpToPowerOfTwo(numTilesY);
235 unsigned totalPow2Tiles = pow2TilesX * pow2TilesY;
236
237 SCOPED_MEM(arena);
238
239 unsigned *swizzleRemap = arena->allocArray<unsigned>(totalPow2Tiles);
240 memset(swizzleRemap, 0xff, sizeof(unsigned) * totalPow2Tiles);
241
242 unsigned tilesFound = 0;
243 for (unsigned tileY = 0; tileY < pow2TilesY; ++tileY) {
244 if (tileY >= numTilesY) {
245 continue;
246 }
247
248 for (unsigned tileX = 0; tileX < pow2TilesX; ++tileX) {
249 if (tileX >= numTilesX) {
250 continue;
251 }
252 swizzleRemap[scene_rdl2::util::convertCoordToSwizzledIndex(tileX, tileY,
253 pow2TilesX, pow2TilesY)] = tilesFound++;
254 }
255 }
256
257 unsigned tilesPlaced = 0;
258 for (unsigned i = 0; i < totalPow2Tiles; ++i) {
259 if (swizzleRemap[i] != unsigned(-1)) {
260 MNRY_ASSERT(swizzleRemap[i] < totalTiles);
261 tileIndices[swizzleRemap[i]] = tilesPlaced++;
262 if (tilesPlaced == totalTiles) {
263 break;
264 }
265 }
266 }
267}

269void
270RandomTileScheduler::generateTileIndices(scene_rdl2::alloc::Arena *arena,
271 unsigned numTilesX, unsigned numTilesY,
272 uint32_t *tileIndices, uint32_t seed) const
273{
274 unsigned totalTiles = numTilesX * numTilesY;
275
276 for (unsigned i = 0; i < totalTiles; ++i) {
277 tileIndices[i] = i;
278 }
279
280 // Important, take the seed into account so results are deterministic.
281 scene_rdl2::util::Random rng(seed);
282 std::shuffle(tileIndices, tileIndices + totalTiles, rng);
283}

285void
286SpiralSquareTileScheduler::generateTileIndices(scene_rdl2::alloc::Arena *arena,
287 unsigned numTilesX, unsigned numTilesY,
288 uint32_t *tileIndices, uint32_t seed) const
289{
290 unsigned maxSide = std::max(numTilesX, numTilesY);
291 unsigned maxSideSquared = maxSide * maxSide;
292 unsigned numTiles = numTilesX * numTilesY;
293
294 SCOPED_MEM(arena);
295
296 unsigned *tempIndices = arena->allocArray<unsigned>(maxSideSquared);
297
298 // Generate a square tiling pattern which surrounds our desired viewport.
299 SpiralRectTileScheduler scheduler;
300 scheduler.generateTileIndices(arena, maxSide, maxSide, tempIndices, seed);
301
302 // Trivial case, the viewport is square.
303 if (numTilesX == numTilesY) {
304 MNRY_ASSERT(maxSideSquared == numTiles);
305 memcpy(tileIndices, tempIndices, sizeof(unsigned) * numTiles);
306 return;
307 }
308
309 //
310 // Extract out the tiles we're interested in.
311 //
312 unsigned minX, maxX, minY, maxY;
313
314 if (numTilesX > numTilesY) {
315 minX = 0;
316 maxX = numTilesX;
317 minY = (maxSide - numTilesY) / 2;
318 maxY = minY + numTilesY;
319 } else {
320 minX = (maxSide - numTilesX) / 2;
321 maxX = minX + numTilesX;
322 minY = 0;
323 maxY = numTilesY;
324 }
325
326 struct UnsortedTile
327 {
328 unsigned mCounter; // Incrementing counter for later index compaction.
329 unsigned mOldIndex;
330 unsigned mNewIndex;
331 };
332
333 UnsortedTile *tiles = arena->allocArray<UnsortedTile>(numTiles);
334
335 unsigned dstIdx = 0;
336 for (unsigned tileY = 0; tileY < maxSide; ++tileY) {
337 if (tileY >= minY && tileY < maxY) {
338 for (unsigned tileX = 0; tileX < maxSide; ++tileX) {
339 if (tileX >= minX && tileX < maxX) {
340 tiles[dstIdx].mCounter = dstIdx;
341 tiles[dstIdx].mOldIndex = tempIndices[tileY * maxSide + tileX];
342 ++dstIdx;
343 }
344 }
345 }
346 }
347
348 MNRY_ASSERT(dstIdx == numTiles);
349
350 //
351 // Compact indices.
352 //
353
354 // Sort so that old indices are in ascending order.
355 std::sort(tiles, tiles + numTiles, [](const UnsortedTile &a, const UnsortedTile &b) -> bool
356 {
357 return a.mOldIndex < b.mOldIndex;
358 });
359
360 // Patch in new indices.
361 for (unsigned i = 0; i < numTiles; ++i) {
362 tiles[i].mNewIndex = i;
363 }
364
365 // Revert to original ordering.
366 std::sort(tiles, tiles + numTiles, [](const UnsortedTile &a, const UnsortedTile &b) -> bool
367 {
368 return a.mCounter < b.mCounter;
369 });
370
371 // Copy indices out to dst array.
372 for (unsigned i = 0; i < numTiles; ++i) {
373 tileIndices[i] = tiles[i].mNewIndex;
374 }
375}

377void
378SpiralRectTileScheduler::generateTileIndices(scene_rdl2::alloc::Arena *arena,
379 unsigned numTilesX, unsigned numTilesY,
380 uint32_t *tileIndices, uint32_t seed) const
381{
382 enum
383 {
384 LEFT,
385 DOWN,
386 RIGHT,
387 UP,
388 NUM_DIRS,
389 };
390
391 // Start at an out corner of the screen and work inwards. We'll reverse the ordering
392 // later in this function.
393 unsigned numTiles = numTilesX * numTilesY;
394 unsigned currX = numTilesX - 1;
395 unsigned currY = numTilesY - 1;
396 unsigned dir = (numTilesX >= numTilesY) ? LEFT : DOWN;
397
398 unsigned currTileIdx = 0;
399
400 // Mark all tiles as need to be filled.
401 memset(tileIndices, 0xff, sizeof(uint32_t) * numTiles);
402
403 while (1) {
404
405 unsigned dstIdx = numTilesX * currY + currX;
406 MNRY_ASSERT(dstIdx < numTiles);
407
408 if (tileIndices[dstIdx] == 0xffffffff) {
409 // Here is where we reverse the ordering so we're starting from the inside
410 // as opposed to the outside.
411 tileIndices[dstIdx] = numTiles - currTileIdx - 1;
412 if (++currTileIdx == numTiles) {
413 break;
414 }
415 }
416
417 //
418 // Figure out where we should place next tile.
419 //
420 switch (dir) {
421 case LEFT:
422 if (currX == 0 ||
423 tileIndices[numTilesX * currY + currX - 1] != 0xffffffff) {
424 dir = DOWN;
425 } else {
426 --currX;
427 }
428 break;
429
430 case DOWN:
431 if (currY == 0 ||
432 tileIndices[numTilesX * (currY - 1) + currX] != 0xffffffff) {
433 dir = RIGHT;
434 } else {
435 --currY;
436 }
437 break;
438
439 case RIGHT:
440 if (currX == numTilesX - 1 ||
441 tileIndices[numTilesX * currY + currX + 1] != 0xffffffff) {
442 dir = UP;
443 } else {
444 ++currX;
445 }
446 break;
447
448 case UP:
449 if (currY == numTilesY - 1 ||
450 tileIndices[numTilesX * (currY + 1) + currX] != 0xffffffff) {
451 dir = LEFT;
452 } else {
453 ++currY;
454 }
455 break;
456
457 default:
458 MNRY_ASSERT(0);
459 }
460 }
461}
