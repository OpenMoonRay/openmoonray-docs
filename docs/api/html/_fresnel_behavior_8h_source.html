20// Abstract base class. Objects of this type serve to describe the Fresnel
21// behavior of a lobe, and to hold the construction arguments for the
22// underlying Fresnel implementation
23class FresnelBehavior
24{
25public:
26 // Describes the class of Fresnel behavior. We can apply different
27 // behaviors when combining lobes based on knowing if a FresnelBehavior
28 // is dielectric or conductor. Dielectrics transmit light, whereas
29 // conductors do not.
30 enum class Class
31 {
32 Dielectric,
33 Conductor,
34 };
