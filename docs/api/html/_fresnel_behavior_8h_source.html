20// Abstract base class. Objects of this type serve to describe the Fresnel
21// behavior of a lobe, and to hold the construction arguments for the
22// underlying Fresnel implementation
23class FresnelBehavior
24{
25public:
26 // Describes the class of Fresnel behavior. We can apply different
27 // behaviors when combining lobes based on knowing if a FresnelBehavior
28 // is dielectric or conductor. Dielectrics transmit light, whereas
29 // conductors do not.
30 enum class Class
31 {
32 Dielectric,
33 Conductor,
34 };
35
36 FresnelBehavior() =default;
37
38 virtual ~FresnelBehavior() =0;
39
40 virtual Class getClass() const =0;
41
42 template <typename T>
43 bool isType() const { return (typeid(T) == typeid(*this)); }
44
45 FresnelBehavior(const FresnelBehavior& other) =delete;
46 FresnelBehavior& operator=(const FresnelBehavior& other) =delete;
47};
48
49// Represents physical dielectric Fresnel behavior
50class Dielectric : public FresnelBehavior
51{
52public:
53 Dielectric(float etaI, float etaT) :
54 mEtaI(etaI),
55 mEtaT(etaT) {}
56
57 ~Dielectric() override {}
58
59 Class getClass() const override { return Class::Dielectric; }
60
61 finline float getEtaI() const { return mEtaI; }
62 finline float getEtaT() const { return mEtaT; }
63
64 void setEtaI(float etaI) { mEtaI = etaI; }
65 void setEtaT(float etaT) { mEtaI = etaT; }
66
67private:
68 float mEtaI;
69 float mEtaT;
70};

72// Represents physical conductor Fresnel behavior. Uses the "Artist
73// Friendly Metallic Fresnel" parameterization proposed by Ole Gulbrandsen
74// in Journal of Computer Graphics Techniques Vol. 3, No. 4, 2014
75// http://jcgt.org
76class Conductor : public FresnelBehavior
77{
78public:
79 Conductor(const scene_rdl2::math::Color& reflectivity,
80 const scene_rdl2::math::Color& edgeTint) :
81 mReflectivity(reflectivity),
82 mEdgeTint(edgeTint) {}
83
84 ~Conductor() override {}
85
86 Class getClass() const override { return Class::Conductor; }
87
88 finline const scene_rdl2::math::Color& getReflectivity() const { return mReflectivity; }
89 finline const scene_rdl2::math::Color& getEdgeTint() const { return mEdgeTint; }
90
91private:
92 math::Color mReflectivity;
93 math::Color mEdgeTint;
