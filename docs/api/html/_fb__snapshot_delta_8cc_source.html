28bool
29Fb::snapshotDelta(Fb &dstFb, FbActivePixels &dstActivePixels, const bool coarsePass) const
30//
31// dstFb : tiled format
32// this : tiled format
33//
34// return false if this and dstFb has different resolusion.
35// const bool coarsePass : only used by activePixels record logic.
36// You don't need to set coarsePass argument if you don't record.
37//
38{
39 if (dstFb.getWidth() != getWidth() || dstFb.getHeight() != getHeight()) {
40 return false; // error
41 }
42
43 //
44 // do snapshot and result ActivePixelMask goes to dstActivePixels
45 //
46# ifdef SNAPSHOT_DELTA_TIMING_TEST
47 static rec_time::RecTimeLog recTimeSnapshotDeltaLog;
48 rec_time::RecTime recTime;
49 recTime.start();
50# endif // end SNAPSHOT_DELTA_TIMING_TEST
51
52 std::vector<int> doSnapshotTbl;
53
54 dstActivePixels.init(dstFb.getWidth(),
55 dstFb.getHeight()); // init pixelInfo,heatMap,weightBuffer,renderOutput
56 doSnapshotTbl.push_back(0);
57
58 if (mPixelInfoStatus) {
59 dstActivePixels.initPixelInfo();
60 dstFb.setupPixelInfo(nullptr,
61 getPixelInfoName()); // setup memory if needed otherwise not clear data itself
62 doSnapshotTbl.push_back(1);
63 } else {
64 dstFb.resetPixelInfo();
65 }
66 if (mHeatMapStatus) {
67 dstActivePixels.initHeatMap();
68 dstFb.setupHeatMap(nullptr,
69 getHeatMapName()); // setup memory if needed otherwise not clear data itself
70 doSnapshotTbl.push_back(2);
71 } else {
72 dstFb.resetHeatMap();
73 }
74 if (mWeightBufferStatus) {
75 dstActivePixels.initWeightBuffer();
76 dstFb.setupWeightBuffer(nullptr,
77 getWeightBufferName()); // setup mem if needed otherwise not clear data itself
78 doSnapshotTbl.push_back(3);
79 } else {
80 dstFb.resetWeightBuffer();
81 }
82 if (mRenderBufferOddStatus) {
83 dstActivePixels.initRenderBufferOdd();
84 dstFb.setupRenderBufferOdd(nullptr);
85 doSnapshotTbl.push_back(4);
86 } else {
87 dstFb.resetRenderBufferOdd();
88 }
89 if (mRenderOutputStatus) {
90 doSnapshotTbl.push_back(5);
91 } else {
92 dstFb.resetRenderOutput();
93 }
94
95# ifdef SINGLE_THREAD
96 for (size_t i = 0; i < doSnapshotTbl.size(); i++) {
97 switch (doSnapshotTbl[i]) {
98 case 0 :
99 snapshotDeltaBeauty(dstFb, dstActivePixels.getActivePixels(), coarsePass);
100 break;
101 case 1 :
102 snapshotDeltaPixelInfo(dstFb, dstActivePixels.getActivePixelsPixelInfo());
103 break;
104 case 2 :
105 snapshotDeltaHeatMap(dstFb, dstActivePixels.getActivePixelsHeatMap());
106 break;
107 case 3 :
108 snapshotDeltaWeightBuffer(dstFb, dstActivePixels.getActivePixelsWeightBuffer());
109 break;
110 case 4 :
111 snapshotDeltaRenderBufferOdd(dstFb, dstActivePixels.getActivePixelsRenderBufferOdd());
112 break;
113 case 5 :
114 snapshotDeltaRenderOutput(dstFb, dstActivePixels);
115 break;
116 }
117 }
118# else // else SINGLE_THREAD
119 tbb::parallel_for((unsigned)0, (unsigned)doSnapshotTbl.size(), [&](unsigned id) {
120 switch (doSnapshotTbl[id]) {
121 case 0 :
122 snapshotDeltaBeauty(dstFb, dstActivePixels.getActivePixels(), coarsePass);
123 break;
124 case 1 :
125 snapshotDeltaPixelInfo(dstFb, dstActivePixels.getActivePixelsPixelInfo());
126 break;
127 case 2 :
128 snapshotDeltaHeatMap(dstFb, dstActivePixels.getActivePixelsHeatMap());
129 break;
130 case 3 :
131 snapshotDeltaWeightBuffer(dstFb, dstActivePixels.getActivePixelsWeightBuffer());
132 break;
133 case 4 :
134 snapshotDeltaRenderBufferOdd(dstFb, dstActivePixels.getActivePixelsRenderBufferOdd());
135 break;
136 case 5 :
137 snapshotDeltaRenderOutput(dstFb, dstActivePixels);
138 break;
139 }
140 });
141# endif // end !SINGLE_THREAD
142
143# ifdef SNAPSHOT_DELTA_TIMING_TEST
144 recTimeSnapshotDeltaLog.add(recTime.end());
145 if (recTimeSnapshotDeltaLog.getTotal() == 24) {
146 std::cerr << ">> Fb.cc snapshotDelta() ave:"
147 << recTimeSnapshotDeltaLog.getAverage() * 1000.0f << " ms" << std::endl;
148 recTimeSnapshotDeltaLog.reset();
149 }
150# endif // end SNAPSHOT_DELTA_TIMING_TEST
151
152 return true;
153}

181bool
182Fb::snapshotDeltaRecDump(const std::string &fileName)
183{
184 if (!mActivePixelsArray) {
185 return false; // not snapshotDeltaRecStart() yet
186 }
187 if (mActivePixelsArray->isStart()) {
188 return false; // not stopped yet
189 }
190 if (!mActivePixelsArray->size()) {
191 return false; // data is empty
192 }
193
194 std::string data;
195 mActivePixelsArray->encode(data);
196
197 // So far only progmcrt_merge computation is used and snapshotDeltaRecDump is calls
198 // from progmcrt_merge computation only. So we always added special ".merge" extension
199 // for output file.
200 std::string outName = fileName + ".merge";
201
202 std::ofstream fout(outName, std::ios::out | std::ios::binary | std::ios::trunc);
203 if (!fout) {
204 std::ostringstream ostr;
205 ostr << ">> Fb.cc snapshotDeltaRecDump() Can't open file:" << outName;
206 logging::Logger::error(ostr.str());
207 return false;
208 }
209
210 fout.write((const char *)data.data(), data.size());
211 if (!fout) {
212 std::ostringstream ostr;
213 ostr << ">> Fb.cc snapshotDeltaRecDump() Can't write data." << " file:" << outName;
214 logging::Logger::error(ostr.str());
215 return false;
216 }
217
218 fout.close();
219
220 mActivePixelsArray = nullptr;
221
222 logging::Logger::error(">> Fb.cc snapshotDeltaRecDump() done");
223
224 return true;
225}

230template <typename T, typename F>
231void
232Fb::snapshotDeltaMain(ActivePixels &dstActivePixels,
233 T *dst,
234 unsigned int *dstNumSample,
235 const ActivePixels &srcActivePixels,
236 const T *src,
237 const unsigned int *srcNumSample,
238 ActivePixels &outActivePixels,
239 F snapshotTileFunc) const
240{
241 //
242 // We don't need to reset outActivePixels because all tile mask will be set anyway.
243 //
244 for (unsigned tileId = 0; tileId < getTileTotal(); ++tileId) {
245 uint64_t srcTileMask = srcActivePixels.getTileMask(tileId);
246
247 uint64_t activePixelMask = 0x0;
248 if (srcTileMask) {
249 T *__restrict dstTile = dst + (tileId << 6);
250 const T *__restrict srcTile = src + (tileId << 6);
251 unsigned int *__restrict dstTileNumSample = dstNumSample + (tileId << 6);
252 const unsigned int *__restrict srcTileNumSample = srcNumSample + (tileId << 6);
253 uint64_t dstTileMask = dstActivePixels.getTileMask(tileId);
254
255 activePixelMask = snapshotTileFunc(dstTile,
256 dstTileNumSample,
257 dstTileMask,
258 srcTile,
259 srcTileNumSample,
260 srcTileMask);
261
262 dstActivePixels.orOp(tileId, activePixelMask);
263 }
264 outActivePixels.setTileMask(tileId, activePixelMask);
265 }
266}
267#else // else SINGLE_THREAD
268template <typename T, typename F>
269void
270Fb::snapshotDeltaMain(ActivePixels &dstActivePixels,
271 T *dst,
272 unsigned int *dstNumSample,
273 const ActivePixels &srcActivePixels,
274 const T *src,
275 const unsigned int *srcNumSample,
276 ActivePixels &outActivePixels,
277 F snapshotTileFunc) const
278{
279 if (!getTileTotal()) return;
280 //
281 // We don't need to reset outActivePixels because all tile mask will be set anyway.
282 //
283 tbb::blocked_range<size_t> range(0, getTileTotal(), 64);
284 tbb::parallel_for(range, [&](const tbb::blocked_range<size_t> &tileRange) {
285 for (size_t tileId = tileRange.begin(); tileId < tileRange.end(); ++tileId) {
286 uint64_t srcTileMask = srcActivePixels.getTileMask(tileId);
287
288 uint64_t activePixelMask = 0x0;
289 if (srcTileMask) {
290 T *__restrict dstTile = dst + (tileId << 6);
291 const T *__restrict srcTile = src + (tileId << 6);
292 unsigned int *__restrict dstTileNumSample = dstNumSample + (tileId << 6);
293 const unsigned int *__restrict srcTileNumSample = srcNumSample + (tileId << 6);
294 uint64_t dstTileMask = dstActivePixels.getTileMask(tileId);
295
296 activePixelMask = snapshotTileFunc(dstTile,
297 dstTileNumSample,
298 dstTileMask,
299 srcTile,
300 srcTileNumSample,
301 srcTileMask);
302
303 dstActivePixels.orOp(tileId, activePixelMask);
304 }
305 outActivePixels.setTileMask(tileId, activePixelMask);
306 }
307 });
308}

314Fb::snapshotAllTileLoop(Fb &dstFb, F func) const
315{
316 for (unsigned tileId = 0; tileId < getTileTotal(); ++tileId) {
317 func(tileId);
318 }

323Fb::snapshotAllTileLoop(Fb &dstFb, F func) const
324{

327 tbb::parallel_for(range, [&](const tbb::blocked_range<size_t> &tileRange) {
328 for (size_t tileId = tileRange.begin(); tileId < tileRange.end(); ++tileId) {
329 func(tileId);
330 }
331 });

336template <typename F>
337void
338Fb::snapshotAllActiveAov(Fb &dstFb, F activeAovFunc) const
339{
340 for (auto &itr : mRenderOutput) {
341 const FbAovShPtr &srcFbAov = itr.second;
342 if (!srcFbAov->getStatus()) continue; // skip non active aov
343
344 // real data AOV buffer or Reference type
345 const std::string &aovName = srcFbAov->getAovName();
346 FbAovShPtr &dstFbAov = dstFb.getAov(aovName); // generate if needed
347
348 activeAovFunc(srcFbAov, dstFbAov);
349 }
350}
351#else // else SINGLE_THREAD
352template <typename F>
353void
354Fb::snapshotAllActiveAov(Fb &dstFb, F activeAovFunc) const
355{
356 std::vector<std::string> activeAovNameArray;
357 for (const auto &itr : mRenderOutput) {
358 const FbAovShPtr &srcFbAov = itr.second;
359 if (!srcFbAov->getStatus()) continue; // skip non active aov
360
361 // real data AOV buffer or Reference type
362 activeAovNameArray.push_back(srcFbAov->getAovName());
363 }
364 if (!activeAovNameArray.size()) return;
365
366 tbb::blocked_range<size_t> range(0, activeAovNameArray.size());
367 tbb::parallel_for(range, [&](const tbb::blocked_range<size_t> &r) {
368 for (size_t activeAovNameId = r.begin(); activeAovNameId < r.end(); ++activeAovNameId) {
369 const std::string &aovName = activeAovNameArray[activeAovNameId];
370 if (!findAov(aovName)) {
371 std::ostringstream ostr;
372 ostr << ">> ============ Fb.h findAov failed2. aovName:>" << aovName << "<";
373 logging::Logger::error(ostr.str());
374 continue;
375 }
376 const FbAovShPtr &srcFbAov = mRenderOutput.at(aovName);
377 FbAovShPtr dstFbAov = dstFb.getAov(aovName); // generate if needed
378
379 activeAovFunc(srcFbAov, dstFbAov);
380 }
381 });
382}
383#endif // end !SINGLE_THREAD

388Fb::snapshotDeltaBeauty(Fb &dstFb, ActivePixels &dstActivePixels, const bool coarsePass) const
389{
390 snapshotDeltaMain
391 (dstFb.mActivePixels, dstFb.mRenderBufferTiled.getData(), dstFb.mNumSampleBufferTiled.getData(),

412void
413Fb::snapshotDeltaPixelInfo(Fb &dstFb, ActivePixels &dstActivePixels) const
414{
415 // We use snapshotAllTileLoop instead of snapshotDeltaMain because we don't have associated numSample info.
416 snapshotAllTileLoop(dstFb, [&](unsigned tileId) {
417 PixelInfo *__restrict dst = dstFb.mPixelInfoBufferTiled.getData() + (tileId << 6);
418 const PixelInfo *__restrict src = mPixelInfoBufferTiled.getData() + (tileId << 6);
419
420 uint64_t currDstTileMask = dstFb.mActivePixelsPixelInfo.getTileMask(tileId);
421 uint64_t currSrcTileMask = mActivePixelsPixelInfo.getTileMask(tileId);
422
423 uint64_t activePixelMask =
424 fb_util::SnapshotUtil::snapshotTilePixelInfo(reinterpret_cast<uint32_t *>(dst),
425 currDstTileMask,
426 reinterpret_cast<const uint32_t *>(src),
427 currSrcTileMask);
428
429 dstFb.mActivePixelsPixelInfo.orOp(tileId,
430 activePixelMask); // update activePixel info (OR operation)
431 dstActivePixels.setTileMask(tileId, activePixelMask);
432 });
433}

502void
503Fb::snapshotDeltaRenderOutput(Fb &dstFb, FbActivePixels &dstFbActivePixels) const
504// This function is used on progmcrt_merge computation
505{
506 //
507 // stageA : Try to make snapshot for active AOV buffers first and save result
508 // inside dstFb/dstFbActivePixels
509 //
510 snapshotAllActiveAov(dstFb, [&](const FbAovShPtr &srcFbAov, FbAovShPtr &dstFbAov) {
511 // activeAovFunc
512 if (srcFbAov->getReferenceType() == FbReferenceType::UNDEF) {
513 // Non-Reference type buffer
514 // We should do snapshot
515
516 // We always need to process numSampleData on merge computation
517 constexpr bool storeNumSampleData = true;
518
519 // need to setup default value before call setup()
520 dstFbAov->setDefaultValue(srcFbAov->getDefaultValue());
521 dstFbAov->setup(nullptr,
522 srcFbAov->getFormat(),
523 srcFbAov->getWidth(),
524 srcFbAov->getHeight(), // setup memory and clean if needed
525 storeNumSampleData);
526
527 // setup closestFilter condition
528 dstFbAov->setClosestFilterStatus(srcFbAov->getClosestFilterStatus());
529
530 // generate if needed : MTsafe
531 FbActivePixels::FbActivePixelsAovShPtr &dstFbActivePixelsAov =
532 dstFbActivePixels.getAov(dstFbAov->getAovName());
533 dstFbActivePixelsAov->init(srcFbAov->getWidth(), srcFbAov->getHeight());
534
535 switch (srcFbAov->getFormat()) {
536 case VariablePixelBuffer::FLOAT :
537 snapshotDeltaMain
538 (dstFbAov->getActivePixels(),
539 dstFbAov->getBufferTiled().getFloatBuffer().getData(),
540 dstFbAov->getNumSampleBufferTiled().getData(),
541 srcFbAov->getActivePixels(),
542 srcFbAov->getBufferTiled().getFloatBuffer().getData(),
543 srcFbAov->getNumSampleBufferTiled().getData(),
544 dstFbActivePixelsAov->getActivePixels(),
545 [](float *dstTile,
546 unsigned int *dstTileNumSample,
547 uint64_t dstTileMask,
548 const float *srcTile,
549 const unsigned int *srcTileNumSample,
550 uint64_t srcTileMask) -> uint64_t {
551 // snapshotTileFunc
552 return fb_util::SnapshotUtil::snapshotTileFloatNumSample
553 (reinterpret_cast<uint32_t *>(dstTile),
554 dstTileNumSample,
555 dstTileMask,
556 reinterpret_cast<const uint32_t *>(srcTile),
557 srcTileNumSample,
558 srcTileMask);
559 });
560 break;
561 case VariablePixelBuffer::FLOAT2 :
562 snapshotDeltaMain
563 (dstFbAov->getActivePixels(),
564 dstFbAov->getBufferTiled().getFloat2Buffer().getData(),
565 dstFbAov->getNumSampleBufferTiled().getData(),
566 srcFbAov->getActivePixels(),
567 srcFbAov->getBufferTiled().getFloat2Buffer().getData(),
568 srcFbAov->getNumSampleBufferTiled().getData(),
569 dstFbActivePixelsAov->getActivePixels(),
570 [](math::Vec2f *dstTile,
571 unsigned int *dstTileNumSample,
572 uint64_t dstTileMask,
573 const math::Vec2f *srcTile,
574 const unsigned int *srcTileNumSample,
575 uint64_t srcTileMask) -> uint64_t {
576 // snapshotTileFunc
577 return fb_util::SnapshotUtil::snapshotTileFloat2NumSample
578 (reinterpret_cast<uint32_t *>(dstTile),
579 dstTileNumSample,
580 dstTileMask,
581 reinterpret_cast<const uint32_t *>(srcTile),
582 srcTileNumSample,
583 srcTileMask);
584 });
585 break;
586 case VariablePixelBuffer::FLOAT3 :
587 snapshotDeltaMain
588 (dstFbAov->getActivePixels(),
589 dstFbAov->getBufferTiled().getFloat3Buffer().getData(),
590 dstFbAov->getNumSampleBufferTiled().getData(),
591 srcFbAov->getActivePixels(),
592 srcFbAov->getBufferTiled().getFloat3Buffer().getData(),
593 srcFbAov->getNumSampleBufferTiled().getData(),
594 dstFbActivePixelsAov->getActivePixels(),
595 [](math::Vec3f *dstTile,
596 unsigned int *dstTileNumSample,
597 uint64_t dstTileMask,
598 const math::Vec3f *srcTile,
599 const unsigned int *srcTileNumSample,
600 uint64_t srcTileMask) -> uint64_t {
601 // snapshotTileFunc
602 return fb_util::SnapshotUtil::snapshotTileFloat3NumSample
603 (reinterpret_cast<uint32_t *>(dstTile),
604 dstTileNumSample,
605 dstTileMask,
606 reinterpret_cast<const uint32_t *>(srcTile),
607 srcTileNumSample,
608 srcTileMask);
609 });
610 break;
611 case VariablePixelBuffer::FLOAT4 :
612 snapshotDeltaMain
613 (dstFbAov->getActivePixels(),
614 dstFbAov->getBufferTiled().getFloat4Buffer().getData(),
615 dstFbAov->getNumSampleBufferTiled().getData(),
616 srcFbAov->getActivePixels(),
617 srcFbAov->getBufferTiled().getFloat4Buffer().getData(),
618 srcFbAov->getNumSampleBufferTiled().getData(),
619 dstFbActivePixelsAov->getActivePixels(),
620 [](math::Vec4f *dstTile,
621 unsigned int *dstTileNumSample,
622 uint64_t dstTileMask,
623 const math::Vec4f *srcTile,
624 const unsigned int *srcTileNumSample,
625 uint64_t srcTileMask) -> uint64_t {
626 // snapshotTileFunc
627 return fb_util::SnapshotUtil::snapshotTileFloat4NumSample
628 (reinterpret_cast<uint32_t *>(dstTile),
629 dstTileNumSample,
630 dstTileMask,
631 reinterpret_cast<const uint32_t *>(srcTile),
632 srcTileNumSample,
633 srcTileMask);
634 });
635 break;
636 default :
637 break;
638 }
639 } else {
640 // Reference type buffer
641 // Just setup fbAov w/ referenceType information. We don't have any actual data
642 // for reference buffer type inside fbAov.
643 FbActivePixels::FbActivePixelsAovShPtr &dstFbActivePixelsAov =
644 dstFbActivePixels.getAov(dstFbAov->getAovName());
645 dstFbActivePixelsAov->init(srcFbAov->getReferenceType());
646
647 dstFbAov->setup(srcFbAov->getReferenceType());
648 }
649 });
650
651 //
652 // StageB : Try to clean up dst unused AOV buffers
653 //
654 unsigned totalActiveAov = 0;
655 for (auto &itr : dstFb.mRenderOutput) {
656 FbAovShPtr &dstFbAov = itr.second;
657 if (!dstFbAov->getStatus()) continue; // skip non active aov
658
659 if (mRenderOutput.find(dstFbAov->getAovName()) == mRenderOutput.end()) {
660 // We can not find this aov buffer inside src.
