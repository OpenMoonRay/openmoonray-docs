37uint64
38TestRender::renderAndCompare(const rdl2::Material *mat,
39 int testNum,
40 int width,
41 int height,
42 int raysPerPixel,
43 bool isIndirect)
44{
45 // allocate results buffer and call the test renderer
46 const int bufferSize = 3 * width * height;
47 uint8_t results[bufferSize];
48
49 double start = time::getTime();
50 uint64 shaderTicks = render(mat, results, width, height, raysPerPixel, false, isIndirect);
51 double end = time::getTime();
52
53 std::cerr << "render test " << testNum << " render time = "
54 << end - start << '\n';
55 std::cerr << "render test " << testNum << " shader ticks = "
56 << shaderTicks << '\n';
57
58 // write the results as a ppm
59 {
60 std::ofstream ppm;
61 std::stringstream filename;
62 filename << "/tmp/TestRender_result" << testNum << ".ppm";
63 ppm.open(filename.str().c_str());
64 ppm << "P6\n";
65 ppm << width << " " << height << std::endl;
66 const int maxPixel = 255; // 8 bit gamma corrected color
67 ppm << maxPixel << std::endl;
68 ppm.write((const char *) results, bufferSize * sizeof(int8_t));
69 ppm.close();
70 }
71
72 // compare with the canonical
73 std::ifstream ppm;
74 uint8_t canonical[bufferSize];
75 std::stringstream filename;
76 filename << "ref/TestRender_canonical" << testNum << ".ppm";
77 ppm.open(filename.str().c_str());
78 ppm.getline((char *) canonical, bufferSize); // P6\0
79 ppm.getline((char *) canonical, bufferSize); // width height
80 ppm.getline((char *) canonical, bufferSize); // max white
81 ppm.read((char *) canonical, bufferSize);
82 // allow +/-1 code of tolerance
83 uint8_t *resPtr = results;
84 uint8_t *canPtr = canonical;
85
86 for (int i = 0; i < bufferSize; ++i) {
87 CPPUNIT_ASSERT(abs(*resPtr - *canPtr) <= 1);
88 resPtr++;
89 canPtr++;
90 }
91
92 return shaderTicks;
93}

95uint64
96TestRender::render(
97 const rdl2::Material *mat,
98 uint8_t *results,
99 int width, int height,
100 int raysPerPixel,
101 bool primeThePump,
102 bool isIndirect)
103{
104 mcrt_common::ThreadLocalState *tls = mcrt_common::getFrameUpdateTLS();
105 shading::TLState *shadingTls = MNRY_VERIFY(tls->mShadingTls.get());
106
107 // initialize for texturing support
108 shadingTls->initTexturingSupport();
109
110 // setup the test render scene options
111 ispc::Scene scene;
112 scene.mMat = (const ispc::Material *) mat;
113 scene.mWidth = width;
114 scene.mHeight = height;
115 scene.mRaysPerPixel = raysPerPixel;
116 scene.mShadingTLState = (ispc::ShadingTLState *)shadingTls;
117
118 // call ispc render function
119 if (primeThePump) {
120 ispc::render(&scene, results, isIndirect);
121 }
122
123 uint64 shaderTicks = ispc::render(&scene, results, isIndirect);
124
125 return shaderTicks;
126}
