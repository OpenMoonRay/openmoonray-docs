26template<typename GridType>
27std::vector<float>
28getGridLuminance(const amorphous::SScatterSampler& sampler,
29 const GridType& emissionGrid)
30{
31 std::vector<float> values;
32 for (auto it = emissionGrid.cbeginValueOn(); it; ++it) {
33 const openvdb::math::Coord& xyz = it.getCoord();
34 const auto rgba = sampler.incand(xyz);
35 values.push_back(luminance(Color(rgba.r(), rgba.g(), rgba.b())));
36 }
37 return values;
38}

75std::unique_ptr<EmissionDistribution>
76AmorphousVolume::computeEmissionDistribution(const scene_rdl2::rdl2::VolumeShader* volumeShader) const
77{
78 if (!mHasEmissionField || !mUseAmorphousSampler || !mAmorphousSampler.mMasterSampler) {
79 // There is no amorphous data but there is a emission from a volume shader.
80 return Primitive::computeEmissionDistribution(volumeShader);
81 }
82
83 openvdb::GridBase::ConstPtr emissionGrid = mAmorphousSampler.getIncandGrid();
84 const amorphous::SScatterSampler& scatterSampler = *mAmorphousSampler.mMasterSampler.get();
85
86 MNRY_ASSERT_REQUIRE(emissionGrid);
87 MNRY_ASSERT_REQUIRE(emissionGrid->transform().isLinear());
88
89 std::unique_ptr<EmissionDistribution> distribution;
90
91 // TODO: fix the dual usage of VdbVolume's mPrimToRender matrices, then we can pass mPrimToRender into
92 // computeEmissionDistributionImpl() in both the calls here. We can't at present because it may be being used to
93 // store something different
94 if (emissionGrid->isType<amorphous::ScalarGridType>()) {
95
96 std::vector<float> values = getGridLuminance(scatterSampler,
97 *static_cast<const amorphous::ScalarGridType*>(emissionGrid.get()));
98
99 distribution = computeEmissionDistributionImpl(
100 getRdlGeometry(),
101 *static_cast<const amorphous::ScalarGridType*>(emissionGrid.get()),
102 mRenderToPrim,
103 values,
104 volumeShader);
105
106 } else if (emissionGrid->isType<amorphous::RGBGridType>()) {
107
108 std::vector<float> values = getGridLuminance(scatterSampler,
109 *static_cast<const amorphous::RGBGridType*>(emissionGrid.get()));
110
111 distribution = computeEmissionDistributionImpl(
112 getRdlGeometry(),
113 *static_cast<const amorphous::RGBGridType*>(emissionGrid.get()),
114 mRenderToPrim,
115 values,
116 volumeShader);
117 }
118 return distribution;
119}

121Color
122AmorphousVolume::evalDensity(mcrt_common::ThreadLocalState* tls,
123 uint32_t volumeId,
124 const Vec3f& pSample) const
125{
126 const openvdb::Vec3d p(pSample[0], pSample[1], pSample[2]);
127 if (mUseAmorphousSampler) {
128 // for shadow attenuation, always use the cheapest interpolation
129 const Color density = mAmorphousSampler.extinct(tls, volumeId, p, Interpolation::POINT);
130 Color retVal = density * sampleBakedDensity(tls, volumeId, p);
131 return retVal;
132 } else {
133 const Color density = Color(mDensitySampler.eval(tls, volumeId, p, Interpolation::POINT));
134 Color retVal = density * sampleBakedDensity(tls, volumeId, p);
135 return retVal;
136 }
137}

139void
140AmorphousVolume::evalVolumeCoefficients(mcrt_common::ThreadLocalState* tls,
141 uint32_t volumeId,
142 const Vec3f& pSample,
143 Color* extinction,
144 Color* albedo,
145 Color* temperature,
146 bool highQuality) const
147{
148 const openvdb::Vec3d p(pSample[0], pSample[1], pSample[2]);
149 Interpolation mode = highQuality ? mInterpolationMode : Interpolation::POINT;
150 if (mUseAmorphousSampler) {
151 *extinction = mAmorphousSampler.extinct(tls, volumeId, p, mode);
152 *albedo = mAmorphousSampler.albedo(tls, volumeId, p, mode);
153 if (temperature) {
154 *temperature = mAmorphousSampler.incand(tls, volumeId, p, mode);
155 }
156 } else {
157 *extinction = Color(mDensitySampler.eval(tls, volumeId, p, mode));
158 *albedo = Color(1.0f);
159 if (temperature) {
160 const auto colorVector = mEmissionSampler.eval(tls, volumeId, p, Interpolation::POINT);
161 *temperature = Color(colorVector.x(), colorVector.y(), colorVector.z());
162 }
163 }
164
165 *extinction *= sampleBakedDensity(tls, volumeId, p);
166}

169AmorphousVolume::evalTemperature(mcrt_common::ThreadLocalState* tls,

182bool
183AmorphousVolume::initAmorphousSampler(const std::string& vdbFilePath,
184 const std::string& densityGridName,
185 const scene_rdl2::rdl2::Geometry& rdlGeometry,
186 openvdb::VectorGrid::Ptr velocityGrid,
187 const std::vector<int>& volumeIds)
188{
189 const float emissionSampleRate = scene_rdl2::math::clamp(
190 mVdbVolumeData->mEmissionSampleRate);
191
192 if (!mAmorphousSampler.initialize(vdbFilePath,
193 densityGridName,
194 rdlGeometry,
195 emissionSampleRate,
196 volumeIds)) {
197 return false;
198 }
199
200 mTopologyGrid = mAmorphousSampler.getDensityGrid();
201 // padding extra voxels for motion blur usage
202 if (velocityGrid) {
203 float shutterOpen, shutterClose;
204 mVdbVelocity->getShutterOpenAndClose(shutterOpen, shutterClose);
205 initMotionBlurBoundary(mTopologyGrid, velocityGrid,
206 shutterOpen, shutterClose);
207 }
208
209 mHasEmissionField = (mAmorphousSampler.getIncandGrid() != nullptr) &&
210 (!mAmorphousSampler.getIncandGrid()->empty());
211
212 return true;
213}

215bool
216AmorphousVolume::initialize(const scene_rdl2::rdl2::Geometry& rdlGeometry,
217 const scene_rdl2::rdl2::Layer* layer,
218 const VolumeAssignmentTable* volumeAssignmentTable)
219{
220 const std::string& vdbFilePath = mVdbVolumeData->mFilePath;
221 if (vdbFilePath.empty()) {
222 return false;
223 }
224
225 openvdb::io::File file(vdbFilePath);
226 openvdb::GridPtrVecPtr grids;
227 openvdb::VectorGrid::Ptr velocityGrid;
228 std::vector<int> volumeIds;
229
230 if (!initializePhase1(vdbFilePath,
231 file,
232 rdlGeometry,
233 layer,
234 volumeAssignmentTable,
235 volumeIds,
236 grids,
237 velocityGrid)) {
238 return false;
239 }
240
241 if (initAmorphousSampler(vdbFilePath,
242 mVdbVolumeData->mDensityGridName,
243 rdlGeometry,
244 velocityGrid,
245 volumeIds)) {
246 mUseAmorphousSampler = true;
247 } else {
248 rdlGeometry.warn("fallback to default grid loading solution");
249 mUseAmorphousSampler = false;
250 if (!initDensitySampler(file,
251 grids,
252 mVdbVolumeData->mDensityGridName,
253 rdlGeometry,
254 velocityGrid,
255 volumeIds)) {
256 return false;
257 }
258
259 if (!mVdbVolumeData->mEmissionGridName.empty()) {
260 if (!initEmissionSampler(file,
261 grids,
262 mVdbVolumeData->mEmissionGridName,
263 rdlGeometry,
264 velocityGrid,
265 volumeIds)) {
266 mHasEmissionField = false;
267 return false;
268 } else {
269 mHasEmissionField = !mEmissionSampler.mGrid->empty();
270 }
271 }
272 }
273
274 initializePhase2(rdlGeometry,
275 layer,
276 volumeAssignmentTable,
277 volumeIds,
278 velocityGrid);
279
280 return true;
281}
