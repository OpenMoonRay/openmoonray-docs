388///
389/// @class PrimitiveAttributeTable
390/// @brief a PrimitiveAttribute hashtable using AttributeKey as lookup index
391///
392class PrimitiveAttributeTable
393{
394public:
395 typedef std::unordered_map<AttributeKey,
396 std::vector<std::unique_ptr<PrimitiveAttributeBase>>,
397 AttributeKeyHash> map_type;
398 typedef typename map_type::key_type key_type;
399 typedef typename map_type::mapped_type mapped_type;
400 typedef typename map_type::iterator iterator;
401 typedef typename map_type::const_iterator const_iterator;
402
403 PrimitiveAttributeTable() = default;
404 ~PrimitiveAttributeTable() = default;
405 PrimitiveAttributeTable(PrimitiveAttributeTable&&) = default;
406 PrimitiveAttributeTable(const PrimitiveAttributeTable&) = delete;
407 PrimitiveAttributeTable& operator=(const PrimitiveAttributeTable&) = delete;
408
409 /// @brief submit a vector data to build a PrimitiveAttribute and stored in
410 /// PrimitiveAttributeTable
411 /// @param key TypedAttributeKey for indexing submitted primitive attribute
412 /// @param rate specify the frequency of submitted primitive attribute
413 /// @param data the content of submitted primitive attribute
414 /// @return whether the attribute is correctly submitted
415 /// into PrimitiveAttributeTable
416 template <typename T>
417 bool addAttribute(TypedAttributeKey<T> key, AttributeRate rate,
418 std::vector<T>&& data)
419 {
420 std::vector<std::unique_ptr<PrimitiveAttributeBase>> dataList;
421 dataList.emplace_back(new PrimitiveAttribute<T>(rate, std::move(data)));
422 if (mMap.find(key) == mMap.end()) {
423 return mMap.emplace(key, std::move(dataList)).second;
424 } else {
425 return false;
426 }
427 }
428
429 /// @brief submit a set of vector data to build a PrimitiveAttribute
430 /// and stored in PrimitiveAttributeTable. Each vector data in the set
431 /// stands for a time sample of this PrimitiveAttribute
432 /// @param key TypedAttributeKey for indexing submitted primitive attribute
433 /// @param rate specify the frequency of submitted primitive attribute
434 /// @param data the content of submitted primitive attribute
435 /// @return whether the attribute is correctly submitted
436 /// into PrimitiveAttributeTable
437 template <typename T>
438 bool addAttribute(TypedAttributeKey<T> key, AttributeRate rate,
439 std::vector<std::vector<T>>&& data)
440 {
441 static_assert(!std::is_integral<T>::value,
442 "integral type is not motionblur interpolatable");
443 static_assert(!is_string<T>::value,
444 "string type is not motionblur interpolatable");
445
446 MNRY_ASSERT_REQUIRE(data.size() <= 2,
447 "only support two time samples for motionblur at this moment");
448
449 std::vector<std::unique_ptr<PrimitiveAttributeBase>> dataList;
450 dataList.reserve(data.size());
451 for (size_t i = 0; i < data.size(); ++i) {
452 dataList.emplace_back(
453 new PrimitiveAttribute<T>(rate, std::move(data[i])));
454 }
455 if (mMap.find(key) == mMap.end()) {
456 return mMap.emplace(key, std::move(dataList)).second;
457 } else {
458 return false;
459 }
460 }
461
462 /// return the typed PrimitiveAttribute reference with specified key
463 template <typename T>
464 PrimitiveAttribute<T>& getAttribute(TypedAttributeKey<T> key,
465 size_t timeIndex = 0)
466 {
467 MNRY_ASSERT(timeIndex < mMap.find(key)->second.size());
468 return mMap[key][timeIndex]->template as<T>();
469 }
470
471 template <typename T>
472 const PrimitiveAttribute<T>& getAttribute(TypedAttributeKey<T> key,
473 size_t timeIndex = 0) const
474 {
475 MNRY_ASSERT(hasAttribute(key));
476 MNRY_ASSERT(timeIndex < mMap.find(key)->second.size());
477 return (mMap.find(key)->second[timeIndex])->template as<T>();
478 }

517private:
518
519 template <typename T>
520 mapped_type copyAttribute(TypedAttributeKey<T> key) const
521 {
522 mapped_type attr;
523 size_t timeSampleCount = getTimeSampleCount(key);
524 attr.reserve(timeSampleCount);
525 for (size_t t = 0; t < timeSampleCount; ++t) {
526 attr.emplace_back(new PrimitiveAttribute<T>(getAttribute(key, t)));
527 }
528 return attr;
529 }
530
531private:
532 map_type mMap;
533};
