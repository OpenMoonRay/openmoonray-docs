58bool
59DebugSamplesRecArray::save(const std::string &filename) const
60{
61 if (mMode != Mode::SAVE) return true; // just in case.
62
63 std::ofstream out(filename.c_str(), std::ios::trunc | std::ios::binary);
64 if (!out) return false;
65
66 std::string buff;
67 scene_rdl2::rdl2::ValueContainerEnq vContainerEnq(&buff);
68 vContainerEnq.enqUInt(mSizeX);
69 vContainerEnq.enqUInt(mSizeY);
70 crawlAllPixels([&](const size_t x, const size_t y) {
71 vContainerEnq.enqUInt((unsigned int)mImg[y][x].size());
72 for (unsigned i = 0; i < mImg[y][x].size(); ++i) {
73 vContainerEnq.enqVec4f(mImg[y][x][i]);
74 }
75 });
76
77 size_t size = vContainerEnq.finalize();
78 out.write((const char *)&size, sizeof(size));
79 out.write(buff.data(), size); // binary data write
80
81 return true;
82}

99bool
100DebugSamplesRecArray::load(const std::string &filename)
101//
102// So far only called from constructor. This means mImg is empty.
103//
104{
105 std::ifstream in(filename.c_str(), std::ios::binary);
106 if (!in) return false;
107
108 std::string buff;
109 size_t size;
110 in.read((char *)&size, sizeof(size));
111 buff.resize(size);
112 in.read(&buff[0], size);
113
114 scene_rdl2::rdl2::ValueContainerDeq vContainerDeq(buff.data(), size);
115 vContainerDeq.deqUInt(mSizeX);
116 vContainerDeq.deqUInt(mSizeY);
117 mImg.resize(mSizeY);
118 for (unsigned y = 0; y < mSizeY; ++y) {
119 mImg[y].resize(mSizeX);
120 for (unsigned x = 0; x < mSizeX; ++x) {
121 unsigned samples;
122 vContainerDeq.deqUInt(samples);
123 mImg[y][x].resize(samples);
124 for (unsigned i = 0; i < samples; ++i) {
125 scene_rdl2::math::Vec4f c;
126 vContainerDeq.deqVec4f(c);
127 // c *= 0.25f; // test
128 mImg[y][x][i] = c;
129 }
130 }
131 }
132
133 return true;
134}

136void
137DebugSamplesRecArray::pixSamplesSave(const unsigned x,
138 const unsigned y,
139 const unsigned sampleId,
140 float &r,
141 float &g,
142 float &b,
143 float &a)
144{
145 if (x >= mSizeX || y >= mSizeY) return;
146
147 PixSamples &pix = mImg[y][x];
148 if (pix.size() != sampleId) {
149 std::cerr << ">> DebugSamplesRecArray.cc ====== RUNTIME-OVERRUN-ERROR ====== pixSamplesSave() failed."
150 << " x:" << x << " y:" << y << " sampleId:" << sampleId << std::endl;
151 return; // duplicate sampling id
152 }
153
154 pix.emplace_back(r, g, b, a);
155}

157void
158DebugSamplesRecArray::pixSamplesLoad(const unsigned x,
159 const unsigned y,
160 const unsigned sampleId,
161 float &r,
162 float &g,
163 float &b,
164 float &a) const
165{
166 if (x >= mSizeX || y >= mSizeY) return;
167
168 const PixSamples &pix = mImg[y][x];
169 if (sampleId >= pix.size()) {
170 std::cerr << ">> DebugSamplesRecArray.cc ====== RUNTIME-OVERRUN-ERROR ====== pixSamplesLoad() failed."
171 << " x:" << x << " y:" << y << " sampleId:" << sampleId << " >= pix.size():" << pix.size()
172 << std::endl;
173 return;
174 }
175
176 const scene_rdl2::fb_util::RenderColor &sample = pix[sampleId];
177 r = sample[0];
178 g = sample[1];
179 b = sample[2];
180 a = sample[3];
181}

213std::string
214DebugSamplesRecArray::memStr(const size_t byte) const
215{
216 std::ostringstream ostr;
217 if (byte < 1024) {
218 ostr << byte << " byte";
219 } else if (byte < (size_t)1024 * (size_t)1024) {
220 float kByte = (float)byte / 1024.f;
221 ostr << std::setw(7) << std::fixed << std::setprecision(2) << kByte << " Kbyte";
222 } else if (byte < (size_t)1024 * (size_t)1024 * (size_t)1024) {
223 float mByte = (float)byte / 1024.f / 1024.f;
224 ostr << std::setw(7) << std::fixed << std::setprecision(2) << mByte << " Mbyte";
225 } else if (byte < (size_t)1024 * (size_t)1024 * (size_t)1024 * (size_t)1024) {
226 float gByte = (float)byte / 1024.f / 1024.f / 1024.f;
227 ostr << std::setw(7) << std::fixed << std::setprecision(2) << gByte << " Gbyte";
228 } else {
229 float tByte = (float)byte / 1024.f / 1024.f / 1024.f / 1024.f;
230 ostr << std::setw(7) << std::fixed << std::setprecision(2) << tByte << " Tbyte";
231 }
232 return ostr.str();
233}
