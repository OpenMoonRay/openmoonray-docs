538 bool updatePrep(UpdateHelper& sceneObjects, int depth)
539 {
540 MNRY_ASSERT_REQUIRE(!mUpdateActive);
541
542 if (mUpdatePrepApplied &&
543 (sceneObjects.getDepth(this) >= depth || sceneObjects.isLeaf(this))) {
544 return updateRequired();
545 }
546 mUpdatePrepApplied = true;
547
548 mAttributeTreeChanged = mAttributeUpdateMask.any();
549 mBindingTreeChanged = mBindingUpdateMask.any();
550
551 bool isLeaf = true;
552 const size_t n = mSceneClass.mAttributes.size();
553 for (size_t i = 0; i < n; ++i) {
554 const Attribute * const attribute = mSceneClass.mAttributes[i];
555 switch (attribute->getType()) {
556 case TYPE_SCENE_OBJECT:
557 {
558 const AttributeKey<SceneObject*> key(*attribute);
559 SceneObject * const object = get(key);
560 if (object) {
561 isLeaf = false;
562 mAttributeTreeChanged |= object->updatePrep(sceneObjects, depth + 1);
563 }
564 }
565 break;
566
567 case TYPE_SCENE_OBJECT_VECTOR:
568 isLeaf = false;
569 updatePrepSequenceContainer<SceneObjectVector>(attribute, sceneObjects, depth + 1);
570 break;
571
572 case TYPE_SCENE_OBJECT_INDEXABLE:
573 isLeaf = false;
574 updatePrepSequenceContainer<SceneObjectIndexable>(attribute, sceneObjects, depth + 1);
575 break;
576
577 default:
578 break;
579 }
580 if (attribute->isBindable() && mBindings[i]) {
581 isLeaf = false;
582 mBindingTreeChanged |= mBindings[i]->updatePrep(sceneObjects, depth + 1);
583 }
584 }
585
586 if (updateRequired()) {
587 if (isLeaf) {
588 sceneObjects.insertLeaf(this);
589 } else {
590 sceneObjects.insert(this, depth);
591 }
592 }
593 return updateRequired();
594 }

602 void applyUpdates()
603 {
604 UpdateHelper sceneObjects;
605
606 this->updatePrep(sceneObjects, 0);
607
608 // Update all leaves
609
610 size_t s = sceneObjects.size();
611 if (s == 0){
612 Logger::info("There is no leaf scene object need to be updated");
613 } else if (s == 1) {
614 Logger::info("Updating 1 leaf scene object...");
615 } else {
616 Logger::info("Updating ", s, " leaf scene objects...");
617 }
618
619 for (auto obj = sceneObjects.cbegin(); obj != sceneObjects.cend(); ++obj)
620 {
621 (*obj)->debug("Updating");
622 (*obj)->update();
623 }
624
625 // Update the objects from bottom up
626 for (int i = static_cast<int>(sceneObjects.getMaxDepth())-1; i >= 0;
627 --i) {
628 size_t s = sceneObjects.size(i);
629 if (s == 0){
630 Logger::info("There is no scene object need to be updated at level ", i);
631 } else if (s == 1) {
632 Logger::info("Updating 1 scene object at level ", i, "...");
633 } else {
634 Logger::info("Updating ", s, " scene objects at level ", i, "...");
635 }
636
637 for (auto obj = sceneObjects.cbegin(i); obj != sceneObjects.cend(i); ++obj)
638 {
639 (*obj)->debug("Updating");
640 (*obj)->update();
641 }
642 }
643 }

922 template <typename Container>
923 bool updatePrepSequenceContainer(const Attribute* attribute,
924 UpdateHelper& sceneObjects,
925 int depth)
926 {
927 bool updateRequired = false;
928 const AttributeKey<Container> key(*attribute);
929 const Container& objectVector = get(key);
930 for (SceneObject * const object : objectVector) {
931 if (object) {
932 updateRequired |= object->updatePrep(sceneObjects, depth);
933 }
934 }
935 mAttributeTreeChanged |= updateRequired;
936 return updateRequired;

1004 // Bitmask indicating which attributes have been set. Used for determining
1005 // which attribute values to pack during serialization.
1006 boost::dynamic_bitset<> mAttributeSetMask;
1007
1008 // Bitmask indicating which attributes have bindings set. Used for
1009 // determining which bindings to pack during serialization.
1010 boost::dynamic_bitset<> mBindingSetMask;
1011
1012 // Bitmask indicating which attributes have changed *since the last
1013 // call to update()*. This is different from mAttributeSetMask, as this
1014 // bitmask and hasChanged() work with update(), while mAttributeSetMask is
1015 // used internally for the purposes of serialization.
1016 boost::dynamic_bitset<> mAttributeUpdateMask;
1017
1018 // Bitmask indicating which bindings have changed *since the last
1019 // call to update()*. This is different from mBindingSetMask, as this
1020 // bitmask and hasBindingChanged() work with update(), while
1021 // mAttributeSetMask is used internally for the purposes of serialization.
1022 boost::dynamic_bitset<> mBindingUpdateMask;
1023
1024 // Used to ensure that calls to set() and setBinding() only happen between
1025 // pairs of beginUpdate() and endUpdate() calls.
1026 bool mUpdateActive;
1027
1028 // Tracks whether the object is "dirty". Things that can make the object
1029 // dirty include creation, attribute changes, and binding changes.
1030 // Committing the changes makes the object "clean".
1031 // This is used by the SceneObject writers to decide what objects to
1032 // serialize, not by updatePrep().
1033 bool mDirty;
1034
1035 // Tracks whether updatePrep() has been called on this object since the
1036 // last resetUpdate() call. Keeps the updatePrep() call tree from going
1037 // down the same branch multiple times. However, if the current depth is
1038 // deeper than the depth recorded in previous calls, we need to go down
1039 // the same branch to update the depths of its childrens. Also Notice this
1040 // only means that updatePrep() has been called. It does not mean that update()
1041 // has been called.
1042 bool mUpdatePrepApplied;
1043
1044 // Track whether any dependencies have been changed hence this object need
1045 // to be updated.
1046 bool mAttributeTreeChanged;
1047 bool mBindingTreeChanged;
1048
1049 // Sometimes a change external to the object can require that this object be
1050 // updated. (E.g. a displacement assignment in a layer.)
1051 bool mUpdateRequested;
