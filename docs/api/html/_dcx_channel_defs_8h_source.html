54//=============================================================================
55//
56// Default channel names for deep metadata IO
57//
58// IMPORTANT NOTE: these names have changed from the original DigiPro2015
59// paper due to Nuke 10 not supporting numeric channel names! Specifically Nuke
60// no longer supports layer or channel names which start with a number.
61//
62// Additionally the channel type for the flags channel is now 32-bit float.
63//
64// To avoid channel reordering conflicts in applications it's best to also
65// change the layer name. Here's the old to new name mappings:
66// old (2.2.1) new (2.2.2)
67// spmask.1 -> deepabuf.sp1
68// spmask.2 -> deepabuf.sp2
69// spmask.3 -> deepmeta.flags
70//
71//=============================================================================
72

74const char* const spMask8Chan1Name = "sp1";
