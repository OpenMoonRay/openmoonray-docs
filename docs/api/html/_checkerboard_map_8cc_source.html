48void
49CheckerboardMap::sample(const scene_rdl2::rdl2::Map *self, moonray::shading::TLState *tls,
50 const moonray::shading::State &state, Color *sample)
51{
52 const CheckerboardMap *me = static_cast<const CheckerboardMap *>(self);
53
54 Vec2f st;
55 switch (me->get(attrTextureEnum)) {
56 case 1:
57 st = asVec2(evalVec3f(me, attrInputTextureCoordinate, tls, state));
58 break;
59 case 0:
60 default:
61 st = state.getSt();
62 break;
63 }
64
65 //Put UVs in 0-1 range
66 float smod = std::fmod(st[0] * me->get(attrUTiles) / 2.0f, 1.0f);
67 float tmod = std::fmod(st[1] * me->get(attrVTiles) / 2.0f, 1.0f);
68
69 //If negative, wrap it into positive range
70 if (smod < 0.0f) smod += 1.0f;
71 if (tmod < 0.0f) tmod += 1.0f;
72
73 if (smod < 0.5f) {
74 if (tmod < 0.5f) {
75 *sample = me->get(attrColorA);
76 } else {
77 *sample = me->get(attrColorB);
78 }
79 } else {
80 if (tmod < 0.5f) {
81 *sample = me->get(attrColorB);
82 } else {
83 *sample = me->get(attrColorA);
84 }
85 }
86}
