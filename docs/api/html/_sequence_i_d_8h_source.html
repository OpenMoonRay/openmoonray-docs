42// Normally, I would be all about type-safety. However, the only purpose for this class' existence is to supply
43// consistent, but random, hash values based on certain state. I'm not too concerned with the types used to specify that
44// state.
45class SequenceID
46{
47public:
48 template <typename... T>
49 explicit SequenceID(T... t) noexcept
50 : mHash(computeHash(t...))
51 {
52 }
53
54 utype operator()(utype seed) const noexcept
55 {
56 return getHash(seed);
57 }
58
59 utype getHash(utype seed) const noexcept
60 {
61 return gphash(mHash, seed);
62 }
63
64 utype private_test() const noexcept
65 {
66 return mHash;
67 }
68
69private:
70 static constexpr utype computeHashImpl() noexcept
71 {
72 // A Mersenne prime. We don't really expect a lot of calls to this level, but a Mersenne prime is probably as
73 // good a choice as any: it's prime, so it will give fairly good strides if used in that fashion, but it's
74 // Mersenne: A smart compiler can optimize any multiplications.
75 return 2147483647u;
76 }
77
78 template <typename T>
79 static constexpr utype computeHashImpl(T t) noexcept
80 {
81 // We could do something clever here, like break 64-bit types into two 32-bit types, but we want this to be
82 // the same in ISPC, where we only use 32-bit types. So we static_cast.
83 const utype mersennePrime7Exp = 19u;
84 const utype mersennePrime8Exp = 31u;
85 const auto h = hashCombine(0x31e149a2u, static_cast<utype>(t));
86 return mersenneMultiply(mersennePrime8Exp, mersenneMultiply(mersennePrime7Exp, h));
87 }
88
89 // TODO: C++17: re-write with fold expressions
90 template <typename T, typename... Rest>
91 static constexpr utype computeHashImpl(T t, Rest... rest) noexcept
92 {
93 // We sandwich the args with the first element so that we get good distributions whether we are increments on
94 // the first or the last elements. E.g.,
95 // SequenceID(2, 1, 3);
96 // SequenceID(2, 1, 4);
97 // SequenceID(2, 1, 5);
98 // vs
99 // SequenceID(2, 1, 3);
100 // SequenceID(3, 1, 3);
101 // SequenceID(4, 1, 3);
102 return hashCombine(hashCombine(computeHashImpl(t), computeHashImpl(rest...)), computeHashImpl(t));
103 }
104
105 template <typename T>
106 static constexpr utype computeHash(T t) noexcept
107 {
108 // We overload the computeHash function for one argument instead of
109 // using the variadic template version. If given one argument, the
110 // variadic template concatenates the value with itself, which gives a
111 // rather poor distribution.
112 const utype mersennePrime7Exp = 19u;
113 const utype mersennePrime8Exp = 31u;
114 const auto h = hashCombine(static_cast<utype>(t), 0x72407263u);
115 return mersenneMultiply(mersennePrime8Exp, mersenneMultiply(mersennePrime7Exp, h));
116 }
117
118 template <typename... T>
119 static constexpr utype computeHash(T... t) noexcept
120 {
121 return computeHashImpl(t...);
122 }
123
124 utype mHash;
125};
