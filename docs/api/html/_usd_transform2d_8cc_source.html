50void
51UsdTransform2d::sample(const scene_rdl2::rdl2::Map* self,
52 moonray::shading::TLState *tls,
53 const moonray::shading::State& state,
54 Color* sample)
55{
56 const UsdTransform2d* me = static_cast<const UsdTransform2d*>(self);
57
58 Color input = evalColor(me, attrIn, tls, state);
59 Vec2d uv(input.r, input.g);
60
61 const float rotation = me->get(attrRotation);
62 if (!isZero(rotation)) {
63 float s, c;
64 const float theta = deg2rad(-rotation);
65 sincos(theta, &s, &c);
66 uv.x = uv.x * c - uv.y * s;
67 uv.y = uv.x * s + uv.y * c;
68 }
69
70 const Vec2f scale = me->get(attrScale);
71 const Vec2f translation = me->get(attrTranslation);
72 uv.x = scale.x * uv.x + translation.x;
73 uv.y = scale.y * uv.y + translation.y;
74
75 *sample = Color(uv.x, uv.y, 0.f);
76}
