115//---------------------------------------------------------------------------
116//
117// Context structure for storing the global ChannelIdx assignment slots
118// and maps for quick access to/from ChannelAliases and ChannelIdxs.
119//
120// A :ref:`channelalias_class` contains layer and channel strings which map
121// to unique :ref:`channelidx_type` indice.
122//
123//---------------------------------------------------------------------------
124
125class DCX_EXPORT ChannelContext
126{
127 public:
128
129 struct ChanOrder
130 {
131 ChannelIdx channel; // ChannelIdx
132 uint32_t order; // Order in layer (TODO: not implemented - required anymore?)
133 };
134
135 struct Layer
136 {
137 std::string name;
138 std::vector<ChanOrder> channels;
139 };

141

143 typedef std::map<ChannelIdx, int> ChannelIdxToListMap;
144 typedef std::map<std::string, int> LayerNameToListMap;

147 public:
148
149 //--------------------------------------------------------------
150 //
151 // Default ctor adds the predefined standard ChannelAliases
152 // to the context.
153 // Pass in false if a subclass of ChannelContext wants to define
154 // a custom ChannelAlias type and manage the predefined set
155 // itself.
156 //
157 //--------------------------------------------------------------
158
159 ChannelContext(bool addStandardChans=true);
160
161 virtual ~ChannelContext();

170
171 //---------------------------------------------------------------

174
175 ChannelSet getChannelSetFromAliases (const ChannelAliasPtrList&);
176 ChannelSet getChannelSetFromAliases (const ChannelAliasPtrSet&);
177
178
179 //---------------------------------------------------------------
180 //
181 // Returns the last assigned ChannelIdx.
182 //
183 // This value can be used to size channel arrays as it represents
184 // the current maximum channel count for this context.
185 // Note that this value does not represent *active* channels,
186 // only channel definitions assigned to ChannelIdx slots, whereas
187 // a ChannelSet is used to define the set of active channels.
188 //
189 // If no arbitrary channels have been added to this context the
190 // value will be Chan_ArbitraryStart-1.
191 //
192 //---------------------------------------------------------------
193
194 ChannelIdx lastAssignedChannel () const;
195
196
197 //---------------------------------------------------------------------
198 //
199 // Get and/or create a channel, returning a ChannelIdx or ChannelAlias.
200 // Returns Chan_Invalid or NULL if there was an error creating it.
201 //
202 // Note that multiple channel names can map to the same ChannelIdx but
203 // each name will have a unique ChannelAlias.
204 // ex. 'R', 'rgba.R', 'rgb.red' will all be mapped to Chan_R with each
205 // getting a unique ChannelAlias.
206 //
207 // Unrecognized channel names like 'mylayer.foo' and 'mylayer.F' will
208 // each be assigned unique ChannelIdxs unless the duplicate
209 // ChannelAliases were added using addChannelAlias() with the same
210 // ChannelIdx from the first created alias.
211 //
212 //---------------------------------------------------------------------
213
214 ChannelIdx getChannel (const char* name);
215 ChannelIdx getChannel (const std::string& name);
216
217 ChannelAlias* getChannelAlias (const char* name);
218 ChannelAlias* getChannelAlias (const std::string& name);

226 //---------------------------------------------------------------------

228 const char* getChannelName (ChannelIdx channel) const;
229 std::string getChannelFullName (ChannelIdx channel) const;

245 //
246 // Add the predefined standard channels to the context.

249
250 void addStandardChannels();
251
252
253 //--------------------------------------------------------------------------------
254 //
255 // Add a new ChannelAlias to the context, either by passing in a pre-allocated
256 // ChannelAlias or having the context construct it.
257 // In both cases the ChannelContext takes ownership of the pointer and
258 // deletes the pointers in the destructor.
259 //
260 // If the assigned channel is Chan_Invalid then no specfic channel slot is
261 // being requested so the next available ChannelIdx is assigned, incrementing
262 // lastAssignedChannel().
263 //
264 // * chan_name user-facing name, not including the layer ('R', 'G', 'red',
265 // 'green', 'alpha', 'Z', 'ZBack', etc)
266 // * layer_name user-facing layer name ('rgba', 'beauty', 'beauty.diffuse')
267 // * channel absolute ChannelIdx - if Chan_Invalid a new ChannelIdx is assigned
268 // * position sorted position index within layer (i.e. 0,1,2)
269 // * io_name name to use for exr file I/O ('R' or 'AR' vs. 'rgba.red' or
270 // 'opacity.R')
271 // * io_type PixelType to use for file I/O
272 //
273 //--------------------------------------------------------------------------------
274
275 ChannelAlias* addChannelAlias (ChannelAlias* alias);
276
277 ChannelAlias* addChannelAlias (const std::string& chan_name,
278 const std::string& layer_name,
279 ChannelIdx channel,

283 int io_part=0);
