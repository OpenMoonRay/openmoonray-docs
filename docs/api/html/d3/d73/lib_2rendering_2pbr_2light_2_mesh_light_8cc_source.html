85// Get uv coordinates of a point p on a quad. The first triangle of the quad is defined
86// by p0, p1, and p3. The second triangle is defined by p2, p3, and p1. If u + v <= 1,
87// then the point lies on the first tringle. If u + v > 1, then it lies on the second
88// triangle. If the quad is degenerate, or p lies outside of the quad, then we fall back
89// to a dummy uv of (0,0).
90scene_rdl2::math::Vec2f getQuadUVCoordinates(const scene_rdl2::math::Vec3f& p0, const scene_rdl2::math::Vec3f& p1, const scene_rdl2::math::Vec3f& p2,
91 const scene_rdl2::math::Vec3f& p3, const scene_rdl2::math::Vec3f& p)
92{
93 // Which triangle in the quad do we use to compute the uv coordinates for p?
94 // Figure out which side of the triangle's diagonal the point is on.
95 // The diagonal is p3 - p1.
96 const scene_rdl2::math::Vec3f diagonal = p3 - p1;
97 const scene_rdl2::math::Vec3f pvec = p - p1;
98 const scene_rdl2::math::Vec3f pvecxdiag = math::cross(pvec, diagonal);
99 // edge on first triangle
100 const scene_rdl2::math::Vec3f e1 = p0 - p1;
101 // edge on second triangle
102 const scene_rdl2::math::Vec3f e2 = p2 - p1;
103 // normal of first triangle
104 const scene_rdl2::math::Vec3f n1 = math::cross(e1, diagonal);
105 // normal of second triangle (pointing in opposite direction)
106 const scene_rdl2::math::Vec3f n2 = math::cross(e2, diagonal);
107 float tri1 = math::dot(pvecxdiag, n1);
108 float tri2 = math::dot(pvecxdiag, n2);
109 // 99% of the time, one dot product will be negative and the other will be positive.
110 // But if there is a precision error, pick the larger one.
111 bool insideTri1 = tri1 > tri2;
112
113 scene_rdl2::math::Vec2f uv;
114 if (insideTri1) {
115 float recipDenom = 1.0f / dot(n1, n1);
116 // w is weight applied to p0
117 float w = tri1 * recipDenom;
118 // v is weight applied to p3
119 float v = math::dot(math::cross(e1, pvec), n1) * recipDenom;
120 // u is weight applied to p1
121 float u = 1.f - v - w;
122 uv.x = u;
123 uv.y = v;
124 } else {
125 float recipDenom = 1.0f / dot(n2, n2);
126 // w is weight applied to p2
127 float w = tri2 * recipDenom;
128 // v is weight applied to p3
129 float v = math::dot(math::cross(e2, pvec), n2) * recipDenom;
130 // u is weight applied to p1
131 float u = 1.f - v - w;
132 // second triangle must flip the uv coordinates
133 uv.x = 1.f - u;
134 uv.y = 1.f - v;
135 }
136
137 // Check that uv is valid. Reasons to have an invalid uv are:
138 // Zero area quad or the point p is outside the quad.
139 // In that case, make a dummy uv.
140 if (!math::isFinite(uv) ||
141 uv.x > 1.f || uv.y > 1.f ||
142 uv.x < 0.f || uv.y < 0.f) {
143
144 uv.x = 0.f;
145 uv.y = 0.f;
146 }
147
148 return uv;
149}

168// Given n normalized vectors and a central axis vector, find the maximum angle
169// between each vector and the axis. This is used to find thetaO of a face that
170// has motion blur. ThetaO spans the motion of the normal of that face from
171// t = 0 to t = 1. If the face is a quad, then the normals from both subtriangles
172// must be considered.
173float getMaxAngle(const scene_rdl2::math::Vec3f& axis, const scene_rdl2::math::Vec3f vectors[], size_t n)
174{
175 float minCosAngle = 1.f;
176 for (size_t i = 0; i < n; ++i) {
177 float cosAngle = dot(axis, vectors[i]);
178 if (cosAngle < minCosAngle) {
179 minCosAngle = cosAngle;
180 }
181 }
182
183 return scene_rdl2::math::acos(scene_rdl2::math::clamp(minCosAngle, -1.f, 1.f));
184}

188// A Cone describes the parameters needed to compute the orientation heuristics
189// when building and the bvh and computing the importance of a node in the bvh.
190// It is the orientation analogue to a bounding box.
191struct Cone {
192 scene_rdl2::math::Vec3f mAxis;
193 float mThetaO;
194 // for the mesh light, we do not need to store thetaE because it is always
195 // pi / 2. For a many light solution we would need to store thetaE for each
196 // bounding cone.
197 //float mThetaE;
198
199 void print() const {
200 std::cout << "axis: " << mAxis << std::endl;
201 std::cout << "thetaO: " << mThetaO << std::endl;
202 //std::cout << "thetaE: " << mThetaE << std::endl;
203 }
204};

206// Taken from Algorithm 1 in Kulla (2017) p. 5
207// Given two cones what is their combined axis and orientation angle?
208Cone
209combineBoundingCones(const Cone* a, const Cone* b)
210{
211 if (b->mThetaO > a->mThetaO) {
212 std::swap(a, b);
213 }
214
215 // cone a is a sphere. Just return it.
216 if (a->mThetaO > scene_rdl2::math::sPi) {
217 return *a;
218 }
219
220 float cosThetaD = scene_rdl2::math::clamp(dot(a->mAxis, b->mAxis), -1.0f, 1.0f);
221 float thetaD = scene_rdl2::math::acos(cosThetaD);
222
223 // For mesh lights, we do not need to compute thetaE because it is always
224 // pi / 2. For a many light solution we need to compute thetaE. I am leaving
225 // this computation as a reminder in case we ever want to use this for a many
226 // lights solution.
227 //float thetaE = max(a.mThetaE, b.mThetaE);
228
229 // The bound of cone a already covers b, return cone a
230 if (thetaD + b->mThetaO <= a->mThetaO) {
231 return *a;
232 }
233
234 // construct new bounding cone
235 float thetaO = (a->mThetaO + thetaD + b->mThetaO) / 2.0f;
236 // thetaO cannot be greater than pi for a sphere.
237 if (scene_rdl2::math::sPi <= thetaO) {
238 return {a->mAxis, scene_rdl2::math::sPi};
239 }
240
241 // rotate a's axis towards b's axis so thetaO covers both
242 scene_rdl2::math::Vec3f rotAxis = cross(a->mAxis, b->mAxis);
243 if (scene_rdl2::math::isZero(length(rotAxis))) {
244 // cone becomes sphere because axes are antiparallel
245 return { scene_rdl2::math::Vec3f(1.f, 0.f, 0.f), scene_rdl2::math::sPi };
246 }
247 rotAxis = rotAxis.normalize();
248 float thetaR = thetaO - a->mThetaO;
249 const scene_rdl2::math::Vec3f axis = scene_rdl2::math::cos(thetaR) * a->mAxis + scene_rdl2::math::sin(thetaR) * cross(rotAxis,
250 a->mAxis);
251 return {axis, thetaO};
252}

329// The SplitCost computes the cost of splitting the bvh node at a particular
330// location. The cost for splitting a node N into nodes NL and NR is
331// c * (EL * SAL * OL + ER * SAR * OR) / (SA * O).
332// c is the cost of splitting along axis x, y or z.
333// E (and EL and ER) is the cumulative energy of all the primitives in the node.
334// SA (and SAL and SAR) is the bounding box surface area of the node.
335// O (and OR and OL) is the solid angle of light emission influence of the node.
336struct SplitCost {
337 SplitCost() : mLeftEnergy(0.0f), mRightEnergy(0.0f),
338 mLeftBbox(scene_rdl2::util::empty), mRightBbox(scene_rdl2::util::empty),
339 mLeftBcone({scene_rdl2::math::Vec3f(0.f), 0.f}), mRightBcone({scene_rdl2::math::Vec3f(0.f), 0.f})
340 {}
341
342 // Compute the cost of the bounding cone
343 float orientationHeuristic(float thetaO) const
344 {
345 // Kulla (2017) p. 6 Eq. 1.
346 float thetaW = scene_rdl2::math::min(thetaO + sThetaE, scene_rdl2::math::sPi);
347 return (scene_rdl2::math::sPi/2) * (4 - 3 * scene_rdl2::math::cos(thetaO) + 2 * (thetaW - thetaO) *
348 scene_rdl2::math::sin(thetaO) - scene_rdl2::math::cos(thetaO - 2 * thetaW));
349 }
350
351 inline float boxArea(const scene_rdl2::math::Vec3f& dim) const
352 {
353 return dim[0] * dim[1] + dim[1] * dim[2] + dim[2] * dim[0];
354 };
355
356 float eval(float splitCoefficient) const
357 {
358 // get cost of orientation
359 float leftOrientation = orientationHeuristic(mLeftBcone.mThetaO);
360 float rightOrientation = orientationHeuristic(mRightBcone.mThetaO);
361 Cone combinedBcone = combineBoundingCones(&mLeftBcone, &mRightBcone);
362 float combinedOrientation = orientationHeuristic(combinedBcone.mThetaO);
363
364 // get cost of surface area
365 const scene_rdl2::math::Vec3f leftDim = mLeftBbox.size();
366 const scene_rdl2::math::Vec3f rightDim = mRightBbox.size();
367
368 float leftBoundSA = boxArea(leftDim);
369 float rightBoundSA = boxArea(rightDim);
370
371 scene_rdl2::math::BBox3f combinedBound = mLeftBbox;
372 combinedBound.extend(mRightBbox);
373 const scene_rdl2::math::Vec3f totalDim = combinedBound.size();
374 float combinedBoundSA = boxArea(totalDim);
375
376 // SAOH
377 return (mLeftEnergy * leftBoundSA * leftOrientation
378 + mRightEnergy * rightBoundSA * rightOrientation)
379 * splitCoefficient / (combinedBoundSA * combinedOrientation);
380 };
381
382 float mLeftEnergy;
383 float mRightEnergy;
384 scene_rdl2::math::BBox3f mLeftBbox;
385 scene_rdl2::math::BBox3f mRightBbox;
386 Cone mLeftBcone;
387 Cone mRightBcone;
388};

421bool
422MeshLight::update(const scene_rdl2::math::Mat4d& world2render)
423{
424 MNRY_ASSERT(mRdlLight);
425
426 // reset everything
427 reset();
428
429 // Store the scene's render to world transform, which is equivalent to the
430 // render to light transform. Please see the explanation about this transform
431 // in MeshLight.h
432 mRender2Light = world2render.inverse();
433
434 // check if light is on and the reference geometry is valid
435 mOn = mRdlLight->get(scene_rdl2::rdl2::Light::sOnKey) && getReferenceGeometry();
436 if (!mOn) {
437 return false;
438 }
439
440 // Compute a dummy radiance using just the color, intensity and exposure.
441 // There is no point in continuing the update if this evaluates to zero.
442 mRadiance = computeLightRadiance(mRdlLight, scene_rdl2::rdl2::Light::sColorKey,
443 scene_rdl2::rdl2::Light::sIntensityKey, scene_rdl2::rdl2::Light::sExposureKey, sNormalizedKey,
444 sApplySceneScaleKey, 1.f /* dummy inverse area */);
445
446 if (isBlack(mRadiance)) {
447 mOn = false;
448 return false;
449 }
450
451 // Update light transform
452 // Accommodating 2 motion steps for now. However we do not have light motion
453 // blur working yet so only the first motion step is used.
454 const scene_rdl2::math::Mat4d& l2w0 = mRdlLight->get(scene_rdl2::rdl2::Node::sNodeXformKey, /* rayTime = */ 0.f);
455 const scene_rdl2::math::Mat4d& l2w1 = mRdlLight->get(scene_rdl2::rdl2::Node::sNodeXformKey, /* rayTime = */ 1.f);
456 const scene_rdl2::math::Mat4f local2Render0 = toFloat(l2w0 * world2render);
457 const scene_rdl2::math::Mat4f local2Render1 = toFloat(l2w1 * world2render);
458 if (!updateParamAndTransforms(local2Render0, local2Render1, 1.0f, 1.0f)) {
459 return false;
460 }
461
462 // Update miscellaneous
463 updateVisibilityFlags();
464 updatePresenceShadows();
465 updateRayTermination();
466 updateMaxShadowDistance();
467
468 // update rdl geometry
469 scene_rdl2::rdl2::SceneObject* geomSo = mRdlLight->get(sGeometryKey);
470 if (geomSo) {
471 mRdlGeometry = geomSo->asA<scene_rdl2::rdl2::Geometry>();
472 }
473
474 // update image map
475 scene_rdl2::rdl2::SceneObject* mapSo = mRdlLight->get(sMapShaderKey);
476 if (mapSo) {
477 // ispc ptr is int64
478 mMapShader = (const int64_t *) mapSo->asA<scene_rdl2::rdl2::Map>();
479 }
480
481 UPDATE_ATTRS_CLEAR_RADIUS
482
483 return true;
484}

585void
586MeshLight::setMesh(geom::internal::Primitive* prim)
587{
588 // If there is no reference mesh, then the light is off
589 if (prim == nullptr) {
590 return;
591 }
592
593 MNRY_ASSERT(prim->getType() == geom::internal::Primitive::POLYMESH);
594
595 // Check parts
596 bool validPart = false;
597 const scene_rdl2::rdl2::StringVector& parts = getPartsList();
598 for (const auto& part : parts) {
599 if (mLayer->getAssignmentId(mRdlGeometry, part) != -1) {
600 validPart = true;
601 break;
602 }
603 }
604 // load in all meshes if parts list is empty
605 validPart |= parts.empty();
606
607 if (!validPart) {
608 return;
609 }
610
611 geom::internal::Mesh* geomMesh = (geom::internal::Mesh*)prim;
612 geom::internal::Mesh::TessellatedMesh mesh;
613 geomMesh->getTessellatedMesh(mesh);
614
615 size_t faceCount = mesh.mFaceCount;
616
617 // If the mesh is empty, early return.
618 if (faceCount == 0) {
619 return;
620 }
621
622 // Is this mesh a triangle mesh or a quad mesh?
623 size_t faceVertexCount = 0;
624 if (mesh.mIndexBufferType == geom::internal::MeshIndexType::TRIANGLE) {
625 faceVertexCount = 3;
626 } else if (mesh.mIndexBufferType == geom::internal::MeshIndexType::QUAD) {
627 faceVertexCount = 4;
628 } else {
629 MNRY_ASSERT(0, "MeshLight: face vertex count is bad");
630 }
631
632 // The id of this mesh
633 int geomID = mGeomMeshes.size();
634 mGeomMeshes.push_back(geomMesh);
635
636 // Does this mesh have deformation motion blur (ie, multiple vertex buffers?)
637 mDeformationMb = (mesh.mVertexBufferDesc.size() > 1);
638
639 // All meshes in the mesh light must have same number of motion steps.
640 // setMesh is called multiple times on the same rdlGeometry, once per
641 // mesh in the procedural. The first mesh determines the number of motion
642 // blur steps. Each subsequent mesh checks its number of motion blur steps
643 // against the first one.
644 if (mMbSteps == 0) {
645 mMbSteps = mesh.mVertexBufferDesc.size();
646 } else {
647 MNRY_ASSERT(mMbSteps == mesh.mVertexBufferDesc.size());
648 }
649
650 // index buffer
651 const int* indices = (const int* )mesh.mIndexBufferDesc.mData;
652 const size_t indexCount = faceCount * faceVertexCount;
653
654 // get unique indices
655 std::set<int> tmpOrderedIndices(indices, indices + indexCount);
656 std::vector<int> orderedIndices(tmpOrderedIndices.begin(), tmpOrderedIndices.end());
657
658 float dt = mMbSteps == 1 ? 1.f : 1.f / float(mMbSteps - 1);
659
660 // vertex buffer
661 size_t motionSampleCount = mesh.mVertexBufferDesc.size();
662 size_t vertexCount = motionSampleCount * mesh.mVertexCount;
663 size_t vertexOffset = mVertices.size();
664 mVertexOffset.push_back(vertexOffset);
665 mVertices.resize(vertexOffset + vertexCount);
666 for (int index : orderedIndices) {
667 for (size_t t = 0; t < motionSampleCount; ++t) {
668 // byte offset of vertex
669 size_t byteOffset = mesh.mVertexBufferDesc[t].mOffset + mesh.mVertexBufferDesc[t].mStride * index;
670
671 // x component of vertex
672 const float* f = reinterpret_cast<const float *>((const char *)mesh.mVertexBufferDesc[t].mData + byteOffset);
673
674 // get and transform vertex
675 const scene_rdl2::math::Vec3f vertex(*f, *(f+1), *(f+2));
676 MNRY_ASSERT(math::isFinite(vertex));
677 const scene_rdl2::math::Vec3f localSpaceVertex = transformPoint(mRender2Light, vertex);
678 mVertices[vertexOffset + index * motionSampleCount + t] = localSpaceVertex;
679
680 // extend bounding boxes in local space and render space
681 mLocalSpaceBounds.extend(localSpaceVertex);
682 float time = float(t) * dt;
683 mRenderSpaceBounds.extend(xformPointLocal2Render(localSpaceVertex, time));
684
685 // If we have transform motion blur but only one vertex buffer, we must expand the bounds at time point 1.0f
686 if (mMb && motionSampleCount == 1) {
687 // If there is rotational motion blur, we should have many motion steps. 10 is the default number of
688 // motion steps for curved motion blur on geometries. If it is linear or scale motion blur then 2 motion
689 // steps is enough.
690 const size_t motionSteps = mMb & LIGHT_MB_ROTATION ? 10 : 2;
691 const float dt = 1.f / float(motionSteps - 1);
692 for (size_t t = 1; t < motionSteps; ++t) {
693 float time = float(t) * dt;
694 mRenderSpaceBounds.extend(xformPointLocal2Render(localSpaceVertex, time));
695 }
696 }
697 }
698 }
699
700 // centroid time is for determining what face centers, areas, normals, and
701 // energies will be used for the importance sampling when we have motion blur.
702 float centroidTime = mDeformationMb ? 0.5f : 0.0f;
703
704 // Face buffer
705 std::vector<Face> faces(faceCount);
706 for (size_t f = 0; f < faceCount; ++f) {
707 // get IDs
708 faces[f].mGeomID = geomID;
709 faces[f].mPrimID = f;
710
711 // get indices
712 faces[f].mIndices = &indices[f * faceVertexCount];
713
714 // get center
715 faces[f].mCenter = getFaceVertex(faces[f], 0, centroidTime) +
716 getFaceVertex(faces[f], 1, centroidTime) +
717 getFaceVertex(faces[f], 2, centroidTime);
718 if (faceVertexCount == 4) {
719 faces[f].mCenter += getFaceVertex(faces[f], 3, centroidTime);
720 faces[f].mCenter *= 0.25f;
721 } else {
722 faces[f].mCenter *= (1.0f / 3.0f);
723 }
724
725 float area = 0.f;
726 if (faceVertexCount == 3 || getFaceVertex(faces[f], 2, centroidTime) ==
727 getFaceVertex(faces[f], 3, centroidTime)) {
728 // triangle or degenerate quad case
729
730 // get area
731 const scene_rdl2::math::Vec3f normal = getNormal(
732 getFaceVertex(faces[f], 0, centroidTime),
733 getFaceVertex(faces[f], 1, centroidTime),
734 getFaceVertex(faces[f], 2, centroidTime));
735
736 area = 0.5f * length(normal);
737
738 // get normal
739 faces[f].mNormal = normalize(normal);
740
741 // get thetaO
742 if (mDeformationMb) {
743 // normal at t = 0
744 const scene_rdl2::math::Vec3f normal0 = getUnitNormal(getFaceVertex(faces[f], 0, 0.f),
745 getFaceVertex(faces[f], 1, 0.f), getFaceVertex(faces[f], 2, 0.f));
746
747 // normal at t = 1
748 const scene_rdl2::math::Vec3f normal1 = getUnitNormal(getFaceVertex(faces[f], 0, 1.f),
749 getFaceVertex(faces[f], 1, 1.f), getFaceVertex(faces[f], 2, 1.f));
750
751 // Accommodate motion of normals in thetaO. ThetaO is the
752 // maximum angle between the normal at the centroid time point
753 // and the normals at the first and last time point.
754 const scene_rdl2::math::Vec3f normals[] = { normal0, normal1 };
755 faces[f].mThetaO = getMaxAngle(faces[f].mNormal, normals, 2);
756 } else {
757 faces[f].mThetaO = sThetaO;
758 }
759 } else {
760 // quad case
761
762 // get area
763 const scene_rdl2::math::Vec3f cross1 = getNormal(
764 getFaceVertex(faces[f], 0, centroidTime),
765 getFaceVertex(faces[f], 1, centroidTime),
766 getFaceVertex(faces[f], 3, centroidTime));
767 area = 0.5 * length(cross1);
768 const scene_rdl2::math::Vec3f cross2 = getNormal(
769 getFaceVertex(faces[f], 2, centroidTime),
770 getFaceVertex(faces[f], 3, centroidTime),
771 getFaceVertex(faces[f], 1, centroidTime));
772 area += 0.5f * length(cross2);
773
774 // get normal
775 // A quad is not necessarily flat, so we average the two normals.
776 faces[f].mNormal = normalize(cross1 + cross2);
777
778 // get thetaO
779 if (mDeformationMb) {
780 // normal of first triangle at t = 0
781 const scene_rdl2::math::Vec3f normal00 = getUnitNormal(getFaceVertex(faces[f], 0, 0.f),
782 getFaceVertex(faces[f], 1, 0.f), getFaceVertex(faces[f], 3, 0.f));
783
784 // normal of second triangle at t = 0
785 const scene_rdl2::math::Vec3f normal10 = getUnitNormal(getFaceVertex(faces[f], 2, 0.f),
786 getFaceVertex(faces[f], 3, 0.f), getFaceVertex(faces[f], 1, 0.f));
787
788 // normal of first triangle at t = 1
789 const scene_rdl2::math::Vec3f normal01 = getUnitNormal(getFaceVertex(faces[f], 0, 1.f),
790 getFaceVertex(faces[f], 1, 1.f), getFaceVertex(faces[f], 3, 1.f));
791
792 // normal of second triangle at t = 1
793 const scene_rdl2::math::Vec3f normal11 = getUnitNormal(getFaceVertex(faces[f], 2, 1.f),
794 getFaceVertex(faces[f], 3, 1.f), getFaceVertex(faces[f], 1, 1.f));
795
796 // Accommodate motion of normals in thetaO. ThetaO is the
797 // maximum angle between the normal at the centroid time point
798 // and the normals at the first and last time point.
799 const scene_rdl2::math::Vec3f normals[] = {normal00, normal10, normal01, normal11};
800 faces[f].mThetaO = getMaxAngle(faces[f].mNormal, normals, 4);
801 } else {
802 // If there is no deformation mb, thetaO is the larger angle
803 // between the two triangle normals and the face normal.
804 const scene_rdl2::math::Vec3f normals[] = { normalize(cross1), normalize(cross2) };
805 faces[f].mThetaO = getMaxAngle(faces[f].mNormal, normals, 2);
806 }
807 }
808 mArea += area;
809 faces[f].mInvArea = 1 / area;
810
811 // get energy
812 // TODO: more fine detail on texture sampling? Take average instead of max?
813 // Currently we sample the texture from each corner of the mesh and from its
814 // center. But it might be better to take many more samples than that.
815 if (mMapShader) {
816 float energy = 0.0f;
817 mcrt_common::ThreadLocalState *tls = mcrt_common::getFrameUpdateTLS();
818
819 // In order to approximate the energy of a face, we sample at each
820 // corner of the face and its center and take the maximum color value.
821 // Due to the location of the samples, it is trivial to compute most
822 // of the uv coordinates.
823
824 // sample texture at first corner of the face
825 const scene_rdl2::math::Color& firstColor = sampleMapShader(tls, geomID, f,
826 getFaceVertex(faces[f], 0, centroidTime), faces[f].mNormal, scene_rdl2::math::Vec2f(0.f, 0.f));
827 energy = scene_rdl2::math::max(energy, luminance(firstColor));
828
829 // sample texture at second corner of the face
830 const scene_rdl2::math::Color& secondColor = sampleMapShader(tls, geomID, f,
831 getFaceVertex(faces[f], 1, centroidTime), faces[f].mNormal, scene_rdl2::math::Vec2f(1.f, 0.f));
832 energy = scene_rdl2::math::max(energy, luminance(secondColor));
833
834 // sample texture at third corner of the face
835 scene_rdl2::math::Vec2f thirdUV;
836 if (faceVertexCount == 3) {
837 // triangle
838 thirdUV = scene_rdl2::math::Vec2f(0.f, 1.f);
839 } else {
840 // quad
841 thirdUV = scene_rdl2::math::Vec2f(1.f, 1.f);
842 }
843 const scene_rdl2::math::Color& thirdColor = sampleMapShader(tls, geomID, f,
844 getFaceVertex(faces[f], 2, centroidTime), faces[f].mNormal, thirdUV);
845 energy = scene_rdl2::math::max(energy, luminance(thirdColor));
846
847 // sample texture at fourth corner if the face is a quad
848 if (faceVertexCount == 4) {
849 const scene_rdl2::math::Color& fourthColor = sampleMapShader(tls, geomID, f,
850 getFaceVertex(faces[f], 3, centroidTime), faces[f].mNormal, scene_rdl2::math::Vec2f(0.f, 1.f));
851 energy = scene_rdl2::math::max(energy, luminance(fourthColor));
852 }
853
854 // sample texture in the middle of the face
855 scene_rdl2::math::Vec2f centerUV;
856 if (faceVertexCount == 3) {
857 // the uv is trivial if the face is a triangle
858 centerUV = scene_rdl2::math::Vec2f(1.f / 3.f, 1.f / 3.f);
859 } else {
860 // if the face is a quad, we need to compute the uv coordinate
861 const scene_rdl2::math::Vec3f& p0 = getFaceVertex(faces[f], 0, centroidTime);
862 const scene_rdl2::math::Vec3f& p1 = getFaceVertex(faces[f], 1, centroidTime);
863 const scene_rdl2::math::Vec3f& p2 = getFaceVertex(faces[f], 2, centroidTime);
864 const scene_rdl2::math::Vec3f& p3 = getFaceVertex(faces[f], 3, centroidTime);
865 const scene_rdl2::math::Vec3f& center = faces[f].mCenter;
866
867 centerUV = getQuadUVCoordinates(p0, p1, p2, p3, center);
868 }
869 const scene_rdl2::math::Color& centerColor = sampleMapShader(tls, geomID, f,
870 faces[f].mCenter, faces[f].mNormal, centerUV);
871 energy = scene_rdl2::math::max(energy, luminance(centerColor));
872 faces[f].mEnergy = energy * area;
873 } else {
874 faces[f].mEnergy = 1.0f * area;
875 }
876 }
877
878 // Append the faces from this mesh to the faces list. This vector will be submitted to the bvh.
879 mFaceOffset.push_back(mFaces.size());
880 mFaces.insert(mFaces.end(), faces.begin(), faces.end());
881 mFaceVertexCount.push_back(faceVertexCount);
882}

884void
885MeshLight::finalize() {
886 // first check if there are any faces. If not, mOn is false.
887 if (mFaces.empty()) {
888 mOn = false;
889 return;
890 }
891
892 if (scene_rdl2::math::isZero(mArea)) {
893 mOn = false;
894 return;
895 }
896
897 mInvArea = 1.0f / mArea;
898 mFaceCount = mFaces.size();
899 mPrimIDToNodeID.resize(mFaceCount);
900
901 // Compute radiance
902 mRadiance = computeLightRadiance(mRdlLight, scene_rdl2::rdl2::Light::sColorKey,
903 scene_rdl2::rdl2::Light::sIntensityKey, scene_rdl2::rdl2::Light::sExposureKey, sNormalizedKey,
904 sApplySceneScaleKey, mInvArea);
905
906 if (isBlack(mRadiance)) {
907 mOn = false;
908 return;
909 }
910
911 // build the mesh light bvh
912 mBVH.clear();
913 buildBVHRecurse(mLocalSpaceBounds, mFaces, 0, mFaceCount, 0);
914 MNRY_ASSERT(verifyBuild(0));
915
916 // Fill in relevant HUD data
917 mBVHPtr = mBVH.data();
918 mBVHSize = mBVH.size();
919 mVerticesPtr = mVertices.data();
920 mVertexOffsetPtr = mVertexOffset.data();
921 mFaceOffsetPtr = mFaceOffset.data();
922 mFaceVertexCountPtr = mFaceVertexCount.data();
923 mPrimIDToNodeIDPtr = mPrimIDToNodeID.data();
924}

927void
928MeshLight::setEmbreeAccelerator(RTCDevice rtcDevice)
929{
930 if (mRtcScene == nullptr) {
931 mRtcScene = rtcNewScene(rtcDevice);
932 rtcSetSceneBuildQuality(mRtcScene, RTC_BUILD_QUALITY_HIGH);
933
934 // iterate over all meshes
935 size_t meshCount = mGeomMeshes.size();
936 for (size_t i = 0; i < meshCount; ++i) {
937 geom::internal::Mesh::TessellatedMesh mesh;
938 mGeomMeshes[i]->getTessellatedMesh(mesh);
939
940 // Create Embree geometry
941 RTCGeometry rtcGeom;
942 RTCFormat indexBufferFormat;
943 if (mFaceVertexCount[i] == 3) {
944 rtcGeom = rtcNewGeometry(rtcDevice, RTC_GEOMETRY_TYPE_TRIANGLE);
945 indexBufferFormat = RTC_FORMAT_UINT3;
946 } else if (mFaceVertexCount[i] == 4) {
947 rtcGeom = rtcNewGeometry(rtcDevice, RTC_GEOMETRY_TYPE_QUAD);
948 indexBufferFormat = RTC_FORMAT_UINT4;
949 } else {
950 MNRY_ASSERT(0);
951 return;
952 }
953
954 // set motion steps.
955 rtcSetGeometryTimeStepCount(rtcGeom, mMbSteps);
956
957 // Set up the mesh index buffer.
958 rtcSetSharedGeometryBuffer(rtcGeom, RTC_BUFFER_TYPE_INDEX, 0,
959 indexBufferFormat,
960 mesh.mIndexBufferDesc.mData,
961 mesh.mIndexBufferDesc.mOffset,
962 mesh.mIndexBufferDesc.mStride,
963 mesh.mFaceCount);
964
965 // Set up the mesh vertex buffers
966 for (size_t t = 0; t < mMbSteps; ++t) {
967 rtcSetSharedGeometryBuffer(rtcGeom, RTC_BUFFER_TYPE_VERTEX, t,
968 RTC_FORMAT_FLOAT3, // xyz
969 (void*) mVerticesPtr,
970 (mVertexOffset[i] + t) * sizeof(scene_rdl2::math::Vec3f), /*offset*/
971 mMbSteps * sizeof(scene_rdl2::math::Vec3f), /*stride*/
972 mesh.mVertexCount);
973 }
974
975 // geometry mask
976 unsigned mask = scene_rdl2::rdl2::ALL_VISIBLE;
977 rtcSetGeometryMask(rtcGeom, mask);
978
979 rtcSetGeometryIntersectFilterFunction(rtcGeom, &rt::backFaceCullingFilter);
980 rtcSetGeometryOccludedFilterFunction(rtcGeom, &rt::backFaceCullingFilter);
981
982 unsigned geomID = rtcAttachGeometry(mRtcScene, rtcGeom);
983 // If this ever fails, we need to create a mapping from embree geomID
984 // to the index into the mGeomMeshes vector.
985 MNRY_ASSERT_REQUIRE(geomID == i);
986 rtcCommitGeometry(rtcGeom);
987 }
988 rtcCommitScene(mRtcScene);
989 }

997
998const scene_rdl2::math::Vec3f
999MeshLight::getFaceVertex(const Face& face, size_t index, float time) const
1000{
1001 size_t vertexOffset = mVertexOffset[face.mGeomID] + face.mIndices[index] * mMbSteps;
1002
1003 if (!mDeformationMb) {
1004 return mVertices[vertexOffset];
1005 }
1006
1007 // Special case for time == 1.0f, because otherwise there will be a segfault
1008 // when trying to access v2 below. time == 1.0f when setting the mesh and
1009 // building the bvh, but it shouldn't occur during mcrt.
1010 if (time == 1.0f) {
1011 return mVertices[vertexOffset + mMbSteps - 1];
1012 }
1013
1014 float invDt = mMbSteps - 1;
1015 float d = time * invDt;
1016 float step = scene_rdl2::math::floor(d);
1017 float t = d - step;
1018
1019 const scene_rdl2::math::Vec3f& v1 = mVertices[vertexOffset + step];
1020 const scene_rdl2::math::Vec3f& v2 = mVertices[vertexOffset + step + 1];
1021 return math::lerp(v1, v2, t);
1022}

1024int
1025MeshLight::naiveSplit(std::vector<Face>& faces,
1026 int start, int end, unsigned splitAxis,
1027 scene_rdl2::math::BBox3f& leftBound, scene_rdl2::math::BBox3f& rightBound) const
1028{
1029 // We won't split a face
1030 MNRY_ASSERT(end - start > 1);
1031
1032 leftBound = scene_rdl2::math::BBox3f(scene_rdl2::util::empty);
1033 rightBound = scene_rdl2::math::BBox3f(scene_rdl2::util::empty);
1034
1035 // put half of the faces in the left node and half in the right
1036 int mid = (start + end) / 2;
1037
1038 // partially sort each face according to the center of the face
1039 std::nth_element(faces.begin() + start, faces.begin() + mid, faces.begin() + end,
1040 [&](const Face &face1, const Face &face2) {
1041 return face1.mCenter[splitAxis] < face2.mCenter[splitAxis];
1042 }
1043 );
1044
1045 // number of motion blur steps
1046 float dt = mDeformationMb ? 1.0f / (mMbSteps - 1) : 0.f;
1047
1048 for (int i = start; i < mid; ++i) {
1049 unsigned fvc = getFaceVertexCount(faces[i]);
1050 for (unsigned j = 0; j < fvc; ++j) {
1051 for (unsigned t = 0; t < mMbSteps; ++t) {
1052 float time = dt * t;
1053 leftBound.extend(getFaceVertex(faces[i], j, time));
1054 }
1055 }
1056 }
1057
1058 for (int i = mid; i < end; ++i) {
1059 unsigned fvc = getFaceVertexCount(faces[i]);
1060 for (unsigned j = 0; j < fvc; ++j) {
1061 for (unsigned t = 0; t < mMbSteps; ++t) {
1062 float time = dt * t;
1063 rightBound.extend(getFaceVertex(faces[i], j, time));
1064 }
1065 }
1066 }
1067
1068 return mid;
1069}
1070
1071int
1072MeshLight::buildBVHRecurse(const scene_rdl2::math::BBox3f& bbox, std::vector<Face>& faces,
1073 int start, int end, int parentIndex)
1074{
1075 if ((end - start) < sBucketsCount) {
1076
1077 //
1078 // If we only have a few primitives in the node, fallback to a naive node
1079 // partition
1080 //
1081
1082 int nodeIndex = mBVH.size();
1083 Node node;
1084 mBVH.push_back(node);
1085 if ((end - start) > 1) {
1086 // compute energy of node
1087 float energy = 0.0f;
1088 for (int i = start; i < end; ++i) {
1089 energy += faces[i].mEnergy;
1090 }
1091
1092 // compute bounding cone of the node
1093 Cone cone = {faces[start].mNormal, faces[start].mThetaO};
1094 for (int i = start + 1; i < end; ++i) {
1095 const Cone faceCone = {faces[i].mNormal, faces[i].mThetaO};
1096 cone = combineBoundingCones(&cone, &faceCone);
1097 }
1098
1099 // fill in node parameters
1100 mBVH[nodeIndex].mParentIndex = parentIndex;
1101 mBVH[nodeIndex].mBbox = bbox;
1102 mBVH[nodeIndex].mBcone = cone;
1103 mBVH[nodeIndex].mEnergy = energy;
1104 mBVH[nodeIndex].mFace = nullptr;
1105
1106 // the longest length of the bounding box
1107 int splitAxis = maxDim(bbox.size());
1108
1109 // split node into left and right nodes
1110 scene_rdl2::math::BBox3f leftBound;
1111 scene_rdl2::math::BBox3f rightBound;
1112 int mid = naiveSplit(faces, start, end, splitAxis, leftBound, rightBound);
1113
1114 // recursively build left branch
1115 buildBVHRecurse(leftBound, faces, start, mid, nodeIndex);
1116 // recursively build right branch
1117
1118 int rightChildIndex = buildBVHRecurse(rightBound, faces, mid, end, nodeIndex);
1119 mBVH[nodeIndex].mRightChildIndex = rightChildIndex;
1120
1121 } else {
1122 // leaf node
1123 mBVH[nodeIndex].mParentIndex = parentIndex;
1124 mBVH[nodeIndex].mEnergy = faces[start].mEnergy;
1125 mBVH[nodeIndex].mFace = &faces[start];
1126 mBVH[nodeIndex].mBbox = bbox;
1127 mBVH[nodeIndex].mBcone = {faces[start].mNormal, faces[start].mThetaO};
1128 // right child nodeIndex is a dummy value that is never used for leaf nodes
1129 mBVH[nodeIndex].mRightChildIndex = -1;
1130 // Map the leaf node ids to the bvh node index
1131 unsigned faceOffset = mFaceOffset[faces[start].mGeomID] + faces[start].mPrimID;
1132 mPrimIDToNodeID[faceOffset] = nodeIndex;
1133 }
1134
1135 return nodeIndex;
1136 }
1137 //
1138 // Split According to the Surface Area Orientation Heuristic
1139 //
1140
1141 // Getting a bound around the centroids of the faces was a recommendation from
1142 // "On fast Construction of SAH-based Bounding Volume Hierarchies"
1143 // by Ingo Wald (2007).
1144 scene_rdl2::math::BBox3f centroidBbox(scene_rdl2::util::empty);
1145 for (int i = start; i < end; ++i) {
1146 centroidBbox.extend(faces[i].mCenter);
1147 }
1148 const scene_rdl2::math::Vec3f& centroidBboxSize = centroidBbox.size();
1149 const scene_rdl2::math::Vec3f& pMin = centroidBbox.lower;
1150
1151 // Initialize the split index of the list of primitives as the start index
1152 int mid = start;
1153
1154 // We iterate over the x, y, and z axes and compute the split cost for each
1155 // So we must keep track of the split cost and the left / right bounding
1156 // regions for each axis.
1157 float minCosts[3];
1158 std::pair<scene_rdl2::math::BBox3f, scene_rdl2::math::BBox3f> minBboxes[3];
1159 std::pair<Cone, Cone> minBcones[3];
1160
1161 // iterate over all 3 coordinate axes
1162 for (int splitAxis = 0; splitAxis < 3; splitAxis++) {
1163 if (centroidBboxSize[splitAxis] == 0.f) {
1164 // The centroids of all the faces are perfectly aligned along
1165 // this axis. We cannot attempt to split along this axis.
1166 // Give the axis infinite cost and continue.
1167 minCosts[splitAxis] = math::pos_inf;
1168 continue;
1169 }
1170
1171 // initialize the minBboxes to empty
1172 minBboxes[splitAxis] = {scene_rdl2::util::empty, scene_rdl2::util::empty};
1173
1174 // Spatially bin primitives into buckets
1175 Bucket buckets[sBucketsCount];
1176 float stride = centroidBboxSize[splitAxis] / sBucketsCount;
1177 for (int i = start; i < end; ++i) {
1178 const Face &face = faces[i];
1179 int32_t bucketIndex = scene_rdl2::math::clamp(std::floor((face.mCenter[splitAxis] - pMin[splitAxis]) /
1180 stride), 0.0f, (float) (sBucketsCount - 1));
1181
1182 buckets[bucketIndex].mEnergy += face.mEnergy;
1183 if (buckets[bucketIndex].mBbox.empty()) {
1184 // if this bucket is empty, initialize with the first bounding cone
1185 buckets[bucketIndex].mBcone = {face.mNormal, faces[i].mThetaO};
1186 } else {
1187 const Cone faceCone = {faces[i].mNormal, faces[i].mThetaO};
1188 buckets[bucketIndex].mBcone = combineBoundingCones(&buckets[bucketIndex].mBcone,
1189 &faceCone);
1190 }
1191 buckets[bucketIndex].mBbox.extend(face.mCenter);
1192 }
1193
1194 // Some buckets maybe empty. Remove those.
1195 Bucket finalBuckets[sBucketsCount];
1196 int bucketsCount = 0;
1197 for (int i = 0; i < sBucketsCount; ++i) {
1198 if (!buckets[i].mBbox.empty()) {
1199 finalBuckets[bucketsCount++] = buckets[i];
1200 }
1201 }
1202
1203 // there should be at least 2 buckets at this point
1204 MNRY_ASSERT(bucketsCount >= 2);
1205
1206 // the splitCoefficienct is a regularizing term that penalizes splitting
1207 // across axes with smaller lengths.
1208 float splitCoefficient = centroidBboxSize[maxDim(centroidBboxSize)] /
1209 centroidBboxSize[splitAxis];
1210
1211 // Then split the node into left and right nodes along the splitAxis.
1212 if (bucketsCount == 2) {
1213 // only two buckets, must split between them
1214 minBboxes[splitAxis] = {finalBuckets[0].mBbox, finalBuckets[1].mBbox};
1215 minBcones[splitAxis] = {finalBuckets[0].mBcone, finalBuckets[1].mBcone};
1216
1217 // compute split cost
1218 SplitCost cost;
1219 cost.mLeftEnergy = finalBuckets[0].mEnergy;
1220 cost.mLeftBbox = finalBuckets[0].mBbox;
1221 cost.mLeftBcone = finalBuckets[0].mBcone;
1222 cost.mRightEnergy = finalBuckets[1].mEnergy;
1223 cost.mRightBbox = finalBuckets[1].mBbox;
1224 cost.mRightBcone = finalBuckets[1].mBcone;
1225 minCosts[splitAxis] = cost.eval(splitCoefficient);
1226
1227 } else {
1228
1229 // The number of candidate splits is splitsCount. We compute the
1230 // cost of each candidate split and keep the minimum cost.
1231 int splitsCount = bucketsCount - 1;
1232 SplitCost costs[splitsCount];
1233
1234 // Fill in the splitCost structs
1235 costs[0].mLeftEnergy = finalBuckets[0].mEnergy;
1236 costs[0].mLeftBbox = finalBuckets[0].mBbox;
1237 costs[0].mLeftBcone = finalBuckets[0].mBcone;
1238 costs[splitsCount - 1].mRightEnergy = finalBuckets[bucketsCount - 1].mEnergy;
1239 costs[splitsCount - 1].mRightBbox = finalBuckets[bucketsCount - 1].mBbox;
1240 costs[splitsCount - 1].mRightBcone = finalBuckets[bucketsCount - 1].mBcone;
1241
1242 for (int i = 1; i < splitsCount; ++i) {
1243 costs[i].mLeftEnergy = finalBuckets[i].mEnergy + costs[i - 1].mLeftEnergy;
1244 costs[i].mLeftBbox = finalBuckets[i].mBbox;
1245 costs[i].mLeftBbox.extend(costs[i - 1].mLeftBbox);
1246 costs[i].mLeftBcone = combineBoundingCones(&finalBuckets[i].mBcone,
1247 &costs[i - 1].mLeftBcone);
1248 }
1249
1250 for (int i = splitsCount - 2; i >= 0; --i) {
1251 costs[i].mRightEnergy = finalBuckets[i + 1].mEnergy + costs[i + 1].mRightEnergy;
1252 costs[i].mRightBbox = finalBuckets[i + 1].mBbox;
1253 costs[i].mRightBbox.extend(costs[i + 1].mRightBbox);
1254 costs[i].mRightBcone = combineBoundingCones(&finalBuckets[i + 1].mBcone,
1255 &costs[i + 1].mRightBcone);
1256 }
1257
1258 // compute min split cost
1259 float minCost = costs[0].eval(splitCoefficient);
1260 int minCostSplit = 0;
1261 for (int i = 1; i < splitsCount; ++i) {
1262 float splitCost = costs[i].eval(splitCoefficient);
1263 if (splitCost < minCost) {
1264 minCost = splitCost;
1265 minCostSplit = i;
1266 }
1267 }
1268
1269 // store the cost and bounds for the minimum cost split
1270 minCosts[splitAxis] = minCost;
1271 minBboxes[splitAxis] = {costs[minCostSplit].mLeftBbox,
1272 costs[minCostSplit].mRightBbox};
1273 minBcones[splitAxis] = {costs[minCostSplit].mLeftBcone,
1274 costs[minCostSplit].mRightBcone};
1275 }
1276 }
1277
1278 // Get the axis with the minimum split cost
1279 int splitAxis = 0;
1280 float minCost = minCosts[splitAxis];
1281 for (int i = 1; i < 3; ++i) {
1282 if (minCosts[i] < minCost) {
1283 splitAxis = i;
1284 minCost = minCosts[i];
1285 }
1286 }
1287
1288 scene_rdl2::math::BBox3f leftBound = minBboxes[splitAxis].first;
1289 scene_rdl2::math::BBox3f rightBound = minBboxes[splitAxis].second;
1290
1291 // partition the faces into left and right nodes
1292 const auto& pMid = std::partition(faces.begin() + start, faces.begin() + end,
1293 [&](const Face &face) {
1294 return face.mCenter[splitAxis] <= leftBound.upper[splitAxis];
1295 }
1296 );
1297
1298 mid = pMid - faces.begin();
1299
1300 // If we did not successfully partition the faces, fall back to a naive partition.
1301 // There is no known scenario where the partition would be unsuccessful, so there
1302 // is an assert here. However, it is best to have a fallback just in case there
1303 // is a splitting scenario not already accounted for.
1304 Cone cone;
1305 if (mid == start || mid == end) {
1306 MNRY_ASSERT(0);
1307 splitAxis = maxDim(centroidBboxSize);
1308 mid = naiveSplit(faces, start, end, splitAxis, leftBound, rightBound);
1309 // compute bounding cone of this node
1310 cone = {faces[start].mNormal, faces[start].mThetaO};
1311 for (int i = start + 1; i < end; ++i) {
1312 const Cone faceCone = {faces[i].mNormal, faces[i].mThetaO};
1313 cone = combineBoundingCones(&cone, &faceCone);
1314 }
1315 } else {
1316 // We partitioned the centers of the faces into left and right bboxes
1317 // We must extend the bboxes to include the corners of the face.
1318
1319 // number of motion blur steps
1320 float dt = mDeformationMb ? 1.0f / (mMbSteps - 1) : 0.f;
1321
1322 for (int i = start; i < mid; ++i) {
1323 unsigned fvc = getFaceVertexCount(faces[i]);
1324 for (unsigned j = 0; j < fvc; ++j) {
1325 for (unsigned t = 0; t < mMbSteps; ++t) {
1326 float time = dt * t;
1327 leftBound.extend(getFaceVertex(faces[i], j, time));
1328 }
1329 }
1330 }
1331
1332 for (int i = mid; i < end; ++i) {
1333 unsigned fvc = getFaceVertexCount(faces[i]);
1334 for (unsigned j = 0; j < fvc; ++j) {
1335 for (unsigned t = 0; t < mMbSteps; ++t) {
1336 float time = dt * t;
1337 rightBound.extend(getFaceVertex(faces[i], j, time));
1338 }
1339 }
1340 }
1341 // compute bounding cone of this node
1342 cone = combineBoundingCones(&minBcones[splitAxis].first, &minBcones[splitAxis].second);
1343 }
1344
1345 // create node
1346 Node node;
1347 node.mParentIndex = parentIndex;
1348 node.mBbox = bbox;
1349 node.mBcone = cone;
1350 node.mFace = nullptr;
1351 node.mEnergy = 0.0f;
1352 for (int i = start; i < end; ++i) {
1353 node.mEnergy += faces[i].mEnergy;
1354 }
1355
1356 int32_t nodeIndex = mBVH.size();
1357 mBVH.push_back(node);
1358
1359 // recursively build left branch
1360 buildBVHRecurse(leftBound, faces, start, mid, nodeIndex);
1361 // recursively build right branch
1362 int rightChildIndex = buildBVHRecurse(rightBound, faces, mid, end, nodeIndex);
1363 mBVH[nodeIndex].mRightChildIndex = rightChildIndex;
1364
1365 return nodeIndex;
1366}

1370{
1371 // TODO: Is there anything else that should be verified?
1372 const Node& node = mBVH[index];
1373 if (node.isLeaf()) {
1374 return true;
1375 } else {

1382 bbox.upper.x >= leftBbox.upper.x &&
1383 bbox.upper.y >= leftBbox.upper.y &&

1388 bbox.upper.x >= rightBbox.upper.x &&

1390 bbox.upper.z >= rightBbox.upper.z);

1401float
1402MeshLight::importance(const scene_rdl2::math::Vec3f& shadingPoint, const scene_rdl2::math::Vec3f* shadingNormal, const Node& node) const
1403{
1404 // Early out if this node has no energy. This would happen if the MapShader
1405 // is black for all the faces contained in this node.
1406 if (node.mEnergy == 0.0f) {
1407 return 0.0f;
1408 }
1409
1410 scene_rdl2::math::Vec3f center = node.getCenter();
1411 // the vector from the shadingPoint to the center of the node
1412 scene_rdl2::math::Vec3f point2center = center - shadingPoint;
1413 float distance2 = lengthSqr(point2center);
1414
1415 // orientation metric
1416 // For more deatils see Kulla (2017) p. 8 Fig. 7 and Eq. 3.
1417 float orientation = 1.0f;
1418 if (!scene_rdl2::math::isEqual(node.mBcone.mThetaO, scene_rdl2::math::sPi) && distance2 != 0) {
1419 float distance = scene_rdl2::math::sqrt(distance2);
1420 point2center /= distance;
1421 // thetaU is the uncertainty angle from the center of the box to its edge
1422 float thetaU = 0.0f;
1423 // radius of the sphere that circumscribes the bounding box
1424 float radius = length(node.mBbox.upper - center);
1425
1426 if (distance >= radius) {
1427 // the shading point is outside the bounding sphere
1428 thetaU = scene_rdl2::math::asin(radius / distance);
1429 } else {
1430 // shading point is inside the bounding sphere
1431 thetaU = scene_rdl2::math::sPi;
1432 }
1433 // thetaI is the angle between the axis and the shading normal
1434 float thetaI = 0.0f;
1435 float shadingNormalContribution = 1.0f;
1436 if (shadingNormal) {
1437 const scene_rdl2::math::Vec3f& normal = *shadingNormal;
1438 thetaI = scene_rdl2::math::acos(scene_rdl2::math::clamp(dot(point2center, normal), -1.0f, 1.0f));
1439 shadingNormalContribution = scene_rdl2::math::abs(scene_rdl2::math::cos(scene_rdl2::math::max(thetaI - thetaU, 0.0f)));
1440 }
1441 // theta is the angle between the bounding cone axis and the center to shading point axis
1442 float theta = scene_rdl2::math::acos(-dot(node.mBcone.mAxis, point2center));
1443 float thetaPrime = scene_rdl2::math::max(theta - node.mBcone.mThetaO - thetaU, 0.0f);
1444 // For the mesh light, thetaE is always the same. For the many lights
1445 // solution we would need to compare against node.mBcone.mThetaE
1446 bool useThetaPrime = thetaPrime < sThetaE;
1447 orientation = shadingNormalContribution * (useThetaPrime ? scene_rdl2::math::cos(thetaPrime) : 0.0f);
1448 }
1449
1450 return node.mEnergy * orientation / distance2;
1451}

1453int
1454MeshLight::drawSampleRecurse(const scene_rdl2::math::Vec3f& shadingPoint, const scene_rdl2::math::Vec3f* shadingNormal,
1455 int currentNodeIndex, float u, float& pdf) const
1456{
1457 const Node& node = mBVH[currentNodeIndex];
1458 if (node.isLeaf()) {
1459 // The face is selected. Multiply by the inverse area of the face to get
1460 // the final pdf = pdf(face) * pdf(point | face)
1461 pdf *= node.mFace->mInvArea;
1462 return currentNodeIndex;
1463 }
1464
1465 int32_t leftIndex = currentNodeIndex + 1;
1466 const Node& leftChild = mBVH[leftIndex];
1467 int32_t rightIndex = node.mRightChildIndex;
1468 const Node& rightChild = mBVH[rightIndex];
1469
1470 float leftImportance = importance(shadingPoint, shadingNormal, leftChild);
1471 float rightImportance = importance(shadingPoint, shadingNormal, rightChild);
1472
1473 MNRY_ASSERT(leftImportance >= 0.0f);
1474 MNRY_ASSERT(rightImportance >= 0.0f);
1475
1476 float pdfL;
1477 if (leftImportance == 0 && rightImportance == 0) {
1478 // edge case
1479 pdfL = 0.5;
1480 } else {
1481 pdfL = leftImportance / (leftImportance + rightImportance);
1482 }
1483
1484 float pdfR = 1.0f - pdfL;
1485
1486 // During the traversal of the BVH, we my encounter situations where
1487 // u, pdfL, or pdfR equals exactly 1 or exactly 0.
1488 // If u == 1 and pdfL == 1, we want to traverse the left branch.
1489 // But if u == 0 and pdfL == 0, we want to traverse the right branch.
1490 if (u <= pdfL && pdfL != 0) {
1491 // traverse left branch
1492 float uRemapped = u / pdfL;
1493 pdf *= pdfL;
1494 return drawSampleRecurse(shadingPoint, shadingNormal, leftIndex,
1495 uRemapped, pdf);
1496 } else {
1497 // traverse right branch
1498 float uRemapped = (u - pdfL) / pdfR;
1499 pdf *= pdfR;
1500 return drawSampleRecurse(shadingPoint, shadingNormal, rightIndex,
1501 uRemapped, pdf);
1502 }
1503}
1504
1505float
1506MeshLight::getPdfOfFace(size_t nodeID, const scene_rdl2::math::Vec3f& p, const scene_rdl2::math::Vec3f* n) const
1507{
1508 // compute pdf by traversing backwards through the bvh and computing
1509 // the importance of each traversed node and its sibling.
1510 MNRY_ASSERT(mBVH[nodeID].isLeaf());
1511 float pdf = 1.0f;
1512 unsigned currNodeIndex = nodeID;
1513 while (currNodeIndex > 0) {
1514 Node currNode = mBVH[currNodeIndex];
1515 MNRY_ASSERT(currNode.mParentIndex < (int)mBVH.size() ||
1516 (currNode.mParentIndex == -1 && currNodeIndex == 0));
1517 Node siblingNode;
1518 const Node& parentNode = mBVH[currNode.mParentIndex];
1519 if (parentNode.mRightChildIndex == currNodeIndex) {
1520 // currNode is the right child, sibling is the left child
1521 siblingNode = mBVH[currNode.mParentIndex + 1];
1522 } else {
1523 // currNode is the left child, sibling is the right child
1524 MNRY_ASSERT(currNodeIndex == currNode.mParentIndex + 1);
1525 siblingNode = mBVH[parentNode.mRightChildIndex];
1526 }
1527
1528 float currImportance = importance(p, n, currNode);
1529 float siblingImportance = importance(p, n, siblingNode);
1530
1531 MNRY_ASSERT(currImportance >= 0.0f);
1532 MNRY_ASSERT(siblingImportance >= 0.0f);
1533
1534 if (currImportance == 0.0f && siblingImportance == 0.0f) {
1535 pdf *= 0.5;
1536 } else {
1537 pdf *= currImportance / (currImportance + siblingImportance);
1538 }
1539
1540 currNodeIndex = currNode.mParentIndex;
1541 }
1542
1543 return pdf;
1544}

1546bool
1547MeshLight::intersect(const scene_rdl2::math::Vec3f &p, const scene_rdl2::math::Vec3f *n, const scene_rdl2::math::Vec3f &wi, float time, float maxDistance,
1548 LightIntersection &isect) const
1549{
1550 RTCRayHit rayHit;
1551
1552 // transform ray origin and direction from render space to the light's local
1553 // space because the vertex buffer is in the light's local space
1554 const scene_rdl2::math::Vec3f transformedP = xformPointRender2Local(p, time);
1555 const scene_rdl2::math::Vec3f transformedWi = xformVectorRender2LocalRot(wi, time);
1556
1557 rayHit.ray.org_x = transformedP.x;
1558 rayHit.ray.org_y = transformedP.y;
1559 rayHit.ray.org_z = transformedP.z;
1560 rayHit.ray.tnear = 0.0f;
1561 rayHit.ray.dir_x = transformedWi.x;
1562 rayHit.ray.dir_y = transformedWi.y;
1563 rayHit.ray.dir_z = transformedWi.z;
1564 rayHit.ray.time = time;
1565 rayHit.ray.tfar = maxDistance;
1566 rayHit.ray.mask = ~0u;
1567 rayHit.ray.id = 0;
1568 rayHit.hit.geomID = RTC_INVALID_GEOMETRY_ID;
1569 rayHit.hit.primID = RTC_INVALID_GEOMETRY_ID;
1570 rayHit.hit.instID[0] = RTC_INVALID_GEOMETRY_ID;
1571
1572 // Call Embree intersection test
1573 LightIntersectContext context;
1574 rtcIntersect1(mRtcScene, &context.mRtcContext, &rayHit);
1575
1576 if (rayHit.hit.geomID == RTC_INVALID_GEOMETRY_ID) {
1577 return false;
1578 }
1579
1580 // compute pdf of this intersection.
1581 // pdf = pdf (point | face) * pdf(face)
1582 int faceOffset = mFaceOffset[rayHit.hit.geomID] + rayHit.hit.primID;
1583 int nodeIndex = mPrimIDToNodeID[faceOffset];
1584 float pdf = 1.0f;
1585 if (n) {
1586 // transform shading point's normal from render space to the light's
1587 // local space.
1588 const scene_rdl2::math::Vec3f transformedN = xformNormalRender2LocalRot(*n, time);
1589 pdf = mBVH[nodeIndex].mFace->mInvArea * getPdfOfFace(nodeIndex, transformedP, &transformedN);
1590 } else {
1591 pdf = mBVH[nodeIndex].mFace->mInvArea * getPdfOfFace(nodeIndex, transformedP, nullptr);
1592 }
1593
1594 // Fill in isect members
1595 isect.N = xformNormalLocal2RenderRot(mBVH[nodeIndex].mFace->mNormal, time);
1596 isect.uv = scene_rdl2::math::Vec2f(rayHit.hit.u, rayHit.hit.v);
1597 isect.distance = xformLocal2RenderScale(rayHit.ray.tfar, time);
1598 isect.pdf = pdf;
1599 isect.primID = rayHit.hit.primID;
1600 isect.geomID = rayHit.hit.geomID;
1601
1602 return true;
1603}

1605bool
1606MeshLight::sample(const scene_rdl2::math::Vec3f &p, const scene_rdl2::math::Vec3f *n, float time, const scene_rdl2::math::Vec3f& r,
1607 scene_rdl2::math::Vec3f &wi, LightIntersection &isect, float rayDirFootprint) const
1608{
1609 MNRY_ASSERT(mOn);
1610
1611 const float r1 = r[0];
1612 const float r2 = r[1];
1613 const float r3 = r[2];
1614
1615 float pdf = 1.0f;
1616 int faceIndex;
1617 // transform shading point's position and normal from render space to the
1618 // light's local space, because the vertex buffer is in the light's local space.
1619 const scene_rdl2::math::Vec3f transformedP = xformPointRender2Local(p, time);
1620 if (n) {
1621 const scene_rdl2::math::Vec3f transformedN = xformNormalRender2LocalRot(*n, time);
1622 faceIndex = drawSampleRecurse(transformedP, &transformedN, 0, r3, pdf);
1623 } else {
1624 faceIndex = drawSampleRecurse(transformedP, nullptr, 0, r3, pdf);
1625 }
1626
1627 MNRY_ASSERT(scene_rdl2::math::isfinite(pdf) && pdf >= 0.0f);
1628 MNRY_ASSERT(mBVH[faceIndex].isLeaf());
1629 const Face& face = *mBVH[faceIndex].mFace;
1630
1631 scene_rdl2::math::Vec2f uv;
1632 scene_rdl2::math::Vec3f hit;
1633 scene_rdl2::math::Vec3f normal;
1634
1635 if (getFaceVertexCount(face) == 3 || getFaceVertex(face, 2, time) == getFaceVertex(face, 3, time)) {
1636 // triangle or degenerate quad case
1637 scene_rdl2::math::Vec3f p1 = getFaceVertex(face, 0, time);
1638 scene_rdl2::math::Vec3f p2 = getFaceVertex(face, 1, time);
1639 scene_rdl2::math::Vec3f p3 = getFaceVertex(face, 2, time);
1640
1641 // the random numbers r1 and r2 can be used as uv coordinates
1642 float u = r1;
1643 float v = r2;
1644 float w = 1 - u - v;
1645 if (w < 0) {
1646 // if outside the triangle, flip the uvs to be inside the triangle
1647 u = 1 - u;
1648 v = 1 - v;
1649 w = -w;
1650 }
1651 hit = w*p1 + u*p2 + v*p3;
1652
1653 if (mDeformationMb) {
1654 // We cannot directly use face.mNormal here because it is the normal at
1655 // time = centroidTime. Therefore we compute it here.
1656 normal = normalize(cross(p2 - p1, p3 - p1));
1657 } else {
1658 normal = face.mNormal;
1659 }
1660
1661 if (mMapShader) {
1662 // get barycentric coordinate for hit
1663 uv.x = u;
1664 uv.y = v;
1665 }
1666 } else {
1667 // quad case
1668 MNRY_ASSERT(getFaceVertexCount(face) == 4);
1669
1670 scene_rdl2::math::Vec3f v0 = getFaceVertex(face, 0, time);
1671 scene_rdl2::math::Vec3f v1 = getFaceVertex(face, 1, time);
1672 scene_rdl2::math::Vec3f v2 = getFaceVertex(face, 2, time);
1673 scene_rdl2::math::Vec3f v3 = getFaceVertex(face, 3, time);
1674
1675 scene_rdl2::math::Vec3f normal013 = cross(v1 - v0, v3 - v0);
1676 scene_rdl2::math::Vec3f normal231 = cross(v3 - v2, v1 - v2);
1677 float area013 = length(normal013);
1678 float area231 = length(normal231);
1679 float areaSum = area013 + area231;
1680 float rSplit = area013 / areaSum;
1681
1682 scene_rdl2::math::Vec3f p1;
1683 scene_rdl2::math::Vec3f p2;
1684 scene_rdl2::math::Vec3f p3;
1685 float u;
1686
1687 // Choose which triangle according to the area of the triangles
1688 if (r1 < rSplit) {
1689 // first triangle
1690 u = r1 / rSplit;
1691 p1 = v0;
1692 p2 = v1;
1693 p3 = v3;
1694 normal = normal013 / area013;
1695 } else {
1696 // second triangle
1697 u = (r1 - rSplit) / (1 - rSplit);
1698 p1 = v2;
1699 p2 = v3;
1700 p3 = v1;
1701 normal = normal231 / area231;
1702 }
1703
1704 float v = r2;
1705 float w = 1 - u - v;
1706 if (w < 0) {
1707 // if outside the triangle, flip the uvs to be inside the triangle
1708 u = 1 - u;
1709 v = 1 - v;
1710 w = -w;
1711 }
1712 hit = w*p1 + u*p2 + v*p3;
1713
1714 if (mMapShader) {
1715 if (r1 < rSplit) {
1716 // first triangle
1717 uv.x = u;
1718 uv.y = v;
1719 } else {
1720 // second triangle
1721 uv.x = 1 - u;
1722 uv.y = 1 - v;
1723 }
1724 }
1725 }
1726
1727 MNRY_ASSERT(isFinite(hit));
1728
1729 // compute wi and hit distance
1730 wi = xformVectorLocal2Render(hit - transformedP, time);
1731 isect.distance = length(wi);
