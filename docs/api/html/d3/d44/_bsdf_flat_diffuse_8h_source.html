29static finline float
30getOrenNayarFalloffIntensity(const scene_rdl2::math::Vec3f& n,
31 const scene_rdl2::math::Vec3f& v,
32 const scene_rdl2::math::Vec3f& l,
33 const float a,
34 const float b,
35 const float terminatorShift)
36{
37 // We are using the Fujii implementation or Oren-Nayar.
38 // The PBRT implementation was also tested but resulted
39 // in more energy loss.
40 // https://mimosa-pudica.net/improved-oren-nayar.html
41 // Note: For the flat diffuse lobe type we introduce an
42 // additional terminatorShift control that reshapes the n dot l
43 // response
44 const float nl = scene_rdl2::math::max(
45 0.0f, scene_rdl2::math::cos(
46 scene_rdl2::math::min(
47 scene_rdl2::math::sHalfPi, scene_rdl2::math::acos(
48 scene_rdl2::math::clamp(scene_rdl2::math::dot(n, l), -1.0f, 1.0f)) *
49 terminatorShift)));
50 const float nv = scene_rdl2::math::max(scene_rdl2::math::dot(n, v), 0.0f);
51 const float s = scene_rdl2::math::dot(l, v) - nl * nv;
52
53 float t = 1.0f;
54 if (s > 0.0f) {
55 t = scene_rdl2::math::max(nl, nv) + scene_rdl2::math::sEpsilon;
56 }
57
58 return nl * (a + b * s/t);
59}

61///
62/// @class FlatDiffuseBsdfLobe BsdfFlatDiffuse.h <shading/bsdf/npr/BsdfFlatDiffuse.h>
63/// @brief This is a totally made-up NPR "flat diffuse" BRDF that
64/// matches Oren-Nayar when flatness = 0. It works by bending
65/// the normal towards the light as flatness goes from 0 to 1.
66///
67class FlatDiffuseBsdfLobe : public OrenNayarBsdfLobe
68{
69public:
70 // Constructor / Destructor
71 FlatDiffuseBsdfLobe(const scene_rdl2::math::Vec3f &N,
72 const scene_rdl2::math::Color& albedo,
73 float roughness,
74 float terminatorShift,
75 float flatness,
76 float flatnessFalloff,
77 bool reflection) :
78 OrenNayarBsdfLobe(N, albedo, roughness, reflection),
79 mTerminatorShift(terminatorShift),
80 mFlatness(flatness),
81 mFlatnessFalloff(flatnessFalloff)
82 {}
83
84 // BsdfLobe API
85 finline scene_rdl2::math::Color eval(const BsdfSlice &slice, const scene_rdl2::math::Vec3f &wi, float *pdf = NULL) const override
86 {
87 // When flatness approaches 1.0 it tends to produce an extremely harsh terminator,
88 // which can lead to polygonal artifacts where self-shadowing occurs. This remapping
89 // attempts to apply a targeted softening effect at the terminator - similar to the
90 // "shadow terminator fix" that we used to soften bump mapping
91
92 // Calculate the flatness falloff off by remapping
93 // the angle between the light and the normal.
94 // Desmos expression:
95 // https://www.desmos.com/calculator/z1iufecqmf
96 const float terminatorShift = mTerminatorShift + 1.0f;
97 const float theta = scene_rdl2::math::min(scene_rdl2::math::sHalfPi, terminatorShift *
98 scene_rdl2::math::acos(scene_rdl2::math::clamp(scene_rdl2::math::dot(mFrame.getN(), wi), -1.0f, 1.0f)));
99 const float a = 1.0f - mFlatnessFalloff;
100 // The power of "a" controls the linearity of the falloff parameter.
101 // Raising to the power of 4 slightly biases the control towards
102 // the low end giving more control over a low falloff value. The
103 // constant 100.0 controls how sharply the falloff occurs when the
104 // parameter is set to 1.0.
105 const float b = 1.0f + a * a * a * a * 100.0f;
106 const float c = scene_rdl2::math::sHalfPi / b;
107 const float d = c - scene_rdl2::math::sHalfPi;
108 const float t = (theta > scene_rdl2::math::sHalfPi - c) ? scene_rdl2::math::cos((theta + d) * b) : 1.0f;
109 const float flatness = mFlatness * t;
110 const scene_rdl2::math::Vec3f flatN = scene_rdl2::math::normalize(scene_rdl2::math::lerp(mFrame.getN(), wi, flatness));
111
112 const float cosThetaWi = getOrenNayarFalloffIntensity(flatN,
113 slice.getWo(), wi,
114 mA, mB,
115 terminatorShift);
116
117 if (pdf != NULL) {
118 *pdf = cosThetaWi * scene_rdl2::math::sOneOverPi;
119 }
120
121 // Note: we assume this lobe has been setup with a OneMinus*Fresnel
122 // as we want to use 1 - specular_fresnel. Also notice we use
123 // cosThetaWo to evaluate the fresnel term, as an approximation of what
124 // hDotWi would be for the specular lobe.
125 float cosThetaWo = 1.0f;
126 if (getFresnel()) {
127 const scene_rdl2::math::Vec3f N = (matchesFlag(REFLECTION)) ? flatN : -flatN;
128 cosThetaWo = scene_rdl2::math::max(dot(N, slice.getWo()), 0.0f);
129 }
130
131 // Soften hard shadow terminator due to shading normals
132 const float Gs = (matchesFlag(REFLECTION)) ?
133 slice.computeShadowTerminatorFix(flatN, wi) :
134 1.0f;
135
136 const scene_rdl2::math::Color result = mAlbedo * scene_rdl2::math::sOneOverPi * Gs *
137 computeScaleAndFresnel(cosThetaWo) *
138 (slice.getIncludeCosineTerm() ? cosThetaWi : 1.0f);
139
140 // This is an ad hoc factor that tries to minimize energy
141 // loss while also minimizing the difference in brightness
142 // compared with lambertian (i.e. flatnesss = 0).
143 const float normalizationFactor = scene_rdl2::math::lerp(1.0f,
144 0.75f,
145 scene_rdl2::math::bias(mFlatness, 0.75f));
146
147 return result * normalizationFactor;
148 }

150 void show(std::ostream& os, const std::string& indent) const override
151 {
152 const scene_rdl2::math::Color& scale = getScale();
153 const scene_rdl2::math::Vec3f& N = mFrame.getN();
154 const Fresnel * const fresnel = getFresnel();
155
156 os << indent << "[FlatDiffuseBsdfLobe] "
157 << (matchesFlag(REFLECTION) ? "(reflection)" : "(transmission)") << "\n";
158 os << indent << " " << "scale: "
159 << scale.r << " " << scale.g << " " << scale.b << "\n";
160 os << indent << " " << "N: "
161 << N.x << " " << N.y << " " << N.z << "\n";
162 os << indent << " " << "albedo: "
163 << mAlbedo.r << " " << mAlbedo.g << " " << mAlbedo.b << "\n";
164 os << indent << "roughness: " << mRoughness << "\n";
165 os << indent << "terminator shift: " << mTerminatorShift << "\n";
166 os << indent << "flatness: " << mFlatness << "\n";
167 os << indent << "flatness falloff: " << mFlatnessFalloff << "\n";
168 if (fresnel) {
169 fresnel->show(os, indent + " ");
170 }
171 }

177};
