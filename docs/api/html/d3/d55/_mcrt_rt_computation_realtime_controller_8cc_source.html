33void
34McrtRtComputationRealtimeController::markReceivedGeometryData()
35{
36 double oldTime = mReceivedGeometryDataTime;
37 mReceivedGeometryDataTime = util::getSeconds();
38
39 if (!mConstantFps) {
40 //
41 // dynamically update fps based on inteval of received geometry data when dynamicFps mode
42 //
43 double currInterval = (oldTime == 0.0)? 1.0 / mFps: mReceivedGeometryDataTime - oldTime;
44 mFps = computeFps(currInterval);
45 }
46}

50bool
51McrtRtComputationRealtimeController::isFrameComplete()
52{
53 double now = util::getSeconds(); // get current time
54
55 if (!mStartGapIntervalCondition) {
56 mGapStartTime = now; // start gap interval
57 mStartGapIntervalCondition = true;
58 }
59
60 {
61 //
62 // |<------- total time budge ------->|
63 // | (mPredictFrameInterval) |
64 // prev frame ....+......................+ . . . . . +.... next frame
65 // ^<---- render time --->^<-- gap -->^
66 // | | interval |
67 // +- frame start | |
68 // (mLastTime) | |
69 // | |
70 // render end --+ |
71 // (mGapStartTime) |
72 // |
73 // frame end --+
74 //
75 // 1 frame = <render time> + <gap interval>
76 //
77 // <render time> is from frame start ~ render end.
78 // <gap interval> is total time budget - render time.
79 // Sampling volume is estimated and start render.
80 // Sometimes it's finished before frame end (gap interval > 0) or
81 // overrun (gap interval is negative) depending on how accurate the estimation is.
82 // Close to zero gap interval is always good and zero is perfect.
83 //
84
85 //
86 // Try to find just frame complete moment
87 //
88 double frameInterval = (mLastTime == 0.0)? 1.0 / mFps : now - mLastTime;
89 double currOverrun = (mPredictFrameInterval == 0.0)? 0.0: frameInterval - mPredictFrameInterval;
90 if (currOverrun < 0.0) {
91 return false; // still inside gap interval. need to wait bit more
92 }
93
94 //
95 // Entire frame is completed. We are update some timing value here.
96 //
97 mLastTime = now; // frame complete time
98 mFrameInterval = frameInterval; // last frame's time length
99 mPureGapInterval = now - mGapStartTime; // compute gap interval
100 mOverrun = currOverrun; // overrun time length
101
102 mStartGapIntervalCondition = false; // reset gap interval condition
103
104 }
105
106 if (mReceivedGeometryDataTime > 0.0) {
107 //
108 // Try to compute adjustment time offset to move geometry data message receive timing as
109 // middle of render start and end.
110 //
111 double durationGeoEnd = now - mReceivedGeometryDataTime;
112 double currentAdjustDuration = (1.0 / mFps * 0.5) - durationGeoEnd; // use current fps
113 mAdjustDuration = computeAdjustDuration(currentAdjustDuration);
114 }
115
116 {
117 //
118 // Compute new gap interval value
119 //
120 // We've already averaged adjustDuration but try to apply one more scale here.
121 // This is very experimental but works and easy to adjust overall behavior.
122 //
123 double adjustScale = (mConstantFps)? 0.125: 0.5; // experimental scale value
124 double adjustDuration = - mOverrun + mAdjustDuration * adjustScale;
125 mPredictFrameInterval = 1.0 / mFps + adjustDuration;
126 mGapInterval = mPureGapInterval + adjustDuration;
127 }
128
129 return true;
130}
