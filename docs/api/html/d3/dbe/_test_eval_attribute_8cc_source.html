71Bsdfv *
72SceneSetup::shade()
73{
74 moonray::shading::TLState *tls = getShadingTLS();
75 alloc::Arena *arena = MNRY_VERIFY(tls)->mArena;
76
77 // mostly 0s with a few notable exceptions in lane0 and lane1
78 moonray::shading::Statev *statev = arena->alloc<moonray::shading::Statev>(CACHE_LINE_SIZE);
79 memset(statev, 0, sizeof(moonray::shading::Statev));
80 statev->mNg.x[0] = 1.0;
81 statev->mNg.y[1] = 1.0;
82 statev->mN.x[0] = 1.0;
83 statev->mN.y[1] = 1.0;
84 statev->mSt.x[0] = .5;
85 statev->mSt.x[1] = .25;
86 statev->mWo.x[0] = 1.0;
87 statev->mWo.x[1] = 1.0;
88 statev->mdPds.y[0] = 1.0; // orthogonal to Ng
89 statev->mdPds.x[1] = 1.0; // orthogonal to Ng
90 // now assume we have a valid lane0 and lane1, fill out the
91 // remaining lanes equal to lane1. it is an error to
92 // pass unitialized lanes into shading.
93 for (unsigned i = 2; i < VLEN; ++i) {
94 statev->mNg.x[i] = statev->mNg.x[1];
95 statev->mN.y[i] = statev->mN.y[1];
96 statev->mSt.x[i] = statev->mSt.x[1];
97 statev->mWo.x[i] = statev->mWo.x[1];
98 statev->mdPds.y[i] = statev->mdPds.y[1];
99 }
100
101 // pass in 1 AOSOA (statevs), return 1 AOSOA (bsdfvs)
102 Bsdfv *bsdfv = arena->alloc<Bsdfv>(CACHE_LINE_SIZE);
103
104 // Alloc an array of BsdfBuilderv
105 BsdfBuilderv *builderv = tls->mArena->allocArray<BsdfBuilderv>(1, CACHE_LINE_SIZE);
106
107 // helper to allow calling ispc function via func ptr below
108 typedef void (__cdecl * BsdfBuilderArrayInitFuncv)
109 (moonray::shading::TLState *tls,
110 unsigned numStatev,
111 const moonray::shading::Statev* state,
112 scene_rdl2::rdl2::Bsdfv* bsdfv,
113 moonray::shading::BsdfBuilderv* bsdfBuilderv,
114 SIMD_MASK_TYPE implicitMask);
115
116 // Initialize the BsdfBuilderv's (and the Bsdfv's) by calling ispc
117 // function via func ptr. This approach allows casting from c++ to
118 // ispc types:
119 // * moonray::shading::TLState -> ispc::ShadingTLState
120 // * moonray::shading::State -> ispc::State
121 BsdfBuilderArrayInitFuncv initBsdfBuilderArray = (BsdfBuilderArrayInitFuncv)ispc::getBsdfBuilderInitFunc();
122 initBsdfBuilderArray(tls, 1, statev, reinterpret_cast<scene_rdl2::rdl2::Bsdfv*>(bsdfv), builderv, scene_rdl2::util::sAllOnMask);
123
124 mMaterial->shadev(tls, 1, reinterpret_cast<const scene_rdl2::rdl2::Statev *>(statev), reinterpret_cast<scene_rdl2::rdl2::BsdfBuilderv *>(builderv));
125 CPPUNIT_ASSERT(bsdfv->mNumLobes == 2); // evalColor and
126 // evalColorComponent
127 return bsdfv;
128}

132void
133TestEvalAttribute::testEvalColor()
134{
135 SceneSetup scene;
136 Bsdfv &bsdf = *(scene.shade());
137
138 // evalColor test results are in lobe0
139 CPPUNIT_ASSERT(bsdf.mLobes[0]->mName == BsdfLobeName::BSDF_LOBE_LAMBERT);
140 LambertBsdfLobev &lobe = *(LambertBsdfLobev *) bsdf.mLobes[0];
141
142 // lane0
143 CPPUNIT_ASSERT(lobe.mScale.r[0] == 0.0 &&
144 lobe.mScale.g[0] == 0.5 &&
145 lobe.mScale.b[0] == 0.0);
146 CPPUNIT_ASSERT(lobe.mFrame.mZ.x[0] == 1.0 &&
147 lobe.mFrame.mZ.y[0] == 0.0 &&
148 lobe.mFrame.mZ.z[0] == 0.0);
149
150 // lane1
151 CPPUNIT_ASSERT(lobe.mScale.r[1] == 0.0 &&
152 lobe.mScale.g[1] == 0.25 &&
153 lobe.mScale.b[1] == 0.0);
154 CPPUNIT_ASSERT(lobe.mFrame.mZ.x[1] == 0.0 &&
155 lobe.mFrame.mZ.y[1] == 1.0 &&
156 lobe.mFrame.mZ.z[1] == 0.0);

159void
160TestEvalAttribute::testEvalColorComponent()
161{
162 SceneSetup scene;
163 Bsdfv &bsdf = *(scene.shade());
164
165 // evalColorComponent results are in lobe1
166 CPPUNIT_ASSERT(bsdf.mLobes[1]->mName == BsdfLobeName::BSDF_LOBE_COOK_TORRANCE);
167 CookTorranceBsdfLobev &lobe = *(CookTorranceBsdfLobev *) bsdf.mLobes[1];
168
169 // When testing mFrame we use isEqual() due to minor error
170 // introduced in adaptNormal() during lobe creation
171
172 const float tol = 0.006f;// things have gotten sloppier when moving to ispc-1.14.1
173
174 // lane0
175 CPPUNIT_ASSERT(lobe.mScale.r[0] == 0.25 &&
176 lobe.mScale.g[0] == 0.0 &&
177 lobe.mScale.b[0] == 0.0);
178 CPPUNIT_ASSERT(scene_rdl2::math::isEqual(lobe.mFrame.mZ.x[0], 1.f, tol) &&
179 scene_rdl2::math::isEqual(lobe.mFrame.mZ.y[0], 0.f, tol) &&
180 scene_rdl2::math::isEqual(lobe.mFrame.mZ.z[0], 0.f, tol));
181 CPPUNIT_ASSERT(lobe.mRoughness[0] == 1.0);
182
183 // lane1
184 CPPUNIT_ASSERT(lobe.mScale.r[1] == 0.125 &&
185 lobe.mScale.g[1] == 0.0 &&
186 lobe.mScale.b[1] == 0.0);
187 CPPUNIT_ASSERT(scene_rdl2::math::isEqual(lobe.mFrame.mZ.x[1], 0.f, tol) &&
188 scene_rdl2::math::isEqual(lobe.mFrame.mZ.y[1], 1.f, tol) &&
189 scene_rdl2::math::isEqual(lobe.mFrame.mZ.z[1], 0.f, tol));
190 CPPUNIT_ASSERT(lobe.mRoughness[1] == 1.0);
191}

431void
432TestEvalAttribute::testEvalNormal()
433{
434 moonray::shading::TLState *tls = getShadingTLS();
435
436 scene_rdl2::rdl2::SceneContext ctx;
437 ctx.setDsoPath(ctx.getDsoPath() + ":dso/map");
438 scene_rdl2::rdl2::Map *map = ctx.createSceneObject("TestMap", "/map")->asA<scene_rdl2::rdl2::Map>();
439 {
440 scene_rdl2::rdl2::SceneObject::UpdateGuard update(map);
441 map->set("mode", 10);
442 map->set("vec3f10", scene_rdl2::rdl2::Vec3f(1.0f, 2.0f, 2.0f));
443 }
444 scene_rdl2::rdl2::Map *map2 = ctx.createSceneObject("TestMap", "/map2")->asA<scene_rdl2::rdl2::Map>();
445 {
446 scene_rdl2::rdl2::SceneObject::UpdateGuard update(map2);
447 map2->set("mode", 11);
448 map2->set("float11", 0.5f);
449 map2->setBinding("vec3f11", map);
450 }
451 map2->applyUpdates();
452 {
453 moonray::shading::Statev statev;
454 const scene_rdl2::math::Vec3f wo = scene_rdl2::math::normalize(scene_rdl2::math::Vec3f(0.f, 1.f, 1.f));
455 for (int i = 0; i < (int) VLEN; ++i) {
456 statev.mN.x[i] = 0.0f;
457 statev.mN.y[i] = 1.0f;
458 statev.mN.z[i] = 0.0f;
459 statev.mdPds.x[i] = 0.0f;
460 statev.mdPds.y[i] = 0.0f;
461 statev.mdPds.z[i] = -1.0f;
462
463 // normal mapping now requires a view direction
464 statev.mWo.x[i] = wo.x;
465 statev.mWo.y[i] = wo.y;
466 statev.mWo.z[i] = wo.z;
467
468 // and a geometric normal
469 statev.mNg.x[i] = statev.mN.x[i];
470 statev.mNg.y[i] = statev.mN.y[i];
471 statev.mNg.z[i] = statev.mN.z[i];
472 }
473 moonray::shading::Colorv result;
474 moonray::shading::samplev(map2, tls, &statev, &result);
475 for (int i = 0; i < (int) VLEN; ++i) {
476 CPPUNIT_ASSERT(scene_rdl2::math::isEqual(result.r[i], -0.374551f));
477 CPPUNIT_ASSERT(scene_rdl2::math::isEqual(result.g[i], 0.918762f));
478 CPPUNIT_ASSERT(scene_rdl2::math::isEqual(result.b[i], -0.12485f));
479 }
480 }
