29TransmissionCookTorranceBsdfLobe::TransmissionCookTorranceBsdfLobe(const Vec3f &N,
30 float roughness,
31 float etaI,
32 float etaT,
33 const Color &tint,
34 float favg,
35 float favginv,
36 float abbeNumber) :
37 BsdfLobe(Type(TRANSMISSION | GLOSSY),
38 DifferentialFlags(0),
39 false,
40 PROPERTY_NORMAL | PROPERTY_ROUGHNESS),
41 mFrame(N),
42 mTint(tint),
43 mInputRoughness(roughness),
44 mEtaI(etaI),
45 mEtaT(etaT),
46 mNeta(mEtaI * rcp(mEtaT))
47
48{
49 mFavg = favg;
50 mFavgInv = favginv;
51 // Apply roughness squaring to linearize roughness response
52 // See "Physically-Based Shading at Disney" Siggraph course notes.
53 mRoughness = mInputRoughness * mInputRoughness;
54
55 if (mRoughness < 0.001f) {
56 mRoughness = 0.001f;
57 }
58
59 mInvRoughness = rcp(mRoughness);
60
61 if (scene_rdl2::math::isZero(abbeNumber)) {
62 mAllowDispersion = false;
63 } else {
64 shading::computeSpectralIOR(etaT,
65 abbeNumber,
66 &mEtaR,
67 &mEtaG,
68 &mEtaB);
69 mAllowDispersion = true;
70 }
71
72 // Use a directional differential scale that varies with roughness
73 mdDFactor = sdDFactorMin + mRoughness * sdDFactorSlope;
74}

78TransmissionCookTorranceBsdfLobe::eval(const BsdfSlice &slice,
79 const Vec3f &wi,
80 float *pdf) const
81{
82 // Based on the paper:
83 // "Microfacet Models for Refraction through Rough Surfaces"
84 // (Walter et al. - EGSR 2007)
85
86 // Prepare for early exit
87 if (pdf != NULL) {
88 *pdf = 0.0f;
89 }
90 return eval(slice, wi, mEtaT, pdf);

93Color
94TransmissionCookTorranceBsdfLobe::eval(const BsdfSlice &slice,
95 const Vec3f &wi,
96 float etaT,
97 float *pdf) const
98{
99 float cosNO = dot(mFrame.getN(), slice.getWo());
100 float cosNI = dot(mFrame.getN(), wi);
101
102 // Exclude cases where wo and wi point to the same side of the surface
103 if (cosNO * cosNI > 0.0f) {
104 return sBlack;
105 }
106
107 // Compute abs of cosines and exclude degenerate cases
108 cosNO = (cosNO > 0.0f ? min(cosNO, sOneMinusEpsilon)
109 : max(cosNO, -sOneMinusEpsilon));
110 cosNI = (cosNI > 0.0f ? min(cosNI, sOneMinusEpsilon)
111 : max(cosNI, -sOneMinusEpsilon));
112 const float absCosNO = scene_rdl2::math::abs(cosNO);
113 if (absCosNO <= sEpsilon) return sBlack;
114 const float absCosNI = scene_rdl2::math::abs(cosNI);
115 if (absCosNI <= sEpsilon) return sBlack;
116
117 // Compute microfacet / half vector, pointing towards the surface side
118 // with the lowest ior
119 Vec3f m;
120 if (!computeNormalizedRefractionHalfVector(mEtaI,
121 slice.getWo(),
122 etaT,
123 wi,
124 m)) {
125 return sBlack;
126 }
127
128 // Make sure microfacet points to the same side of the surface wrt. N and wo
129 m = (mEtaI > etaT ? -m : m);
130
131 // Exclude cases where microfacet m is pointing to the opposite side of
132 // the surface (wrt. N). This corresponds to the X+(m.n) term of the normal
133 // distribution function D(m); see eq. (25), (30) or (33)
134 const float cosThetaM = dot(m, mFrame.getN());
135 if (cosThetaM <= sEpsilon) return sBlack;
136
137 // Note: computing cosMI and cosMO after potentially flipping m may change
138 // their sign compared to the semantics in the paper, but this is necessary
139 // for the next exclusion test below, and doesn't affect the outcome of
140 // computing denom further below.
141 const float cosMI = dot(m, wi);
142 const float cosMO = dot(m, slice.getWo());
143
144 // Exclude cases where wo or wi is not on the same side as m and N.
145 // This corresponds to the X+(v, m, n) term of the shadow-masking
146 // function G1(v, m); see eq. (27) or (34)
147 if (cosNO * cosMO <= sEpsilon || cosNI * cosMI <= sEpsilon) {
148 return sBlack;
149 }
150
151 // By construction above (we flipped m towards wo), we know that
152 // cosMI < 0 and cosMO > 0, so we don't need to use abs()
153 const float absCosMI = -cosMI;
154 if (absCosMI <= sEpsilon) return sBlack;
155 const float absCosMO = cosMO;
156 if (absCosMO <= sEpsilon) return sBlack;
157
158
159 // eq. 25: calculate D(m):
160 const float alpha2 = mRoughness * mRoughness;
161 const float cosThetaM2 = cosThetaM * cosThetaM;
162 const float cosThetaM4 = cosThetaM2 * cosThetaM2;
163 const float minusTanThetaM2 = (cosThetaM2 - 1.0f) * rcp(cosThetaM2);
164 const float D = scene_rdl2::math::exp(minusTanThetaM2 * rcp(alpha2)) *
165 rcp(sPi * alpha2 * cosThetaM4);
166
167 // eq. 26, 27: now calculate G1(i,m) and G1(o,m)
168 const float ao = absCosNO * mInvRoughness * rsqrt(1.0f - cosNO * cosNO);
169 const float ai = absCosNI * mInvRoughness * rsqrt(1.0f - cosNI * cosNI);
170 const float G1o = (ao < 1.6f) ? (3.535f * ao + 2.181f * ao * ao) *
171 rcp(1.0f + 2.276f * ao + 2.577f * ao * ao) : 1.0f;
172 const float G1i = (ai < 1.6f) ? (3.535f * ai + 2.181f * ai * ai) *
173 rcp(1.0f + 2.276f * ai + 2.577f * ai * ai) : 1.0f;
174 const float G = G1o * G1i;
175
176 // Compute the Cook-Torrance bsdf, using equation (21)
177 // Note: we assume this lobe has been setup with a OneMinus*Fresnel
178 // as we want to use 1 - specular_fresnel. Also notice we use
179 // cosThetaWo to evaluate the fresnel term, as an approximation of what
180 // hDotWi would be for the specular lobe.
181// float denom = mEtaI * cosMO + etaT * cosMI;
182 float denom = mEtaI * cosMO + etaT * cosMI;
183 denom = rcp(denom * denom);
184 const float etaTSqr = etaT * etaT;
185 Color fresnel = computeScaleAndFresnel(absCosMO);
186 Color f = absCosMO * absCosMI * etaTSqr * denom *
187 fresnel * G * D * rcp(absCosNO) *
188 (slice.getIncludeCosineTerm() ? 1.0f : rcp(absCosNI));
189
190 f += TransmissionCookTorranceEnergyCompensation::evalT(
191 absCosNO, absCosNI,
192 mInputRoughness,
193 mEtaI, mEtaT,
194 mFavg,
195 mFavgInv,
196 slice.getIncludeCosineTerm());
197
198 // Compute pdf of sampling wi by solving for pdf(wi) == po(o) in eq. (41),
199 // by substituting fs(i,o,n) in eq. (41) with eq. (21).
200 // pdf(wi) = etaT^2 * D(m) * |m.n| * |wi.m|
201 // / (etaI * (wo.m) + etaT * (wi.m))^2
202 if (pdf != NULL) {
203#if PBR_CT_SAMPLE_WIDENED_ROUGHNESS
204 // We cannot re-use D computed above without adjusting for the widened
205 // roughness we use in sample(), otherwise our pdf is biased
206 const float alpha = CookTorranceBsdfLobe::widenRoughness(mRoughness,
207 absCosNO);
208 const float alpha2 = alpha * alpha;
209 const float D = scene_rdl2::math::exp(minusTanThetaM2 * rcp(alpha2)) *
210 rcp(sPi * alpha2 * cosThetaM4);
211#endif
212 *pdf = D * cosThetaM * etaTSqr * absCosMI * denom;
213 }
214
215 // apply transmission color tint hack
216 return f * mTint;
217}

219Color
220TransmissionCookTorranceBsdfLobe::sample(const BsdfSlice &slice,
221 float r1, float r2,
222 Vec3f &wi,
223 float &pdf) const
224{
225 const float cosNO = dot(mFrame.getN(), slice.getWo());
226 const float absCosNO = scene_rdl2::math::abs(cosNO);
227
228 Color dispersionColor = scene_rdl2::math::sWhite;
229 float neta = mNeta;
230 float etaT = mEtaT;
231 float spectralPDF = 1.0f;
232 pdf = 0.0f;
233 if (mAllowDispersion) {
234 shading::sampleSpectralIOR(r1,
235 mEtaR,
236 mEtaG,
237 mEtaB,
238 &etaT,
239 &spectralPDF,
240 dispersionColor);
241
242 neta = mEtaI / etaT;
243 }
244
245
246 Color val;
247 // generate a random microfacet normal m (eq. 28, 29)
248 // we take advantage of cos(atan(x)) == 1/sqrt(1+x^2)
249 // and sin(atan(x)) == x/sqrt(1+x^2)
250 const float alpha = CookTorranceBsdfLobe::widenRoughness(mRoughness,
251 absCosNO);
252 const float alpha2 = alpha * alpha;
253 const float tanThetaMSqr = -alpha2 * scene_rdl2::math::log(1.0f - r1);
254 const float tanThetaM = scene_rdl2::math::sqrt(tanThetaMSqr);
255 const float cosThetaM = rsqrt(1.0f + tanThetaMSqr);
256 const float sinThetaM = cosThetaM * tanThetaM;
257 const float phiM = 2.0f * sPi * r2;
258
259 // Compute the half vector. By construction it is facing the same side
260 // as N and as wo
261 Vec3f m = computeLocalSphericalDirection(cosThetaM, sinThetaM, phiM);
262 m = mFrame.localToGlobal(m);
263
264 // Compute the transmission direction
265 float cosMO, cosMI;
266 if (!computeRefractionDirection(m,
267 slice.getWo(),
268 neta,
269 wi,
270 cosMO, cosMI)) {
271 wi = Vec3f(zero);
272 pdf = 0.0f;
273 return Color(0.0f, 0.0f, 0.0f);
274 }
275
276 Color result = eval(slice, wi, etaT, &pdf);
277 // Dispersion PDF
278 pdf *= spectralPDF;
279
280 val = (dispersionColor * result);
281 return val;
282}

284bool
285TransmissionCookTorranceBsdfLobe::getProperty(Property property, float *dest) const
286{
287 bool handled = true;
288
289 switch (property)
290 {
291 case PROPERTY_ROUGHNESS:
292 *dest = mInputRoughness;
293 *(dest + 1) = mInputRoughness;
294 break;
295 case PROPERTY_NORMAL:
296 {
297 const Vec3f &N = mFrame.getN();
298 *dest = N.x;
299 *(dest + 1) = N.y;
300 *(dest + 2) = N.z;
301 }
302 break;
303 default:
304 handled = BsdfLobe::getProperty(property, dest);
305 break;
306 }
307
308 return handled;
309}
