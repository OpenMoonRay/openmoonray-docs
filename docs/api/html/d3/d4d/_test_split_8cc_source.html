49void
50TestSplit::testRoundtrip()
51{
52 // Create the context, load a class, and create some objects.
53 SceneContext context;
54 const SceneClass* sc = context.createSceneClass("ExtensiveObject");
55 SceneObject* apple = context.createSceneObject("ExtensiveObject", "/seq/shot/apple");
56 SceneObject* banana = context.createSceneObject("ExtensiveObject", "/seq/shot/banana");
57 Geometry* teapot = context.createSceneObject("FakeTeapot", "/seq/shot/teapot")->asA<Geometry>();
58 Material* material = context.createSceneObject("FakeMaterial", "/seq/shot/material")->asA<Material>();
59 LightSet* lightset = context.createSceneObject("LightSet", "/seq/shot/lightset")->asA<LightSet>();
60 Layer* layer = context.createSceneObject("Layer", "/seq/shot/layer")->asA<Layer>();
61
62 AttributeKey<String> stringKey = sc->getAttributeKey<String>("string");
63 AttributeKey<Vec3fVector> vec3fVectorKey = sc->getAttributeKey<Vec3fVector>("vec3f_vector");
64
65 std::string stringDefault = apple->get(stringKey);
66 Vec3fVector vec3fVectorDefault = apple->get(vec3fVectorKey);
67
68 apple->beginUpdate();
69 apple->set(stringKey, std::string("apple"));
70 apple->set(vec3fVectorKey, mShortVec);
71 apple->endUpdate();
72 banana->beginUpdate();
73 banana->set(stringKey, std::string("banana"));
74 banana->set(vec3fVectorKey, mLongVec);
75 banana->endUpdate();
76
77 layer->beginUpdate();
78 for (unsigned i = 0; i < 50; i++) {
79 layer->assign(teapot, "part" + std::to_string(i), material, lightset);
80 }
81 layer->endUpdate();
82
83 writeSceneToFile(context, "roundtrip_split", false, true);
84
85 // Create a fresh SceneContext and read in the Ascii file.
86 SceneContext asciiContext;
87 AsciiReader areader(asciiContext);
88 areader.fromFile("roundtrip_split.rdla");
89
90 SceneObject* asciiApple = asciiContext.getSceneObject("/seq/shot/apple");
91 SceneObject* asciiBanana = asciiContext.getSceneObject("/seq/shot/banana");
92 Geometry* asciiTeapot = asciiContext.getSceneObject("/seq/shot/teapot")->asA<Geometry>();
93 Layer* asciiLayer = asciiContext.getSceneObject("/seq/shot/layer")->asA<Layer>();
94
95 // everything except banana.vec3fVector should be in the ascii file
96 CPPUNIT_ASSERT(apple->get(stringKey) == asciiApple->get(stringKey));
97 CPPUNIT_ASSERT(apple->get(vec3fVectorKey) == asciiApple->get(vec3fVectorKey));
98 CPPUNIT_ASSERT(banana->get(stringKey) == asciiBanana->get(stringKey));
99 CPPUNIT_ASSERT(vec3fVectorDefault == asciiBanana->get(vec3fVectorKey));
100
101 for (unsigned i = 0; i < 50; i++) {
102 // will throw if entry doesn't exist
103 asciiLayer->lookup(asciiTeapot, "part" + std::to_string(i));
104 }
105
106 // Create a fresh SceneContext and read in the Binary file.
107 SceneContext binContext;
108 BinaryReader breader(binContext);
109 breader.fromFile("roundtrip_split.rdlb");
110
111 SceneObject* binApple = binContext.getSceneObject("/seq/shot/apple");
112 SceneObject* binBanana = binContext.getSceneObject("/seq/shot/banana");
113 Layer* binLayer = binContext.getSceneObject("/seq/shot/layer")->asA<Layer>();
114
115 // only banana.vec3fVector should be in the binary file
116 CPPUNIT_ASSERT(stringDefault == binApple->get(stringKey));
117 CPPUNIT_ASSERT(vec3fVectorDefault == binApple->get(vec3fVectorKey));
118 CPPUNIT_ASSERT(stringDefault == binBanana->get(stringKey));
119 CPPUNIT_ASSERT(banana->get(vec3fVectorKey) == binBanana->get(vec3fVectorKey));
120
121 Layer::GeometrySet geoms;
122 binLayer->getAllGeometries(geoms);
123 CPPUNIT_ASSERT(geoms.empty());
124}
