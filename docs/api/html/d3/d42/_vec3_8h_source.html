295 // MoonRay: begin *****
296 template <typename T>
297 inline Vec3<T> cross(const Vec3<T>& a, const Vec3<T>& b)
298 {
299 // Accurate Differences of Products with Kahan's Algorithm
300 // https://pharr.org/matt/blog/2019/11/03/difference-of-floats
301 // This version of cross may be a touch slower with increased precision.
302 // Profiling did not demonstrate a noticeable slowdown.
303 return Vec3<T>(differenceOfProducts(a.y, b.z, a.z, b.y),
304 differenceOfProducts(a.z, b.x, a.x, b.z),
305 differenceOfProducts(a.x, b.y, a.y, b.x));
306 }

316 template <typename T> __forceinline Vec3<T>
317 safeNormalize(const Vec3<T> & a, const T eps = T(epsilon))
318 {
319 const T lengthSqr = dot(a, a);
320 return (lengthSqr <= eps * eps) ? Vec3<T>(T(0)) : a * rsqrt(lengthSqr);
