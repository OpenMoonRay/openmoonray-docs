43bool
44QuadMesh::bakePosMap(int width, int height, int udim,
45 TypedAttributeKey<Vec2f> stKey,
46 Vec3fa *posResult, Vec3f *nrmResult) const
47{
48 const Attributes *primitiveAttributes = getAttributes();
49 MNRY_ASSERT_REQUIRE(primitiveAttributes->isSupported(stKey));
50 // throw out the anchor boys....
51 for (size_t faceId = 0; faceId < getTessellatedMeshFaceCount(); ++faceId) {
52 int baseFaceId = getBaseFaceId(faceId);
53 const uint32_t id0 = mIndices[faceId * sQuadVertexCount];
54 const uint32_t id1 = mIndices[faceId * sQuadVertexCount + 1];
55 const uint32_t id2 = mIndices[faceId * sQuadVertexCount + 2];
56 const uint32_t id3 = mIndices[faceId * sQuadVertexCount + 3];
57
58 const Vec3f pos0 = mVertices(id0).asVec3f();
59 const Vec3f pos1 = mVertices(id1).asVec3f();
60 const Vec3f pos2 = mVertices(id2).asVec3f();
61 const Vec3f pos3 = mVertices(id3).asVec3f();
62
63 // FIXME: we are forced to fetch st1 and st3 twice.
64 Vec2f st0, st1, st2, st3;
65 getQuadAttributes(stKey, baseFaceId, faceId, st0, st1, st3,
66 0.0f, /* isFirst = */ true);
67 getQuadAttributes(stKey, baseFaceId, faceId, st2, st3, st1,
68 0.0f, /* isFrist = */ false);
69
70 // check if this face exists in this udim and if so
71 // transform the st coordinates into the normalized range
72 if (udimxform(udim, st0) && udimxform(udim, st1) &&
73 udimxform(udim, st2) && udimxform(udim, st3)) {
74 const Vec3f *nrm0 = nullptr;
75 const Vec3f *nrm1 = nullptr;
76 const Vec3f *nrm2 = nullptr;
77 const Vec3f *nrm3 = nullptr;
78 Vec3f nrmData0, nrmData1, nrmData2, nrmData3;
79 if (nrmResult) {
80 getQuadNormal(baseFaceId, faceId, nrmData0, nrmData1, nrmData3,
81 0.0f, /* isFirst = */ true);
82 getQuadNormal(baseFaceId, faceId, nrmData2, nrmData3, nrmData1,
83 0.0f, /* isFirst = */ false);
84 nrm0 = &nrmData0;
85 nrm1 = &nrmData1;
86 nrm2 = &nrmData2;
87 nrm3 = &nrmData3;
88 }
89 // first triangle: <0, 1, 3>
90 scene_rdl2::math::BBox2f roiST(
91 Vec2f(std::min(st0.x, std::min(st1.x, st3.x)),
92 std::min(st0.y, std::min(st1.y, st3.y))),
93 Vec2f(std::max(st0.x, std::max(st1.x, st3.x)),
94 std::max(st0.y, std::max(st1.y, st3.y))));
95 rasterizeTrianglePos(roiST, width, height, st0, st1, st3,
96 pos0, pos1, pos3, nrm0, nrm1, nrm3, posResult, nrmResult);
97 // second triangle: <2, 3, 1>
98 roiST = scene_rdl2::math::BBox2f(
99 Vec2f(std::min(st2.x, std::min(st3.x, st1.x)),
100 std::min(st2.y, std::min(st3.y, st1.y))),
101 Vec2f(std::max(st2.x, std::max(st3.x, st1.x)),
102 std::max(st2.y, std::max(st3.y, st1.y))));
103 rasterizeTrianglePos(roiST, width, height, st2, st3, st1,
104 pos2, pos3, pos1, nrm2, nrm3, nrm1, posResult, nrmResult);
105 }
106 }
107
108 return true;
109}

111void
112QuadMesh::setRequiredAttributes(int primId, float time, float u, float v,
113 float w, bool isFirst, Intersection& intersection) const
114{
115 int tessFaceId = primId;
116 int baseFaceId = getBaseFaceId(tessFaceId);
117 // In Embree, quad is internally handled as a pair of
118 // two triangles (v0,v1,v3) and (v2,v3,v1).
119 // The (u',v') coordinates of the second triangle
120 // is corrected by u = 1-u' and v = 1-v' to produce
121 // a quad parametrization where u and v go from 0 to 1.
122 // That's to say, if u+v > 1,
123 // intersection happens in the second triangle,
124 // and (u',v') in the second triangle should be (1-u, 1-v)
125 // In order to get face varying attributes by offset,
126 // we still need to keep the unused vertex,
127 // but set its weight to 0 to remove its influence
128 uint32_t id1 = mIndices[sQuadVertexCount * tessFaceId ];
129 uint32_t id2 = mIndices[sQuadVertexCount * tessFaceId + 1];
130 uint32_t id3 = mIndices[sQuadVertexCount * tessFaceId + 2];
131 uint32_t id4 = mIndices[sQuadVertexCount * tessFaceId + 3];
132 uint32_t isecId1, isecId2, isecId3;
133 uint32_t varyingId1, varyingId2, varyingId3, varyingId4;
134 // weights for interpolator
135 float varyingW[4];
136 float vertexW[4];
137
138 if (isFirst) {
139 // Triangle (0,1,3)
140 vertexW[0] = w;
141 vertexW[1] = u;
142 vertexW[2] = 0.0f;
143 vertexW[3] = v;
144 intersection.setIds(id1, id2, id4);
145 } else {
146 // Triangle (2,3,1)
147 vertexW[0] = 0.0f;
148 vertexW[1] = v;
149 vertexW[2] = w;
150 vertexW[3] = u;
151 intersection.setIds(id3, id4, id2);
152 }
153
154 if (mIsTessellated) {
155 // the surface uv of intersection point on base face
156 Vec2f uv =
157 vertexW[0] * mFaceVaryingUv[sQuadVertexCount * tessFaceId ] +
158 vertexW[1] * mFaceVaryingUv[sQuadVertexCount * tessFaceId + 1] +
159 vertexW[2] * mFaceVaryingUv[sQuadVertexCount * tessFaceId + 2] +
160 vertexW[3] * mFaceVaryingUv[sQuadVertexCount * tessFaceId + 3];
161 // bilinear interpolation weight
162 varyingW[0] = (1.0f - uv[0]) * (1.0f - uv[1]);
163 varyingW[1] = ( uv[0]) * (1.0f - uv[1]);
164 varyingW[2] = ( uv[0]) * ( uv[1]);
165 varyingW[3] = (1.0f - uv[0]) * ( uv[1]);
166 varyingId1 = mBaseIndices[sQuadVertexCount * baseFaceId ];
167 varyingId2 = mBaseIndices[sQuadVertexCount * baseFaceId + 1];
168 varyingId3 = mBaseIndices[sQuadVertexCount * baseFaceId + 2];
169 varyingId4 = mBaseIndices[sQuadVertexCount * baseFaceId + 3];
170 } else {
171 varyingW[0] = vertexW[0];
172 varyingW[1] = vertexW[1];
173 varyingW[2] = vertexW[2];
174 varyingW[3] = vertexW[3];
175 varyingId1 = id1;
176 varyingId2 = id2;
177 varyingId3 = id3;
178 varyingId4 = id4;
179 }
180
181 int partId = (mFaceToPart.size() > 0) ? mFaceToPart[baseFaceId] : 0;
182 const Attributes* primitiveAttributes = getAttributes();
183 // Interpolate PrimitiveAttribute
184 // Interpolator get attributes by offsetting indices,
185 // directly passing sub triangle vertex id doesn't guarantee
186 // To support face varying attributes for quad,
187 // id should be passed in face index rotation order,
188 // as well as four weights.
189 // e.g. intersect at second triangle in face(v0-v1-v2-v3),
190 // we should pass indices (vid0, vid1, vid2, vid3),
191 // and weights (0, v, w, u)
192 MeshInterpolator interpolator(primitiveAttributes,
193 time, partId, baseFaceId,
194 varyingId1, varyingId2, varyingId3, varyingId4,
195 varyingW[0], varyingW[1], varyingW[2], varyingW[3], tessFaceId,
196 id1, id2, id3, id4,
197 vertexW[0], vertexW[1], vertexW[2], vertexW[3]);
198 intersection.setRequiredAttributes(interpolator);
199
200 if (primitiveAttributes->isSupported(StandardAttributes::sNormal)) {
201 Vec3f N;
202 interpolator.interpolate(StandardAttributes::sNormal,
203 reinterpret_cast<char*>(&N));
204 N.normalize();
205 intersection.setN(N);
206 }

209void
210QuadMesh::postIntersect(mcrt_common::ThreadLocalState& tls,
211 const scene_rdl2::rdl2::Layer* pRdlLayer, const mcrt_common::Ray& ray,
212 Intersection& intersection) const
213{
214 int primId = ray.primID;
215 int tessFaceId = primId;
216 int baseFaceId = getBaseFaceId(tessFaceId);
217 int assignmentId = getFaceAssignmentId(tessFaceId);
218 intersection.setLayerAssignments(assignmentId, pRdlLayer);
219
220 const auto& material =
221 intersection.getMaterial()->get<shading::RootShader>();
222 const AttributeTable *table = material.getPrimitiveAttributeTable();
223 intersection.setTable(&tls.mArena, table);
224 overrideInstanceAttrs(ray, intersection);
225
226 // barycentric coordinate
227 float u = ray.u;
228 float v = ray.v;
229 float w = 1.0f - u - v;
230 bool isFirst = w > 0.0f;
231 if (!isFirst) {
232 u = 1.0f - u;
233 v = 1.0f - v;
234 w = -w;
235 }
236
237 setRequiredAttributes(primId, ray.time, u, v, w, isFirst, intersection);
238
239 const Attributes* primitiveAttributes = getAttributes();
240 Vec3f Ng, N, dPds, dPdt;
241 Vec2f St;
242 Ng = normalize(ray.getNg());
243 // shading normal
244 if (primitiveAttributes->isSupported(StandardAttributes::sNormal)) {
245 N = intersection.getN();
246 } else {
247 N = Ng;
248 }
249
250 Vec2f st1, st2, st3;
251 getQuadST(baseFaceId, tessFaceId, st1, st2, st3, isFirst);
252 St = w * st1 + u * st2 + v * st3;
253
254 uint32_t isecId1, isecId2, isecId3;
255 intersection.getIds(isecId1, isecId2, isecId3);
256 if (isMotionBlurOn()) {
257 float t = ray.time;
258 const Vec3f& p10 = mVertices(isecId1, 0);
259 const Vec3f& p20 = mVertices(isecId2, 0);
260 const Vec3f& p30 = mVertices(isecId3, 0);
261 // form a ReferenceFrame and use its tangent/binormal as dpds dpdt
262 // if we fail to calculate the st partial derivatives
263 Vec3f dp0[2];
264 if (!computeTrianglePartialDerivatives(
265 p10, p20, p30, st1, st2, st3, dp0)) {
266 scene_rdl2::math::ReferenceFrame frame(N);
267 dp0[0] = frame.getX();
268 dp0[1] = frame.getY();
269 }
270 const Vec3f& p11 = mVertices(isecId1, 1);
271 const Vec3f& p21 = mVertices(isecId2, 1);
272 const Vec3f& p31 = mVertices(isecId3, 1);
273 Vec3f dp1[2];
274 if (!computeTrianglePartialDerivatives(
275 p11, p21, p31, st1, st2, st3, dp1)) {
276 scene_rdl2::math::ReferenceFrame frame(N);
277 dp1[0] = frame.getX();
278 dp1[1] = frame.getY();
279 }
280 dPds = (1.0f - t) * dp0[0] + t * dp1[0];
281 dPdt = (1.0f - t) * dp0[1] + t * dp1[1];
282 } else {
283 const Vec3f& p1 = mVertices(isecId1);
284 const Vec3f& p2 = mVertices(isecId2);
285 const Vec3f& p3 = mVertices(isecId3);
286 // form a ReferenceFrame and use its tangent/binormal as dpds dpdt
287 // if we fail to calculate the st partial derivatives
288 Vec3f dp[2];
289 if (!computeTrianglePartialDerivatives(
290 p1, p2, p3, st1, st2, st3, dp)) {
291 scene_rdl2::math::ReferenceFrame frame(N);
292 dp[0] = frame.getX();
293 dp[1] = frame.getY();
294 }
295 dPds = dp[0];
296 dPdt = dp[1];
297 }
298
299 intersection.setDifferentialGeometry(Ng, N, St, dPds, dPdt, true);
300
301 // calculate dfds/dfdt for primitive attributes that request differential
302 if (table->requestDerivatives()) {
303 computeAttributesDerivatives(table, st1, st2, st3,
304 baseFaceId, tessFaceId, ray.time, isFirst, intersection);
305 }
306
307 // fill in ray epsilon to avoid self intersection
308 const scene_rdl2::rdl2::Geometry* geometry = intersection.getGeometryObject();
309 MNRY_ASSERT(geometry != nullptr);
310 intersection.setEpsilonHint(geometry->getRayEpsilon());
311
312 // polyon vertices
313 if (table->requests(StandardAttributes::sNumPolyVertices)) {
314 intersection.setAttribute(StandardAttributes::sNumPolyVertices, 4);
315 }
316 if (table->requests(StandardAttributes::sPolyVertexType)) {
317 intersection.setAttribute(StandardAttributes::sPolyVertexType,
318 static_cast<int>(StandardAttributes::POLYVERTEX_TYPE_POLYGON));
319 }
320 for (int iVert = 0; iVert < 4; iVert++) {
321 if (table->requests(StandardAttributes::sPolyVertices[iVert])) {
322 uint32_t id = mIndices[4 * tessFaceId + iVert];
323 // may need to move the vertices to render space
324 // for instancing object since they are ray traced in local space
325 const Vec3f v = ray.isInstanceHit() ? transformPoint(ray.ext.l2r, mVertices(id).asVec3f())
326 : mVertices(id).asVec3f();
327 intersection.setAttribute(StandardAttributes::sPolyVertices[iVert], v);
328 }
329 }
330
331 // motion vectors
332 if (table->requests(StandardAttributes::sMotion)) {
333 const Vec3f motion = computeMotion(mVertices, isecId1, isecId2, isecId3,
334 w, u, v, ray);
335 intersection.setAttribute(StandardAttributes::sMotion, motion);
336 }
337}

339bool
340QuadMesh::computeIntersectCurvature(const mcrt_common::Ray& ray,
341 const Intersection& intersection, Vec3f& dnds, Vec3f& dndt) const
342{
343 int tessFaceId = ray.primID;
344 int baseFaceId = getBaseFaceId(tessFaceId);
345 uint32_t id1, id2, id3;
346 intersection.getIds(id1, id2, id3);
347 bool isFirst = ray.u + ray.v <= 1.0f;
348 Vec2f st1, st2, st3;
349 getQuadST(baseFaceId, tessFaceId, st1, st2, st3, isFirst);
350 Vec3f n1, n2, n3;
351 getQuadNormal(baseFaceId, tessFaceId, n1, n2, n3, ray.time, isFirst);
352 Vec3f dn[2];
353 if (computeTrianglePartialDerivatives(n1, n2, n3, st1, st2, st3, dn)) {
354 dnds = dn[0];
355 dndt = dn[1];
356 return true;
357 } else {
358 return false;
359 }

363QuadMesh::getST(int tessFaceId, float u, float v, Vec2f& st) const
364{
365 int baseFaceId = getBaseFaceId(tessFaceId);
366 float w = 1.0f - u - v;
367 bool isFirst = w > 0.f;
368 if (!isFirst) {
369 // Triangle (2,3,1)
370 u = 1.0f - u;
371 v = 1.0f - v;
372 w = -w;
373 }

375 getQuadST(baseFaceId, tessFaceId, st1, st2, st3, isFirst);
376 st = w * st1 + u * st2 + v * st3;

379void
380QuadMesh::getNeighborVertices(int baseFaceId, int tessFaceId,
381 int tessFaceVIndex, int& vid, int& vid1, int& vid2, int& vid3,
382 Vec2f& st, Vec2f& st1, Vec2f& st2, Vec2f& st3) const
383{
384 size_t offset = tessFaceId * sQuadVertexCount;
385 if (tessFaceVIndex == 2) {
386 if (mIndices[offset + 2] == mIndices[offset + 3]) {
387 // the degened quad introduced during tessellation
388 // first triangle on quad (0, 1, 3), which is (0, 1, 2)
389 getQuadST(baseFaceId, tessFaceId, st1, st2, st3, true);
390 vid1 = mIndices[offset + 0];
391 vid2 = mIndices[offset + 1];
392 vid3 = mIndices[offset + 2];
393 vid = vid3;
394 st = st3;
395 } else {
396 // second triangle on quad (2, 3, 1)
397 getQuadST(baseFaceId, tessFaceId, st1, st2, st3, false);
398 vid1 = mIndices[offset + 2];
399 vid2 = mIndices[offset + 3];
400 vid3 = mIndices[offset + 1];
401 vid = vid1;
402 st = st1;
403 }
404 } else {
405 // first triangle on quad (0, 1, 3)
406 getQuadST(baseFaceId, tessFaceId, st1, st2, st3, true);
407 vid1 = mIndices[offset + 0];
408 vid2 = mIndices[offset + 1];
409 vid3 = mIndices[offset + 3];
410 if (tessFaceVIndex == 0) {
411 vid = vid1;
412 st = st1;
413 } else if (tessFaceVIndex == 1) {
414 vid = vid2;
415 st = st2;
416 } else if (tessFaceVIndex == 3) {
417 vid = vid3;
418 st = st3;
419 }
420 }

423void
424QuadMesh::splitNGons(size_t outputFaceCount,
425 PolygonMesh::FaceToPartBuffer& faceToPart,
426 const PolygonMesh::FaceVertexCount& faceVertexCount,
427 PolygonMesh::IndexBuffer& indices,
428 LayerAssignmentId& layerAssignmentId,
429 PrimitiveAttributeTable& primitiveAttributeTable)
430{
431 size_t inputFaceCount = faceVertexCount.size();
432 size_t inputIndexCount = indices.size();
433 indices.resize(sQuadVertexCount * outputFaceCount);
434 if (faceToPart.size() > 0) {
435 faceToPart.resize(outputFaceCount);
436 }
437
438 size_t inputIndexOffset = inputIndexCount;
439 size_t outputIndexOffset = indices.size() - 1;
440 size_t outputF2POffset = faceToPart.size() - 1;
441 // quadrangulate indices
442 for (int f = inputFaceCount - 1; f >= 0; --f) {
443 size_t fvCount = faceVertexCount[f];
444 inputIndexOffset -= fvCount;
445 size_t v = fvCount;
446 // when quadragulating an ngon with odd number n,
447 // there will be one quad that has a duplicated vertex
448 if (fvCount & 1) {
449 indices[outputIndexOffset--] = indices[inputIndexOffset + v - 1];
450 indices[outputIndexOffset--] = indices[inputIndexOffset + v - 1];
451 indices[outputIndexOffset--] = indices[inputIndexOffset + v - 2];
452 indices[outputIndexOffset--] = indices[inputIndexOffset];
453 if (faceToPart.size() > 0) {
454 faceToPart[outputF2POffset--] = faceToPart[f];
455 }
456 v -= 1;
457 }
458 for (; v >= sQuadVertexCount; v -= 2) {
459 indices[outputIndexOffset--] = indices[inputIndexOffset + v - 1];
460 indices[outputIndexOffset--] = indices[inputIndexOffset + v - 2];
461 indices[outputIndexOffset--] = indices[inputIndexOffset + v - 3];
462 indices[outputIndexOffset--] = indices[inputIndexOffset];
463 if (faceToPart.size() > 0) {
464 faceToPart[outputF2POffset--] = faceToPart[f];
465 }
466 }
467 }
468
469 // quadrangulate layerAssignmentId if it's in per face frequency
470 if (layerAssignmentId.getType() == LayerAssignmentId::Type::VARYING) {
471 auto& faceAssignmentIds = layerAssignmentId.getVaryingId();
472 faceAssignmentIds.resize(outputFaceCount);
473 size_t outputFaceOffset = outputFaceCount - 1;
474 for (int f = inputFaceCount - 1; f >= 0; --f) {
475 size_t quadCount = (faceVertexCount[f] - 1) >> 1;
476 for (int v = quadCount; v > 0; --v) {
477 faceAssignmentIds[outputFaceOffset--] = faceAssignmentIds[f];
478 }
479 }
480 }
481
482 // quadrangulate uniform and facevarying primitive attributes
483 for (auto& kv : primitiveAttributeTable) {
484 for (auto& attribute : kv.second) {
485 AttributeRate rate = attribute->getRate();
486 if (rate == RATE_UNIFORM) {
487 attribute->resize(outputFaceCount);
488 size_t outputFaceOffset = outputFaceCount - 1;
489 for (int f = inputFaceCount - 1; f >= 0; --f) {
490 size_t quadCount = (faceVertexCount[f] - 1) >> 1;
491 for (size_t v = quadCount; v > 0; --v) {
492 attribute->copyInPlace(f, outputFaceOffset--);
493 }
494 }
495 } else if (rate == RATE_FACE_VARYING) {
496 attribute->resize(sQuadVertexCount * outputFaceCount);
497 inputIndexOffset = inputIndexCount;
498 outputIndexOffset = indices.size() - 1;
499 for (int f = inputFaceCount - 1; f >= 0; --f) {
500 size_t fvCount = faceVertexCount[f];
501 inputIndexOffset -= fvCount;
502 size_t v = fvCount;
503 // when quadragulating an ngon with odd number n,
504 // there will be one quad that has a duplicated vertex
505 if (fvCount & 1) {
506 attribute->copyInPlace(inputIndexOffset + v - 1,
507 outputIndexOffset--);
508 attribute->copyInPlace(inputIndexOffset + v - 1,
509 outputIndexOffset--);
510 attribute->copyInPlace(inputIndexOffset + v - 2,
511 outputIndexOffset--);
512 attribute->copyInPlace(inputIndexOffset,
513 outputIndexOffset--);
514 v -= 1;
515 }
516 for (; v >= sQuadVertexCount; v -= 2) {
517 // quadrangulate a ngon into quad fan
518 attribute->copyInPlace(inputIndexOffset + v - 1,
519 outputIndexOffset--);
520 attribute->copyInPlace(inputIndexOffset + v - 2,
521 outputIndexOffset--);
522 attribute->copyInPlace(inputIndexOffset + v - 3,
523 outputIndexOffset--);
524 attribute->copyInPlace(inputIndexOffset,
525 outputIndexOffset--);
526 }
527 }
528 }
529 }
530 }

533void
534QuadMesh::generateIndexBufferAndSurfaceSamples(
535 const std::vector<PolyFaceTopology>& quadTopologies,
536 const PolyTessellatedVertexLookup& tessellatedVertexLookup,
537 PolygonMesh::IndexBuffer& indices,
538 std::vector<PolyMesh::SurfaceSample>& surfaceSamples,
539 std::vector<int>& tessellatedToBaseFace,
540 std::vector<Vec2f>* faceVaryingUv) const
541{
542 tessellatedToBaseFace.clear();
543 // we need to use arena to allocate temporary data structure for
544 // outer/interior ring stitching
545 mcrt_common::ThreadLocalState *tls = mcrt_common::getFrameUpdateTLS();
546 scene_rdl2::alloc::Arena *arena = &tls->mArena;
547
548 size_t tessellatedVertexCount =
549 tessellatedVertexLookup.getTessellatedVertexCount();
550 surfaceSamples.resize(tessellatedVertexCount);
551 for (size_t f = 0; f < quadTopologies.size(); ++f) {
552 const PolyFaceTopology& quadTopology = quadTopologies[f];
553 // handle uniform tessellation in more optimized topology
554 if (tessellatedVertexLookup.noRingToStitch(quadTopology)) {
555 SCOPED_MEM(arena);
556 int colVertexCount;
557 int rowVertexCount;
558 int * vids = nullptr;
559 collectUniformTessellatedVertices(arena,
560 quadTopology, f, tessellatedVertexLookup,
561 vids, rowVertexCount, colVertexCount);
562 float du = 1.0f / (float)(colVertexCount - 1);
563 float dv = 1.0f / (float)(rowVertexCount - 1);
564 int offset = 0;
565 for (int i = 0; i < rowVertexCount; ++i) {
566 for (int j = 0; j < colVertexCount; ++j) {
567 int vid = vids[offset++];
568 surfaceSamples[vid].mFaceId = f;
569 surfaceSamples[vid].mUv = Vec2f(j * du, i * dv);
570 }
571 }

674template <typename T>
675T bilinearInterpolate(const Vec2f& uv,
676 const T& a0, const T& a1, const T& a2, const T& a3)
677{
678 return (1.0f - uv[0]) * (1.0f - uv[1]) * a0 +
679 ( uv[0]) * (1.0f - uv[1]) * a1 +
680 ( uv[0]) * ( uv[1]) * a2 +
681 (1.0f - uv[0]) * ( uv[1]) * a3;
682}
