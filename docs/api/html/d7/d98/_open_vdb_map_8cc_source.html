62void
63OpenVdbMap::update()
64{
65 mXform = fauxstd::make_unique<moonray::shading::Xform>(this, nullptr, nullptr, nullptr);
66 mIspc.mXform = mXform->getIspcXform();
67
68 if (hasChanged(attrVdbSource) ||
69 hasChanged(attrTexture) ||
70 hasChanged(attrOpenVdbGeometry)) {
71
72 mSampler.reset();
73
74 // If using OpenVdbGeometry, we'll append its transform, otherwise not.
75 const Mat4d* grid2world = nullptr;
76
77 // Resolve the vdb filename and potentially a grid2world transform
78 std::string vdbFilename;
79 if (get(attrVdbSource) == VDB_SOURCE_OPENVDB_GEOMETRY) {
80 const Geometry* geom = get(attrOpenVdbGeometry) ?
81 get(attrOpenVdbGeometry)->asA<Geometry>() :
82 nullptr;
83 if (!geom) {
84 error("You must specify an OpenVdbGeometry object");
85 } else if (!moonray::shading::isOpenVdbGeometry(geom)) {
86 error("The object specified for 'open_vdb_geometry' is not an OpenVdbGeometry object");
87 } else {
88 AttributeKey<String> modelAttrKey = moonray::shading::getModelAttributeKey(geom);
89 vdbFilename = geom->get<std::string>(modelAttrKey);
90 grid2world = &geom->get(Node::sNodeXformKey);
91 }
92 } else { // VDB_SOURCE_TEXTURE
93 vdbFilename = get(attrTexture);
94 if (vdbFilename.empty()) {
95 error("You must specify a valid .vdb filename");
96 }
97 }
98
99 // make sure we successfully resolved a filename
100 if (!vdbFilename.empty()) {
101
102 // We only need the render2world transform if using P
103 const Mat4d* render2world = nullptr;
104 if (get(attrTextureCoordSource) == ispc::OPENVDBMAP_COORD_P) {
105 const SceneContext* ctx = getSceneClass().getSceneContext();
106 render2world = ctx->getRender2World();
107 }
108
109 // Construct an uninitialized OpenVdbSampler, then try to initialize it.
110 // If initialization fails for any reason, destroy this useless object
111 // immediately. Two-phase construct-and-init design pattern
112 mSampler = fauxstd::make_unique<moonray::shading::OpenVdbSampler>();
113
114 std::string errorMsg;
115 if (mSampler->initialize(vdbFilename,
116 get(attrGridName),
117 render2world,
118 grid2world,
119 errorMsg)) {
120 } else {
121 mSampler.reset();
122 if (get(attrShowWarnings)) {
123 warn("Unable to load grid from file ", vdbFilename);
124 }
125 }
126 }
127 }
128
129 if (hasChanged(attrTextureCoordSource)) {
130 mRequiredAttributes.clear();
131 mOptionalAttributes.clear();
132 if (get(attrTextureCoordSource) == ispc::OPENVDBMAP_COORD_REFP) {
133 mRequiredAttributes.push_back(mIspc.mRefPKey);
134 }
135 }
136}

138// called by scalar and vector sample funcs
139void
140OpenVdbMap::sampleVdb(const Map* map,
141 moonray::shading::TLState *tls,
142 const float px, const float py, const float pz,
143 scene_rdl2::math::Color *out)
144{
145 const OpenVdbMap* me = static_cast<const OpenVdbMap*>(map);
146
147 if (!me->mSampler) {
148 *out = me->get(attrDefaultValue);
149 return;
150 }
151
152 const Vec3f P(px, py, pz);
153
154 if (me->get(attrShowActiveField)) {
155 *out = me->mSampler->getIsActive(tls, P) ?
156 scene_rdl2::math::Color(1.f, 1.f, 1.f) :
157 scene_rdl2::math::Color(0.f, 0.f, 0.f);
158 } else {
159 *out = me->mSampler->sample(tls, P,
160 static_cast<moonray::shading::OpenVdbSampler::Interpolation>(me->get(attrInterpolation)));
161 }
162}

164void
165OpenVdbMap::sample(const Map* map, moonray::shading::TLState *tls,
166 const moonray::shading::State& state, scene_rdl2::math::Color* sample)
167{
168 const OpenVdbMap* me = static_cast<const OpenVdbMap*>(map);
169
170 Vec3f P = state.getP();
171
172 if (me->get(attrTextureCoordSource) == ispc::OPENVDBMAP_COORD_REFP) {
173 state.getRefP(P);
174 } else if (me->get(attrTextureCoordSource) == ispc::OPENVDBMAP_COORD_INPUT) {
175 P = evalVec3f(me, attrInputTextureCoordinate, tls, state);
176 }
177
178 sampleVdb(map, tls, P.x, P.y, P.z, sample);
179}
