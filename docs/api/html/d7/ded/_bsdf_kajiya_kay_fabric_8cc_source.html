19///
20/// @class KajiyaKayFabricBsdfLobe BsdfFabric.h <pbr/fabric/BsdfFabric.h>
21/// @brief The KajiyaKay Cosine Specular Lobe - cos(thetaD)^N
22/// thetaD = thetaO - thetaIPrime, theta wrt hair tangent
23///
24
25// Based on the paper - "Rendering fur with three dimensional textures" (Kajiya et al)
26// https://www.cs.drexel.edu/~david/Classes/CS586/Papers/p271-kajiya.pdf
27// Note - we include a custom normalization factor that we've calculated for this model.
28
29//----------------------------------------------------------------------------
30// Constructor / Destructor
31KajiyaKayFabricBsdfLobe::KajiyaKayFabricBsdfLobe(const Vec3f &N,
32 const Vec3f &T,
33 const Vec3f &threadDirection,
34 const float threadElevation,
35 const float roughness) :
36 FabricBsdfLobe(N, T,
37 threadDirection, threadElevation,
38 roughness)
39{
40 // Convert the Roughness->Exponent
41 // Square the roughness for a visually linear effect
42 const float oneOverR2 = 1.0 / (mRoughness*mRoughness);
43
44 // Using the Conversion from:
45 // Walter et al, EGSR 2007, "Microfacet Models for Refraction through Rough Surfaces"
46 mSpecularExponent = 2.0*oneOverR2*oneOverR2;
47
48 calculateNormalizationFactor();
49}

51// Initialize Specular Exponent and NormalizationFactor
52void
53KajiyaKayFabricBsdfLobe::calculateNormalizationFactor()
54{
55 // Normalization factor = gamma((n+3)/2) / gamma((n+2)/2) / pi^1.5
56 mNormalizationFactor = scene_rdl2::math::exp(lgammaf(mSpecularExponent*0.5f+1.5f) - lgammaf(mSpecularExponent*0.5+1.0f));
57 // 1/powf(pi,1.5) = 0.1795871;
58 mNormalizationFactor *= 0.1795871f;
59}

61// KajiyaKay Specular Lobe - cos(thetaD)^N, thetaD = abs(thetaO - thetaIPrime)
62// Regular cosine lobe specular, looks similar to DWAFabric but softer peaks
63Color
64KajiyaKayFabricBsdfLobe::eval(const BsdfSlice &slice,
65 const Vec3f &wi,
66 float *pdf) const
67{
68 // Prepare for early exit
69 if (pdf != nullptr) {
70 *pdf = 0.0f;
71 }
72 const float cosThetaWi = dot(mFrame.getN(), wi);
73 if (cosThetaWi <= sEpsilon) return sBlack;
74 const float cosThetaWo = dot(mFrame.getN(), slice.getWo());
75 if (cosThetaWo <= sEpsilon) return sBlack;
76
77 // BRDF = cos^N(thetaD)
78 // BRDF = cos^N(thetaIPrime - thetaO)
79 // BRDF = (cos(thetaIPrime)*cos(thetaO) + sin(thetaIPrime)*sin(thetaO))^N
80 const float cosThetaO = clamp(dot(mThreadDirection, slice.getWo()), -0.99f, 0.99f);
81 const float sinThetaO = scene_rdl2::math::sqrt(1 - cosThetaO*cosThetaO);
82
83 // IPrime is I reflected across the normal plane
84 const float cosThetaIPrime = (-1) * clamp(dot(mThreadDirection, wi), -0.99f, 0.99f);
85 const float sinThetaIPrime = scene_rdl2::math::sqrt(1 - cosThetaIPrime*cosThetaIPrime);
86
87 // cos(thetaD) = cos(thetaIPrime - thetaO);
88 const float cosThetaD = cosThetaO*cosThetaIPrime + sinThetaO*sinThetaIPrime;
89
90 if (cosThetaD < sEpsilon || cosThetaD > 1.f) {
91 return sBlack;
92 }
93
94 const float cosThetaDPow = powf(cosThetaD, mSpecularExponent);
95 float fabricBRDF = mNormalizationFactor * cosThetaDPow;
96
97 // PDF
98 if (pdf != nullptr && sinThetaIPrime > sEpsilon) {
99 // We sample theta_d, so to convert the PDF to sample theta_i we divide by
100 // the Jacobian (dOmegaI/dOmegaD) = sin(thetaI) / sin(thetaD)
101 // PDF for sampling theta_d is (n+1)/2pi * powf(costhetaD, n)
102 *pdf = (mSpecularExponent+1)*sOneOverTwoPi*cosThetaDPow;
103
104 // Converting the PDF from sampling theta_d to sampling theta_i
105 // Divide by jacobian (dOmegaI/dOmegaD)
106 const float sinThetaD = scene_rdl2::math::sqrt(clamp(1 - cosThetaD*cosThetaD, 0.0f, 1.0f));
107 *pdf *= sinThetaD;
108 *pdf /= sinThetaIPrime; // sinThetaI = sinThetaIPrime
109 // Control Variance Spikes - www.pbrt.org/hair.pdf (Section 1.8)
110 // If the ratio of pdf in theta_d with the BRDF gets too high, clamp it
111 // to control variance spikes. A ratio more than 4 seems satisfactory in tests.
112 if (fabricBRDF/(*pdf) > 4) {
113 *pdf = fabricBRDF/4;
114 }
115 }
116
117 const Fresnel *F = getFresnel();
118 const Color fresnel = (F != nullptr ? F->eval(cosThetaWo) : Color(1.0f));
119 const Color f = fresnel * getScale() * fabricBRDF;
120
121 // Soften hard shadow terminator due to shading normals
122 const float Gs = slice.computeShadowTerminatorFix(mFrame.getN(), wi);
123
124 return Gs * f;
125}

127Color
128KajiyaKayFabricBsdfLobe::sample(const BsdfSlice &slice,
129 float r1, float r2,
130 Vec3f &wi,
131 float &pdf) const
132{
133 const float cosNO = dot(mFrame.getN(), slice.getWo());
134 if (cosNO <= sEpsilon) {
135 pdf = 0.0f;
136 return sBlack;
137 }
138
139 Vec3f omegaI;
140 sampleVector(slice.getWo(), r1, r2, omegaI);
141
142 // Convert back in Global Space
143 wi = normalize(mFrame.localToGlobal(omegaI));
144
145 return eval(slice, wi, &pdf);
146}

148void
149KajiyaKayFabricBsdfLobe::sampleVector(const Vec3f& wo,
150 float r1, float r2,
151 Vec3f& wi) const
152{
153 // Theta is wrt the hair tangent
154 const float thetaO = safeACos(dot(mThreadDirection, wo));
155
156 // We sample theta_d and then convert it into the sample vector omega_i
157 const float thetaD = scene_rdl2::math::acos(powf(r1, 1/(mSpecularExponent+1)));
158
159 // theta_d = fabs(theta_o - theta_i_prime)
160 // so, theta_i_prime = theta_o - theta_d || theta_i_prime = theta_o + theta_d
161 float thetaIPrime = thetaO + thetaD;
162 if (thetaIPrime >= sPi) {
163 thetaIPrime = thetaO - thetaD;
164 }
165 if ((thetaD < thetaO) && (r2 < 0.5)) {
166 thetaIPrime = thetaO - thetaD;
167 }
168
169 float cosThetaI, sinThetaI;
170 sincos(thetaIPrime, &sinThetaI, &cosThetaI);
171
172 // Phi Sampling
173 const float phiM = sPi*r2;
174 float cosPhi, sinPhi;
175 sincos(phiM, &sinPhi, &cosPhi);
176
177 // Theta wrt tangent
178 wi.x = -cosThetaI;
179 wi.y = sinThetaI * cosPhi;
180 wi.z = sinThetaI * sinPhi;
181}
