76 /*! 4-wide bool type emulated with 16-wide bit vectors. */
77 struct sseb_t
78 {
79 __mmask16 v;
80
81 ////////////////////////////////////////////////////////////////////////////////
82 /// Constructors, Assignment & Cast Operators
83 ////////////////////////////////////////////////////////////////////////////////
84
85 __forceinline sseb_t ( const __mmask16 a ) : v(a) {}
86 __forceinline sseb_t ( const sseb_t& other ) { v = other.v; }
87 __forceinline sseb_t& operator=( const sseb_t& other ) { v = other.v; return *this; }
88 __forceinline sseb_t ( const sseb_m& other ) {
89 __m512i i = _mm512_extload_epi32(&other,_MM_UPCONV_EPI32_NONE,_MM_BROADCAST_4X16,_MM_HINT_NONE);
90 v = _mm512_cmp_epi32_mask(i,_mm512_set1_epi32(0),_MM_CMPINT_NE);
91 }
92 };
93

95 /// Assignment Operators
96 ////////////////////////////////////////////////////////////////////////////////
97
98 __forceinline sseb_m::sseb_m ( const sseb_t& other ) {
99 __m512i i = _mm512_mask_blend_epi32(other.v, _mm512_set1_epi32(0), _mm512_set1_epi32(-1));
100 _mm512_mask_extpackstorelo_epi32(this,0xf,i,_MM_DOWNCONV_EPI32_NONE, _MM_HINT_NONE);
101 }
102 __forceinline sseb_m& sseb_m::operator=( const sseb_t& other ) {
103 __m512i i = _mm512_mask_blend_epi32(other.v, _mm512_set1_epi32(0), _mm512_set1_epi32(-1));
104 _mm512_mask_extpackstorelo_epi32(this,0xf,i,_MM_DOWNCONV_EPI32_NONE, _MM_HINT_NONE);
105 return *this;
106 }
