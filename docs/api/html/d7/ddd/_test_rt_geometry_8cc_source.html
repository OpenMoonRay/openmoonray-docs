41 void visitPolygonMesh(PolygonMesh& p) override {
42 // update test quad face from original position
43 // (-1, 1, -1) ------------ (1, 1, -1)
44 // | |
45 // | |
46 // | |
47 // | |
48 // | |
49 // (-1,-1, -1) ------------ (1,-1, -1)

81 void generate(const GenerateContext& generateContext,
82 const moonray::shading::XformSamples& parent2render)
83 {
84 int testMode = generateContext.getRdlGeometry()->get(attrTestMode);
85 if (testMode == 0) {
86 createPolygonTestCase(generateContext, parent2render);
87 } else if (testMode == 1) {
88 createInstanceTestCase(generateContext, parent2render);
89 } else if (testMode == 2) {
90 createNestedInstanceTestCase(generateContext, parent2render);
91 }
92 }

94 void update(const UpdateContext& updateContext,
95 const moonray::shading::XformSamples& parent2render)
96 {
97 // Implement this method to update primitives created from
98 // generate call
99 PolygonTestCaseUpdate updater(parent2render);
100 forEachPrimitive(updater);
101 mDeformed = true;
102 }

106 void createPolygonTestCase(const GenerateContext& generateContext,

108 // create quad face
109 // (-1, 1, -1) ------------ (1, 1, -1)
110 // | |
111 // | |

122 vertices.push_back(Vec3fa( 1, 1, -1, 0));
123 std::unique_ptr<PolygonMesh> mesh = createPolygonMesh(

181 void createNestedInstanceTestCase(const GenerateContext& generateContext,
182 const moonray::shading::XformSamples& parent2render) {
183 // form a shared quad face to instance around, in its local space:
184 // (-1, 1, -1) ------------ (1, 1, -1)
185 // | |
186 // | |
187 // | P1 |
188 // | |
189 // | |
190 // (-1,-1, -1) ------------ (1,-1, -1)
191
192 PolygonMesh::FaceVertexCount faceVertexCount({4});
193 PolygonMesh::IndexBuffer indices({0, 1, 2, 3});
194 PolygonMesh::VertexBuffer vertices;
195 vertices.push_back(Vec3fa(-1, 1, -1, 0));
196 vertices.push_back(Vec3fa(-1, -1, -1, 0));
197 vertices.push_back(Vec3fa( 1, -1, -1, 0));
198 vertices.push_back(Vec3fa( 1, 1, -1, 0));
199 std::unique_ptr<PolygonMesh> P1 = createPolygonMesh(
200 std::move(faceVertexCount), std::move(indices),
201 std::move(vertices), LayerAssignmentId(0));
202
203 std::shared_ptr<SharedPrimitive> sharedP1 = createSharedPrimitive(
204 std::move(P1));
205 // create primitive group with two instances of above shared quad P1,
206 // and add in one regular primitive P2:
207 // PG
208 //
209 // (-1, 6, -1) ------------ (1, 6, -1)
210 // | P2 |
211 // | |
212 // (-1, 5, -1) ------------ (1, 5, -1)
213 //
214 //
215 //
216 // (-1, 3, -1) ------------ (1, 3, -1)
217 // | |
218 // | |
219 // | P1 |
220 // | |
221 // | |
222 // (-1, 1, -1) ------------ (1, 1, -1)
223 //
224 //
225 //
226 //
227 // (-1,-1, -1) ------------ (1,-1, -1)
228 // | |
229 // | |
230 // | P1 |
231 // | |
232 // | |
233 // (-1,-3, -1) ------------ (1,-3, -1)
234
235 std::unique_ptr<PrimitiveGroup> PG = createPrimitiveGroup();
236
237 PolygonMesh::FaceVertexCount faceVertexCount2({4});
238 PolygonMesh::IndexBuffer indices2({0, 1, 2, 3});
239 PolygonMesh::VertexBuffer vertices2;
240 vertices2.push_back(Vec3fa(-1, 6, -1, 0));
241 vertices2.push_back(Vec3fa(-1, 5, -1, 0));
242 vertices2.push_back(Vec3fa( 1, 5, -1, 0));
243 vertices2.push_back(Vec3fa( 1, 6, -1, 0));
244 std::unique_ptr<PolygonMesh> P2 = createPolygonMesh(
245 std::move(faceVertexCount2), std::move(indices2),
246 std::move(vertices2), LayerAssignmentId(0));
247 PG->addPrimitive(std::move(P2));
248 PG->addPrimitive(createInstance(Mat43::translate(Vec3f(0, 2, 0)),
249 sharedP1));
250 PG->addPrimitive(createInstance(Mat43::translate(Vec3f(0,-2, 0)),
251 sharedP1));
252
253
254 // now create two instances with above primitive group PG
255 // PG PG
256 //
257 // (-3, 6, -1) ------------ (-1, 6,-1) (1, 6, -1) ------------ (3, 6, -1)
258 // | P2 | | P2 |
259 // | | | |
260 // (-3, 5, -1) ------------ (-1, 5,-1) (1, 5, -1) ------------ (3, 5, -1)
261 //
262 //
263 //
264 // (-3, 3, -1) ------------ (-1, 3,-1) (1, 3, -1) ------------ (3, 3, -1)
265 // | | | |
266 // | | | |
267 // | P1 | | P1 |
268 // | | | |
269 // | | | |
270 // (-3, 1, -1) ------------ (-1, 1,-1) (1, 1, -1) ------------ (3, 1,-1)
271 //
272 //
273 //
274 //
275 // (-3,-1, -1) ------------ (-1,-1,-1) (1,-1, -1) ------------ (3,-1,-1)
276 // | | | |
277 // | | | |
278 // | P1 | | P1 |
279 // | | | |
280 // | | | |
281 // (-3,-3, -1) ------------ (-1,-3,-1) (1,-3, -1) ------------ (3,-3,-1)
282
283 std::shared_ptr<SharedPrimitive> sharedPG =
284 createSharedPrimitive(std::move(PG));
285 addPrimitive(
286 createInstance(Mat43::translate(Vec3f(-2, 0, 0)), sharedPG),
287 generateContext.getMotionBlurParams(), parent2render);
288 addPrimitive(
289 createInstance(Mat43::translate(Vec3f(2, 0, 0)), sharedPG),
290 generateContext.getMotionBlurParams(), parent2render);
291 }
