29///
30/// @class RetroReflectionBsdfLobe BsdfRetroreflection.h <shading/BsdfRetroreflection.h>
31/// @brief retro-reflective normalized phong lobe
32/// References:
33/// [1] Lafortune et al, ReportCW 1994, "Using the Modified Phong Reflectance Model for Physically Based Rendering"
34/// [2] Walter et al, EGSR 2007, "Microfacet Models for Refraction through Rough Surfaces"
35/// [3] Igehy, Siggraph 1999, "Tracing Ray Differentials"
36class RetroreflectionBsdfLobe : public BsdfLobe
37{
38public:
39 // Constructor / Destructor
40 RetroreflectionBsdfLobe(const scene_rdl2::math::Vec3f &N, const float roughness) :
41 BsdfLobe(Type(REFLECTION | GLOSSY), DifferentialFlags(0), false,
42 PROPERTY_NORMAL | PROPERTY_ROUGHNESS),
43
44 mN(N),
45 mInputRoughness(roughness)
46 {
47 // Apply roughness squaring to linearize roughness response
48 // See "Physically-Based Shading at Disney" Siggraph course notes.
49 mRoughness = mInputRoughness * mInputRoughness;
50 if (mRoughness < 0.001) {
51 mRoughness = 0.001;
52 }
53
54 // Use a directional differential scale that varies with roughness
55 // Taken from CookTorrance roughness-based differential computation
56 mdDFactor = sdDFactorMin + mRoughness * sdDFactorSlope;
57 }
58
59 // BsdfLobe API
60 scene_rdl2::math::Color eval(const BsdfSlice &slice, const scene_rdl2::math::Vec3f &wi, float *pdf = NULL) const override
61 {
62
63 // Prepare for early exit
64 if (pdf != NULL) {
65 *pdf = 0.0f;
66 }
67
68 const float cosThetaI = dot(mN, wi);
69 if (cosThetaI <= 0.0f) return scene_rdl2::math::sBlack;
70
71 const float cosOmegaO_OmegaI = dot(slice.getWo(), wi);
72 if (cosOmegaO_OmegaI <= 0.0f) return scene_rdl2::math::sBlack;
73
74 //convert roughness to phong exponent (see [2])
75 const float alphaP = 2.0f * scene_rdl2::math::rcp(mRoughness * mRoughness) - 2.0f;
76
77 // section 3.1 in [1]
78 const float power = scene_rdl2::math::pow(cosOmegaO_OmegaI, alphaP) *
79 scene_rdl2::math::sOneOverTwoPi;
80 const float normalizationFactor2 = (alphaP + 2.0f);
81 const scene_rdl2::math::Color f = computeScaleAndFresnel(cosThetaI) *
82 normalizationFactor2 * power *
83 (slice.getIncludeCosineTerm() ? cosThetaI : 1.0);
84
85 // Compute pdf of sampling ([1])
86 // section 3.3.1 in [1])
87 if (pdf != NULL) {
88 const float normalizationFactor1 = (alphaP + 1.0f);
89 *pdf = normalizationFactor1 * power;
90 }
91
92 // Soften hard shadow terminator due to shading normals
93 const float Gs = slice.computeShadowTerminatorFix(mN, wi);
94
95 return Gs * f;
96 }
97
98
99 scene_rdl2::math::Color sample(const BsdfSlice &slice, float r1, float r2,
100 scene_rdl2::math::Vec3f &wi, float &pdf) const override
101 {
102 const float cosNO = dot(mN, slice.getWo());
103 if (cosNO <= 0.0f) {
104 wi = scene_rdl2::math::Vec3f(scene_rdl2::math::zero);
105 pdf = 0.0f;
106 return scene_rdl2::math::sBlack;
107 }
108
109 // sample the retro-reflective phong lobe
110 // section 3.3.1 in [1]
111 const float alphaP = 2.0f * scene_rdl2::math::rcp(mRoughness * mRoughness) - 2.0f;
112 const float cosThetaM = pow(r1, scene_rdl2::math::rcp(alphaP + 1.0f));
113 const float sinThetaM = sqrt(1.0f - cosThetaM*cosThetaM);
114 const float phiM = 2.0f * scene_rdl2::math::sPi * r2;
115
116 scene_rdl2::math::Vec3f m = computeLocalSphericalDirection(cosThetaM, sinThetaM, phiM);
117
118 //sample along the outgoing vector
119 scene_rdl2::math::ReferenceFrame frame(slice.getWo());
120 wi = frame.localToGlobal(m);
121
122 return eval(slice, wi, &pdf);
123 }
124
125
126 scene_rdl2::math::Color albedo(const BsdfSlice &slice) const override
127 {
128 float cosThetaWo = dot(mN, slice.getWo());
129 return computeScaleAndFresnel(scene_rdl2::math::abs(cosThetaWo));
130 }
131
132 void differentials(const scene_rdl2::math::Vec3f &wo, const scene_rdl2::math::Vec3f &wi,
133 float r1, float r2, const scene_rdl2::math::Vec3f &dNdx, const scene_rdl2::math::Vec3f &dNdy,
134 scene_rdl2::math::Vec3f &dDdx, scene_rdl2::math::Vec3f &dDdy) const override
135 {
136 // Reverse incoming differentials, based on [3]
137 // Factors taken from CookTorrance roughness-based differential computation
138 // TODO: Not sure this is even close to correct.
139 dDdx *= -mdDFactor;
140 dDdy *= -mdDFactor;
141 }
142
143 bool getProperty(Property property, float *dest) const override
144 {
145 bool handled = true;
146 switch (property)
147 {
148 case PROPERTY_ROUGHNESS:
149 *dest = mInputRoughness;
150 *(dest + 1) = mInputRoughness;
151 break;
152 case PROPERTY_NORMAL:
153 {
154 *dest = mN.x;
155 *(dest + 1) = mN.y;
156 *(dest + 2) = mN.z;
157 }
158 break;
159 default:
160 handled = BsdfLobe::getProperty(property, dest);
161 break;
162 }
163
164 return handled;
165 }
166
167 void show(std::ostream& os, const std::string& indent) const override
168 {
169 const scene_rdl2::math::Color& scale = getScale();
170 const Fresnel * const fresnel = getFresnel();
171 os << indent << "[RetroreflectionBsdfLobe]\n";
172 os << indent << " " << "scale: "
173 << scale.r << " " << scale.g << " " << scale.b << "\n";
174 if (fresnel) {
175 fresnel->show(os, indent + " ");
176 }
177 }
178
179private:
180 // Derive a directional differential scale that varies according to roughness
181 // Copied from Cook-Torrance BRDF (BsdfCookTorrance.h)
182 static constexpr float sdDFactorMin = 1.0f;
183 static constexpr float sdDFactorMax = 8.0f;
184 static constexpr float sdDFactorSlope = (sdDFactorMax - sdDFactorMin);
185
186 scene_rdl2::math::Vec3f mN;
187 float mInputRoughness;
188 float mRoughness;
189 float mdDFactor;
190};
