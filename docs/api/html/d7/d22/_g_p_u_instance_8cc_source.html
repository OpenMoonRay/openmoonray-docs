13bool
14GPUInstance::build(CUstream cudaStream,
15 OptixDeviceContext context,
16 std::string* errorMsg)
17{
18 // If we have already visited, do nothing.
19 if (mIsBuilt) {
20 return true;
21 }
22 mIsBuilt = true;
23
24 // build() the referenced group. If it has already been built,
25 // e.g. it is referenced by another group/instance, then
26 // mGroup->build() does nothing.
27 if (!mGroup->build(cudaStream, context, errorMsg)) {
28 // error
29 return false;
30 }
31
32 if (mHasMotionBlur) {
33 // It's ugly, but this is how NVIDIA says you are supposed to set this up.
34 // 12 * sizeof(float) = Optix 4x3 transform matrix size
35 size_t transformSizeInBytes = sizeof(OptixMatrixMotionTransform)
36 + (sNumMotionKeys - 2) * 12 * sizeof(float);
37 OptixMatrixMotionTransform *transform =
38 (OptixMatrixMotionTransform*) alloca(transformSizeInBytes);
39
40 transform->child = mGroup->mTopLevelIAS;
41 transform->motionOptions.numKeys = sNumMotionKeys;
42 transform->motionOptions.timeBegin = 0.f;
43 transform->motionOptions.timeEnd = 1.f;
44 transform->motionOptions.flags = OPTIX_MOTION_FLAG_NONE;
45 for (int i = 0; i < sNumMotionKeys; i++) {
46 mXforms[i].toOptixTransform(transform->transform[i]);
47 }
48
49 mMMTTraversableBuf.allocAndUpload((char*)(transform), transformSizeInBytes);
50
51 if (optixConvertPointerToTraversableHandle(context,
52 mMMTTraversableBuf.deviceptr(),
53 OPTIX_TRAVERSABLE_TYPE_MATRIX_MOTION_TRANSFORM,
54 &mMMTTraversable) != OPTIX_SUCCESS) {
55 return false;
56 }
57 }
58
59 return true;
60}
