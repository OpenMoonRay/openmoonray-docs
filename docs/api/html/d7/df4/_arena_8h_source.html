58// Container of memory blocks. This is shared amongst threads so is fully thread
59// safe. It allows blocks which are reclaimed from one thread to be handed out
60// to a separate thread.
61class ArenaBlockPool : private util::RefCount<ArenaBlockPool, util::AlignedDeleter<ArenaBlockPool>>
62{
63public:
64 finline explicit ArenaBlockPool(unsigned blockSize = DEFAULT_ARENA_BLOCK_SIZE) :
65 mBlockSize(blockSize)
66 {
67 MNRY_ASSERT_REQUIRE(blockSize && util::isPowerOfTwo(blockSize));
68 mTotalBlocks = 0;
69 }
70
71 finline ~ArenaBlockPool()
72 {
73 cleanUp();
74 }
75
76 // Deallocates all blocks.
77 finline void cleanUp()
78 {
79 // Make sure all existing blocks have been handed back to us.
80 MNRY_ASSERT(mFreeBlocks.size() == mTotalBlocks);
81
82 // Delete all blocks.
83 ArenaBlock *block = nullptr;
84 do {
85 block = (ArenaBlock *)mFreeBlocks.pop();
86 delete block;
87 } while (block);
88
89 mTotalBlocks = 0;
90 }
91
92 finline unsigned getMemoryUsage() const
93 {
94 return mTotalBlocks * mBlockSize;
95 }
96
97 finline unsigned getBlockSize() const
98 {
99 return mBlockSize;
100 }
101
102 finline ArenaBlock *allocateBlock()
103 {
104 ArenaBlock *block = (ArenaBlock *)mFreeBlocks.pop();
105 if (!block) {
106 block = new ArenaBlock(mBlockSize, CACHE_LINE_SIZE);
107 ++mTotalBlocks;
108 }
109
110 return block;
111 }
112
113 finline void freeBlock(ArenaBlock *block)
114 {
115 mFreeBlocks.push(block);
116 }
117
118protected:
119 unsigned mBlockSize;
120 tbb::atomic<unsigned> mTotalBlocks;
121
122 CACHE_ALIGN util::ConcurrentSList mFreeBlocks;
123};

236finline uint8_t *
237Arena::alloc(unsigned size, unsigned alignment)
238{
239 MNRY_ASSERT(isValid());
240 MNRY_ASSERT(mPtr);
241
242 // Alloc sizes and return addresses are on 4-byte boundaries.
243 size = std::max<unsigned>(size, 4);
244 alignment = std::max<unsigned>(alignment, 4);
245
246 align(alignment);
247
248 uint8_t *ret = mPtr;
249 mPtr += size;
250
251 // Did alloc succeed?
252 if (mPtr > mEnd) {
253
254 // Alloc failed, get a brand new fresh block to try and satisfy it.
255 allocNewBlock();
256
257 align(alignment);
258
259 ret = mPtr;
260 mPtr += size;
261
262 if (mPtr > mEnd) {
263 logging::Logger::error("Block size too small to satisfy allocation in arena allocator, ",
264 size, " wanted (", alignment, " byte aligned), ",
265 mBlockPool->getBlockSize(), " block size.\n");
266 return nullptr;
267 }
268 }
269
270#ifdef DEBUG
271 // Debug only - clear out memory.
272 memset(ret, 0xac, size);
273#endif
274
275 MNRY_ASSERT(isValid());
276
277 return ret;
278}

280finline void
281Arena::setPtr(uint8_t *ptr)
282{
283 MNRY_ASSERT(mBlockPool);
284
285 if (ptr == nullptr) {
286
287 if (!mBlocks.empty()) {
288
289 // Rewind until we have only a single block and set the pointer to
290 // the start of that block. Avoids a pathological case where we
291 // continuously clear and alloc new blocks redundantly.
292 while (mBlocks.size() > 1) {
293 ArenaBlock *block = mBlocks.back();
294 mBlockPool->freeBlock(block);
295 mBlocks.pop_back();
296 }
297
298 MNRY_ASSERT(mBlocks.size() == 1);
299 setActiveBlock(mBlocks.back());
300 }
301
302 } else {
303
304 MNRY_ASSERT(isValidPtr(ptr));
305
306 while (1) {
307 if (ptr >= mBase && ptr <= mEnd) {
308 mPtr = ptr;
309 return;
310 }
311
312 ArenaBlock *block = mBlocks.back();
313 mBlockPool->freeBlock(block);
314 mBlocks.pop_back();
315
316 // If this asserts, ptr wasn't within any of our blocks.
317 MNRY_ASSERT(!mBlocks.empty());
318
319 setActiveBlock(mBlocks.back());
320 }
321 }

332finline bool
333Arena::isValidPtr(const void *ptr) const
334{
335 MNRY_ASSERT(isValid());
336
337 // Check if pointer belongs in top block.
338 // We are comparing less than or equal to with mPtr to account for the case
339 // that zero allocations were made.
340 if (ptr >= mBase && ptr <= mPtr) {
341 return true;
342 }
343
344 // Check all remaining blocks.
345 int i = int(mBlocks.size()) - 2;
346 while (i >= 0) {
347 const ArenaBlock *block = mBlocks[i];
348 if (ptr >= block->mMemory && ptr <= block->mMemory + block->mSize) {
349 return true;
350 }
351 --i;
352 }
353
354 return false;
355}

357finline void
358Arena::resetInternal()
359{
360 mBase = nullptr;
361 mEnd = nullptr;
362 mPtr = nullptr;
363
364 if (!mBlockPool) {
365 return;
366 }
367
368 for (auto it = mBlocks.begin(); it != mBlocks.end(); ++it) {
369 mBlockPool->freeBlock(*it);
370 }
371
372 mBlocks.clear();
373
374 // Don't clear mBlockPool here.
375}

514finline unsigned
515FixedArena::getFree(unsigned alignment) const
516{
517 MNRY_ASSERT(util::isPowerOfTwo(alignment));
518
519 alignment = std::max<unsigned>(alignment, 4);
520 unsigned mask = ~(alignment - 1);
521
522 ptrdiff_t low = ptrdiff_t((size_t(mLow) + (alignment - 1)) & mask); // align up
523 ptrdiff_t high = ptrdiff_t(size_t(mHigh) & mask); // align down
524
525 return unsigned(std::max(high - low, ptrdiff_t(0)));
526}

528finline uint8_t *
529FixedArena::alloc(unsigned size, unsigned alignment)
530{
531 MNRY_ASSERT(mHigh >= mLow);
532
533 // Alloc sizes and return addresses are on 4-byte boundaries.
534 size = std::max<unsigned>(size, 4);
535 alignment = std::max<unsigned>(alignment, 4);
536
537 uint8_t *oldLow = mLow;
538
539 align(alignment);
540
541 uint8_t *ret = mLow;
542 mLow += size;
543
544 // Success?
545 if (mLow <= mHigh) {
546#ifdef DEBUG
547 memset(ret, 0xa0, size); // Debug only - clear out memory.
548#endif
549 return ret;
550 }
551
552 // revert old state
553 mLow = oldLow;
554
555 logging::Logger::warn("Out of memory in fixed arena allocator, %d wanted (%d byte aligned), %d available.\n",
556 size, alignment, getFree(alignment));
557
558 return nullptr;
559}

561finline uint8_t *
562FixedArena::allocHigh(unsigned size, unsigned alignment)
563{
564 MNRY_ASSERT(mHigh >= mLow);
565
566 size = std::max<unsigned>(size, 4u);
567 alignment = std::max<unsigned>(alignment, 4u);
568
569 uint8_t *oldHigh = mHigh;
570
571 alignHigh(size, alignment);
572
573 if (mHigh >= mLow) {
574#ifdef DEBUG
575 memset(mHigh, 0xad, size); // Debug only - clear out memory.
576#endif
577 return mHigh;
578 }
579
580 mHigh = oldHigh;
581
582 logging::Logger::warn("Out of memory in fixed arena allocator, %d wanted (%d byte aligned), %d available.\n",
583 size, alignment, getFree(alignment));
584
585 return nullptr;
