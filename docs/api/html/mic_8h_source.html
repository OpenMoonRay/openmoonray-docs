131 template<const int lane>
132 __forceinline mic_f lane_shuffle_gather(const mic_f &v0,const mic_f &v1,const mic_f &v2,const mic_f &v3)
133 {
134 mic_f t = lshuf<lane,lane,lane,lane>(v0);
135 t = lshuf<lane,lane,lane,lane>(0xf0,t,v1);
136 t = lshuf<lane,lane,lane,lane>(0xf00,t,v2);
137 t = lshuf<lane,lane,lane,lane>(0xf000,t,v3);
138 return t;
139 }
