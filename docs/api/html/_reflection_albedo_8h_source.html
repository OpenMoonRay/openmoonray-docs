11/*
12 * This file encodes the average directional albedo and
13 * the normalization factor required for the Cook-Torrance Microfacet BRDF.
14 * As of 10/14/19, we provide tables for Beckmann and GGX NDFs.
15 * This class provides accessors to encapsulate the
16 * indexing into the albedo table. Values can be looked up as:
17 * E(cosThetaO, roughness)
18 * oneMinusEavg(roughness)
19 * These tables have been calculated using moonshine_devtools/compute_albedo
20 * Ref:
21 * A Microfacet Based Coupled Specular-Matte BRDF Model, Kelemen'01
22 *

33class ReflectionAlbedo
34{
35public:
36 static float E(ispc::MicrofacetDistribution type,
37 float cosTheta,
38 float roughness)
39 {
40 float w;
41 size_t iLow, iHigh;
42 index(cosTheta, roughness,
43 w, iLow, iHigh);
44 float e = 0.0f;
45 switch (type) {
46 case ispc::MICROFACET_DISTRIBUTION_BECKMANN:
47 e = scene_rdl2::math::lerp(mBeckmannE[iLow],
48 mBeckmannE[iHigh],
49 w);
50 break;
51 case ispc::MICROFACET_DISTRIBUTION_GGX:
52 e = scene_rdl2::math::lerp(mGGXE[iLow],
53 mGGXE[iHigh],
54 w);
55 break;
56 }
57 return e;
58 }
59
60 static float oneMinusAvg(ispc::MicrofacetDistribution type,
61 float roughness)
62 {
63 float w;
64 const size_t iLow = rIndex(roughness, w);
65 const size_t iHigh = scene_rdl2::math::min(iLow+1, COMP-1);
66 float e = 0.0f;
67 switch (type) {
68 case ispc::MICROFACET_DISTRIBUTION_BECKMANN:
69 e = scene_rdl2::math::lerp(mBeckmannOneMinusEavg[iLow],
70 mBeckmannOneMinusEavg[iHigh],
71 w);
72 break;
73 case ispc::MICROFACET_DISTRIBUTION_GGX:
74 e = scene_rdl2::math::lerp(mGGXOneMinusEavg[iLow],
75 mGGXOneMinusEavg[iHigh],
76 w);
77 break;
78 }
79 return e;
80 }
81
82private:
83 static void index(float cosTheta,
84 float roughness,
85 float& w,
86 size_t& iLow,
87 size_t& iHigh)
88 {
89 const size_t lIndex = rIndex(roughness, w);
90 const size_t hIndex = scene_rdl2::math::min(lIndex+1, COMP*COMP-1);
91 const float if1 = lIndex * COMP +
92 cosTheta * (COMP - 1);
93 const float if2 = hIndex * COMP +
94 cosTheta * (COMP - 1);
95 // Take cosTheta's ceiling for the index
96 iLow = static_cast<size_t>(ceil(if1));
97 iHigh = static_cast<size_t>(ceil(if2));
98 }
99
100 static size_t rIndex(float roughness,
101 float& weight)
102 {
103 const float i = roughness * (COMP - 1);
104 const float indexI = floor(i);
105 weight = (i-indexI);
106 return static_cast<size_t>(indexI);
107 }
108 static const size_t COMP = 33;
109 static const float mBeckmannE[COMP*COMP];
110 static const float mGGXE[COMP*COMP];
111 static const float mBeckmannOneMinusEavg[COMP];
112 static const float mGGXOneMinusEavg[COMP];
113};
