120class FalloffCurve
121{
122public:
123 FalloffCurve() :
124 mType(FALLOFF_CURVE_TYPE_LINEAR) {}
125
126 FalloffCurve(FalloffCurveType curveType)
127 {
128 init(curveType);
129 }
130
131 void init(FalloffCurveType curveType)
132 {
133 mType = (FalloffCurveType)scene_rdl2::math::min(curveType, FALLOFF_CURVE_TYPE_NUM_TYPES - 1);
134 }
135
136 /// HUD validation and type casting
137 static uint32_t hudValidation(bool verbose);
138 HUD_AS_ISPC_METHODS(FalloffCurve);
139
140
141 float eval(float t) const
142 {
143 t = scene_rdl2::math::saturate(t);
144
145 switch (mType) {
146 case FALLOFF_CURVE_TYPE_NONE:
147 return (t > 0.0f) ? 1.0f : 0.0f;
148 case FALLOFF_CURVE_TYPE_LINEAR:
149 return t;
150 case FALLOFF_CURVE_TYPE_EASEIN:
151 return t * t;
152 case FALLOFF_CURVE_TYPE_EASEOUT:
153 return t * (2.0f - t);
154 case FALLOFF_CURVE_TYPE_EASEINOUT:
155 return t * t * (3.0f - 2.0f * t);
156 default:
157 return t;
158 }
159 }
160
161protected:
162 FALLOFF_CURVE_MEMBERS;
163
164};

174class OldFalloffCurve
175{
176public:
177 OldFalloffCurve() :
178 mType(OLD_FALLOFF_CURVE_TYPE_NONE), mExp(1.0f), mG1(0.0f), mInvH0(0.0f) {}
179
180 OldFalloffCurve(OldFalloffCurveType curveType, float exp)
181 {
182 init(curveType, exp);
183 }
184
185 void init(OldFalloffCurveType curveType, float exp)
186 {
187 mType = (OldFalloffCurveType)scene_rdl2::math::min(curveType, OLD_FALLOFF_CURVE_TYPE_NUM_TYPES - 1);
188 mExp = exp;
189 mG1 = g(1.0f); // initialize this before mInvH0!
190 mInvH0 = 1.0f / h(0.0f);
191 }
192
193 /// HUD validation and type casting
194 static uint32_t hudValidation(bool verbose);
195 HUD_AS_ISPC_METHODS(OldFalloffCurve);
196
197
198 float eval(float t) const
199 {
200 t = scene_rdl2::math::saturate(t);
201
202 switch (mType) {
203 case OLD_FALLOFF_CURVE_TYPE_NONE:
204 return t;
205 case OLD_FALLOFF_CURVE_TYPE_EASEOUT:
206 return scene_rdl2::math::pow(3.0f * t * t - 2.0f * t * t * t, mExp);
207 case OLD_FALLOFF_CURVE_TYPE_GAUSSIAN:
208 return scene_rdl2::math::pow((scene_rdl2::math::cos((1.0f - t) * scene_rdl2::math::sPi) + 1.0f) * 0.5f,
209 mExp);
210 case OLD_FALLOFF_CURVE_TYPE_LINEAR:
211 return scene_rdl2::math::pow(t, mExp);
212 case OLD_FALLOFF_CURVE_TYPE_SQUARED:
213 return t * t;
214 case OLD_FALLOFF_CURVE_TYPE_NATURAL:
215 return naturalOldFalloff(t);
216 }
217 return t;
218 }
219
220protected:
221 OLD_FALLOFF_CURVE_MEMBERS;
222
223private:
224 //
225 // We use the following function in the [0..1] interval:
226 // natural_falloff(x) = h(x) / h(0)
227 // h(x) = g(x) - g(1)
228 // g(x) = f(tail * x + 1)
229 // f(x) = 1.0 / (x ** power)
230 // Here tail and g(x) are chosen so that
231 // g'(0) is a large negative value (high slope)
232 // g'(1) is close to 0 (small slope)
233 // In gnuplot:
234 // tail = 7
235 // power = 2
236 // plot [x=0:1][0:1] ((1.0 / ((tail*x+1)**power)) - (1.0 / ((tail*1+1)**power))) / ((1.0 / ((tail*0+1)**power)) -
237 // (1.0 / ((tail*1+1)**power)))
238 //
239 #define OLD_FALLOFF_HEAD 1.0f
240 #define OLD_FALLOFF_TAIL 7.0f
241
242 float naturalOldFalloff(float x) const { return 1.0f - (h(x) * mInvH0); }
243
244 float f(float x) const { return 1.0f / scene_rdl2::math::pow(x, mExp); }
245 float g(float x) const { return f(OLD_FALLOFF_TAIL * x + OLD_FALLOFF_HEAD); }
246 float h(float x) const { return g(x) - mG1; }
247};

251finline scene_rdl2::math::Color
252computeLightRadiance(const scene_rdl2::rdl2::Light *light,
253 const scene_rdl2::rdl2::AttributeKey<scene_rdl2::rdl2::Rgb> &colorKey,
254 const scene_rdl2::rdl2::AttributeKey<scene_rdl2::rdl2::Float> &intensityKey,
255 const scene_rdl2::rdl2::AttributeKey<scene_rdl2::rdl2::Float> &exposureKey,
256 const scene_rdl2::rdl2::AttributeKey<scene_rdl2::rdl2::Bool> &normalizedKey,
257 const scene_rdl2::rdl2::AttributeKey<scene_rdl2::rdl2::Bool> &applySceneScaleKey,
258 float invSurfaceArea)
259{
260 // Compute color, intensity and exposure contributions
261 scene_rdl2::math::Color color = light->get<scene_rdl2::rdl2::Rgb>(colorKey);
262 float exposure = light->get<scene_rdl2::rdl2::Float>(exposureKey);
263 color *= light->get<scene_rdl2::rdl2::Float>(intensityKey) * powf(2.0f, exposure);
264
265 // When normalized, we can change the size of the emitter without
266 // changing the amount of total energy cast into the scene.
267 bool normalized = light->get<scene_rdl2::rdl2::Bool>(normalizedKey);
268
269 if (!normalized) {
270 // Interpret as radiance.
271 return color;
272 }
273
274 // Interpret as flux
275 // For backwards compatibility, we can multiply the surface area
276 // by the scene scale and pi for purposes of flux computation.
277 // Our goal is to eliminate this behavior (or at least default it to false)
278 // so we match other renderer's implemntation of normalized lights.
279 if (light->get<scene_rdl2::rdl2::Bool>(applySceneScaleKey)) {
280 const scene_rdl2::rdl2::SceneVariables &sv = light->getSceneClass().getSceneContext()->getSceneVariables();
281 float sceneScale = sv.get(scene_rdl2::rdl2::SceneVariables::sSceneScaleKey);
282 invSurfaceArea *= 1.0f / (sceneScale * sceneScale);
283 invSurfaceArea *= scene_rdl2::math::sOneOverPi;
284 }
285
286 return color * invSurfaceArea;
287}

304// Approximation which breaks down the closer p gets to the light but should
305// be sufficiently accurate as the light gets more distant.
306finline float
307getPlanarApproxSubtendedSolidAngle(const scene_rdl2::math::Vec3f &p, // point we are illuminating
308 const scene_rdl2::math::Vec3f &lightPos,
309 const scene_rdl2::math::Vec3f &lightDir,
310 float lightArea)
311{
312 scene_rdl2::math::Vec3f negWi = p - lightPos;
313 float distSq = negWi.lengthSqr();
314 if (distSq < scene_rdl2::math::sEpsilon) {
315 return scene_rdl2::math::sTwoPi;
316 }
317 float dist = scene_rdl2::math::sqrt(distSq);
318 float solidAngle = dot(negWi, lightDir) * lightArea / (distSq * dist);
319 return scene_rdl2::math::clamp(solidAngle, 0.0f, scene_rdl2::math::sTwoPi);
320}
