225 template<typename L> __forceinline XformComponent<L> slerp(const XformComponent<L>& a, const XformComponent<L>& b, const ScalarT t) {
226 XformComponent<L> l2p;
227 l2p.t = lerp(a.t, b.t, t);
228 l2p.r = normalize(slerp(a.r, b.r, t));
229 l2p.s = lerp(a.s, b.s, t);

307 __forceinline BBox3f
308 transformBBox(const XformT<Mat3<Vec3f> >& m, const BBox3f& bb)
309 {
310 const auto xa = m.row0() * bb.lower.x;
311 const auto xb = m.row0() * bb.upper.x;
312
313 const auto ya = m.row1() * bb.lower.y;
314 const auto yb = m.row1() * bb.upper.y;
315
316 const auto za = m.row2() * bb.lower.z;
317 const auto zb = m.row2() * bb.upper.z;
318
319 return BBox3f(
320 min(xa, xb) + min(ya, yb) + min(za, zb) + m.row3(),
321 max(xa, xb) + max(ya, yb) + max(za, zb) + m.row3()
322 );
323 }
