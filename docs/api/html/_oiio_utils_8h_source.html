77inline bool
78OiioUtils::getMetadata(const OIIO::ImageSpec &spec, const std::string &name, float out[3])
79{
80 // We need to set all of BASETYPE, AGGREGATE and VECSEMANTICS for TypeDesc in this case somehow.
81 const OIIO::ParamValue *p =
82 spec.find_attribute(name, OIIO::TypeDesc(OIIO::TypeDesc::FLOAT, OIIO::TypeDesc::VEC3, OIIO::TypeDesc::VECTOR));
83 if (!p) return false;
84 out[0] = ((const float *)p->data())[0];
85 out[1] = ((const float *)p->data())[1];
86 out[2] = ((const float *)p->data())[2];
87 return true;
88}

92OiioUtils::getPixChanOffset(const OIIO::ImageSpec &spec, const std::string &channelName)
93{
94 for (int offset = 0; offset < spec.nchannels; ++offset) {
95 if (spec.channelnames[offset] == channelName) {
96 return offset;
97 }

99 return -1; // error : Could not find target channel
100}
