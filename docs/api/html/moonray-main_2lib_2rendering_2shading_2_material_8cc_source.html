38Material::~Material()
39{
40 // Not thread safe, all objects should be created/destroyed on a single thread.
41 if (mShadeQueue) {
42
43 {
44 tbb::mutex::scoped_lock lock(sShadeQueueMutex);
45
46 // Check the shade queue size also since it may have already been destroyed
47 // during global program destruction time.
48 if (sShadeQueues.size()) {
49
50 // Remove ourselves from global list of shade queues.
51 for (auto it = sShadeQueues.begin(); it != sShadeQueues.end(); ++it) {
52 if (*it == mShadeQueue) {
53 sShadeQueues.erase(it);
54 break;
55 }
56 }
57 }
58 }
59 scene_rdl2::util::alignedFreeArray(mShadeEntries);
60 scene_rdl2::util::alignedFreeDtor(mShadeQueue);
61 }
62
63 {
64 tbb::mutex::scoped_lock lock(sMaterialListMutex);
65
66 // Remove ourselves from global list of Materials.
67 for (auto it = sAllMaterials.begin(); it != sAllMaterials.end(); ++it) {
68 if (*it == this) {
69 sAllMaterials.erase(it);
70 break;
71 }
72 }
73 }
74}

76void
77Material::deferEntriesForLaterProcessing(mcrt_common::ThreadLocalState *tls,
78 unsigned numEntries,
79 SortedRayState *entries)
80{
81 if (!numEntries) {
82 return;
83 }
84
85 EXCL_ACCUMULATOR_PROFILE(tls, EXCL_ACCUM_DEFER_SHADE_ENTRIES);
86
87 // Keep track of calls, if this gets too high then we'll need to investigate.
88 ++sDeferredEntryCalls;
89 sTotalDeferredEntries += numEntries;
90
91 unsigned deferredCalls = sDeferredEntryCalls;
92
93 const unsigned maxCallsToLog = 5;
94
95 if (deferredCalls < maxCallsToLog) {
96 mcrt_common::ExclusiveAccumulators *acc = MNRY_VERIFY(getExclusiveAccumulators(tls));
97 unsigned accStack = acc->getStackSize();
98 scene_rdl2::logging::Logger::warn("Call to Material::deferEntriesForLaterProcessing encountered (accumulator stack = ", accStack, ", handler stack = ", tls->mHandlerStackDepth, ".");
99 } else if (deferredCalls == maxCallsToLog) {
100 scene_rdl2::logging::Logger::warn("Multiple calls to Material::deferEntriesForLaterProcessing encountered, no more will be reported this frame.");
101 }
102
103 {
104 tbb::mutex::scoped_lock lock(mDeferredEntryMutex);
105 mDeferredEntries.insert(mDeferredEntries.end(), entries, entries + numEntries);
106 }
107}

109void
110Material::retrieveDeferredEntries(mcrt_common::ThreadLocalState *tls,
111 scene_rdl2::alloc::Arena *arena,
112 unsigned &numEntries,
113 SortedRayState *&entries)
114{
115 if (mDeferredEntries.empty()) {
116 return;
117 }
118
119 EXCL_ACCUMULATOR_PROFILE(tls, EXCL_ACCUM_DEFER_SHADE_ENTRIES);
120
121 mDeferredEntryMutex.lock();
122
123 unsigned numDeferredEntries = (unsigned)mDeferredEntries.size();
124 unsigned totalEntries = numEntries + numDeferredEntries;
125
126 SortedRayState *allEntries = arena->allocArray<SortedRayState>(totalEntries);
127
128 memcpy(allEntries, &mDeferredEntries[0], numDeferredEntries * sizeof(SortedRayState));
129 mDeferredEntries.clear();
130
131 // Don't hold the lock any longer than we strictly have to.
132 mDeferredEntryMutex.unlock();
133
134 memcpy(allEntries + numDeferredEntries, entries, numEntries * sizeof(SortedRayState));
135
136 numEntries = totalEntries;
137 entries = allEntries;
138
139 // The shadeBundleHandler expects sorted entries so redo that sort since
140 // we're concatenating various SortRayState arrays from different threads.
141 std::sort(entries, entries + numEntries,
142 [](const SortedRayState &a, const SortedRayState &b) -> bool
143 {
144 return EXTRACT_KEY32(a, offsetof(SortedRayState, mSortKey)) <
145 EXTRACT_KEY32(b, offsetof(SortedRayState, mSortKey));
146 });
147
148 MNRY_ASSERT( (scene_rdl2::util::isSorted32<SortedRayState, offsetof(SortedRayState, mSortKey)>(numEntries, entries)) );
149}

177bool
178Material::areAllShadeQueuesEmpty()
179{
180 for (auto it = sShadeQueues.begin(); it != sShadeQueues.end(); ++it) {
181 if (!(*it)->isEmpty()) {
182 return false;
183 }
184 }
185
186 // We also need to check for the case where a Material's queue may be empty
187 // but it could still have some entries stored for deferred processing.
188 for (auto it = sAllMaterials.begin(); it != sAllMaterials.end(); ++it) {
189 if (!(*it)->mDeferredEntries.empty()) {
190 return false;
191 }
192 }
193
194 return true;
195}

197unsigned
198Material::flushNonEmptyShadeQueue(mcrt_common::ThreadLocalState *tls)
199{
200 // Always force sFlushCycleIdx to increment by at least 1.
201 size_t startIdx = sFlushCycleIdx++;
202
203 scene_rdl2::alloc::Arena *arena = &tls->mArena;
204 SCOPED_MEM(arena);
205
206 for (size_t i = 0; i < sShadeQueues.size(); ++i) {
207
208 size_t cycleIdx = startIdx + i;
209 ShadeQueue *queue = sShadeQueues[cycleIdx % sShadeQueues.size()];
210#if 1
211 unsigned flushed = queue->drain(tls, VLEN, arena);
212#else
213 unsigned flushed = queue->flush(tls, arena);
214#endif
215
216 if (flushed) {
217
218 // There is a harmless race here (so we can avoid any heavier synchronization).
219 // Overall it's fine and won't cause any problems.
220 if (cycleIdx > sFlushCycleIdx) {
221 sFlushCycleIdx = cycleIdx;
222 }
223 return flushed;
224 }
225 }
226
227 // We also need to check for the case where a Material's queue may be empty
228 // but it could still have some entries stored for deferred processing.
229 // In this case we move the deferred entries into the actual shader
230 // queues and return signal that there is still shade queues which contain
231 // entries.
232 unsigned numDeferredEntries = 0;
233 for (auto it = sAllMaterials.begin(); it != sAllMaterials.end(); ++it) {
234 if (!(*it)->mDeferredEntries.empty()) {
235 Material *material = *it;
236 unsigned numEntries = 0;
237 SortedRayState *entries = nullptr;
238 material->retrieveDeferredEntries(tls, arena, numEntries, entries);
239
240 // This check is here since another thread may have stolen the
241 // deferred entries in the meantime.
242 if (numEntries && entries) {
243 numDeferredEntries += numEntries;
244 ShadeQueue *queue = material->mShadeQueue;
245 MNRY_VERIFY(queue)->addEntries(tls, numEntries, entries, arena);
246 }
247 }
248 }
249
250 // This is expected to be zero 99.99% of the time.
251 return numDeferredEntries;
252}

254void
255Material::allocShadeQueue(unsigned shadeQueueSize, ShadeQueue::Handler handler)
256{
257 // This should only ever get called once at most per material.
258 MNRY_ASSERT(!mShadeQueue);
259
260 mShadeQueue = scene_rdl2::util::alignedMallocCtor<ShadeQueue>(CACHE_LINE_SIZE);
261 mShadeEntries = scene_rdl2::util::alignedMallocArray<ShadeQueue::EntryType>(shadeQueueSize, CACHE_LINE_SIZE);
262 mShadeQueue->init(shadeQueueSize, mShadeEntries);
263 mShadeQueue->setHandler(handler, this);
264
265 sShadeQueues.push_back(mShadeQueue);
266
267 // TODO: We don't support deleting primitives currently. We'll need to add
268 // some extra logic to maintain a valid bundled id when that gets implemented.
269 mMaterialId = uint32_t(sShadeQueues.size());
270}
