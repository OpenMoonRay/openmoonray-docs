32

35
36 // Will be 3 or 4
37 int mVertsPerFace;
38
39 // Indices for tessellated faces. Size of mIndexBuffer is numFaces * mVertsPerFace.

42

44 size_t mVertexCount;
45
46 // 0 if no motion blur

48

54
55 // WARNING: Baked geometry vertices and normals are in RENDER SPACE. You may want to
56 // transform back to object space. Be careful with transformNormal() as it uses
57 // the inverse xform!
58
59 // Mapping from tessellated face id to base face id.

61
62 // Mapping of face to part.
63 std::vector<int> mFaceToPart;
64
65 // Mesh attributes.
66 std::vector<std::unique_ptr<BakedAttribute>> mAttrs;
67};
