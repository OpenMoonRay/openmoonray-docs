28void
29RenderProgressEstimation::updatePassInfo(const RenderFrameTimingRecord &timingRec)
30{
31 mLastPassCompleteTime = scene_rdl2::util::getSeconds();
32 mCompletedSamplesPerTile = timingRec.getNumSamplesPerTile();
33 mAveragedSampleCost = timingRec.getAveragedSampleCost();
34
35 unsigned remainedSamplesPerTile = 0;
36 if (mFinalTileSamples > mCompletedSamplesPerTile) {
37 remainedSamplesPerTile = mFinalTileSamples - mCompletedSamplesPerTile - mStartTileSamples;
38 }
39
40 mEstimatedTimeOfCompletion = remainedSamplesPerTile * mAveragedSampleCost + mLastPassCompleteTime; // current best guess
41}

57double
58RenderProgressEstimation::getEstimatedCompletedFractionOfNow() const // return 0.0 ~ 1.0
59{
60 if (mFrameComplete) return 1.0;
61 if (getWholeRenderTime() <= 0.0) return 0.0; // not finished estimation stage yet. (We don't have estimation info yet).
62 return getElapsedRenderTime() / getWholeRenderTime(); // compute fraction based on completion render time estimation
63}

67{

72std::string
73RenderProgressEstimation::getModeBanner()
74{
75 static constexpr char msgRen[] = "Rendering";
76 static constexpr int msgLen = sizeof(msgRen) - 1;
77
78 std::string banner;
79 if (mMode == ModeType::RENDERING) {
80 banner = msgRen;
81 } else {
82 mCallCounter = (mLastMode != mMode)? 0: mCallCounter + 1;
83 std::string work = std::string(msgLen, ' ');
84 switch (mMode) {
85 case ModeType::REVERT_FILM:
86 work += "Reading resume file ...";
87 break;
88 case ModeType::CHECKPOINT_OUTPUT:
89 work += "Checkpoint out " + std::to_string(mCheckpointTileSamples) + " tileSamples ...";
90 break;
91 case ModeType::SIGNAL_INTERRUPTION:
92 work += "Interrupt by SIGNAL ...";
93 break;
94 default : break; // never happened
95 }
96 work += std::string(msgLen, ' ');
97 banner = work.substr(mCallCounter % (work.size() - msgLen), msgLen);
98 }
99 mLastMode = mMode;
100
101 return banner;
102}
