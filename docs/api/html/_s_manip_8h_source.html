15//
16// How to use the SManip class:
17//
18// Define a function that returns std::ios_base& and takes std::ios_base& as
19// the first argument. You can define any other arguments you like after the
20// first. E.g.
21//
22// std::ios_base& myfunc(std::ios_base& outs, int& x, double y) { ... }
23//
24// This is the function that will do the stream manipulation.
25//
26// Define a function that will be used as your stream manipulator. In this
27// function, declare an SManip with template parameters that match your
28// function parameters exactly:
29//
30// SManip<int&, double> mymanip(int& x, double y)
31// {
32// return SManip<int&, double>(&myfunc, x, y);
33// }
34//
35// Call with your manip:
36// int x = 42;
37// std::cout << mymanip(x, 3.14) << '\n';
38//

61// Manipulator taking arguments
62//
63// There are no explicit references in this class, as we make the user
64// fully-specify the types in the template arguments. If the user wants l-value
65// or r-value reference semantics, they have to supply them in the type.
66template <typename... Args>
67struct SManip
68{
69 using Function = std::ios_base& (*)(std::ios_base&, Args...);
70
71 Function mF;
72 std::tuple<Args...> mArgs;
73
74 SManip(Function ff, Args... ii) :
75 mF(ff),
76 mArgs(std::forward<Args>(ii)...)
77 {
78 }
79};

83inline std::ostream& operator<<(std::ostream& os, SManip<Args...>&& m)
