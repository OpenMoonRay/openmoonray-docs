93void PrimitiveAttributeTable::copy(PrimitiveAttributeTable& result ) const
94{
95 for (const auto& kv : mMap) {
96 AttributeKey key = kv.first;
97 switch (key.getType()) {
98 case scene_rdl2::rdl2::TYPE_BOOL:
99 result.mMap.emplace(key, copyAttribute(
100 TypedAttributeKey<bool>(key)));
101 break;
102 case scene_rdl2::rdl2::TYPE_INT:
103 result.mMap.emplace(key, copyAttribute(
104 TypedAttributeKey<int>(key)));
105 break;
106 case scene_rdl2::rdl2::TYPE_LONG:
107 result.mMap.emplace(key, copyAttribute(
108 TypedAttributeKey<long>(key)));
109 break;
110 case scene_rdl2::rdl2::TYPE_FLOAT:
111 result.mMap.emplace(key, copyAttribute(
112 TypedAttributeKey<float>(key)));
113 break;
114 case scene_rdl2::rdl2::TYPE_STRING:
115 result.mMap.emplace(key, copyAttribute(
116 TypedAttributeKey<std::string>(key)));
117 break;
118 case scene_rdl2::rdl2::TYPE_RGB:
119 result.mMap.emplace(key, copyAttribute(
120 TypedAttributeKey<scene_rdl2::math::Color>(key)));
121 break;
122 case scene_rdl2::rdl2::TYPE_RGBA:
123 result.mMap.emplace(key, copyAttribute(
124 TypedAttributeKey<scene_rdl2::math::Color4>(key)));
125 break;
126 case scene_rdl2::rdl2::TYPE_VEC2F:
127 result.mMap.emplace(key, copyAttribute(
128 TypedAttributeKey<scene_rdl2::math::Vec2f>(key)));
129 break;
130 case scene_rdl2::rdl2::TYPE_VEC3F:
131 result.mMap.emplace(key, copyAttribute(
132 TypedAttributeKey<scene_rdl2::math::Vec3f>(key)));
133 break;
134 case scene_rdl2::rdl2::TYPE_MAT4F:
135 result.mMap.emplace(key, copyAttribute(
136 TypedAttributeKey<scene_rdl2::math::Mat4f>(key)));
137 break;
138 default:
139 MNRY_ASSERT(false, (std::string("unsupported attribute type ") +
140 std::string(attributeTypeName(key.getType())) +
141 std::string(" for atttribute ") +
142 std::string(key.getName())).c_str());
143 break;
144 }
145 }
146}
