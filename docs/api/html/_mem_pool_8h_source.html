8// This is a multi-threaded allocator optimized specifically dealing with latent
9// state in a ray tracing context.
10//
11// Here are the requirements:
12//
13// - Minimal overhead.
14// - Allocations are all equally sized. (We'll be storing ray state in there.)
15// - We can't make any assumptions about the length of time an allocation may
16// need to live for. They'll come and go at varying rates.
17// - Allocations may be made from multiple threads concurrently without blocking.
18// - We can't make any assumptions about the relative frequencies of allocations
19// amongst various threads. Some threads may make orders of magnitude more
20// allocations than others. Allocations patterns may vary within the same run.
21// - Allocations may be freed on different threads than they were allocated from.
22// - The max number of allocations is known upfront (i.e. how may rays can we keep
23// alive in the system at any one time.)
24//
25//
26// Structure:
27//
28// Let's call the memory returned from an allocation an entry. A Block is a
29// container of multiple entries. Entries can be allocated and freed using the
30// Block interface. A block itself isn't thread safe apart from one of its calls:
31// Block::addToPendingFreeList. This can be called currently to return memory to
32// a specific block. Memory freed via this called is added to a separate list
33// which then gets properly returned in a call named processPendingFreeList.
34// This call itself isn't thread safe but gets called in a thread local fashion.
35//
36// A MemBlockManager owns all the MemBlocks in the system. It can allocate and free blocks
37// in a fully thread-safe manner. It exists to serve the MemPool class which is
38// the class which the application will typically deal with. MemPools are
39// allocated in thread local storage and therefore don't need to do any locking.
40// MemPools request blocks via the MemBlockManager when in need of memory, and give
41// back MemBlocks which they no longer need to the MemBlockManager so that other threads
42// can reuse them.
43//
44// TODO:
45// - Update SList implementation to be lockless.
46//

58//
59// A node which can be inserted into a doubly linked list. Intrusive.
60//
61struct LinkedListNode
62{
63 LinkedListNode()
64 {
65 mPrev = mNext = this;
66 }
67
68 // Careful - if this node is currently in another list, then that list will be corrupted.
69 void reset()
70 {
71 mPrev = mNext = this;
72 }
73
74 // Are there any other node in the list which this node belongs to?
75 bool isAlone() const
76 {
77 return mNext == this;
78 }
79
80 void removeSelf()
81 {
82 MNRY_ASSERT(mPrev && mNext);
83 mNext->mPrev = mPrev;
84 mPrev->mNext = mNext;
85 mPrev = mNext = this;
86 }
87
88 // Inserts self before argument node.
89 void insertSelfBefore(LinkedListNode *node)
90 {
91 MNRY_ASSERT(node && node->mPrev && node->mNext);
92 node->mPrev->mNext = this;
93 mPrev = node->mPrev;
94 node->mPrev = this;
95 mNext = node;
96 }
97
98 // Appends self after argument node.
99 void appendSelfAfter(LinkedListNode *node)
100 {
101 MNRY_ASSERT(node && node->mPrev && node->mNext);
102 node->mNext->mPrev = this;
103 mNext = node->mNext;
104 node->mNext = this;
105 mPrev = node;
106 }
107
108 // Data.
109 LinkedListNode * mNext;
110 LinkedListNode * mPrev;
111};

115template <typename INTR_TYPE, typename LEAF_TYPE>
116class CACHE_ALIGN MemBlock : public LinkedListNode
117{
118public:
119 // Not thread safe.
120 void init(void *entryMemory, unsigned entryStride)
121 {
122 mEntryMemory = (uint8_t *)entryMemory;
123 mEntryStride = uint32_t(entryStride);
124
125 fullReset();
126 }

128 // Not thread safe.
129 void fullReset()
130 {
131 mNext = mPrev = this;
132 mNumFreeEntries = NUM_ENTRIES;
133
134 mInternalFull = 0;
135 mInternalEmpty = INTR_TYPE(-1);
136 memset(mUsedEntries, 0, sizeof(mUsedEntries));
137
138 mInternalFree = 0;
139 memset(mFreeEntries, 0, sizeof(mFreeEntries));
140
141#ifdef DEBUG
142 memset(mEntryMemory, 0xbc, NUM_ENTRIES * mEntryStride);
143#endif
144 }
145
146 // Not thread safe.
147 void fastReset()
148 {
149 mNext = mPrev = this;
150 mNumFreeEntries = NUM_ENTRIES;
151 mInternalFull = 0;
152 mInternalEmpty = INTR_TYPE(-1);
153 mInternalFree = 0;
154
155#ifdef DEBUG
156 for (unsigned i = 0; i < NUM_LEAF_NODES; ++i) {
157 MNRY_ASSERT(mUsedEntries[i] == 0);
158 }
159
160 for (unsigned i = 0; i < NUM_LEAF_NODES; ++i) {
161 MNRY_ASSERT(mFreeEntries[i] == 0);
162 }
163
164 memset(mEntryMemory, 0xbd, NUM_ENTRIES * mEntryStride);
165#endif
166 }

168 // Returns the number of entries which it was able to allocate for the cases
169 // where the full number of items couldn't be allocated. The entries parameter
170 // must be large enough to store numEntries pointers.
171 unsigned allocList(unsigned numEntries, void **entries)
172 {
173 MNRY_ASSERT(isValid());
174 MNRY_ASSERT(numEntries);
175
176 //
177 // Fast path when only a single entry is desired or available.
178 //
179
180 if (numEntries == 1 || mNumFreeEntries < 2) {
181
182 if (mNumFreeEntries == 0) {
183 return 0;
184 }
185
186 --mNumFreeEntries;
187
188 // Find first leaf node with free entries.
189 const unsigned leafNodeIdx = util::countLeadingZerosUnsafe(INTR_TYPE(~mInternalFull));
190
191 // Find first free entry within leaf node.
192 const unsigned entryIdx = util::countLeadingZerosUnsafe(LEAF_TYPE(~mUsedEntries[leafNodeIdx]));
193
194 // Compute combined entry index relative to 0'th entry. This is the index
195 // of the entry we'll return.
196 const unsigned masterIdx = (leafNodeIdx << ENTRIES_PER_LEAF_NODE_SHIFT) + entryIdx;
197 MNRY_ASSERT(isIndexAllocated(masterIdx) == false);
198
199 // Update bitfields to record new allocation.
200 mUsedEntries[leafNodeIdx] |= sLeafMSB >> entryIdx;
201
202 // Check if this leaf has had all its bits allocated.
203 const INTR_TYPE internalBit = sInternalMSB >> leafNodeIdx;
204 if (mUsedEntries[leafNodeIdx] == LEAF_TYPE(-1)) {
205 mInternalFull |= internalBit;
206 }
207
208 // Clear leaf node bit from mInternalEmpty, since the corresponding leaf can't
209 // be completely empty anymore.
210 mInternalEmpty &= ~internalBit;
211
212 entries[0] = mEntryMemory + (mEntryStride * masterIdx);
213
214 MNRY_ASSERT(isValid());
215
216 return 1;
217 }
218
219 //
220 // Allocation of multiple elements is required.
221 //
222
223 MNRY_ASSERT(numEntries > 1);
224
225 numEntries = std::min(numEntries, mNumFreeEntries);
226 unsigned entriesToAllocate = numEntries;
227 mNumFreeEntries -= numEntries;
228
229 // Pass 1 : Search for empty leaves and fill them up completely.
230 while (mInternalEmpty && entriesToAllocate >= ENTRIES_PER_LEAF_NODE) {
231
232 const unsigned leafNodeIdx = util::countLeadingZerosUnsafe(INTR_TYPE(mInternalEmpty));
233 MNRY_ASSERT(mUsedEntries[leafNodeIdx] == 0);
234
235 const unsigned masterIdx = leafNodeIdx << ENTRIES_PER_LEAF_NODE_SHIFT;
236 for (unsigned i = 0; i < ENTRIES_PER_LEAF_NODE; ++i) {
237 entries[i] = mEntryMemory + (mEntryStride * (masterIdx + i));
238 }
239
240 const INTR_TYPE internalBit = sInternalMSB >> leafNodeIdx;
241 mInternalEmpty &= ~internalBit;
242 mInternalFull |= internalBit;
243
244 mUsedEntries[leafNodeIdx] = LEAF_TYPE(-1);
245
246 entries = &entries[ENTRIES_PER_LEAF_NODE]; // entries += ENTRIES_PER_LEAF_NODE;
247 entriesToAllocate -= ENTRIES_PER_LEAF_NODE;
248 }
249
250 // Pass 2 : Search for partially used leaves and fill them up as needed.
251 while (entriesToAllocate) {
252
253 // Find first leaf node with free entries.
254 const unsigned leafNodeIdx = util::countLeadingZerosUnsafe(INTR_TYPE(~mInternalFull));
255
256 LEAF_TYPE freeEntries = ~mUsedEntries[leafNodeIdx];
257
258 MNRY_ASSERT(freeEntries);
259
260 do {
261 const unsigned entryIdx = util::countLeadingZerosUnsafe(freeEntries);
262 const unsigned masterIdx = (leafNodeIdx << ENTRIES_PER_LEAF_NODE_SHIFT) + entryIdx;
263
264 MNRY_ASSERT(isIndexAllocated(masterIdx) == false);
265
266 *entries = mEntryMemory + (mEntryStride * masterIdx);
267 entries = &entries[1]; // ++entries;
268 --entriesToAllocate;
269
270 freeEntries &= ~(sLeafMSB >> entryIdx);
271
272 } while (freeEntries && entriesToAllocate);
273
274 // Update bitfields to record new allocations.
275 mUsedEntries[leafNodeIdx] = ~freeEntries;
276
277 const INTR_TYPE internalBit = sInternalMSB >> leafNodeIdx;
278 if (freeEntries == 0) {
279 mInternalFull |= internalBit;
280 }
281
282 mInternalEmpty &= ~internalBit;
283 }
284
285 MNRY_ASSERT(isValid());
286
287 return numEntries;
288 }

290 // Thread safe.
291 void addToPendingFreeList(unsigned numEntries, void **entries)
292 {
293 MNRY_ASSERT(numEntries);
294
295 Mutex::scoped_lock lock(mPendingFreeMutex);
296
297 for (unsigned i = 0; i < numEntries; ++i) {
298
299 void *entry = entries[i];
300
301 MNRY_ASSERT(isEntryValid(entry));
302
303 unsigned masterIdx = unsigned(((uint8_t *)entry - mEntryMemory) / mEntryStride);
304 MNRY_ASSERT(masterIdx < NUM_ENTRIES);
305
306 const unsigned leafNodeIdx = masterIdx >> ENTRIES_PER_LEAF_NODE_SHIFT;
307 const LEAF_TYPE leafNodeBit = sLeafMSB >> (masterIdx & (ENTRIES_PER_LEAF_NODE - 1));
308
309 mInternalFree |= (sInternalMSB >> leafNodeIdx);
310 mFreeEntries[leafNodeIdx] |= leafNodeBit;
311 }
312 }
313
314 //
315 // Not thread safe. It must be called on the thread which owns this block.
316 //
317 // This takes all the entries in the pending free list and makes them
318 // available for allocation again. It doesn't stall however and other
319 // threads are free to continue adding further entries to the pending free
320 // list whilst this function is executing. Returns true if any entries were
321 // freed.
322 //
323 // Processing the free list should be incremental, if there's no work to do,
324 // it shouldn't take any time. This way we can just call it freely without
325 // having to tune too much.
326 //
327 // Returns the number of entries which we're freed during this call.
328 //
329 unsigned processPendingFreeList()
330 {
331 MNRY_ASSERT(isValid());
332
333 // Speculative early-out.
334 if (mInternalFree == 0 || mNumFreeEntries == NUM_ENTRIES) {
335 return 0;
336 }
337
338 //
339 // We have a few options here:
340 //
341 // 1: Lock and process.
342 // 2: Lock and copy, then unlock and process.
343 // 3: Double buffer pending free list and atomically switch pointers.
344 //
345 // Approach 1 is the most straight forward to implement so let's try that
346 // one first.
347 //
348
349 unsigned numFreed = 0;
350
351 mPendingFreeMutex.lock();
352
353 while (mInternalFree) {
354
355 const unsigned leafNodeIdx = util::countLeadingZerosUnsafe(mInternalFree);
356
357 LEAF_TYPE freeNodes = mFreeEntries[leafNodeIdx];
358
359 mFreeEntries[leafNodeIdx] = 0;
360 mUsedEntries[leafNodeIdx] &= ~freeNodes;
361
362 const INTR_TYPE internalBit = sInternalMSB >> leafNodeIdx;
363
364 mInternalFull &= ~internalBit;
365 mInternalFree &= ~internalBit;
366
367 if (mUsedEntries[leafNodeIdx] == 0) {
368 mInternalEmpty |= internalBit;
369 }
370
371 numFreed += util::countOnBits(freeNodes);
372 }
373
374 mNumFreeEntries += numFreed;
375
376 mPendingFreeMutex.unlock();
377
378 MNRY_ASSERT(isValid());
379
380 return numFreed;
381 }
382
383 // Not thread safe.
384 bool isValid() const
385 {
386 MNRY_STATIC_ASSERT((NUM_ENTRIES & 31) == 0);
387
388 MNRY_ASSERT(mNext && mPrev);
389 MNRY_ASSERT(mEntryMemory);
390 MNRY_ASSERT(mEntryStride);
391
392 //
393 // Check number of free entries is correct.
394 //
395 unsigned numU32s = NUM_ENTRIES / 32;
396
397 unsigned numUsed = 0;
398 const uint32_t *entries = (const uint32_t *)mUsedEntries;
399
400 for (unsigned i = 0; i < numU32s; ++i) {
401 numUsed += util::countOnBits(entries[i]);
402 }
403
404 MNRY_ASSERT(getNumFreeEntries() == (NUM_ENTRIES - numUsed));
405
406 //
407 // Check that mInternalFull and mInternalEmpty are consistent with leaf nodes.
408 //
409 MNRY_ASSERT((mInternalFull & mInternalEmpty) == 0);
410
411 for (unsigned i = 0; i < NUM_LEAF_NODES; ++i) {
412
413 const INTR_TYPE internalBit = sInternalMSB >> i;
414 if ((internalBit & mInternalFull) != 0) {
415 MNRY_ASSERT(mUsedEntries[i] == LEAF_TYPE(-1));
416 } else {
417 MNRY_ASSERT(mUsedEntries[i] != LEAF_TYPE(-1));
418 }
419
420 if ((internalBit & mInternalEmpty) != 0) {
421 MNRY_ASSERT(mUsedEntries[i] == 0);
422 } else {
423 MNRY_ASSERT(mUsedEntries[i] != 0);
424 }
425 }
426
427 return true;
428 }

434 return mInternalFull == INTR_TYPE(-1);

468private:
469 // Returns true if this entry is in the address space of this block.
470 bool isEntryValid(const void *entry) const
471 {
472 MNRY_ASSERT(entry >= mEntryMemory && entry < mEntryMemory + NUM_ENTRIES * mEntryStride);
473 MNRY_ASSERT((uint64_t((const uint8_t *)entry - mEntryMemory) % mEntryStride) == 0);
474 return true;
475 }
476
477 bool isIndexAllocated(unsigned masterIdx) const
478 {
479 const unsigned leafNodeIdx = masterIdx >> ENTRIES_PER_LEAF_NODE_SHIFT;
480 return (mUsedEntries[leafNodeIdx] & (sLeafMSB >> (masterIdx & (ENTRIES_PER_LEAF_NODE - 1)))) != 0;
481 }
482
483 enum
484 {
485 NUM_LEAF_NODES = sizeof(INTR_TYPE) * 8,
486 ENTRIES_PER_LEAF_NODE = sizeof(LEAF_TYPE) * 8,
487 ENTRIES_PER_LEAF_NODE_SHIFT = math::compile_time::log2i(ENTRIES_PER_LEAF_NODE),
488 NUM_ENTRIES = NUM_LEAF_NODES * ENTRIES_PER_LEAF_NODE,
489 };
490
491 // All data offsets are relative to this address.
492 uint8_t * mEntryMemory;
493
494 // Each entry we can hand out is assumed to be a constant size.
495 uint32_t mEntryStride;
496
497 // The number of free entries we currently have available to hand out, not
498 // counting any entries in the pending free list.
499 unsigned mNumFreeEntries;
500
501 //
502 // These members form a 2-deep bitfield hierarchy. The top level bitfield is
503 // called the "internal" bitfield and bitfields at the second level are
504 // called "leaf nodes". If a bit is set in the mInternalFull bitfield, it
505 // signifies that the corresponding leaf node bitfield is completely full up,
506 // i.e. it doesn't have anymore free space. Correspondingly if a bit is set
507 // in the mInternalEmpty bitfield, it signifies that the corresponding leaf
508 // bitfield is completely empty. So we get something like this:
509 //
510 // Full Empty Leaf node meaning
511 //
512 // 0 0 Leaf node isn't full or empty, i.e, some bits are allocated.
513 // 0 1 Leaf node is completely empty.
514 // 1 0 Leaf node is completely full.
515 // 1 1 Invalid state.
516 //
517 // Each bit in each entry in the mUsedEntries array corresponds to a single
518 // entry in our memory pool. An on bit means the entry is currently allocated.
519 //
520 INTR_TYPE mInternalFull;
521 INTR_TYPE mInternalEmpty;
522 LEAF_TYPE mUsedEntries[NUM_LEAF_NODES];
523
524 //
525 // Pending free-list. This is a record of elements belonging to this block
526 // which have been freed from various threads but are awaiting insertion into
527 // the main allocation hierarchy, and thus can't be handed out just yet.
528 // Any access to these members should be protected with mPendingFreeMutex.
529 //
530 typedef tbb::spin_mutex Mutex;
531
532 Mutex mPendingFreeMutex;
533 INTR_TYPE mInternalFree;
534 LEAF_TYPE mFreeEntries[NUM_LEAF_NODES];
535
536 // Explicitly store these constants separately since they man be bigger than
537 // what can be held in an enum.
538 static const INTR_TYPE sInternalMSB = INTR_TYPE(1ULL << (NUM_LEAF_NODES - 1));
539 static const LEAF_TYPE sLeafMSB = LEAF_TYPE(1ULL << (ENTRIES_PER_LEAF_NODE - 1));
540};

544//
545// A MemBlockManager owns all the Blocks in the system. It can allocate and free blocks in a
546// fully thread-safe manner. It exists to serve the MemPool class which is the class
547// which the application will typically deal with.
548//
549// Internally, given the address of a chunk of memory to be freed, it knows how to
550// map it back to it's owning block.
551//
552
553template<typename BLOCK_TYPE>
554class CACHE_ALIGN MemBlockManager
555{
556public:
557 typedef BLOCK_TYPE BlockType;
558
559 MemBlockManager() :
560 mNumBlocks(0),
561 mBlockMemory(nullptr),
562 mEntryMemory(nullptr),
563 mEntryStride(0),
564 mEntryToBlockDivider(0)
565 {
566 }
567
568 // Use queryEntryMemoryRequired to compute size needed for entryMemory.
569 // Not thread safe.
570 void init(unsigned numBlocks,
571 BlockType *blockMemory,
572 void *entryMemory,
573 unsigned entryStride)
574 {
575 // We use the first 8 bytes to store the next pointer for entries in ConcurrentSList.
576 MNRY_STATIC_ASSERT(sizeof(BlockType) >= 8);
577
578 mNumBlocks = MNRY_VERIFY(numBlocks);
579 mBlockMemory = MNRY_VERIFY(blockMemory);
580 mEntryMemory = MNRY_VERIFY((uint8_t *)entryMemory);
581 mEntryStride = MNRY_VERIFY(entryStride);
582 mEntryToBlockDivider = MNRY_VERIFY(mEntryStride * NUM_ENTRIES_PER_BLOCK);
583
584 fullReset();
585 }
586
587 // Forceably reclaims and initializes all blocks. Only call this when you know that
588 // none are still in use. Not thread safe.
589 void fullReset()
590 {
591 mFreeBlocks.init();
592
593 // Insert all entries into free list in reverse order so that they get handed
594 // out contiguously.
595 for (unsigned i = 1; i <= mNumBlocks; ++i) {
596 size_t blockIdx = size_t(mNumBlocks - i);
597 BlockType *block = mBlockMemory + blockIdx;
598 size_t entryOffset = blockIdx * NUM_ENTRIES_PER_BLOCK * mEntryStride;
599 uint8_t *entryMemory = mEntryMemory + entryOffset;
600 block->init(entryMemory, mEntryStride);
601 mFreeBlocks.push((util::SList::Entry *)block);
602 }
603
604 MNRY_ASSERT(mFreeBlocks.size() == mNumBlocks);
605 }
606
607 // Forceably reclaims all blocks. Faster than calling FullReset since it assumes
608 // all contained blocks are empty. If this is not the case, call fullReset().
609 // Not thread safe.
610 void fastReset()
611 {
612 mFreeBlocks.init();
613
614 // Insert all entries into free list in reverse order so that they get handed
615 // out contiguously.
616 for (unsigned i = 1; i <= mNumBlocks; ++i) {
617 unsigned blockIdx = mNumBlocks - i;
618 BlockType *block = mBlockMemory + blockIdx;
619 block->fastReset();
620 mFreeBlocks.push(block);
621 }
622
623 MNRY_ASSERT(mFreeBlocks.size() == mNumBlocks);
624 }
625
626 unsigned getMemoryUsage() const
627 {
628 return (mNumBlocks * sizeof(BLOCK_TYPE)) +
629 queryEntryMemoryRequired(mNumBlocks, mEntryStride) +
630 sizeof(*this);
631 }
632
633 // Thread safe.
634 BlockType *allocateBlock()
635 {
636 BlockType *block = (BlockType *)mFreeBlocks.pop();
637 if (block) {
638 // Reset pointers so it can be inserted into a the doubly linked
639 // list again. (Inserting it into the mFreeBlocks list would have
640 // corrupted these!)
641 block->reset();
642
643 MNRY_ASSERT(isValidBlockAddress(block));
644 MNRY_ASSERT(block->isValid());
645 MNRY_ASSERT(block->isEmpty());
646 }
647 return block;
648 }
649
650 // Thread safe.
651 void freeBlock(BlockType *block)
652 {
653 MNRY_ASSERT(isValidBlockAddress(block));
654 MNRY_ASSERT(block->isValid());
655
656 block->removeSelf();
657
658 if (block->isEmpty()) {
659 block->fastReset();
660 } else {
661 block->fullReset();
662 }
663
664 mFreeBlocks.push((util::SList::Entry *)block);
665 }
666
667 // This call does the extra work of routing the deallocation to the block
668 // it was originally allocated from. Thread safe.
669 void freeList(unsigned numEntries, void **entries)
670 {
671 if (numEntries == 1) {
672 freeSingleEntry(entries[0]);
673
674 } else if (numEntries <= 64) {
675
676 // Naive simple loop over entries with one lock per entry.
677 for (unsigned i = 0; i < numEntries; ++i) {
678 freeSingleEntry(entries[i]);
679 }
680
681 } else {
682
683 // Sort entries by address so that all entries from a particular block
684 // will be contiguous in memory. This allows us to reduce the number of
685 // locks to one per block as opposed to one per entry.
686
687 intptr_t *addrs = (intptr_t *)entries;
688 intptr_t *end = addrs + numEntries;
689
690 std::sort(addrs, end);
691
692 while (addrs != end) {
693
694 unsigned blockIdx = getOwningBlockIndex((void *)(*addrs));
695 intptr_t *currAddr = addrs + 1;
696 while (currAddr != end && blockIdx == getOwningBlockIndex((void *)(*currAddr))) {
697 ++currAddr;
698 }
699
700 unsigned entriesInBlock = unsigned(currAddr - addrs);
701 MNRY_ASSERT(entriesInBlock);
702
703#ifdef DEBUG
704 for (unsigned i = 0; i < entriesInBlock; ++i) {
705 memset((void *)(addrs[i]), 0xbe, mEntryStride);
706 }
707#endif
708
709 BlockType *block = mBlockMemory + blockIdx;
710 MNRY_ASSERT(isValidBlockAddress(block));
711
712 block->addToPendingFreeList(entriesInBlock, (void **)addrs);
713
714 addrs += entriesInBlock;
715 }
716
717 MNRY_ASSERT(intptr_t(addrs) == intptr_t(&entries[numEntries]));
718 }
719 }
720
721 // Thread safe.
722 unsigned getOwningBlockIndex(const void *entry) const
723 {
724 // Map the entry to its owning block.
725 unsigned blockIdx = unsigned(intptr_t(entry) - intptr_t(mEntryMemory)) / mEntryToBlockDivider;
726 MNRY_ASSERT(blockIdx < mNumBlocks);
727 return blockIdx;
728 }
729
730 // Thread safe.
731 bool isValidBlockAddress(const BlockType *block) const
732 {
733 MNRY_ASSERT(block);
734 MNRY_ASSERT(block >= (BlockType *)mBlockMemory && block < (mBlockMemory + mNumBlocks));
735 MNRY_ASSERT(((intptr_t)block - (intptr_t)mBlockMemory) % sizeof(BlockType) == 0);
736 return true;
737 }

739 static constexpr size_t queryEntryMemoryRequired(size_t numBlocks, size_t entryStride)
740 {
741 return numBlocks * NUM_ENTRIES_PER_BLOCK * entryStride;
742 }

744protected:
745 void freeSingleEntry(void *entry)
746 {
747 MNRY_ASSERT(entry >= mEntryMemory);
748
749#ifdef DEBUG
750 memset(entry, 0xbe, mEntryStride);
751#endif
752
753 // Map the entry to its owning block.
754 unsigned blockIdx = getOwningBlockIndex(entry);
755
756 BlockType *block = mBlockMemory + blockIdx;
757 MNRY_ASSERT(isValidBlockAddress(block));
758
759 block->addToPendingFreeList(1, &entry);
760 }

779// Per thread data. All allocation related calls on this interface should only
780// be called by this thread which owns this object. Since memory which has been
781// allocated on this thread may be passed around to other threads, it supports
782// deallocating memory allocations made from other threads also.

784template<typename BLOCK_TYPE>
785class LocalUntypedMemPool
786{
787public:

810 void cleanUp()
811 {
812 if (mBlockManager && mActiveBlock) {
813 BlockType *block = mActiveBlock;
814 do {
815 BlockType *next = (BlockType *)block->mNext;
816 mBlockManager->freeBlock(block);
817 block = next;

819 }

822 mActiveBlock = nullptr;

824 mNumAllocated = 0;

827 // Full reset. Deallocate all blocks explicitly.
828 void fullReset()
829 {
830 // Give back all blocks we have taken so far.
831 if (mBlockManager && mActiveBlock) {
832 BlockType *block = mActiveBlock;
833 do {
834 BlockType *next = (BlockType *)block->mNext;
835 mBlockManager->freeBlock(block);
836 block = next;
837 } while (block != mActiveBlock);
838 }

840 fastReset();
841 }

843 // Don't give back any blocks, assume that they have been stolen from us.
844 void fastReset()
845 {
846 resetStats();
847
848 // Give back all blocks we have taken so far.
849 if (mBlockManager) {
850
851 // Always allocate a single initial block.
852 INC_COUNTER(BLOCKS_ALLOCATED);
853 mActiveBlock = mBlockManager->allocateBlock();
854
855 // If this fails, we need to allocate more blocks at startup.
856 MNRY_ASSERT(mActiveBlock && mActiveBlock->isEmpty());
857
858 mNumReserved = BlockType::getNumEntries();
859 mNumAllocated = 0;
860
861 MNRY_ASSERT(isBlockListValid());
862 }
863 }
864
865 // This function encapsulates the heuristics for making allocations, it is
866 // key for good perf. Returns true if the request could be satisfied or
867 // false otherwise. The entries parameter must be large enough to store
868 // numEntries pointers.
869 bool untypedAllocList(unsigned numEntries, void **entries)
870 {
871 MNRY_ASSERT(isValid());
872
873 MNRY_ASSERT(mActiveBlock->isValid());
874 MNRY_ASSERT(mBlockManager->isValidBlockAddress(mActiveBlock));
875
876 unsigned remainingAllocs = numEntries;
877 void **baseEntry = entries;
878
879 //
880 // Case A (best) : Try and trivially allocate from the current block.
881 //
882
883 unsigned numAllocated = mActiveBlock->allocList(remainingAllocs, entries);
884 ADD_TO_COUNTER(CASE_A_ALLOCS, numAllocated);
885
886 remainingAllocs -= numAllocated;
887 mNumAllocated += numAllocated;
888
889 if (remainingAllocs == 0) {
890 return true;
891 }
892
893 entries = &entries[numAllocated]; // entries += numAllocated;
894
895 // Don't cycle beyond (num blocks - 1) for any given allocation. This ensures that all
896 // blocks will get their pending free lists flushed equally.
897 BlockType *endBlock = (BlockType *)mActiveBlock->mPrev;
898
899 cycleToNextBlock();
900
901 //
902 // Case B (worse) : Cycle through remaining blocks and try and allocate from them.
903 //
904
905 while (mActiveBlock != endBlock) {
906
907 // Process pending frees from this new block.
908 INC_COUNTER(PROCESS_PENDING_FREE_LIST);
909 mNumAllocated -= mActiveBlock->processPendingFreeList();
910 MNRY_ASSERT(mNumAllocated <= mNumReserved); // Assert we haven't underflowed beyond 0.
911
912 // Give back completely free blocks to the parent block manager *if* it's
913 // not the only block remaining in the list.
914 if (mActiveBlock->isEmpty() && !mActiveBlock->isAlone()) {
915 returnEmptyBlock();
916 continue;
917 }
918
919 numAllocated = mActiveBlock->allocList(remainingAllocs, entries);
920 ADD_TO_COUNTER(CASE_B_ALLOCS, numAllocated);
921
922 remainingAllocs -= numAllocated;
923 mNumAllocated += numAllocated;
924
925 if (remainingAllocs == 0) {
926 return true;
927 }
928
929 entries = &entries[numAllocated]; // entries += numAllocated;
930
931 cycleToNextBlock();
932 }
933
934 //
935 // Case C (worst) : Satisfy all remaining allocations by grabbing brand new empty blocks.
936 //
937
938 while (remainingAllocs) {
939
940 BlockType *freshBlock = mBlockManager->allocateBlock();
941
942 if (__builtin_expect((freshBlock != nullptr), 1)) {
943
944 INC_COUNTER(BLOCKS_ALLOCATED);
945
946 // Insert into head of block list so it becomes the new active block.
947 freshBlock->appendSelfAfter(mActiveBlock);
948 mActiveBlock = freshBlock;
949
950 numAllocated = freshBlock->allocList(remainingAllocs, entries);
951 ADD_TO_COUNTER(CASE_C_ALLOCS, numAllocated);
952
953 remainingAllocs -= numAllocated;
954 mNumReserved += BlockType::getNumEntries();
955 mNumAllocated += numAllocated;
956 entries = &entries[numAllocated]; // entries += numAllocated;
957
958 continue;
959 }
960
961 // We're completely out of memory!
962 INC_COUNTER(FAILED_BLOCK_ALLOCS);
963 ADD_TO_COUNTER(FAILED_ENTRY_ALLOCS, remainingAllocs);
964
965 // Free all the entries we've just allocated to avoid leaking memory.
966 MNRY_ASSERT(remainingAllocs && numEntries >= remainingAllocs);
967 untypedFreeList(numEntries - remainingAllocs, baseEntry);
968
969 return false;
970 }
971
972 MNRY_ASSERT(isValid());
973
974 return true;
975 }

977 // All allocations are made through the LocalUntypedMemPool interface for the current thread.
978 // Deallocations can be made from any thread at any time through this call.
979 void untypedFreeList(unsigned numEntries, void **entries)
980 {
981 ADD_TO_COUNTER(FREE_CALLS, numEntries);
982 mBlockManager->freeList(numEntries, entries);
983 }

985 //
986 // Statistics gathering:
987 //
988 // The total number of successful allocs is (FAST_ALLOCS + PENDING_ALLOCS + NEW_BLOCK_ALLOCS).
989 //
990 enum
991 {
992 // MemBlock requests made to MemBlockManager.
993 BLOCKS_ALLOCATED,
994
995 // The number of allocations made using the fast path. This higher this
996 // count, the better.
997 CASE_A_ALLOCS,
998
999 // The number of allocations made as a result of a call to MemBlock::processPendingFreeList.
1000 // This also equals the number of times MemBlock::processPendingFreeList returned true.
1001 CASE_B_ALLOCS,
1002
1003 // The number of allocations which required us to allocate a brand new block.
1004 CASE_C_ALLOCS,
1005
1006 // All the frees which have been processed. When there are no outstanding memory
1007 // allocations, this number should equal (CASE_A_ALLOCS + CASE_B_ALLOCS + CASE_C_ALLOCS).
1008 FREE_CALLS,
1009
1010 // The number of times we've had to call MemBlock::processPendingFreeList to
1011 // try and free up more memory. Less calls are better.
1012 PROCESS_PENDING_FREE_LIST,
1013
1014 // The number of empty blocks we were able to return to the MemBlockManager so
1015 // that other threads could use the memory.
1016 RETURNED_EMPTY_BLOCKS,
1017
1018 // This happens if we are completely out of local memory after pending frees
1019 // have been processed and there are no available blocks in the block manager.
1020 FAILED_BLOCK_ALLOCS,
1021
1022 // The is the total number of entries which we've failed to allocate.
1023 FAILED_ENTRY_ALLOCS,
1024
1025 NUM_COUNTERS,
1026 };

1176 bool isBlockListValid() const
1177 {
1178 // We assume there is always at least a single block allocated.
1179 MNRY_ASSERT(mActiveBlock);
1180
1181 unsigned numForward = 0;
1182 LinkedListNode *node = mActiveBlock;
1183
1184 do {
1185 ++numForward;
1186 node = node->mNext;
1187 } while (node != mActiveBlock);
1188
1189 unsigned numBackward = 0;
1190 node = mActiveBlock;
1191
1192 do {
1193 ++numBackward;
1194 node = node->mPrev;
1195 } while (node != mActiveBlock);
1196
1197 MNRY_ASSERT(numForward == numBackward);
1198 MNRY_ASSERT(numForward * BlockType::getNumEntries() == mNumReserved);

1200 return true;
1201 }

1204 MemBlockManager<BlockType> *mBlockManager;
1205

1207 // One of a cyclic doubly linked list of nodes.
1208 BlockType * mActiveBlock;

1211 // manager. It's the sum of the available entries and the allocated entries.
1212 unsigned mNumReserved;
1213
1214 // The number of entries we've allocated at this moment in time. It doesn't
1215 // account for entries which are waiting in the pending free list.
1216 unsigned mNumAllocated;
1217
1218 Stats mStats;
