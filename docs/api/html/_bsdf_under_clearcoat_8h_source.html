106 // Fresnel Object that reverses the IORs for rays exiting Clearcoat
107 Fresnel* mExitingFresnel;
108 // Ratio of etaI/etaT
109 float mNeta;
110
111 // When sampling the Under lobe through the Clearcoat interface,
112 // it is very likely to run into situations where the "sampled" vector
113 // from the under lobe results in TIR when exiting the Clearcoat interface and
114 // might have to be discarded. At this point, in lieu of generating another sample
115 // until we find one that exits, or discarding this sample entirely,
116 // we assume that the ray undergoes TIR through the interface, bounces around
117 // multiple times and eventually comes out through the interface in the same
118 // direction. This helps energy conservation and stops the under lobe from
119 // going too dark because of discarded samples.
120
121 // Note that we disable this option when unit testing the UnderClearcoat interface
122 // to ensure the consistency tests pass for the sample() and eval() functions.
123 bool mPassThroughTIRWhenSampling;
