917HalfOpenViewport SceneVariables::getRezedSubViewport() const
918{
919 HalfOpenViewport regionViewport = getRezedRegionWindow();
920
921 HalfOpenViewport screen(0, 0, regionViewport.width(), regionViewport.height());
922
923 math::Vec2i debugPixel;
924 if (getDebugPixel(debugPixel)) {
925 if (screen.contains(debugPixel.x, debugPixel.y)) {
926 return HalfOpenViewport(debugPixel.x, debugPixel.y, debugPixel.x + 1, debugPixel.y + 1);
927 }
928 }
929
930 const std::vector<int>& viewport = get(sSubViewport);
931 if (viewport[0] == std::numeric_limits<int>::lowest()) {
932 return screen;
933 }
934
935 // Clip rezed sub-viewport to eventual screen window.
936 float invRes = 1.f / get(sResKey);
937 int minX = int(float(viewport[0]) * invRes);
938 int minY = int(float(viewport[1]) * invRes);
939 int maxX = int(float(viewport[2]) * invRes);
940 int maxY = int(float(viewport[3]) * invRes);
941
942 minX = math::max(minX, screen.mMinX);
943 minY = math::max(minY, screen.mMinY);
944 maxX = math::min(maxX, screen.mMaxX);
945 maxY = math::min(maxY, screen.mMaxY);
946
947 MNRY_ASSERT(minX >= 0);
948 MNRY_ASSERT(minY >= 0);
949
950 return HalfOpenViewport(minX, minY, maxX, maxY);
951}
