56inline BsdfLobe::Type
57getSurfaceFlags(const BsdfSlicev &slicev, const Bsdfv &bsdfv, const scene_rdl2::math::Vec3f &wi, int lane)
58{
59 if (getIsSpherical(bsdfv, lane)) {
60 return static_cast<BsdfLobe::Type>(slicev.mFlags[lane]);
61 }
62
63 const scene_rdl2::math::Vec3f ng(slicev.mNg.x[lane], slicev.mNg.y[lane], slicev.mNg.z[lane]);
64 const scene_rdl2::math::Vec3f wo(slicev.mWo.x[lane], slicev.mWo.y[lane], slicev.mWo.z[lane]);
65 return BsdfLobe::Type(slicev.mFlags[lane] &
66 ((scene_rdl2::math::dot(ng, wo) * scene_rdl2::math::dot(ng, wi) > 0.0f) ?
67 ~BsdfLobe::TRANSMISSION : ~BsdfLobe::REFLECTION));
68
69}
