93// This function round a floating point number to a certain lowest significant bit from the right
94// Rounding is away from zero
95finline float roundFloat(const float in, const uint8_t lsb)
96{
97 float out = in;
98 unsigned int *outInt = reinterpret_cast<unsigned int*>(&out);
99 if (((*outInt) & 0x7f800000) == 0) return 0; // make all denormalized coding zero
100 if (((*outInt) | 0x807fffff) == 0xffffffff) return out; // Inf and NaN remains the same
101 *outInt += 1<<lsb;
102 *outInt &= ((unsigned int)(-1))<<lsb;
103 return out;
104}

106// This function performs Kahan algorithm to accumulate the data in serial
107// c is the running compensation for lost low-order bits
108finline void kahanSum(float& sum, float& c, const float in)
109{
110 float y = in - c;
111 float t = sum + y;
112 c = (t - sum) - y;
113 sum = t;
114}

116// This class exists to aid fast and safe transposition of AOS to SOA data.
117// - By making this object 32-bit exactly, we can use SIMD intrinsics to do the
118// transposition directly without needing to write another custom code path.
119// - By being able to store 32 flags in a single 32-bit value, it cuts down on
120// storage required compared to bools, and hence saves SOA transposition bandwidth.
121// - Bools are ill defined in ISPC so we don't want to be using ISPC structures
122// with bools, which implies removing them from C++ structures which we want
123// to transpose.
124//
