142void
143TestSets::testStaticGeometry()
144{
145 Geometry* teapot1 = mContext->createSceneObject("FakeTeapot", "/seq/shot/teapot1")->asA<Geometry>();
146 Geometry* teapot2 = mContext->createSceneObject("FakeTeapot", "/seq/shot/teapot2")->asA<Geometry>();
147 Geometry* teapot3 = mContext->createSceneObject("FakeTeapot", "/seq/shot/teapot3")->asA<Geometry>();
148
149 GeometrySet* asset = mContext->createSceneObject("GeometrySet", "/seq/shot/asset")->asA<GeometrySet>();
150
151 {
152 SceneObject::UpdateGuard guard1(teapot1);
153 teapot1->set(Geometry::sStaticKey, true);
154
155 SceneObject::UpdateGuard guard2(teapot2);
156 teapot2->set(Geometry::sStaticKey, true);
157
158 SceneObject::UpdateGuard guard3(teapot3);
159 teapot3->set(Geometry::sStaticKey, true);
160
161 SceneObject::UpdateGuard guard4(asset);
162 asset->add(teapot1);
163 asset->add(teapot2);
164 asset->add(teapot3);
165 }
166
167 CPPUNIT_ASSERT_EQUAL(true, asset->isStatic());
168
169 {
170 SceneObject::UpdateGuard guard5(teapot2);
171 teapot2->set(Geometry::sStaticKey, false);
172 }
173
174 CPPUNIT_ASSERT_EQUAL(false, asset->isStatic());
175}
