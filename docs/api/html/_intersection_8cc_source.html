33void
34Intersection::transferAndComputeDerivatives(mcrt_common::ThreadLocalState *tls,
35 mcrt_common::RayDifferential *ray,
36 float textureDiffScale)
37{
38 EXCL_ACCUMULATOR_PROFILE(tls, EXCL_ACCUM_COMPUTE_RAY_DERIVATIVES);
39
40 MNRY_ASSERT(mFlags.get(GeomInitialized));
41
42 // Transfer ray differentials to hit point
43 ray->transfer(getP(), getNg());
44
45 // Initialize texture derivatives to 0 so we sample at the finest mip level.
46 mdSdx = mdTdx = mdSdy = mdTdy = 0.0f;
47
48 if (!ray->hasDifferentials()) {
49 // If we fail to compute the aux ray positions, it's probably
50 // because one of the aux rays is at a super glancing angle wrt the
51 // hit plane. If that's the case, we should be sampling at our coarsest
52 // mip level.
53 mdSdx = 1024.0f;
54 mdTdy = 1024.0f;
55
56 } else if (hasGeometryDerivatives()) {
57 // The intersection has valid data to compute texture derivatives.
58
59 // transfer is assumed to be already called on the ray
60 const scene_rdl2::math::Vec3f dpdx = ray->getdPdx();
61 const scene_rdl2::math::Vec3f dpdy = ray->getdPdy();
62
63 // Take a short cut if dpdu and dpdv are very close to forming
64 // an orthogonal basis. When that's the case, simple projection will
65 // suffice to find our derivatives.
66 if (isZero(dot(mdPds, mdPdt), sIntersectionEpsilon)) {
67
68 float dotDpdu = dot(mdPds, mdPds);
69 float dotDpdv = dot(mdPdt, mdPdt);
70 if (dotDpdu > sIntersectionEpsilon && dotDpdv > sIntersectionEpsilon) {
71 float invDotDpdu = 1.0f / dot(mdPds, mdPds);
72 float invDotDpdv = 1.0f / dot(mdPdt, mdPdt);
73 mdSdx = dot(dpdx, mdPds) * invDotDpdu;
74 mdTdx = dot(dpdx, mdPdt) * invDotDpdv;
75 mdSdy = dot(dpdy, mdPds) * invDotDpdu;
76 mdTdy = dot(dpdy, mdPdt) * invDotDpdv;
77 }
78 } else {
79 // We have an non-orthgonal basis, plus the system is over constrained.
80 // Find largest component of normal (since it's orthogonal to both dpdu
81 // and dpdv) and discard that axis. See pbrt 2nd ed. pg 508 for details.
82
83 scene_rdl2::math::Vec4f A;
84 scene_rdl2::math::Vec2f bx, by;
85
86 // component wise multiply to get rid of negative signs
87 const scene_rdl2::math::Vec3f n2 = mNg * mNg;
88 if (n2.x > n2.y && n2.x > n2.z) {
89 // discard x axis, use y and z axes
90 A = scene_rdl2::math::Vec4f(mdPds.y, mdPdt.y, mdPds.z, mdPdt.z);
91 bx = scene_rdl2::math::Vec2f(dpdx.y, dpdx.z);
92 by = scene_rdl2::math::Vec2f(dpdy.y, dpdy.z);
93 } else if (n2.y > n2.z) {
94 // discard y axis, use x and z axes
95 A = scene_rdl2::math::Vec4f(mdPds.x, mdPdt.x, mdPds.z, mdPdt.z);
96 bx = scene_rdl2::math::Vec2f(dpdx.x, dpdx.z);
97 by = scene_rdl2::math::Vec2f(dpdy.x, dpdy.z);
98 } else {
99 // discard z axis, use x and y axes
100 A = scene_rdl2::math::Vec4f(mdPds.x, mdPdt.x, mdPds.y, mdPdt.y);
101 bx = scene_rdl2::math::Vec2f(dpdx.x, dpdx.y);
102 by = scene_rdl2::math::Vec2f(dpdy.x, dpdy.y);
103 }
104
105 float detA = A.x * A.w - A.y * A.z;
106 if (!isZero(detA, sIntersectionEpsilon)) {
107 MNRY_VERIFY(solve2x2LinearSystem(&A.x, &mdSdx, &bx.x));
108 MNRY_VERIFY(solve2x2LinearSystem(&A.x, &mdSdy, &by.x));
109 }
110 }
111 }
112
113 // Texture derivatives are always assumed to be valid by this point,
114 // regardless if we had valid ray differentials or valid geometry derivatives.
115 MNRY_ASSERT(finite(mdSdx) && finite(mdTdx) && finite(mdSdy) && finite(mdTdy));
116
117 // Scale the texture derivatives
118 mdSdx *= textureDiffScale;
119 mdSdy *= textureDiffScale;
120 mdTdx *= textureDiffScale;
121 mdTdy *= textureDiffScale;
122}

138Intersection::adaptToonNormal(const scene_rdl2::math::Vec3f &Ns) const
139{
140 // This flag is set by the integrator if the ToonMaterial has
141 // indicated that no light culling should be performed. If so,
142 // the normal is not adapted. This is useful for certain NPR effects
143 // that use an arbitrary normal.
144 if (!mFlags.get(UseAdaptNormal)) {
145 return Ns;
146 }
147
148 return adaptNormal(Ns);

151scene_rdl2::math::Vec3f
152Intersection::adaptNormal(const scene_rdl2::math::Vec3f &Ns) const
153{
154 // Ns is the result of a normal mapping operation.
155 // The result may or may not be physically plausible from
156 // our particular view point. This function detects this
157 // case and minimally bends the shading normal back towards
158 // the physically plausible geometric normal. This unfortunately
159 // breaks bi-directional algorithms. See "Microfacet-based
160 // Normal Mapping for Robust Monte Carlo Path Tracing" for
161 // an apporach to normal mapping that can work bi-directionally.
162 //
163 // The solution employed here is descrbied in
164 // "The Iray Light Transport System" Section A.3 "Local Shading
165 // Normal Adaption".
166 //
167 const scene_rdl2::math::Vec3f &wo = getWo();
168 const scene_rdl2::math::Vec3f &Ng = getNg();
169
170 scene_rdl2::math::Vec3f result = Ns;
171
172 // Compute reflection vector r, which is wo reflected about Ns
173 const scene_rdl2::math::Vec3f r = 2.f * dot(Ns, wo) * Ns - wo;
174
175 // If r is above the horizon of Ng, we're in good shape
176 const float rDotNg = dot(r, Ng);
177
178 // compute a bent reflection vector that is just
179 // slightly (based on eps) above the Ng horizon
180 const float eps = 0.01f;
181 if (rDotNg < eps) {
182 // this appears to work, but has the wrong normalization
183 // if this becomes a problem, the correct normalization can be
184 // computed via:
185 // const float s = scene_rdl2::math::sqrt(scene_rdl2::math::max(0.f, (1 - rDotNg * rDotNg)) / (1 - eps * eps));
186 // const float alpha = eps * s - rDotNg;
187 // scene_rdl2::math::Vec3f rBent = r + alpha * Ng;
188 //
189 scene_rdl2::math::Vec3f rBent = r + (eps - rDotNg) * Ng;
190
191 // Normalize it.
192 // Worth noting: since we are going to normalize(wo + rBent)
193 // we could alternatively scale wo via wo = wo * |rBent| and obtain
194 // the same result for bentNs.
195 rBent = scene_rdl2::math::normalize(rBent);
196
197 // Compute bent Ns as halfway vector between wo and rBent
198 scene_rdl2::math::Vec3f bentNs = scene_rdl2::math::normalize(wo + rBent);
199 result = bentNs;
200 }
201
202 return result;
203}
