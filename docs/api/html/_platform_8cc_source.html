196 static double getFrequencyInMHz()
197 {
198 struct timeval tvstart, tvstop;
199 unsigned long long int cycles[2];
200
201 gettimeofday(&tvstart, NULL);
202 cycles[0] = rdtsc();
203 gettimeofday(&tvstart, NULL);
204 usleep(250000);
205 gettimeofday(&tvstop, NULL);
206 cycles[1] = rdtsc();
207 gettimeofday(&tvstop, NULL);
208
209 const unsigned long microseconds = ((tvstop.tv_sec-tvstart.tv_sec)*1000000) + (tvstop.tv_usec-tvstart.tv_usec);
210 unsigned long mhz = (unsigned long) (cycles[1]-cycles[0]) / microseconds;
211
212 //std::cout << "MIC frequency is " << mhz << " MHz" << std::endl;
213 return (double)mhz;
214 }
