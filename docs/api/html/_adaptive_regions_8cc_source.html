28inline void doFlush(mcrt_common::ThreadLocalState* tls)
29{
30 auto driver = getRenderDriver();
31 MNRY_ASSERT(driver);
32
33 unsigned flushed;
34 do {
35 flushed = shading::Material::flushNonEmptyShadeQueue(tls);
36 flushed += driver->flushXPUQueue(tls, &tls->mArena);
37 flushed += tls->mPbrTls->flushLocalQueues();
38 } while (flushed);
39}

42void AdaptiveRegions::init(scene_rdl2::math::BBox2i renderBounds, bool vectorized)
43{
44 mVectorized = vectorized;
45 mRegions.init(renderBounds);
46 for (int i = 0; i < mRegions.getNumRegions(); ++i) {
47 const scene_rdl2::math::BBox2i bounds = mRegions.getOverlappingRegionBounds(i);
48
49 mTrees[i] = AdaptiveRegionTree(bounds);
50 mRegionError[i] = std::numeric_limits<float>::max();
51 mDone[i] = false;
52 mRegionTileCount[i].value = mNumTiles[i] = tilesHorizontal(extents(bounds, 0)) *
53 tilesVertical(extents(bounds, 1));
54 mAdaptiveTreeUpdateCounter[i] = 0;
55 }
56 disableAdjustUpdateTiming();
57}

64void AdaptiveRegions::update(const scene_rdl2::math::BBox2i& tile,
65 const scene_rdl2::fb_util::Tiler& tiler,
66 const scene_rdl2::fb_util::RenderBuffer& renderBuf,
67 const scene_rdl2::fb_util::FloatBuffer& numSamplesBuf,
68 const scene_rdl2::fb_util::RenderBuffer& renderBufOdd,
69 float targetError,
70 mcrt_common::ThreadLocalState* tls,
71 const unsigned endSampleIdx)
72{
73 // This function must be called for EVERY tile, even if the tile ends up doing no rendering work. If this isn't
74 // adhered to, the load balancing will become off.
75
76 // Our regions overlap to help reduce seams. getOverlappedRegions returns an array of (potentially redundant)
77 // valid region indices for this tile, and adds a -1 at the end of valid indices. Instead of sorting the array
78 // and removing redundant elements, we keep track of which regions we have already visited and skip those.
79 //
80 // Sorting a nine-element array, even with a hard-coded sorting network, and then running std::unique on it is
81 // much less efficient than just going through the array once and keeping track of what we've seen.
82 //
83 // The overlapped regions do not incur more rendering costs, it simply expands are area of interest when
84 // calculating error. When a tile is rendered, it is only checked against its primary region (getSampleArea).
85
86 IndexArray indices;
87 mRegions.getOverlappedRegions(tile, indices);
88
89 VisitedArray visited = { false };
90 MNRY_ASSERT(std::all_of(visited.cbegin(), visited.cend(), [](bool b) { return b == false; }));
91
92 // Keep track of what region this thread was last in for queue flushing purposes.
93 static thread_local int localRegionID = -1;
94 for (int idx : indices) {
95 if (idx < 0) {
96 break;
97 } else if (visited[idx]) {
98 continue;
99 }
100 visited[idx] = true;
101
102 if (mNumTiles[idx] == 0) {
103 // This can happen when, for example, we're doing debug pixels and we have a screen-space quad with no
104 // area.
105 continue;
106 }
107
108 // Decrement the number of tiles that have been rendered in this quad. If we're the last thread, we update
109 // the AdaptiveRegionTree.
110 if (--mRegionTileCount[idx].value == 0) {
111 if (mVectorized) {
112 doFlush(tls);
113 }
114
115 EXCL_ACCUMULATOR_PROFILE(tls, EXCL_EXCL_LOCK_ADAPTIVE_TREE);
116 mMutexPool[idx].exclusiveLockAll();
117 EXCL_ACCUMULATOR_PROFILE(tls, EXCL_BUILD_ADAPTIVE_TREE);
118
119 // The memory ordering specifies how non-atomic memory is accessed around an atomic read or write.
120 // Relaxed memory order specifies neither ordering nor synchronization, but we don't need to do that in
121 // these cases, as the lock above ensures proper ordering and synchronization. If we put more
122 // constrained memory ordering, we're simply paying the price twice.
123 int qpc = mRegionTileCount[idx].value.load(std::memory_order_relaxed);
124 //PRINT3(idx, qpc, mNumTiles[idx]);
125 while (qpc <= 0) {
126 // More threads may have come through between first read of mRegionTileCount and the lock, causing
127 // the value to go negative, so we have to increment until positive to account for these possible
128 // accesses. This means that fewer than mNumTiles[idx] are rendered by threads on the next pass,
129 // which is a form of automatic load balancing.
130 qpc += mNumTiles[idx];
131 }
132 mRegionTileCount[idx].value.store(qpc, std::memory_order_relaxed);
133
134 if (mUpdateSentinel.shouldUpdate(endSampleIdx)) {
135 // useful debug dump functions for input of adaptive tree update
136 // saveRenderBufferByPPM(idx, tiler, renderBuf, false);
137 // saveRenderBufferByPPM(idx, tiler, renderBufOdd, true);
138 // saveFloatBufferByPPM(idx, tiler, numSamplesBuf);
139
140 // We have to update adaptive tree
141 const float error = mTrees[idx].update(tiler, renderBuf, numSamplesBuf, renderBufOdd, targetError);
142 mRegionError[idx].store(error, std::memory_order_relaxed);
143 mDone[idx].store(mTrees[idx].done(), std::memory_order_relaxed);
144
145 // savePixelErrorsByPPM(idx); // useful debug dump all pixelError info to the disk as image
146 ++mAdaptiveTreeUpdateCounter[idx];
147 }
148 mMutexPool[idx].unlockAll();
149 } else if (mVectorized && idx != localRegionID) {
150 // We don't want to flush too often, so we do it once per thread per adaptive region by keeping track of the
151 // last region we were in. This falls down a bit if we restart the render (we may miss a flush).
152 doFlush(tls);
153#pragma warning push
154#pragma warning disable 1711
155 // warning #1711: assignment to statically allocated variable
156 localRegionID = idx;
157#pragma warning pop
158 }
159 }
160}

162void
163AdaptiveRegions::updateAll(const scene_rdl2::fb_util::Tiler& tiler,
164 const scene_rdl2::fb_util::RenderBuffer& renderBuf,
165 const scene_rdl2::fb_util::FloatBuffer& numSamplesBuf,
166 const scene_rdl2::fb_util::RenderBuffer& renderBufOdd,
167 float targetError)
168//
169// This function is only called at initialization stage (i.e. before start MCRT) of resume render
170// by single thread
171//
172{
173 for (int idx = 0; idx < sMaxNRegions; ++idx) { // Can we do by MT?
174 const float error = mTrees[idx].update(tiler, renderBuf, numSamplesBuf, renderBufOdd, targetError);
175 mRegionError[idx].store(error, std::memory_order_relaxed);
176
177 // savePixelErrorsByPPM(idx); // useful debug dump all pixelError info to the disk as image
178 ++mAdaptiveTreeUpdateCounter[idx];
179 }

184//
185// For debugging purpose function.
186// Save pixelError information to the disk with some message.
187//
188{
