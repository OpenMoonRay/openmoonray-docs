131bool
132State::getRefP(math::Vec3f &refP) const
133{
134 bool retVal = false;
135 if (isProvided(shading::StandardAttributes::sRefP)) {
136 refP = getAttribute(shading::StandardAttributes::sRefP);
137 retVal = true;
138 } else {
139 refP = math::zero;
140 }
141 return retVal;
142}

144bool
145State::getRefN(math::Vec3f &refN) const
146{
147 refN = math::zero;
148
149 if (isProvided(shading::StandardAttributes::sRefN)) {
150 refN = getAttribute(shading::StandardAttributes::sRefN);
151 } else if (isProvided(shading::StandardAttributes::sRefP)) {
152 // Compute refN from refP partials
153 math::Vec3f dPds, dPdt;
154 getdVec3fAttrds(shading::StandardAttributes::sRefP, dPds);
155 getdVec3fAttrdt(shading::StandardAttributes::sRefP, dPdt);
156 if (math::isZero(math::lengthSqr(dPds))) {
157 // Invalid zero vector. Avoid divide by zero and NaNs
158 // that would result from math::normalize().
159 return false;
160 }
161 dPds = math::normalize(dPds);
162 // For curves, since dPdt is zero, we use the camera direction
163 // Wo and make it orthogonal to dPds
164 if (math::isZero(math::lengthSqr(dPdt))) {
165 dPdt = getWo();
166 dPdt = math::cross(dPds, math::cross(dPds, math::normalize(dPdt)));
167
168 }
169 refN = math::cross(dPds, math::normalize(dPdt));
170 if (math::isZero(math::lengthSqr(refN))) {
171 // dPds and dPdt are identical so their cross product is a zero vector.
172 return false;
173 }
174 } else {
175 return false;
176 }
177
178 if (isEntering() == false) {
179 // Flip reference space normals on exiting a surface
180 refN = -refN;
181 }
182
183 return true;
184}

186math::Vec3f
187State::getdPdx() const
188{
189 const math::Vec3f dpds = getdPds();
190 const math::Vec3f dpdt = getdPdt();
191 const float dsdx = getdSdx();
192 const float dtdx = getdTdx();
193
194 const math::Vec3f dpdx = dpds * dsdx + dpdt * dtdx;
195
196 return dpdx;
197}
198
199math::Vec3f
200State::getdPdy() const
201{
202 const math::Vec3f dpds = getdPds();
203 const math::Vec3f dpdt = getdPdt();
204 const float dsdy = getdSdy();
205 const float dtdy = getdTdy();
206
207 const math::Vec3f dpdy = dpds * dsdy + dpdt * dtdy;
208
209 return dpdy;
210}
