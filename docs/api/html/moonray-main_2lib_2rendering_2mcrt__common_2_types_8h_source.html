68// Optimized AOS <--> SOA transposes cause 64 bit addresses to be split up.
69// Here is a class to help deal with this issue.

71// Use for non-const addresses. We must use the corresponding non-const class on
72// the ISPC side.
73struct Address64v
74{
75 void *get(unsigned idx) { return (void *)(((uint64_t(mHigh[idx])) << 32) | mLow[idx]); }
76 void set(unsigned idx, void *addr) { uint64_t p = uint64_t(addr); mHigh[idx] = p >> 32; mLow[idx] = p & 0xffffffff; }
77
