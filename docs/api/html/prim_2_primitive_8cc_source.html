39std::unique_ptr<EmissionDistribution>
40Primitive::computeEmissionDistribution(const scene_rdl2::rdl2::VolumeShader* volumeShader) const
41{
42 // We voxelize the geometry bounding box into approx 1000 voxels.
43 // (The reason it's only approximate is that in initializing the variable 'res' below,
44 // we perform a rounding operation in each axis.)
45 int numVoxels = 1000;
46
47 // Generate (up to) 2 bounding boxes and their associated distToRender matrices (for t=0 and t=1)
48 BBox3f bboxes[2];
49 scene_rdl2::math::Mat4f distToRender[2];
50 size_t numTimeSteps = getMotionSamplesCount();
51 for (size_t timeStep = 0; timeStep < numTimeSteps; timeStep++) {
52 BBox3f bbox = computeAABBAtTimeStep(timeStep);
53 bboxes[timeStep] = bbox;
54 Vec3f low = bbox.lower;
55 Vec3f upp = bbox.upper;
56 Vec3f len = upp - low;
57 float s = std::cbrt(len.x * len.y * len.z / numVoxels);
58
59 scene_rdl2::math::Vec3i res(scene_rdl2::math::ceil(len.x / s), scene_rdl2::math::ceil(len.y / s), scene_rdl2::math::ceil(len.z / s));
60
61 distToRender[timeStep] = scene_rdl2::math::Mat4f( s , 0.0f , 0.0f , 0.0f,
62 0.0f , s , 0.0f , 0.0f,
63 0.0f , 0.0f , s , 0.0f,
64 low.x, low.y, low.z, 1.0f);
65 }
66
67 Vec3f low, upp;
68 if (numTimeSteps == 1) {
69 low = bboxes[0].lower;
70 upp = bboxes[0].upper;
71 distToRender[1] = distToRender[0];
72 } else {
73 // Values for the bounding box at t = 0.5, used for generating the histogram for the emission distribution
74 low = lerp(bboxes[0].lower, bboxes[1].lower, 0.5f);
75 upp = lerp(bboxes[0].upper, bboxes[1].upper, 0.5f);
76 }
77 Vec3f len = upp - low;
78 float sCubed = len.x * len.y * len.z / numVoxels;
79 float s = std::cbrt(sCubed);
80 scene_rdl2::math::Vec3i res(scene_rdl2::math::ceil(len.x / s), scene_rdl2::math::ceil(len.y / s), scene_rdl2::math::ceil(len.z / s));
81 std::vector<float> histogram(res.x * res.y * res.z, 1.0f);
82
83 // Bake volume shader emission into the grid.
84 shading::TLState *tls = mcrt_common::getFrameUpdateTLS()->mShadingTls.get();
85 float *hist = histogram.data();
86 for (int k = 0; k < res.z; ++k) {
87 float z = low.z + k*s;
88 for (int j = 0; j < res.y; ++j) {
89 float y = low.y + j*s;
90 for (int i = 0; i < res.x; ++i) {
91 float x = low.x + i*s;
92 shading::Intersection isect;
93 isect.setP(Vec3f(x,y,z));
94 *hist++ = scene_rdl2::math::luminance(volumeShader->emission(tls, shading::State(&isect), scene_rdl2::math::Color(1.0f)));
95 }
96 }
97 }
98
99 // Construct and return emission distribution
100 float invUnitVolume = 1.0f / sCubed;
101 return std::unique_ptr<EmissionDistribution>(new DenseEmissionDistribution(
102 res, distToRender, invUnitVolume, histogram));
103}

105void
106Primitive::bakeVolumeShaderDensityMap(const scene_rdl2::rdl2::VolumeShader* volumeShader,
107 const scene_rdl2::math::Mat4f& primToRender,
108 const MotionBlurParams& motionBlurParams,
109 const VolumeAssignmentTable* volumeAssignmentTable,
110 const int assignmentId)
111{
112 constexpr int sDefaultRez = 100;
113
114 if (volumeShader->isHomogenous() || getType() != POLYMESH) {
115 // If there is no map binding, or the primitive does not support volume shaders (is not a mesh),
116 // do not bake velocity grid nor density grid. Store uniform color for density instead.
117 // There is no need to initialize isect nor get shading tls in order to evaluate a solid color.
118 shading::Intersection isect;
119 shading::TLState *tls = nullptr;
120 mDensityColor = volumeShader->extinct(tls, shading::State(&isect), scene_rdl2::math::Color(1.f));
121 return;
122 }
123
124 // local space bounds
125 // shared primitives already return their aabb in their local space
126 const scene_rdl2::math::BBox3f localBbox = getIsReference() ?
127 computeAABB() : scene_rdl2::math::transformBBox(primToRender.inverse(), computeAABB());
128 const scene_rdl2::math::Vec3f& localMin = localBbox.lower;
129 const scene_rdl2::math::Vec3f& localMax = localBbox.upper;
130
131 // world space bounds
132 const scene_rdl2::math::Mat4f& primToWorld = scene_rdl2::math::toFloat(mRdlGeometry->get(scene_rdl2::rdl2::Node::sNodeXformKey));
133 const scene_rdl2::math::BBox3f worldSpaceBBox = scene_rdl2::math::transformBBox(primToWorld, localBbox);
134 const Vec3f worldSpaceSize = worldSpaceBBox.upper - worldSpaceBBox.lower;
135 float maxSize = scene_rdl2::math::max(worldSpaceSize.x, scene_rdl2::math::max(worldSpaceSize.y, worldSpaceSize.z));
136
137 // set resolution of grid
138 int rez = sDefaultRez;
139 int bakeResolutionMode = volumeShader->getBakeResolutionMode();
140 switch (bakeResolutionMode) {
141 case 1:
142 {
143 // Divisions
144 rez = volumeShader->getBakeDivisions();
145 break;
146 }
147 case 2:
148 {
149 // Voxel size
150 float voxelSize = volumeShader->getBakeVoxelSize();
151 rez = scene_rdl2::math::ceil(maxSize / voxelSize);
152 break;
153 }
154 }
155
156 if (rez <= 0) {
157 // Bad user input. Don't bake grid.
158 mDensityColor = scene_rdl2::math::Color(1.0);
159 return;
160 }
161
162 // Get volume ids. There is a separate velocity/density sampler for each volume id.
163 const std::vector<int>& volumeIds = volumeAssignmentTable->getVolumeIds(assignmentId);
164
165 // Velocity grid for motion blur.
166 createVelocityGrid(rez / 2, motionBlurParams, volumeIds);
167
168 if(!volumeShader->hasExtinctionMapBinding()) {
169 // If there is no map binding for density, don't bake. Store uniform color instead.
170 // There is no need to initialize isect nor get shading tls in order to evaluate a solid color.
171 shading::Intersection isect;
172 shading::TLState *tls = nullptr;
173 mDensityColor = volumeShader->extinct(tls, shading::State(&isect), scene_rdl2::math::Color(1.f));
174 return;
175 }
176
177 // make resolution for grid
178 const Vec3f upperBound = static_cast<float>(rez) * worldSpaceSize / maxSize;
179 const openvdb::math::CoordBBox vdbBbox(0, 0, 0, upperBound.x, upperBound.y, upperBound.z);
180
181 // Make transform of grid
182 // 1. Scale grid to be same size as primitive (localMax - localMin)
183 // 2. Translate grid to same location as primitive in local space.
184 // 3. Apply local to render space transform (if primitive is not shared)
185 const scene_rdl2::math::Mat4f l2r = getIsReference() ? scene_rdl2::math::Mat4f(scene_rdl2::math::one) : primToRender;
186 openvdb::math::Transform::Ptr transform = openvdb::math::Transform::createLinearTransform(
187 // step 3
188 openvdb::math::Mat4f(
189 l2r.vx.x, l2r.vx.y, l2r.vx.z, l2r.vx.w,
190 l2r.vy.x, l2r.vy.y, l2r.vy.z, l2r.vy.w,
191 l2r.vz.x, l2r.vz.y, l2r.vz.z, l2r.vz.w,
192 l2r.vw.x, l2r.vw.y, l2r.vw.z, l2r.vw.w
193 )
194 );
195 // step 2
196 transform->preTranslate(openvdb::Vec3f(localMin.x, localMin.y, localMin.z));
197 // step 1
198 const Vec3f scale = (localMax- localMin) / upperBound;
199 transform->preScale(openvdb::Vec3d(scale.x, scale.y, scale.z));
200
201 // initialize density grid
202 mBakedDensityGrid = openvdb::Vec3SGrid::create();
203 mBakedDensityGrid->denseFill(vdbBbox, openvdb::Vec3f(1.f, 1.f, 1.f), true);
204 mBakedDensityGrid->setTransform(transform);
205 // compute feature size from transform
206 const openvdb::Vec3d voxelSize = transform->voxelSize();
207 mFeatureSize = scene_rdl2::math::max(voxelSize.x(), voxelSize.y(), voxelSize.z());
208 volumeAssignmentTable->setFeatureSizes(this, volumeIds, mFeatureSize);
209 // bake map shader
210 openvdb::tools::foreach(mBakedDensityGrid->beginValueOn(),
211 [&](const openvdb::Vec3SGrid::ValueOnIter& it) {
212 // get xyz position of grid in render space
213 // world space == render space
214 const openvdb::Vec3d pd = mBakedDensityGrid->indexToWorld(it.getCoord());
215 // If the primitive is a reference, it is generated and baked
216 // in local space. Unfortunately, some shaders (such as projection maps
217 // and the OpenVdbMap) require a true render space position in order
218 // to work correctly. So to handle this we'll pretend that the reference
219 // primitive is located at the world origin and transform it into render
220 // space using the world2render transform (which should be what primToRender is).
221 const scene_rdl2::math::Vec3f p = getIsReference() ?
222 scene_rdl2::math::transformPoint(primToRender, scene_rdl2::math::Vec3f(pd.x(), pd.y(), pd.z())) :
223 scene_rdl2::math::Vec3f(pd.x(), pd.y(), pd.z());
224 // sample volume shader
225 shading::Intersection isect;
226 isect.init(getRdlGeometry());
227 isect.setP(p);
228 shading::TLState *tls = mcrt_common::getFrameUpdateTLS()->mShadingTls.get();
229 const scene_rdl2::math::Color result = volumeShader->extinct(tls, shading::State(&isect), scene_rdl2::math::Color(1.f));
230 // set result on grid
231 it.setValue(openvdb::Vec3f(result.r, result.g, result.b));
232 });
233
234 mBakedDensityGrid->pruneGrid();
235 mBakedDensitySampler.initialize(mBakedDensityGrid,
236 volumeIds,
237 STATS_BAKED_DENSITY_GRID_SAMPLES);
238}

250scene_rdl2::math::Color
251Primitive::evalDensity(mcrt_common::ThreadLocalState* tls,
252 uint32_t volumeId,
253 const Vec3f& pSample) const
254{
255 if (mBakedDensitySampler.mIsValid) {
256 const openvdb::Vec3d p(pSample[0], pSample[1], pSample[2]);
257 const openvdb::Vec3f density = mBakedDensitySampler.eval(tls,
258 volumeId,
259 p,
260 geom::internal::Interpolation::POINT);
261 return scene_rdl2::math::Color(density.x(), density.y(), density.z());
262 } else {
263 return mDensityColor;
264 }
