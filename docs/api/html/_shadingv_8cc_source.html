90void
91moonray::shading::shadev(const rdl2::Material *material,
92 moonray::shading::TLState *tls,
93 int numStatev,
94 const moonray::shading::Statev *statev,
95 rdl2::Bsdfv *bsdfv)
96{
97
98 // hook up attribute offset table to evaluate primitive attributes
99 // in ispc.
100 tls->getAttributeOffsetsFromRootShader(*material);
101
102 // safely cast the opaque rdl2 types
103 {
104 EXCL_ACCUMULATOR_PROFILE(tls, EXCL_ACCUM_SHADING);
105 tls->startIspcAccumulator();
106
107 // Our Bsdv array is already allocated and initialized, but for
108 // shading each on e needs to be "wrapped" inside a BsdfBuilder.
109
110 // Alloc an array of BsdfBuilderv
111 BsdfBuilderv *builderv = tls->mArena->allocArray<BsdfBuilderv>(numStatev, CACHE_LINE_SIZE);
112
113 // Initialize the BsdfBuilderv's (and the Bsdfv's) by calling ispc
114 // function via func ptr. This approach allows casting from c++ to
115 // ispc types:
116 // * shading::TLState -> ispc::ShadingTLState
117 // * shading::State -> ispc::State
118 BsdfBuilderArrayInitFuncv initBsdfBuilderArray = (BsdfBuilderArrayInitFuncv)ispc::getBsdfBuilderInitFunc();
119 initBsdfBuilderArray(tls, numStatev, statev, bsdfv, builderv, scene_rdl2::util::sAllOnMask);
120 material->shadev(tls,
121 numStatev,
122 reinterpret_cast<const rdl2::Statev *>(statev),
123 reinterpret_cast< rdl2::BsdfBuilderv *>(builderv));
124 tls->stopIspcAccumulator();
125 }
126
127 // Evaluate and store the post scatter extra aovs on the bsdf object.
128 // They will be accumulated after ray scattering.
129 {
130 EXCL_ACCUMULATOR_PROFILE(tls, EXCL_ACCUM_AOVS);
131 const moonray::shading::Material &ext = material->get<moonray::shading::Material>();
132 const std::vector<Material::ExtraAov> &extraAovs = ext.getPostScatterExtraAovs();
133 if (!extraAovs.empty()) {
134 // colorv layout:
135 // N = numStatev, M = num extra Aovs
136 // bsdf0/aov0,1,2,3...,M-1,bsdf1/aov0,1,2,3..., M-1,...,bsdfN-1,aov0,1,2,3,...,M-1
137 math::Colorv *colorv = tls->mArena->allocArray<Colorv>(numStatev * extraAovs.size(), CACHE_LINE_SIZE);
138 // labelId layout
139 // label ids are the same for each bsdf
140 // label0, label1, ... labelM-1
141 int *labelIds = tls->mArena->allocArray<int>(extraAovs.size(), CACHE_LINE_SIZE);
142 for (size_t i = 0; i < extraAovs.size(); ++i) {
143 const Material::ExtraAov &ea = extraAovs[i];
144 labelIds[i] = ea.mLabelId;
145 for (int j = 0; j < numStatev; ++j) {
146 const size_t coff = j * extraAovs.size() + i;
147 samplev(ea.mMap, tls, statev + j, colorv + coff);
148 }
149 }
150 for (int j = 0; j < numStatev; ++j) {
151 Bsdfv_setPostScatterExtraAovs(((moonray::shading::Bsdfv *) bsdfv) + j,
152 extraAovs.size(),
153 labelIds,
154 colorv + j * extraAovs.size());
155 }
156 }
157 }
158
159 tls->clearAttributeOffsets();
160
161 // hack in the bsdf label ids
162 {
163 EXCL_ACCUMULATOR_PROFILE(tls, EXCL_ACCUM_AOVS);
164 internal::setBsdfLabels(material,
165 tls,
166 numStatev,
167 statev,
168 bsdfv,
169 0); // parentLobeCount is always 0 since this is
170 // called directly from ShadeBundleHandler and
171 // this is therefore always the top level material
172 }
173}
