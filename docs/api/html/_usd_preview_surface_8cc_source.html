147void
148UsdPreviewSurface::shade(const scene_rdl2::rdl2::Material* self,
149 moonray::shading::TLState *tls,
150 const State& state,
151 BsdfBuilder& bsdfBuilder)
152{
153 const UsdPreviewSurface* me =
154 static_cast<const UsdPreviewSurface*>(self);
155
156 // Transform inputNormal from tangent space to shade space
157 scene_rdl2::math::Vec3f inputNormal = evalVec3f(me, attrNormal, tls, state);
158 const scene_rdl2::math::Vec3f &stateN = state.getN();
159 const scene_rdl2::math::ReferenceFrame frame(stateN, scene_rdl2::math::normalize(state.getdPds()));
160 const scene_rdl2::math::Vec3f N = scene_rdl2::math::normalize(frame.localToGlobal(inputNormal));
161
162 const scene_rdl2::math::Color albedo = evalColor(me, attrDiffuseColor, tls, state);
163 const float ior = evalFloat(me, attrIor, tls, state);
164 const float roughness = evalFloat(me, attrRoughness, tls, state);
165 const float opacityThreshold = evalFloat(me, attrOpacityThreshold, tls, state);
166 const float opacity = opacityThreshold > 0.0f ? 1.0f : evalFloat(me, attrOpacity, tls, state);
167
168 // Emission
169 const scene_rdl2::math::Color emissiveColor = evalColor(me, attrEmissiveColor, tls, state);
170 if (!scene_rdl2::math::isBlack(emissiveColor)) {
171 bsdfBuilder.addEmission(evalColor(me, attrEmissiveColor, tls, state));
172 }
173
174 // Clearcoat
175 const float clearcoat = evalFloat(me, attrClearcoat, tls, state);
176 if (!scene_rdl2::math::isZero(clearcoat)) {
177 const float clearcoatRoughness = evalFloat(me, attrClearcoatRoughness, tls, state);
178 if (scene_rdl2::math::isZero(clearcoatRoughness)) { // mirror
179 addMirrorDielectricLobe(bsdfBuilder, N, ior,
180 clearcoat, // weight
181 1.0f, // opacity
182 aovOuterSpecular, aovSpecularTransmission,
183 ispc::BSDFBUILDER_ADDITIVE);
184 } else { // microfacet
185 addMicrofacetDielectricLobe(bsdfBuilder, N, ior,
186 clearcoatRoughness, clearcoat,
187 1.0f, // opacity
188 aovOuterSpecular, aovSpecularTransmission,
189 ispc::BSDFBUILDER_ADDITIVE);
190 }
191 }
192
193 const float weight = 1.0f;
194
195 // Specular
196 const ispc::SpecularWorkflow specularWorkflow =
197 static_cast<ispc::SpecularWorkflow>(me->get(attrUseSpecularWorkflow));
198 if (specularWorkflow == ispc::SPECULAR_WORKFLOW_SPECULAR) {
199 // We can't support this until we implement Schlick fresnel so do nothing for now.
200 // Phase 2: implement Schlick which takes specular color directly
201 } else { // SPECULAR_WORKFLOW_METALNESS
202 const float metallic = evalFloat(me, attrMetallic, tls, state);
203 if (scene_rdl2::math::isZero(metallic)) {
204 // If "metallic" is zero then use dielectric
205 if (scene_rdl2::math::isZero(roughness)) {
206 addMirrorDielectricLobe(bsdfBuilder, N, ior, weight, opacity);
207 } else {
208 addMicrofacetDielectricLobe(bsdfBuilder, N, ior, roughness, weight, opacity);
209 }
210 } else if (scene_rdl2::math::isOne(metallic)) {
211 // If "metallic" is one then use artist friendly conductor
212 if (roughness < scene_rdl2::math::sEpsilon) {
213 addMirrorConductorLobe(bsdfBuilder, N, albedo, weight);
214 } else {
215 addMicrofacetConductorLobe(bsdfBuilder, N, albedo, roughness, weight);
216 }
217 return; // Exit early for fully metallic surfaces
218 } else {
219 // If "metallic" is between zero and one then blend dielectric
220 // and conductor based with "metallic" as the weight
221 if (roughness < scene_rdl2::math::sEpsilon) {
222 addMirrorConductorLobe(bsdfBuilder, N, albedo, metallic);
223 addMirrorDielectricLobe(bsdfBuilder, N, ior, weight, opacity);
224 } else {
225 addMicrofacetConductorLobe(bsdfBuilder, N, albedo, roughness, metallic);
226 addMicrofacetDielectricLobe(bsdfBuilder, N, ior, roughness, weight, opacity);
227 }
228 }
229 }
230
231 // Diffuse
232 if (!scene_rdl2::math::isBlack(albedo)) {
233 const moonray::shading::LambertianBRDF diffuseRefl(N, albedo);
234 bsdfBuilder.addLambertianBRDF(diffuseRefl,
235 weight,
236 ispc::BSDFBUILDER_PHYSICAL,
237 aovDiffuse);
238 }
