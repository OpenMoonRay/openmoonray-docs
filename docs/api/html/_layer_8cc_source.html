132int32_t
133Layer::assign(Geometry* geometry, const String& partName, const LayerAssignment& layerAssignment)
134{
135 if (!mUpdateActive) {
136 std::stringstream errMsg;
137 errMsg << "Can only make assignment ('" << geometry->getName() <<
138 "', '" << partName << "') in Layer '" << mName << "' between"
139 " beginUpdate() and endUpdate() calls.";
140 throw except::RuntimeError(errMsg.str());
141 }
142
143 // Assign the geometry and part
144 // For geometry with a volume shader, we ignore the parts which causes
145 // moonray to just use the entire geometry. Individual parts are generally
146 // not closed shapes which causes problems with the volume integrator
147 // because MoonRay treats parts as individual pieces of geometry. Skipping the
148 // part list uses the entire geometry as one "welded together" piece.
149 // It is OK to repeatedly call TraceSet::assign() with the same geometry and empty
150 // part because it is smart enough to lookup the existing idx before attempting
151 // to add a new entry.
152 int32_t idx = layerAssignment.mVolumeShader ? TraceSet::assign(geometry, "") :
153 TraceSet::assign(geometry, partName);
154
155 // Get mutable references to the attribute vectors.
156 auto& surfaceShaders = getMutable(sSurfaceShadersKey);
157 auto& lightSets = getMutable(sLightSetsKey);
158 auto& displacements = getMutable(sDisplacementsKey);
159 auto& volumeShaders = getMutable(sVolumeShadersKey);
160 auto& lightFilterSets = getMutable(sLightFilterSetsKey);
161 auto& shadowSets = getMutable(sShadowSetsKey);
162 auto& shadowReceiverSets = getMutable(sShadowReceiverSetsKey);
163
164 if (idx < static_cast<int32_t>(surfaceShaders.size())) {
165 // assignment is for existing geometry / part pair
166 bool shouldDirtyAssignments = false;
167 if (surfaceShaders[idx] != layerAssignment.mMaterial) {
168 surfaceShaders[idx] = layerAssignment.mMaterial;
169 shouldDirtyAssignments = true;
170 }
171 if (lightSets[idx] != layerAssignment.mLightSet) {
172 lightSets[idx] = layerAssignment.mLightSet;
173 shouldDirtyAssignments = true;
174 }
175 if (displacements[idx] != layerAssignment.mDisplacement) {
176 displacements[idx] = layerAssignment.mDisplacement;
177 shouldDirtyAssignments = true;
178 }
179 if (volumeShaders[idx] != layerAssignment.mVolumeShader) {
180 volumeShaders[idx] = layerAssignment.mVolumeShader;
181 shouldDirtyAssignments = true;
182 }
183 if (lightFilterSets[idx] != layerAssignment.mLightFilterSet) {
184 lightFilterSets[idx] = layerAssignment.mLightFilterSet;
185 shouldDirtyAssignments = true;
186 }
187 if (shadowSets[idx] != layerAssignment.mShadowSet) {
188 shadowSets[idx] = layerAssignment.mShadowSet;
189 shouldDirtyAssignments = true;
190 }
191 if (shadowReceiverSets[idx] != layerAssignment.mShadowReceiverSet) {
192 shadowReceiverSets[idx] = layerAssignment.mShadowReceiverSet;
193 shouldDirtyAssignments = true;
194 }
195
196 // IMPORTANT: Binary reader requires this attributes serialized. It can not call this
197 // method if the data is not present
198 if (shouldDirtyAssignments) {
199 dirtyAssignments();
200 }
201 } else {
202 // assignment is for new geometry / part pair
203 dirtyAssignments();
204
205 // Assignment doesn't exist yet, so create it.
206 surfaceShaders.push_back(layerAssignment.mMaterial);
207 lightSets.push_back(layerAssignment.mLightSet);
208 displacements.push_back(layerAssignment.mDisplacement);
209 volumeShaders.push_back(layerAssignment.mVolumeShader);
210 lightFilterSets.push_back(layerAssignment.mLightFilterSet);
211 shadowSets.push_back(layerAssignment.mShadowSet);
212 shadowReceiverSets.push_back(layerAssignment.mShadowReceiverSet);
213
214 MNRY_ASSERT(surfaceShaders.size() == idx + 1);
215 }
216
217 return idx;
218}

373void
374Layer::updatePrepAssignments(UpdateHelper& sceneObjects, int depth)
375{
376 MNRY_ASSERT_REQUIRE(!mUpdateActive);
377
378 mLightSetsChanged = hasChanged(sLightSetsKey);
379 mLightFilterSetsChanged = hasChanged(sLightFilterSetsKey);
380 mShadowSetsChanged = hasChanged(sShadowSetsKey);
381 mShadowReceiverSetsChanged = hasChanged(sShadowReceiverSetsKey);
382
383 // After previous update, resetAssignmentUpdates should have been called
384 // to clean up these tables.
385 MNRY_ASSERT(mChangedRootShaders.empty());
386 MNRY_ASSERT(mChangedOrDeformedGeometries.empty());
387
388 // Loop through all of the shaders in the scene and check if they are in the update graph. If so, flag so that
389 // we can update the primitive attribute tables in renderPrep(). Also flag the associated geometry for reload in
390 // renderPrep().
391 bool changed = false;
392 const auto& surfaceShaders = get(sSurfaceShadersKey);
393 const auto& geometries = get(sGeometriesKey);
394 const auto& displacements = get(sDisplacementsKey);
395 const auto& volumeShaders = get(sVolumeShadersKey);
396 for (size_t i = 0; i < surfaceShaders.size(); ++i) {
397 Geometry * const geometry = geometries[i]->asA<Geometry>();
398 if (surfaceShaders[i] != nullptr) {
399 Material * const material = surfaceShaders[i]->asA<Material>();
400 if (material && material->updatePrep(sceneObjects, depth + 1)) { // true if object is in update graph
401 mChangedRootShaders.insert(material);
402 // Geometries depend on materials because material request primitive
403 // attributes from the geometry. That means if a material changes
404 // it might request a new primitive attribute from the geometry
405 // and so the geometry would need to be reloaded and retessellated.
406 // At this point we do not know which primitive attributes the material
407 // requests, that occurs during the update calls, so we add this
408 // geometry to the list of changed or deformed geometries just in case.
409 mChangedOrDeformedGeometries[geometry] = i;
410 changed = true;
411 }
412 }
413 if (volumeShaders[i] != nullptr) {
414 VolumeShader * const volumeShader = volumeShaders[i]->asA<VolumeShader>();
415 if (volumeShader && volumeShader->updatePrep(sceneObjects, depth + 1)) { // true if object is in update graph
416 mChangedRootShaders.insert(volumeShader);
417 // Geometries depend on volumeShaders because we bake the maps into the geometry itself
418 mChangedOrDeformedGeometries[geometry] = i;
419 changed = true;
420 }
421 }
422 if (geometry) {
423 if (isDeformed(geometry)) {
424 mChangedOrDeformedGeometries[geometry] = i;
425 changed = true;
426 } else if (geometry->updatePrep(sceneObjects, depth + 1)) {
427 // true if the dirtied attributes involve geometry change
428 if (geometry->requiresGeometryUpdate(sceneObjects, depth + 1)) {
429 mChangedOrDeformedGeometries[geometry] = i;
430 }
431 changed = true;
432 }
433 }
434 if (displacements[i] != nullptr) {
435 Displacement * const displacement = displacements[i]->asA<Displacement>();
436 if (displacement && displacement->updatePrep(sceneObjects, depth + 1)) { // true if object is in update graph
437 mChangedRootShaders.insert(displacement);
438 mChangedOrDeformedGeometries[geometry] = i;
439 // geometry must re-tessellate even though no attrs or bindings have changed
440 geometry->requestUpdate();
441 changed = true;
442 }
443 }
444 }
445 // Flag LightSets, LightFilterSets, ShadowSets, and ShadowReceiverSets that need to be updated in preFrame()
446 for (SceneObject * const lightSetObj : get(sLightSetsKey)) {
447 if (lightSetObj) {
448 LightSet * lightSet = lightSetObj->asA<LightSet>();
449 if (lightSet->updatePrepLight(sceneObjects, depth + 1)) {
450 mLightSetsChanged = true;
451 changed = true;
452 }
453
454 for (SceneObject * const light : lightSet->getLights()) {
455 if (light->hasChanged(Light::sLightFiltersKey)) {
456 mLightFilterSetsChanged = true;
457 changed = true;
458 }
459 }
460 }
461 }
462
463 if (!mLightFilterSetsChanged) {
464 for (SceneObject * const lightFilterSet : get(sLightFilterSetsKey)) {
465 if (lightFilterSet && lightFilterSet->asA<LightFilterSet>()->
466 updatePrepLightFilter(sceneObjects, depth + 1)) {
467 mLightFilterSetsChanged = true;
468 changed = true;
469 }
470 }
471 }
472
473 for (SceneObject * const shadowSetObj : get(sShadowSetsKey)) {
474 if (shadowSetObj) {
475 ShadowSet * shadowSet = shadowSetObj->asA<ShadowSet>();
476 if (shadowSet->haveLightsChanged()) {
477 mShadowSetsChanged = true;
478 changed = true;
479 }
480 }
481 }
482
483 for (SceneObject * const shadowReceiverSetObj : get(sShadowReceiverSetsKey)) {
484 if (shadowReceiverSetObj) {
485 const scene_rdl2::rdl2::ShadowReceiverSet * shadowReceiverSet =
486 shadowReceiverSetObj->asA<scene_rdl2::rdl2::ShadowReceiverSet>();
487 if (shadowReceiverSet->haveGeometriesChanged()) {
488 mShadowReceiverSetsChanged = true;
489 changed = true;
490 }
491 }
492 }
493
494 // This is an optimization to avoid calling a full blown updatePrep(),
495 // which would require an unnecessary full loop over root shaders, geometries
496 // and lightSets again.
497 updatePrepFast(changed, false, sceneObjects, depth);
498}

511bool
512Layer::updatePrepFast(bool attributeTreeChanged,
513 bool bindingTreeChanged,
514 UpdateHelper& sceneObjects,
515 int depth)
516{
517 MNRY_ASSERT_REQUIRE(!mUpdateActive);
518 // early out
519 if (mUpdatePrepApplied &&
520 (sceneObjects.getDepth(this) >= depth || sceneObjects.isLeaf(this))) {
521 return updateRequired();
522 }
523 mUpdatePrepApplied = true;
524
525 mAttributeTreeChanged = attributeTreeChanged || mAttributeUpdateMask.any();
526 mBindingTreeChanged = bindingTreeChanged || mBindingUpdateMask.any();
527
528 if (mAttributeTreeChanged || mBindingTreeChanged) {
529 sceneObjects.insert(this, depth);
530 }
531 return updateRequired();
532}

556static void
557addRootShaderToSet(RootShader *s, Layer::RootShaderSet &rootShaders)
558{
559 // add this rootshader to rootshaders and any rootshader
560 // that it directly references - recurse.
561 auto res = rootShaders.insert(s);
562 if (!res.second) return; // s was already in root shader
563
564 const auto &sc = s->getSceneClass();
565 for (auto it = sc.beginAttributes(); it != sc.endAttributes(); ++it) {
566 if ((*it)->getType() == TYPE_SCENE_OBJECT) {
567 SceneObject *o = s->get<SceneObject *>(AttributeKey<SceneObject *>(**it));
568 if (o && o->isA<RootShader>()) {
569 addRootShaderToSet(o->asA<RootShader>(), rootShaders);
570 }
571 }
572 }
573}

625void
626Layer::getAllGeometryToRootShaders(GeometryToRootShadersMap & g2s)
627{
628 const auto& geometries = get(sGeometriesKey);
629 const auto& surfaceShaders = get(sSurfaceShadersKey);
630 const auto& displacements = get(sDisplacementsKey);
631 const auto& volumeShaders = get(sVolumeShadersKey);
632 for (size_t i = 0; i < geometries.size(); ++i) {
633 Geometry * const geometry = geometries[i]->asA<Geometry>();
634 if (surfaceShaders[i]) {
635 RootShader * const rootShader = surfaceShaders[i]->asA<RootShader>();
636 g2s[geometry].insert(rootShader);
637 }
638 if (displacements[i]) {
639 RootShader * const rootShader = displacements[i]->asA<RootShader>();
640 g2s[geometry].insert(rootShader);
641 }
642 if (volumeShaders[i]) {
643 RootShader * const rootShader = volumeShaders[i]->asA<RootShader>();
644 g2s[geometry].insert(rootShader);
645 }
646 if (g2s.find(geometry) == g2s.end()) {
647 g2s.emplace(geometry, RootShaderSet());
648 }
649 }

652void
653Layer::getChangedGeometryToRootShaders(GeometryToRootShadersMap & g2s)
654{
655 // If nothing is modified, do nothing.
656 // This check includes geometries with attribute, binding, and/or deform changes.
657 if (mChangedOrDeformedGeometries.empty()) return;
658
659 const auto& surfaceShaders = get(sSurfaceShadersKey);
660 const auto& displacements = get(sDisplacementsKey);
661 const auto& volumeShaders = get(sVolumeShadersKey);
662 for (auto iter = mChangedOrDeformedGeometries.begin();
663 iter != mChangedOrDeformedGeometries.end(); ++iter) {
664 // check for geometry which has attribute or binding updated
665 // or a request to update by the shader
666 // ignore those with only geometry data deformed.
667 Geometry* geom = iter->first;
668 int index = iter->second;
669 // Even if a geometry is added to mChangedOrDeformedGeometries, it still might
670 // not need to be updated. The three reasons why a geometry needs to be updated is if
671 // 1) An attribute that requires a geometry update changes. Note that if an attribute
672 // changes that does not require a geometry update, special care is taken to
673 // set the mAttributeTreeChanged flag to false.
674 // 2) An attribute binding changes
675 // 3) A shader requests that the geometry is updated.
676 // 3 is a special case. If any change is made to a geometry's assigned material,
677 // that geometry is added to mChangedOrDeformedGeometries. After this happens,
678 // we check if the material requests the geometry update. See SceneContext::applyUpdates
679 // and Layer::updatePrepAssignment for more details.
680 if (geom->updateRequired()) {
681 if (surfaceShaders[index]) {
682 RootShader * const rootShader = surfaceShaders[index]->asA<RootShader>();
683 g2s[geom].insert(rootShader);
684 }
685 if (displacements[index]) {
686 RootShader * const rootShader = displacements[index]->asA<RootShader>();
687 g2s[geom].insert(rootShader);
688 }
