19AttributeRate
20pickRate(
21 const scene_rdl2::rdl2::SceneObject* object,
22 const std::string& keyName,
23 size_t size0,
24 size_t size1,
25 const geom::RateCounts& rates)
26{
27 if (size0 != size1) {
28 Logger::warn(object->getName(), '.', keyName, ": time samples not equal size, ", size0, " != ", size1);
29 if (size1 < size0) size0 = size1;
30 }
31 return pickRate(object, keyName, size0, rates);
32}

34AttributeRate
35pickRate(
36 const scene_rdl2::rdl2::SceneObject* object,
37 const std::string& keyName,
38 size_t size,
39 const geom::RateCounts& rates)
40{
41 if (size == 0) {
42 Logger::warn(object->getName(), '.', keyName, ": invalid size 0");
43 return AttributeRate::RATE_UNKNOWN;
44 } else if (size == 1) {
45 return AttributeRate::RATE_CONSTANT;
46 } else if (size == rates.faceVaryingCount) {
47 return AttributeRate::RATE_FACE_VARYING;
48 } else if (size == rates.partCount) {
49 return AttributeRate::RATE_PART;
50 } else if (size == rates.uniformCount) {
51 return AttributeRate::RATE_UNIFORM;
52 } else if (size == rates.vertexCount) {
53 return AttributeRate::RATE_VERTEX;
54 } else if (size == rates.varyingCount) {
55 return AttributeRate::RATE_VARYING;
56 }
57 // Pick one that fits. Tried in assumed largest->smallest order. Some geometry
58 // may produce a different order but it is probably ok that the interpolation
59 // guess is not the closest one. Also 1 always turns into constant even if others
60 // have counts of 1.
61 size_t best;
62 AttributeRate rate;
63 if (rates.faceVaryingCount > 1 && size > rates.faceVaryingCount) {
64 best = rates.faceVaryingCount;
65 rate = AttributeRate::RATE_FACE_VARYING;
66 } else if (rates.vertexCount > 1 && size > rates.vertexCount) {
67 best = rates.vertexCount;
68 rate = AttributeRate::RATE_VERTEX;
69 } else if (rates.varyingCount > 1 && size > rates.varyingCount) {
70 best = rates.varyingCount;
71 rate = AttributeRate::RATE_VARYING;
72 } else if (rates.uniformCount > 1 && size > rates.uniformCount) {
73 best = rates.uniformCount;
74 rate = AttributeRate::RATE_UNIFORM;
75 } else if (rates.partCount > 1 && size > rates.partCount) {
76 best = rates.partCount;
77 rate = AttributeRate::RATE_PART;
78 } else {
79 best = 1;
80 rate = AttributeRate::RATE_CONSTANT;
81 }
82 Logger::warn(object->getName(), '.', keyName, ": invalid size ", size, " truncated to ", best);
83 return rate;
84}

88 const std::string& keyName,
89 size_t size,
90 size_t correctSize)
91{
92 if (size == correctSize) return true;

100void
101processArbitraryData(
102 const scene_rdl2::rdl2::SceneObject* geometry,
103 const scene_rdl2::rdl2::AttributeKey<scene_rdl2::rdl2::SceneObjectVector> attributeKey,
104 shading::PrimitiveAttributeTable& primitiveAttributeTable,
105 const geom::RateCounts& rates,
106 bool useFirstFrame,
107 bool useSecondFrame)
108{
109 const scene_rdl2::rdl2::SceneObjectVector& arbitraryData = geometry->get(attributeKey);
110 for (auto sceneObject : arbitraryData) {
111 const scene_rdl2::rdl2::UserData* userData = sceneObject->asA<scene_rdl2::rdl2::UserData>();
112 if (!userData) {
113 continue;
114 }
115
116 if (userData->hasBoolData()) {
117 shading::TypedAttributeKey<scene_rdl2::rdl2::Bool> key(userData->getBoolKey());
118 // bool vector is a std::deque
119 const scene_rdl2::rdl2::BoolVector& constData = userData->getBoolValues();
120 std::vector<scene_rdl2::rdl2::Bool> data(constData.begin(), constData.end());
121 primitiveAttributeTable.addAttribute(
122 key,
123 pickRate(geometry, key.getName(), data.size(), rates),
124 std::move(data));
125 }
126
127 if (userData->hasIntData()) {
128 shading::TypedAttributeKey<scene_rdl2::rdl2::Int> key(userData->getIntKey());
129 scene_rdl2::rdl2::IntVector data = userData->getIntValues();
130 primitiveAttributeTable.addAttribute(
131 key,
132 pickRate(geometry, key.getName(), data.size(), rates),
133 std::move(data));
134 }
135
136 {
137 std::vector<scene_rdl2::rdl2::FloatVector> samples;
138 if (useFirstFrame && userData->hasFloatData0()) {
139 samples.push_back(userData->getFloatValues0());
140 }
141 if (useSecondFrame && userData->hasFloatData1()) {
142 samples.push_back(userData->getFloatValues1());
143 }
144 if (!samples.empty()) {
145 shading::TypedAttributeKey<float> key(userData->getFloatKey());
146 size_t size0 = samples[0].size();
147 size_t size1 = samples.size() > 1 ? samples[1].size() : size0;
148 primitiveAttributeTable.addAttribute(
149 key,
150 pickRate(geometry, key.getName(), size0, size1, rates),
151 std::move(samples));
152 }
153 }
154
155 if (userData->hasStringData()) {
156 shading::TypedAttributeKey<std::string> key(userData->getStringKey());
157 scene_rdl2::rdl2::StringVector data = userData->getStringValues();
158 primitiveAttributeTable.addAttribute(
159 key,
160 pickRate(geometry, key.getName(), data.size(), rates),
161 std::move(data));
162 }

176 primitiveAttributeTable.addAttribute(

180 }
181 }

186 samples.push_back(userData->getVec2fValues0());

189 samples.push_back(userData->getVec2fValues1());
