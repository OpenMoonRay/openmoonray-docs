57void
58IntensityLightFilter::update(const LightFilterMap& /*lightFilters*/,
59 const Mat4d& world2render)
60{
61 if (!mRdlLightFilter) {
62 return;
63 }
64
65 float intensity = mRdlLightFilter->get<scene_rdl2::rdl2::Float>(sIntensityKey);
66 float exposure = mRdlLightFilter->get<scene_rdl2::rdl2::Float>(sExposureKey);
67 math::Color color = mRdlLightFilter->get<scene_rdl2::rdl2::Rgb>(sColorKey);
68 bool invert = mRdlLightFilter->get<scene_rdl2::rdl2::Bool>(sInvertKey);
69
70 mRadianceMod = intensity * color * powf(2.0f, exposure);
71 if (invert) {
72 if (!isZero(mRadianceMod.r)) {
73 mRadianceMod.r = 1.f / mRadianceMod.r;
74 }
75 if (!isZero(mRadianceMod.g)) {
76 mRadianceMod.g = 1.f / mRadianceMod.g;
77 }
78 if (!isZero(mRadianceMod.b)) {
79 mRadianceMod.b = 1.f / mRadianceMod.b;
80 }
81 }
82}
