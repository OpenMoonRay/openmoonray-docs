29///
30/// @class EyeCausticBsdfLobe BsdfEyeCaustic.h <rendeirng/shading/BsdfEyeCaustic.h>
31/// @brief normalized (N.L)^Exponent lobe
32/// This lobe creates a Phong-like specular lobe, but along the surface normal instead of
33/// the 'reflection vector'. It is called "EyeCaustic" because it is empirically modeled
34/// based on the caustics observed in eyeballs from refractions through the cornea.
35/// When used in conjunction with 'iris-bulge' (concave bending of the flat iris-geo)
36/// via normal mapping, the concave normals, combined with the eye caustic lobe,
37/// produce the effect of light 'pooling' on the iris, opposite to the light-reflection
38/// on the cornea. This gives the caustic-like behavior desired in our characters' eyes.
39/// References for Normalized Phong and Sampling:
40/// [1] Lafortune et al, ReportCW 1994, "Using the Modified Phong Reflectance Model for Physically Based Rendering"
41/// [2] Walter et al, EGSR 2007, "Microfacet Models for Refraction through Rough Surfaces"
42class EyeCausticBsdfLobe : public BsdfLobe
43{
44public:
45 // Constructor / Destructor
46 EyeCausticBsdfLobe(const scene_rdl2::math::Vec3f& N,
47 const scene_rdl2::math::Vec3f& irisN,
48 const scene_rdl2::math::Color& causticColor,
49 const float exponent) :
50 BsdfLobe(Type(REFLECTION | GLOSSY),
51 DifferentialFlags(0),
52 false,
53 PROPERTY_NORMAL),
54 mFrame(irisN),
55 mCausticColor(causticColor),
56 mN(N)
57 {
58 // As mentioned in [2], Phong exponents relates to Beckmann roughness using:
59 // exponent = 2 * pow(roughness, -2) -2
60 // Using a minimum roughness of 0.01, gives us the maximum exponent allowed for this lobe.
61 const float maxExponent = 20000.0f;
62 mExponent = scene_rdl2::math::clamp(exponent, 0.1f, maxExponent);
63 const float normalizedExponent = mExponent * scene_rdl2::math::rcp(maxExponent);
64 // Use a directional differential scale that varies with roughness
65 // Taken from CookTorrance roughness-based differential computation
66 mdDFactor = sdDFactorMin + normalizedExponent * sdDFactorSlope;
67
68 mNormalizationFactor = (mExponent + 2) * scene_rdl2::math::sOneOverTwoPi;
69 }
70
71
72
73 // BsdfLobe API
74 scene_rdl2::math::Color eval(const BsdfSlice &slice, const scene_rdl2::math::Vec3f &wi, float *pdf = NULL) const override
75 {
76 // Prepare for early exit
77 if (pdf != NULL) {
78 *pdf = 0.0f;
79 }
80
81 const float cosThetaO = dot(mN, slice.getWo());
82 if (cosThetaO <= 0.0f) return scene_rdl2::math::sBlack;
83
84 const float eyeCausticBrdfCosThetaI = dot(mFrame.getN(), wi);
85 if (eyeCausticBrdfCosThetaI <= 0.0f) return scene_rdl2::math::sBlack;
86
87 // Normalization Factor from [1]
88 const float normalizedPhong = mNormalizationFactor * powf(eyeCausticBrdfCosThetaI, mExponent);
89
90 // Note: we assume this lobe has been setup with a OneMinus*Fresnel
91 // as we want to use 1 - specular_fresnel. Also notice we use
92 // cosThetaWo to evaluate the fresnel term, as an approximation of what
93 // hDotWi would be for the specular lobe.
94 const scene_rdl2::math::Color f = mCausticColor * computeScaleAndFresnel(cosThetaO) * normalizedPhong *
95 (slice.getIncludeCosineTerm() ? eyeCausticBrdfCosThetaI : 1.0);
96
97 // Compute pdf of sampling ([1])
98 // section 3.3.1 in [1])
99 if (pdf != NULL) {
100 const float normalizationFactor = (mExponent + 1) * scene_rdl2::math::sOneOverTwoPi;
101 *pdf = normalizationFactor * powf(eyeCausticBrdfCosThetaI, mExponent);
102 }
103
104 // Soften hard shadow terminator due to shading normals
105 const float Gs = slice.computeShadowTerminatorFix(mFrame.getN(), wi);
106
107 return Gs * f;
108 }
109
110
111 scene_rdl2::math::Color sample(const BsdfSlice &slice, float r1, float r2,
112 scene_rdl2::math::Vec3f &wi, float &pdf) const override
113 {
114 const float cosNO = dot(mN, slice.getWo());
115 if (cosNO <= 0.0f) {
116 wi = scene_rdl2::math::Vec3f(scene_rdl2::math::zero);
117 pdf = 0.0f;
118 return scene_rdl2::math::sBlack;
119 }
120
121 // see [1] for PDF inversion
122 const float cosThetaM = scene_rdl2::math::pow(r1, scene_rdl2::math::rcp(mExponent+1));
123 const float sinThetaM = scene_rdl2::math::sqrt(1 - cosThetaM*cosThetaM);
124 const float phiM = scene_rdl2::math::sTwoPi * r2;
125
126 scene_rdl2::math::Vec3f m = computeLocalSphericalDirection(cosThetaM, sinThetaM, phiM);
127
128 // Sample along the normal
129 wi = mFrame.localToGlobal(m);
130
131 return eval(slice, wi, &pdf);
132 }
133
134
135 scene_rdl2::math::Color albedo(const BsdfSlice &slice) const override
136 {
137 float cosThetaWo = dot(mN, slice.getWo());
138 return mCausticColor * computeScaleAndFresnel(scene_rdl2::math::abs(cosThetaWo));
139 }
140
141 void differentials(const scene_rdl2::math::Vec3f &wo, const scene_rdl2::math::Vec3f &wi,
142 float r1, float r2, const scene_rdl2::math::Vec3f &dNdx, const scene_rdl2::math::Vec3f &dNdy,
143 scene_rdl2::math::Vec3f &dDdx, scene_rdl2::math::Vec3f &dDdy) const override
144 {
145 // Light reflected along N, D = N
146 // Scaled by roughness
147 dDdx = dNdx * mdDFactor;
148 dDdy = dNdy * mdDFactor;
149
150 }

153 {
154 bool handled = true;
155 switch (property)
156 {
157 case PROPERTY_NORMAL:
158 {
159 *dest = mFrame.getN().x;
160 *(dest + 1) = mFrame.getN().y;
161 *(dest + 2) = mFrame.getN().z;
162 }
163 break;
164 default:
165 handled = BsdfLobe::getProperty(property, dest);
166 break;
167 }
168
169 return handled;
170 }

172 void show(std::ostream& os, const std::string& indent) const override
173 {
174 const scene_rdl2::math::Color& scale = getScale();
175 const Fresnel * const fresnel = getFresnel();
176 os << indent << "[EyeCausticBsdfLobe]\n";
177 os << indent << " " << "scale: "
178 << scale.r << " " << scale.g << " " << scale.b << "\n";
179 os << indent << " " << "caustic color: "
180 << mCausticColor.r << " " << mCausticColor.g << " " << mCausticColor.b << "\n";
181 os << indent << " " << "exponent: " << mExponent << "\n";
182 os << indent << " " << "iris N: " << mFrame.getN().x << " " << mFrame.getN().y << " "<< mFrame.getN().z << "\n";
183 os << indent << " " << "N: " << mN.x << " " << mN.x << " " << mN.z << "\n";
184
185 if (fresnel) {
186 fresnel->show(os, indent + " ");
187 }
188 }
189
190private:
191 // Derive a directional differential scale that varies according to roughness
192 // Copied from Cook-Torrance BRDF (BsdfCookTorrance.h)
193 static constexpr float sdDFactorMin = 1.0f;
194 static constexpr float sdDFactorMax = 8.0f;
195 static constexpr float sdDFactorSlope = (sdDFactorMax - sdDFactorMin);
196
197 scene_rdl2::math::ReferenceFrame mFrame;
198 scene_rdl2::math::Color mCausticColor;
199 scene_rdl2::math::Vec3f mN;
200 float mExponent;
201 float mNormalizationFactor;
202 float mdDFactor;
203};
