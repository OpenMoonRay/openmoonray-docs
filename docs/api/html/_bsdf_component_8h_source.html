27// Abstract base class. Objects of this type serve to describe a component
28// of a Bsdf. This description includes constructor arguments for one or
29// more particular lobe types, as well as any associated Fresnel objects,
30// and any associated BsdfComponentModifiers (see below).
31class BsdfComponent
32{
33public:
34 explicit BsdfComponent(const Iridescence* const iridescence = nullptr) :
35 mIridescence(iridescence)
36 {}
37
38 virtual ~BsdfComponent() =0;
39
40 finline const Iridescence* getIridescence() const { return mIridescence; }
41
42 BsdfComponent(const BsdfComponent& other) =delete;
43 BsdfComponent& operator=(const BsdfComponent& other) =delete;
44
45private:
46 const Iridescence* const mIridescence;
47};
