78 template <typename F>
79 static size_t crawlAllActivePixels(const ActivePixels &activePixels, F activePixelFunc) {
80 size_t totalActivePix = 0;
81 uint64_t mask = 0x0;
82 for (unsigned tileId = 0; tileId < activePixels.getNumTiles(); ++tileId) {
83 if ((mask = activePixels.getTileMask(tileId))) {
84 unsigned tilePixOffset = tileId << 6;
85 for (unsigned pixOffset = 0; pixOffset < 64; ++pixOffset) {
86 if (!mask) break; // early exit
87 if (mask & static_cast<uint64_t>(0x1)) {
88 totalActivePix++;
89 unsigned currPixOffset = tilePixOffset + pixOffset;
90 activePixelFunc(currPixOffset);
91 } // active pixel
92 mask >>= 1;
93 } // pixOffset
94 } // active tile
95 } // tileId
96 return totalActivePix;
97 }

121finline void
122ActivePixels::init(const unsigned width, const unsigned height)
123{
124 if (mOriginalWidth == width && mOriginalHeight == height) {
125 return; // skip
126 }
127
128 mOriginalWidth = width;
129 mOriginalHeight = height;
130 mAlignedWidth = (width + 7) & ~7;
131 mAlignedHeight = (height + 7) & ~7;
132
133 mNumTilesX = calcNumTilesX(mAlignedWidth);
134 mNumTilesY = calcNumTilesY(mAlignedHeight);
135 unsigned totalTiles = mNumTilesX * mNumTilesY;
136
137 mTiles.resize(totalTiles, static_cast<uint64_t>(0x0));
138}

154finline void
155ActivePixels::reset(const std::vector<char> &activeTilesTbl)
156{
157 // Basically activeTilesTbl.size() should be same as mTiles.size().
158 MNRY_ASSERT(activeTilesTbl.size() == mTiles.size());
159
160 for (size_t i = 0; i < activeTilesTbl.size(); ++i) {
161 if (activeTilesTbl[i]) mTiles[i] = 0x0;
162 }
163}

202finline bool
203ActivePixels::orOp(const ActivePixels &src)
204{
205 if (src.mOriginalWidth != mOriginalWidth ||
206 src.mOriginalHeight != mOriginalHeight ||
207 src.mAlignedWidth != mAlignedWidth ||
208 src.mAlignedHeight != mAlignedHeight ||
209 src.mNumTilesX != mNumTilesX ||
210 src.mNumTilesY != mNumTilesY) {
211 return false;
212 }
213
214 if (src.mTiles.size() != mTiles.size()) {
215 return false;
216 }
217
218 uint64_t * __restrict dstPtr = &mTiles[0];
219 const uint64_t * __restrict srcPtr = &src.mTiles[0];
220 for (size_t tileId = 0; tileId < mTiles.size(); ++tileId) {
221 *dstPtr++ |= *srcPtr++;
222 }
223 return true;
224}
