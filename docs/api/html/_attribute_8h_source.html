315template <typename T>
316const T&
317Attribute::getDefaultValue() const
318{
319 // Sanity checks.
320 if (attributeType<T>() != mType) {
321 std::stringstream errMsg;
322 errMsg << "Attribute::getDefaultValue() invoked with incorrect type '" <<
323 attributeTypeName<T>() << "'. Attribute '" << mName <<
324 "' is of type '" << attributeTypeName(mType) << "'.";
325 throw except::TypeError(errMsg.str());
326 }
327
328 return *static_cast<T*>(mDefault);
329}
