16// Helper function stolen from pbrt3 that emulates the behavior of
17// std::upper_bound(). The implementation here adds some bounds checking
18// for corner cases (e.g., making sure that a valid interval is selected
19// even in the case the predicate evaluates to true or false for all entries),
20// which would normally have to follow a call to std::upper_bound()
21template <typename Predicate>
22int findInterval(int size, const Predicate& pred)
23{
24 int first = 0, len = size;
25 while (len > 0) {
26 int half = len >> 1, middle = first + half;
27 if (pred(middle)) {
28 first = middle + 1;
29 len -= half + 1;
30 } else {
31 len = half;
32 }
33 }
34 return scene_rdl2::math::clamp(first - 1, 0, size - 2);
35}

101finline void
102toUnitDisk(float& u, float& v)
103{
104 if (u == 0.5f && v == 0.5f) {
105 u = 0.0f;
106 v = 0.0f;
107 } else {
108 float phi,r;
109 const float a = 2.0f*u - 1.0f;
110 const float b = 2.0f*v - 1.0f;
111
112 if (scene_rdl2::math::abs(a) > scene_rdl2::math::abs(b)) {
113 r = a;
114 phi = (scene_rdl2::math::sPi/4.0f)*(b/a);
115 } else {
116 r = b;
117 phi = (scene_rdl2::math::sPi/2.0f) - (scene_rdl2::math::sPi/4.0f)*(a/b);
118 }
119
120 float cosPhi;
121 float sinPhi;
122
123 scene_rdl2::math::sincos(phi, &sinPhi, &cosPhi);
124
125 u = r*cosPhi;
126 v = r*sinPhi;
127 }
128}

135__forceinline float
136distb1(float r)
137{
138 float u = r;
139 for (int j = 0; j < 5; ++j) {
140 u = (11.0f * r + u * u * (6.0f + u * (8.0f - 9.0f * u)))
141 / (4.0f + 12.0f * u * (1 + u * (1.0f - u)));
142 }
143 return u;
144}
145}

150finline float
151cubicBSplineWarp(float r)
152{
153 if (r < 1.0f / 24.0f) {
154 return scene_rdl2::math::pow(24.0f * r, 0.25f) - 2.0f;
155 } else if (r < 0.5f) {
156 return detail::distb1(24.0f / 11.0f * (r - 1.0f / 24.0f)) - 1.0f;
157 } else if (r < 23.0f / 24.0f) {
158 return 1.0f - detail::distb1(24.0f / 11.0f * (23.0f / 24.0f - r));
159 } else {
160 return 2.0f - scene_rdl2::math::pow(24.0f * (1.0f - r), 0.25f);
161 }
162}

167finline float
168quadraticBSplineWarp(float r)
169{
170 if (r < 1.0f / 6.0f) {
171 return scene_rdl2::math::pow(6.0f * r, 1.0f / 3.0f) - 3.0f / 2.0f;
172 } else if (r < 5.0f / 6.0f) {
173 float u = (6.0f * r - 3.0f) / 4.0f;
174 for (int j = 0; j < 4; ++j) {
175 u = (8.0f * u * u * u - 12.0f * r + 6.0f) / (12.0f * u * u - 9.0f);
176 }
177 return u;
178 } else {
179 return 3.0f / 2.0f - scene_rdl2::math::pow(6.0f * (1.0f - r), 1.0f / 3.0f);
180 }
181}

209finline float
210vanDerCorputSequence(uint32_t n, uint32_t scramble = 0)
211{
212 // Reverse bits of _n_
213 n = (n << 16) | (n >> 16);
214 n = ((n & 0x00ff00ff) << 8) | ((n & 0xff00ff00) >> 8);
215 n = ((n & 0x0f0f0f0f) << 4) | ((n & 0xf0f0f0f0) >> 4);
216 n = ((n & 0x33333333) << 2) | ((n & 0xcccccccc) >> 2);
217 n = ((n & 0x55555555) << 1) | ((n & 0xaaaaaaaa) >> 1);
218 n ^= scramble;
219
220 // From Kollig & Keller
221 // return (float) n / (float) 0x100000000LL;
222 // From PBRT
223 //return scene_rdl2::math::min(((n >> 8) & 0xffffff) / float(1 << 24),
224 //scene_rdl2::math::sOneMinusEpsilon);
225 return scene_rdl2::util::bitsToFloat(n);
226}

229finline float
230sobolSequence(uint32_t n, uint32_t scramble = 0)
231{
232 for (uint32_t v = (1u << 31); n != 0; n >>= 1, v ^= v >> 1)
233 if (n & 0x1)
234 scramble ^= v;
235
236 // From Kollig & Keller
237 // return (float) scramble / (float) 0x100000000LL;
238 // From PBRT
239 //return scene_rdl2::math::min(((scramble >> 8) & 0xffffff) / float(1 << 24),
240 //scene_rdl2::math::sOneMinusEpsilon);
241 return scene_rdl2::util::bitsToFloat(scramble);
242}

276__forceinline float
277radicalInverse(uint32_t n, uint32_t base)
278{
279 float val = 0.0f;
280 const float invBase = 1.0f / static_cast<float>(base);
281 float invBi = invBase;
282
283 while (n > 0) {
284 const uint32_t di = (n % base);
285 val += di * invBi;
286 n *= invBase;
287 invBi *= invBase;
288 }
289
290 return val;
291}

294__forceinline float
295permutedRadicalInverse(uint32_t n, uint32_t base, const uint32_t* p)
296{
297 float val = 0.0f;
298 const float invBase = 1.0f / static_cast<float>(base);
299 float invBi = invBase;
300
301 while (n > 0) {
302 const uint32_t di = p[n % base];
303 val += di * invBi;
304 n *= invBase;
305 invBi *= invBase;
306 }
307
308 // For the case where the permutation table permutes the digit 0 to
309 // another digit, account for the infinite sequence of that digit
310 // trailing at the end of the radical inverse value.
311 val += p[0] * base / (base-1.0f) * invBi;
312 return val;
313}

316// Returns false if matrix had non-uniform scale, or true otherwise.
317finline bool
318extractUniformScale(const scene_rdl2::math::Mat4f &mat, float *scale)
319{
320 float x = mat.row0().length();
321 float y = mat.row1().length();
322 float z = mat.row2().length();
323
324 if (!scene_rdl2::math::isEqual(x, y, 0.0001f) || !scene_rdl2::math::isEqual(x, z, 0.0001f)) {
325 return false;
326 }
327
328 *scale = (x + y + z) * (1.0f / 3.0f);
329 return true;
330}

341// If integrating over a surface instead of solid angle, we need to scale
342// by this factor. If the return value could result in a denormalized result,
343// return 0. Also see absAreaToSolidAngleScale.
344finline float
345areaToSolidAngleScale(const scene_rdl2::math::Vec3f &wi, const scene_rdl2::math::Vec3f &lightNormal, float dist)
346{
347 float denom = -dot(wi, lightNormal);
348 return (denom < 0.00001f) ? 0.0f : (dist * dist) / denom;
349}
350
351// Like the areaToSolidAngleScale function but use this one if there is a
352// possibility that wi and lightNormal are within 90 degrees of each other.
353// If that's not possible, then using the above version is slightly faster since
354// it bypasses the abs call.
355finline float
356absAreaToSolidAngleScale(const scene_rdl2::math::Vec3f &wi, const scene_rdl2::math::Vec3f &lightNormal, float dist)
357{
358 float denom = scene_rdl2::math::abs(dot(wi, lightNormal));
359 return (denom < 0.00001f) ? 0.0f : (dist * dist) / denom;
360}

363finline bool
364intersectUnitSphere(const scene_rdl2::math::Vec3f &pos, const scene_rdl2::math::Vec3f &dir,
365 scene_rdl2::math::Vec3f *hit, float *dist)
366{
367 MNRY_ASSERT(isNormalized(dir));
368
369 float b = scene_rdl2::math::dot(dir, pos) * 2.0f;
370 float c = pos.lengthSqr() - 1.0f;
371
372 // Exit if ray outside sphere (c > 0) and is pointing away (b > 0).
373 if (c > 0.0f && b > 0.0f) {
374 return false;
375 }
376
377 // Find quadratic discriminant
378 float discrim = b * b - 4.0f * c;
379 if (discrim < 0.0f) {
380 return false;
381 }
382
383 float sqrtDiscrim = scene_rdl2::math::sqrt(discrim);
384 float t0 = -0.5f * ((b < 0.0f) ? (b - sqrtDiscrim) : (b + sqrtDiscrim));
385 float t1 = c / t0;
386
387 if (t0 > t1) {
388 std::swap(t0, t1);
389 }
390
391 if (t1 <= 0.0f) {
392 return false;
393 }
394
395 float t = (t0 < 0.0f) ? t1 : t0;
396
397 // Mathematically, we shouldn't need to normalize here. In practice, it helps
398 // with robustness since it ensures that all components are within [-1, 1].
399 // This is not the case under normal circumstances due to floating point
400 // inaccuracies.
401 *hit = normalize(pos + dir * t);
402 *dist = t;
403
404 return true;
405}

437// Returns a higher precision inverse when only rotations and translations
438// are present in the matrix. This method first checks for this condition and
439// falls back to the generic inverse function if the condition fails. If the
440// generic inverse is used, res will contain false.
441template<typename Scalar>
442scene_rdl2::math::Mat4<scene_rdl2::math::Vec4<Scalar> >
443rtInverse(const scene_rdl2::math::Mat4<scene_rdl2::math::Vec4<Scalar> > &m, bool &res)
444{
445 scene_rdl2::math::QuaternionT<Scalar> rot;
446 scene_rdl2::math::Mat3<scene_rdl2::math::Vec3<Scalar> > scale;
447 const scene_rdl2::math::Mat3<scene_rdl2::math::Vec3<Scalar> > m3(m.vx.x, m.vx.y, m.vx.z,
448 m.vy.x, m.vy.y, m.vy.z,
449 m.vz.x, m.vz.y, m.vz.z);
450 scene_rdl2::math::decompose(m3, scale, rot);
451 const Scalar TOL = 1e-5;
452 if (!scene_rdl2::math::isOne(scale[0][0], TOL) || !scene_rdl2::math::isOne(scale[1][1], TOL) ||
453 !scene_rdl2::math::isOne(scale[2][2], TOL)) {
454 // fallback to generic inverse
455 res = false;
456 return m.inverse();
457 }
458
459 // otherwise the inverse is just tInv * rTranpose
460 scene_rdl2::math::Mat3<scene_rdl2::math::Vec3<Scalar> > rInv(m3.transposed());
461 scene_rdl2::math::Vec3<Scalar> tInv(-m.vw.x, -m.vw.y, -m.vw.z);
462
463 res = true;
464 return
465 scene_rdl2::math::Mat4<scene_rdl2::math::Vec4<Scalar> >(1, 0, 0, 0,
466 0, 1, 0, 0,
467 0, 0, 1, 0,
468 tInv.x, tInv.y, tInv.z, 1) *
469
470 scene_rdl2::math::Mat4<scene_rdl2::math::Vec4<Scalar> >(rInv[0].x, rInv[0].y, rInv[0].z, 0,
471 rInv[1].x, rInv[1].y, rInv[1].z, 0,
472 rInv[2].x, rInv[2].y, rInv[2].z, 0,
473 0, 0, 0, 1);
474}
