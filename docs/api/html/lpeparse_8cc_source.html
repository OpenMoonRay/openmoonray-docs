98LPexp *
99Parser::buildStop(LPexp *etype, LPexp *scatter, const std::list<LPexp*> &custom)
100{
101 lpexp::Cat *cat = new lpexp::Cat();
102 cat->append(etype);
103 cat->append(scatter);
104 for (std::list<LPexp*>::const_iterator i = custom.begin(); i != custom.end(); ++i)
105 cat->append(*i);
106
107 if (custom.size() < 5)
108 cat->append (new lpexp::Repeat (new lpexp::Wildexp (m_basic_labels)));
109 cat->append(new lpexp::Symbol(Labels::STOP));
110 return cat;
111}

115LPexp *
116Parser::parseSymbol()
117{
118 bool iscustom;
119 ustring sym = parseRawSymbol(iscustom);
120 if (m_ingroup) {
121 if (sym == udot)
122 return new lpexp::Wildexp(m_minus_stop);
123 else
124 return new lpexp::Symbol(sym);
125 } else {
126 if (iscustom) {
127 std::list<LPexp *> custom;
128 custom.push_back(new lpexp::Symbol(sym));
129 return buildStop(new lpexp::Wildexp(m_minus_stop), new lpexp::Wildexp(m_minus_stop), custom);
130 } else {
131 LPexp *basics[2] = {NULL, NULL};
132 if (sym != ".") {
133 SymbolToInt::const_iterator i = m_label_position.find(sym);
134 if (i == m_label_position.end()) {
135 m_error = std::string("Unrecognized basic label: ") + sym.c_str();
136 return NULL;
137 }
138 int pos = i->second;
139 basics[pos] = new lpexp::Symbol(sym);
140 }
141 for (int k = 0; k < 2; ++k)
142 if (!basics[k])
143 basics[k] = new lpexp::Wildexp(m_minus_stop);
144 std::list<LPexp *> empty;
145 return buildStop (basics[0], basics[1], empty);
146 }
147 }
148}

152ustring
153Parser::parseRawSymbol(bool &iscustom)
154{
155 std::string sym;
156 if (head() == '\'') {
157 next();
158 while (hasInput() && head() != '\'') {
159 sym += head();
160 next();
161 }
162 if (!hasInput()) {
163 m_error = "Reached end of line looking for ' to end a literal";
164 return Labels::NONE;
165 }
166 next();
167 iscustom = true;
168 } else {
169 sym += head();
170 next();
171 iscustom = false;
172 }
173 // hacky alias for NONE label
174 if (!iscustom && sym == "x")
175 return Labels::NONE;
176 return ustring(sym);
177}

181LPexp *
182Parser::parseCat()
183{
184 //lpexp::Cat *cat = new lpexp::Cat();
185 std::vector<LPexp *> explist;
186 char endchar;
187 if (head() == '(') {
188 next();
189 endchar = ')';
190 }
191 else endchar = 0;
192
193 while (hasInput() && head() != endchar) {
194 if (head() == '|') {
195 if (!explist.size()) {
196 m_error = "No left expression to or with |";
197 for (size_t i=0; i < explist.size(); ++i)
198 delete explist[i];
199 return NULL;
200 }
201 next();
202 LPexp *e = _parse();
203 if (error()) {
204 for (size_t i=0; i < explist.size(); ++i)
205 delete explist[i];
206 return NULL;
207 }
208 if (explist.back()->getType() == lpexp::OR)
209 ((lpexp::Orlist*)explist.back())->append(e);
210 else {
211 lpexp::Orlist *orexp = new lpexp::Orlist();
212 orexp->append(explist.back());
213 orexp->append(e);
214 explist[explist.size() - 1] = orexp;
215 }
216 } else {
217 LPexp *e = _parse();
218 if (error()) {
219 for (size_t i=0; i < explist.size(); ++i)
220 delete explist[i];
221 return NULL;
222 }
223 explist.push_back(e);
224 }
225 }
226 if (hasInput() && head() == endchar)
227 next();
228 else if (endchar != 0) {
229 m_error = "Reached end of line looking for )";
230 for (size_t i=0; i < explist.size(); ++i)
231 delete explist[i];
232 return NULL;
233 }
234 lpexp::Cat *cat = new lpexp::Cat();
235 for (size_t i=0; i < explist.size(); ++i)
236 cat->append(explist[i]);
237 return cat;
238}

242LPexp *
243Parser::parseGroup()
244{
245 assert(head() == '<');
246 if (m_ingroup) {
247 m_error = "No groups allowed inside of groups";
248 return NULL;
249 }
250 int basicpos = 0;
251 LPexp *basics[2] = {NULL, NULL};
252 std::list<LPexp *> custom;
253#define THROWAWAY() do{\
254 for (int i=0;i<2;++i) if(basics[i]) delete basics[i];\
255 for (std::list<LPexp *>::iterator i = custom.begin();i!=custom.end();++i) delete *i;\
256 m_ingroup = false;\
257 return NULL;\
258 }while(0)
259
260 m_ingroup = true;
261 next();
262 while (hasInput() && head() != '>') {
263 LPexp *e = _parse();
264 if (error()) THROWAWAY();
265 if (basicpos < 2)
266 basics[basicpos++] = e;
267 else
268 custom.push_back(e);
269 }
270
271 if (!hasInput()) {
272 m_error = "Reached end of line looking for > to end a group";
273 THROWAWAY();
274 }
275 next();
276 m_ingroup = false;
277 for (; basicpos < 2; ++basicpos)
278 basics[basicpos] = new lpexp::Wildexp(m_minus_stop);
279 return buildStop(basics[0], basics[1], custom);
280}

284LPexp *
285Parser::parseNegor()
286{
287 assert (head() == '^');
288 SymbolSet symlist;
289 symlist.insert(Labels::STOP); // never allowed
290 int pos = -1;
291 next();
292 while (hasInput() && head() != ']') {
293 bool iscustom;
294 ustring sym = parseRawSymbol(iscustom);
295 if (error()) return NULL;
296 symlist.insert(sym);
297 if (iscustom) {
298 if (symlist.size() > 2 && pos != -1)
299 std::cerr << "[pathexp] you are mixing labels of different type in [...]" << std::endl;
300 pos = -1;
301 } else {
302 SymbolToInt::const_iterator found = m_label_position.find(sym);
303 if (found == m_label_position.end()) {
304 m_error = "Unrecognized basic label";
305 return NULL;
306 }
307 if (symlist.size() > 2 && found->second != pos)
308 std::cerr << "[pathexp] you are mixing labels of different type in [...]" << std::endl;
309 pos = found->second;
310 }
311 }
312 if (!hasInput()) {
313 m_error = "Reached end of line looking for ] to end an negative or list'";
314 return NULL;
315 }
316 if (symlist.size() < 2) {
317 m_error = "Empty or list [^] not allowed";
318 return NULL;
319 }
320 next();
321 lpexp::Wildexp *wildcard = new lpexp::Wildexp(symlist);
322 if (m_ingroup)
323 return wildcard;
324 else {
325 std::list<LPexp *> custom;
326 if (pos < 0) { // is a custom label
327 custom.push_back(wildcard);
328 return buildStop(new lpexp::Wildexp(m_minus_stop), new lpexp::Wildexp(m_minus_stop), custom);
329 } else {
330 LPexp *basics[2] = {NULL, NULL};
331 basics[pos] = wildcard;
332 for (int i = 0; i < 2; ++i)
333 if (!basics[i])
334 basics[i] = new lpexp::Wildexp(m_minus_stop);
335 return buildStop(basics[0], basics[1], custom);
336 }
337 }
338}

342LPexp *
343Parser::parseOrlist()
344{
345 assert(head() == '[');
346 next();
347 if (hasInput() && head() == '^')
348 return parseNegor();
349 else {
350 lpexp::Orlist *orlist = new lpexp::Orlist();
351 while (hasInput() && head() != ']') {
352 LPexp *e = _parse();
353 if (error()) {
354 delete orlist;
355 return NULL;
356 }
357 orlist->append(e);
358 }
359 if (!hasInput()) {
360 m_error = "Reached end of line looking for ] to end an or list";
361 delete orlist;
362 return NULL;
363 }
364 next();
365 return orlist;
366 }
367}

371std::pair<int, int>
372Parser::parseRange()
373{
374 assert(head() == '{');
375 next();
376 std::string firstnum = "";
377 while (hasInput() && '0' <= head() && head() <= '9') {
378 firstnum += head();
379 next();
380 }
381 std::string secondnum = "";
382 if (hasInput() && head() == ',') {
383 next();
384 while (hasInput() && '0' <= head() && head() <= '9') {
385 secondnum += head();
386 next();
387 }
388 if (!secondnum.size())
389 secondnum = "-1";
390 }
391 if (!hasInput() || head() != '}' || !firstnum.size()) {
392 m_error = "Bad {} range definition";
393 return std::pair<int, int>(-1, -1);
394 }
395 next();
396 if (secondnum.size())
397 return std::pair<int, int>(atoi(firstnum.c_str()), atoi(secondnum.c_str()));
398 else
399 return std::pair<int, int>(atoi(firstnum.c_str()), atoi(firstnum.c_str()));
400}

404LPexp *
405Parser::parseModifier(LPexp *e)
406{
407 if (hasInput()) {
408 if (head() == '*') {
409 next();
410 return new lpexp::Repeat(e);
411 } else if (head() == '{') {
412 std::pair<int, int> range = parseRange();
413 if (error()) return NULL;
414 if (range.second < 0) {
415 lpexp::Cat *cat = new lpexp::Cat();
416 cat->append(new lpexp::NRepeat(e, range.first, range.first));
417 cat->append(new lpexp::Repeat(e->clone()));
418 return cat;
419 } else
420 return new lpexp::NRepeat(e, range.first, range.second);
421 } else if (head() == '+') {
422 next();
423 lpexp::Cat *cat = new lpexp::Cat();
424 cat->append(e);
425 cat->append(new lpexp::Repeat(e->clone()));
426 return cat;
427 } else
428 return e;
429 }
430 else
431 return e;
432}

436LPexp *
437Parser::_parse()
438{
439 LPexp *e;
440 if (head() == '(')
441 e = parseCat();
442 else if (head() == '[')
443 e = parseOrlist();
444 else if (head() == '<')
445 e = parseGroup();
446 else
447 e = parseSymbol();
448 if (error())
449 return NULL;
450 return parseModifier(e);
451}
