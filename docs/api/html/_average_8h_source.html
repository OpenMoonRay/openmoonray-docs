26/* Average is a template class that tracks the sum of values as well
27 * as how many times it was incremented (via += operator).
28 * This class is useful for tracking timings for functions calls or blocks of
29 * code that get call more then once.
30 * It allows you to add the time in your desired units and then increments
31 * the number of times it has been tracked. It allows you to get the total
32 * sum (sum of all add() calls) as well as the Average.
33 */
34template <typename T>
35class Average
36{
37public:
38 __forceinline Average()
39 {
40 reset();
41 }
42
43 __forceinline Average(unsigned int count, T sum)
44 {
45 mCount = count;
46 mSum = sum;
47 }
48
49 __forceinline ~Average()
50 {
51 }
52
53 __forceinline void reset()
54 {
55 mCount = 0;
56 mSum = 0;
57 }
58
59 __forceinline unsigned long getCount() const { return mCount; }
60 __forceinline T getSum() const { return mSum; }
61 double getAverage() const
62 {
63 if (mCount == 0) {
64 return 0.f;
65 } else {
66 return mSum / mCount;
67 }
68 }
69
70 __forceinline void operator+=(T value)
71 {
72 mSum += value;
73 mCount++;
74 }
75
76 __forceinline void operator+=(const Average<T> &other)
77 {
78 mSum += other.mSum;
79 mCount += other.mCount;
80 }
81
82
83protected:
84 unsigned int mCount;
85 T mSum;
86};

137/* MixMaxAverage much like the Average class it a template class that
138 * in addition adds the ability to track the max and min values that have
139 * been observed. A little more heavy weight due to the extra storage.
140 */
141template <typename T>
142class MixMaxAverage
143{
144public:
145 __forceinline MixMaxAverage()
146 {
147 reset();
148 }
149
150 ~MixMaxAverage() {};
151
152 __forceinline void reset() {
153 mCount = 0;
154 mSum = 0;
155 mMin = std::numeric_limits<T>::max();
156 mMax = std::numeric_limits<T>::min();
157 }
158
159 __forceinline unsigned long getCount() const { return mCount; }
160 __forceinline T getSum() const { return mSum; }
161 __forceinline T getMin() const { return mMin; }
162 __forceinline T getMax() const { return mMax; }
163 __forceinline double getAverage() const
164 {
165 if (mCount == 0) {
166 return 0.f;
167 } else {
168 return double(mSum) / mCount;
169 }
170 }
171
172 __forceinline void operator+=(T value)
173 {
174 mCount++;
175 mSum += value;
176 value = scene_rdl2::math::max(value, mMax);
177 value = scene_rdl2::math::min(value, mMin);
178 }
179
180protected:
181 unsigned long mCount;
182 T mSum;
183 T mMin;
184 T mMax;
185};
