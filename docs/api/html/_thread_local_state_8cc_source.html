49// These are lightweight objects which we put into a tbb::enumerable_thread_specific
50// container so that we can map OS thread ids to consistent top level ThreadLocalState
51// objects. We have one type for frame updates, and a second type for rendering.
52struct FrameUpdateTLSProxy
53{
54 FrameUpdateTLSProxy() :
55 mTLSIndex(gNextFrameUpdateTLSIndex.fetch_and_increment())

78 void cleanUpOverflowPool()
79 {
80 // mOverflowPool is an array of pointers which are all initialized to nullptr.
81 // Iterate and free up objects until we hit the first nullptr, then our work is done.
82 for (unsigned i = 0; i < MAX_OVERFLOW_POOL_SIZE; ++i) {
83 if (mOverflowPool[i] == nullptr) {
84 break;
85 }
86 alignedFreeDtor<ThreadLocalState>(mOverflowPool[i]);
87 }
88
89 memset(mOverflowPool, 0, sizeof(mOverflowPool));
90 }

165 mAovQueueSize = 1;

258bool
259ThreadLocalState::checkForHandlerStackOverflowRisk()
260{
261 ExclusiveAccumulators *acc = MNRY_VERIFY(getExclusiveAccumulators(this));
262 unsigned accStack = acc->getStackSize();
263
264 // A little padding is added here for safety since each additional handler
265 // put on the C stack may result in multiple profiler accumulators being
266 // added to the accumulator stack, which would also be in risk of asserting
267 // when overflowed.
268 const unsigned accStackThreshold = (MAX_EXCL_ACCUM_STACK_SIZE - 20);
269
270 // Check for underflow.
271 MNRY_STATIC_ASSERT(accStackThreshold < 1000000);
272
273 // Check for both accumulator and handler stack overflows.
274 return accStack >= accStackThreshold ||
275 mHandlerStackDepth >= MAX_HANDLER_STACK_SIZE;
276}

280void
281initTLS(const TLSInitParams &initParams)
282{
283 if (gPrivate.mTLSInitialized) {
284 return;
285 }
286
287 MNRY_ASSERT_REQUIRE(!gPrivate.mRenderPhaseOfFrame);
288
289 MOONRAY_START_THREADSAFE_STATIC_WRITE;
290
291 gPrivate.mInitParams = initParams;
292
293 if (gPrivate.mInitParams.mDesiredNumTBBThreads == 0) {
294 gPrivate.mInitParams.mDesiredNumTBBThreads = tbb::task_scheduler_init::default_num_threads();
295 }
296
297 MNRY_ASSERT_REQUIRE(gPrivate.mInitParams.mDesiredNumTBBThreads);
298
299 gNextFrameUpdateTLSIndex = 0;
300
301 // Initialize tick counter.
302 gPrivate.mTicksPerSecond.init();
303
304 //
305 // Preallocate all render TLS objects.
306 //
307
308 unsigned numThreads = getNumTBBThreads();
309
310 gPrivate.mTLSList = reinterpret_cast<ThreadLocalState *>
311 (scene_rdl2::util::alignedMalloc(sizeof(ThreadLocalState) * numThreads,
312 CACHE_LINE_SIZE));
313
314 for (unsigned i = 0; i < numThreads; ++i) {
315 new (&gPrivate.mTLSList[i]) ThreadLocalState(i, true);
316 }
317
318 // Create special GUI TLS here.
319 gPrivate.mGuiTLS = reinterpret_cast<ThreadLocalState *>
320 (scene_rdl2::util::alignedMalloc(sizeof(ThreadLocalState),
321 alignof(ThreadLocalState)));
322 new (gPrivate.mGuiTLS) ThreadLocalState(numThreads, false);
323
324 // Verify that overflow pool is empty.
325 MNRY_STATIC_ASSERT(MAX_OVERFLOW_POOL_SIZE > 0);
326 MNRY_ASSERT_REQUIRE(gPrivate.mOverflowPool[0] == nullptr);
327
328 // Setup profiling accumulators.
329 gAccumulatorHandles.init(getNumTBBThreads(), getNumTLSAllocated());
330
331 // TLS is now initialized.
332 gPrivate.mTLSInitialized = true;
333
334 MOONRAY_FINISH_THREADSAFE_STATIC_WRITE;
335}

337void
338cleanUpTLS()
339{
340 if (!gPrivate.mTLSInitialized) {
341 return;
342 }
343
344 MOONRAY_START_THREADSAFE_STATIC_WRITE;
345
346 gAccumulatorHandles.cleanUp();
347
348 gPrivate.mFrameUpdateTLSMapping.clear();
349
350 MNRY_ASSERT_REQUIRE(gPrivate.mGuiTLS);
351 gPrivate.mGuiTLS->~ThreadLocalState();
352 scene_rdl2::util::alignedFree(gPrivate.mGuiTLS);
353
354 MNRY_ASSERT_REQUIRE(gPrivate.mTLSList);
355 unsigned numThreads = getNumTBBThreads();
356 for (unsigned i = 0; i < numThreads; ++i) {
357 gPrivate.mTLSList[i].~ThreadLocalState();
358 }
359 scene_rdl2::util::alignedFree(gPrivate.mTLSList);
360
361 // Clean up any extraneous update phase TLS objects.
362 gPrivate.cleanUpOverflowPool();
363
364 // Reset internal TLS related data.
365 gPrivate.~Private();
366 new (&gPrivate) Private;
367
368 MNRY_ASSERT_REQUIRE(gPrivate.mTLSInitialized == false);
369
370 MOONRAY_FINISH_THREADSAFE_STATIC_WRITE;
371}

443ThreadLocalState *getFrameUpdateTLS()
444{
445 MNRY_ASSERT_REQUIRE(gPrivate.mTLSInitialized);
446 MNRY_ASSERT_REQUIRE(!gPrivate.mRenderPhaseOfFrame);
447
448 FrameUpdateTLSProxy *proxy = &gPrivate.mFrameUpdateTLSMapping.local();
449 MNRY_ASSERT_REQUIRE(proxy);
450
451 const unsigned globalTLSIdx = proxy->mTLSIndex;
452
453 // 99.99% case.
454 if (globalTLSIdx < getNumTBBThreads()) {
455
456 ThreadLocalState *tls = gPrivate.mTLSList + globalTLSIdx;
457 updateToLatestOIIOTLS(tls);
458 return tls;
459 }
460
461 //
462 // There can be some strange conditions where additional TBB threads are spawned
463 // and added to the TBB thread pool on rare occasion.
464 //
465 // See https://software.intel.com/en-us/forums/intel-threading-building-blocks/topic/782789
466 // for details. To counteract this, we allocate new TLS objects as needed on the fly so that
467 // The function always returns a valid TLS object regardless. These go into an overflow pool
468 // which is cleared when the update phase completes.
469 //
470 // The caveat to this scheme is that we don't know up front the complete range of thread
471 // indices which will be returned, and they may not be tightly packed. This is relevant
472 // if you need to access pre-allocated arrays using the returned thread index. Be careful!!
473 //
474
475 const unsigned overflowIdx = globalTLSIdx - getNumTBBThreads();
476 MNRY_ASSERT_REQUIRE(overflowIdx < MAX_OVERFLOW_POOL_SIZE);
477
478 if (gPrivate.mOverflowPool[overflowIdx]) {
479
480 MNRY_ASSERT_REQUIRE(gPrivate.mOverflowPool[overflowIdx]->mThreadIdx == globalTLSIdx);
481 updateToLatestOIIOTLS(gPrivate.mOverflowPool[overflowIdx]);
482 return gPrivate.mOverflowPool[overflowIdx];
483 }
484
485 //
486 // At this point we have no choice but to create a new TLS object and add it
487 // to the overflow pool.
488 //
489
490 ThreadLocalState *overflowTLS = alignedMallocCtorArgs<ThreadLocalState>(CACHE_LINE_SIZE, globalTLSIdx, false);
491
492 updateToLatestOIIOTLS(overflowTLS);
493
494 MOONRAY_THREADSAFE_STATIC_WRITE(gPrivate.mOverflowPool[overflowIdx] = overflowTLS);
495
496 return overflowTLS;
497}
