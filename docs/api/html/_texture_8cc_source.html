14int
15getTextureOptionIndex(bool isDisplacement,
16 const shading::State &state)
17{
18 const shading::Intersection::PathType pathType = state.getIntersection()->getPathType();
19
20 if (isDisplacement) {
21 return TrilinearIsotropic;
22 }
23
24 switch (pathType) {
25 case Intersection::IndirectDiffuse:
26 return ClosestMipClosestTexel;
27
28 case Intersection::IndirectGlossy:
29 return LinearMipClosestTexel;
30
31 case Intersection::Primary:
32 return TrilinearAnisotropic;
33
34 case Intersection::IndirectMirror:
35 return LinearMipClosestTexel;
36
37 case Intersection::LightPath:
38 return LinearMipClosestTexel;
39
40 default:
41 MNRY_ASSERT(0 && "Missing switch case");
42 return 0;
43 }
44}

76bool
77checkTextureWindow(texture::TextureSampler* textureSampler,
78 texture::TextureHandle* handle,
79 const std::string &filename,
80 std::string &errorMsg)
81{
82 int dataWindow[4] = {0};
83 textureSampler->getTextureInfo(handle, "datawindow", dataWindow);
84 if (dataWindow[0] < 0 || dataWindow[1] < 0 ||
85 dataWindow[2] < 0 || dataWindow[3] < 0) {
86 std::ostringstream os;
87 os << "FATAL: \"" << filename <<
88 "\" has negative pixels in dataWindow: (" <<
89 dataWindow[0] << " " << dataWindow[1] << " " <<
90 dataWindow[2] << " " << dataWindow[3] << ")";
91 errorMsg = os.str();
92 return false;
93 }
94
95 int displayWindow[4] = {0};
96 textureSampler->getTextureInfo(handle, "displaywindow", displayWindow);
97 if (displayWindow[0] < 0 || displayWindow[1] < 0 ||
98 displayWindow[2] < 0 || displayWindow[3] < 0) {
99 std::ostringstream os;
100 os << "FATAL: \"" << filename <<
101 "\" has negative pixels in displayWindow: (" <<
102 displayWindow[0] << " " << displayWindow[1] << " " <<
103 displayWindow[2] << " " << displayWindow[3] << ")";
104 errorMsg = os.str();
105 return false;
106 }
107
108 return true;
109}

111std::string
112processTexturePrepareResult(OIIO::DwaPrepareResult &result)
113{
114 switch(result) {
115 case OIIO::DwaPrepareResult::DwaFailed :
116 return "Failed to prepare the texture. Perhaps file is missing?";
117 case OIIO::DwaPrepareResult::DwaIncompatibleWrapping :
118 return "Failed to prepare texture: Wrapping mode must be "
119 "clamp or periodic [pow2]. "
120 "Falling back to Scalar OIIO execution.";
121 case OIIO::DwaPrepareResult::DwaIncompatibleNoTiling :
122 return "Failed to prepare texture: Texture must be tiled. "
123 "Falling back to Scalar OIIO execution.";
124 case OIIO::DwaPrepareResult::DwaIncompatibleTileSize :
125 return "Failed to prepare texture: Texture tile size must be "
126 "a power of 2 (typically 64). "
127 "Falling back to Scalar OIIO execution.";
128 case OIIO::DwaPrepareResult::DwaIncompatibleChannelType :
129 return "Failed to prepare texture: Channel type must be "
130 "either uint8, uint16, half, float or int8. "
131 "Falling back to Scalar OIIO execution.";
132 case OIIO::DwaPrepareResult::DwaIncompatibleVolumetric :
133 return "Failed to prepare texture: Texture with volumetric "
134 "data is not supported. "
135 "Falling back to Scalar OIIO execution.";
136 case OIIO::DwaPrepareResult::DwaIncompatibleRawAccess :
137 return "Failed to prepare texture: Texture does not provide "
138 "raw data access. "
139 "Falling back to Scalar OIIO execution.";
140 default :
141 // of course this will never happen
142 return "Failed to prepare texture: Unhandled OIIO::DwaPrepareResult encountered";
143 }
144}
