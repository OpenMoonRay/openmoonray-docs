45static const int AOV_MAX_RANGE_TYPE = 100;
46
47// An aovSchema is an array of integers/AovSchemaIds that tells us what value
48// goes in a particular float array location. Some aov schema IDs are known
49// at compile time (e.g. AOV_SCHEMA_ID_STATE_P - the
50// state's position variable). Some are not (e.g. some particular primitive
51// color attribute index). Those that are not known
52// at compile time are known as a "RANGE_TYPE". Their integer values occupy
53// a reserved range of IDs. For example, any ID in the range
54// of [100, 199) corresponds to a primitive attribute float.
55// In the code, an AovSchemaId variable can be represented as an int or as the
56// AovSchemaId type. We often use an int for the general case because the
57// range type values aren't known at compile time.
58// Multiple AOVs can use the same AovSchemaId if they output the same value.
59
60// The actual value of these IDs is not important, but the order is
61// assumed. Do not re-arrange them without extensive inspection of Aov.cc.
62enum AovSchemaId {
63 AOV_SCHEMA_ID_UNKNOWN = 0,
64 AOV_SCHEMA_ID_BEAUTY,
65 AOV_SCHEMA_ID_ALPHA,
66 AOV_SCHEMA_ID_VARIANCE,
67 // state position (vec3f)
68 AOV_SCHEMA_ID_STATE_P,
69 // state geometric normal (vec3f)
70 AOV_SCHEMA_ID_STATE_NG,
71 // state shading normal (vec3f)
72 AOV_SCHEMA_ID_STATE_N,
73 // state St coord (vec2f)
74 AOV_SCHEMA_ID_STATE_ST,
75 // state dpds (vec3f)
76 AOV_SCHEMA_ID_STATE_DPDS,
77 // state dpdt (vec3f)
78 AOV_SCHEMA_ID_STATE_DPDT,
79 // state ds, dt (floats)
80 AOV_SCHEMA_ID_STATE_DSDX,
81 AOV_SCHEMA_ID_STATE_DSDY,
82 AOV_SCHEMA_ID_STATE_DTDX,
83 AOV_SCHEMA_ID_STATE_DTDY,
84 // wireframe output is handled like a state attribute aov
85 AOV_SCHEMA_ID_WIREFRAME,
86 // state world position (vec3f)
87 AOV_SCHEMA_ID_STATE_WP,
88 // state depth (-1.f * P.z)
89 AOV_SCHEMA_ID_STATE_DEPTH,
90 // motion vectors are handled like a state attribute aov
91 AOV_SCHEMA_ID_STATE_MOTION,
92
93 // Beyond here, everything should be a range type.
94 // Danger! We only have room for AOV_MAX_RANGE_TYPE of each of these!
95 // E.g. We can only support 100 different light AOV schema IDs.
96
97 // primitive attributes
98 AOV_SCHEMA_ID_PRIM_ATTR_FLOAT = 1 * AOV_MAX_RANGE_TYPE,
99 AOV_SCHEMA_ID_PRIM_ATTR_VEC2F = 2 * AOV_MAX_RANGE_TYPE,
100 AOV_SCHEMA_ID_PRIM_ATTR_VEC3F = 3 * AOV_MAX_RANGE_TYPE,
101 AOV_SCHEMA_ID_PRIM_ATTR_RGB = 4 * AOV_MAX_RANGE_TYPE,
102 // material aovs
103 AOV_SCHEMA_ID_MATERIAL_AOV_FLOAT = 5 * AOV_MAX_RANGE_TYPE,
104 AOV_SCHEMA_ID_MATERIAL_AOV_VEC2F = 6 * AOV_MAX_RANGE_TYPE,
105 AOV_SCHEMA_ID_MATERIAL_AOV_VEC3F = 7 * AOV_MAX_RANGE_TYPE,
106 AOV_SCHEMA_ID_MATERIAL_AOV_RGB = 8 * AOV_MAX_RANGE_TYPE,
107 // lpe aovs
108 AOV_SCHEMA_ID_VISIBILITY_AOV = 9 * AOV_MAX_RANGE_TYPE,
109 AOV_SCHEMA_ID_LIGHT_AOV = 10 * AOV_MAX_RANGE_TYPE,
110};
