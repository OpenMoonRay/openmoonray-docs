107 bp::list
108 PySceneContext_getGeometrySetIndicesForLayer(rdl2::SceneContext& self, rdl2::Layer* layer)
109 {
110 if (layer == nullptr) {
111 return {};
112 }
113
114 bp::list res;
115 const std::vector<rdl2::GeometrySet*> geomSetVec = self.getGeometrySetsForLayer(layer);
116
117 std::size_t counter = 0;
118 for (auto iter = self.beginGeometrySet(); iter != self.endGeometrySet(); ++iter) {
119 if (std::find(geomSetVec.cbegin(), geomSetVec.cend(), *iter) != geomSetVec.cend()) {
120 res.append(counter);
121 }
122
123 ++counter;
124 }
125
126 return res;

138 "Once the data has been loaded or updated, the rendering libraries should be "
