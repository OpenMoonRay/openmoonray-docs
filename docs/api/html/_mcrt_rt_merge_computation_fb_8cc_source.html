36bool
37RenderFb::storePartialFrame(const PartialFrame &partial)
38{
39 int machineId = partial.mMachineId;
40 if (machineId < 0 || machineId >= mNumMachines) {
41 return false;
42 }
43
44 for (const auto &buffer: partial.mBuffers) {
45 // extract buffers by type
46 if (!std::strcmp(buffer.mName, "beauty")) {
47 auto &buf = mUpstreamBuffers[machineId];
48 buf.resize(buffer.mDataLength);
49 memcpy(&buf[0], buffer.mData.get(), buffer.mDataLength);
50 // we always send status as STARTED.
51
52 } else {
53 // skip so far
54 }
55 }
56 mReceived[machineId] = true;
57 mProgress[machineId] = partial.getProgress();
58
59 mCompleteTotal = 0;
60 mProgressTotal = 0.0f;
61 for (int i = 0; i < mNumMachines; ++i) {
62 if (mReceived[i]) {
63 mCompleteTotal++;
64 mProgressTotal += mProgress[i];
65 }
66 }
67
68 if (mCompleteTotal == mNumMachines) {
69 return true; // complete all data
70 }
71 return false; // not yet or error
72}

74float
75RenderFb::unpackSparseTiles(const std::vector<std::vector<fb_util::Tile>> &tiles,
76 fb_util::VariablePixelBuffer &tiledFrame)
77{
78 // Copy each upstream buffer into the final frame buffer.
79 for (size_t i = 0; i < mUpstreamBuffers.size(); ++i) {
80 if (!tiles[i].empty() && mUpstreamBuffers[i].size() != 0) {
81 tiledFrame.unpackSparseTiles(&mUpstreamBuffers[i][0], tiles[i]);
82 }
83 }
84
85 return mProgressTotal;
86}

128bool
129RenderFbArray::storePartialFrame(const PartialFrame &partial,
130 const math::Viewport &vp)
131{
132 math::Viewport partialVp(partial.mHeader.mViewport.mMinX,
133 partial.mHeader.mViewport.mMinY,
134 partial.mHeader.mViewport.mMaxX,
135 partial.mHeader.mViewport.mMaxY);
136 if (vp != partialVp) {
137 /* Useful debug code
138 std::cout << ">>>=== McrtRtMergeComputationFb.cc vp != partialVp" << std::endl;
139 std::cout << ">>> partial"
140 << " mMinX:" << partial.mHeader.mViewport.mMinX
141 << " mMinY:" << partial.mHeader.mViewport.mMinY
142 << " mMaxX:" << partial.mHeader.mViewport.mMaxX
143 << " mMaxY:" << partial.mHeader.mViewport.mMaxY
144 << " vp"
145 << " minX:" << vp.mMinX
146 << " minY:" << vp.mMinY
147 << " maxX:" << vp.mMaxX
148 << " maxY:" << vp.mMaxY
149 << std::endl;
150 */
151 // Somehow partialFrame's viewport information is empty.
152 // We skip this test for temporal quick fix reason. Toshi (22/Sep/16)
153 // return false; // invalid viewport data
154 }
155
156 int globalFrameId = partial.mHeader.mFrameId;
157
158 {
159 int machineId = partial.mMachineId;
160 MOONRAY_LOG_INFO("Merge partialFrame machineId:%d globalFrameId:%d", machineId, globalFrameId );
161 }
162
163 if (mStartFrameId <= -1) { // initialize table
164 MOONRAY_LOG_INFO("Merge initialize multi-fb table");
165 mStartFrameId = globalFrameId;
166 mEndFrameId = mStartFrameId + mTotalCacheFrame - 1;
167 for (int i = 0; i < mTotalCacheFrame; ++i) {
168 mFbTbl[i] = &mFbArray[i];
169 mFbTbl[i]->setup(mNumMachines);
170 }
171 }
172 if (globalFrameId < mStartFrameId) {
173 return false; // too old data
174 }
175
176 int shiftCount = 0; // for debug
177 if (globalFrameId > mEndFrameId) { // we have to update table
178 int shiftOffset = globalFrameId - mEndFrameId;
179 for (int i = 0; i < shiftOffset; ++i) {
180 shiftFbTbl();
181 shiftCount++; // for debug
182 }
183 }
184
185 int localFrameId = getLocalFrameId(globalFrameId);
186 RenderFb *cFb = mFbTbl[localFrameId];
187
188 //------------------------------
189
190 if (cFb->storePartialFrame(partial)) {
191 return true;
192 }
193
194 return false;
195}

197void
198RenderFbArray::shiftFbTbl()
199{
200 RenderFb *ptr = mFbTbl[0];
201
202 {
203 int cFrameId = mStartFrameId;
204 if (!ptr->isComplete()) {
205 MOONRAY_LOG_ERROR("drop uncomplete frame:%d", cFrameId);
206 }
207 }
208
209 int lastId = mFbTbl.size() - 1;
210 for (int i = 0; i < lastId; ++i) {
211 mFbTbl[i] = mFbTbl[i+1];
212 }
213 ptr->reset(); // do reset for new recycled entry
214 mFbTbl[lastId] = ptr;
215
216 mStartFrameId++;
217 mEndFrameId++;
218}

220bool
221RenderFbArray::completeTestAndShift()
222{
223 int total = mFbTbl.size();
224 int shiftTotal = -1;
225 for (int i = 0; i < total; ++i) {
226 RenderFb * cFb = mFbTbl[i];
227 if (cFb->isComplete()) {
228 shiftTotal = i;
229 break;
230 }
231 }
232
233 if (shiftTotal == -1) {
234 // could not find completed renderFb
235 return false;
236 }
237
238 for (int i = 0; i < shiftTotal; ++i) {
239 shiftFbTbl();
240 }
241
242 return true; // found completed renderFb and already shifted
243}
