20//
21// This allocator is designed for renderPrep cache dequeue situation like following condition.
22// - Cache data file is mmapped (memory mapped by mmap()) and it's read-only
23// - This mmapped memory is accessed via CacheDequeue object
24// - Construct vector and others by this CacheAllocator and just set data address from mmapped memory.
25// - We would like to skip initialization because set address already has data and
26// we should not initialize that info (actually this memory is read-only).
27// - Created vector(and others) are accessed by read-only
28// - If we don't use CacheDequeue mode, behave as a regular allocator.
29//
30template <typename T>
31class CacheAllocator
32{
33public:
34 typedef std::false_type propagate_on_container_copy_assignment;
35 typedef std::true_type propagate_on_container_move_assignment;
36 typedef std::true_type propagate_on_container_swap;
37 typedef std::false_type is_always_equal;
38
39 using value_type = T;
40
41 CacheAllocator() noexcept :
42 mCacheDequeue(nullptr)
43 {}
44 CacheAllocator(CacheDequeue *cacheDequeue) noexcept :
45 mCacheDequeue(cacheDequeue)
46 {}
47 template <typename U> CacheAllocator(const CacheAllocator<U> &u) noexcept :
48 mCacheDequeue(u.mCacheDequeue)
49 {}
50
51 T *allocate(std::size_t n)
52 {
53 if (!mCacheDequeue) return static_cast<T *>(::operator new(sizeof(T) * n));
54 return reinterpret_cast<T *>(getMem(sizeof(T) * n));
55 }
56 void deallocate(T *p, std::size_t n) noexcept
57 {
58 if (!mCacheDequeue) ::operator delete(p);
59 }
60
61 template <typename U, typename... Args>
62 void construct(U *p, Args &&... args)
63 {
64 if (!mCacheDequeue) {
65 ::new(p) T(std::forward<Args>(args)...);
66 } else {
67 // disable construct() function when cacheDeq enable mode
68 }
69 }
70
71 size_t max_size() const noexcept
72 {
73 if (!mCacheDequeue) {
74 return std::numeric_limits<size_t>::max() / sizeof(T);
75 }
76 return mCacheDequeue->getRestSize() / sizeof(T);
77 }
78
79 std::string show() const
80 {
81 std::ostringstream ostr;
82 ostr << "CacheAllocator (addr:0x" << std::hex << (uintptr_t)this << std::dec
83 << ") {\n";
84 if (!mCacheDequeue) {
85 ostr << " mCacheDequeue:(empty)\n";
86 } else {
87 ostr << str_util::addIndent(mCacheDequeue->show()) << '\n';
88 }
89 ostr << "}";
90 return ostr.str();
91 }
92
93 CacheDequeue *getCacheDequeue() const { return mCacheDequeue; }
94
95protected:
96 CacheDequeue *mCacheDequeue;
97
98 uintptr_t getMem(size_t size) { return (uintptr_t)mCacheDequeue->skipByteData(size); }
99};

101template <typename T, typename U>
102bool operator ==(const CacheAllocator<T> &a, const CacheAllocator<U> &b)
103{
104 if (a.getCacheDequeue() == b.getCacheDequeue()) return true;
105 if (!a.getCacheDequeue() || !b.getCacheDequeue()) return false;
106 return a.getCacheDequeue()->isSameEncodedData(*(b.getCacheDequeue()));
107}

109template <typename T, typename U>

111{
112 return !(a == b);
113}
