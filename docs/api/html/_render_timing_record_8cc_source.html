117void
118RenderFrameTimingRecord::newStint()
119{
120 mRenderFrameStartTime = scene_rdl2::util::getSeconds();
121 mLastFrameUpdate = (mRenderFrameEndTime < 0.0)? 0.0: mRenderFrameStartTime - mRenderFrameEndTime;
122
123 mRenderFrameTimeBudget = 0.0;
124 mRenderFrameDuration = 0.0;
125
126 // We should not initialize mRenderPasses because we have to keep past mRenderPasses info
127 // up to at least 64 samples per tile for next checkpoint stint
128
129 mEstimatedSampleCost = 0.0; // estimated sample cost by estimation phase
130
131 mTimeBudget = 0.0; // current timebudget for renderPasses() (sec)
132 mOverheadCost = 0.0; // current overhead cost for single renderPasses() (sec)
133 mSampleCost = 0.0; // curreng best estimation for single sample per tile (sec)
134}

196void
197RenderFrameTimingRecord::updateRenderPassesTable(const unsigned newSamplesPerTile,
198 const unsigned requestedKeepSamplesPerTile)
199//
200// Keep passes at least last requestedKeepSamplesPerTile samples per tile.
201//
202{
203 if (newSamplesPerTile > requestedKeepSamplesPerTile || !mRenderPasses.size()) {
204 // We don't need old passes information or mRenderPasses is empty
205 mRenderPasses.resize(1);
206 mRenderPasses.shrink_to_fit();
207
208 } else {
209 // find last passes we need to keep
210 size_t keepTotal = 0;
211 size_t totalSamplesPerTile = newSamplesPerTile;
212 for (size_t i = 0; i < mRenderPasses.size(); ++i) {
213 keepTotal = i + 1;
214 totalSamplesPerTile += mRenderPasses[i].getSamplesPerTile();
215 if (totalSamplesPerTile >= requestedKeepSamplesPerTile) {
216 break;
217 }
218 }
219
220 // remove unused items (i.e. more than 64 samples)
221 mRenderPasses.resize(keepTotal + 1);
222 mRenderPasses.shrink_to_fit();
223
224 // shift all items in the tables and make room for current pass
225 for (size_t i = mRenderPasses.size() - 1; i > 0; --i) {
226 mRenderPasses[i] = mRenderPasses[i - 1];
227 }
228 }
229
230 mRenderPasses[0].init();
231
232#ifdef DEBUG_MSG
233 std::cerr << ">> RenderTimingRecord.cc updateRenderPassesTable() v:" << newSamplesPerTile << " (";
234 size_t total = 0;
235 for (size_t i = 0; i < mRenderPasses.size(); ++i) {
236 if (i == 0) {
237 std::cerr << newSamplesPerTile;
238 total += newSamplesPerTile;
239 } else {
240 std::cerr << ' ' << mRenderPasses[i].getSamplesPerTile();
241 total += mRenderPasses[i].getSamplesPerTile();
242 }
243 }
244 std::cerr << ") total:" << total << std::endl;
245#endif // end DEBUG_MSG
246}
