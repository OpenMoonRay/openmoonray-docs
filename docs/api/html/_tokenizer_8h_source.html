106 // This function is very c-like, because there are times when we can compute the length at compile-time (e.g. a
107 // string literal). This ensures we can do so, instead of, say, passing in a template type and computing the length
108 // here.
109 static bool isToken(char c, const char* const tokens, std::size_t tokensLength)
110 {
111 for (std::size_t i = 0; i < tokensLength; ++i) {
112 if (c == tokens[i]) {
113 return true;
114 }
115 }
116 return false;
117 }

119 template <typename TokenRecognizer>
120 static std::vector<std::string> tokenize(const std::string& s, TokenRecognizer tokenRecognizer)
121 {
122 // Speaking in terms of whitespace...
123 // first should point to the next non-whitespace character
124 // last should point to the following first whitespace character
125
126 std::vector<std::string> ret;
127 using const_iterator = std::string::const_iterator;
128 const_iterator first = std::find_if_not(s.cbegin(), s.cend(), tokenRecognizer);
129 while (first != s.cend()) {
130 const_iterator last = std::find_if(first, s.cend(), tokenRecognizer);
131 ret.emplace_back(first, last);
132 first = std::find_if_not(last, s.cend(), tokenRecognizer);
133 }
134 return ret;
135 }
