125 finline uint64_t onOffSwitchMask(const uint64_t mask64) const {
126 // This is equivalent to (mask64)? (uint64_t)0xffffffffffffffff: (uint64_t)0x0;
127 return static_cast<uint64_t>(((static_cast<int64_t>(mask64) & static_cast<int64_t>(0x8000000000000000)) |
128 (static_cast<int64_t>(0x0) - static_cast<int64_t>(mask64))) >> 63);
129 }
130
131 finline uint64_t countRightZeroBit(uint64_t mask64) const {
132 //
133 // bit scan forward : This function does not work when mask64 == 0 because bsf return undefined value.
134 //
135 uint64_t result;
136 asm volatile("bsfq %1, %0": "=r"(result): "r"(mask64));
137 return result;
138 }

141finline int
142TileExtrapolationPhaseManager::search_maskBundle2(const uint64_t activePixelMask, const uint64_t pixelSearchMaskPixId[]) const
143//
144// bundle mask total 2
145//
146{
147 for (size_t phaseId = 0; phaseId < mPhases.size(); ++phaseId) {
148 const TileExtrapolationPhase &currPhase = mPhases[phaseId];
149
150 if (activePixelMask & currPhase.mPhaseMask) {
151 uint64_t currMaskA = activePixelMask & pixelSearchMaskPixId[currPhase.mStartMaskId];
152 uint64_t currMaskB = activePixelMask & pixelSearchMaskPixId[currPhase.mStartMaskId + 1];
153
154 uint64_t switchMaskA = onOffSwitchMask(currMaskA);
155
156 uint64_t resultMask = currMaskB;
157 resultMask = currMaskA | (~switchMaskA & resultMask);
158
159 return static_cast<int>(countRightZeroBit(resultMask));
160 }
161 }
162 return -1;
163}

379 void searchActiveNearestPixel(const uint64_t activePixelMask, int extrapolatePixIdArray[64],
380 const int minX = 0, const int maxX = 8, const int minY = 0, const int maxY = 8 ) const {
381 //
382 // activePixelMask : input : 8x8pixels = 64bit, each bit represented pixel is active(1) or not(0).
383 // extrapolatePixIdArray : output : result of pixel id which need to fill
384 // minX, maxX, minY, maxY : active boundary of tile
385 //
386 // I run loop test and set following bundary activePixTotal number to switch search logic. Toshi (Oct/26/2017)
387 //
388 // Test is done like this.
389 // 1) randomly generated active pixel pattern by loop
390 // 2) for each pattern, try all of the logic which are bundle mask total from 2 to 8 and no bundle case.
391 // 3) compare result and find best(fastest) logic for that pattern.
392 // 4) try multiple times (i.e., run 2~3 multiple times) and find optimal logic for each activePixTotal
393 //
394 // Test run is done on tealpseudo (Intel(R) Xeon(R) CPU E5-2697 v3 @ 2.60GHz) at Oct/26/2017.
395 // activePixTotal : 1 : bundle 3
396 // 2~18 : bundle 2
397 // 19~64 : bundle 1
398 //
399 int activePixTotal = static_cast<int>(countBit64(activePixelMask));
400 if (activePixTotal >= 19) {
401 return searchActiveNearestPixel_maskBundle1(activePixelMask, extrapolatePixIdArray, minX, maxX, minY, maxY);
402 } else if (activePixTotal >= 2) {
403 return searchActiveNearestPixel_maskBundle2(activePixelMask, extrapolatePixIdArray, minX, maxX, minY, maxY);
404 }
405 return searchActiveNearestPixel_maskBundle3(activePixelMask, extrapolatePixIdArray, minX, maxX, minY, maxY);
406 }
