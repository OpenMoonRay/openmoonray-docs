69void
70LensDistribution::update()
71{
72 try {
73 mBokeh = mCamera->get(mBokehKey);
74
75 if (mBokeh) {
76 // Shape
77 mBokehSides = mCamera->get(mBokehSidesKey);
78 mBokehImage = mCamera->get(mBokehImageKey);
79
80 // Controls
81 mBokehAngle = scene_rdl2::math::deg2rad(mCamera->get(mBokehAngleKey));
82 mBokehWeightLocation = mCamera->get(mBokehWeightLocationKey);
83 mBokehWeightStrength = mCamera->get(mBokehWeightStrengthKey);
84
85 if (!mBokehImage.empty()) {
86 try {
87 mBokehImageDist.reset(new ImageDistribution(mBokehImage, Distribution2D::PLANAR));
88 } catch (const scene_rdl2::except::IoError&) {
89 scene_rdl2::Logger::error("Failed to open: ", mBokehImage, ". Switching to Disk mode.");
90 mBokehImage = "";
91 }
92 } else if (mBokehSides >= 3) {
93 bokehPolygonBuilder();
94 } else if (mBokehSides > 0) {
95 scene_rdl2::Logger::error("Unable to create a shape with less than 3 vertices. Switching to Disk mode.");
96 }
97 }
98 } catch (const scene_rdl2::except::KeyError&) {
99 mBokeh = false;
100 }
101}

103float
104LensDistribution::sampleLens(float &u, float &v) const
105{
106 MNRY_ASSERT(0.0f <= u && u < 1.0f);
107 MNRY_ASSERT(0.0f <= v && v < 1.0f);
108
109 if (mBokeh) {
110 // Input Coordinates - [0, 1)^2
111 float localU = u;
112 float localV = v;
113
114 if (!mBokehImage.empty()) {
115 // Image Mode
116 scene_rdl2::math::Vec2f uv;
117
118 mBokehImageDist->sample(localU, localV, 0, &uv, nullptr, TEXTURE_FILTER_NEAREST);
119
120 // Output - [-1, 1]^2
121 localU = (2.0f * uv.x) - 1.0f;
122 localV = (2.0f * uv.y) - 1.0f;
123 } else if (mBokehSides >= 3) {
124 // Polygon Mode
125 // Graphics Gems I p. 650
126 // Find the closest index and the next counter-clockwise index
127 int curIndex = std::min(static_cast<int>(localU * mBokehSides), mBokehSides);
128 int nextIndex = (curIndex + 1) % mBokehSides;
129
130 // Map u to a point in the sub-triangle
131 localU = (localU * mBokehSides) - static_cast<float>(curIndex);
132 localV = scene_rdl2::math::sqrt(localV); // Necessary to weight all portions of triangle equally
133
134 // Barycentric coordinates of the triangle
135 float a = 1.0f - localV;
136 float b = (1.0f - localU) * localV;
137 float c = (localU * localV);
138
139 // Vertices of the triangle
140 scene_rdl2::math::Vec2f A = scene_rdl2::math::Vec2f(0.0f, 0.0f);
141 scene_rdl2::math::Vec2f B = mBokehVertices[curIndex];
142 scene_rdl2::math::Vec2f C = mBokehVertices[nextIndex];
143
144 // Mapping the point to a new location within the triangle
145 // Output - [-1, 1]^2
146 localU = a * A.x + b * B.x + c * C.x;
147 localV = a * A.y + b * B.y + c * C.y;
148 } else {
149 // Disk Mode - Default. Also used if Image cannot be found or Shape cannot be generated.
150 // Output - [-1, 1]^2
151 toUnitDisk(localU, localV);
152 }
153
154 u = localU;
155 v = localV;
156
157 // For the pronounced edge effect, the sample's weight needs to be altered depending on position
158 // Find the distance of (u, v) from origin
159 float localDistance = scene_rdl2::math::sqrt(scene_rdl2::math::pow(scene_rdl2::math::abs(localU), 2) + scene_rdl2::math::pow(scene_rdl2::math::abs(localV), 2));
160
161 float weightLocation = mBokehWeightLocation; // mean
162 float weightStrength = mBokehWeightStrength; // stddev
163
164 // Weighting Calculation
165 // PDF of a normal function
166 // TODO: Edge detection for image and polygon modes
167 // TODO: Integrate to 1 so weight values do not need to be manipulated by user
168 float weight = (1.0f / (weightStrength * scene_rdl2::math::sqrt(scene_rdl2::math::sTwoPi)))
169 * scene_rdl2::math::exp((-1.0f * scene_rdl2::math::pow(localDistance - weightLocation, 2)) /
170 (2.0f * scene_rdl2::math::pow(weightStrength, 2)))
171 + 0.5f;
172
173 return weight;
174 } else {
175 // TODO: Keith, evaluate Cranley-Patterson rotations or rotational rotations
176 // If bokeh is off, just map to a unit disk
177 toUnitDisk(u, v);
178
179 return 1.0f;
180 }
181}

183void
184LensDistribution::bokehPolygonBuilder()
185{
186 float mBokehStep = scene_rdl2::math::sTwoPi / mBokehSides; // Figure out the step size
187 mBokehVertices.reset(new scene_rdl2::math::Vec2f[mBokehSides]);
188
189 // Calculate the vertices for the polygon
190 float curStep = mBokehAngle;
191 for (int i = 0; i < mBokehSides; ++i) {
192 scene_rdl2::math::sincos(curStep, &mBokehVertices[i].y, &mBokehVertices[i].x);
193 curStep += mBokehStep;
194 }
195}
