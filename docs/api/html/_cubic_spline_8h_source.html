69 template<typename CvType>
70 CvType evalCubic(const float& t,
71 const CvType& cv0, const CvType& cv1,
72 const CvType& cv2, const CvType& cv3) const
73 {
74 float w0, w1, w2, w3;
75 evalWeights(t, w0, w1, w2, w3);
76 return w0 * cv0 + w1 * cv1 + w2 * cv2 + w3 * cv3;
77 }

79 template<typename CvType>
80 CvType evalCubicDeriv(const float& t,
81 const CvType& cv0, const CvType& cv1,
82 const CvType& cv2, const CvType& cv3) const
83 {
84 float w0, w1, w2;
85 evalDerivWeights(t, w0, w1, w2);
86 return w0 * (cv1 - cv0) + w1 * (cv2 - cv1) + w2 * (cv3 - cv2);
87 }

186// CubicSpline is similar to nonperiodic cubic curves in Renderman
187// each span has 4 control points
188// one or more spans form a chain, each chain is a curve
189// one or more chains form a CubicSpline (Curves) primitive
190// To interpolate primitive attributes in different rates:
191// constant : should supply a single value for the entire Curves primitive
192// uniform : should supply a total of nCurves values for the entire Curves primitive
193// varying : should supply sum(nSpansi + 1) values
194// facevarying : should supply sum(nSpansi + 1) values
195// vertex : should supply sum(nVerticesi) values
196//
197// For example: a CubicSpline with 2 chains (nCurves = 2)
198// chain0 has 3 spans (nSpans0 = 3, nVertices0 = 3 * 3 + 1 = 10 for bezier or 3 + 3 = 6 for b-spline)
199// chain1 has 1 spans (nSpans1 = 1, nVertices1 = 3 * 1 + 1 = 4 for bezier or 1 + 3 = 4 for b-spline)
200// constant attribute should supply 1 value
201// uniform attribute should supply 2 values
202// varying/facevarying attribute should supply (3 + 1) + (1 + 1) = 6 values
203// vertex attribute should supply 10 + 4 = 14 values for bezier or 6 + 4 = 10 values for b-spline
204class CubicSplineInterpolator : public shading::Interpolator
205{
206public:
207 CubicSplineInterpolator(const shading::Attributes *attr, float time,
208 int chain, int varyingOffset, int faceVaryingOffset, float u,
209 int vertexOffset, float w0, float w1, float w2, float w3):
210 shading::Interpolator(attr, time, 0, chain, 2, mVaryingIndex, mLinearWeights,
211 2, mFaceVaryingIndex, mLinearWeights,
212 chain, 4, mCurveVertices, mCurveWeights)
213 {
214
215 mVaryingIndex[0] = varyingOffset;
216 mVaryingIndex[1] = varyingOffset + 1;
217 mFaceVaryingIndex[0] = faceVaryingOffset;
218 mFaceVaryingIndex[1] = faceVaryingOffset + 1;
219 mLinearWeights[0] = 1.0f - u;
220 mLinearWeights[1] = u;
221 for (int i = 0; i < 4; i++) {
222 mCurveVertices[i] = vertexOffset + i;
223 }
224 mCurveWeights[0] = w0;
225 mCurveWeights[1] = w1;
226 mCurveWeights[2] = w2;
227 mCurveWeights[3] = w3;
228 }
229private:
230 int mVaryingIndex[2];
231 int mFaceVaryingIndex[2];
232 float mLinearWeights[2];
233 int mCurveVertices[4];
234 float mCurveWeights[4];
235};
