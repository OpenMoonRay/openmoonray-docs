321 __forceinline void transpose4(const avxf& r0, const avxf& r1, const avxf& r2, const avxf& r3, avxf& c0, avxf& c1, avxf& c2, avxf& c3)
322 {
323 avxf l02 = unpacklo(r0,r2);
324 avxf h02 = unpackhi(r0,r2);
325 avxf l13 = unpacklo(r1,r3);
326 avxf h13 = unpackhi(r1,r3);
327 c0 = unpacklo(l02,l13);
328 c1 = unpackhi(l02,l13);
329 c2 = unpacklo(h02,h13);
330 c3 = unpackhi(h02,h13);
331 }

333 __forceinline void transpose(const avxf& r0, const avxf& r1, const avxf& r2, const avxf& r3, const avxf& r4, const avxf& r5, const avxf& r6, const avxf& r7,
334 avxf& c0, avxf& c1, avxf& c2, avxf& c3, avxf& c4, avxf& c5, avxf& c6, avxf& c7)
335 {
336 avxf h0,h1,h2,h3; transpose4(r0,r1,r2,r3,h0,h1,h2,h3);
337 avxf h4,h5,h6,h7; transpose4(r4,r5,r6,r7,h4,h5,h6,h7);
338 c0 = shuffle<0,2>(h0,h4);
339 c1 = shuffle<0,2>(h1,h5);
340 c2 = shuffle<0,2>(h2,h6);
341 c3 = shuffle<0,2>(h3,h7);
342 c4 = shuffle<1,3>(h0,h4);
343 c5 = shuffle<1,3>(h1,h5);
344 c6 = shuffle<1,3>(h2,h6);
345 c7 = shuffle<1,3>(h3,h7);
346 }

439 // Atan() + Atan2() AVX Intrinsic Emulator
440 // Intel provides an atan and atan2 intrinsic in their short vector math library.
441 // In order to achieve the same functionality, an emulator was created:
442 //
443 // https://github.com/xoolive/geodesy/blob/master/src/sse2_math.h
444 //
445 // This emulator was appropriated from this implementation and modified
446 // to achieve compatibility with our avxf type.
447

449 // https://github.com/xoolive/geodesy/blob/master/license.txt

451 #define ALIGN32_BEG
452 #define ALIGN32_END __attribute__((aligned(32)))
453
454 // Masks
455 static const ALIGN32_BEG int _mm256_cst_sign_mask[8] ALIGN32_END =
456 { static_cast<int>(0x80000000), static_cast<int>(0x80000000), static_cast<int>(0x80000000), static_cast<int>(0x80000000),
457 static_cast<int>(0x80000000), static_cast<int>(0x80000000), static_cast<int>(0x80000000), static_cast<int>(0x80000000) };
458
459 // Numerical constants
460 static const __m256 _mm256_cst_zero = _mm256_setzero_ps();
461 static const __m256 _mm256_cst_one = _mm256_set1_ps( 1.0f);
462 static const __m256 _mm256_cst_mone = _mm256_set1_ps(-1.0f);
463 static const __m256 _mm256_cst_two = _mm256_set1_ps( 2.0f);
464 static const __m256 _mm256_cst_mtwo = _mm256_set1_ps(-2.0f);
465
466 // Tangent-based numerical constants
467 static const __m256 _mm256_cst_tan3pio8 = _mm256_set1_ps(2.414213562373095f);
468 static const __m256 _mm256_cst_tanpio8 = _mm256_set1_ps(0.4142135623730950f);
469
470 // Pi-based numerical constants
471 static const __m256 _mm256_cst_pi = _mm256_set1_ps(3.14159265358979f);
472 static const __m256 _mm256_cst_mpi = _mm256_set1_ps(-3.14159265358979f);
473 static const __m256 _mm256_cst_pio2 = _mm256_set1_ps(1.5707963267948966f);
474 static const __m256 _mm256_cst_mpio2 = _mm256_set1_ps(-1.5707963267948966f);
475 static const __m256 _mm256_cst_pio4 = _mm256_set1_ps(0.7853981633974483f);
476
477 // Minimax coefficients
478 static const __m256 _mm256_cst_atancof_p0 = _mm256_set1_ps( 8.05374449538e-2f);
479 static const __m256 _mm256_cst_atancof_p1 = _mm256_set1_ps(-1.38776856032e-1f);
480 static const __m256 _mm256_cst_atancof_p2 = _mm256_set1_ps( 1.99777106478e-1f);
481 static const __m256 _mm256_cst_atancof_p3 = _mm256_set1_ps(-3.33329491539e-1f);

483 __forceinline avxf
484 atan( __m256 x)
485 {
486 // Figure out if x is negative and if so flip it
487 __m256 signbit = _mm256_and_ps(x, *(const __m256*) _mm256_cst_sign_mask); // Determine the signbit for each element
488 x = _mm256_andnot_ps(*(const __m256*) _mm256_cst_sign_mask, x); // Get the absolute value
489
490 __m256 y = _mm256_cst_zero; // y = <0, 0, 0, 0, 0, 0, 0, 0>
491
492 // Range reduction
493 __m256 x2 = _mm256_div_ps(_mm256_cst_mone, x); // x2[n] = (1/x[n])
494 __m256 x3 = _mm256_div_ps(_mm256_sub_ps(x, _mm256_cst_one),
495 _mm256_add_ps(x, _mm256_cst_one)); // x3[n] = (x[n] - 1)/(x[n] + 1)
496
497 // if (x > tan(3*pi/8) {
498 // x = x2;
499 // y = pi/2;
500 // }
501 __m256 mask = _mm256_cmp_ps(x, _mm256_cst_tan3pio8, _CMP_GT_OQ); // mask[n] = (x[n] > 2.4142)
502 x = _mm256_blendv_ps(x, x2, mask);
503 y = _mm256_blendv_ps(y, _mm256_cst_pio2, mask);
504
505 // if (x > tan(pi/8) {
506 // x = x3;
507 // y = pi/4;
508 // }
509 mask = _mm256_cmp_ps(x, _mm256_cst_tanpio8, _CMP_GT_OQ); // mask[n] = (x[n] > 0.4142)
510 x = _mm256_blendv_ps(x, x3, mask);
511 y = _mm256_blendv_ps(y, _mm256_cst_pio4, mask);
512
513 // Polynomial computation
514 __m256 z = _mm256_mul_ps(x, x); // z = x^2
515 __m256 num;
516
517#if defined (__FMA__) // L = 25, CPI = 2.5
518 num = _mm256_fmadd_ps(z, _mm256_cst_atancof_p0, _mm256_cst_atancof_p1); // num = (p0 * x^2) + p1
519 num = _mm256_fmadd_ps(num, z, _mm256_cst_atancof_p2); // num = (p0 * x^4) + (p1 * x^2) + p2
520 num = _mm256_fmadd_ps(num, z, _mm256_cst_atancof_p3); // num = (p0 * x^6) + (p1 * x^4) + (p2 * x^2) + p3
521 num = _mm256_mul_ps(num, z); // num = (p0 * x^8) + (p1 * x^6) + (p2 * x^4) + (p3 * x^2)
522 num = _mm256_fmadd_ps(num, x, x); // num = (p0 * x^9) + (p1 * x^7) + (p2 * x^5) + (p3 * x^3) + x
523#else // L = 37, CPI = 6.5
524 num = _mm256_mul_ps(_mm256_cst_atancof_p0, z); // num = p0 * x^2
525 num = _mm256_add_ps(num, _mm256_cst_atancof_p1); // num = (p0 * x^2) + p1
526 num = _mm256_mul_ps(num, z); // num = (p0 * x^4) + (p1 * x^2)
527 num = _mm256_add_ps(num, _mm256_cst_atancof_p2); // num = (p0 * x^4) + (p1 * x^2) + p2
528 num = _mm256_mul_ps(num, z); // num = (p0 * x^6) + (p1 * x^4) + (p2 * x^2)
529 num = _mm256_add_ps(num, _mm256_cst_atancof_p3); // num = (p0 * x^6) + (p1 * x^4) + (p2 * x^2) + p3
530 num = _mm256_mul_ps(num, z); // num = (p0 * x^8) + (p1 * x^6) + (p2 * x^4) + (p3 * x^2)
531 num = _mm256_mul_ps(num, x); // num = (p0 * x^9) + (p1 * x^7) + (p2 * x^5) + (p3 * x^3)
532 num = _mm256_add_ps(num, x); // num = (p0 * x^9) + (p1 * x^7) + (p2 * x^5) + (p3 * x^3) + x
533#endif
534
535 y = _mm256_add_ps(y, num); // y += num
536 y = _mm256_xor_ps(y, signbit); // y *= -1 (maybe)
537
538 return y;
539 }

541 __forceinline
542 avxf atan2(const avxf& y, const avxf& x)
543 {
544 __m256 mask = _mm256_cmp_ps(x, _mm256_cst_zero, _CMP_LT_OQ); // mask[n] = (x[n] < 0) - x-dependent mask
545 __m256 w = _mm256_blendv_ps(_mm256_cst_pi, _mm256_cst_mpi, y);
546 w = _mm256_blendv_ps(_mm256_cst_zero, w, x);
547
548 __m256 q = _mm256_div_ps(y, x); // q = y/x - Possible NaN, handled via masks
549 q = _mm256_add_ps(w, atan(q)); // q = w + atan(q)
550
551 __m256 mask2; // y-dependent mask
552
553 // atan2(-y, 0) -> -pi/2
554 mask = _mm256_cmp_ps(x, _mm256_cst_zero, _CMP_EQ_OQ); // mask[n] = (x[n] == 0)
555 mask2 = _mm256_and_ps(mask, _mm256_cmp_ps(y, _mm256_cst_zero, _CMP_LT_OQ)); // mask2[n] = (mask & (y[n] < 0))
556 q = _mm256_blendv_ps(q, _mm256_cst_mpio2, mask2);
557
558 // atan2(+y, 0) -> pi/2
559 mask2 = _mm256_and_ps(mask, _mm256_cmp_ps(y, _mm256_cst_zero, _CMP_GT_OQ)); // mask2[n] = (mask & (y[n] > 0))
560 q = _mm256_blendv_ps(q, _mm256_cst_pio2, mask2);
561
562 // atan2(0, 0) -> 0
563 mask2 = _mm256_and_ps(mask, _mm256_cmp_ps(y, _mm256_cst_zero, _CMP_EQ_OQ)); // mask2[n] = (mask & (y[n] == 0)
564 q = _mm256_blendv_ps(q, _mm256_cst_zero, mask2);
565
566 // atan2(0, -x) -> pi
567 mask = _mm256_cmp_ps(x, _mm256_cst_zero, _CMP_LT_OQ); // mask[n] = (x[n] < 0)
568 mask2 = _mm256_and_ps(mask, _mm256_cmp_ps(y, _mm256_cst_zero, _CMP_EQ_OQ)); // mask2[n] = (mask[n] & (y[n] == 0))
569 q = _mm256_blendv_ps(q, _mm256_cst_pi, mask2);
570
571 return q;
572 }
