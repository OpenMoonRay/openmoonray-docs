62void
63DecayLightFilter::update(const LightFilterMap& /*lightFilters*/,
64 const Mat4d& world2render)
65{
66 if (!mRdlLightFilter) {
67 return;
68 }
69 mFalloffNear = mRdlLightFilter->get<scene_rdl2::rdl2::Bool>(sFalloffNearKey);
70 mFalloffFar = mRdlLightFilter->get<scene_rdl2::rdl2::Bool>(sFalloffFarKey);
71 mNearStart = mRdlLightFilter->get<scene_rdl2::rdl2::Float>(sNearStartKey);
72 mNearEnd = scene_rdl2::math::max(mRdlLightFilter->get<scene_rdl2::rdl2::Float>(sNearEndKey), mNearStart);
73 mFarStart = mRdlLightFilter->get<scene_rdl2::rdl2::Float>(sFarStartKey);
74 // If there is a near falloff, we want the far falloff to start afterwards,
75 // otherwise there is a discontinuity in the light decay.
76 if (mFalloffNear) {
77 mFarStart = scene_rdl2::math::max(mFarStart, mNearEnd);
78 }
79 mFarEnd = scene_rdl2::math::max(mRdlLightFilter->get<scene_rdl2::rdl2::Float>(sFarEndKey), mFarStart);
80}

82bool
83DecayLightFilter::canIlluminate(const CanIlluminateData& data) const
84{
85 // Approximate light and shading point as two spheres. We can compute the
86 // minimum and maximum distances between any 2 points in those two spheres.
87 float distance = scene_rdl2::math::length(data.lightPosition - data.shadingPointPosition);
88 float buffer = data.lightRadius + data.shadingPointRadius;
89 float minDistance = buffer > distance ? 0.f : distance - buffer;
90 float maxDistance = distance + buffer;
91
92 bool success = true;
93
94 if (mFalloffNear) {
95 success &= maxDistance >= mNearStart;
96 }
97
98 if (mFalloffFar) {
99 success &= minDistance <= mFarEnd;
100 }
101
102 return success;
103}

106Color
107DecayLightFilter::eval(const EvalData& data) const
108{
109 float dist = data.isect->distance;
110
111 if ((mFalloffNear && dist < mNearStart) ||
112 (mFalloffFar && dist > mFarEnd)) {
113 return Color(0.0f);
114
115 } else if (mFalloffNear && dist < mNearEnd) {
116 return Color((dist - mNearStart) / (mNearEnd - mNearStart));
117
118 } else if (mFalloffFar && dist > mFarStart) {
119 return Color((mFarEnd - dist) / (mFarEnd - mFarStart));
120 }
121
122 return Color(1.0f);
123}
