56void *
57ValueContainerEnq::saveSceneObjectVL(void *ptr, const SceneObject *obj)
58{
59 if (obj) {
60 const std::string &klassName = obj->getSceneClass().getName();
61 const std::string &objName = obj->getName();
62
63 unsigned long klassNameSize = static_cast<unsigned long>(klassName.size());
64 unsigned long objNameSize = static_cast<unsigned long>(objName.size());
65
66 ptr = updatePtr(ptr, ValueContainerUtil::variableLengthEncoding(klassNameSize, ptr));
67 ptr = updatePtr(ptr, ValueContainerUtil::variableLengthEncoding(objNameSize, ptr));
68
69 if (klassNameSize) ptr = saveCharN(ptr, klassName.c_str(), klassNameSize);
70 if (objNameSize) ptr = saveCharN(ptr, objName.c_str(), objNameSize);
71
72 VALUE_CONTAINER_ENQ_DEBUG_MSG("klass:>" << klassName << "< obj:>" << objName << "<\n");
73 } else {
74 ptr = updatePtr(ptr, ValueContainerUtil::variableLengthEncoding(static_cast<unsigned long>(0), ptr));
75 ptr = updatePtr(ptr, ValueContainerUtil::variableLengthEncoding(static_cast<unsigned long>(0), ptr));
76 VALUE_CONTAINER_ENQ_DEBUG_MSG("klass:>< obj:><\n");
77 }
78 return ptr;
79}
