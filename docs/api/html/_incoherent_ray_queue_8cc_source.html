25//
26// This table was generated by calling the reference version of computeDirSortKey
27// on all directions over an octant (including applying a morton space filling
28// curve to the results). The resultant 16-bit values are stored in a 16x16 table
29// indexed by quantizing the x and z directional components of the incoming
30// normalized direction.
31//
32// See /work/rd/raas/pics/mark/ray_sorting/RaySortingPolarMortonLookupTable.tga
33// for visualizations.
34//
35// The code used to generated the directional lookup table looks like:
36//
37// const uint32_t TABLE_SIZE = 16;
38// uint32_t table[TABLE_SIZE][TABLE_SIZE];
39//
40// float scl, ofs;
41// scene_rdl2::math::getScaleOffset(0.f, float(TABLE_SIZE - 1), 0.f, 1.f, &scl, &ofs);
42//
43// for (unsigned y = 0; y < TABLE_SIZE; ++y) {
44// float dy = float(y) * scl + ofs;
45// for (unsigned x = 0; x < TABLE_SIZE; ++x) {
46// float dx = float(x) * scl + ofs;
47// float len = length(scene_rdl2::math::Vec2f(dx, dy));
48//
49// scene_rdl2::math::Vec3f dir;
50// if (len >= 1.f) {
51// dir = normalize(scene_rdl2::math::Vec3f(dx, dy, 0.f));
52// } else {
53// float dz = sqrtf(1.f - (dx * dx) - (dy * dy));
54// dir = scene_rdl2::math::Vec3f(dx, dy, dz);
55// }
56//
57// // see computeDirSortKey(const scene_rdl2::math::Vec3f &dir) below.
58// uint32_t dirSortKey = computeReferenceDirSortKey(dir);

99__forceinline uint32_t
100MAYBE_UNUSED mortonEncode2D(uint32_t x, uint32_t y)
101{
102 MNRY_ASSERT(x <= 0xffff);
103 MNRY_ASSERT(y <= 0xffff);
104
105 uint32_t res = (y << 16) | x;
106 uint32_t tmp;
107
108 tmp = (res ^ (res >> 8)) & 0x0000ff00; res = res ^ tmp ^ (tmp << 8);
109 tmp = (res ^ (res >> 4)) & 0x00f000f0; res = res ^ tmp ^ (tmp << 4);
110 tmp = (res ^ (res >> 2)) & 0x0c0c0c0c; res = res ^ tmp ^ (tmp << 2);
111 tmp = (res ^ (res >> 1)) & 0x22222222; res = res ^ tmp ^ (tmp << 1);
112
113 return res;
114}

116__forceinline uint32_t
117MAYBE_UNUSED getOctant(const scene_rdl2::math::Vec3f &dir)
118{
119 uint32_t negX = dir.x < 0.f ? 1 : 0;
120 uint32_t negY = dir.y < 0.f ? 2 : 0;
121 uint32_t negZ = dir.z < 0.f ? 4 : 0;
122 return negX | negY | negZ;
123}

169__forceinline uint32_t
170MAYBE_UNUSED computePosSortKey(const scene_rdl2::math::Vec3f &p)
171{
172 MNRY_ASSERT(p.x >= 0.f && p.x < 256.f);
173 uint32_t px = uint32_t(p.x) >> 2;
174 MNRY_ASSERT(px < 64);
175
176// MNRY_ASSERT(p.y >= 0.f && p.y < 256.f);
177// uint32_t py = uint32_t(p.y) >> 2;
178// MNRY_ASSERT(py < 64);
179
180 MNRY_ASSERT(p.z >= 0.f && p.z < 256.f);
181 uint32_t pz = uint32_t(p.z) >> 2;
182 MNRY_ASSERT(pz < 64);
183
184 uint32_t morton = mortonEncode2D(px, pz);
185
186 MNRY_ASSERT(morton <= 4095);
187
188 return morton;

224void
225IncoherentRayQueue::addEntries(mcrt_common::ThreadLocalState *tls,
226 unsigned numEntries, RayState **entries,
227 scene_rdl2::alloc::Arena *arena)
228{
229 // Convert all incoming RayStates into SortedRayStates (in place) to make
230 // room for sort keys.
231 WrappedRayState *wrappedRs = reinterpret_cast<WrappedRayState *>(entries);
232
233#ifdef RAY_SORTING
234
235 EXCL_ACCUMULATOR_PROFILE(tls, EXCL_ACCUM_RAY_SORT_KEY_GEN);
236
237 RayState *baseRayState = indexToRayState(0);
238 for (unsigned i = 0; i < numEntries; ++i) {
239
240 //
241 // Compute sort key...
242 //
243
244 RayState *rs = entries[i];
245 wrappedRs[i].mSortedRs.mRsIdx = shading::RayStateIndex(rs - baseRayState);
246
247 const scene_rdl2::math::Vec3f &pos = rs->mRay.getOrigin();
248 const scene_rdl2::math::Vec3f &dir = rs->mRay.getDirection();
249
250 uint32_t octant = getOctant(dir) << 28;
251 uint32_t p = computePosSortKey(pos * sPositionScale + sPositionOffset) << 16;
252 uint32_t d = computeDirSortKey(dir) << 0;
253
254 MNRY_ASSERT((octant & p) == 0);
255 MNRY_ASSERT((octant & d) == 0);
256 MNRY_ASSERT((p & d) == 0);
257
258 //
259 // Octant is located in the up most bits since that's the primary criteria
260 // the embree streaming implementation uses internally to sort rays.
261 // Next we have a choice between position and direction. Position seems
262 // to be the most logical choice since a BVH is a spatial data structure
263 // after all. Direction then goes in the lower bits. The overall
264 // structure looks like:
265 //
266 // bit 31 free
267 // bits 28-30 octant
268 // bits 16-27 position (morton xz)
269 // bits 0-15 direction (morton polar)
270 //
271 wrappedRs[i].mSortedRs.mSortKey = octant | p | d;
272 }
273
274#endif
275
276 BaseIncoherentRayQueue::addEntries(tls, numEntries, wrappedRs, arena);
277}

279// Split the world extents into 256 blocks along each axis. Note that these likely aren't
280// perfectly cubic in shape.
281void
282IncoherentRayQueue::setBounds(const scene_rdl2::math::BBox3f &bbox)
283{
284 MNRY_ASSERT(bbox.lower.x <= bbox.upper.x);
285 MNRY_ASSERT(bbox.lower.y <= bbox.upper.y);
286 MNRY_ASSERT(bbox.lower.z <= bbox.upper.z);
287
288 scene_rdl2::math::Vec3f extents = bbox.upper - bbox.lower;
289 float maxExtent = reduce_max(extents);
290 float padding = std::max(maxExtent * 0.05f, 0.001f);
291 float halfExt = maxExtent * 0.5f + padding;
292 scene_rdl2::math::Vec3f midpoint = center(bbox);
293
294 scene_rdl2::math::getScaleOffset(midpoint.x - halfExt, midpoint.x + halfExt, 0.f, 255.99f,
295 &sPositionScale.x, &sPositionOffset.x);
296 scene_rdl2::math::getScaleOffset(midpoint.y - halfExt, midpoint.y + halfExt, 0.f, 255.99f,
297 &sPositionScale.y, &sPositionOffset.y);
298 scene_rdl2::math::getScaleOffset(midpoint.z - halfExt, midpoint.z + halfExt, 0.f, 255.99f,
299 &sPositionScale.z, &sPositionOffset.z);
300
301 //std::cout << "bounds = " << bbox << std::endl;
302}
