99void
100MeasuredMaterial::shade(const scene_rdl2::rdl2::Material* self, moonray::shading::TLState *tls,
101 const State& state, BsdfBuilder &bsdfBuilder)
102{
103 moonray::shading::Bsdf *bsdf = const_cast<moonray::shading::Bsdf*>(bsdfBuilder.getBsdf());
104
105 const MeasuredMaterial* me = static_cast<const MeasuredMaterial*>(self);
106
107 alloc::Arena *arena = getArena(tls);
108
109 // Evaluate normal map if any
110 Vec3f N = evalNormal(me, attrInputNormal, attrInputNormalDial,
111 attrInputNormalSpace, tls, state);
112 Vec3f T = normalize(state.getdPds());
113 scene_rdl2::math::ReferenceFrame frame(N, T);
114 Vec2f dir = me->get(attrAnisotropicDirection);
115 Vec3f anisotropicDirection =
116 frame.localToGlobal(normalize(Vec3f(dir.x, dir.y, 0.0f)));
117
118 // Diffuse
119 if (me->mBsdfTableDiffuse != nullptr) {
120 Color color = evalColorComponent(me, attrDiffuse,
121 attrDiffuseFactor, attrDiffuseColor, tls, state);
122 if (!isBlack(color)) {
123 auto lobe = arena->allocWithArgs<moonray::shading::MeasuredBsdfLobe>(N, anisotropicDirection,
124 me->mBsdfTableDiffuse,
125 evalFloat(me, attrDiffuseHueShift, tls, state),
126 evalFloat(me, attrDiffuseSaturation, tls, state));
127 lobe->setScale(color);
128 bsdf->addLobe(lobe);
129 }
130 }
131
132 // Specular
133 if (me->mBsdfTableSpecular != nullptr &&
134 (!state.isCausticPath() || me->get(attrCastsCaustics))) {
135
136 Color color = evalColorComponent(me, attrSpecular,
137 attrSpecularFactor, attrSpecularColor, tls, state);
138 if (!isBlack(color)) {
139 auto lobe = arena->allocWithArgs<moonray::shading::MeasuredBsdfLobe>(N, anisotropicDirection,
140 me->mBsdfTableSpecular,
141 evalFloat(me, attrSpecularHueShift, tls, state),
142 evalFloat(me, attrSpecularSaturation, tls, state));
143 lobe->setScale(color);
144 bsdf->addLobe(lobe);
145 }
146 }
147}
