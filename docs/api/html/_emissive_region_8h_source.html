23// EmissiveRegion is a primitive with emissive volume assignment
24// The integrator will integrate the emission contribution from EmissiveRegion
25// through importance sampling like regular light source so that we can
26// use emissive volume to illuminate the scene with better converge rate
27class EmissiveRegion
28{
29public:
30 // This constructor is used by emissive regions that do
31 // not point to instanced volume primitives. So it can just use
32 // the transform from the emission distribution unmodified.
33 EmissiveRegion(const Primitive* primitive,
34 const scene_rdl2::rdl2::VolumeShader* volumeShader,
35 int volumeId, int visibilityMask):
36 mVolumeId(volumeId),
37 mVisibilityMask(visibilityMask),
38 mEmissionDistribution(primitive->computeEmissionDistribution(volumeShader)),
39 mTransform(mEmissionDistribution->getTransform())
40 {}
41 // This constructor is used by emissive regions that point
42 // to shared volume primitives. It needs to use a distToRender transform
43 // that is different than the one stored in the emission distribution.
44 EmissiveRegion(int volumeId, int visibilityMask,
45 std::shared_ptr<EmissionDistribution> emissionDistribution,
46 const EmissionDistribution::Transform &transform):
47 mVolumeId(volumeId),
48 mVisibilityMask(visibilityMask),
49 mEmissionDistribution(emissionDistribution),
50 mTransform(transform)
51 {
52 }
53
54 int size() const
55 {
56 return mEmissionDistribution->count();
57 }
58
59 void sample(const Vec3f& p, float u1, float u2, float u3,
60 Vec3f& wi, float& pdfWi, float& tEnd, float time) const
61 {
62 mEmissionDistribution->sample(mTransform, p, u1, u2, u3, wi, pdfWi, tEnd, time);
63 }
64
65 float pdf(const Vec3f& p, const Vec3f& wi, float& tEnd, float time) const
66 {
67 return mEmissionDistribution->pdf(mTransform, p, wi, tEnd, time);
68 }
69
70 Vec3f getEquiAngularPivot(float u1, float u2, float u3, float time) const
71 {
72 return mEmissionDistribution->sample(mTransform, u1, u2, u3, time);
73 }
74
75 bool canIlluminateSSS() const
76 {
77 // treat bssrdf as a special case of diffuse reflection
78 return (mVisibilityMask & scene_rdl2::rdl2::DIFFUSE_REFLECTION) != 0;
79 }
80
81 bool canIlluminateVolume() const
82 {
83 // PHASE_REFLECTION and PHASE_TRANSMISSION always on and off in pair
84 // so we don't need to verify both
85 return (mVisibilityMask & scene_rdl2::rdl2::PHASE_REFLECTION) != 0;
86 }
87
88 int mVolumeId;
89 int mVisibilityMask;
90 std::shared_ptr<EmissionDistribution> mEmissionDistribution;
91 EmissionDistribution::Transform mTransform;
92};
