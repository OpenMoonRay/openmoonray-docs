49void
50UsdPrimvarReader_vector::update()
51{
52 if (hasChanged(attrVarName)) {
53 mOptionalAttributes.clear();
54 const std::string& varname = get(attrVarName);
55 // We handle these special case "varname" values:
56 // "dpds"
57 // "dpdt"
58 // "dnds"
59 // "dndt"
60 // "velocity"
61 // "acceleration"
62 // "motion"
63 if (varname == "dpds") {
64 mIspc.mAttributeMapType = ispc::PRIMVAR_MAP_TYPE_DPDS;
65 } else if (varname == "dpdt") {
66 mIspc.mAttributeMapType = ispc::PRIMVAR_MAP_TYPE_DPDT;
67 } else if (varname == "dnds") {
68 mIspc.mAttributeMapType = ispc::PRIMVAR_MAP_TYPE_DNDS;
69 } else if (varname == "dndt") {
70 mIspc.mAttributeMapType = ispc::PRIMVAR_MAP_TYPE_DNDT;
71 } else if (varname == "velocity") {
72 mIspc.mAttributeMapType = ispc::PRIMVAR_MAP_TYPE_VELOCITY;
73 mIspc.mPrimitiveAttributeIndex = moonray::shading::StandardAttributes::sVelocity;
74 mOptionalAttributes.push_back(mIspc.mPrimitiveAttributeIndex);
75 } else if (varname == "acceleration") {
76 mIspc.mAttributeMapType = ispc::PRIMVAR_MAP_TYPE_ACCELERATION;
77 mIspc.mPrimitiveAttributeIndex = moonray::shading::StandardAttributes::sAcceleration;
78 mOptionalAttributes.push_back(mIspc.mPrimitiveAttributeIndex);
79 } else if (varname == "motion") {
80 mIspc.mAttributeMapType = ispc::PRIMVAR_MAP_TYPE_MOTIONVEC;
81 mIspc.mPrimitiveAttributeIndex = moonray::shading::StandardAttributes::sMotion;
82 mOptionalAttributes.push_back(mIspc.mPrimitiveAttributeIndex);
83 } else {
84 // Not a special case so create key from the
85 // "varname" value
86 mIspc.mAttributeMapType = ispc::PRIMVAR_MAP_TYPE_PRIMITIVE_ATTRIBUTE;
87 moonray::shading::TypedAttributeKey<Vec3f> attributeKey(varname);
88 mIspc.mPrimitiveAttributeIndex = attributeKey;
89 mOptionalAttributes.push_back(attributeKey);
90 }
91 ::moonshine::primvar::createLogEvent("vector",
92 varname,
93 mIspc.mMissingAttributeEvent,
94 mLogEventRegistry);
95 }
96}

98void
99UsdPrimvarReader_vector::sample(const scene_rdl2::rdl2::Map *self, moonray::shading::TLState *tls,
100 const moonray::shading::State &state, Color *sample)
101{
102 const UsdPrimvarReader_vector* me = static_cast<const UsdPrimvarReader_vector*>(self);
103
104 const int attributeMapType = me->mIspc.mAttributeMapType;
105 if (attributeMapType == ispc::PRIMVAR_MAP_TYPE_DPDS) {
106 const Vec3f& dpds = state.getdPds();
107 sample->r = dpds.x;
108 sample->g = dpds.y;
109 sample->b = dpds.z;
110 } else if (attributeMapType == ispc::PRIMVAR_MAP_TYPE_DPDT) {
111 const Vec3f& dpdt = state.getdPdt();
112 sample->r = dpdt.x;
113 sample->g = dpdt.y;
114 sample->b = dpdt.z;
115 } else if (attributeMapType == ispc::PRIMVAR_MAP_TYPE_DNDS) {
116 const Vec3f& dnds = state.getdNds();
117 sample->r = dnds.x;
118 sample->g = dnds.y;
119 sample->b = dnds.z;
120 } else if (attributeMapType == ispc::PRIMVAR_MAP_TYPE_DNDT) {
121 const Vec3f& dndt = state.getdNdt();
122 sample->r = dndt.x;
123 sample->g = dndt.y;
124 sample->b = dndt.z;
125 } else {
126 int key = me->mIspc.mPrimitiveAttributeIndex;
127 if (state.isProvided(key)) {
128 const Vec3f& p = state.getAttribute(moonray::shading::TypedAttributeKey<Vec3f>(key));
129 sample->r = p.x;
130 sample->g = p.y;
131 sample->b = p.z;
132 } else {
133 // the primitive attribute is unavailable, use fallback parameter
134 const Vec3f& v = evalVec3f(me, attrFallback, tls, state);
135 sample->r = v.x;
136 sample->g = v.y;
137 sample->b = v.z;
138 if (me->get(attrWarnWhenUnavailable)) {
139 moonray::shading::logEvent(me, tls, me->mIspc.mMissingAttributeEvent);
140 }
141 }
142 }
143}
