19/// This class serves the functionality of tracking status of volume in 3d space
20/// (ie. which volume regions a point or an interval is belong to)
21/// we need to consider the scenario where a point or an interval can belong to
22/// multiple volume regions (fire inside smoke, for example)
23class VolumeRegions
24{
25public:
26 VolumeRegions() : mBitMask({0}) {}
27
28 size_t getMaxRegionsCount() const { return 32 * mBitMask.size(); }
29
30 void reset() {
31 memset(mBitMask.data(), 0, sizeof(uint32_t) * mBitMask.size());
32 }
33
34 bool isOn(int volumeId) const {
35 MNRY_ASSERT(volumeId < (int)getMaxRegionsCount());
36 return (mBitMask[volumeId >> 5]) & (1 << (volumeId & 31));
37 }
38
39 bool isVacuum() const {
40 for (uint32_t i = 0; i < mBitMask.size(); ++i) {
41 if (mBitMask[i] != 0) {
42 return false;
43 }
44 }
45 return true;
46 }
47
48 void turnOn(int volumeId) {
49 MNRY_ASSERT(volumeId < (int)getMaxRegionsCount());
50 mBitMask[volumeId >> 5] |= 1 << (volumeId & 31);
51 }
52
53 void turnOff(int volumeId) {
54 MNRY_ASSERT(volumeId < (int)getMaxRegionsCount());
55 mBitMask[volumeId >> 5] &= ~(1 << (volumeId & 31));
56 }
57
58 void toggle(int volumeId) {
59 MNRY_ASSERT(volumeId < (int)getMaxRegionsCount());
60 mBitMask[volumeId >> 5] ^= 1 << (volumeId & 31);
61 }
62
63 int getRegionsCount() const {
64 int regionsCount = 0;
65 for (uint32_t i = 0; i < mBitMask.size(); ++i) {
66 regionsCount += scene_rdl2::util::countOnBits(mBitMask[i]);
67 }
68 return regionsCount;
69 }
70
71 int getVolumeIds(int* volumeIds) const {
72 // locate all 1 bits in bit mask
73 int regionsCount = 0;
74 for (uint32_t i = 0; i < mBitMask.size(); ++i) {
75 uint32_t v = mBitMask[i];
76 uint32_t baseIdx = (i << 5);
77 while (v != 0) {
78 // Note: __builtin_ctz(0) is undefined but that's ok since we're
79 // handling that case.
80 uint32_t idx = __builtin_ctz(v); // index of lowest 1-bit
81 volumeIds[regionsCount++] = baseIdx + idx;
82 v &= v - 1; // set lowest 1-bit to zero
83 }
84 }
85 return regionsCount;
86 }
87
88private:
89 // TODO 32 * 16 = 512 volume regions are probably not sufficient for extreme
90 // production case, we can extend it in the future.
91 std::array<uint32_t, 16> mBitMask;
92};
