143class StatisticsPrimaryRayTotal
144{
145public:
146 StatisticsPrimaryRayTotal() { reset(); }
147
148 void reset() {
149 mMsgTotal = 0;
150 mUpdateTotal = 0;
151 mTotalPrimarySample = 0;
152 mSampleGroundTotal = 0.0;
153 }
154
155 inline void
156 update(const size_t primarySample, const size_t msgInterval, const size_t msgTotal) {
157 if (mMsgTotal >= msgTotal) return;
158
159 mTotalPrimarySample += primarySample;
160 mUpdateTotal++;
161 if ((mUpdateTotal % msgInterval) == 0) {
162 double sample = (double)mTotalPrimarySample / (double)msgInterval;
163
164 mSampleGroundTotal += sample;
165 double sampleAverage = mSampleGroundTotal / (double)(mMsgTotal + 1);
166
167# define _F15_5 std::setw(15) << std::fixed << std::setprecision(5)
168
169 std::ostringstream ostr;
170 ostr << mMsgTotal << " " << _F15_5 << sample << " ave:" << _F15_5 << sampleAverage << " primayRayTotal";
171 std::cout << ostr.str().c_str() << std::endl;
172 MOONRAY_LOG_INFO(ostr.str().c_str());
173
174 mUpdateTotal = 0;
175 mTotalPrimarySample = 0;
176 mMsgTotal++;
177 }
178 }
179
180protected:
181 size_t mMsgTotal;
182 size_t mUpdateTotal;
183 size_t mTotalPrimarySample;
184 double mSampleGroundTotal;
185};

187class StatisticsBusyMessage
188{
189public:
190 StatisticsBusyMessage() : mOnMessageCounter(0), mOnIdleCounter(0) {}
191
192 void onMessageUpdate() { mOnMessageCounter++; };
193 void onIdleUpdate() {
194 mOnIdleCounter++;
195 if (mOnIdleCounter < mOnMessageCounter) {
196 int c = mOnMessageCounter - mOnIdleCounter;
197
198 std::ostringstream ostr;
199 ostr << ">>> busy onMessage() execution:" << c;
200 MOONRAY_LOG_INFO(ostr.str().c_str());
201 }
202 mOnIdleCounter = mOnMessageCounter; // reset onIdle counter as onMessage counter here
203 }
204
205protected:
206 size_t mOnMessageCounter;
207 size_t mOnIdleCounter;
208};
