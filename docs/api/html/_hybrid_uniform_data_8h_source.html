19// Class used to automatically check data layout across C++ and ISPC hybrid
20// uniform data types. Typical usage is through the HUD_VALIDATOR() macro below.
21template <typename CppHudType, uint32_t (*ispcHudValidation)(bool)>
22class HudValidator
23{
24public:
25 HudValidator() {
26 // Change this to true if you want to see all the member offsets
27 static const bool verbose = false;
28 uint32_t cppCrc = CppHudType::hudValidation(verbose);
29 uint32_t ispcCrc = ispcHudValidation(verbose);
30 if (cppCrc != ispcCrc) {
31 printf("Fatal error: Hybrid uniform data layout mismatch:\n");
32 cppCrc = CppHudType::hudValidation(true);
33 ispcCrc = ispcHudValidation(true);
34 abort();
35 }
36 }
37};

40// Macro used to create a static instance of the HudValidator, which runtime
41// initialization will automatically trigger validation that the data layout
42// matches across C++ and ISPC hybrid uniform data types.
43//
44// The given C++ Type must have a static method with the following signature:
45// static uint32_t hudValidation(bool verbose);
46// which uses the macros defined in <HybridUniformData.hh>:
47// HUD_BEGIN_VALIDATION();
48// HUD_VALIDATE();
49// HUD_END_VALIDATION;
50//
51// The corresponding ispc::Type must have an identical function with the following
52// signature:
53// export uniform uint32_t
54// Type_hudValidation(uniform bool verbose)
55//
56// Usage examples can be found throughout the moonbase codebase.
57#define HUD_VALIDATOR(Type) \
58 static scene_rdl2::util::HudValidator<Type, ispc::Type##_hudValidation> s##Type##HudValidator
59

61// Used to validate at compile-time that a C++ type and its corresponding ISPC
62// uniform type have a matching memory layout. This is needed to validate simple
63// templated types as the above runtime crc-based system does not work for
64// templated types.
65#define HUD_VALIDATE_STATIC(Type, Member) \
66 MNRY_STATIC_ASSERT(offsetof(Type, Member) == \
67 offsetof(ispc::Type, Member))
