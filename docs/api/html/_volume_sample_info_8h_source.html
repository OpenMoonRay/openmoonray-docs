18/// This class provides information to sample field value for one particular
19/// combination of volume/ray/primitive. It is filled out during volume BVH
20/// traversal stage and stored in ThreadLocalStorage so that all the volume
21/// sample/integration work for one particular ray can reuse the same info

28 float featureSize, bool canCastShadow, bool isVDB) {

32 mSampleRayDir = sampleRayDir;
33 mFeatureSize = featureSize;
34 mCanCastShadow = canCastShadow;

36 }
