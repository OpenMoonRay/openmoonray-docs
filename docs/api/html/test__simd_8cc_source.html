18float
19avxAtanError()
20{
21 __m256 avx_results;
22 float* avx_result = (float*)&avx_results;
23
24 float libm_result;
25 float rel_error;
26 float abs_error;
27 float max_rel_error = 0.0f;
28
29 for (float x = -5.0f; x < 5.0f; x += 0.001f) {
30 libm_result = atan(x);
31 avx_results = atan(_mm256_set1_ps(x));
32
33 if (libm_result == 0.0f) continue;
34 abs_error = fabs(avx_result[(rand() % 8)] - libm_result);
35 rel_error = fabs(abs_error / libm_result);
36 max_rel_error = std::max(max_rel_error, rel_error);
37 }
38
39 return max_rel_error;
40}
