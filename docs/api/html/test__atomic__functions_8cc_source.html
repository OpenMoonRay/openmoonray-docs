19template <typename RandomIterator, typename F, typename... Args>
20std::vector<std::thread> distributeThreads(F f,
21 RandomIterator first,
22 RandomIterator last,
23 std::size_t nthreads,
24 Args&&... args)
25{
26 std::size_t nelements = std::distance(first, last);
27
28 std::vector<std::thread> threads;
29 threads.reserve(nthreads);
30
31 while (nelements > 0) {
32 const auto nt = nelements/nthreads;
33 threads.emplace_back(f, first, first + nt, std::forward<Args>(args)...);
34 first += nt;
35 --nthreads;
36 nelements -= nt;
37 }
38
39 MNRY_ASSERT(nthreads == 0);
40 MNRY_ASSERT(first == last);
41
42 return threads;
43}

105template <typename T>
106void runTestAdd()
107{
108 using Container = std::vector<T>;
109
110 constexpr std::size_t nvalues = 2'999;
111 Container v(nvalues);
112 std::iota(v.begin(), v.end(), T(1));
113
114 std::mt19937 rng(42);
115 std::shuffle(v.begin(), v.end(), rng);
116
117 // Triangle number summation
118 const T expectedValue = (T(nvalues) * (T(nvalues) + T(1))) / T(2);
119
120 T sumValue = 0;
121
122 const int nthreads = std::max(8u, std::thread::hardware_concurrency());
123 std::atomic<bool> ready(false);
124 auto f = [&ready, &sumValue](auto first, auto last) { findSum(ready, sumValue, first, last); };
125 auto threads = distributeThreads(f, v.cbegin(), v.cend(), nthreads);
126 ready.store(true, std::memory_order_relaxed);
127 for (auto& t: threads) {
128 t.join();
129 }
130
131 CPPUNIT_ASSERT(expectedValue == sumValue);
132}

186void runTestClosest()
187{
188 struct alignas(16) Array
189 {
190 explicit Array(float distance) noexcept
191 : values{distance * 11.0f, distance * 9.0f, distance * 5.0f, distance}
192 {
193 }
194
195 float values[4];
196 };
197
198 using Container = std::vector<Array>;
199
200 constexpr std::size_t nvalues = 10'007;
201 Container v;
202 v.reserve(nvalues);
203 for (std::size_t i = 0; i < nvalues; ++i) {
204 // Let's not start at 0: it makes all of our values zero the way we have defined our constructor, which is no
205 // fun.
206 v.push_back(Array(i + 1));
207 }
208
209 Array result(50);
210
211 // Make sure we start with a value who has a depth value that is somewhere in the middle.
212 CPPUNIT_ASSERT(result.values[3] > v.front().values[3]);
213 CPPUNIT_ASSERT(result.values[3] < v.back().values[3]);
214
215 std::mt19937 rng(42);
216 std::shuffle(v.begin(), v.end(), rng);
217
218 const int nthreads = std::max(8u, std::thread::hardware_concurrency());
219 std::atomic<bool> ready(false);
220 auto f = [&ready, &result](auto first, auto last) { findClosest(ready, result, first, last); };
221 auto threads = distributeThreads(f, v.cbegin(), v.cend(), nthreads);
222 ready.store(true, std::memory_order_relaxed);
223 for (auto& t: threads) {
224 t.join();
225 }
226
227 for (std::size_t i = 0; i < 4; ++i) {
228 CPPUNIT_ASSERT_EQUAL(result.values[0], 11.0f);
229 CPPUNIT_ASSERT_EQUAL(result.values[1], 9.0f);
230 CPPUNIT_ASSERT_EQUAL(result.values[2], 5.0f);
231 CPPUNIT_ASSERT_EQUAL(result.values[3], 1.0f);
232 }
233}
