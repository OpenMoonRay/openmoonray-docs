56// The following are a set of macros that handle hybrid varying data layout
57// validation across C++ / ISPC types. It uses a CRC based on the offset of
58// each validated data member within the type as well as the size of the
59// type. See the HVD_VALIDATOR() macro in <HybridVaryingData.h> for more details.
60
61#define HVD_UPDATE_CRC() \
62 crc = crc ^ (ofs << 13); \
63 crc = (crc >> 1) | ((crc & 0x1) << 31)
64
65#ifdef ISPC
66
67 #define HVD_BEGIN_VALIDATION(type_name, vlen) \
68 uniform uint32_t crc = 0xffaaf0af; \
69 uniform uint32_t ofs = 0; \
70 if (verbose) print(#type_name " (ISPC):\n")
71
72 #define HVD_VALIDATE(type_name, member_name) \
73 ofs = (uniform uint32_t)((uniform intptr_t)(&((((varying type_name *uniform)(0))->member_name)))); \
74 if (verbose) { \
75 print(" " #member_name ": % / %\n", ofs, sizeof(varying type_name)); \
76 } \
77 ofs += sizeof(varying type_name); \
78 HVD_UPDATE_CRC()
79
80 #define HVD_END_VALIDATION \
81 if (verbose) print(" CRC = %\n", crc); \
82 return crc
83
84#else
85
86 #define HVD_BEGIN_VALIDATION(type_name, vlen) \
87 uint32_t crc = 0xffaaf0af; \
88 uint32_t ofs = 0; \
89 uint32_t numLanes = vlen; \
90 if (verbose) printf(#type_name " (C++):\n")
91
92 #define HVD_VALIDATE(type_name, member_name) \
93 ofs = (uint32_t)((intptr_t)(&((((type_name *)(0))->member_name)))); \
94 ofs *= numLanes; \
95 if (verbose) { \
96 printf(" " #member_name ": %d / %d\n", \
97 (int)ofs, (int)(sizeof(type_name) * numLanes)); \
98 } \
99 ofs += sizeof(type_name) * numLanes; \
100 HVD_UPDATE_CRC()
101
102 #define HVD_END_VALIDATION \
103 if (verbose) printf(" CRC = %u\n", crc); \
104 return crc
105
106#endif
