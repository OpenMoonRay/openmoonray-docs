61inline bool
62UpdateSentinel::shouldUpdate(const unsigned endSampleIdx)
63{
64 if (!mAdjustUpdateTimingCondition) {
65 return true; // If adjust update timing logic condition is disable, we always evaluate adaptiveTree
66 }
67
68 auto itr = std::lower_bound(mAdaptiveIterationPixSampleIdTable.begin(),
69 mAdaptiveIterationPixSampleIdTable.end(),
70 endSampleIdx);
71 if (*itr != endSampleIdx) {
72 return false; // endSampleId is not a boundary of adaptiveTree update sample number interval
73 }
74 return true;
75}
