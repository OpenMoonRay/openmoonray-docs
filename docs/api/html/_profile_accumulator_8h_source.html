8// Functionality for profiling the render portion of the frame. There are some
9// assumptions which make it unsuitable for the update portion. These are:
10//
11// 1. All threads are running flat out.

14// This allows us to divide "thread wall clock time" by the thread count

123//
124// We provide 2 separate parameters here since when initializing the global
125// TLS pool, we create 2 extra TLS objects also initTLSPhase2
126// (see ThreadLocalState.cc). One for use by the main thread, which is important
127// during the scene building phase of a frame. The other is a special GUI thread
128// TLS to allow picking, and scene intersections.
129//
130// If code which uses scoped accumulators is invoked from either of the 2 above
131// thread, then we'd write past the end of our ThreadLocalAccumulator
132// array if we only allocated an array sized for numThreads. Therefore to
133// avoid this issue, we allow the number of TLS objects to be allocated separately.
134// These extra TLS objects (over the number of render threads) are ignored when
135// tallying up the profiling stats.

137// The root accumulator is returned.
138//

182// Private implementation:
183
184// Per thread data for a particular accumulator, doesn't support nesting.
185struct CACHE_ALIGN ThreadLocalAccumulator
186{
187 ThreadLocalAccumulator()
188 {
189 reset();
190 }
191
192 void reset()
193 {
194 // This function clears the entire cache line!!
195 MNRY_STATIC_ASSERT(alignof(ThreadLocalAccumulator) == CACHE_LINE_SIZE);
196 MNRY_STATIC_ASSERT(sizeof(*this) == CACHE_LINE_SIZE);
197 MNRY_ASSERT(gNumAccumulatorsActive == 0);
198 memset(this, 0, CACHE_LINE_SIZE);
199 }
200
201 __forceinline void start()
202 {
203 // If this assert triggers, this accumulator is probably nested.
204 MNRY_ASSERT(!mTimerActive);
205
206 MNRY_DURING_ASSERTS(++gNumAccumulatorsActive);
207 mLastStartTime = getProfileAccumulatorTicks();
208 ++mTotalCallCount;
209 mTimerActive = true;
210 }
211
212 __forceinline void stop()
213 {
214 // If this assert triggers, this accumulator is probably nested.
215 MNRY_ASSERT(mTimerActive);
216
217 mTimerActive = false;
218 uint64_t endTime = getProfileAccumulatorTicks();
219 MNRY_ASSERT(endTime >= mLastStartTime);
220 mTotalTime += endTime - mLastStartTime;
221 MNRY_ASSERT(--gNumAccumulatorsActive >= 0);
222 }
223
224 __forceinline bool canStart() const
225 {
226 return !mTimerActive;
227 }
228
229 __forceinline bool canStop() const
230 {
231 return mTimerActive;
232 }
233
234 uint64_t mLastStartTime;
235 uint64_t mTotalTime;
236 unsigned mTotalCallCount;
237 bool mTimerActive;
238};
239
240// Ensure that the derived class also fits in a cache line!
241MNRY_STATIC_ASSERT(sizeof(ThreadLocalAccumulator) <= CACHE_LINE_SIZE);

260//
261// Helper class which adds RAII functionality.
262//
263class ScopedAccumulator
264{
265public:
266 __forceinline ScopedAccumulator(unsigned threadIdx, Accumulator *acc) :
267 mTLAccumulator(nullptr)
268 {
269 if (gAccumulatorsActive) {
270 mTLAccumulator = &MNRY_VERIFY(acc)->mThreadLocal[threadIdx];
271 mTLAccumulator->start();
272 }
273 }
274
275 // The startPaused parameter is not used directly. It's here so we can
276 // differentiate between the different constructors.
277 __forceinline ScopedAccumulator(unsigned threadIdx, Accumulator *acc, bool startPaused) :
278 mTLAccumulator(nullptr)
279 {
280 if (gAccumulatorsActive) {
281 mTLAccumulator = &MNRY_VERIFY(acc)->mThreadLocal[threadIdx];
282 }
283 }
284
285 __forceinline ~ScopedAccumulator()
286 {
287 // The ThreadLocalAccumulator may be in a stopped state already. We only
288 // need to stop it here if it's not.
289 if (mTLAccumulator && mTLAccumulator->canStop()) {
290 mTLAccumulator->stop();
291 }
292 }
293
294 // Use the ACCUMULATOR_START macro to trigger this.
295 __forceinline void start()
296 {
297 MNRY_ASSERT(mTLAccumulator && mTLAccumulator->canStart());
298 mTLAccumulator->start();
299 }
300
301 // Use the ACCUMULATOR_STOP macro to trigger this.
302 __forceinline void stop()
303 {
304 MNRY_ASSERT(mTLAccumulator && mTLAccumulator->canStop());
305 mTLAccumulator->stop();
306 mTLAccumulator = nullptr;
307 }
308
309 // Use the ACCUMULATOR_PAUSE macro to trigger this.
310 __forceinline void pause()
311 {
312 MNRY_ASSERT(mTLAccumulator && mTLAccumulator->canStop());
313 mTLAccumulator->stop();
314 }
315
316 // Use the ACCUMULATOR_UNPAUSE macro to trigger this.
317 __forceinline void unpause()
318 {
319 MNRY_ASSERT(mTLAccumulator && mTLAccumulator->canStart());
320 mTLAccumulator->start();
321 }
322
323private:
324 ThreadLocalAccumulator *mTLAccumulator;
325};
