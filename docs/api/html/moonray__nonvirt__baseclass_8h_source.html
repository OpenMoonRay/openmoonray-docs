11//
12// These macros are meant to be used when inheriting from simple POD classes. They prevent
13// the warning about a base class not having a virtual destructor. The implication is you
14// shouldn't be holding onto a dynamically allocated instance of the derived class
15// using a pointer to this base class as this will result in the derived class'
16// destructor(s) not getting called on deletion. These macros are for when you know you
17// won't violate this rule.
18//
19
20#if defined(__ICC)
21
22//
23// use these defines to bracket a region of code where a class derives from a non-virtual
24// class. (Most of the time this is not what you want to do, but for small objects it can be.)
25// Keep the region as small as possible
26//
27#define MOONRAY_START_INHERIT_FROM_NONVIRTUAL_BASECLASS __pragma(warning(disable:444))
28#define MOONRAY_FINISH_INHERIT_FROM_NONVIRTUAL_BASECLASS __pragma(warning(default:444))
29
30// simpler version for one-line cases
