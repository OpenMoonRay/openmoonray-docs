57/// Convenience function for iterating over all existing shading TLS instances.
58template <typename Body>
59finline void forEachTLS(Body const &body)
60{
61 unsigned numTLS = mcrt_common::getNumTBBThreads();
62 mcrt_common::ThreadLocalState *tlsList = mcrt_common::getTLSList();
63 for (unsigned i = 0; i < numTLS; ++i) {
64 auto geomTls = tlsList[i].mGeomTls.get();
65 if (geomTls) {
66 body(geomTls);
67 }
68 }
69}
