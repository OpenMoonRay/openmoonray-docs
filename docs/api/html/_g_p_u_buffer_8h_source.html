17// GPUBuffer manages a buffer on the GPU, with methods to copy to/from that
18// GPU buffer. You can't manipulate GPU memory directly from the host side.
19
20template <typename T>
21class GPUBuffer
22{
23public:
24 GPUBuffer() :
25 mCount {0},
26 mPtr {nullptr}
27 {}
28
29 // move constructor and assignment
30 GPUBuffer(GPUBuffer&& other)
31 {
32 mCount = other.mCount;
33 mPtr = other.mPtr;
34 other.mCount = 0;
35 other.mPtr = nullptr;
36 }
37
38 GPUBuffer& operator=(GPUBuffer&& other)
39 {
40 if (this != &other) {
41 free();
42 mCount = other.mCount;
43 mPtr = other.mPtr;
44 other.mCount = 0;
45 other.mPtr = nullptr;
46 }
47 return *this;
48 }
49
50 // non-copyable
51 GPUBuffer(const GPUBuffer&) = delete;
52 GPUBuffer& operator=(const GPUBuffer&) = delete;
53
54 ~GPUBuffer()
55 {
56 free();
57 }

59 // you MUST check for cudaSuccess for sufficient GPU VRAM
60 cudaError_t alloc(const size_t count)
61 {
62 free();
63 if (count > 0) {
64 mCount = count;
65 return cudaMalloc(&mPtr, sizeInBytes());
66 }
67 return cudaSuccess;
68 }
69
70 void free()
71 {
72 cudaError_t err = cudaFree(mPtr);
73 MNRY_ASSERT(err == cudaSuccess);
74 mCount = 0;
75 mPtr = nullptr;
76 }
77
78 // you MUST check for cudaSuccess for sufficient GPU VRAM
79 cudaError_t resize(const size_t count)
80 {
81 if (count != mCount) {
82 free();
83 return alloc(count);
84 }
85 // do nothing if the size hasn't changed
86 return cudaSuccess;
87 }
88
89 void upload(const T* src)
90 {
91 // we already know the size from the previous alloc()
92 MNRY_ASSERT(mPtr != nullptr);
93 MNRY_ASSERT(src != nullptr);
94 // *should* always succeed unless there is a bug
95 cudaError_t err = cudaMemcpy(mPtr, (void*)const_cast<T*>(src), sizeInBytes(),
96 cudaMemcpyHostToDevice);
97 MNRY_ASSERT(err == cudaSuccess);
98 }
99
100 void upload(const T* src, const size_t count)
101 {
102 // count is <= the buffer count
103 MNRY_ASSERT(mPtr != nullptr);
104 MNRY_ASSERT(src != nullptr);
105 MNRY_ASSERT(count <= mCount);
106 // *should* always succeed unless there is a bug
107 cudaError_t err = cudaMemcpy(mPtr, (void*)const_cast<T*>(src), count * sizeof(T),
108 cudaMemcpyHostToDevice);
109 MNRY_ASSERT(err == cudaSuccess);
110 }
111
112 void upload(const std::vector<T>& v)
113 {
114 MNRY_ASSERT(mPtr != nullptr);
115 MNRY_ASSERT(mCount == v.size());
116 upload(v.data());
117 }
118
119 // you MUST check for cudaSuccess for sufficient GPU VRAM
120 cudaError_t allocAndUpload(const T* src)
121 {
122 MNRY_ASSERT(src != nullptr);
123 cudaError_t err = alloc(1);
124 if (err != cudaSuccess) {
125 return err;
126 }
127 upload(src);
128 return cudaSuccess;
129 }
130
131 // you MUST check for cudaSuccess for sufficient GPU VRAM
132 cudaError_t allocAndUpload(const T* src, const size_t count)
133 {
134 MNRY_ASSERT(src != nullptr);
135 cudaError_t err = alloc(count);
136 if (err != cudaSuccess) {
137 return err;
138 }
139 if (count > 0) {
140 upload(src);
141 }
142 return cudaSuccess;
143 }
144
145 // you MUST check for cudaSuccess for sufficient GPU VRAM
146 cudaError_t allocAndUpload(const std::vector<T>& v)
147 {
148 cudaError_t err = alloc(v.size());
149 if (err != cudaSuccess) {
150 return err;
151 }
152 if (v.size() > 0) {
153 upload(v.data());
154 }
155 return cudaSuccess;
156 }
157
158 void download(T* dst) const
159 {
160 MNRY_ASSERT(mPtr != nullptr);
161 MNRY_ASSERT(dst != nullptr);
162 // *should* always succeed unless there is a bug
163 cudaError_t err = cudaMemcpy((void*)dst, mPtr, sizeInBytes(), cudaMemcpyDeviceToHost);
164 MNRY_ASSERT(err == cudaSuccess);
165 }
166
167 void download(T* dst, const size_t count) const
168 {
169 // count is <= the buffer count
170 MNRY_ASSERT(mPtr != nullptr);
171 MNRY_ASSERT(dst != nullptr);
172 MNRY_ASSERT(count <= mCount);
173 // *should* always succeed unless there is a bug
174 cudaError_t err = cudaMemcpy((void*)dst, mPtr, count * sizeof(T), cudaMemcpyDeviceToHost);
175 MNRY_ASSERT(err == cudaSuccess);
176 }
177
178 size_t sizeInBytes() const

203 return reinterpret_cast<CUdeviceptr>(mPtr);
204 }

206private:
207 size_t mCount;
208 T *mPtr;
209};
