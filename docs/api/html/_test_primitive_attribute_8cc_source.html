74void
75TestPrimitiveAttribute::setUp()
76{
77 mcrt_common::ThreadLocalState *tls = mcrt_common::getFrameUpdateTLS();
78 mTls = MNRY_VERIFY(tls->mShadingTls.get());
79 scene_rdl2::alloc::Arena *arena = mTls->mArena;
80
81 const int nkeys = keysPerStride * nStrides;
82
83 // stride layout:
84 // bool: stored as uint8_t, padded to 32 bits
85 // int: stored as int32_t, no padding
86 // float: stored as float, no padding
87 // Vec2f: stored as Vec2f, no padding
88 // Vec3f: stored as Vec3f, no padding
89 // Mat4f: stored as Mat4f, no padding
90 // Color: stored as Color, no padding
91 int *offsets = arena->allocArray<int>(nkeys);
92 mTls->mAttributeOffsets = offsets;
93 const size_t strideSize = sizeof(int32_t) + sizeof(int32_t) + sizeof(float) +
94 sizeof(Vec2f) + sizeof(Vec3f) + sizeof(Mat4f) + sizeof(Color);
95
96 for (size_t stride = 0; stride < nStrides; ++stride) {
97 const int sk = stride * keysPerStride;
98
99 offsets[sk + 0] = stride * strideSize;
100 offsets[sk + 1] = offsets[sk + 0] + sizeof(int32_t);
101 offsets[sk + 2] = offsets[sk + 1] + sizeof(int32_t);
102 offsets[sk + 3] = offsets[sk + 2] + sizeof(float);
103 offsets[sk + 4] = offsets[sk + 3] + sizeof(Vec2f);
104 offsets[sk + 5] = offsets[sk + 4] + sizeof(Vec3f);
105 offsets[sk + 6] = offsets[sk + 5] + sizeof(Mat4f);
106 }
107 const size_t dataSize = strideSize * nStrides;
108
109 uint8_t *data[VLEN];
110 for (size_t lane = 0; lane < VLEN; ++lane) {
111 data[lane] = arena->allocArray<uint8_t>(dataSize);
112
113 // mimics Address64_set from Util.isph
114 intptr_t addr = (intptr_t) data[lane];
115 mStatev.mData.mHigh[lane] = (addr >> 32);
116 mStatev.mData.mLow[lane] = (addr & 0xffffffff);
117 }
118
119 for (size_t stride = 0; stride < nStrides; ++stride) {
120 for (size_t lane = 0; lane < VLEN; ++lane) {
121
122 uint8_t *d = data[lane];
123 const int sk = stride * keysPerStride;
124
125 // fill data out with test data
126 *reinterpret_cast<char *>(d + offsets[sk + 0]) = getBool(stride, lane);
127 *reinterpret_cast<int *>(d + offsets[sk + 1]) = getInt(stride, lane);
128 *reinterpret_cast<float *>(d + offsets[sk + 2]) = getFloat(stride, lane);
129 *reinterpret_cast<Vec2f *>(d + offsets[sk + 3]) = getVec2f(stride, lane);
130 *reinterpret_cast<Vec3f *>(d + offsets[sk + 4]) = getVec3f(stride, lane);
131 *reinterpret_cast<Mat4f *>(d + offsets[sk + 5]) = getMat4f(stride, lane);
132 *reinterpret_cast<Color *>(d + offsets[sk + 6]) = getColor(stride, lane);
133 }
134 }
135}
