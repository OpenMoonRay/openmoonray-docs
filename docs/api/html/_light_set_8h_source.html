103 // Set the accelerator for this light set. The accelerator may contain more lights than the LightSet itself.
104 // This is because the accelerator's light list is generated during render prep, and includes all of the original
105 // lights specified in the rdl2::LightSet. The LightSet has some lights culled away during mcrt, so contains a
106 // subset of those lights. We need to map the index of the lights in the LightAccelerator to the index of the light
107 // in the LightSet. The lightIdMap does this. Lights that exist in the LightAccelerator but not in the LightSet are
108 // mapped to -1.
109 finline void setAccelerator(const LightAccelerator* acc, const int* lightIdMap)
110 {
111 mAccelerator = acc;
112 mAcceleratorLightIdMap = lightIdMap;
113 }
