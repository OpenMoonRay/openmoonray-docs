29// LineSegments is similar to nonperiodic linear curves in Renderman
30// each span is a linear line segment with 2 control points
31// one or more spans form a chain, each chain is a curve
32// one or more chains form a LineSegments (Curves) primitive
33// To interpolate primitive attributes in different rates:
34// constant : should supply a single value for the entire Curves primitive
35// uniform : should supply a total of nCurves values for the entire Curves primitive
36// varying : should supply sum(nSpansi + 1) values
37// facevarying : should supply sum(nSpansi + 1) values
38// vertex : should supply sum(nVerticesi) values
39//
40// For example: a LineSegments with 2 chains (nCurves = 2)
41// chain0 has 3 spans (nSpans0 = 3, nVertices0 = 3 + 1 = 4)
42// chain1 has 1 spans (nSpans1 = 1, nVertices1 = 1 + 1 = 2)
43// constant attribute should supply 1 value
44// uniform attribute should supply 2 values
45// varying/facevarying/vertex attribute should supply 4 + 2 = 6 values
46class LineSegmentsInterpolator : public shading::Interpolator
47{
48public:
49 LineSegmentsInterpolator(const Attributes *attr, float time,
50 int chain, float u, int vertexOffset):
51 shading::Interpolator(attr, time,
52 0, // part
53 chain, // coarseFace
54 2, // numVaryings
55 mCurveVertices, // varyings
56 mLinearWeights, // varyingWeights
57 2, // numFaceVaryings
58 mCurveVertices, // faceVaryings
59 mLinearWeights, // faceVaryingWeights
60 chain, // tessellatedFace
61 2, // numVertices
62 mCurveVertices, // vertices
63 mLinearWeights) // vertexWeights
64 {
65 mLinearWeights[0] = 1.0f - u;
66 mLinearWeights[1] = u;
67 mCurveVertices[0] = vertexOffset;
68 mCurveVertices[1] = vertexOffset + 1;
69 }
70private:
71 float mLinearWeights[2];
72 int mCurveVertices[2];
73};

76LineSegments::LineSegments(
77 Curves::Type type,
78 geom::Curves::SubType subtype,
79 geom::Curves::CurvesVertexCount&& curvesVertexCount,
80 geom::Curves::VertexBuffer&& vertices,
81 LayerAssignmentId&& layerAssignmentId,
82 PrimitiveAttributeTable&& primitiveAttributeTable):
83 Curves(type,
84 static_cast<Curves::SubType>(subtype),
85 std::move(curvesVertexCount),
86 std::move(vertices),
87 std::move(layerAssignmentId),
88 std::move(primitiveAttributeTable))
89{
90 // allocate/calculate index buffer
91 size_t curvesCount = getCurvesCount();
92 std::vector<uint32_t> spansInChain(curvesCount, 0);
93 size_t spanCount = 0;
94 for (size_t i = 0; i < curvesCount; ++i) {
95 spansInChain[i] = mCurvesVertexCount[i] - 1;
96 spanCount += spansInChain[i];
97 }
98 mSpanCount = spanCount;
99 // allocate/fill the index buffer
100 mIndexBuffer.reserve(mSpanCount);
101 size_t vertexOffset = 0;
102 for (size_t i = 0; i < curvesCount; ++i) {
103 for (size_t j = 0; j < spansInChain[i]; ++j) {
104 mIndexBuffer.emplace_back(vertexOffset++, i, j);
105 }
106 vertexOffset += 1;
107 }
108
109 if (mLayerAssignmentId.getType() == LayerAssignmentId::Type::VARYING) {
110 MNRY_ASSERT_REQUIRE(
111 mLayerAssignmentId.getVaryingId().size() == curvesCount);
112 }
113
114 // facevarying/varying/vertex are the same thing for LineSegments
115 for (auto& kv : mPrimitiveAttributeTable) {
116 for (auto& attribute : kv.second) {
117 if (attribute->getRate() == RATE_VARYING ||
118 attribute->getRate() == RATE_FACE_VARYING) {
119 attribute->setRate(RATE_VERTEX);
120 }
121 }
122 }
123 size_t vertexCount = getVertexCount();
124 setAttributes(Attributes::interleave(mPrimitiveAttributeTable,
125 0, curvesCount, vertexCount, std::vector<size_t>(), vertexCount));
126}

128void
129LineSegments::postIntersect(mcrt_common::ThreadLocalState &tls,
130 const scene_rdl2::rdl2::Layer* layer, const mcrt_common::Ray& ray,
131 Intersection& intersection) const
132{
133 int spanId = ray.primID;
134 const IndexData& index = mIndexBuffer[spanId];
135 const uint32_t chain = index.mChain;
136
137 int assignmentId =
138 mLayerAssignmentId.getType() == LayerAssignmentId::Type::CONSTANT ?
139 mLayerAssignmentId.getConstId() :
140 mLayerAssignmentId.getVaryingId()[chain];
141 intersection.setLayerAssignments(assignmentId, layer);
142
143 const AttributeTable *table =
144 intersection.getMaterial()->get<shading::Material>().getPrimitiveAttributeTable();
145 intersection.setTable(&tls.mArena, table);
146
147 const uint32_t vertexOffset = index.mVertex;
148 intersection.setIds(vertexOffset, 0, 0);
149 overrideInstanceAttrs(ray, intersection);
150
151 Vec3fa cv[2];
152 if (!isMotionBlurOn()) {
153 // Grab the CVs
154 cv[0] = mVertexBuffer(vertexOffset + 0);
155 cv[1] = mVertexBuffer(vertexOffset + 1);
156 } else {
157 // only support two time samples for motionblur at this moment
158 MNRY_ASSERT(getMotionSamplesCount() == 2);
159 float w1 = ray.time;
160 float w0 = 1.0f - w1;
161 cv[0] = mVertexBuffer(vertexOffset + 0, 0) * w0 +
162 mVertexBuffer(vertexOffset + 0, 1) * w1;
163
164 cv[1] = mVertexBuffer(vertexOffset + 1, 0) * w0 +
165 mVertexBuffer(vertexOffset + 1, 1) * w1;
166 }
167
168 Vec3f Ng, N, dPds, dPdt;
169 Vec2f St;
170 // Partial with respect to S is simply the tangent vector
171 dPds = cv[1] - cv[0];
172
173 // We now render curves as ray-aligned flat ribbons.
174 // Use the flipped ray direction to compute the normal. But, this may be an
175 // instance, so we need to transform the ray direction into the local
176 // space. It gets transformed back to render space later on.
177 // Note that it looks like we should be using r2l to get from render to
178 // local space, but recall that when transforming normals we use the transpose
179 // of the inverse, which is l2r transposed. The transformNormal() function
180 // takes the inverse of the transform (l2r) and applies the transpose.
181 const Vec3f flippedRayDir = ray.isInstanceHit() ?
182 normalize(transformNormal(ray.ext.l2r, -ray.dir)) :
183 normalize(-ray.dir);
184
185 // Lacking much better options, we'll construct dPdt wrt the other two
186 // components of the ostensibly orthonormal frame we have
187 dPdt = cross(dPds, flippedRayDir);
188 if (unlikely(lengthSqr(dPdt) < scene_rdl2::math::sEpsilon)) {
189 // Degened dPds or dPdt. The reason can be either clustered cv points
190 // or ray direction is parallel to dPds. Use ReferenceFrame as fallback
191 N = flippedRayDir;
192 scene_rdl2::math::ReferenceFrame frame(N);
193 dPds = frame.getX();
194 dPdt = frame.getY();
195 } else {
196 // Cross the partials to get the normal
197 N = cross(dPdt, dPds);
198 }
199 N.normalize();
200 // Geometric normal equals shading normal for a line segment
201 Ng = N;
202
203 const uint32_t spansInChain = mCurvesVertexCount[chain] - 1;
204 const float stSpanRange = 1.0f / spansInChain;
205 const uint32_t indexInChain = index.mSpan;
206 const float stStart = stSpanRange * indexInChain;
207 St[0] = stStart + ray.u * stSpanRange;
208 St[1] = 0.0f;
209
210 intersection.setDifferentialGeometry(Ng, N, St, dPds, dPdt, true);
211
212 const scene_rdl2::rdl2::Geometry* geometry = intersection.getGeometryObject();
213 MNRY_ASSERT(geometry != nullptr);
214 if (geometry->getRayEpsilon() <= 0.0f) {
215 // Use the 1.5 times the curve radius as the next ray epsilon hint.
216 // This lets shadow and continuation rays escape
217 // the intersected hair, instead of self-intersecting.
218 float curveRadius = cv[0].w * (1.0f - ray.u) + cv[1].w * ray.u;
219 intersection.setEpsilonHint(1.5f * curveRadius);
220 } else {
221 intersection.setEpsilonHint( geometry->getRayEpsilon() );
222 }
223
224 LineSegmentsInterpolator interpolator(getAttributes(), ray.time, chain,
225 ray.u, vertexOffset);
226 intersection.setRequiredAttributes(interpolator);
227
228 // calculate dfds/dfdt for primitive attributes that request differential
229 if (table->requestDerivatives()) {
230 // ds = stSpanRange -> 1 / ds = 1 / stSpanRange = spansInChain
231 computeAttributesDerivatives(table, (float)spansInChain,
232 vertexOffset, ray.time, intersection);
233 }
234
235 // for wireframe AOV/shaders
236 if (table->requests(StandardAttributes::sNumPolyVertices)) {
237 intersection.setAttribute(StandardAttributes::sNumPolyVertices, 2);
238 }
239 if (table->requests(StandardAttributes::sPolyVertexType)) {
240 intersection.setAttribute(StandardAttributes::sPolyVertexType,
241 static_cast<int>(StandardAttributes::POLYVERTEX_TYPE_LINE));
242 }
243 for (int iVert = 0; iVert < 2; iVert++) {
244 if (table->requests(StandardAttributes::sPolyVertices[iVert])) {
245 // may need to move the vertices to render space
246 // for instancing object since they are ray traced in local space
247 const Vec3f v = ray.isInstanceHit() ? transformPoint(ray.ext.l2r, cv[iVert].asVec3f())
248 : cv[iVert].asVec3f();
249 intersection.setAttribute(StandardAttributes::sPolyVertices[iVert], v);
250 }
251 }
252
253 // motion vectors
254 if (table->requests(StandardAttributes::sMotion)) {
255 Vec3f pos0, pos1;
256 const Vec3f *pos1Ptr = nullptr;
257 if (isMotionBlurOn()) {
258 float wc = ray.getTime() - sHalfDt;
259 const Vec3fa v0 = lerp(
260 lerp(mVertexBuffer(vertexOffset + 0, 0), mVertexBuffer(vertexOffset + 0, 1), wc),
261 lerp(mVertexBuffer(vertexOffset + 1, 0), mVertexBuffer(vertexOffset + 1, 1), wc),
262 ray.u);
263 wc = ray.getTime() + sHalfDt;
264 const Vec3fa v1 = lerp(
265 lerp(mVertexBuffer(vertexOffset + 0, 0), mVertexBuffer(vertexOffset + 0, 1), wc),
266 lerp(mVertexBuffer(vertexOffset + 1, 0), mVertexBuffer(vertexOffset + 1, 1), wc),
267 ray.u);
268 pos0 = v0.asVec3f();
269 pos1 = v1.asVec3f();
270 pos1Ptr = &pos1;
271 } else {
272 const Vec3fa v0 = lerp(
273 mVertexBuffer(vertexOffset + 0, 0), mVertexBuffer(vertexOffset + 1, 0), ray.u);
274 pos0 = v0.asVec3f();
275 pos1Ptr = nullptr;
276 }
277
278 // Motion vectors only support a single instance level, hence we only care
279 // about ray.instance0.
280 const Instance *instance = (ray.isInstanceHit())?
281 static_cast<const Instance *>(ray.ext.instance0OrLight) : nullptr;
282 const Vec3f motion = computePrimitiveMotion(pos0, pos1Ptr, ray.getTime(), instance);
283 intersection.setAttribute(StandardAttributes::sMotion, motion);
284 }
285}

287bool
288LineSegments::computeIntersectCurvature(const mcrt_common::Ray& ray,
289 const Intersection& intersection, Vec3f& dNds, Vec3f& dNdt) const
290{
291 uint32_t id1, id2, id3;
292 intersection.getIds(id1, id2, id3);
293
294 const uint vertexOffset = id1;
295
296 Vec3fa cv0, cv1;
297 if (!isMotionBlurOn()) {
298 // Grab the CVs
299 cv0 = mVertexBuffer(vertexOffset + 0);
300 cv1 = mVertexBuffer(vertexOffset + 1);
301 } else {
302 // only support two time samples for motionblur at this moment
303 MNRY_ASSERT(getMotionSamplesCount() == 2);
304 float w1 = ray.time;
305 float w0 = 1.0f-w1;
306 cv0 = mVertexBuffer(vertexOffset + 0, 0) * w0 +
307 mVertexBuffer(vertexOffset + 0, 1) * w1;
308
309 cv1 = mVertexBuffer(vertexOffset + 1, 0) * w0 +
310 mVertexBuffer(vertexOffset + 1, 1) * w1;
311 }
312
313 Vec3fa centerP = (1.0f - ray.u) * cv0 + ray.u * cv1;
314 Vec3fa normal = Vec3fa(intersection.getP(), 0.f) - centerP;
315
316 // Modelled locally as a cylinder, there is no variance of the normal
317 // down the length of the cylinder
318 dNds = Vec3f(0.0f, 0.0f, 0.0f);
319
320 // We do vary as we extend closer to the edge
321 Vec3fa posPlusDeltaT = Vec3fa(intersection.getP() + intersection.getdPdt(), 0.f);
322 dNdt = (posPlusDeltaT - centerP) - normal;
323 return true;
324}

326void
327LineSegments::computeAttributesDerivatives(const AttributeTable* table,
328 float invDs, int vertexOffset,
329 float time, Intersection& intersection) const
330{
331 Attributes* attrs = getAttributes();
332 for (auto key: table->getDifferentialAttributes()) {
333 if (!attrs->isSupported(key)) {
334 continue;
335 }
336 if (attrs->getRate(key) == RATE_VERTEX) {
337 switch (key.getType()) {
338 case scene_rdl2::rdl2::TYPE_FLOAT:
339 computeVertexAttributeDerivatives(
340 TypedAttributeKey<float>(key),
341 invDs, vertexOffset, time, intersection);
342 break;
343 case scene_rdl2::rdl2::TYPE_RGB:
344 computeVertexAttributeDerivatives(
345 TypedAttributeKey<scene_rdl2::math::Color>(key),
346 invDs, vertexOffset, time, intersection);
347 break;
348 case scene_rdl2::rdl2::TYPE_RGBA:
349 computeVertexAttributeDerivatives(
350 TypedAttributeKey<scene_rdl2::math::Color4>(key),
351 invDs, vertexOffset, time, intersection);
352 break;
353 case scene_rdl2::rdl2::TYPE_VEC2F:
354 computeVertexAttributeDerivatives(
355 TypedAttributeKey<scene_rdl2::math::Vec2f>(key),
356 invDs, vertexOffset, time, intersection);
357 break;
358 case scene_rdl2::rdl2::TYPE_VEC3F:
359 computeVertexAttributeDerivatives(
360 TypedAttributeKey<scene_rdl2::math::Vec3f>(key),
361 invDs, vertexOffset, time, intersection);
362 break;
363 case scene_rdl2::rdl2::TYPE_MAT4F:
364 computeVertexAttributeDerivatives(
365 TypedAttributeKey<scene_rdl2::math::Mat4f>(key),
366 invDs, vertexOffset, time, intersection);
367 break;
368 default:
369 break;
370 }
371 }
372 }
