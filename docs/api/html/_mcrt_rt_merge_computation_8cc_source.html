46inline fb_util::VariablePixelBuffer::Format
47convertImageEncoding(BaseFrame::ImageEncoding encoding)
48{
49 switch (encoding)
50 {
51 case BaseFrame::ENCODING_RGBA8: return fb_util::VariablePixelBuffer::RGBA8888;
52 case BaseFrame::ENCODING_RGB888: return fb_util::VariablePixelBuffer::RGB888;

54 }
55 return fb_util::VariablePixelBuffer::UNINITIALIZED;
56}

105void
106McrtRtMergeComputation::configure(const object::Object& aConfig)
107{
108 MOONRAY_LOG_INFO(">>> McrtRtMergeComputation <<< ...");
109
110 mMotionCaptureMode = false;
111 const std::string sApplicationMode = "applicationMode";
112 if (!aConfig[sApplicationMode].isNull()) {
113 if (aConfig[sApplicationMode].value().asInt() == 1) {
114 mMotionCaptureMode = true;
115 } else {
116 MOONRAY_LOG_ERROR("APPLICATION MODE SET TO UNDEFIND");
117 }
118 }
119
120 if (!aConfig["numMachines"].isNull()) {
121 mNumMachines = aConfig["numMachines"];
122 MNRY_ASSERT_REQUIRE(mNumMachines > 0);
123 mUpstreamBuffers.resize(mNumMachines);
124 mUpstreamPixelInfoBuffers.resize(mNumMachines);
125 mTiles.resize(mNumMachines);
126 mStatus.reset(new BaseFrame::Status[mNumMachines]);
127 mProgress.reset(new float[mNumMachines]);
128 for (unsigned int i = 0; i < mNumMachines; ++i) {
129 mStatus[i] = BaseFrame::ERROR;
130 mProgress[i] = 0.0;
131 }
132 int totalCacheFrame = 6;
133 mFbArray.reset(new RenderFbArray(totalCacheFrame, mNumMachines));
134
135 } else {
136 MNRY_ASSERT_REQUIRE(false,
137 "numMachines is a required config setting for the mcrt_merge computation");
138 }
139
140 if (!aConfig["imageEncoding"].isNull()) {
141 mImageEncoding = static_cast<BaseFrame::ImageEncoding>(static_cast<int>((aConfig["imageEncoding"])));
142 }
143
144 if (!aConfig["fps"].isNull()) {
145 mFps = aConfig["fps"];
146 }

149bool
150McrtRtMergeComputation::fpsIntervalPassed()
151{
152 // In a single machine setup, frame gating is handled here.
153 double now = util::getSeconds();
154 if (now - mLastTime < (1.0f / mFps)) {
155 return false;
156 }
157 mLastTime = now;
158 return true;

161void
162McrtRtMergeComputation::onIdle()
163{
164 // If we don't have any upstream buffers, don't bother.
165 if (mMotionCaptureMode) {
166 onIdle_mocap();
167 return;
168 }
169
170 if (mNumMachines == 0 || mUpstreamBuffers.empty() || mFinalFrame.getArea() == 0 || !fpsIntervalPassed()) {
171 //MOONRAY_LOG_DEBUG("Returning early: mNumMachines: %d | mUpstreamBuffers.empty(): %s | mFinalFrame.getArea(): %d | fpsIntervalPassed: %s", mNumMachines, boolToString(mUpstreamBuffers.empty()), mFinalFrame.getArea(), boolToString(fpsIntervalPassed()));
172 //MOONRAY_LOG_DEBUG_STR("Returning early: mNumMachines:" << mNumMachines << " | mUpstreamBuffers.empty(): " << boolToString(mUpstreamBuffers.empty()) << " | mFinalFrame.getArea(): " << mFinalFrame.getArea() << " | fpsIntervalPassed: " << boolToString(fpsIntervalPassed()));
173 return;
174 }
175
176 float progress = 0.f;
177 bool hasUpstreamBuffer = false;
178 for (unsigned int i = 0; i < mNumMachines; ++i) {
179 // This makes sure we at least have some
180 // part of the frame to show.
181 if (mUpstreamBuffers[i].size() != 0) {
182 hasUpstreamBuffer = true;
183 }
184 progress += mProgress[i];
185 }
186
187 if (!hasUpstreamBuffer) {
188 MOONRAY_LOG_DEBUG("Returning early: hasUpstreamBuffer %s", boolToString(hasUpstreamBuffer));
189 return;
190 }
191 progress = std::min(progress, 1.0f);
192
193 BaseFrame::Status status = BaseFrame::FINISHED;
194 for (unsigned int i = 0; i < mNumMachines; ++i) {
195 // It seems like it is possible that by the time
196 // the first frame is ready to be sent after applying
197 // an update all the nodes could be on
198 // BaseFrame::RENDERING but it is
199 // Important that at least one STARTED notice is sent
200 // at the begining
201 if (mNeedsStartedSent) {
202 status = BaseFrame::STARTED;
203 mNeedsStartedSent = false;
204 break;
205 }
206 // If at least one machine is not finished make sure we don't report finished.
207 if (mStatus[i] != BaseFrame::FINISHED) {
208 status = mStatus[i];
209 }
210 }
211
212 // If we haven't gotten any new frames just return
213 if (!mHasPartialFrame) {
214 return;
215 }
216
217 mHasPartialFrame = false;
218
219 // Send it downstream.
220 RenderedFrame::Ptr frameMsg(new RenderedFrame);
221
222 frameMsg->mHeader.mRezedViewport = mRezedViewport;
223 frameMsg->mHeader.mStatus = status;
224 frameMsg->mHeader.mProgress = progress;
225
226 // Copy each upstream buffer into the final frame buffer.
227 for (size_t i = 0; i < mUpstreamBuffers.size(); ++i) {
228 if (!mTiles[i].empty() && mUpstreamBuffers[i].size() != 0) {
229 mTiledFrame.unpackSparseTiles(&mUpstreamBuffers[i][0], mTiles[i]);
230 }
231 }
232
233 // Untile buffer.
234 fb_util::Tiler tiler(mFinalFrame.getWidth(), mFinalFrame.getHeight());
235 mFinalFrame.untile(mTiledFrame, tiler, true);
236
237# ifdef ALPHA_VIEW_TEST
238 tmpHack(&mFinalFrame); // test code to view alpha value
239# endif // end ALPHA_VIEW_TEST
240
241
242 // Image buffer as unsigned char
243 moonray::network::DataPtr buffer;
244
245 // Size of image buffer in bytes
246 size_t bufferLength;
247
248 math::Viewport fullViewport(0, 0, mFrameWidth - 1, mFrameHeight - 1);
249
250 // If we're using an ROI viewport, then copy the contents of the final frame into our smaller buffer
251 if (mUsingROI) {
252 frameMsg->mHeader.setViewport(mViewport.mMinX, mViewport.mMinY, mViewport.mMaxX, mViewport.mMaxY);
253 MOONRAY_LOG_DEBUG("Using ROI Viewport: (%d, %d, %d, %d) (%d x %d)", mViewport.mMinX, mViewport.mMinY, mViewport.mMaxX, mViewport.mMaxY, mViewport.width(), mViewport.height());
254 fb_util::copyRoiBuffer<uint8_t>
255 (mViewport, fullViewport, mFinalFrame.getSizeOfPixel(),
256 mRoiPixelBuffer.getData(),
257 mFinalFrame.getData(), bufferLength);
258 buffer = mRoiPixelBuffer.getDataShared();
259 } else { // Otherwise, use the full buffer
260 frameMsg->mHeader.mViewport.reset();
261 buffer = mFinalFrame.getDataShared();
262 bufferLength = mFinalFrame.getArea() * mFinalFrame.getSizeOfPixel();
263 }
264
265 frameMsg->addBuffer(buffer, bufferLength, "beauty", mImageEncoding);
266
267 if (mHasPixelInfo) {
268 // Copy each upstream pixel buffer into the final pixel info buffer.
269 for (size_t i = 0; i < mUpstreamPixelInfoBuffers.size(); ++i) {
270 if (!mTiles[i].empty() && mUpstreamPixelInfoBuffers[i].size() != 0) {
271 fb_util::unpackSparseTiles(&mPixelInfoTiledFrame, &mUpstreamPixelInfoBuffers[i][0], mTiles[i]);
272 }
273 }
274
275 // Untile buffer.
276 fb_util::Tiler tiler(mFinalPixelInfoFrame.getWidth(), mFinalPixelInfoFrame.getHeight());
277 fb_util::untile(&mFinalPixelInfoFrame, mPixelInfoTiledFrame, tiler, true,
278 [](const fb_util::PixelInfoBuffer::PixelType &pixel, unsigned)
279 -> const fb_util::PixelInfoBuffer::PixelType & {
280 return pixel;
281 });
282
283
284 // Handle ROI for the pixel buffer
285 if (mUsingROI) {
286 fb_util::copyRoiBuffer<float>
287 (mViewport, fullViewport, 1,
288 reinterpret_cast<float *>(mRoiPixelInfoBuffer.getData()),
289 reinterpret_cast<float *>(mFinalPixelInfoFrame.getData()), bufferLength);
290 buffer = mRoiPixelInfoBuffer.getDataSharedAs<uint8_t>();
291 } else {
292 buffer = mFinalPixelInfoFrame.getDataSharedAs<uint8_t>();
293 bufferLength = mFinalPixelInfoFrame.getArea() * sizeof(decltype(mFinalPixelInfoFrame)::PixelType); // 1 float per pixel
294 }
295
296 MOONRAY_LOG_DEBUG("Adding pixel info buffer");
297 frameMsg->addBuffer(buffer, bufferLength, "depth", BaseFrame::ENCODING_FLOAT);
298 mHasPixelInfo = false;
299 }
300
301 // TODO... At this point we have a full sized linear buffer containing the final frame.
302 // If there is an active ROI, we'll need to extract the relevant pixels into the
303 // smaller ROI buffer right here.
304
305 MOONRAY_LOG_DEBUG("Sending frame!");
306 send(frameMsg);
307
308 if (mFirstFrame == true) {
309 moonray::network::GenericMessage::Ptr firstFrameMsg(new moonray::network::GenericMessage);
310 firstFrameMsg->mValue = "MCRT Rendered First Frame";
311 send (firstFrameMsg);
312 mFirstFrame = false;
313 MOONRAY_LOG_INFO("McrtMerge Sent first frame message");
314 }

317void
318McrtRtMergeComputation::onIdle_mocap()
319{
320 if (mNumMachines == 0 || !mFbArray || mFinalFrame.getArea() == 0) {
321 return;
322 }
323
324 if (!mFbArray->isActive()) {
325 return;
326 }
327
328 double now = util::getSeconds();
329 if (now - mLastTime < (1.0f / 60.0f)) {
330 return;
331 }
332
333 if (!mFbArray->completeTestAndShift()) {
334 return; // could not find completed renderFb yet.
335 }
336
337 mLastTime = now;
338
339 RenderFb * cRenderFb = mFbArray->getLocal(0); // get first frame data in the fbArray
340 float progress = cRenderFb->unpackSparseTiles(mTiles, mTiledFrame);
341 mFbArray->shiftFbTbl(); // shift one frame
342
343 // basically, mocap case, we always report as FINISHED.
344 /*
345 BaseFrame::Status status = BaseFrame::FINISHED;
346 if (mNeedsStartedSent) {
347 // It is important that at least one STARTED notice is sent at the begining
348 status = BaseFrame::STARTED;
349 mNeedsStartedSent = false;
350 }
351 */
352 BaseFrame::Status status = BaseFrame::STARTED;
353
354 // Untile buffer.
355 fb_util::Tiler tiler(mFinalFrame.getWidth(), mFinalFrame.getHeight());
356 mFinalFrame.untile(mTiledFrame, tiler, true);
357
358 // Send it downstream.
359 RenderedFrame::Ptr frameMsg(new RenderedFrame);
360
361 frameMsg->mHeader.mRezedViewport = mRezedViewport;
362 frameMsg->mHeader.mStatus = status;
363 frameMsg->mHeader.mProgress = progress;
364
365 // Image buffer as unsigned char
366 moonray::network::DataPtr buffer;
367
368 // Size of image buffer in bytes
369 size_t bufferLength;
370
371 math::Viewport fullViewport(0, 0, mFrameWidth - 1, mFrameHeight - 1);
372
373 // If we're using an ROI viewport, then copy the contents of the final frame into our smaller buffer
374 if (mUsingROI) {
375 frameMsg->mHeader.setViewport(mViewport.mMinX, mViewport.mMinY, mViewport.mMaxX, mViewport.mMaxY);
376 MOONRAY_LOG_DEBUG("Using ROI Viewport: (%d, %d, %d, %d) (%d x %d)",
377 mViewport.mMinX, mViewport.mMinY, mViewport.mMaxX, mViewport.mMaxY,
378 mViewport.width(), mViewport.height());
379 fb_util::copyRoiBuffer<uint8_t>(mViewport, fullViewport, mFinalFrame.getSizeOfPixel(),
380 mRoiPixelBuffer.getData(),
381 mFinalFrame.getData(), bufferLength);
382 buffer = mRoiPixelBuffer.getDataShared();
383 } else { // Otherwise, use the full buffer
384 frameMsg->mHeader.mViewport.reset();
385 buffer = mFinalFrame.getDataShared();
386 bufferLength = mFinalFrame.getArea() * mFinalFrame.getSizeOfPixel();
387 }
388
389 frameMsg->addBuffer(buffer, bufferLength, "beauty", mImageEncoding);
390
391 // Not support pixelInfo mode for mocap
392
393 MOONRAY_LOG_DEBUG("Sending frame!");
394 send(frameMsg);
395
396 if(mFirstFrame == true ) {
397 moonray::network::GenericMessage::Ptr firstFrameMsg(new moonray::network::GenericMessage);
398 firstFrameMsg->mValue = "MCRT Rendered First Frame";
399 send(firstFrameMsg);
400 mFirstFrame = false;
401 MOONRAY_LOG_INFO("McrtMerge Sent first frame message");
402 }

405void
406McrtRtMergeComputation::onMessage(const Message::Ptr aMsg)
407{
408 if (mMotionCaptureMode) {
409 onMessage_mocap(aMsg);
410 return;
411 }
412
413 if (aMsg->id() == PartialFrame::ID) {
414 PartialFrame& partial = static_cast<PartialFrame&>(*aMsg);
415 onViewportChanged(partial);
416
417 int machineId = partial.mMachineId;
418 //MOONRAY_LOG_DEBUG("machineId: %s vs mNumMachines; %d", machineId, mNumMachines);
419 MNRY_ASSERT_REQUIRE((unsigned)machineId < mNumMachines);
420
421 for (const auto& buffer: partial.mBuffers) {
422 // TODO: Extract buffers by type
423 if (!std::strcmp(buffer.mName, "beauty")) {
424 auto &buf = mUpstreamBuffers[machineId];
425 MNRY_ASSERT(buffer.mDataLength);
426 buf.resize(buffer.mDataLength);
427 memcpy(&buf[0], buffer.mData.get(), buffer.mDataLength);
428 if (partial.getStatus() == BaseFrame::STARTED) {
429 mNeedsStartedSent = true;
430 }
431 mHasPartialFrame = true;
432 mStatus[machineId] = partial.getStatus();
433 mProgress[machineId] = partial.getProgress();
434 } else if (!std::strcmp(buffer.mName, "depth")) {
435 auto &buf = mUpstreamPixelInfoBuffers[machineId];
436 MNRY_ASSERT(buffer.mDataLength &&
437 (buffer.mDataLength % sizeof(fb_util::PixelInfoBuffer::PixelType) == 0));
438 buf.resize(buffer.mDataLength);
439 memcpy(&buf[0], buffer.mData.get(), buffer.mDataLength);
440 mHasPixelInfo = true;
441 }
442 }
443 }

459void
460McrtRtMergeComputation::onViewportChanged(const PartialFrame& msg)
461{
462
463 bool shouldReinit = false;
464 const BaseFrame::Viewport& viewport(msg.getViewport());
465
466 // Check if we're using an ROI, and/or if it's different than our current viewport
467 math::Viewport roiViewport;
468 if (viewport.hasViewport()) {
469 roiViewport = math::Viewport(viewport.minX(), viewport.minY(), viewport.maxX(), viewport.maxY());
470 if (roiViewport != mViewport) {
471 shouldReinit = true;
472 }
473 // If we were using ROI, and we no longer have an input viewport, then we're cancelling ROI
474 } else if (mUsingROI) {
475 shouldReinit = true;
476 }
477
478 if (msg.getRezedViewport().height() != mFrameHeight || msg.getRezedViewport().width() != mFrameWidth) {
479 shouldReinit = true;
480 }
481
482 // If we haven't changed our frame width or height, or we're not using a different ROI, early exit
483 if (!shouldReinit) {
484 return;
485 }
486
487 /* Useful debug code
488 std::cout << ">>> McrtRtMergeComputation.cc onViewportChanged() viewport changed!" << std::endl;
489 std::cout << ">>> McrtRtMergeComputation.cc"
490 << " mFrameWidth:" << mFrameWidth << " -> " << msg.getRezedViewport().width()
491 << " mFrameHeight:" << mFrameHeight << " -> " << msg.getRezedViewport().height()
492 << " minX:" << viewport.minX()
493 << " minY:" << viewport.minY()
494 << " maxX:" << viewport.maxX()
495 << " maxY:" << viewport.maxY()
496 << std::endl;
497 */
498
499 mRezedViewport = msg.getRezedViewport();
500 mFrameWidth = mRezedViewport.width();
501 mFrameHeight = mRezedViewport.height();
502
503 if (viewport.hasViewport()) {
504 mViewport = roiViewport;
505 mUsingROI = true;
506 } else {
507 mViewport = math::Viewport(0, 0, mFrameWidth - 1, mFrameHeight - 1);
508 mUsingROI = false;
509 }
510
511 MOONRAY_LOG_DEBUG_STR("onViewportChanged (" << mFrameWidth << " x " << mFrameHeight << ")");
512
513 MNRY_ASSERT(mUpstreamBuffers.size() == mNumMachines);
514
515 // Reset upstream buffers.
516 for (unsigned int i = 0; i < mNumMachines; ++i) {
517 mUpstreamBuffers[i].clear();
518 mUpstreamPixelInfoBuffers[i].clear();
519 }
520
521 // Create tile configurations for each upstream buffer.
522 //
523 // TODO: The tiling mode here needs to be kept in sync with the tiling
524 // mode which the partial frames were generated with. Ideally this
525 // information would be transmitted in a message but for now it's
526 // just hardcoded.
527 auto tileSchedulerType = rndr::TileScheduler::SPIRAL_SQUARE;
528 mTileScheduler = rndr::TileScheduler::create(tileSchedulerType);
529
530 for (unsigned int i = 0; i < mNumMachines; ++i) {
531 SCOPED_MEM(&mArena);
532 mTiles[i].clear();
533 if (mTileScheduler->generateTiles(&mArena, mFrameWidth, mFrameHeight, mViewport, i, mNumMachines)) {
534 mTiles[i] = mTileScheduler->getTiles();
535 }
536 }
537
538 // Init buffers.
539 unsigned tiledWidth = util::alignUp(mFrameWidth, COARSE_TILE_SIZE);
540 unsigned tiledHeight = util::alignUp(mFrameHeight, COARSE_TILE_SIZE);
541
542 auto format = convertImageEncoding(mImageEncoding);
543 mTiledFrame.init(format, tiledWidth, tiledHeight);
544 mFinalFrame.init(format, mFrameWidth, mFrameHeight);
545
546 mPixelInfoTiledFrame.init(tiledWidth, tiledHeight);
547 mFinalPixelInfoFrame.init(mFrameWidth, mFrameHeight);
548
549 // Reinitialize our ROI buffers on any change
550 if (mUsingROI) {
551 mRoiPixelBuffer.init(convertImageEncoding(mImageEncoding), (unsigned)mViewport.width(), (unsigned)mViewport.height());
552 mRoiPixelInfoBuffer.init((unsigned)mViewport.width(), (unsigned)mViewport.height());
553 } else {
554 // We're not using ROI at the moment, so free the buffers that may have been allocated
555 mRoiPixelBuffer.cleanUp();
556 mRoiPixelInfoBuffer.cleanUp();
557 }
558
559 mLastTime = util::getSeconds();
560}
