50 //
51 // Crawl all pixels by pixel access function with tiled memory layout consideration.
52 // Pixel operation is defined by pixFunc argument.
53 // mPixels is non-tiled memory layout but this idea might be worth if pixFunc access tiled layout memory
54 // This template is only used for debug purpose so far.
55 //
56 template <typename F>
57 void
58 crawlAllPixels(F pixFunc) {
59 const OIIO::ImageSpec &spec = mSpec;
60 int totalTileY = (spec.height + 7) / 8;
61 int totalTileX = (spec.width + 7) / 8;
62
63#ifdef SINGLE_THREAD
64 constexpr bool parallel = false;
65#else
66 constexpr bool parallel = true;
67#endif
68 simpleLoop2<parallel>(0, totalTileY, [&](unsigned tileY) {
69 int startY = tileY * 8;
70 int endY = (startY + 8 < spec.height)? startY + 8: spec.height;
71
72 for (int tileX = 0; tileX < totalTileX; ++tileX) {
73 int startX = tileX * 8;
74 int endX = (startX + 8 < spec.width)? startX + 8: spec.width;
75
76 for (int pixY = startY; pixY < endY; ++pixY) {
77 for (int pixX = startX; pixX < endX; ++pixX) {
78 int pixCount = (spec.height - pixY - 1) * spec.width + pixX; // flip Y to access pix
79 int pixOffsetId = pixCount * spec.nchannels;
80 pixFunc(pixX, pixY, spec.width, &mPixels[pixOffsetId]);
81 } // pixX
82 } // pixY
83 } // tileX
84 });
85 }
86
87 //
88 // Crawl all pixels by scanline segment access function with tiled memory layout consideration.
89 // Scanline operation is defined by scanlineFunc argument.
90 // mPixels is non-tiled memory layout but this idea is very worth if scanlineFunc access tiled layout memory
91 // This template is used by RenderOutputDriver read API.
92 //
93 template <typename F>
94 void
95 crawlAllTiledScanline(F scanlineFunc) {
96 const OIIO::ImageSpec &spec = mSpec;
97 int totalTileY = (spec.height + 7) / 8;
98 int totalTileX = (spec.width + 7) / 8;
99
100#ifdef SINGLE_THREAD
101 constexpr bool parallel = false;
102#else
103 constexpr bool parallel = true;
104#endif
105 simpleLoop2<parallel>(0, totalTileY, [&](unsigned tileY) {
106 int startY = tileY * 8;
107 int endY = (startY + 8 < spec.height)? startY + 8: spec.height;
108
109 for (int tileX = 0; tileX < totalTileX; ++tileX) {
110 int startX = tileX * 8;
111 int endX = (startX + 8 < spec.width)? startX + 8: spec.width;
112
113 for (int pixY = startY; pixY < endY; ++pixY) {
114 int startPixCount = (spec.height - pixY - 1) * spec.width + startX; // flip Y to access pix
115 int startPixOffsetId = startPixCount * spec.nchannels;
116 scanlineFunc(startX, endX, pixY, spec.nchannels, &mPixels[startPixOffsetId]);
117 } // pixY
118 } // tileX
119 });
120 }
