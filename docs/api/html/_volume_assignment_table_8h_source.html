31// This class offers a fast way to track volume IDs through
32// instance primitive intersections.
33class VolumeIdFSM
34{
35public:
36 VolumeIdFSM();
37
38 // Adds a transition sequence to the FSM. A transition sequence
39 // is a sequence of instance primitives ending in a volumeId
40 void add(const std::vector<const geom::internal::Instance *> &sequence, int volumeId);
41
42 // Transition to instance "to"
43 // Returns next valid state (>= 0) based on this transition. If the
44 // transition is invalid, return -1.
45 int transition(int state, const geom::internal::Instance *to) const;
46
47 // Is this an end state?
48 bool isLeaf(int state) const;
49
50 // What is the volumeId of this state?
51 // If the state is a leaf state it will be the volumeId otherwise
52 // it will be -1.
53 int getVolumeId(int state) const;
54
55private:
56 struct Node
57 {
58 using InstanceId = const geom::internal::Instance *;
59 using NodeIdx = int;
60 using TransitionMap = std::unordered_map<InstanceId, NodeIdx>;
61
62 Node(): mVolumeId(-1) {}
63
64 int addTransition(InstanceId to, NodeIdx nodeIdx);
65 NodeIdx findTransition(InstanceId to) const;
66
67 // valid next nodes
68 TransitionMap mTransitions;
69
70 // If this node is a leaf node, mTransitions will be
71 // empty and this value is the volumeId
72 int mVolumeId;
73 };
74
75 std::vector<Node> mNodes;
76};

79/// This class serves the functionality of book keeping the number of
80/// unique volume assignments (part/material). It offers fast lookup
81/// from layer assignment id to volume id (0 to n-1 where n = number of
82/// unique volume assignments) and vise versa.
83class VolumeAssignmentTable
84{
85public:
86 void initLookupTable(const scene_rdl2::rdl2::Layer* layer);
87
88 void setFeatureSizes(geom::internal::Primitive* prim,
89 const std::vector<int>& volumeIds,
90 const float fs) const;
91
92 const scene_rdl2::rdl2::VolumeShader* lookupWithVolumeId(int volumeId) const {
93 return mVolumeShaders[mVolumeIdToAssignmentId[volumeId]];
94 }
95
96 const scene_rdl2::rdl2::VolumeShader* lookupWithAssignmentId(int assignmentId) const {
97 return mVolumeShaders[assignmentId];
98 }
99
100 const ShadowLinking& lookupShadowLinkingWithVolumeId(int volumeId) const {
101 return mShadowLinkings[mVolumeIdToAssignmentId[volumeId]];
102 }
103
104 size_t getVolumeCount() const { return mVolumeIdToAssignmentId.size(); }
105
106 const std::vector<int> &getVolumeIds(int assignmentId) const {
107 return mAssignmentIdToVolumeIds[assignmentId];
108 }
109
110 int getAssignmentId(int volumeId) const {
111 return mVolumeIdToAssignmentId[volumeId];
112 }
113
114 int getVisibilityMask(int volumeId) const {
115 return mVisibilityMasks[volumeId];
116 }
117
118 // For a given volumeId return true if volumeId is an instance.
119 // If true, then evaluate the instance xform at time, returning the
120 // result in primToRender.
121 bool evalInstanceXform(int volumeId, float time, Mat43 &primToRender) const;
122
123 // A state machine is used to compute volume ids from a sequence of
124 // instance primitive intersections
125 const VolumeIdFSM &getInstanceVolumeIds() const { return mInstanceVolumeIds; }
126
127private:
128 // In the case of instanced volumes, a layer assignment might
129 // correspond to multiple volume ids. Each instance primitive requires
130 // its own volume id.
131 std::vector<std::vector<int>> mAssignmentIdToVolumeIds;
132
133 // A given volume id can correspond to only a single assignment id.
134 // In the case of instancing, the assignment id is the assignment id
135 // of the reference primitive.
136 std::vector<int> mVolumeIdToAssignmentId;
137
138 // Given a volume id, what is its visibility mask?
139 std::vector<int> mVisibilityMasks;
140
141 // Based on assignment id
142 std::vector<const scene_rdl2::rdl2::VolumeShader*> mVolumeShaders;
143
144 // Based on assignment id
145 std::vector<ShadowLinking> mShadowLinkings;
146
147 // Given a volume id, there might be a motion xform stack due to
148 // instancing
149 std::vector<std::vector<MotionTransform>> mInstanceXforms;
150
151 VolumeIdFSM mInstanceVolumeIds;
152};
