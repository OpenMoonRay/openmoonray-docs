54static void
55sampleDistribution2D(const Distribution2D &dist, const FloatArray &r1, const FloatArray &r2,
56 FloatArray &u, FloatArray &v)
57{
58 uint32_t size = uint32_t(r1.size());
59
60 u.resize(size);
61 v.resize(size);
62
63 for (uint32_t i=0; i < size; ++i) {
64 Vec2f uv;
65 float pdf;
66 dist.sample(r1[i], r2[i], &uv, &pdf, TEXTURE_FILTER_NEAREST);
67 u[i] = uv[0];
68 v[i] = uv[1];
69 }
70}

203 template <typename Iter>
204 Distribution1D createDistribution(Iter first, Iter last)
205 {
206 Distribution1D dist(std::distance(first, last));
207 for (std::size_t i = 0; first != last; ++first, ++i) {
208 dist.setWeight(i, *first);
209 }
210 dist.tabulateCdf();
211 return dist;
212 }

215// Stochastically sample the distribution to estimate its mean/expected value.
216float
217stochasticDiscreteMean(const std::vector<float>& vals, unsigned samples)
218{
219 const Distribution1D dist = createDistribution(vals.begin(), vals.end());
220 scene_rdl2::util::Random rnd(0xdeadbeef);
221
222 float mean = 0.0f;
223 for (unsigned i = 0; i < samples; ++i) {
224 const std::size_t idx = dist.sampleDiscrete(rnd.getNextFloat());
225 testAssert(idx < vals.size(), "Invalid index");
226 mean += vals[idx];
227 }
228
229 return mean / static_cast<float>(samples);
230}
231
232// This function should return close to the sum of the values.
233// P(x1) * x1 / P(x1)
234// + P(x2) * x2 / P(x2)
235// + ...
236// + P(xn) * xn / P(xn)
237// = x1 + x2 + ... + xn
238float
239stochasticDiscreteSum(const std::vector<float>& vals, unsigned samples)
240{
241 const Distribution1D dist = createDistribution(vals.begin(), vals.end());
242 scene_rdl2::util::Random rnd(0xdeadbeef);
243
244 float mean = 0.0f;
245 for (unsigned i = 0; i < samples; ++i) {
246 float pdf;
247 const std::size_t idx = dist.sampleDiscrete(rnd.getNextFloat(), &pdf);
248 testAssert(idx < vals.size(), "Invalid index");
249 mean += vals[idx] / pdf;
250 }
251
252 return mean / static_cast<float>(samples);
253}

256// Calculate expected value of sampling from vals.
257float
258discreteExpectedValue(const std::vector<float>& vals)
259{
260 const float sum = std::accumulate(vals.begin(), vals.end(), 0.0f);
261 return std::accumulate(vals.begin(), vals.end(), 0.0f, [sum](float init, float x) { return init + x/sum * x; });
262}
263
264// Calculate variance of sampling from vals.
265float
266discreteVariance(const std::vector<float>& vals)
267{
268 const float sum = std::accumulate(vals.begin(), vals.end(), 0.0f);
269 const float mean = discreteExpectedValue(vals);
270
271 float s2 = 0.0f;
272 for (std::vector<float>::const_iterator it = vals.begin(); it != vals.end(); ++it) {
273 const float x = *it;
274 const float px = x/sum;
275 s2 += px * (x - mean)*(x - mean);
276 }
277
278 return s2;
279}

281// Using the central limit theorem, if we average enough results, we get a normal distribution.
282// We're going to use this fact for testing the distribution.
283void
284testDiscreteMean(const std::vector<float>& vals)
285{
286 const unsigned kNumSamples = 100000;
287
288 // Determine the actual mean and variance for the distribution.
289 const float mean = discreteExpectedValue(vals);
290 const float s2 = discreteVariance(vals);
291 const float s = scene_rdl2::math::sqrt(s2);
292
293 // The standard deviation of the mean is 1/sqrt(N) smaller than the
294 // standard deviation of the individual samples.
295 const float sOfMean = s / scene_rdl2::math::sqrt(static_cast<float>(kNumSamples));
296 const float testedMean = stochasticDiscreteMean(vals, kNumSamples);
297
298 // The mean should be within 2 standard deviations about 95% of the time.
299 // This means that this test is _expected_ to fail about 5% of the time!
300 const float rangeMin = mean - 2.0f * sOfMean;
301 const float rangeMax = mean + 2.0f * sOfMean;
302
303 testAssert(testedMean >= rangeMin && testedMean <= rangeMax, "Mean not in expected range (this test will fail 5 percent of the time)");
304}

306// Place samples into buckets (this is easy for sampling discrete. They have to
307// be arbitrarily sized for sampling continuous). Use the distribution of in
308// the buckets in a chi-square test.
309void
310testDiscreteChiSquare(const std::vector<float>& vals, unsigned samples)
311{
312 if (vals.size() < 1) {
313 return;
314 }
315
316 const Distribution1D dist = createDistribution(vals.begin(), vals.end());
317 scene_rdl2::util::Random rnd(0xdeadbeef);
318
319 std::vector<unsigned> buckets(vals.size());
320
321 for (unsigned i = 0; i < samples; ++i) {
322 const std::size_t idx = dist.sampleDiscrete(rnd.getNextFloat());
323 testAssert(idx < vals.size(), "Invalid index");
324 ++buckets[idx];
325 }
326
327 float chiSquare = 0.0f;
328 for (std::size_t i = 0; i < buckets.size(); ++i) {
329 const float expectedNumSamples = dist.pdfDiscrete(i) * static_cast<float>(samples);
330 const float numSamplesObserved = static_cast<float>(buckets[i]);
331 const float d = numSamplesObserved - expectedNumSamples;
332 if (expectedNumSamples > 0) {
333 chiSquare += (d*d)/expectedNumSamples;
334 }
335 }
336
337 // For a large amount of samples, a chi-square distribution with b buckets
338 // approximates a normal distribution with a mean of b-1 and a variance of
339 // 2b-2.
340 const float chiSquareMean = static_cast<float>(buckets.size() - 1u);
341 const float chiSquareS2 = static_cast<float>(2u*buckets.size() - 2u);
342 const float chiSquareS = scene_rdl2::math::sqrt(chiSquareS2);
343
344 // The mean should be within 2 standard deviations about 95% of the time.
345 // This means that this test is _expected_ to fail about 5% of the time!
346 const float rangeMin = chiSquareMean - 2.0f * chiSquareS;
347 const float rangeMax = chiSquareMean + 2.0f * chiSquareS;
348
349 testAssert(chiSquare >= rangeMin && chiSquare <= rangeMax, "Chi-square not in expected range (this test will fail 5 percent of the time)");
350}
