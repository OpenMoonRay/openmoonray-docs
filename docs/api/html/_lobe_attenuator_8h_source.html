23// Abstract base class. Objects of this type serve to
24// handle the attenuation of a BsdfLobe or Bssrdf or VolumeSubsurface
25class LobeAttenuator
26{
27public:
28 LobeAttenuator():
29 mN(0.0f, 0.0f, 1.0f),
30 mOneMinusFresnel(nullptr)
31 {}
32
33 LobeAttenuator(const scene_rdl2::math::Vec3f& N):
34 mN(N),
35 mOneMinusFresnel(nullptr)
36 {}
37
38 // Make this an abstract class not be instantiated
39 virtual ~LobeAttenuator() =0;
40
41 // Default Attenuation Operators
42 // Can be Overridden by Child Attenuators
43 virtual BsdfLobe*
44 operator() (scene_rdl2::alloc::Arena * arena,
45 BsdfLobe* lobe) const
46 {
47 if (mOneMinusFresnel == nullptr)
48 return lobe;
49
50 // Default BSDF Attenuator
51 lobe = arena->allocWithArgs<UnderBsdfLobe>(lobe, mN);
52 lobe->setFresnel(mOneMinusFresnel);
53 return lobe;
54 }
55
56 virtual void
57 operator() (scene_rdl2::alloc::Arena * arena,
58 Bssrdf* bssrdf) const
59 {
60 if (mOneMinusFresnel == nullptr)
61 return;
62
63 // Default BSSRDF Attenuator
64 auto fresnel =
65 static_cast<MultipleTransmissionFresnel*>(bssrdf->getTransmissionFresnel());
66 // Assumes all BSSRDF's have a MultipleTransmissionFresnel attached.
67 MNRY_ASSERT(fresnel);
68 fresnel->add(mOneMinusFresnel);
69 }
70
71 virtual void
72 operator() (scene_rdl2::alloc::Arena * arena,
73 VolumeSubsurface* vs) const
74 {
75 if (mOneMinusFresnel == nullptr)
76 return;
77
78 // Default VolumeSubsurface Attenuator
79 auto fresnel =
80 static_cast<MultipleTransmissionFresnel*>(vs->getTransmissionFresnel());
81 // Assumes all VS's have a MultipleTransmissionFresnel attached.
82 MNRY_ASSERT(fresnel);
83 fresnel->add(mOneMinusFresnel);
84 }
85
86 LobeAttenuator(const LobeAttenuator& other) =delete;
87 LobeAttenuator& operator=(const LobeAttenuator& other) =delete;
88
89protected:
90 scene_rdl2::math::Vec3f mN;
91 Fresnel* mOneMinusFresnel;
92};
93
94// Compulsary definition of a pure virtual destructor to abstract the base class
95LobeAttenuator::~LobeAttenuator() {}

152class VelvetAttenuator : public LobeAttenuator
153{
154public:
155 VelvetAttenuator(scene_rdl2::alloc::Arena *arena,
156 const VelvetBRDF& velvet,
157 float weight) :
158 LobeAttenuator(velvet.getN())
159 {
160 if (velvet.getUseAbsorbingFibers() == false) {
161 // Lets more energy through based on the velvet color
162 const scene_rdl2::math::Color& c = velvet.getColor();
163 weight *= scene_rdl2::math::max(c.r,
164 scene_rdl2::math::max(c.g,
165 c.b));
166 }
167 mOneMinusFresnel =
168 arena->allocWithArgs<OneMinusVelvetFresnel>(velvet.getRoughness());
169
170 mOneMinusFresnel->setWeight(weight);
171 }
172
173 VelvetAttenuator(const VelvetAttenuator& other) =delete;
174 VelvetAttenuator& operator=(const VelvetAttenuator& other) =delete;
175
176 ~VelvetAttenuator() override {}
177
178 // Use Standard Attenuation Operations
179 using LobeAttenuator::operator();

182class ClearcoatAttenuator : public LobeAttenuator
183{
184public:
185 ClearcoatAttenuator(
186 scene_rdl2::alloc::Arena *arena,
187 const scene_rdl2::math::Vec3f& N,
188 float roughness,
189 float etaI,
190 float etaT,
191 bool refracts,
192 float thickness,
193 const scene_rdl2::math::Color& attenuationColor,
194 Fresnel * fresnel) :
195 LobeAttenuator(N),
196 mEtaI(etaI),
197 mEtaT(etaT),
198 mRefracts(refracts),
199 mThickness(thickness),
200 mAttenuationColor(attenuationColor)
201 {
202 if (scene_rdl2::math::isZero(roughness)) {
203 mOneMinusFresnel = arena->allocWithArgs<OneMinusFresnel>(fresnel);
204 } else {
205 mOneMinusFresnel =
206 arena->allocWithArgs<OneMinusRoughFresnel>(fresnel, roughness);
207 }
208 }
209
210 ClearcoatAttenuator(const ClearcoatAttenuator& other) =delete;
211 ClearcoatAttenuator& operator=(const ClearcoatAttenuator& other) =delete;
212
213 ~ClearcoatAttenuator() override {}
214
215 // Use Standard Attenuation Operations
216 using LobeAttenuator::operator();
217
218 // Overrride BSDF Attenuation
219 BsdfLobe*
220 operator() (scene_rdl2::alloc::Arena * arena, BsdfLobe* lobe) const override
221 {
222 if (mRefracts && !scene_rdl2::math::isEqual(mEtaI, mEtaT)) {
223 const bool isReflectionLobe = lobe->matchesFlag(BsdfLobe::REFLECTION);
224 if (isReflectionLobe) {
225 lobe = arena->allocWithArgs<UnderClearcoatBsdfLobe>(
226 lobe, arena, mN,
227 mEtaI, mEtaT,
228 mThickness, mAttenuationColor);
229 } else {
230 // we are wrapping a transmission lobe
231 lobe = arena->allocWithArgs<UnderClearcoatTransmissionBsdfLobe>(
232 lobe, arena, mN,
233 mEtaI, mEtaT,
234 mThickness, mAttenuationColor);
235 }
236 } else {
237 // no refraction
238 lobe = arena->allocWithArgs<UnderBsdfLobe>(lobe,
239 mN,
240 mThickness,
241 mAttenuationColor);
242 }
243
244 lobe->setFresnel(mOneMinusFresnel);
245
246 return lobe;
247 }
248
249 // We don't currently support doing proper clearcoat refraction/absorption
250 // with Subsurface - but for now we at least use standard attenuation
251
252private:
253 float mEtaI;
254 float mEtaT;
255 bool mRefracts;
256 float mThickness;
257 scene_rdl2::math::Color mAttenuationColor;
258};
