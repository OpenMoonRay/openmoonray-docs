33 // Compute signed interocular offset due to stereo (translation in camera
34 // space along x axis).
35 // Warning: This function doesn't work in CENTER mode.
36 finline float
37 computeInterocularOffset(StereoView stereoView,
38 float interocularDistance)
39 {
40 MNRY_ASSERT(stereoView != StereoView::CENTER);
41
42 float interocularOffset =
43 (stereoView == StereoView::LEFT ? -0.5f : 0.5f) * interocularDistance;
44
45 return interocularOffset;
46 }

98 void DomeMaster3DCamera::updateImpl(const Mat4d& world2render)
99 {
100 mStereoView = getStereoViewImpl();
101 mInterocularOffset = (mStereoView == StereoView::CENTER) ? 0 :
102 computeInterocularOffset(mStereoView, getRdlCamera()->get(sStereoInterocularDistanceKey));
103
104 mImageResolutionWidthReciprocal = 1.0f / getApertureWindowWidth();
105 mImageResolutionHeightReciprocal = 1.0f / getApertureWindowHeight();
106
107 // Convert to Radians to use w/ trig functions...
108 mFOVHorizontalAngleRadians = scene_rdl2::math::degreesToRadians(getRdlCamera()->get(sFOVHorizontalAngleKey));
109 mFOVVerticalAngleRadians = scene_rdl2::math::degreesToRadians(getRdlCamera()->get(sFOVVerticalAngleKey));
110
111 mZenithMode = getRdlCamera()->get( sZenithModeKey );
112 mFlipRayX = getRdlCamera()->get( sFlipRayXKey );
113 mFlipRayY = getRdlCamera()->get( sFlipRayYKey );
114
115 mParallaxDistance = getRdlCamera()->get( sParallaxDistanceKey );
116 mInterocularDistanceFileName = getRdlCamera()->get(sCameraSeparationMapFileNameKey);
117
118 texture::TextureSampler* textureSampler = texture::getTextureSampler();
119 mTextureOption.apply_gamma = false;
120 mTextureOption.swrap = OIIO::TextureOpt::Wrap::WrapClamp;
121 mTextureOption.twrap = OIIO::TextureOpt::Wrap::WrapClamp;
122 mTextureOption.interpmode = OIIO::TextureOpt::InterpMode::InterpBilinear;
123 mOIIOTextureSystem = textureSampler->getTextureSystem();
124 std::string errorString;
125 if (mInterocularDistanceFileName != "") {
126 mTextureHandle =
127 textureSampler->getHandle(mInterocularDistanceFileName,
128 errorString,
129 mOIIOTextureSystem->get_perthread_info() );
130 if (mTextureHandle == nullptr) {
131 getRdlCamera()->error("FATAL: DomeMaster3DCamera failed to open texture file \"" ,
132 mInterocularDistanceFileName, "\" (" , errorString ,
133 ") at line: %d", __LINE__);
134 }
135 }

138 void
139 DomeMaster3DCamera::computePhiAndTheta(float x,
140 float y,
141 float& sinPhi,
142 float& cosPhi,
143 float& sinTheta,
144 float& cosTheta) const {
145 // remap pixel coordinates to new domain: [-1,-1] and [1,1]...
146 x = 2.0 * x * mImageResolutionWidthReciprocal - 1.0;
147 y = 2.0 * y * mImageResolutionHeightReciprocal - 1.0;
148
149 double phi, theta;
150 // convert to spherical coordinates...(bounded by Horizontal & Vertical FOV)
151 phi = x * (mFOVHorizontalAngleRadians / 2.0);
152 if (mZenithMode) {
153 theta = M_PI_2 - y * (mFOVVerticalAngleRadians / 2.0);
154 } else {
155 theta = y * (mFOVVerticalAngleRadians / 2.0);
156 }
157 sincos(phi, &sinPhi, &cosPhi);
158 sincos(theta, &sinTheta, &cosTheta);
159 }

183 void DomeMaster3DCamera::createRayImpl(mcrt_common::RayDifferential* dstRay,
184 float x,
185 float y,
186 float time,
187 float /*lensU*/,
188 float /*lensV*/) const
189 {
190 // Compute transforms
191 Mat4f ct2render; // "camera space at ray time" --> render space
192 if (getMotionBlur()) {
193 ct2render = computeCamera2Render(time);
194 } else {
195 time = 0.0f;
196 ct2render = getCamera2Render();
197 }
198
199 float sinPhi, cosPhi, sinTheta, cosTheta, tmp;
200 computePhiAndTheta(x, y, sinPhi, cosPhi, sinTheta, cosTheta);
201
202 float interocularMapDistanceColor[3] = { 1.0, 1.0, 1.0 };
203 if (mTextureHandle != nullptr) {
204 mOIIOTextureSystem->texture (mTextureHandle,
205 mOIIOTextureSystem->get_perthread_info(),
206 mTextureOption,
207 ( x * mImageResolutionWidthReciprocal ),
208 ( y * mImageResolutionHeightReciprocal ),
209 mImageResolutionWidthReciprocal, 0.0,
210 0.0, mImageResolutionHeightReciprocal,
211 mImageSpec.nchannels, interocularMapDistanceColor );
212 }
213 Vec3f eyeOrigin( mInterocularOffset * interocularMapDistanceColor[0], 0.0, 0.0 );
214 if (mStereoView != StereoView::CENTER) {
215 // This section causes the left & right eyes to "rotate" about the up-axis
216 // Eg. As we look 90 degrees to left, the left eye comes to the origin in X
217 // and moves backward in Z by .5 the interocular distance...
218 if (mZenithMode) {
219 tmp = eyeOrigin.x * cosPhi - eyeOrigin.y * sinPhi;
220 eyeOrigin.y = (eyeOrigin.y * cosPhi + eyeOrigin.x * sinPhi);
221 eyeOrigin.x = tmp;
222 } else {
223 tmp = eyeOrigin.x * cosPhi - eyeOrigin.z * sinPhi;
224 eyeOrigin.z = (eyeOrigin.z * cosPhi + eyeOrigin.x * sinPhi);
225 eyeOrigin.x = tmp;
226 }
227 }
228
229#if 0
230 Vec3f rayDirection = createDirection(eyeOrigin, x, y);
231 // Head Tilt Support
232 Vec3f headTarget( sinPhi * sinTheta,
233 -cosPhi * sinTheta,
234 rayDirection.z );
235 float headTilt = getRdlCamera()->get(sHeadTiltMapKey);
236 headTilt = (headTilt - 0.5) * M_PI;
237 Mat3f tilt;
238 tilt.setToRotation(headTarget, headTilt);
239 eyeOrigin = scene_rdl2::math::transformVector(tilt, eyeOrigin);
240#endif
241
242 flipXVector(eyeOrigin);
243 flipYVector(eyeOrigin);
244
245 const Vec3f cameraOrigin = transformPoint(ct2render, eyeOrigin);
246 *dstRay = mcrt_common::RayDifferential(cameraOrigin,
247 transformVector(ct2render, createDirection(eyeOrigin, x, y)),
248 cameraOrigin, transformVector(ct2render, createDirection(eyeOrigin,
249 x + 1.0f, y)),
250 cameraOrigin, transformVector(ct2render, createDirection(eyeOrigin,
251 x, y + 1.0f)),
252 getNear(), getFar(), time, 0);
253 }

260 Vec3f DomeMaster3DCamera::createDirection(const Vec3f& camOrigin, float x, float y) const
261 {
262 float sinPhi, cosPhi, sinTheta, cosTheta;
263 computePhiAndTheta(x, y,
264 sinPhi, cosPhi,
265 sinTheta, cosTheta);
266 Vec3f rayDirection;
267 if (mZenithMode) {
268 rayDirection = Vec3f(sinPhi * sinTheta, - cosPhi * sinTheta, -cosTheta);
269 } else {
270 rayDirection = Vec3f(sinPhi * cosTheta, sinTheta, - cosPhi * cosTheta);
271 }
272
273 applyParallax(rayDirection);
274
275 flipXVector(rayDirection);
276 flipYVector(rayDirection);
277
278 rayDirection -= camOrigin;
279 rayDirection.normalize();
280
281 return rayDirection;
282 }
