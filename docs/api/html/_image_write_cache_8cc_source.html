118void
119ImageWriteCacheImageSpec::updateHash(scene_rdl2::grid_util::Sha1Gen &sha1Gen) const
120{
121 sha1Gen.updateInt2(mWidth, mHeight);
122 sha1Gen.update<int>(mTotalNumChans);
123 sha1Gen.updateInt2(mSpecX, mSpecY);
124 sha1Gen.updateInt2(mSpecFullX, mSpecFullY);
125 sha1Gen.updateInt2(mSpecFullWidth, mSpecFullHeight);
126
127 for (size_t i = 0; i < mChanFormat.size(); ++i) {
128 sha1Gen.update<ChannelFormat>(mChanFormat[i]);
129 }
130 sha1Gen.updateStrVec(mChanNames);
131 if (!mName.empty()) {
132 sha1Gen.updateStr(mName);
133 }
134 sha1Gen.updateStr(mCompression);
135 if (isCompressionDwa()) {
136 sha1Gen.update<float>(mLevel);
137 }
138
139 if (!mAttrNames.empty()) {
140 sha1Gen.updateStrVec(mAttrNames);
141 sha1Gen.updateStrVec(mAttrTypes);
142 sha1Gen.updateStrVec(mAttrValues);
143 sha1Gen.updateStr(mMetaDataName);
144 }
145 if (!mResumeAttr.empty()) {
146 for (size_t i = 0; i < mResumeAttr.size(); i += 3) {
147 sha1Gen.updateStr3(mResumeAttr[i],
148 mResumeAttr[i+1],
149 mResumeAttr[i+2]);
150 }
151 }
152}

476bool
477ImageWriteCacheTmpFileItem::copyFile(const std::string &dstName, std::string &errMsg) const
478//
479// Copy tmpFile data to destination location but name is not a final dstName.
480// Destination filename is created based on dstName by genCopyDestName() into the same directory
481// of dstName. After this copyFile() is properly completed, we are going to do rename() and
482// create dstName at later stage.
483//
484{
485 MNRY_ASSERT(mTmpFileFd >= 0);
486
487 std::string copyDestName = genCopyDestName(dstName);
488
489 int dstFd = open(copyDestName.c_str(), O_WRONLY | O_CREAT, 0666);
490 if (dstFd < 0) {
491 errMsg = scene_rdl2::util::buildString("Could not create copy destination file '",
492 copyDestName.c_str(), "' ", strerror(errno));
493 return false;
494 }
495
496 struct stat stat_src;
497 if (fstat(mTmpFileFd, &stat_src) == -1) {
498 close(dstFd);
499 errMsg = scene_rdl2::util::buildString("Could not get tmpFile size for '",
500 copyDestName.c_str(), "' ", strerror(errno));
501 return false;
502 }
503
504 size_t sendSize = stat_src.st_size;
505 size_t sentSize = 0;
506 while (sendSize > 0) {
507 off_t pos = sentSize;
508 ssize_t size = sendfile(dstFd, mTmpFileFd, &pos, sendSize);
509 if (size == -1) {
510 close(dstFd);
511 errMsg = scene_rdl2::util::buildString("Failed tmpFile copy to destination location '",
512 copyDestName.c_str(), "' ", strerror(errno));
513 return false;
514 }
515 if (size == sendSize) break; // sendfile completed
516 sentSize += size;
517 sendSize -= size;
518 }
519
520 if (close(dstFd) == -1) {
521 errMsg = scene_rdl2::util::buildString("Failed close file of '", copyDestName.c_str(), "' ",
522 strerror(errno));
523 return false;
524 }
525
526# ifdef IMAGE_WRITE_DETAIL_MESSAGE
527 std::string msg = scene_rdl2::util::buildString("Copied: tmpFile to '", copyDestName.c_str(), "'");
528 scene_rdl2::logging::Logger::info(msg);
529 if (isatty(STDOUT_FILENO)) std::cout << msg << std::endl;
530# endif // end IMAGE_WRITE_DETAIL_MESSAGE
531
532 return true;
533}

564void
565ImageWriteCache::setupEnqCacheImageMemory(const int width, const int height)
566{
567 mWidth = width;
568 mHeight = height;
569
570 int tileAlignedHeight = (mHeight + 7) & ~7;
571
572 // I tested several different Y block size but same size of tile Y looks best
573 // If you want to change YblockSize, pick 8 (= tile size) * N would be best.
574 // This maximizes framebuffer access coherency because our framebuffer memory layout is tiled format.
575 mYBlockSize = 8 * 1;
576 mYBlockTotal = (tileAlignedHeight + mYBlockSize - 1) / mYBlockSize;
577
578 mDataFullArray.resize(mYBlockTotal);
579 mDataHalfArray.resize(mYBlockTotal);
580
581 mCacheQueueFullBuffArray.resize(mYBlockTotal);
582 mCacheQueueHalfBuffArray.resize(mYBlockTotal);
583
584 //------------------------------
585
586 for (int yBlockId = 0; yBlockId < mYBlockTotal; ++yBlockId) {
587 new(&mCacheQueueFullBuffArray[yBlockId].mCEnq) std::unique_ptr<scene_rdl2::cache::CacheEnqueue>;
588 mCacheQueueFullBuffArray[yBlockId].mCEnq.reset(new scene_rdl2::cache::CacheEnqueue(&mDataFullArray[yBlockId]));
589
590 new(&mCacheQueueHalfBuffArray[yBlockId].mCEnq) std::unique_ptr<scene_rdl2::cache::CacheEnqueue>;
591 mCacheQueueHalfBuffArray[yBlockId].mCEnq.reset(new scene_rdl2::cache::CacheEnqueue(&mDataHalfArray[yBlockId]));
592
593 // If we have blockData size info (i.e. this ImageWriteCache is not a 1st data and
594 // it is 2nd or later), we do pre allocate memory here in order to avoid unnecessary
595 // reallocation. However this preallocation should be executed after CacheEnqueue
596 // construction has been completed (See CacheEnqueue constructor for more detail).
597 if (mBlockDataSizeFull) {
598 mDataFullArray[yBlockId].resize(mBlockDataSizeFull);
599 }
600 if (mBlockDataSizeHalf) {
601 mDataHalfArray[yBlockId].resize(mBlockDataSizeHalf);
602 }
603 }
604}

606void
607ImageWriteCache::expandInternalDataIfNeeded(int yBlockId,
608 size_t addByteFull, // byte
609 size_t addByteHalf) // byte
610{
611 size_t currCapaFull, currCapaHalf; // byte
612 getCapacityInternalData(yBlockId, currCapaFull, currCapaHalf);
613
614 size_t currDataSizeFull = enqFullBuff(yBlockId)->currentSize(); // byte
615 size_t currDataSizeHalf = enqHalfBuff(yBlockId)->currentSize(); // byte
616
617 size_t remainFull = currCapaFull - currDataSizeFull; // byte
618 size_t remainHalf = currCapaHalf - currDataSizeHalf; // byte
619
620 // We only resize if needed.
621 if (remainFull < addByteFull) {
622 mDataFullArray[yBlockId].resize(currCapaFull + addByteFull - remainFull);
623 }
624 if (remainHalf < addByteHalf) {
625 mDataHalfArray[yBlockId].resize(currCapaHalf + addByteHalf - remainHalf);
626 }
627}

638void
639ImageWriteCache::freeInternalData()
640{
641 // Save cache buffer size for future preallocation of imageWriteCache
642 calcFinalBlockInternalDataSize();
643
644 // We don't need cache data itself and need to free here
645 // We should call shrink_to_fit() in order to exactly free internal memory of vectors.
646 mData.clear();
647 mData.shrink_to_fit();
648
649 mDataFullArray.clear();
650 mDataFullArray.shrink_to_fit();
651 mDataHalfArray.clear();
652 mDataHalfArray.shrink_to_fit();
653
654 mCacheQueueFullBuffArray.clear();
655 mCacheQueueFullBuffArray.shrink_to_fit();
656 mCacheQueueHalfBuffArray.clear();
657 mCacheQueueHalfBuffArray.shrink_to_fit();
658
659 mWidth = 0;
660 mHeight = 0;
661 mYBlockSize = 0;
662 mYBlockTotal = 0;
663
664 mImageWriteCacheSpecs.clear();
665 mImageWriteCacheSpecs.shrink_to_fit();
666
667 malloc_trim(0); // Return unused memory from malloc() arena to OS
668}

670void
671ImageWriteCache::calcFinalBlockInternalDataSize()
672{
673 if (mDataFullArray.size()) {
674 mBlockDataSizeFull = mDataFullArray[0].size();
675 }
676 if (mDataHalfArray.size()) {
677 mBlockDataSizeHalf = mDataHalfArray[0].size();
678 }
679
680# ifdef RUNTIME_VERIFY_BLOCKDATASIZE

794void
795ImageWriteCache::outputDataFinalize()
796//
797// for ImageWriteDriver
798//
799{
800 if (getRenderDriver()->getCheckpointController().isMemorySnapshotActive()) {
801 // signal-based checkpoint action enabled
802 ProcKeeper::ProcKeeperShPtr keeper = ProcKeeper::get();
803 if (keeper) {
804 keeper->startImageWrite(this); // Let keeper know image writing action started
805 }
806 }
807
808 //
809 // We have to consider two stage outputs and overwrite conditions.
810 // See comment of CheckpointController::fileOutputMain() for more detail.
811 //
812 mRenderOutputDriver->writeCheckpointDeq(this, false);
813 mRenderOutputDriver->loggingErrorAndInfo(this);
814
815 if (!mTwoStageOutput) {
816 if (!mCheckpointOverwrite) {
817 // non two stage mode multi version file output
818 mRenderOutputDriver->writeCheckpointDeq(this, true);
819 mRenderOutputDriver->loggingErrorAndInfo(this);
820 }
821 } else {
822 // TwoStage output is on and data was written out to tmpFile.
823 allFinalizeCheckpointFile(); // copy and rename for checkpoint file
824 if (!mCheckpointOverwrite) {
825 // copy and rename for checkpoint multi version file
826 allFinalizeCheckpointMultiVersionFile();
827 }
828 }
829 if (!mPostCheckpointScript.empty()) {
830 runPostCheckpointScript();
831 }
832
833 if (getRenderDriver()->getCheckpointController().isMemorySnapshotActive()) {
834 // signal-based checkpoint action enabled
835 ProcKeeper::ProcKeeperShPtr keeper = ProcKeeper::get();
836 if (keeper) {
837 keeper->finishImageWrite(); // Let keeper know image writing action finished
838# ifdef DEBUG_MSG_PROGRESS
839 std::cerr << ">> ImageWriteCache.cc outputDataFinalize() after finishImageWrite()\n";
840# endif // end DEBUG_MSG_PROGRESS
841 }
842 }

930void
931ImageWriteCache::timeUpdateBuffWrite(float fraction)
932{
933 ProgressStage stage;
934 if (fraction <= 0.0f) {
935 // We categorize very first fraction = 0.0 for buff stage as BUFF0.
936 // And categorize second or more fraction value 0.0 as BUFF1.
937 // We don't know exactly inside openimageio but BUFF0 is a compression stage and
938 // BUFF1 is a phase of beginning stage of write buffer action I guess.
939 stage = ((mCurrBuffWriteProgressCounter.load(std::memory_order_relaxed) == 0)
940 ? ProgressStage::BUFF0 : ProgressStage::BUFF1);
941 } else {
942 stage = ProgressStage::BUFF2;
943 }
944 mCurrBuffWriteProgressFraction = scene_rdl2::math::clamp(static_cast<int>(fraction * 10.0f), 0, 10);
945 mCurrBuffWriteProgressCounter.fetch_add(1, std::memory_order_relaxed);
946
947 updateProgress(stage);
948
949# ifdef DEBUG_MSG_PROGRESS
950 std::cerr << ">> ImageWriteCache.cc update:" << mRecTimeBuffWrite.end()
951 << " fraction:" << fraction << '\n';
952# endif // end DEBUG_MSG_PROGRESS
953}

1133bool
1134ImageWriteCache::allFinalizeFile(bool resumable,
1135 const TmpFileActionCallBack &copyCallBack,
1136 const TmpFileActionCallBack &renameCallBack) const
1137{
1138 if (resumable) {
1139 // This is a resumable output situation. Output files will be used for
1140 // input of resume rendering.
1141 // In order to reduce the risk of resume render failure, we do finalize by
1142 // copy-all and rename-all solution.
1143
1144 // First of all we are going to copy all tmp files to the destination temp files.
1145 if (!crawlAllTmpFileItems(copyCallBack)) {
1146 return false;
1147 }
1148
1149 // Then 2nd step is that rename all destination temp files to final name.
1150 if (!crawlAllTmpFileItems(renameCallBack)) {
1151 return false;
1152 }
1153
1154 } else {
1155 // This is a non resumable output situation.
1156 // We do minimize the risk of left over the part file (copied file) in the unexpected event.
1157 // We are using copy and rename for individual file and loop over all temporary files.
1158 bool err = false;
1159 crawlAllTmpFileItems([&](const ImageWriteCacheTmpFileItemShPtr item, std::string &errMsg) {
1160 // Even some of the tmpFile finalize failed (copy failed or rename failed),
1161 // we would like to continue other file's finalize.
1162 if (!copyCallBack(item, errMsg)) { err = true; return true; }
1163 if (!renameCallBack(item, errMsg)) { err = true; return true; }
1164 return true;
1165 });
1166 if (err) return false;
1167 }
1168
1169 return true;
1170}
1171
1172bool
1173ImageWriteCache::crawlAllTmpFileItems(const TmpFileActionCallBack &callBack) const
1174{
1175 if (!mTwoStageOutput) return true;
1176
1177 bool result = true;
1178 for (size_t id = 0; id < mTmpFileItemArray.size(); ++id) {
1179 std::string errMsg;
1180 if (!callBack(mTmpFileItemArray[id], errMsg)) {
1181 scene_rdl2::logging::Logger::error(errMsg);
1182 result = false;
1183 }
1184 }
1185 return result;
1186}
