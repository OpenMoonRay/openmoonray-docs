22// Constructor / Destructor
23FabricVelvetBsdfLobe::FabricVelvetBsdfLobe(const Vec3f &N,
24 const float roughness,
25 const Color& color) :
26 BsdfLobe(Type(REFLECTION | GLOSSY),
27 DifferentialFlags(IGNORES_INCOMING_DIFFERENTIALS),
28 /* isSpherical = */ false,
29 PROPERTY_COLOR | PROPERTY_PBR_VALIDITY |
30 PROPERTY_NORMAL | PROPERTY_ROUGHNESS),
31 mFrame(N),
32 mColor(color)
33{
34
35 mRoughness = clamp(roughness, 0.05f, 1.0f);
36
37 mSpecularExponent = 1.0f / mRoughness;
38 const float* nTable = reinterpret_cast<const float*>(ispc::BsdfFabricVelvet_normalizationTable());
39 const int index = static_cast<int>(scene_rdl2::math::floor(mRoughness*ispc::VELVET_NORMALIZATION_TABLE_SIZE - 1));
40 mNormalizationFactor = 1.0f / nTable[index];
41
42 // Derive a directional differential scale that varies according to width
43 mdDFactor = sdDFactorMin + mRoughness * sdDFactorSlope;
44}

46Color
47FabricVelvetBsdfLobe::eval(const BsdfSlice &slice,
48 const Vec3f &wi,
49 float *pdf) const
50{
51 if (pdf != NULL) {
52 *pdf = 0.0f;
53 }
54
55 const Vec3f N = mFrame.getN();
56
57 const float cosThetaWi = scene_rdl2::math::dot(N, wi);
58 if (cosThetaWi <= scene_rdl2::math::sEpsilon) return scene_rdl2::math::sBlack;
59
60 const float cosThetaWo = scene_rdl2::math::dot(N, slice.getWo());
61 if (cosThetaWo <= sEpsilon) return scene_rdl2::math::sBlack;
62
63 Vec3f H;
64 if (!computeNormalizedHalfVector(slice.getWo(), wi, H)) {
65 return sBlack;
66 }
67
68 // This is the Velvet BRDF for grazing angle highlights
69 // BRDF = sin(N, H)^exponent
70 const float cosNDotH = clamp(scene_rdl2::math::dot(N, H), -.99f, .99f);
71 const float sinSquared = (1.0f - cosNDotH*cosNDotH);
72 const float sinNDotH = scene_rdl2::math::sqrt(sinSquared);
73
74 float velvetFactor = mNormalizationFactor * scene_rdl2::math::pow(sinNDotH, mSpecularExponent);
75
76 // MOONSHINE-921
77 // Add an extra cosineThetaWi term to soften the shadow terminator
78 // Note - this breaks bidirectionality but artists prefer this look.
79 velvetFactor *= cosThetaWi;
80
81 if (pdf != NULL) {
82 *pdf = cosThetaWi * scene_rdl2::math::sOneOverPi;
83 }
84
85 const Color f = computeScaleAndFresnel(cosThetaWo) *
86 velvetFactor * mColor *
87 (slice.getIncludeCosineTerm() ? cosThetaWi : 1.0f);
88
89 return f;
90}

92Color
93FabricVelvetBsdfLobe::sample(const BsdfSlice &slice,
94 float r1, float r2,
95 Vec3f &wi,
96 float &pdf) const
97{
98 // cosine-weighted hemisphere sampling
99 wi = mFrame.localToGlobal(sampleLocalHemisphereCosine(r1, r2));
100 return eval(slice, wi, &pdf);
101}

104Color
105FabricVelvetBsdfLobe::albedo(const BsdfSlice &slice) const
106{
107 float cosThetaWo = scene_rdl2::math::max(dot(mFrame.getN(), slice.getWo()), 0.0f);
108 return (computeScaleAndFresnel(cosThetaWo) * mColor *
109 VelvetAlbedo::at(cosThetaWo, mRoughness));
110}

112// no surface curvature required (similar to the hair lobe derivatives)
113void
114FabricVelvetBsdfLobe::differentials(const Vec3f &wo, const Vec3f &wi,
115 float r1, float r2,
116 const Vec3f &dNdx, const Vec3f &dNdy,
117 Vec3f &dDdx, Vec3f &dDdy) const
118{
119 // It's complex to capture the full derivative. Instead we use the
120 // derivative of a mirror reflection about the H vector, and scale the
121 // length of the directional derivative proportionally with roughness.
122 Vec3f H;
123 if (!computeNormalizedHalfVector(wo, wi, H)) {
124 H = mFrame.getN();
125 }
126 computeReflectionDirectionDifferentialNoCurvature(wo, wi, H, dDdx, dDdy);
127
128 dDdx *= mdDFactor;
129 dDdy *= mdDFactor;
130}
