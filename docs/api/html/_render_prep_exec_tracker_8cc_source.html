292std::string
293RenderPrepExecTracker::Impl::show() const
294{
295 auto showCancelCodePosWithId = [&]() -> std::string {
296 std::ostringstream ostr;
297 ostr << showCancelCodePos(mCancelCodePos)
298 << " (cancelCodePos:" << static_cast<int>(mCancelCodePos) << ")";
299 return ostr.str();
300 };
301 auto showCurrentCancelCondition = [&]() -> std::string {
302 if (!mRenderPrepCancelCallBack) return "?";
303 if (mRenderPrepCancelCallBack()) return "cancel";
304 return "run";
305 };
306
307 std::ostringstream ostr;
308 ostr << "RenderPrepExecTracker {\n"
309 << " mRenderPrepStatsCallBack:" << (mRenderPrepStatsCallBack ? "set" : "empty") << '\n'
310 << " mRenderPrepCancelCallBack:" << (mRenderPrepCancelCallBack ? "set" : "empty")
311 << " status:" << showCurrentCancelCondition() << '\n'
312 << " mMsgHandlerCallBack:" << (mMsgHandlerCallBack ? "set" : "empty") << '\n'
313 << " mRunRenderPrep:" << showCondition(mRunRenderPrep) << '\n'
314 << " mRunApplyUpdate:" << showCondition(mRunApplyUpdate) << '\n'
315 << " mRunLoadGeom0:" << showCondition(mRunLoadGeom0) << '\n'
316 << " mRunLoadGeom1:" << showCondition(mRunLoadGeom1) << '\n'
317 << " mRunFinalizeChange0:" << showCondition(mRunFinalizeChange0) << '\n'
318 << " mRunFinalizeChange1:" << showCondition(mRunFinalizeChange1) << '\n'
319 << " mCancelCodePos:" << showCancelCodePosWithId() << '\n'
320 << "}";
321 return ostr.str();
322}

326RenderPrepExecTracker::Impl::RESULT
327RenderPrepExecTracker::Impl::updateRunStatus(const CancelCodePos &callerCodePos,
328 Condition &updateTarget,
329 const Condition &finishCondition,
330 const Condition &cancelCondition) const
331{
332 RESULT result = checkRunStatus(callerCodePos);
333 if (result == RESULT::FINISHED) {
334 updateTarget = finishCondition;
335 } else {
336 updateTarget = cancelCondition;
337 }
338
339 renderPrepStatsUpdate();
340
341 if (mMsgHandlerCallBack && result == RESULT::CANCELED) {
342 std::ostringstream ostr;
343 ostr << ">> RenderPrepExecTracker.cc RenderPrepExecTracker::Impl::updateRunStatus()"
344 << " callerCodePos:" << showCancelCodePos(callerCodePos)
345 << " result:" << showResult(result);
346 (mMsgHandlerCallBack)(ostr.str());
347# ifdef DEBUG_MSG
348 std::cerr << ostr.str() << '\n';
349# endif // end DEBUG_MSG
350 }
351
352 return result;
353}
354
355RenderPrepExecTracker::Impl::RESULT
356RenderPrepExecTracker::Impl::checkRunStatus(const CancelCodePos &callerCodePos) const
357{
358 if (callerCodePos == mCancelCodePos) {
359 // This is a case of cancelTest enable for debugging purpose.
360 std::cerr << ">> RenderPrepExecTracker.cc checkRunStatus debug CANCELED\n";
361 return RESULT::CANCELED;
362 }
363
364 if (!mRenderPrepCancelCallBack) {
365 // no cancel condition check -> always run
366 return RESULT::FINISHED;
367 }
368
369 if (mRenderPrepCancelCallBack()) {
370 return RESULT::CANCELED;
371 }
372
373 return RESULT::FINISHED;
374}

376void
377RenderPrepExecTracker::Impl::renderPrepStatsUpdate() const
378{
379 if (!mRenderPrepStatsCallBack) return;
380
381 scene_rdl2::grid_util::RenderPrepStats rPrepStats = calcRenderPrepStats();
382 (mRenderPrepStatsCallBack)(rPrepStats);
383

386# endif // end DEBUG_MSG
387}

389scene_rdl2::grid_util::RenderPrepStats
390RenderPrepExecTracker::Impl::calcRenderPrepStats() const
391//
392// convert this class's internal status to the RenderPrepStats which we need to set as
393// the argument of the callback function.
394//
395{
396 using RenderPrepStats = scene_rdl2::grid_util::RenderPrepStats;
397 using Stage = RenderPrepStats::Stage;
398
399 switch (mRunRenderPrep) {
400 case Condition::INIT : return RenderPrepStats(Stage::NOT_ACTIVE);
401 case Condition::START_CANCELED : return RenderPrepStats(Stage::RENDER_PREP_START_CANCELED);
402 case Condition::END : return RenderPrepStats(Stage::RENDER_PREP_DONE); // early exit
403 case Condition::END_CANCELED : return RenderPrepStats(Stage::RENDER_PREP_DONE_CANCELED); // early exit
404 default : break;
405 }
406
407 // renderPrep started
408 switch (mRunApplyUpdate) {
409 case Condition::INIT : return RenderPrepStats(Stage::RENDER_PREP_START);
410 case Condition::START : return RenderPrepStats(Stage::RENDER_PREP_APPLYUPDATE);
411 case Condition::START_CANCELED : return RenderPrepStats(Stage::RENDER_PREP_APPLYUPDATE_CANCELED);
412 case Condition::END_CANCELED : return RenderPrepStats(Stage::RENDER_PREP_APPLYUPDATE_DONE_CANCELED);
413 default : break;
414 }
415
416 // applyUpdate endded
417 switch (mRunLoadGeom0) {
418 case Condition::INIT : return RenderPrepStats(Stage::RENDER_PREP_APPLYUPDATE_DONE);
419 case Condition::START : return RenderPrepStats(Stage::RENDER_PREP_LOAD_GEOM0);
420 case Condition::START_CANCELED : return RenderPrepStats(Stage::RENDER_PREP_LOAD_GEOM0_CANCELED);
421 case Condition::END_CANCELED: return RenderPrepStats(Stage::RENDER_PREP_LOAD_GEOM0_DONE_CANCELED);
422 default : break;
423 }
424
425 // loadGeom0 ended
426 switch (mRunLoadGeom1) {
427 case Condition::INIT : return RenderPrepStats(Stage::RENDER_PREP_LOAD_GEOM0_DONE);
428 case Condition::START : return RenderPrepStats(Stage::RENDER_PREP_LOAD_GEOM1);
429 case Condition::START_CANCELED : return RenderPrepStats(Stage::RENDER_PREP_LOAD_GEOM1_CANCELED);
430 case Condition::END_CANCELED : return RenderPrepStats(Stage::RENDER_PREP_LOAD_GEOM1_DONE_CANCELED);
431 default : break;
432 }
433
434 // loadGeom1 ended
435 switch (mRunFinalizeChange0) {
436 case Condition::INIT : return RenderPrepStats(Stage::RENDER_PREP_LOAD_GEOM1_DONE);
437 case Condition::START : return RenderPrepStats(Stage::GM_FINALIZE0_START);
438 case Condition::START_CANCELED : return RenderPrepStats(Stage::GM_FINALIZE0_START_CANCELED);
439 case Condition::END_CANCELED : return RenderPrepStats(Stage::GM_FINALIZE0_DONE_CANCELED);
440 default : break;
441 }
442
443 // finalizeChange0 ended
444 switch (mRunFinalizeChange1) {
445 case Condition::INIT : return RenderPrepStats(Stage::GM_FINALIZE0_DONE);
446 case Condition::START : return RenderPrepStats(Stage::GM_FINALIZE1_START);
447 case Condition::START_CANCELED : return RenderPrepStats(Stage::GM_FINALIZE1_START_CANCELED);
448 case Condition::END_CANCELED : return RenderPrepStats(Stage::GM_FINALIZE1_DONE_CANCELED);
449 default : break;
450 }
451
452 // finalizeChange1 ended
453 return RenderPrepStats(Stage::GM_FINALIZE1_DONE);
454}

456void
457RenderPrepExecTracker::Impl::parseConfigure()
458{
459 mParse.description("renderPrepExecTracker command");
460
461 mParse.opt("cancelTest", "<posId>", "set test cancel action, posId = 0 disables cancelTest",
462 [&](Arg &arg) -> bool {
463 int posId = (arg++).as<int>(0);
464 int max = static_cast<int>(CancelCodePos::MAX);
465 if (posId < 0 || max <= posId) {
466 return arg.msg(std::string("WARNING : posId out of range:") + std::to_string(posId) +
467 " max:" + std::to_string(max - 1) + '\n');
468 }
469 mCancelCodePos = static_cast<CancelCodePos>(posId);
470 return true;
471 });
472 mParse.opt("cancelCodePosIdList", "", "show cancel code position id list",
473 [&](Arg &arg) -> bool { return arg.msg(showCancelCodePosIdList() + '\n'); });
474 mParse.opt("show", "", "show internal parameters",
475 [&](Arg &arg) -> bool { return arg.msg(show() + '\n'); });
476}

478std::string
479RenderPrepExecTracker::Impl::showCancelCodePosIdList() const
480{
481 std::ostringstream ostr;
482 int size = static_cast<int>(CancelCodePos::MAX);
483 int w = std::to_string(size).size();
484 ostr << "cancelCodePos id table (size:" << size << ") {\n";
485 for (int i = 0; i < size; ++i) {
486 ostr << " " << std::setw(w) << i << " : " << showCancelCodePos(static_cast<CancelCodePos>(i)) << '\n';
487 }
488 ostr << "}";
489 return ostr.str();
490}
