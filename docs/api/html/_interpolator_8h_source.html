27public:
28 template <typename T>
29 bool interpolate(TypedAttributeKey<T> key, char* data) const {
30 if (unlikely(!mAttr->isSupported(key))) {
31 return false;
32 }
33 bool isValidInterpolation = true;
34 AttributeRate rate = mAttr->getRate(key);
35 switch (rate) {
36 case RATE_CONSTANT:
37 interpolateConstant(key, data);
38 break;
39 case RATE_PART:
40 interpolatePart(key, data);
41 break;
42 case RATE_UNIFORM:
43 interpolateUniform(key, data);
44 break;
45 case RATE_VARYING:
46 interpolateVaryings(key, data);
47 break;
48 case RATE_FACE_VARYING:
49 interpolateFaceVaryings(key, data);
50 break;
51 case RATE_VERTEX:
52 interpolateVertices(key, data);
53 break;
54 default:
55 isValidInterpolation = false;
56 break;
57 }
58 return isValidInterpolation;
59 }
60
61protected:
62 Interpolator(const shading::Attributes *attr, float time, int part,
63 int coarseFace, int numVaryings,
64 const int *varyings,
65 const float *varyingWeights,
66 int numFaceVaryings,
67 const int *faceVaryings,
68 const float *faceVaryingWeights,
69 int tessellatedFace, int numVertices,
70 const int *vertices,
71 const float *vertexWeights) :
72 mAttr(attr),
73 mTime(time),
74 mPart(part),
75 mCoarseFace(coarseFace),
76 mNumVaryings(numVaryings),
77 mVaryings(varyings),
78 mVaryingWeights(varyingWeights),
79 mNumFaceVaryings(numFaceVaryings),
80 mFaceVaryings(faceVaryings),
81 mFaceVaryingWeights(faceVaryingWeights),
82 mTessellatedFace(tessellatedFace),
83 mNumVertices(numVertices),
84 mVertices(vertices),

88 template <typename T>
89 void interpolateConstant(TypedAttributeKey<T> key, char* data) const {
90 int timeSampleCount = mAttr->getTimeSampleCount(key);
91 if (timeSampleCount == 1) {
92 mAttr->getConstant(key, data);
93 } else {
94 // TODO only support two time samples for motion blur case right now
95 MNRY_ASSERT(timeSampleCount == 2);
96 T& result = *(reinterpret_cast<T*>(data));
97 result = (T)(
98 (1.0f - mTime) * mAttr->getConstant(key, 0) +
99 ( mTime) * mAttr->getConstant(key, 1));
100 }
101 }

106 if (timeSampleCount == 1) {

133 template <typename T>
134 void interpolateVaryings(TypedAttributeKey<T> key, char* data) const {
135 T& result = *(reinterpret_cast<T*>(data));
136 result = T(scene_rdl2::math::zero);
137 int timeSampleCount = mAttr->getTimeSampleCount(key);
138 if (timeSampleCount == 1) {
139 for (int i = 0; i < mNumVaryings; i++) {
140 int v = mVaryings[i];
141 float w = mVaryingWeights[i];
142 result = result + (T)(w * mAttr->getVarying(key, v));
143 }
144 } else {
145 // TODO only support two time samples for motion blur case right now
146 MNRY_ASSERT(timeSampleCount == 2);
147 for (int i = 0; i < mNumVaryings; i++) {
148 int v = mVaryings[i];
149 float w = mVaryingWeights[i];
150 result = result + (T)(w * (
151 (1.0f - mTime) * mAttr->getVarying(key, v, 0) +
152 ( mTime) * mAttr->getVarying(key, v, 1)));
153 }
154 }
155 }

222template <>
223inline bool Interpolator::interpolate(TypedAttributeKey<bool> key, char* data) const {
224 if (unlikely(!mAttr->isSupported(key))) {
225 return false;
226 }
227 bool isValidInterpolation = true;
228 MNRY_ASSERT(mAttr->getTimeSampleCount(key) == 1);
229 AttributeRate rate = mAttr->getRate(key);
230 switch (rate) {
231 case RATE_CONSTANT:
232 mAttr->getConstant(key, data);
233 break;
234 case RATE_PART:
235 mAttr->getPart(key, mPart, data);
236 break;
237 case RATE_UNIFORM:
238 mAttr->getUniform(key, mCoarseFace, data);
239 break;
240 default:
241 MNRY_ASSERT(false, (std::string("Bool Atttribute ") + std::string(key.getName()) +
242 std::string(" only supports constant/uniform rate\n")).c_str());
243 isValidInterpolation = false;
244 break;
245 }
246 return isValidInterpolation;
247}

277class QuadricInterpolator : public Interpolator
278{
279public:
280 QuadricInterpolator(const shading::Attributes *attr, float time, float u , float v):
281 Interpolator(attr, time,
282 0, // part
283 0, // coarseFace
284 4, // numVaryings
285 mIndices.data(), // varyings
286 mWeights.data(), // varyingWeights
287 4, // numFaceVaryings
288 mIndices.data(), // faceVaryings
289 mWeights.data(), // faceVaryingWeights
290 0, // tessellatedFace
291 4, // numVertices
292 mIndices.data(), // vertices
293 mWeights.data()) // vertexWeights
294 {
295 mWeights[0] = (1.0f - u) * (1.0f - v);
296 mWeights[1] = ( u) * (1.0f - v);
297 mWeights[2] = ( u) * ( v);
298 mWeights[3] = (1.0f - u) * ( v);
299 }
300
301private:
302 std::array<float, 4> mWeights;
303 static constexpr std::array<int, 4> mIndices = { {0, 1, 2, 3} };
304};
