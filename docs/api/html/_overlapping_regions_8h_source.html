20constexpr int roundUpDivision(int x, int y) noexcept
21{
22 return 1 + ((x - 1) / y);
23}
24
25inline int roundUp(int numToRound, int multiple) noexcept
26{
27 if (multiple == 0) {
28 return numToRound;
29 }
30
31 const int remainder = numToRound % multiple;
32 if (remainder == 0) {
33 return numToRound;
34 }
35
36 return numToRound + multiple - remainder;
37}

39/// @class OverlappingRegions
40///
41/// \tparam sRegionOverlap How many pixels the regions overlap
42/// \tparam sTileWidth How wide a tile is
43/// \tparam sTileHeight How high a tile is
44/// \tparam sDesiredNRegionsPerDimension The desired number of regions per dimension. E.g. "2" will attempt to subdivide
45/// the area into 2*2 = 4 regions.
46///
47/// OverlappingRegions divides an integral region into at most sDesiredNRegionsPerDimension^2 sub regions.
48/// These regions will overlap by sRegionOverlap.
49///
50/// This class is used by adaptive sampling. We divide into regions to reduce thread contention, and we overlap the
51/// regions to help eliminate seams caused by this subdivision.
52template <int sRegionOverlap, int sTileWidth, int sTileHeight, int sDesiredNRegionsPerDimension>
53class OverlappingRegions
54{
55public:
56 static_assert(sRegionOverlap % sTileWidth == 0, "Overlap must be a multiple of tile width");
57 static_assert(sRegionOverlap % sTileHeight == 0, "Overlap must be a multiple of tile height");
58
59 using IndexArray = std::array<int, 9>; // 9 to check all neighboring grid cells
60
61 OverlappingRegions() = default;
62
63 explicit OverlappingRegions(scene_rdl2::math::BBox2i bounds)
64 {
65 init(bounds);
66 }
67
68 // Expect bounds to be inclusive-exclusive in min and max [min, max).
69 void init(scene_rdl2::math::BBox2i bounds)
70 {
71 const int width = extents(bounds, 0);
72 const int height = extents(bounds, 1);
73
74 MNRY_ASSERT(width > 0);
75 MNRY_ASSERT(height > 0);
76
77 // +----------------+----------------+
78 // | | |
79 // +------------------------------+ |
80 // | | | |
81 // | | | |
82 // | | | |
83 // +---------------------------------+
84 // | | | |
85 // | | | |
86 // | | | |
87 // | | | |
88 // | | | |
89 // +----------------+-------------+--+
90 // 0, 0
91 // Inner box is the render bounds
92 // Outer box is the regions we create
93
94 // Since we want our regions to overlap, the smallest region we want
95 // should be a tile + our overlap.
96 constexpr int minRegionWidth = sTileWidth + sRegionOverlap;
97 constexpr int minRegionHeight = sTileHeight + sRegionOverlap;
98
99 if (minRegionWidth >= width || minRegionHeight >= height) {
100 mRegionWidth = width;
101 mRegionHeight = height;
102 mNumRegionsPerDimension = 1;
103 } else {
104 // We keep the regions numbers the same in each dimension, no matter the image ratio.
105 // This makes indexing easier.
106 int regionsPerDimension = std::min(width/minRegionWidth, height/minRegionHeight);
107 regionsPerDimension = std::min(sDesiredNRegionsPerDimension, regionsPerDimension);
108 regionsPerDimension = std::max(1, regionsPerDimension);
109
110 mRegionWidth = roundUpDivision(width, regionsPerDimension);
111 mRegionHeight = roundUpDivision(height, regionsPerDimension);
112
113 // Round the regions up to tile sizes. This makes the overlap calculation easier, and makes tile queries for
114 // a single region unambiguous.
115 mRegionWidth = roundUp(mRegionWidth, sTileWidth);
116 mRegionHeight = roundUp(mRegionHeight, sTileHeight);
117 mNumRegionsPerDimension = regionsPerDimension;
118 }
119
120 mOverallBounds = std::move(bounds);
121 }
122
123 int getNumRegions() const
124 {
125 return mNumRegionsPerDimension*mNumRegionsPerDimension;
126 }
127
128 /// Gets the bounding box (with overlap) for a region at a specific index.
129 /// \param idx The index of the bounding box in [0, getNumRegions).
130 /// \return The overlapping bounding box for a region at idx.
131 scene_rdl2::math::BBox2i getOverlappingRegionBounds(int idx) const
132 {
133 MNRY_ASSERT(idx >= 0);
134 MNRY_ASSERT(idx < getNumRegions());
135
136 int idxX;
137 int idxY;
138 getRegionCoords(idx, mNumRegionsPerDimension, idxX, idxY);
139
140 const scene_rdl2::math::Vec2i lower{idxX * mRegionWidth, idxY * mRegionHeight};
141 const scene_rdl2::math::Vec2i upper = lower + scene_rdl2::math::Vec2i{mRegionWidth, mRegionHeight};
142
143 scene_rdl2::math::BBox2i bounds{mOverallBounds.lower + lower, mOverallBounds.lower + upper};
144 bounds = enlarge(bounds, scene_rdl2::math::Vec2i(sRegionOverlap, sRegionOverlap));
145 return intersect(bounds, mOverallBounds);
146 }
147
148 /// Get the bounds of the entire region (i.e. the union of all sub-regions)
149 /// \return The bounds of the entire region represented by the instance of this class
150 const scene_rdl2::math::BBox2i& getOverallBounds() const
151 {
152 return mOverallBounds;
153 }
154
155 /// Get the (single) region index in which a point falls (ignoring region overlap)
156 /// \return The internal index for the region in which p falls
157 int getRegionIndex(scene_rdl2::math::Vec2i p) const
158 {
159 p -= mOverallBounds.lower;
160 const int x = p[0]/mRegionWidth;
161 const int y = p[1]/mRegionHeight;
162 return y * mNumRegionsPerDimension + x;
163 }
164
165 /// Get the (single) region index in which a tile falls (ignoring region overlap)
166 /// This implies that the region bounds are aligned to tile multiples
167 /// \return The internal index for the region in which the tile falls
168 int getRegionIndex(const scene_rdl2::math::BBox2i& tile) const
169 {
170 return getRegionIndex(tile.lower);
171 }
172
173 /// Get a list of all regions into which tile lies (including region overlap)
174 /// \param tile The tile to check
175 /// \param indices The return value. This array may have redundant entries. The entries are valid up until a
176 /// negative value is encountered.
177 void getOverlappedRegions(const scene_rdl2::math::BBox2i& tile, IndexArray& indices) const
178 {
179 const scene_rdl2::math::Vec2i& p = tile.lower;
180
181 int idx = 0;
182 for (int y : {-sRegionOverlap, 0, sRegionOverlap}) {
183 for (int x : {-sRegionOverlap, 0, sRegionOverlap}) {
184 const scene_rdl2::math::Vec2i toCheck = p + scene_rdl2::math::Vec2i{x, y};
185 if (conjointExclusive(mOverallBounds, toCheck)) {
186 indices[idx++] = getRegionIndex(toCheck);
187 }
188 }
189 }
190 // Mark the end of the array.
191 if (idx < 9) indices[idx] = -1;
192 }
193
194private:
195 /// Convert a one-dimensional sub-region index into x and y array values (for the lower point of the sub-region)
196 /// \param idx The sub-region index
197 /// \param numRegionsPerDimension The number of sub-regions per dimension (same in both dimensions)
198 /// \param x Return value of sub-region's lower x value
199 /// \param y Return value of sub-region's lower y value
200 static void getRegionCoords(int idx, int numRegionsPerDimension, int& x, int& y) noexcept
201 {
202 x = idx%numRegionsPerDimension;
203 y = idx/numRegionsPerDimension;
204 }
205
206 int mRegionWidth;
207 int mRegionHeight;
208 int mNumRegionsPerDimension;
209 scene_rdl2::math::BBox2i mOverallBounds;
210};
