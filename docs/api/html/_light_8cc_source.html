75Vec3f
76Light::slerpDirection(float time) const
77{
78 if (!(mMb & LIGHT_MB_ROTATION)) return mDirection;
79
80 const math::Quaternion3f q = math::slerp(mOrientation[0], mOrientation[1], time);
81 // row 2 of Mat3(q) - see Mat3.h
82 return math::Vec3f(2.0f * (q.i * q.k + q.r * q.j),
83 2.0f * (q.j * q.k - q.r * q.i),
84 1.0 - 2.0f * (q.i * q.i + q.j *q.j));
85}

120 if (!mRdlLight->hasChanged(scene_rdl2::rdl2::Light::sTextureKey)
121 && (mDistributionMapping == distributionMapping)
122 && !mRdlLight->hasChanged(scene_rdl2::rdl2::Light::sGammaKey)
123 && !mRdlLight->hasChanged(scene_rdl2::rdl2::Light::sContrastKey)
124 && !mRdlLight->hasChanged(scene_rdl2::rdl2::Light::sSaturationKey)
125 && !mRdlLight->hasChanged(scene_rdl2::rdl2::Light::sGainKey)
126 && !mRdlLight->hasChanged(scene_rdl2::rdl2::Light::sOffsetKey)
127 && !mRdlLight->hasChanged(scene_rdl2::rdl2::Light::sTemperatureKey)
128 && !mRdlLight->hasChanged(scene_rdl2::rdl2::Light::sTextureRotationKey)
129 && !mRdlLight->hasChanged(scene_rdl2::rdl2::Light::sTextureTranslationKey)
130 && !mRdlLight->hasChanged(scene_rdl2::rdl2::Light::sTextureCoverageKey)
131 && !mRdlLight->hasChanged(scene_rdl2::rdl2::Light::sTextureRepsUKey)
132 && !mRdlLight->hasChanged(scene_rdl2::rdl2::Light::sTextureRepsVKey)
133 && !mRdlLight->hasChanged(scene_rdl2::rdl2::Light::sTextureMirrorUKey)
134 && !mRdlLight->hasChanged(scene_rdl2::rdl2::Light::sTextureMirrorVKey)
135 && !mRdlLight->hasChanged(scene_rdl2::rdl2::Light::sTextureBorderColorKey)) {
136 return true;
137 }
138

140
141 delete mDistribution;

152 mRdlLight->get(scene_rdl2::rdl2::Light::sGammaKey),
153 mRdlLight->get(scene_rdl2::rdl2::Light::sContrastKey),
154 mRdlLight->get(scene_rdl2::rdl2::Light::sSaturationKey),
155 mRdlLight->get(scene_rdl2::rdl2::Light::sGainKey),
156 mRdlLight->get(scene_rdl2::rdl2::Light::sOffsetKey),
157 mRdlLight->get(scene_rdl2::rdl2::Light::sTemperatureKey),
158 mRdlLight->get(scene_rdl2::rdl2::Light::sTextureRotationKey),
159 mRdlLight->get(scene_rdl2::rdl2::Light::sTextureTranslationKey),
160 mRdlLight->get(scene_rdl2::rdl2::Light::sTextureCoverageKey),
161 mRdlLight->get(scene_rdl2::rdl2::Light::sTextureRepsUKey),
162 mRdlLight->get(scene_rdl2::rdl2::Light::sTextureRepsVKey),
163 mRdlLight->get(scene_rdl2::rdl2::Light::sTextureMirrorUKey),
164 mRdlLight->get(scene_rdl2::rdl2::Light::sTextureMirrorVKey),
165 mRdlLight->get(scene_rdl2::rdl2::Light::sTextureBorderColorKey));
166 } catch (scene_rdl2::except::KeyError &e) {
167 mRdlLight->error(e.what());
168 }
169
170 if (mDistribution == nullptr || !mDistribution->isValid()) {
171 delete mDistribution;
172 mDistribution = nullptr;
173 mRadiance = math::sBlack;
174 mOn = false;
175 return false;
176 }

178 return true;

230bool
231LocalParamLight::updateTransforms(const Mat4f &local2Render, int ti)
232{
233 // update transforms at time index (ti)
234 // ti == 0 is at normalized rayTime = 0.f
235 // ti == 1 is at normalized rayTime = 1.f;
236 MNRY_ASSERT(getRdlLight());
237
238 if (!extractUniformScale(local2Render, &mLocal2RenderScale[ti])) {
239 getRdlLight()->warn("Non-uniform scale on light - setting to off.");
240 mOn = false;
241 return false;
242 }
243 mRender2LocalScale[ti] = 1.0f / math::max(sEpsilon, mLocal2RenderScale[ti]);
244
245 Mat4f local2RenderRot = local2Render * Mat4f::scale(Vec4f(mRender2LocalScale[ti]));
246 if (!isOne(math::abs(local2RenderRot.det()), 0.01f)) {
247 getRdlLight()->warn("Invalid transform on light - setting to off.");
248 mOn = false;
249 return false;
250 }
251
252 mLocal2Render[ti] = Xform3f(Mat3f(asVec3(local2Render.row0()),
253 asVec3(local2Render.row1()),
254 asVec3(local2Render.row2())),
255 asVec3(local2Render.row3()));
256 // WARNING: this can be inaccurate, may lead to float precision jitter
257 mRender2Local[ti] = mLocal2Render[ti].inverse();
258
259 // WARNING: Includes translation, ok because only used with transformVector
260 mLocal2RenderRot[ti] = Xform3f(Mat3f(asVec3(local2RenderRot.row0()),
261 asVec3(local2RenderRot.row1()),
262 asVec3(local2RenderRot.row2())),
263 asVec3(local2RenderRot.row3()));
264
265 // WARNING: Includes translation, ok because only used with transformVector
266 // and also getXformRender2Local which uses the translation.
267 mRender2LocalRot[ti] = mRender2Local[ti] * Xform3f::scale(Vec3f(mLocal2RenderScale[ti]));
268
269 return true;
270}

272bool
273LocalParamLight::updateParamAndTransforms(const Mat4f &local2Render0,
274 const Mat4f &local2Render1,
275 float halfWidth,
276 float halfHeight)
277{
278 if (!updateTransforms(local2Render0, /* ti = */ 0)) return false;
279
280 mPosition[0] = mLocal2Render[0].row3();
281 mOrientation[0] = normalize(math::Quaternion3f(mLocal2RenderRot[0].row0(),
282 mLocal2RenderRot[0].row1(), mLocal2RenderRot[0].row2()));
283 mDirection = mLocal2RenderRot[0].row2();
284
285 // setup mMb
286 mMb = LIGHT_MB_NONE;
287 const scene_rdl2::rdl2::SceneVariables &vars =
288 getRdlLight()->getSceneClass().getSceneContext()->getSceneVariables();
289 const bool mb = vars.get(scene_rdl2::rdl2::SceneVariables::sEnableMotionBlur) &&
290 getRdlLight()->get(scene_rdl2::rdl2::Light::sMbKey) &&
291 !isEqual(local2Render0, local2Render1);
292
293 if (mb) {
294 if (!updateTransforms(local2Render1, /* ti = */ 1)) return false;
295
296 mPosition[1] = mLocal2Render[1].row3();
297 mOrientation[1] = normalize(math::Quaternion3f(mLocal2RenderRot[1].row0(),
298 mLocal2RenderRot[1].row1(), mLocal2RenderRot[1].row2()));
299 if (dot(mOrientation[0], mOrientation[1]) < 0) {
300 mOrientation[1] *= -1.f;
301 }
302
303 // set specific mb bits
304 if (!isEqual(mPosition[0], mPosition[1])) mMb |= LIGHT_MB_TRANSLATION;
305 if (!isEqual(mLocal2RenderRot[0].l, mLocal2RenderRot[1].l)) mMb |= LIGHT_MB_ROTATION;
306 if (!isEqual(mLocal2RenderScale[0], mLocal2RenderScale[1])) mMb |= LIGHT_MB_SCALE;
307 }
308
309 // Uv mapping
310 getScaleOffset(-halfWidth, halfWidth, 1.0f, 0.0f, &mUvScale.x, &mUvOffset.x);
311 getScaleOffset(-halfHeight, halfHeight, 1.0f, 0.0f, &mUvScale.y, &mUvOffset.y);
312
313 return true;
314}

318// point transformations
319Vec3f
320LocalParamLight::slerpPointLocal2Render(const Vec3f &p, float time) const
321{
322 // transformPoint(Xform3f x, Vec3f p) is p * x.l + x.t
323 const float s = (mMb & LIGHT_MB_SCALE) ?
324 lerp(mLocal2RenderScale[0], mLocal2RenderScale[1], time) :
325 mLocal2RenderScale[0];
326 const Mat3f r = (mMb & LIGHT_MB_ROTATION) ?
327 Mat3f(slerp(mOrientation[0], mOrientation[1], time)) :
328 mLocal2RenderRot[0].l;
329 const Vec3f t = (mMb & LIGHT_MB_TRANSLATION) ?
330 lerp(mPosition[0], mPosition[1], time) :
331 mPosition[0];
332
333 return (p * s) * r + t;
334}
335
336Vec3f
337LocalParamLight::slerpPointRender2Local(const Vec3f &p, float time) const
338{
339 // inverse of Local2Render is (p + t) * r * s
340 // where t = -position, r = render2localRot.l, s = render2localScale
341 const Vec3f t = (mMb & LIGHT_MB_TRANSLATION) ?
342 -1.f * lerp(mPosition[0], mPosition[1], time) :
343 -1.f * mPosition[0];
344 const Mat3f r = (mMb & LIGHT_MB_ROTATION)?
345 Mat3f(slerp(mOrientation[0], mOrientation[1], time)).transposed() : // inverse is transpose
346 mRender2LocalRot[0].l;
347 const float s = (mMb & LIGHT_MB_SCALE) ?
348 rcp(lerp(mLocal2RenderScale[0], mLocal2RenderScale[1], time)) :
349 mRender2LocalScale[0];
350
351 return ((p + t) * r) * s;
352}

408// normal transforms
409Vec3f
410LocalParamLight::slerpNormalLocal2Render(const Vec3f &n, float time) const
411{
412 // note: transformNormal in the math library simply does M * n,
413 // assuming it is passed the inverse of the desired transform.
414 // xformNormalLocal2Render means to transform a normal from local
415 // to render space.
416
417 // transformNormal is r * s * n
418 // where r is render2localRot and s is the render2localScale
419 const Mat3f r = (mMb & LIGHT_MB_ROTATION) ?
420 Mat3f(slerp(mOrientation[0], mOrientation[1], time)).transposed() : // inverse is transpose
421 mRender2LocalRot[0].l;
422 const float s = (mMb & LIGHT_MB_SCALE) ?
423 rcp(lerp(mLocal2RenderScale[0], mLocal2RenderScale[1], time)) :
424 mRender2LocalScale[0];
425
426 return r * (s * n);
427}
