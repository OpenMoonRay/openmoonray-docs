19/// @class DwaFabricBsdfLobe BsdfFabric.h <pbr/fabric/BsdfFabric.h>
20/// @brief DreamWorks specific fabric specular - (1 - |sin(thetaH)|)^N
21/// thetaH = 0.5*(thetaO+thetaI), theta wrt normal plane to hair tangent
22/// Results in really sharp specular peaks with a long tail, preferable for fabrics.
23
24// Precomputed Normalization Integrals for This Model
25// Precomputed List of 1 / (4*pi*integral (1-sin(x))^N*(1-2*sin^2(x))), x from 0 to pi/4
26// N from 0 to 30

53// Initialize Specular Exponent and NormalizationFactor
54void
55DwaFabricBsdfLobe::calculateNormalizationFactor()
56{
57 // Roughness remapping, [0,1] -> [1,30]
58 // Square the roughness for a visually linear effect
59 const float rgh = 1.0f - mRoughness;
60 mSpecularExponent = scene_rdl2::math::ceil(1.0f + 29.0f * rgh*rgh);
61
62 mNormalizationFactor =
63 precomputedDwaFabricNormalizationFactor[static_cast<int>(clamp(mSpecularExponent-1, 0.0f, 30.0f))];
64}

66// (1-|sin(thetaH)|)^N lobe
67Color
68DwaFabricBsdfLobe::eval(const BsdfSlice &slice,
69 const Vec3f &wi,
70 float *pdf) const
71{
72 // Prepare for early exit
73 if (pdf != nullptr) {
74 *pdf = 0.0f;
75 }
76 const float cosThetaWi = dot(mFrame.getN(), wi);
77 if (cosThetaWi <= sEpsilon) return sBlack;
78 const float cosThetaWo = dot(mFrame.getN(), slice.getWo());
79 if (cosThetaWo <= sEpsilon) return sBlack;
80
81 Vec3f H;
82 if (!computeNormalizedHalfVector(slice.getWo(), wi, H)) {
83 return scene_rdl2::math::sBlack;
84 }
85
86 const float cosHWo = dot(H, slice.getWo());
87 if (cosHWo < sEpsilon) return scene_rdl2::math::sBlack;
88
89 // thetaH is wrt the normal plane, so this dot product equals cos(pi/2 - thetaH) = sin(thetaH)
90 const float sinThetaH = clamp(dot(H, mThreadDirection), -.99f, .99f);
91 const float oneMinusAbsSinThetaH = (1.0f - fabs(sinThetaH));
92 if (oneMinusAbsSinThetaH < sEpsilon) return scene_rdl2::math::sBlack;
93 const float oneMinusAbsSinThetaHPow = scene_rdl2::math::pow(oneMinusAbsSinThetaH, mSpecularExponent);
94
95 const float fabricBRDF = mNormalizationFactor * oneMinusAbsSinThetaHPow;
96
97 // PDF for sampling theta_h is (n+1)/pi * powf(1-fabs(sinThetaH), n)
98 if (pdf != nullptr) {
99 *pdf = (mSpecularExponent+1.0f)*sOneOverPi*oneMinusAbsSinThetaHPow;
100 // Convert the PDF from sampling thetaH to sampling thetaI
101 // Divide by Jacobian (dOmegaI/dOmegaH)
102 *pdf *= 0.25 * rcp(cosHWo);
103 }
104
105 const Color f = computeScaleAndFresnel(cosThetaWo) *
106 fabricBRDF * mColor *
107 (slice.getIncludeCosineTerm() ? cosThetaWi : 1.0f);
108
109 // Soften hard shadow terminator due to shading normals
110 const float Gs = slice.computeShadowTerminatorFix(mFrame.getN(), wi);
111
112 return Gs * f;
113}

115Color
116DwaFabricBsdfLobe::sample(const BsdfSlice &slice,
117 float r1, float r2,
118 Vec3f &wi,
119 float &pdf) const
120{
121 const float cosNO = dot(mFrame.getN(), slice.getWo());
122 if (cosNO <= sEpsilon) {
123 pdf = 0.0f;
124 return sBlack;
125 }
126
127 Vec3f omegaI;
128 sampleVector(slice.getWo(), r1, r2, omegaI);
129
130 // Convert back in Global Space
131 wi = normalize(mFrame.localToGlobal(omegaI));
132
133 return eval(slice, wi, &pdf);
134}

137void
138DwaFabricBsdfLobe::sampleVector(const Vec3f& wo,
139 float r1, float r2,
140 Vec3f& wi) const
141{
142 // Theta is wrt the normal plane
143 const float thetaO = safeASin((dot(mThreadDirection, wo)));
144
145 // We sample thetaH and use it to calculate thetaI
146 float thetaH = scene_rdl2::math::asin(1.0f - scene_rdl2::math::pow(r1, 1.0f/(mSpecularExponent+1.0f)));
147 if (r2 < 0.5f) thetaH *= -1.0f;
148
149 const float thetaI = 2.0f * thetaH - thetaO;
150 float cosThetaI, sinThetaI;
151 sincos(thetaI, &sinThetaI, &cosThetaI);
152
153 // Phi Sampling
154 const float phiM = sPi*r2;
155 float cosPhi, sinPhi;
156 sincos(phiM, &sinPhi, &cosPhi);
157
158 // Theta wrt normal plane
159 wi.x = sinThetaI;
160 wi.y = cosThetaI * cosPhi;
161 wi.z = cosThetaI * sinPhi;
162}
