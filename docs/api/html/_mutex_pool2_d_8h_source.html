23 template <int log2MutexCount>
24 inline unsigned getMutexIndex(unsigned px, unsigned py) noexcept
25 {
26 constexpr int mutexCount = getMutexCount(log2MutexCount);
27
28 // This function allows us to look up which mutex to use from a
29 // power-of-two sized array. It will allow us to access the mutexes in a
30 // way that neighboring cells in a 2D grid don't access the same mutex.
31 static_assert(scene_rdl2::util::isPowerOfTwo(mutexCount), "Must be a power of two");
32
33 // The "& ~1" isn't strictly necessary, but it generates more efficient code on every value I've tested. It
34 // makes the number even.
35 // The "| 1" makes the number odd, so that it's co-prime with the power of two mutexCount.
36 const unsigned idx = py * (mutexCount/(log2MutexCount & ~1) | 1) + px;
37
38 // mutexCount should be a power of two. The compiler should change
39 // this mod operation into a mask.
40 return idx % mutexCount;
41 }
