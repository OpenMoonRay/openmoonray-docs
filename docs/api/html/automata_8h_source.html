74/// This struct represent a wildcard (for wildcard transitions)
75/// but it is NOT the transition itself. Light path expressions also
76/// use this class to create the transitions. A wildcard matches
77/// any symbol which is not listed in its black list (m_minus).
78struct Wildcard {
79 Wildcard() {};
80 Wildcard(SymbolSet &minus):m_minus(minus) {};
81
82 bool matches(ustring symbol)const
83 {
84 // true if the given symbol is not in m_minus
85 return (m_minus.find(symbol) == m_minus.end());
86 };
87
88 // Black list of unincluded symbols
89 SymbolSet m_minus;
90};

174 /// return all the symbols that lead to other states starting
175 /// in any of the states in the given set. Also if there are
176 /// wildcard transitions, give back a new wildcard that guarantees
177 /// that its matches match all the wildcards in the set.
178 /// So the symbols returned in out_symbols do NOT overlap those that
179 /// match the returned wildcard (if present). And the union of out_symbols
180 /// and those matched by the wildcard, are all the valid transitions from
181 /// the given state set

216/// Deterministic Finite Automata
217///
218/// This is a ready to use for parsing finite state automata where every
219/// symbol takes you to a single next state at most. It allows for linear
220/// time parsing of light path expressions
221class DfAutomata {

225 class State {
226 friend class DfAutomata;
227 friend class DfOptimizedAutomata;
228 public:
229
230 State(int id)
231 {
232 m_id = id;
233 m_wildcard_trans = -1;
234 }
235
236 /// Get the transition for a symbol
237 //
238 /// Returns -1 if no transitions with that symbol. That means the
239 /// symbol is not recognized at this point of the automata
240 int getTransition(ustring symbol)const;
241
242 // Same semantics as in NdfAutomata::State
243 void addTransition(ustring symbol, State *state);
244 // WARNING: this has an optimized representation and has to be called
245 // always AFTER all the normal transitions have been added
246 void addWildcardTransition(Wildcard *wildcard, State *state);
247
248 /// Optimize transitions
249 ///
250 /// Sometimes the existing wildcard can be extended to wrap some of
251 /// the single symbol transitions. This function performs that optimization
252 void removeUselessTransitions();
253
254 void addRule(void *rule) { m_rules.push_back(rule); };
255 const RuleSet &getRules()const { return m_rules; };
256 int getId()const { return m_id; };
257 std::string tostr()const;
258
259 protected:
260
261 int m_id;
262 // Only one transition per symbol here
263 SymbolToInt m_symbol_trans;
264 int m_wildcard_trans;
265 // A final state here might contain several final states
266 // from the original NDF automata. Therefore, we need a list
267 // of rules here
268 RuleSet m_rules;
269 };
