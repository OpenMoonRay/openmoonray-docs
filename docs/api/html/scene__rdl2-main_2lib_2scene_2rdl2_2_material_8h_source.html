118 virtual void setFataled(bool fataled) {
119 if (fataled) {
120 mOriginalShadeFunc = mShadeFunc;
121 mOriginalShadeFuncv = mShadeFuncv;
122 mShadeFunc = mSceneClass.getSceneContext()->getFatalShadeFunc();
123 mShadeFuncv = nullptr;
124 mOriginalPresenceFunc = mPresenceFunc;
125 mPresenceFunc = mSceneClass.getSceneContext()->getFatalPresenceFunc();
126 mOriginalIorFunc = mIorFunc;
127 mIorFunc = mSceneClass.getSceneContext()->getFatalIorFunc();
128 mOriginalPreventLightCullingFunc = mPreventLightCullingFunc;
129 mPreventLightCullingFunc = mSceneClass.getSceneContext()->getFatalPreventLightCullingFunc();
130 } else {
131 // If we're no longer fataled and we stored away
132 // a shade func, restore it.
133 if (mOriginalShadeFunc != nullptr) {
134 mShadeFunc = mOriginalShadeFunc;
135 }
136 if (mOriginalShadeFuncv != nullptr) {
137 mShadeFuncv = mOriginalShadeFuncv;
138 }
139 if (mOriginalPresenceFunc != nullptr) {
140 mPresenceFunc = mOriginalPresenceFunc;
141 }
142 if (mOriginalIorFunc != nullptr) {
143 mIorFunc = mOriginalIorFunc;
144 }
145 if (mOriginalPreventLightCullingFunc != nullptr) {
146 mPreventLightCullingFunc = mOriginalPreventLightCullingFunc;
147 }
148 }
149 }
