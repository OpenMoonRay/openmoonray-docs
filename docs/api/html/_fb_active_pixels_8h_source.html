162finline FbActivePixels::FbActivePixelsAovShPtr &
163FbActivePixels::getAov(const std::string &aovName)
164{
165 std::lock_guard<std::mutex> lock(mMutex);
166
167 if (mActivePixelsRenderOutput.find(aovName) == mActivePixelsRenderOutput.end()) {
168 // Very first time to access this AOV -> create
169 mActivePixelsRenderOutput[aovName] = FbActivePixelsAovShPtr(new FbActivePixelsAov(aovName));
170 }
171 mActivePixelsRenderOutput[aovName]->setActive();
172 mRenderOutputStatus = true;
173
174 return mActivePixelsRenderOutput[aovName];
175}

177finline void
178FbActivePixels::updateRenderOutputStatus(std::function<bool(const std::string &aovName, bool status)> evalStatusFunc)
179{
180 int totalActiveAov = 0;
181 for (auto &itr : mActivePixelsRenderOutput) {
182 if (evalStatusFunc((itr.second)->getAovName(), (itr.second)->getStatus())) {
183 totalActiveAov++;
184 } else {
185 (itr.second)->reset(); // mark as non-active (and not free internal memory yet)
186 }
187 }
188 mRenderOutputStatus = (totalActiveAov)? true: false;
189}

191finline void
192FbActivePixels::activeRenderOutputCrawler(std::function<void(const std::string &aovName,
193 const ActivePixels &activePixels)> func)
194{
195 for (auto &itr : mActivePixelsRenderOutput) {
196 if (!(itr.second)->getStatus()) continue; // skip non active aov
197 func((itr.second)->getAovName(), (itr.second)->getActivePixels());
198 }
199}

201finline void
202FbActivePixels::resetAllRenderOutput()
203{
204 for (auto &itr : mActivePixelsRenderOutput) {
205 (itr.second)->reset();
206 }
207 mRenderOutputStatus = false;
208}
