69//-----------------------------------------------------------------------------
70//
71// PixelTile class to manage deep pixels.
72//
73// This class is intended for simplifing common pixel-region loops that
74// step from pixel to pixel processing multiple channels simultaneously,
75// vs. using the IlmImfUtil deep classes which seem organized more for
76// per-plane processing and more targeted for deep texture use.
77//
78// See the DeepImageTile classes for an example that wraps the IlmImfUtil
79// DeepImage class.
80//
81//-----------------------------------------------------------------------------
82
83class DCX_EXPORT DeepTile : public PixelTile
84{
85 public:
86
87 //-------------------------------------------------------------------------------
88 //
89 // Hints at how the tile should be spatially accessed during writing of pixels.
90 // Some memory arrangements allow only sequential writing of the entire tile
91 // buffer in order while others want it restricted per-scanline.
92 //
93 //-------------------------------------------------------------------------------
94
95 enum WriteAccessMode
96 {
97 WRITE_DISABLED, // Cannot write to this tile
98 WRITE_RANDOM, // Any x,y pixel can be written to randomly
99 WRITE_RANDOM_SCANLINE, // Scanlines can be written randomly but pixels must be written in order (x argument ignored)
100 WRITE_SEQUENTIAL // Entire tile must be written in sequential order (x & y arguments ignored)
101 };
102
103
104 public:

108 // Data and display windows are set to invalid value [0 0 -1 -1]
109 //
110
111 DeepTile (ChannelContext& channel_ctx,

113 bool yAxisUp=true);
114
115 //
116 // Gets resolution info and channel set from header.
117 // The set of ChannelAliases is used to construct a map of ChannelIdx's to
118 // ChannelAliases. If multiple input ChannelAliases have the same ChannelIdx
119 // destination only the first one is accepted and the others are ignored.
120 //
121
122 DeepTile (const Imf::Header&,
123 ChannelContext& channel_ctx,
124 WriteAccessMode write_access_mode=WRITE_DISABLED,
125 bool yAxisUp=true);

131 // destination only the first one is accepted and the others are ignored.
132 //
133 // display_window is only used to set the window-top reference.
134 //
135
136 DeepTile (const IMATH_NAMESPACE::Box2i& display_window,
