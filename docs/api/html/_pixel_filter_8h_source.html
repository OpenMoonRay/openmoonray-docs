39class BoxPixelFilter final : public PixelFilter
40{
41public:
42 explicit BoxPixelFilter(float footprint) :
43 PixelFilter(footprint)
44 {
45 }
46
47private:
48 void applyFilter(utype /*dataSize*/, float* /*data*/) const override
49 {
50 // It's box...do nothing.
51 }
52
53 void applyOffset(utype dataSize, float* data) const override
54 {
55 const float widthMultiplier = getDesiredFootprint();
56
57 std::transform(data, data + dataSize, data,
58 [widthMultiplier](float x) {
59 // The mean is 0.5. To scale property, we need to transform to
60 // 0, scale, and then transform back.
61 return (x - 0.5f) * widthMultiplier + 0.5f;
62 });
63 }
64};
