43PixelScramble::Container PixelScramble::generate8ElementPermutations()
44{
45 Container values;
46 values.resize(s8Factorial);
47
48 for (std::uint32_t i = 0; i < s8Factorial; ++i) {
49 const Array8 factoradic = permutations::base10ToFactoradic(i);
50 const Array8 permutation = permutations::convertFactoradicNumberToPermutation(factoradic);
51 values.at(i) = permutation;
52 }
53 return values;
54}
