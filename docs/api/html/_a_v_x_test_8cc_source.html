18// Here is a very simple implementation which produces the correct results.
19// It serves as a baseline for performance and correctness.
20//
21
22void
23doAVXRefAOSToSOA( unsigned numElems,
24 const AOSData *__restrict aosData,
25 SOABlock8 *__restrict soaBlocks,
26 SortOrder *sortOrder,
27 scene_rdl2::alloc::Arena *arena,
28 Ticks *ticks )
29{
30 ticks->mPreSort = time::getTicks();
31
32 // Sorting phase:
33 std::sort(sortOrder, sortOrder + numElems, [](const SortOrder &a, const SortOrder &b) -> bool {
34 return a.mSortKey < b.mSortKey;
35 });
36
37 ticks->mPostSort = time::getTicks();
38
39 // Transposition phase:
40 for (unsigned i = 0; i < numElems; ++i) {
41
42 unsigned blockIdx = i >> AVX_VLEN_SHIFT;
43 unsigned laneIdx = i & AVX_VLEN_MASK;
44
45 const AOSData *aos = aosData + sortOrder[i].mElemIdx;
46
47 for (unsigned j = 0; j < sizeof(AOSData) / sizeof(uint32_t); ++j) {
48 soaBlocks[blockIdx].mData[j][laneIdx] = aos->mData[j];
49 }
50 }
51
52 // Smear final entry over trailing SOA entries.
53 if ((numElems & AVX_VLEN_MASK) != 0) {
54
55 SOABlock8 &finalSoa = soaBlocks[numElems >> AVX_VLEN_SHIFT];
56 unsigned finalLaneIdx = (numElems - 1) & AVX_VLEN_MASK;
57
58 for (unsigned i = 0; i < sizeof(AOSData) / sizeof(uint32_t); ++i) {
59 uint32_t ref = finalSoa.mData[i][finalLaneIdx];
60 for (unsigned j = finalLaneIdx + 1; j < AVX_VLEN; ++j) {
61 finalSoa.mData[i][j] = ref;
62 }
63 }
64 }
65
66 ticks->mPostTranspose = time::getTicks();
67}
