24inline float
25computeAreaHeuristic(Vec2f dx, Vec2f dy)
26{
27 // Treat dx and dy as edges of a parallelogram and compute the area.
28 float len = max(length(dx), 0.00000001f);
29 Vec2f normA = dx / len;
30
31 Vec2f perp(-normA.y, normA.x);
32 float height = fabsf(dot(perp, dy));
33 return len * height;
34}

36inline float
37computeMinHeuristic(Vec2f dx, Vec2f dy)
38{
39 float dx2 = lengthSqr(dx);
40 float dy2 = lengthSqr(dy);
41 return min(dx2, dy2);
42}

44inline float
45computeMaxHeuristic(Vec2f dx, Vec2f dy)
46{
47 float dx2 = lengthSqr(dx);
48 float dy2 = lengthSqr(dy);
49 return max(dx2, dy2);
50}

52inline float
53computeAvgHeuristic(Vec2f dx, Vec2f dy)
54{
55 float dx2 = lengthSqr(dx);
56 float dy2 = lengthSqr(dy);
57 return (dx2 + dy2) * 0.5f;
58}

60inline float
61computeCustomMipSelector(const Vec2f dx, const Vec2f dy)
62{
63#if DEFAULT_MIP_FILTER == MIP_FILTER_MIN
64
65 float driver = computeMinHeuristic(dx, dy);
66
67#elif DEFAULT_MIP_FILTER == MIP_FILTER_MAX
68
69 float driver = computeMaxHeuristic(dx, dy);
70
71#elif DEFAULT_MIP_FILTER == MIP_FILTER_AVG
72
73 float driver = computeAvgHeuristic(dx, dy);
74
75#elif DEFAULT_MIP_FILTER == MIP_FILTER_AREA
76
77 float driver = computeAreaHeuristic(dx, dy);
78
79#else
80
81 float driver = 0.f;
82
83#endif
84
85 float maxTexDimension = rsqrt(max(driver, 0.0000000001f));
86 float mipSelector = log2f(max(maxTexDimension, 1.f));
87
88 return clamp(mipSelector, 0.f, 15.999999f);
89}

91inline float
92computeOIIOMipSelector(Vec2f dx, Vec2f dy)
93{
94 float dsdx = dx.x;
95 float dtdx = dx.y;
96 float dsdy = dy.x;
97 float dtdy = dy.y;
98
99 //
100 // See adjust_width in texturesys.cpp:
101 //
102
103 // Clamp degenerate derivatives so they don't cause mathematical problems
104 static const float eps = 1.0e-8f, eps2 = eps*eps;
105 float dxlen2 = dsdx*dsdx + dtdx*dtdx;
106 float dylen2 = dsdy*dsdy + dtdy*dtdy;
107 if (dxlen2 < eps2) { // Tiny dx
108 if (dylen2 < eps2) {
109 // Tiny dx and dy: Essentially point sampling. Substitute a
110 // tiny but finite filter.
111 dsdx = eps; dsdy = 0;
112 dtdx = 0; dtdy = eps;
113 } else {
114 // Tiny dx, sane dy -- pick a small dx orthogonal to dy, but
115 // of length eps.
116 float scale = eps / sqrtf(dylen2);
117 dsdx = dtdy * scale;
118 dtdx = -dsdy * scale;
119 }
120 } else if (dylen2 < eps2) {
121 // Tiny dy, sane dx -- pick a small dy orthogonal to dx, but of
122 // length eps.
123 float scale = eps / sqrtf(dxlen2);
124 dsdy = -dtdx * scale;
125 dtdy = dsdx * scale;
126 }
127
128 //
129 // See TextureSystemImpl::texture_lookup_trilinear_mipmap in texturesys.cpp.
130 //
131
132 float sfilt = std::max (fabsf(dsdx), fabsf(dsdy));
133 float tfilt = std::max (fabsf(dtdx), fabsf(dtdy));
134
135#if DEFAULT_MIP_FILTER == OIIO_CONSERVATIVE
136 float filtwidth = std::max(sfilt, tfilt);
137#else
138 float filtwidth = std::min(sfilt, tfilt);
139#endif
140
141 float maxTexDimension = 1.f / max(filtwidth, 0.0000000001f);
142 float mipSelector = log2f(max(maxTexDimension, 1.f));
143
144 return clamp(mipSelector, 0.f, 15.999999f);
145}

152// A mip selector contains the lowest resolution mip level to look up and the
153// interpolation factor between them, like so:
154//
155// floor(mipSelector) max dimension of lowest mip
156//
157// 0 1
158// 1 2
159// 2 4
160// 3 8
161// 4 16
162// 5 32
163// 6 64
164// 7 128
165// 8 256
166// etc... etc...
167//
168// Lerp factor = frac(mipSelector)
169float
170computeMipSelector(float dsdx, float dtdx, float dsdy, float dtdy)
171{
172 Vec2f dx(dsdx, dtdx);
173 Vec2f dy(dsdy, dtdy);
174#if (DEFAULT_MIP_FILTER < START_OIIO_FILTER)
175 return computeCustomMipSelector(dx, dy);
176#elif (DEFAULT_MIP_FILTER == OIIO_CONSERVATIVE || DEFAULT_MIP_FILTER == OIIO_NON_CONSERVATIVE)
177 return computeOIIOMipSelector(dx, dy);
178#else
179 #error Invalid value set for DEFAULT_MIP_FILTER.
180#endif
181}
