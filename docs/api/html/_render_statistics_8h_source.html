412 enum class StopFrameTag : int {
413 WHOLE = 0, // 0:whole
414 MDRIVER_STOPFRAME, // 1:mDriver->stopFrame()
415 MPBRSCENE_POSTFRAME, // 2:mPbrScene->postFrame()
416 ACCUMULATE_PBR_STATS_DATA, // 3:accumulate pbr stats data
417 COLLECTSHADERSTATS, // 4:collectShaderStats()
418 REPORTSHADINGLOGS, // 5:reportShadingLogs()
419 LOG_FLUSH, // 6:log & flush() staff
420 RESET, // 7:reset
421
422 SIZE // total tag size
423 };
424
425 enum class RenderPrepTag : int {
426 WHOLE = 0, // 0:whole
427 START_UPDATE_PHASE_OF_FRAME, // 1:startUpdatePhaseOfFrame()
428 START_RENDERPREP, // 2:startRenderPrep()
429 RESET_STATISTICS, // 3:resetStatistics()
430 RESET_RENDER_OUTPUT_DRIVER, // 4:reset RenderOutputDriver
431 FLAG_STATUS_UPDATE, // 5:flag status update
432 RESET_SHADER_STATS_AND_LOGS, // 6:resetShaderStatsAndLogs
433 LOAD_GEOMETRIES, // 7:loadGeometries()
434 REPORT_GEOMETRY_MEMORY, // 8:reportGeometryMemory()
435 BUILD_MATERIAL_AOV_FLAGS, // 9:buildMaterialAovFlags()
436 BUILD_GEOMETRY_EXTENSIONS, // A:buildGeometryExtensions()
437 RESET_SHADER_STATS_AND_LOGS2, // B:resetShaderStatsAndLogs()
438 PBR_STATISTICS_RESET, // C:pbr statistics reset
439 UPDATE_PBR, // D:update PBR
440
441 SIZE // total tag size
442 };

444 void recTimeStart() { mRecTimeWhole.start(); mRecTime.start(); }

486// The RenderDriver keeps track of these.
487struct RealtimeFrameStats
488{
489 // How long did we spend updating the scene earlier in this frame.
490 double mUpdateDuration;
491
492 // Offset time during mUpdateDuration to adjust render timing
493 double mUpdateDurationOffset;
494
495 // How much time is left over for rendering after the time spent on the update,
496 // given our current FPS setting.
497 double mRenderBudget;
498
499 // Time which RenderDriver::realtimeRenderFrame() was invoked.
500 double mRenderFrameStartTime;
501
502 // Time that we started rendering the first realtime pass. (estimation phase)
503 double mFirstPassStartTime;
504
505 // Time that we completed the first realtime pass. (completed estimation phase)
506 double mFirstPassEndTime;
507
508 // The measured cost to render one sample for all tiles for the first pass.
509 // This includes queue draining time for the first pass so is likely a
510 // slight over estimation.
511 double mPredictedSampleCost;
512
513 // The measured cost to render one sample for all tiles for the rest of the
514 // frame. This should match up closely with mPredictedSampleCost
515 // (or mPredictedSampleCost should be computed more accurately).
516 double mActualSampleCost;
517
518 // The time we'll end rendering based on the sample cost of the first pass.
519 double mPredictedEndTime;
520
521 // The actual end time we've finished rendering (timing of isComplete() returns true inside
522 // RenderDriver::realtimeRenderFrame()'s while() loop.
523 // This should match up closely with mPredictedEndTime (or mPredictedEndTime should be computed
524 // more accurately).
525 double mActualEndTime;
526
527 // The measured average cost of non active duration of each render thread.
528 double mOverheadDuration;
529
530 // The number of passes we ended up rendering (including the "first" pass).
531 unsigned mNumRenderPasses;
532
533 // The total number of samples per tile we ended up rendering. This is
534 // constant for all tiles this frame.
535 unsigned mSamplesPerTile;
536
537 // The total number of samples for this frame.
538 unsigned mSamplesAll;
539};
