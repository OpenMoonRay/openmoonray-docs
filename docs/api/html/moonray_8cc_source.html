40void
41RaasCommandLineApplication::render(rndr::RenderContext & renderContext)
42{
43 // Don't even bother if we can't write the output file in non interactive mode
44
45 scene_rdl2::rdl2::SceneVariables& sceneVars = renderContext.getSceneContext().getSceneVariables();
46
47 if (!scene_rdl2::util::writeTest(sceneVars.get(scene_rdl2::rdl2::SceneVariables::sOutputFile), true)) {
48 std::ostringstream oss;
49 oss << "Output file '" <<
50 sceneVars.get(scene_rdl2::rdl2::SceneVariables::sOutputFile) <<
51 "' is not writable!";
52 scene_rdl2::Logger::error(oss.str());
53 throw scene_rdl2::except::IoError(oss.str());
54 }
55
56 std::shared_ptr<rndr::RenderDriver> driver = rndr::getRenderDriver();
57
58 renderContext.startFrame();
59
60 while (!renderContext.isFrameComplete()) {
61 // Poll for completion every 100 ms.
62 usleep(100000);
63 // TODO: Do we want a 'verbosity' setting. Maybe we don't want to print status always
64 printStatusLine(renderContext, driver->getLastFrameMcrtStartTime(), false);
65 }
66 // Inform the status line that we're done rendering.
67 printStatusLine(renderContext, driver->getLastFrameMcrtStartTime(), true);
68
69 renderContext.stopFrame();
70 renderOutput(renderContext);
71}

73void
74RaasCommandLineApplication::renderOutput(rndr::RenderContext &renderContext)
75{
76 scene_rdl2::fb_util::RenderBuffer outputBuffer;
77 renderContext.snapshotRenderBuffer(&outputBuffer, true, true);
78
79 // write the main output file
80 scene_rdl2::rdl2::SceneContext const &sceneContext = renderContext.getSceneContext();
81 std::string outputFile =
82 sceneContext.getSceneVariables().get(scene_rdl2::rdl2::SceneVariables::sOutputFile);
83 const scene_rdl2::rdl2::SceneObject *metadata = sceneContext.getSceneVariables().getExrHeaderAttributes();
84 const scene_rdl2::math::HalfOpenViewport aperture = renderContext.getRezedApertureWindow();
85 const scene_rdl2::math::HalfOpenViewport region = renderContext.getRezedRegionWindow();
86
87 int error = writeImageWithMessage(&outputBuffer, outputFile, metadata, aperture, region);
88
89 // write any arbitrary RenderOutput objects
90 const pbr::DeepBuffer *deepBuffer = renderContext.getDeepBuffer();
91 pbr::CryptomatteBuffer *cryptomatteBuffer = renderContext.getCryptomatteBuffer();
92 scene_rdl2::fb_util::HeatMapBuffer heatMapBuffer;
93 renderContext.snapshotHeatMapBuffer(&heatMapBuffer, /*untile*/ true, /*parallel*/ true); // internally only do snapshot when it has heatmapAOV
94 scene_rdl2::fb_util::FloatBuffer weightBuffer;
95 renderContext.snapshotWeightBuffer(&weightBuffer, /*untile*/ true, /*parallel*/ true); // internally only do snapshot when it has weightAOV
96 std::vector<scene_rdl2::fb_util::VariablePixelBuffer> aovBuffers;
97 renderContext.snapshotAovBuffers(aovBuffers, /*untile*/ true, /*parallel*/ true);
98 scene_rdl2::fb_util::RenderBuffer renderBufferOdd;
99 renderContext.snapshotRenderBufferOdd(&renderBufferOdd, /*untile*/ true, /*parallel*/ true);
100 std::vector<scene_rdl2::fb_util::VariablePixelBuffer> displayFilterBuffers;
101 renderContext.snapshotDisplayFilterBuffers(displayFilterBuffers, /*untile*/ true, /*parallel*/ true);
102 error += writeRenderOutputsWithMessages(renderContext.getRenderOutputDriver(),
103 deepBuffer, cryptomatteBuffer, &heatMapBuffer,
104 &weightBuffer, &renderBufferOdd, aovBuffers,
105 displayFilterBuffers);
106
107 // throw a file io error if anything failed to write, this will cause main to
108 // exit with a non-zero error code.
109 if (error) {
110 throw scene_rdl2::except::IoError("Failed to write output images");
111 }
112}

114void
115RaasCommandLineApplication::run()
116{
117 // Run global init (creates a RenderDriver) This *must* be called on the same thread we
118 // intend to call RenderContext::startFrame from.
119 rndr::initGlobalDriver(mOptions);
120
121 logInitMessages();
122
123 // Create a RenderContext. Since the RenderContext internally holds a
124 // ref on the RenderDriver, scope it to this block so it gets destroyed
125 // before we call rndr::cleanUpRenderDriver.
126 {
127 rndr::RenderContext renderContext(mOptions, &mInitMessages);
128#ifdef ENABLE_ATHENA_LOGGING // option defined by build system
129 constexpr auto loggingConfig = rndr::RenderContext::LoggingConfiguration::ATHENA_ENABLED;
130#else
131 constexpr auto loggingConfig = rndr::RenderContext::LoggingConfiguration::ATHENA_DISABLED;
132#endif
133 renderContext.initialize(mInitMessages, loggingConfig);
134
135 // TODO: allow progressive mode rendering in moonray also.
136 renderContext.setRenderMode(rndr::RenderMode::BATCH);
137
138 render(renderContext);
139
140 for (const std::string & deltasFile : mOptions.getDeltasFiles()) {
141 Logger::info("Applying deltas from '" + deltasFile + "'.");
142 renderContext.updateScene(deltasFile);
143 render(renderContext);
144 }
145 }
146
147 rndr::cleanUpGlobalDriver();
148}
