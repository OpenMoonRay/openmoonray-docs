50void
51TileWorkQueue::init(RenderMode mode,
52 unsigned numTiles,
53 unsigned numPasses,
54 unsigned numRenderThreads,
55 const Pass *passes,
56 const size_t *samplesPerPass)
57{
58#ifdef RUNTIME_VERIFY
59 if (TileGroupRuntimeVerify::get()) {
60 TileGroupRuntimeVerify::get()->verify();
61 }
62 TileGroupRuntimeVerify::init(numTiles);
63#endif // end RUNTIME_VERIFY
64
65 MNRY_ASSERT(numPasses && numTiles && numRenderThreads);
66
67 const unsigned minTilesPerGroup = 1;
68 unsigned maxTilesPerGroup = 0;
69 unsigned desiredSamplesPerGroup = 0;
70
71 switch (mode) {
72 case RenderMode::BATCH:
73 maxTilesPerGroup = 128;
74 desiredSamplesPerGroup = 512;
75 break;
76
77 case RenderMode::PROGRESSIVE:
78 maxTilesPerGroup = 128;
79 desiredSamplesPerGroup = 512;
80 break;
81
82 case RenderMode::PROGRESSIVE_FAST:
83 maxTilesPerGroup = 128;
84 desiredSamplesPerGroup = 512;
85 break;
86
87 case RenderMode::REALTIME:
88 maxTilesPerGroup = 2;
89 desiredSamplesPerGroup = 8;
90 break;
91
92 case RenderMode::PROGRESS_CHECKPOINT:
93 if (passes[0].getNumSamplesPerTile() == 1) {
94 // This is a very initial estimation stage
95 // So this makes task as 10 tiles/thread for each thread.
96 // See also RenderDriver.cc initPasses()
97 maxTilesPerGroup = 10;
98 desiredSamplesPerGroup = 5;
99 } else {
100 // Otherwise back to same setting of BATCH and PROGRESSIVE
101 maxTilesPerGroup = 128;
102 desiredSamplesPerGroup = 512;
103 }
104 break;
105
106 default:
107 MNRY_ASSERT(0);
108 }
109
110 MNRY_ASSERT(minTilesPerGroup > 0);
111 MNRY_ASSERT(maxTilesPerGroup >= minTilesPerGroup);
112
113 mNumPasses = numPasses;
114 mNumTiles = numTiles;
115
116 //
117 // Compute tile groups within each pass.
118 //
119
120 unsigned groupIdx = 0;
121
122 for (unsigned i = 0; i < mNumPasses; ++i) {
123
124 const Pass &pass = passes[i];
125 PassInfo *passInfo = &mPassInfos[i];
126
127 unsigned samplesPerPixel = pass.mEndSampleIdx - pass.mStartSampleIdx;
128 unsigned samplesPerTile = (pass.mEndPixelIdx - pass.mStartPixelIdx) * samplesPerPixel;
129 unsigned tilesPerGroup = std::lround(static_cast<float>(desiredSamplesPerGroup) /
130 static_cast<float>(samplesPerTile));
131 tilesPerGroup = scene_rdl2::math::clamp<unsigned>(tilesPerGroup, minTilesPerGroup, maxTilesPerGroup);
132 MNRY_ASSERT(tilesPerGroup);
133
134 unsigned numGroups = mNumTiles / tilesPerGroup;
135 if (numGroups * tilesPerGroup < mNumTiles) {
136 ++numGroups;
137 MNRY_ASSERT(numGroups * tilesPerGroup >= mNumTiles);
138 }
139
140 passInfo->mPass = pass;
141
142 passInfo->mStartGroupIdx = groupIdx;
143 passInfo->mEndGroupIdx = groupIdx + numGroups;
144 passInfo->mTilesPerGroup = tilesPerGroup;
145
146 groupIdx = passInfo->mEndGroupIdx;
147 }
148
149 reset();
150
151 if (mRuntimeDebug) {
152 std::cerr << show() << '\n';
153 }
154}

183TileGroup
184TileWorkQueue::reserveNextTileGroup()
185{
186 // TODO: C++17 change to is_always_lock_free and do a static assert, although this is desired behavior, we're
187 // probably okay even if it's not true.
188 MNRY_ASSERT(mOffset.is_lock_free());
189
190 constexpr unsigned minTilesPerGroup = 1u;
191 TileGroup group;
192
193 OffsetData currentOffset = mOffset;
194 while (true) {
195 MNRY_ASSERT(mOffset.is_lock_free());
196
197 // Get pass information
198 const PassInfo& passInfo = mPassInfos[currentOffset.mPass];
199 const unsigned numGroupTiles = std::max(minTilesPerGroup, passInfo.mTilesPerGroup);
200
201 // This will load-balance in that each tile group will only differ by
202 // +/-1 from the (number of tiles)/(number of group tiles).
203 //
204 // groupsLeftInPass will eventually be 1, which leaves that group to
205 // get the remainder of the tiles.
206 const auto tilesLeftInPass = mNumTiles - currentOffset.mTile;
207 const auto groupsLeftInPass = roundUpDivision(tilesLeftInPass, numGroupTiles);
208
209 OffsetData nextOffset(currentOffset);
210 if (groupsLeftInPass <= 1) {
211 // We need to start a new pass next time
212 nextOffset.mTile = 0;
213 ++nextOffset.mPass;
214 } else {
215 const auto numTiles = tilesLeftInPass/groupsLeftInPass;
216 nextOffset.mTile = currentOffset.mTile + numTiles;
217 MNRY_ASSERT(nextOffset.mTile < mNumTiles);
218 }
219
220 // Check to see if we can actually claim this offset though CAS.
221 if (mOffset.compare_exchange_strong(currentOffset, nextOffset)) {
222 // We have successfully claimed this offset. Update our group data for returning.
223 group.mPassIdx = currentOffset.mPass;
224 group.mStartTileIdx = currentOffset.mTile;
225 group.mEndTileIdx = (nextOffset.mTile == 0) ? mNumTiles : (nextOffset.mTile);
226
227 MNRY_ASSERT(group.mEndTileIdx > group.mStartTileIdx);
228 MNRY_ASSERT(group.mStartTileIdx < mNumTiles);
229 MNRY_ASSERT(group.mEndTileIdx <= mNumTiles);
230 return group;
231 }
232 }

235bool
236TileWorkQueue::getNextTileGroup(unsigned threadIdx, TileGroup *group, unsigned lastCoursePassIdx)
237{
238 MNRY_ASSERT(mNumPasses);
239
240 // mGroupClampIdx is set by the main thread.
241 auto groupIdx = mGlobalGroupIdx.load();
242 do {
243 if (groupIdx >= mGroupClampIdx) {
244 return false;
245 }
246 } while (!mGlobalGroupIdx.compare_exchange_weak(groupIdx, groupIdx + 1u));
247 *group = reserveNextTileGroup();
248
249#ifdef RUNTIME_VERIFY
250 TileGroupRuntimeVerify::get()->push(*group);
251#endif // end RUNTIME_VERIFY
252
253 /*
254 // A debug variable to see how many groups we let through: this seems
255 // consistent on the intial render, but varies on the resume render.
256 // (Of course, they get printed out of order).
257 static std::atomic<int> count(0);
258 PRINT(++count);
259 */
260
261 const bool startOfFinePass = group->mStartTileIdx == 0 && group->mPassIdx == lastCoursePassIdx;
262 group->mFirstFinePass = startOfFinePass;
263
264 return true;
265}

307 << scene_rdl2::str_util::addIndent(showPassInfo(mPassInfos[i]), 2) << '\n';
