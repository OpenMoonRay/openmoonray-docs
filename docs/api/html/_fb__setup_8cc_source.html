18 mPixelInfoName = name;

185template <typename ResizeBuffFunc, typename InitWholeBuffFunc, typename InitPartialBuffFunc>
186void
187Fb::setupBufferMain(const PartialMergeTilesTbl *partialMergeTilesTbl,
188 bool &bufferStatus,
189 const ActivePixels &activePixels,
190 unsigned numOfBuffers,
191 ResizeBuffFunc resizeBuffFunc,
192 InitWholeBuffFunc initWholeBuffFunc,
193 InitPartialBuffFunc initPartialBuffFunc)
194//
195// This function finally setup internal memory and clear data and set buffer condition as active
196// (i.e. bufferStatus goes to true).
197//
198{
199 bool needPartialInit = false;
200 bool needWholeInit = false;
201 if (!bufferStatus) {
202 // Previous bufferStatus condition is false. This means we have to properly setup memory inside
203 // this function. In order to this, we need at least one of the needPartialInit and needWholeInit
204 // flag should be true. First of all, set needpartialInit = true here. If this condition is not
205 // good, needWholeInit will be on instead later.
206 needPartialInit = true;
207 }
208
209 //
210 // resize buffer (i.e. memory allocation)
211 //
212 if (activePixels.getWidth() != mRezedViewport.width() ||
213 activePixels.getHeight() != mRezedViewport.height()) {
214 //
215 // Regardless of current buffer condition (bufferStatus = true/false), we have to setup this buffer
216 // because we have to make this buffer as access ready condition (bufferStatus = true).
217 //
218 // We only check activePixels resolution because other associated buffer
219 // always has same resolution of activePixels actually.
220 //
221 unsigned width = mRezedViewport.width();
222 unsigned height = mRezedViewport.height();
223 unsigned alignedWidth = (width + 7) & ~7; // tile aligned (8x8) size
224 unsigned alignedHeight = (height + 7) & ~7; // tile aligned (8x8) size
225# ifdef SINGLE_THREAD
226 for (unsigned bufferId = 0; bufferId < numOfBuffers; ++bufferId) {
227 resizeBuffFunc(bufferId, width, height, alignedWidth, alignedHeight);
228 }
229# else // else SINGLE_THREAD
230 tbb::parallel_for((unsigned)0, numOfBuffers, [&](unsigned bufferId) {
231 resizeBuffFunc(bufferId, width, height, alignedWidth, alignedHeight);
232 });
233# endif // end !SINGLE_THREAD
234
235 needPartialInit = false;
236 needWholeInit = true;
237 }
238
239 //
240 // initialize buffer
241 //
242 if (!partialMergeTilesTbl) {
243 if (needPartialInit) {
244 needWholeInit = true;
245 needPartialInit = false;
246 }
247 }
248
249 if (needWholeInit) {
250# ifdef SINGLE_THREAD
251 for (unsigned bufferId = 0; bufferId < numOfBuffers; ++bufferId) {
252 initWholeBuffFunc(bufferId);
253 }
254# else // else SINGLE_THREAD
255 tbb::parallel_for((unsigned)0, numOfBuffers, [&](unsigned bufferId) {
256 initWholeBuffFunc(bufferId);
257 });
258# endif // end !SINGLE_THREAD
259 } else if (needPartialInit) {
260# ifdef SINGLE_THREAD
261 for (unsigned bufferId = 0; bufferId < numOfBuffers; ++bufferId) {
262 initPartialBuffFunc(bufferId);
263 }
264# else // else SINGLE_THREAD
265 tbb::parallel_for((unsigned)0, numOfBuffers, [&](unsigned bufferId) {
266 initPartialBuffFunc(bufferId);
267 });
268# endif // end !SINGLE_THREAD
269 }
270
271 bufferStatus = true;
272}
