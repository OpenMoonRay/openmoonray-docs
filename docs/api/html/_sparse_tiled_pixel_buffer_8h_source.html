24// Pack sparse tile data into the supplied memory block, given a tiled source
25// buffer and the corresponding tile list.
26// The buffer passed in much be numTiles * 64 * sizeof(PIXEL_TYPE) in length.
27// No allocation or deallocation is performed inside of this function.
28template<typename PIXEL_TYPE>
29inline bool
30packSparseTiles(PIXEL_TYPE *dstPackedBuffer,
31 PixelBuffer<PIXEL_TYPE> const &srcTiledBuffer,
32 const std::vector<Tile> &tiles)
33{
34 MNRY_ASSERT(dstPackedBuffer);
35 MNRY_ASSERT(srcTiledBuffer.getWidth() % 8 == 0);
36 MNRY_ASSERT(srcTiledBuffer.getHeight() % 8 == 0);
37
38 unsigned numTiles = (unsigned)tiles.size();
39 if (numTiles == 0) {
40 return false;
41 }
42
43 Tiler tiler(srcTiledBuffer.getWidth(), srcTiledBuffer.getHeight());
44
45 const PIXEL_TYPE *src = srcTiledBuffer.getData();
46 PIXEL_TYPE *dst = dstPackedBuffer;
47 for (unsigned i = 0; i < numTiles; ++i) {
48 const Tile &tile = tiles[i];
49 unsigned ofs = tiler.linearCoordsToCoarseTileOffset(tile.mMinX, tile.mMinY);
50 memcpy(dst, src + ofs, sizeof(PIXEL_TYPE) * 64);
51 dst += 64;
52 }
53
54 MNRY_ASSERT(dst == dstPackedBuffer + numTiles * 64);
55
56 return true;
57}

59// Unpacked tile data to a destination tiled buffer.
60// dst_tiled_buffer must be pre-initialized to be desired (tiled) dimensions.
61template<typename PIXEL_TYPE>
62inline bool
63unpackSparseTiles(PixelBuffer<PIXEL_TYPE> *dstTiledBuffer,
64 const PIXEL_TYPE *srcPackedData,
65 const std::vector<Tile> &tiles)
66{
67 MNRY_ASSERT(dstTiledBuffer);
68 MNRY_ASSERT(dstTiledBuffer->getWidth() % 8 == 0);
69 MNRY_ASSERT(dstTiledBuffer->getHeight() % 8 == 0);
70
71 unsigned numTiles = (unsigned)tiles.size();
72 if (numTiles == 0 || dstTiledBuffer->getArea() == 0) {
73 return false;
74 }
75
76 unsigned w = dstTiledBuffer->getWidth();
77 unsigned h = dstTiledBuffer->getHeight();
78
79 Tiler tiler(w, h);
80
81 const PIXEL_TYPE *src = srcPackedData;
82 PIXEL_TYPE *dst = dstTiledBuffer->getData();
83 for (unsigned i = 0; i < numTiles; ++i) {
84 const Tile &tile = tiles[i];
85 MNRY_ASSERT(tile.mMaxX <= w && tile.mMaxY <= h);
86 unsigned ofs = tiler.linearCoordsToCoarseTileOffset(tile.mMinX, tile.mMinY);
87 memcpy(dst + ofs, src, sizeof(PIXEL_TYPE) * 64);
88 src += 64;
89 }
90
91 MNRY_ASSERT(src == srcPackedData + numTiles * 64);
92
93 return true;
94}
