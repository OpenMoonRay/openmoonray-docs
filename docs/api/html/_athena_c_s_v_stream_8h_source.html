108 }
109private:
110 template <typename... Args>
111 void do_open(Args&&... args)
112 {
113 try {
114 mBuf.reset(new AthenaCSVStreamBuf(std::forward<Args>(args)...));
115 this->init(mBuf.get());
116 this->clear();
117 mOpened = true;
118 } catch (const std::exception&) {
119 this->setstate(std::ios_base::failbit);
120 mOpened = false;
121 } catch (...) {
122 this->setstate(std::ios_base::failbit);
123 mOpened = false;
124 }
125 }
