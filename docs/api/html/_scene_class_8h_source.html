43/**
44 * The SceneClass represents all the metadata and structure of SceneObjects of
45 * a particular type. It is analogous to a C++ class for render objects that
46 * are declared at runtime.
47 *
48 * In addition to allowing the declaration of attributes, it also handles
49 * a lot of the messy details around stamping out SceneObjects and accessing
50 * specific attribute values. Those are all internal details to RDL though,
51 * and aren't exposed through the public API.
52 *
53 * Once the SceneClass is "complete", no more attribute declarations can
54 * occur. The SceneContext will handle this for you automatically, just be
55 * aware that the only place you can declare attributes is inside your
56 * declaration function (rdl_declare() for DSOs, ClassDeclareFunc for builtins).
57 *
58 * Thread Safety:
59 * - The model is very similar to much of the rest of RDL. The read-only
60 * API is explicitly defined by const methods, and reading from multiple
61 * threads is safe.
62 * - If anyone is writing to a SceneClass (such as declaring new attributes or
63 * modifying metadata in the attributes themselves), while you're reading
64 * it... game over. RDL does not synchronize that for you.
65 */

88 /**
89 * Declares an attribute of type T.
90 *
91 * The flags may include things like whether the attribute is blurrable or
92 * bindable. Blurrable attributes store multiple values (one per timestep).
93 * Bindable attributes can have other SceneObjects bound to them in addition
94 * to having a value.
95 *
96 * The objectType is optional, and only relevant if the attribute's type is
97 * SceneObject* or SceneObjectVector. In that case, the objectType
98 * defines interface constraints on what kinds of SceneObjects can be set
99 * as a value.
100 *
101 * The aliases are optional. If non-empty, attribute aliases will be set
102 * for this attribute. The aliases must not collide with any other attribute
103 * name or alias in the SceneClass.
104 *
105 * The initial value of this attribute will be a sane default for the type,
106 * such as 0 for numeric types, "" for strings, etc.
107 *
108 * @param name The name of the attribute.
109 * @param flags Attribute flags, such as blurrable or bindable.
110 * @param objectType The type of SceneObjects that can be set if the
111 * attribute type (T) is SceneObject* or SceneObjectVector.
112 * @param aliases list of aliases for the attribute name
113 * @return An AttributeKey for fast, type safe gets and sets on any
114 * SceneObject of this SceneClass.
115 */

117 finline AttributeKey<T> declareAttribute(const std::string& name,
118 AttributeFlags flags = FLAGS_NONE,
119 SceneObjectInterface objectType = INTERFACE_GENERIC,
120 const std::vector<std::string> &aliases = {});

615template <typename T, typename F>
616AttributeKey<T>
617SceneClass::createAttribute(const std::string& name, AttributeFlags flags,
618 const std::vector<std::string> &aliases,
619 F attributeConstructor)
620{
621 // Ensure attribute name is valid
622 MNRY_ASSERT_REQUIRE(validName(name), (std::string("Attribute name '") + name +
623 std::string("' does not conform to the format [a-zA-Z][a-zA-Z0-9_]*")).c_str());
624
625 // Ensure it's safe to manipulate the attribute declarations.
626 if (mComplete) {
627 std::stringstream errMsg;
628 errMsg << "Cannot declare attributes on SceneClass '" << mName <<
629 "' after declarations are finished.";
630 throw except::RuntimeError(errMsg.str());
631 }
632
633 // Does an attribute with this name or aliases already exist?
634 AttributeMapConstIterator iter = mNameMap.find(name);
635 if (iter == mNameMap.end()) {
636 // check aliases
637 for (const auto &a : aliases) {
638 iter = mNameMap.find(a);
639 if (iter != mNameMap.end()) {
640 break;
641 }
642 }
643 }
644 if (iter != mNameMap.end()) {
645 std::stringstream errMsg;
646 errMsg << "Duplicate declaration of Attribute '" << name <<
647 "' in SceneClass '" << mName << "'.";
648 throw except::KeyError(errMsg.str());
649 }
650
651 // What is the index of this new attribute?
652 uint32_t index = mAttributes.size();
653
654 // Compute its location in memory.
655 std::pair<uint32_t, std::size_t> result = computeOffsetAndSize<T>(flags);
656 uint32_t offset = result.first;
657 std::size_t size = result.second;
658
659 // Try to create the attribute.
660 Attribute* attribute = nullptr;
661 try {
662 attribute = attributeConstructor(index, offset);
663 } catch (const except::TypeError&) {
664 delete attribute; // Clean up.
665 attribute = nullptr;
666 throw; // Rethrow.
667 }
668
669 // Add the attribute to the list of attributes and lookup map.
670 mAttributes.push_back(attribute);
671 mNameMap.insert(AttributeMapItem(name, attribute));
672 // Aliases
673 for (const auto &a : aliases) {
674 mNameMap.insert(AttributeMapItem(a, attribute));
675 }
676
677 // Track the amount of space used to store the attribute's value. (We don't
678 // just "+= size" here because the offset is absolute and includes any
679 // padding.)
680 mAttributeStorageSize = offset + size;
681
682 // Hand back an AttributeKey for accessing this attribute.
683 return AttributeKey<T>(*attribute);
684}

693SceneObject*
694SceneClass::createObject(const std::string& name)
695{
696 // Sanity check.
697 if (!mComplete) {
698 std::stringstream errMsg;
699 errMsg << "Cannot create SceneObject '" << name << "' until its"
700 " SceneClass '" << mName << "' has finished being declared.";

703

769template <typename T>
770void
771SceneClass::setGroup(const std::string& groupName, AttributeKey<T> attributeKey)
772{
773 // Find the group index, inserting into the group names if this is the
774 // first time the group is referenced.
775 std::size_t groupIndex = 0;
776 auto result = std::find(mGroupNames.begin(), mGroupNames.end(), groupName);
777 if (result == mGroupNames.end()) {
778 // Not found, insert it.
779 mGroupNames.push_back(groupName);
780 groupIndex = mGroupNames.size() - 1;
781 } else {
782 // Found, set the group index.
783 groupIndex = result - mGroupNames.begin();
784 }
785
786 // Insert the Attribute* into the group map at the group index.
787 mGroupMap.insert(std::make_pair(groupIndex, getAttribute(attributeKey)));
788}

864 if (isEqualToValue(&(base[timestep]), value)) {
