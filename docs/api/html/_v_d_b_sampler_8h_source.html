45template<typename GridT>
46class TypedVDBSampler : public VDBSampler
47{
48private:
49 typedef typename GridT::ValueType ValueT;
50
51public:
52 TypedVDBSampler(const GridT& grid)
53 {
54 size_t threadCount = mcrt_common::getMaxNumTLS();
55 mSamplers.reserve(threadCount);
56 for (size_t i = 0; i < threadCount; ++i) {
57 mSamplers.emplace_back(grid);
58 }
59 }
60
61 ~TypedVDBSampler() override {}
62
63 bool getIsActive(const uint32_t threadIdx,
64 const openvdb::Vec3d& pos) override
65 {
66 const openvdb::Vec3d p = mSamplers[threadIdx].mGrid.worldToIndex(pos);
67 const openvdb::Coord coord(p.x(), p.y(), p.z());
68 return mSamplers[threadIdx].mAccessor.isValueOn(coord);
69 }
70
71 ValueT sample(const uint32_t threadIdx, const openvdb::Vec3d& pos,
72 Interpolation mode) const
73 {
74 switch (mode) {
75 case Interpolation::Point :
76 return mSamplers[threadIdx].evalPoint(pos);
77 case Interpolation::Box :
78 return mSamplers[threadIdx].evalBox(pos);
79 case Interpolation::Quadratic :
80 return mSamplers[threadIdx].evalQuadratic(pos);
81 default:
82 return mSamplers[threadIdx].evalPoint(pos);
83 }
84 }

86private:
87 struct GridSampler

89 typedef typename GridT::ConstAccessor ConstAccessor;

91 typedef openvdb::tools::GridSampler<ConstAccessor, openvdb::tools::BoxSampler> BoxSampler;
92 typedef openvdb::tools::GridSampler<ConstAccessor, openvdb::tools::QuadraticSampler> QuadraticSampler;
93
94 GridSampler(const GridT& grid):
95 mGrid(grid),
