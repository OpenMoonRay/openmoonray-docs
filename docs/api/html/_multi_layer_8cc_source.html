61void
62MultiLayerBssrdf::finalize()
63{
64 float sum = 0;
65 mMaxRadius = 0;
66 for (unsigned int i = 0; i < mLobeCount; i++) {
67 sum += mWeights[i];
68 if (mBssrdfs[i]->getMaxRadius() > mMaxRadius) mMaxRadius = mBssrdfs[i]->getMaxRadius();
69 }
70 if (sum > 0) {
71 float norm = 1.0f/sum;
72 float cdf = 0;
73 //mCdfs = new float[mWeights.size()];
74 for (unsigned int i = 0; i < mLobeCount; i++) {
75 mWeights[i] *= norm;
76 cdf += mWeights[i];
77 mCdfs[i] = cdf;
78 }
79 }
80}

82unsigned int
83MultiLayerBssrdf::selectIndex(float r1) const
84{
85 switch (mLobeCount) {
86 case 1:
87 return 0;
88 case 2:
89 return r1 < mCdfs[0] ? 0 : 1;
90 case 3:
91 return (r1 < mCdfs[0] ? 0 :
92 (r1 < mCdfs[1] ? 1 : 2));
93 }
94 const float *ptr = std::upper_bound(mCdfs, mCdfs + mLobeCount, r1);
95 unsigned int layerIndex = ptr - mCdfs;
96 return max(layerIndex, mLobeCount - 1);
97}

100#if !BSSRDF_SAMPLE_UNIFORM
101float
102MultiLayerBssrdf::sampleLocal(float r1, float r2, Vec3f &dPi, float &r) const
103{
104 if (mLobeCount == 0) return 1;
105
106 unsigned int layerIndex = selectIndex(r1);
107 float curCdf = mWeights[layerIndex];
108 if (layerIndex == 0) {
109 r1 = r1/curCdf;
110 } else {
111 r1 = (r1-mCdfs[layerIndex-1])/curCdf;
112 }
113 float childPdf = mBssrdfs[layerIndex]->sampleLocal(r1, r2, dPi, r);
114 float totalPdf = childPdf * curCdf;
115 for (unsigned int i = 0; i < mLobeCount; i++) {
116 if (i != layerIndex) {
117 totalPdf += mWeights[i] * mBssrdfs[i]->pdfLocal(r);
118 }
119 }
120 return totalPdf;
121}

123float
124MultiLayerBssrdf::pdfLocal(float r) const
125{
126 // We compute the estimator with MIS, under Veach's one-sample model.
127 // See paper: "Optimally combining sampling techniques for MC rendering"
128 // Here we use estimator (16) and therefore: pdf = ci * pi / wi
129 // With the balance heuristic it amounts to the weighted average of the
130 // pdf for each channel: pdf = sum(ci * pi)
131 float pdf = 0.0f;
132 for (unsigned int i = 0; i < mLobeCount; ++i) {
133 float ci = mWeights[i];
134 pdf += ci * mBssrdfs[i]->pdfLocal(r);
135 }
136 return pdf;
137}

155// Uniform sampling
156float
157MultiLayerBssrdf::sampleLocal(float r1, float r2, Vec3f &dPi, float &r) const
158{
159 Vec2f p;
160 p[0] = (r1 - 0.5f) * 2.0f * mMaxRadius;
161 p[1] = (r2 - 0.5f) * 2.0f * mMaxRadius;
162 r = p.length();
163
164 // Compute position
165 dPi = p[0] * mFrame.getX() + p[1] * mFrame.getY();
166
167 float pdf = 1.0f / (4.0f * mMaxRadius * mMaxRadius);
168 return pdf;
169}

180bool
181MultiLayerBssrdf::getProperty(Property property, float *dest) const
182{
183 Color values(0.0f);
184 Color totalValues(0.0f);
185 bool handled = true;
186
187 switch (property)
188 {
189 case PROPERTY_COLOR:
190 case PROPERTY_RADIUS:
191 for (unsigned int t = 0; t < mLobeCount; ++t) {
192 mBssrdfs[t]->getProperty(property, (float*) &values);
193 totalValues += mWeights[t] * values;
194 }
195 dest[0] = totalValues[0];
196 dest[1] = totalValues[1];
197 dest[2] = totalValues[2];
198 break;
199 default:
200 handled = Bssrdf::getProperty(property, dest);
201 }
202
203 return handled;
204}
