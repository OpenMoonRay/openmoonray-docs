70static std::string
71trim(const std::string &in)
72{
73 // trim leading and trailing white space
74 const std::string ws = " \t";
75 const std::size_t beg = in.find_first_not_of(ws);
76 if (beg == std::string::npos) {
77 return "";
78 }
79 const std::size_t end = in.find_last_not_of(ws);
80 return in.substr(beg, end - beg + 1);
81}

83int
84StateMachine::Impl::addExpression(std::string const &lp, int id)
85{
86 MNRY_ASSERT(!mBuilt);
87
88 int result = 0;
89 const std::vector<osl::ustring> userEvents { mExtraLabel, mMaterialLabel };
90
91 osl::Parser p(&userEvents);
92 osl::lpexp::LPexp *l = p.parse(lp.c_str());
93 if (!l) {
94 result = -1;
95 } else {
96#ifdef DEBUG
97 // check that id is not currently in use
98 for (auto const &exp: mExpressions) {
99 MNRY_ASSERT(std::get<1>(exp) != id);
100 }
101#endif
102 mExpressions.push_back(std::make_tuple(lp, id, l));
103 }
104
105 // scan lp for labels, any trimmed value between two tick chars
106 // is a label
107 auto t0 = lp.find_first_of('\'');
108 while (t0 != std::string::npos) {
109 auto t1 = lp.find_first_of('\'', t0 + 1);
110 MNRY_ASSERT(t1 != std::string::npos); // otherwise p.parse would have failed
111 addLabel(trim(lp.substr(t0 + 1, t1 - t0 - 1)));
112 t0 = lp.find_first_of('\'', t1 + 1);
113 }
114
115 return result;
116}

131int
132StateMachine::Impl::getLabelId(std::string const &label) const
133{
134 int result = -1;
135 osl::ustring us(label);
136 auto it = std::find(mLabels.begin(), mLabels.end(), us);
137 if (it != mLabels.end()) {
138 result = static_cast<int>(it - mLabels.begin());
139 }
140 return result;
141}

143void
144StateMachine::Impl::build()
145{
146 MNRY_ASSERT(!mBuilt);
147
148 osl::NdfAutomata ndf;
149 osl::DfAutomata fsm;
150
151 if (!mExpressions.empty()) {
152 // build a single ndf
153 for (auto const &exp: mExpressions) {
154 osl::lpexp::LPexp *l = std::get<2>(exp);
155 int id = std::get<1>(exp);
156 osl::lpexp::Rule rule(l, reinterpret_cast<void *>(static_cast<intptr_t>(id)));
157 rule.genAuto(ndf);
158 }
159
160 // build the fsm
161 ndfautoToDfauto(ndf, fsm);
162
163 // MNRY_ASSERT(mFsm.size() > 0);
164 MNRY_ASSERT(fsm.size() > 0);
165
166 // build the optimized fsm
167 mOptFsm.compileFrom(fsm);
168
169 mBuilt = true;
170 }

173int
174StateMachine::Impl::transition(int stateId, EventType ev, EventScatteringType evs, int labelId) const
175{
176 MNRY_ASSERT(mBuilt);
177
178 int newStateId = stateId;
179 if (newStateId < 0) return newStateId;
180
181 // event
182 switch (ev) {
183 case EVENT_TYPE_CAMERA:
184 newStateId = mOptFsm.getTransition(newStateId, osl::Labels::CAMERA);
185 break;
186 case EVENT_TYPE_REFLECTION:
187 newStateId = mOptFsm.getTransition(newStateId, osl::Labels::REFLECT);
188 break;
189 case EVENT_TYPE_TRANSMISSION:
190 newStateId = mOptFsm.getTransition(newStateId, osl::Labels::TRANSMIT);
191 break;
192 case EVENT_TYPE_VOLUME:
193 newStateId = mOptFsm.getTransition(newStateId, osl::Labels::VOLUME);
194 break;
195 case EVENT_TYPE_LIGHT:
196 newStateId = mOptFsm.getTransition(newStateId, osl::Labels::LIGHT);
197 break;
198 case EVENT_TYPE_EMISSION:
199 newStateId = mOptFsm.getTransition(newStateId, osl::Labels::OBJECT);
200 break;
201 case EVENT_TYPE_BACKGROUND:
202 newStateId = mOptFsm.getTransition(newStateId, osl::Labels::BACKGROUND);
203 break;
204 case EVENT_TYPE_EXTRA:
205 newStateId = mOptFsm.getTransition(newStateId, mExtraLabel);
206 break;
207 case EVENT_TYPE_MATERIAL:
208 newStateId = mOptFsm.getTransition(newStateId, mMaterialLabel);
209 break;
210 case EVENT_TYPE_NONE:
211 break;
212 default:
213 newStateId = -1; // broken
214 break;
215 }
216
217 if (newStateId < 0) return newStateId;
218
219 // scattering event
220 switch (evs) {
221 case EVENT_SCATTERING_TYPE_NONE:
222 newStateId = mOptFsm.getTransition(newStateId, osl::Labels::NONE);
223 break;
224 case EVENT_SCATTERING_TYPE_DIFFUSE:
225 newStateId = mOptFsm.getTransition(newStateId, osl::Labels::DIFFUSE);
226 break;
227 case EVENT_SCATTERING_TYPE_GLOSSY:
228 newStateId = mOptFsm.getTransition(newStateId, osl::Labels::GLOSSY);
229 break;
230 case EVENT_SCATTERING_TYPE_MIRROR:
231 newStateId = mOptFsm.getTransition(newStateId, osl::Labels::SINGULAR);
232 break;
233 case EVENT_SCATTERING_TYPE_STRAIGHT:
234 newStateId = mOptFsm.getTransition(newStateId, osl::Labels::STRAIGHT);
235 break;
236 default:
237 newStateId = -1; // broken
238 }
239
240 if (newStateId < 0) return newStateId;
241
242 // labelId
243 if (labelId >= 0) {
244 MNRY_ASSERT(static_cast<unsigned int>(labelId) < mLabels.size());
245 newStateId = mOptFsm.getTransition(newStateId, mLabels[labelId]);
246 }
247
248 if (newStateId < 0) return newStateId;
249
250 // and finish evey hit event with a stop
251 newStateId = mOptFsm.getTransition(newStateId, osl::Labels::STOP);
252
253 return newStateId;
254}

256bool
257StateMachine::Impl::isValid(int stateId, int id) const
258{
259 MNRY_ASSERT(mBuilt);
260
261 bool result = false;
262
263 if (stateId >= 0) {
264 int nrules = 0;
265 void * const * rules = mOptFsm.getRules(stateId, nrules);
266 for (int i = 0; i < nrules; ++i) {
267 intptr_t rule = reinterpret_cast<intptr_t>(rules[i]);
268 if (rule == id) {
269 result = true;
270 break;
271 }
272 }
273 }
274
275 return result;
276}
