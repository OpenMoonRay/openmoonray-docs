21//
22// Properties and assumptions:
23//
24// - ALLOWS CYCLES!
25//
26// - Payload size: There is a copy on insertion so small payloads work better
27// for this. Additionally there is a second copy on flush
28// which allow cycles in the graph. We heavily depend on this
29// behavior for the RayQueue as the mechanism to avoid stalling.
30// If the payload is large and you don't need to support
31// cycles then consider using LocalLargeEntryQueue instead.
32// - Sorting: Optional with an inline sortkey, sortkey not required if
33// sorting is off.
34// - Thread safe: No. Everything is assumed to be thread local. Entries cannot
35// be added by threads other than ourselves.
36//

153protected:
154 unsigned flushInternal(mcrt_common::ThreadLocalState *tls,
155 uint32_t numNewEntries, T *newEntries,
156 scene_rdl2::alloc::Arena *arena)
157 {
158 EXCL_ACCUMULATOR_PROFILE(tls, EXCL_ACCUM_QUEUE_LOGIC);
159
160 SCOPED_MEM(arena);
161
162 uint32_t totalEntries = mNumQueued + numNewEntries;
163
164 MNRY_ASSERT(totalEntries);
165
166 // We always want to copy entries since there may be cycles which allows
167 // other code to add to this queue further down in the callstack. This is
168 // something this queue supports and encourages.
169 EntryType *entries = arena->allocArray<EntryType>(totalEntries);
170
171 // Copy initial entries.
172 memcpy(entries, mEntries, sizeof(EntryType) * mNumQueued);
173
174 // Copy additional entries.
175 if (numNewEntries) {
176 memcpy(entries + mNumQueued, newEntries, sizeof(EntryType) * numNewEntries);
177 }
178
179 unsigned entriesToFlush = totalEntries;
180
181 // Only flush a multiple of the number of lanes we have unless we're doing an
182 // explicit flush (totalEntries < mQueueSize) or queue is exactly full.
183 mNumQueued = 0;
184 if (totalEntries > mQueueSize) {
185
186 unsigned potentiallyQueued = totalEntries & (VLEN - 1);
187
188 // Copy the left overs back into the primary queue. It's safe to do this
189 // now since we always copy the entries before calling the handler.
190 if (potentiallyQueued && potentiallyQueued < mQueueSize) {
191 mNumQueued = potentiallyQueued;
192 entriesToFlush -= potentiallyQueued;
193 memcpy(mEntries, entries + entriesToFlush, sizeof(EntryType) * potentiallyQueued);
194 }
195 }
196
197 MNRY_ASSERT(mNumQueued < VLEN);
198 MNRY_ASSERT(mNumQueued < mQueueSize);
199 MNRY_ASSERT(mNumQueued + entriesToFlush == totalEntries);
200
201 if (SORTED) {
202 scene_rdl2::util::inPlaceSort32<EntryType, SORT_KEY_OFFSET, 200>(entriesToFlush, entries, arena);
203 }
204
205 // Call handler. The entries are only valid for the duration of this call.
206 // Other threads may also call this handler simultaneously with different entries.
207 ++tls->mHandlerStackDepth;
208 (*mHandler)(tls, entriesToFlush, entries, mHandlerData);
209 MNRY_ASSERT(tls->mHandlerStackDepth > 0);
210 --tls->mHandlerStackDepth;
211
212 return unsigned(entriesToFlush);
213 }

216 uint32_t mMaxEntries;
217 EntryType * mEntries; // all data offsets are relative to this address
218 Handler mHandler;
219 void * mHandlerData;
220 uint32_t mNumQueued;
221};

366 unsigned flushUnsortedInternal(mcrt_common::ThreadLocalState *tls,
367 uint32_t numNewEntries, T *newEntries,
368 scene_rdl2::alloc::Arena *arena)
369 {
370 EXCL_ACCUMULATOR_PROFILE(tls, EXCL_ACCUM_QUEUE_LOGIC);
371
372 SCOPED_MEM(arena);
373
374 uint32_t totalEntries = mNumQueued + numNewEntries;
375
376 MNRY_ASSERT(totalEntries);
377
378 // Create a temp list of pointers to pass to handler.
379 EntryType **entries = arena->allocArray<EntryType *>(totalEntries);
380
381 // Copy old pointers into entries array.
382 for (uint32_t i = 0; i < mNumQueued; ++i) {
383 entries[i] = &mEntries[i];
384 }
385
386 // Copy additional entries.
387 for (uint32_t i = 0; i < numNewEntries; ++i) {
388 entries[mNumQueued + i] = &newEntries[i];
389 }
390
391 unsigned entriesToFlush = totalEntries;
392
393 // Only flush a multiple of the number of lanes we have unless we're doing an
394 // explicit flush (totalEntries < mQueueSize) or queue is exactly full.
395 mNumQueued = 0;
396 if (totalEntries > mQueueSize) {
397
398 unsigned potentiallyQueued = totalEntries & (VLEN - 1);
399 if (potentiallyQueued && potentiallyQueued < mQueueSize) {
400 mNumQueued = potentiallyQueued;
401 entriesToFlush -= potentiallyQueued;
402 }
403 }
404
405 // Call handler. The entries are only valid for the duration of this call.
406 // Other threads may also call this handler simultaneously with different entries.
407 ++tls->mHandlerStackDepth;
408 (*mHandler)(tls, entriesToFlush, entries, mHandlerData);
409 MNRY_ASSERT(tls->mHandlerStackDepth > 0);
410 --tls->mHandlerStackDepth;
411
412 // Copy the left overs back into the primary queue.
413 for (uint32_t i = 0; i < mNumQueued; ++i) {
414 mEntries[i] = *entries[entriesToFlush + i];
415 }
416
417 // If you hit any of these asserts, make sure this queue type doesn't appear
418 // in any cycles in the data flow graph.
419 MNRY_ASSERT(mNumQueued < VLEN);
420 MNRY_ASSERT(mNumQueued < mQueueSize);
421 MNRY_ASSERT(mNumQueued + entriesToFlush == totalEntries);
422
423 return unsigned(entriesToFlush);
424 }

426 unsigned flushSortedInternal(mcrt_common::ThreadLocalState *tls,
427 uint32_t numNewEntries, T *newEntries,
428 scene_rdl2::alloc::Arena *arena)
429 {
430 EXCL_ACCUMULATOR_PROFILE(tls, EXCL_ACCUM_QUEUE_LOGIC);
431
432 SCOPED_MEM(arena);
433
434 uint32_t totalEntries = mNumQueued + numNewEntries;
435
436 MNRY_ASSERT(totalEntries);
437
438 struct SortedEntry
439 {
440 EntryType * mEntry;
441 uint32_t mSortKey;
442 };
443
444 // Create a temp list of pointers to pass to handler.
445 SortedEntry *sortedEntries = arena->allocArray<SortedEntry>(totalEntries);
446
447 // Copy old pointers into sortedEntries array.
448 for (uint32_t i = 0; i < mNumQueued; ++i) {
449 sortedEntries[i].mEntry = &mEntries[i];
450 sortedEntries[i].mSortKey = ((uint32_t *)&mEntries[i])[SORT_KEY_OFFSET >> 2];
451 }
452
453 // Copy additional sortedEntries.
454 for (uint32_t i = 0; i < numNewEntries; ++i) {
455 sortedEntries[mNumQueued + i].mEntry = &newEntries[i];
456 sortedEntries[mNumQueued + i].mSortKey = ((uint32_t *)&newEntries[i])[SORT_KEY_OFFSET >> 2];
457 }
458
459 unsigned entriesToFlush = totalEntries;
460
461 // Only flush a multiple of the number of lanes we have unless we're doing an
462 // explicit flush (totalEntries < mQueueSize) or queue is exactly full.
463 mNumQueued = 0;
464 if (totalEntries > mQueueSize) {
465
466 unsigned potentiallyQueued = totalEntries & (VLEN - 1);
467
468 // Copy the left overs back into the primary queue. It's safe to do this
469 // now since we always copy the entries before calling the handler.
470 if (potentiallyQueued && potentiallyQueued < mQueueSize) {
471 mNumQueued = potentiallyQueued;
472 entriesToFlush -= potentiallyQueued;
473 }
474 }
475
476 scene_rdl2::util::inPlaceSort32<SortedEntry, sizeof(EntryType *), 100>(entriesToFlush,
477 sortedEntries, arena);
478
479 // Now the entry list is sorted, remove the sort keys in place from it
480 // before we pass the list to the handler.
481 EntryType **entries = (EntryType **)sortedEntries;
482 for (uint32_t i = 1; i < entriesToFlush; ++i) {
483 entries[i] = sortedEntries[i].mEntry;
484 }
485
486 // Call handler. The entries are only valid for the duration of this call.
487 // Other threads may also call this handler simultaneously with different entries.
488 ++tls->mHandlerStackDepth;
489 (*mHandler)(tls, entriesToFlush, entries, mHandlerData);
490 MNRY_ASSERT(tls->mHandlerStackDepth > 0);
491 --tls->mHandlerStackDepth;
492
493 // Copy the left overs back into the primary queue.
494 for (uint32_t i = 0; i < mNumQueued; ++i) {
495 mEntries[i] = *sortedEntries[entriesToFlush + i].mEntry;
496 }
497
498 // If you hit any of these asserts, make sure this queue type doesn't appear
499 // in any cycles in the data flow graph.
500 MNRY_ASSERT(mNumQueued < VLEN);
501 MNRY_ASSERT(mNumQueued < mQueueSize);
502 MNRY_ASSERT(mNumQueued + entriesToFlush == totalEntries);
503
504 return unsigned(entriesToFlush);
505 }

509 EntryType * mEntries; // all data offsets are relative to this address
510 Handler mHandler;
511 void * mHandlerData;
512 uint32_t mNumQueued;
513};

606 // Optimized for batch entries, single entries are considered the exception
607 // rather than the rule.
608 void addEntries(mcrt_common::ThreadLocalState *tls, unsigned numEntries,
609 const T *entries, scene_rdl2::alloc::Arena *arena)
610 {
611 EXCL_ACCUMULATOR_PROFILE(tls, EXCL_ACCUM_QUEUE_LOGIC);
612
613 MNRY_ASSERT(numEntries);
614 MNRY_ASSERT(mNumQueued < mQueueSize);
615
616 // We are locking here, but the cost gets amortized over numEntries,
617 // and also should typically be contended very seldomly.
618 mMutex.lock();
619
620 uint32_t freeIdx = mNumQueued;
621 uint32_t totalEntries = freeIdx + numEntries;
622
623 if (totalEntries < mQueueSize) {
624 // Copy data into free idx.
625 memcpy(mEntries + freeIdx, entries, numEntries * sizeof(EntryType));
626
627 mNumQueued = totalEntries;
628
629 mMutex.unlock();
630
631 return;
632 }
633
634 // Flush as soon as the queue is full.
635 flushInternal(tls, numEntries, entries, arena);
636
637 MNRY_ASSERT(mNumQueued < mQueueSize);
638 }

640 // Explicit flush of what's currently in the queue.
641 unsigned flush(mcrt_common::ThreadLocalState *tls, scene_rdl2::alloc::Arena *arena)
642 {
643 EXCL_ACCUMULATOR_PROFILE(tls, EXCL_ACCUM_QUEUE_LOGIC);
644
645 MNRY_ASSERT(mNumQueued < mQueueSize);
646
647 // Speculative early out check to avoid taking an unnecessary lock.
648 if (isEmpty()) {
649 return 0;
650 }
651
652 mMutex.lock();
653
654 // Reliable isEmpty check.
655 if (isEmpty()) {
656 mMutex.unlock();
657 return 0;
658 }
659
660 // This call unlocks mMutex internally.
661 return flushInternal(tls, 0, nullptr, arena);
662 }

664 // Like flush but only empties up to the specified number of entries.
665 unsigned drain(mcrt_common::ThreadLocalState *tls,
666 unsigned maxEntriesToFlush,
667 scene_rdl2::alloc::Arena *arena)
668 {
669 EXCL_ACCUMULATOR_PROFILE(tls, EXCL_ACCUM_QUEUE_LOGIC);
670
671 MNRY_ASSERT(mNumQueued < mQueueSize);
672
673 // Speculative early out check to avoid taking an unnecessary lock.
674 if (isEmpty()) {
675 return 0;
676 }
677
678 mMutex.lock();
679
680 // Reliable isEmpty check.
681 if (isEmpty()) {
682 mMutex.unlock();
683 return 0;
684 }
685
686 SCOPED_MEM(arena);
687
688 unsigned entriesToFlush = std::min(maxEntriesToFlush, mNumQueued);
689 MNRY_ASSERT(entriesToFlush);
690
691 // We always want to copy entries in the shared queue case so that other
692 // threads don't have to wait on the handler to complete before they can
693 // start adding entries again.
694 EntryType *entries = arena->allocArray<EntryType>(entriesToFlush);
695
696 // Copy entries to flush into new arena memory.
697 memcpy(entries, mEntries, sizeof(EntryType) * entriesToFlush);
698
699 // Shift remaining entries to start of mEntries buffer.
700 unsigned remainingEntries = mNumQueued - entriesToFlush;
701 if (remainingEntries) {
702 memmove(mEntries, mEntries + entriesToFlush, sizeof(EntryType) * remainingEntries);
703 }
704
705 mNumQueued = remainingEntries;
706
707 MNRY_ASSERT(mNumQueued < mQueueSize);
708
709 mMutex.unlock();
710
711 if (SORTED) {
712 std::sort(entries, entries + entriesToFlush, [](const T &a, const T &b) -> bool
713 {
714 return EXTRACT_KEY32(a, SORT_KEY_OFFSET) < EXTRACT_KEY32(b, SORT_KEY_OFFSET);
715 });
716 }
717
718 // Call handler. The entries are only valid for the duration of this call.
719 // Other threads may also call this handler simultaneously with different entries.
720 ++tls->mHandlerStackDepth;
721 (*mHandler)(tls, entriesToFlush, entries, mHandlerData);
722 MNRY_ASSERT(tls->mHandlerStackDepth > 0);
723 --tls->mHandlerStackDepth;
724
725 return entriesToFlush;
726 }

766 if (potentiallyQueued && potentiallyQueued < mQueueSize) {
767 mNumQueued = potentiallyQueued;
768 entriesToFlush -= potentiallyQueued;
769 memcpy(mEntries, entries + entriesToFlush, sizeof(EntryType) * potentiallyQueued);
770 }
771 }
772
773 MNRY_ASSERT(mNumQueued < VLEN);
774 MNRY_ASSERT(mNumQueued < mQueueSize);
775 MNRY_ASSERT(mNumQueued + entriesToFlush == totalEntries);
776
777 mMutex.unlock();
778
779 if (SORTED) {
780 scene_rdl2::util::inPlaceSort32<EntryType, SORT_KEY_OFFSET, 200>(entriesToFlush, entries, arena);
781 }
782
783 // Call handler. The entries are only valid for the duration of this call.
784 // Other threads may also call this handler simultaneously with different entries.
785 ++tls->mHandlerStackDepth;
786 (*mHandler)(tls, (unsigned)entriesToFlush, entries, mHandlerData);
787 MNRY_ASSERT(tls->mHandlerStackDepth > 0);
788 --tls->mHandlerStackDepth;
789
790 return unsigned(entriesToFlush);
791 }
792
793 tbb::atomic<uint32_t> mQueueSize; // must not exceed mMaxEntries
794 uint32_t mMaxEntries;
795 EntryType * mEntries; // all data offsets are relative to this address
796 Handler mHandler;
797 void * mHandlerData;
798 uint32_t mNumQueued;
799
800 // Move onto separate cache line.
801 CACHE_ALIGN tbb::spin_mutex mMutex;
802};
