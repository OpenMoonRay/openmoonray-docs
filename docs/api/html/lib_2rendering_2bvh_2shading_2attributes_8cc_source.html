20size_t
21Attributes::getMemory() const
22{
23 size_t mem = sizeof(Attributes) +
24 scene_rdl2::util::getVectorElementsMemory(mKeyOffset) +
25 scene_rdl2::util::getVectorElementsMemory(mTimeSampleCount) +
26 scene_rdl2::util::getVectorElementsMemory(mKeyRate) +
27 scene_rdl2::util::getVectorElementsMemory(mBeginFaceVarying);
28
29 // 5 vectors for actual primitive attributes data
30 mem += scene_rdl2::util::getVectorElementsMemory(mConstants);
31 mem += scene_rdl2::util::getVectorElementsMemory(mParts);
32 mem += scene_rdl2::util::getVectorElementsMemory(mUniforms);
33 mem += scene_rdl2::util::getVectorElementsMemory(mVaryings);
34 mem += scene_rdl2::util::getVectorElementsMemory(mFaceVaryings);
35 mem += scene_rdl2::util::getVectorElementsMemory(mVertices);
36 return mem;
37}

39template <typename size_type>
40std::unique_ptr<Attributes>
41Attributes::interleave(const PrimitiveAttributeTable& primitiveAttributeTable,
42 size_t numParts, size_t numUniforms, size_t numVaryings,
43 const std::vector<size_type>& numFaceVaryings, size_t numVertices)
44{
45 int largestKey = -1;
46 for (const auto& kv : primitiveAttributeTable) {
47 AttributeKey key = kv.first;
48 if (!kv.second.empty() && key > largestKey) {
49 largestKey = key;
50 }
51 }
52 size_t attrNumKeys = largestKey + 1;
53 size_t attrMaxNumTimes = 0;
54 std::vector<size_t> attrTimeSampleCount(attrNumKeys, 0);
55 for (const auto& kv : primitiveAttributeTable) {
56 size_t timeSampleCount =
57 primitiveAttributeTable.getTimeSampleCount(kv.first);
58 attrTimeSampleCount[kv.first] = timeSampleCount;
59 if (timeSampleCount > attrMaxNumTimes) {
60 attrMaxNumTimes = timeSampleCount;
61 }
62 }
63 std::vector<size_t> attrKeyOffset(attrNumKeys * attrMaxNumTimes, (size_t)-1);
64 std::vector<AttributeRate> attrKeyRate(attrNumKeys, RATE_UNKNOWN);
65 size_t attrNumFaces = numUniforms;
66 std::vector<size_t> attrBeginFaceVarying;
67 attrBeginFaceVarying.reserve(attrNumFaces);
68 size_t attrNumFaceVaryings = 0;
69 for (size_t f = 0; f < numFaceVaryings.size(); ++f) {
70 attrBeginFaceVarying.push_back(attrNumFaceVaryings);
71 attrNumFaceVaryings += numFaceVaryings[f];
72 }
73
74 // make sure data in allocated buffer aligned
75 // we sort the key by attribute size, and build key->offset table
76 // with size descending order (to avoid wasted padding space)
77 size_t dataSize[RATE_LAST] = {};
78 std::map<size_t, std::vector<AttributeKey> > attrSizeMap;
79 for (const auto& kv : primitiveAttributeTable) {
80 AttributeKey key = kv.first;
81 attrSizeMap[key.getSize()].push_back(key);
82 }
83
84 for (auto rit = attrSizeMap.rbegin(); rit != attrSizeMap.rend(); ++rit) {
85 size_t attrSize = rit->first;
86 for (auto k : rit->second) {
87 size_t nT = primitiveAttributeTable.getTimeSampleCount(k);
88 AttributeRate rate = primitiveAttributeTable.getRate(k);
89 attrKeyRate[k] = rate;
90 // Set the key offsets for each time sample. Time samples
91 // are adjacent to each other and separated by the key data size.
92 for (size_t t = 0; t < nT; ++t) {
93 attrKeyOffset[t * attrNumKeys + k] =
94 dataSize[rate] + t * attrSize;
95 }
96 // When a key has fewer time samples that attrMaxNumTimes,
97 // set the offsets of all subsequent times to point to the last one.
98 // This allows us to take time samples of non-motion-blurred
99 // attributes without requiring a check for MB on the attribute.
100 for (size_t t = nT; t < attrMaxNumTimes; ++t) {
101 attrKeyOffset[t * attrNumKeys + k] =
102 dataSize[rate] + (nT - 1) * attrSize;
103 }
104 dataSize[rate] += attrSize * nT;
105 }
106 }
107 for (size_t i = 0; i < RATE_LAST; i++) {
108 dataSize[i] = scene_rdl2::util::alignUp(dataSize[i], sizeof(float));
109 }
110 // vertex rate alignment is a bit weird since the OSD primitive attribute
111 // tessellator require data to be Vec2f aligned
112 dataSize[RATE_VERTEX] = scene_rdl2::util::alignUp(dataSize[RATE_VERTEX], sizeof(Vec2f));
113
114 size_t attrNumParts = numParts;
115 size_t attrNumVaryings = numVaryings;
116 size_t attrNumVertices = numVertices;
117 size_t attrPartStride = dataSize[RATE_PART];
118 size_t attrUniformStride = dataSize[RATE_UNIFORM];
119 size_t attrVaryingStride = dataSize[RATE_VARYING];
120 size_t attrFaceVaryingStride = dataSize[RATE_FACE_VARYING];
121 size_t attrVertexStride = dataSize[RATE_VERTEX];
122
123 size_t constantSize = dataSize[RATE_CONSTANT];
124 size_t partSize = dataSize[RATE_PART]*attrNumParts;
125 size_t uniformSize = dataSize[RATE_UNIFORM]*attrNumFaces;
126 size_t varyingSize = dataSize[RATE_VARYING]*attrNumVaryings;
127 size_t faceVaryingSize = dataSize[RATE_FACE_VARYING]*attrNumFaceVaryings;
128 size_t vertexSize = dataSize[RATE_VERTEX]*attrNumVertices;
129
130 Vector<char> attrConstants(constantSize);
131 Vector<char> attrParts(partSize);
132 Vector<char> attrUniforms(uniformSize);
133 Vector<char> attrVaryings(varyingSize);
134 Vector<char> attrFaceVaryings(faceVaryingSize);
135 Vector<char> attrVertices(vertexSize);
136
137 // Fill in data
138 for (const auto& kv : primitiveAttributeTable) {
139 AttributeKey key = kv.first;
140 AttributeRate keyRate = primitiveAttributeTable.getRate(key);
141 for (size_t t = 0; t < kv.second.size(); ++t) {
142 const auto& primitiveAttribute = kv.second[t];
143 switch (keyRate) {
144 case RATE_CONSTANT:
145 {
146 size_t offset = attrKeyOffset[t * attrNumKeys + key];
147 primitiveAttribute->fetchData(0, &attrConstants[offset]);
148 }
149 break;
150 case RATE_PART:
151 {
152 for (size_t p = 0; p < attrNumParts; p++) {
153 size_t offset = p * attrPartStride +
154 attrKeyOffset[t * attrNumKeys + key];
155 primitiveAttribute->fetchData(p, &attrParts[offset]);
156 }
157 }
158 break;
159 case RATE_UNIFORM:
160 {
161 for (size_t f = 0; f < attrNumFaces; f++) {
162 size_t offset = f * attrUniformStride +
163 attrKeyOffset[t * attrNumKeys + key];
164 primitiveAttribute->fetchData(f, &attrUniforms[offset]);
165 }
166 }
167 break;
168 case RATE_VARYING:
169 {
170 for (size_t v = 0; v < attrNumVaryings; v++) {
171 size_t offset = v * attrVaryingStride +
172 attrKeyOffset[t * attrNumKeys + key];
173 primitiveAttribute->fetchData(v, &attrVaryings[offset]);
174 }
175 }
176 break;
177 case RATE_FACE_VARYING:
178 {
179 size_t fv = 0;
180 for (size_t f = 0; f < attrNumFaces; f++) {
181 size_t nV = numFaceVaryings[f];
182 size_t beginVarying = attrBeginFaceVarying[f];
183 for (size_t v = 0; v < nV; v++) {
184 size_t offset =
185 (v + beginVarying) * attrFaceVaryingStride +
186 attrKeyOffset[t * attrNumKeys + key];
187 primitiveAttribute->fetchData(fv,
188 &attrFaceVaryings[offset]);
189 fv++;
190 }
191 }
192 }
193 break;
194 case RATE_VERTEX:
195 {
196 for (size_t v = 0; v < attrNumVertices; v++) {
197 size_t offset = v * attrVertexStride +
198 attrKeyOffset[t * attrNumKeys + key];
199 primitiveAttribute->fetchData(v, &attrVertices[offset]);
200 }
201 }
202 break;
203 default:
204 MNRY_ASSERT(false, "Rate unknown");
205 break;
206 }
207 }
208 }
209
210 std::unique_ptr<Attributes> result(new Attributes(attrNumKeys, std::move(attrKeyOffset),
211 std::move(attrTimeSampleCount), std::move(attrKeyRate),
212 attrMaxNumTimes, attrNumParts, attrNumFaces, attrNumVaryings,
213 attrNumFaceVaryings, std::move(attrBeginFaceVarying),
214 attrNumVertices, attrPartStride, attrUniformStride, attrVaryingStride,
215 attrFaceVaryingStride, attrVertexStride,
216 std::move(attrConstants), std::move(attrParts), std::move(attrUniforms),
217 std::move(attrVaryings), std::move(attrFaceVaryings),
218 std::move(attrVertices)));
219 return result;
220}

222void Attributes::negateNormal()
223{
224 TypedAttributeKey<Vec3f> key = StandardAttributes::sNormal;
225 if (!isSupported(key)) {
226 return;
227 }
228
229 size_t timeSampleCount = getTimeSampleCount(key);
230 AttributeRate keyRate = getRate(key);
231 switch (keyRate) {
232 case RATE_CONSTANT:
233 {
234 for (size_t t = 0; t < timeSampleCount; ++t) {
235 Vec3f& vec3 = getConstantInternal(key, t);
236 vec3 = -vec3;
237 }
238 }
239 break;
240 case RATE_UNIFORM:
241 {
242 for (size_t t = 0; t < timeSampleCount; ++t) {
243 for (size_t f = 0; f < mNumFaces; ++f) {
244 Vec3f& vec3 = getUniformInternal(key, f, t);
245 vec3 = -vec3;
246 }
247 }
248 }
249 break;
250 case RATE_VARYING:
251 {
252 for (size_t t = 0; t < timeSampleCount; ++t) {
253 for (size_t v = 0; v < mNumVaryings; ++v) {
254 Vec3f& vec3 = getVaryingInternal(key, v, t);
255 vec3 = -vec3;
256 }
257 }
258 }
259 break;
260 case RATE_FACE_VARYING:
261 {
262 for (size_t t = 0; t < timeSampleCount; ++t) {
263 for (size_t v = 0; v < mNumFaceVaryings; ++v) {
264 Vec3f& vec3 = getFaceVaryingInternal(key, v, t);
265 vec3 = -vec3;
266 }
267 }
268 }
269 break;
270 case RATE_VERTEX:
271 {
272 for (size_t t = 0; t < timeSampleCount; ++t) {
273 for (size_t v = 0; v < mNumVertices; ++v) {
274 Vec3f& vec3 = getVertexInternal(key, v, t);
275 vec3 = -vec3;
276 }
277 }
278 }
279 break;
280 default:
281 MNRY_ASSERT(false, "Rate unknown");
282 break;
283 }

286void Attributes::reverseFaceVaryingAttributes()
287{
288 if (mFaceVaryings.empty()) {
289 return;
290 }
291
292 Vector<char> tmp(mFaceVaryingStride);
293 for (size_t f = 0; f < mNumFaces; f++) {
294 size_t begin = mFaceVaryingStride * mBeginFaceVarying[f];
295 size_t end = f == mNumFaces - 1 ? mFaceVaryings.size() :
296 mFaceVaryingStride * mBeginFaceVarying[f+1];
297 size_t nV = (end - begin) / mFaceVaryingStride;
298
299 for (size_t v = 0; v < nV / 2; v++) {
300 memcpy(tmp.data(), &mFaceVaryings[begin + v*mFaceVaryingStride], mFaceVaryingStride);
301 memcpy(&mFaceVaryings[begin + v*mFaceVaryingStride],
302 &mFaceVaryings[end - (v+1)*mFaceVaryingStride], mFaceVaryingStride);
303 memcpy(&mFaceVaryings[end - (v+1)*mFaceVaryingStride], tmp.data(), mFaceVaryingStride);
304 }
305 }

308static void
309transformVec3(Vec3f& vec3, Vec3Type type, const mcrt_common::Mat43& xform,
310 const mcrt_common::Mat43& invXform)
311{
312 if (type == Vec3Type::POINT) {
313 vec3 = transformPoint(xform, vec3);
314 } else if (type == Vec3Type::VECTOR) {
315 vec3 = transformVector(xform, vec3);
316 } else if (type == Vec3Type::NORMAL) {
317 vec3 = transformNormal(invXform, vec3);
318 } else {
319 MNRY_ASSERT(false, "unknown Vec3Type");
320 }
321}

323void
324Attributes::transformAttributes(const shading::XformSamples& xforms,
325 float shutterOpenDelta, float shutterCloseDelta,
326 const std::vector<Vec3KeyType>& keyTypePairs)
327{
328 shading::XformSamples invXforms;
329 invXforms.reserve(xforms.size());
330 for (size_t i = 0; i < xforms.size(); ++i) {
331 invXforms.push_back(xforms[i].inverse());
332 }
333
334 // In rare cases the number of xforms can be less than the number of attribute
335 // time samples. This can happen if there is a problem with the motion blur
336 // data on the vertices and we fell back to the static motion blur case.
337 // In that case we choose the closest xform available as we can still interpolate
338 // the attribute values even though they use the same xform.
339 // i.e. xt = std::min(xforms.size() - 1, t);
340
341 for (auto keyTypePair : keyTypePairs) {
342 TypedAttributeKey<Vec3f> key = keyTypePair.first;
343 Vec3Type type = keyTypePair.second;
344 if (!isSupported(key)) {
345 continue;
346 }
347 size_t timeSampleCount = getTimeSampleCount(key);
348 AttributeRate keyRate = getRate(key);
349 switch (keyRate) {
350 case RATE_CONSTANT:
351 {
352 for (size_t t = 0; t < timeSampleCount; ++t) {
353 size_t xt = std::min(xforms.size() - 1, t);
354 Vec3f& vec3 = getConstantInternal(key, t);
355 transformVec3(vec3, type, xforms[xt], invXforms[xt]);
356 }
357 }
358 break;
359 case RATE_UNIFORM:
360 {
361 for (size_t t = 0; t < timeSampleCount; ++t) {
362 size_t xt = std::min(xforms.size() - 1, t);
363 for (size_t f = 0; f < mNumFaces; ++f) {
364 Vec3f& vec3 = getUniformInternal(key, f, t);
365 transformVec3(vec3, type, xforms[xt], invXforms[xt]);
366 }
367 }
368 }
369 break;
370 case RATE_VARYING:
371 {
372 for (size_t t = 0; t < timeSampleCount; ++t) {
373 size_t xt = std::min(xforms.size() - 1, t);
374 for (size_t v = 0; v < mNumVaryings; ++v) {
375 Vec3f& vec3 = getVaryingInternal(key, v, t);
376 transformVec3(vec3, type, xforms[xt], invXforms[xt]);
377 }
378 }
379 }
380 break;
381 case RATE_FACE_VARYING:
382 {
383 for (size_t t = 0; t < timeSampleCount; ++t) {
384 size_t xt = std::min(xforms.size() - 1, t);
385 for (size_t v = 0; v < mNumFaceVaryings; ++v) {
386 Vec3f& vec3 = getFaceVaryingInternal(key, v, t);
387 transformVec3(vec3, type, xforms[xt], invXforms[xt]);
388 }
389 }
390 }
391 break;
392 case RATE_VERTEX:
393 {
394 for (size_t t = 0; t < timeSampleCount; ++t) {
395 size_t xt = std::min(xforms.size() - 1, t);
396 for (size_t v = 0; v < mNumVertices; ++v) {
397 Vec3f& vec3 = getVertexInternal(key, v, t);
398 transformVec3(vec3, type, xforms[xt], invXforms[xt]);
399 }
400 }
401 }
402 break;
403 default:
404 MNRY_ASSERT(false, "Rate unknown");
405 break;
406 }
407 }

410}

412void
413Attributes::motionBlurInterpolate(
414 float shutterOpenDelta, float shutterCloseDelta)
415{
416 for (size_t k = 0; k < mNumKeys; ++k) {
417 AttributeKey key(k);
418 if (!isSupported(key)) {
419 continue;
420 }
421 // no need to interpolate static primitive attributes
422 if (getTimeSampleCount(key) < 2) {
423 continue;
424 }
425 switch (key.getType()) {
426 case rdl2::TYPE_FLOAT:
427 motionBlurInterpolate(TypedAttributeKey<float>(key),
428 shutterOpenDelta, shutterCloseDelta);
429 break;
430 case rdl2::TYPE_RGB:
431 motionBlurInterpolate(TypedAttributeKey<math::Color>(key),
432 shutterOpenDelta, shutterCloseDelta);
433 break;
434 case rdl2::TYPE_RGBA:
435 motionBlurInterpolate(TypedAttributeKey<math::Color4>(key),
436 shutterOpenDelta, shutterCloseDelta);
437 break;
438 case rdl2::TYPE_VEC2F:
439 motionBlurInterpolate(TypedAttributeKey<Vec2f>(key),
440 shutterOpenDelta, shutterCloseDelta);
441 break;
442 case rdl2::TYPE_VEC3F:
443 motionBlurInterpolate(TypedAttributeKey<Vec3f>(key),
444 shutterOpenDelta, shutterCloseDelta);
445 break;
446 case rdl2::TYPE_MAT4F:
447 motionBlurInterpolate(TypedAttributeKey<math::Mat4f>(key),
448 shutterOpenDelta, shutterCloseDelta);
449 break;
450 default:
451 break;
452 }
453 }
