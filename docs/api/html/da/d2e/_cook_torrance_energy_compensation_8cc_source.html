11/*
12 * This file is used to compensate for energy loss from
13 * microfacet reflection lobes as outlined in Walter'07.
14 * We use the Kelemen'01 energy compensation BRDF as described in
15 * Kulla'17 to be added to the regular microfacet BRDF.
16 * This file describes the term to be added in reflection
17 * to be perfectly energy 'preserving'. It also provides
18 * sampling curves, PDF and a one-sample weight to sample
19 * this term efficiently.
20 * Ref:
21 * (1) A Microfacet Based Coupled Specular-Matte BRDF Model
22 * with Importance Sampling, Kelemen, 01
23 * (2) Revisiting Physically Based Shading, PBS Kulla'17
24 */

35// BRDF Compensation Term as outlined in
36// Kelemen'01 Section 2.2
37// (1 - E(cosThetaO)*(1-E(cosThetaI))/(1 - Eavg) * cosThetaI
38scene_rdl2::math::Color
39CookTorranceEnergyCompensation::eval(
40 ispc::MicrofacetDistribution type,
41 const float cosNO,
42 const float cosNI,
43 float roughness,
44 const scene_rdl2::math::Color& favg,
45 bool includeCosineTerm)
46{
47 float compensation =
48 (1.0f - ReflectionAlbedo::E(type,
49 cosNO, roughness)) *
50 (1.0f - ReflectionAlbedo::E(type,
51 cosNI, roughness));
52 const float divisor = ReflectionAlbedo::oneMinusAvg(type,
53 roughness);
54 if (!scene_rdl2::math::isZero(divisor)) {
55 compensation /= divisor;
56 } else {
57 compensation = 0.0f;
58 }
59
60 const float eavg = 1.0f - ReflectionAlbedo::oneMinusAvg(type,
61 roughness);
62 const Color multiplier = favg * favg * eavg /
63 (scene_rdl2::math::sWhite - favg*(1.0f - eavg));
64
65 const Color comp = compensation * multiplier * sOneOverPi;
66
67 return (comp *
68 (includeCosineTerm ? cosNI : 1.0f));
69}

71// Sampling Curves Determined by fitting a curve
72// to tabulated CDFInverse data calculated using
73// moonshine_devtools/compute_cdf based on the PDF
74// outlined in Kelemen'01
75void
76CookTorranceEnergyCompensation::sample(
77 ispc::MicrofacetDistribution type,
78 const scene_rdl2::math::ReferenceFrame& frame,
79 float r1, float r2,
80 float r,
81 scene_rdl2::math::Vec3f& wi)
82{
83 float cosThetaI;
84 switch(type) {
85 case ispc::MICROFACET_DISTRIBUTION_BECKMANN:
86 cosThetaI = pow(r1,
87 2.767f*r*r-5.4f*r+2.99412f);
88 break;
89 case ispc::MICROFACET_DISTRIBUTION_GGX:
90 default:
91 cosThetaI = pow(r1,
92 1.1933f*r*r-2.0969f*r+1.3698f);
93 }
94
95 const float sinThetaI = sqrt(1.0f - cosThetaI*cosThetaI);
96 const float phiI = 2.0f * sPi * r2;
97
98 const Vec3f m = computeLocalSphericalDirection(cosThetaI, sinThetaI, phiI);
99 wi = frame.localToGlobal(m);
100}

102// PDF -
103// Kelemen'01 Section 2.2.1
104// (1-E(cosThetaI))/(pi*(1 - Eavg)) * cosThetaI
105float
106CookTorranceEnergyCompensation::pdf(
107 ispc::MicrofacetDistribution type,
108 float cosNI,
109 float roughness)
110{
111 float result = 0.0f;
112 const float norm = sPi * ReflectionAlbedo::oneMinusAvg(type,
113 roughness);
114 if (isZero(norm)) return 0.0f;
115 result = (cosNI * (1.0f - ReflectionAlbedo::E(type,
116 cosNI,
117 roughness)));
118 result /= norm;
119 return max(0.0f, result);
120}

122float
123CookTorranceEnergyCompensation::weight(
124 float roughness)
125{
126 const float w = max(0.0f, (roughness-0.5f))*1.0f;
127 return w;
128}
