33// Cheap approximation of the diffuse albedo.
34Color HairDiffuseLobe::albedo(const BsdfSlice &slice) const
35{
36 Vec3f I = slice.getWo();
37 float cosTheta = dot(I, mHairDir);
38 return getScale() * scene_rdl2::math::sqrt(scene_rdl2::math::abs(1.0f - cosTheta * cosTheta)) / sPiSqr;
39}

42Color HairDiffuseLobe::evalBsdf(const Vec3f& wo,
43 const Vec3f &wi,
44 bool includeCosineTerm) const
45{
46 const Vec3f &hairDir = mHairDir;
47
48 const Vec3f hd_cross_i = cross(hairDir, wi);
49
50 Color hairColor = mHairColorRefl;
51 if (!isEqual(mHairColorRefl, mHairColorTrans)) {
52
53 const Vec3f hd_cross_o = cross(hairDir, wo);
54
55 const float denominator =
56 scene_rdl2::math::max(length(hd_cross_i) * length(hd_cross_o),
57 sEpsilon);
58
59 // Compute the proportion of transmission color vs. reflection color.
60 const float kappa = dot(hd_cross_i, hd_cross_o) / denominator;
61 const float frontIntensity = (1.0f + kappa) / 2.0f;
62 const float backIntensity = 1.0f - frontIntensity;
63
64 hairColor = frontIntensity * mHairColorRefl +
65 backIntensity * mHairColorTrans;
66 }
67
68 // Here we want a lambertian reflectivity and we need to include
69 // the cosine term per the slice. If we don't include the cosine term, then
70 // the mMaxDiffAngle feature is incompatible (a hack anyways) and not
71 // enabled at all.
72 //result /= cosTerm;
73 Color result = hairColor;
74 if (includeCosineTerm) {
75 // Note: cos(thetaI) == sin(sin_hd_i)
76 float min_sin_hd_i = scene_rdl2::math::cos(mMaxDiffAngle);
77 float cos_hd_i = dot(hairDir, wi);
78 float sin_hd_i = max(min_sin_hd_i, scene_rdl2::math::sqrt(scene_rdl2::math::abs(1 - cos_hd_i * cos_hd_i)));
79 result *= sin_hd_i / sPiSqr;
80 } else {
81 result *= 1.0f / sPiSqr;
82 }
83
84 return getScale() * result;
85}

88float HairDiffuseLobe::evalPdf(const Vec3f &wi) const
89{
90#ifdef PBR_HAIR_DIFFUSE_USE_UNIFORM_SAMPLING
91
92 return 1.0f / (4.0f * sPi);
93
94#else
95 Vec3f hairDir = mHairDir;
96 float cosTheta = dot(wi, hairDir);
97 float pdf = scene_rdl2::math::sqrt(std::max(1.0f - cosTheta*cosTheta, 0.f));
98
99 return pdf / sPiSqr;
100#endif
101}

114Color HairDiffuseLobe::sample(const BsdfSlice &slice, float r1, float r2, Vec3f &wi, float &pdf) const
115{
116 HairState hairState(slice.getWo(),
117 mHairDir);
118
119#ifdef PBR_HAIR_DIFFUSE_USE_UNIFORM_SAMPLING
120
121 Vec3f uvw = shading::sampleSphereUniform(r1, r2);
122 wi = (uvw[0] * mHairDir) +
123 (uvw[1] * hairState.hairNormal()) +
124 (uvw[2] * hairState.hairBinormal());
125
126 // We shouldn't need to normalize!
127 MNRY_ASSERT(isNormalized(wi));
128
129 pdf = 1.0f / (4.0f * sPi);
130
131#else
132
133 float sinPhi, cosPhi;
134
135 // Basically, we're sampling a Lambertian in the longitudinal
136 // plane and uniform circular in the azimuthal plane.
137 float u = 2.0f * (r1 - 0.5f);
138 float v = r2;
139
140 float sinTheta = u;
141 float cosTheta = scene_rdl2::math::sqrt(1.0f - u*u);
142 float phi = v * sTwoPi;
143 sincos(phi, &sinPhi, &cosPhi);
144
145 // Compute the light direction vector for shading.
146 float uWgt = sinTheta;
147 float vWgt = cosTheta * sinPhi;
148 float wWgt = cosTheta * cosPhi;
149 wi = (uWgt * mHairDir) +
150 (vWgt * hairState.hairNormal()) +
151 (wWgt * hairState.hairBinormal());
152 wi.normalize();
153
154 pdf = cosTheta / sPiSqr;
155#endif
156 return evalBsdf(slice.getWo(),
157 wi,
158 slice.getIncludeCosineTerm());
159}

161bool
162HairDiffuseLobe::getProperty(Property property, float *dest) const
163{
164 bool handled = true;
165
166 switch (property)
167 {
168 case PROPERTY_COLOR:
169 {
170 *dest = mHairColorRefl[0];
171 *(dest + 1) = mHairColorRefl[1];
172 *(dest + 2) = mHairColorRefl[2];
173 }
174 break;
175 case PROPERTY_ROUGHNESS:
176 {
177 *dest = 1.0f;
178 *(dest + 1) = 1.0f;
179 }
180 break;
181 case PROPERTY_NORMAL:
182 {
183 *dest = mHairDir.x;
184 *(dest + 1) = mHairDir.y;
185 *(dest + 2) = mHairDir.z;
186 }
187 break;
188 case PROPERTY_PBR_VALIDITY:
189 {
190 // TODO: decide if refl/trans colors are valid
191 *dest = 0.0f;
192 *(dest + 1) = 1.0f;
193 *(dest + 2) = 0.0f;
194 }
195 default:
196 handled = BsdfLobe::getProperty(property, dest);
197 break;
198 }
199
200 return handled;
201}
