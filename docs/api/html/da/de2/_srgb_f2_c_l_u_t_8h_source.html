15//
16// -- Generating lookup table for conversion from linear float to sRGB uchar
17//
18// This class is designed for generating lookup table for direct conversion from 32bit
19// single float (linear space) to sRGB space 8bit unsigned char.
20// Also includes several different test and verify functions.
21//
22// This class is not used runtime of moonray. But I would like to keep this code inside
23// moonray because it's very useful to understand basic idea of direct float to uchar
24// lookup table from linear to sRGB conversion. Toshi (Sep/25/2020)
25//
26class SrgbF2CLUT
27{
28public:
29 SrgbF2CLUT() { mUni.mI = 0; }
30
31 //------------------------------
32 //
33 // raw level set/get/show for single float value
34 //
35 void set(const float f) { mUni.mF = f; }
36 void set(const int signBit, const int exponent, const int mantissa) {
37 mUni.mI = (((signBit & 0x1) << 31) | ((exponent & 0xff) << 23) | (mantissa & 0x7fffff));
38 }
39
40 float getF() const { return mUni.mF; }
41
42 std::string show() const;
43
44 //------------------------------
45 //
46 // table generation
47 //
48 static std::string tblGen15bit();
49
50protected:
51 union {
52 float mF;
53 unsigned int mI;
54 } mUni;
55
56 std::string showBit() const;
57 std::string showSignBit() const { return showMask(31, 31, mUni.mI); }
58 std::string showExponentBit() const { return showMask(30, 23, mUni.mI); }
59 std::string showMantissaBit() const { return showMask(22, 0, mUni.mI); }
60 std::string showAllBit() const { return showMask(31, 0, mUni.mI); }
61 std::string showMask(const int left, const int right, const unsigned int d) const;
62}; // SrgbF2CLUT
