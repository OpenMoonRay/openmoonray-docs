18class TlSvr
19//
20// This class provides the functionality of server side telnet connection.
21// Only support p2p (point to point) connection so far and not support multiple
22// incoming input from multiple clients. Only support IPv4 so far.
23// Using this class, it is very easy to implement interactive command line console
24// functionality to the non interactive application.
25//
26// The following is a pseudo code of typical implementation for command line console by TlSvr.
27// This implementation is one of the threads of the server process.
28//
29// TlSvr svr;
30// if (!svr.open(20000)) return; // port is 20000
31//
32// while (1) {
33// std::string cmdLine;
34// int recvByte = svr.recv(cmdLine);
35// if (recvByte == 0) {
36// usleep(10000); // 10ms sleep for yielding CPU resource
37// } else if (recvByte < 0) {
38// // error
39// break;
40// } else {
41// // parse cmdLine here and do something ...
42// svr.send("..test..test..test\n"); // you can send back string to the telnet client.
43// if (cmdLine == "exit") break;
44// }
45// }
46// svr.close();
47//
48{
49public:
50 // These 2 callback definitions are used to define message display functionality.
51 // We have 2 categories. info and error.
52 // Info callback is called for information dump purposes like a new connection established.
53 // Error callback is called when an error has happened internally.
54 // You don't need to set a call back function if you don't want to.
55 using INFOMSG_CALLBACK = std::function<void(const std::string &)>;
56 using ERRMSG_CALLBACK = std::function<void(const std::string &)>;
57
58 TlSvr();
59 ~TlSvr();
60
61 //
62 // You can use serverPortNum = 0 for auto search of available port by the kernel.
63 // In this case, you can figure out the result port number by the return value of this API.
64 // Port is not opened until when it is needed (delayed open). If you specify a port number
65 // for the open API, this port is not opened until the very first recv operation.
66 // This open API only supports internet domain socket and not unix domain.
67 //
68 // If you don't call open(), all other public API internally skip all socket related execution.
69 // This means skip open() is an easiest way to disable all TlSvr functionality.
70 //
71 int open(const int serverPortNum,
72 INFOMSG_CALLBACK infoMsgCallBack = nullptr,
73 ERRMSG_CALLBACK errMsgCallBack = nullptr); // return opened port number. 0 is error
74
75 // non blocking receive : return recv byte or 0:empty -1:EOF -2:otherError
76 int recv(std::string & recvStr,
77 INFOMSG_CALLBACK infoMsgCallBack = nullptr,
78 ERRMSG_CALLBACK errMsgCallBack = nullptr);
79
80 // blocking send
81 // Send only works after established socket connection by recv API.
82 bool send(const std::string & sendStr,
83 INFOMSG_CALLBACK infoMsgCallBack = nullptr,
84 ERRMSG_CALLBACK errMsgCallBack = nullptr);
85
86 void close();
87
88 bool isConnectionEstablised() const; // true:establised false:not_yet
89
90protected:
91 bool setupServerPort(INFOMSG_CALLBACK infoMsgCallBack, ERRMSG_CALLBACK errMsgCallBack);
92 bool socketBindAndListen(INFOMSG_CALLBACK infoMsgCallBack, ERRMSG_CALLBACK errMsgCallBack);
93 bool acceptSocket(INFOMSG_CALLBACK infoMsgCallBack, ERRMSG_CALLBACK errMsgCallBack);
94 void connectionClosed(INFOMSG_CALLBACK infoMsgCallBack);
95 int socketCheck(ERRMSG_CALLBACK errMsgCallBack);
96
97 int mPort; // server port
98 int mBaseSock; // server socket fd
99 int mSock; // incoming socket fd
100 bool mConnectionReady; // connection-established(true) or not(false)
101 int mRecvSize; // received data size
102 bool mEndLine; // end line condition
103};
