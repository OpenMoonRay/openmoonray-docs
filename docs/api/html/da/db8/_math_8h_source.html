309 /*! bit reverse operation */
310 template<class T>
311 __forceinline T bitReverse(const T& vin)
312 {
313 T v = vin;
314 v = ((v >> 1) & 0x55555555) | ((v & 0x55555555) << 1);
315 v = ((v >> 2) & 0x33333333) | ((v & 0x33333333) << 2);
316 v = ((v >> 4) & 0x0F0F0F0F) | ((v & 0x0F0F0F0F) << 4);
317 v = ((v >> 8) & 0x00FF00FF) | ((v & 0x00FF00FF) << 8);
318 v = ( v >> 16 ) | ( v << 16);
319 return v;
320 }

322 /*! bit interleave operation */
323 template<class T>
324 __forceinline T bitInterleave(const T& xin, const T& yin, const T& zin)
325 {
326 T x = xin, y = yin, z = zin;
327 x = (x | (x << 16)) & 0x030000FF;
328 x = (x | (x << 8)) & 0x0300F00F;
329 x = (x | (x << 4)) & 0x030C30C3;
330 x = (x | (x << 2)) & 0x09249249;
331
332 y = (y | (y << 16)) & 0x030000FF;
333 y = (y | (y << 8)) & 0x0300F00F;
334 y = (y | (y << 4)) & 0x030C30C3;
335 y = (y | (y << 2)) & 0x09249249;
336
337 z = (z | (z << 16)) & 0x030000FF;
338 z = (z | (z << 8)) & 0x0300F00F;
339 z = (z | (z << 4)) & 0x030C30C3;
340 z = (z | (z << 2)) & 0x09249249;
341
342 return x | (y << 1) | (z << 2);
343 }

410// C++ implementation to match ISPC version of erf() in Math.isph.
411__forceinline float erf(const float x) {
412 // Algorithm 7.1.25 in Abromowitz and Stegun, Handbook of Mathematical
413 // Functions With Formulas, Graphs, and Mathematical Tables (pg. 299)
414 // (maximum error: 2.5×10−5)
415 // http://www.math.ubc.ca/~cbm/aands/abramowitz_and_stegun.pdf (pg. 88)
416 const float p = 0.47047f;
417 const float a1 = 0.3480242f;
418 const float a2 = -0.0958798f;
419 const float a3 = 0.7478556f;
420 float t = 1.0f / (1.0f + p * abs(x));
421 float result = 1.0f - t * (a1 + t * (a2 + t * a3)) * exp( -(x * x) );
422 return copysignf(result, x);
423}

433#if defined(__FMA__)
434// Return a*b - c*d in a way that avoids catastrophic cancellation.
435// https://pharr.org/matt/blog/2019/11/03/difference-of-floats
436template <typename T, typename U, typename V, typename W>
437inline auto differenceOfProducts(const T& a, const U& b, const V& c, const W& d) noexcept
438{
439 const auto cd = c * d;
440 const auto err = nmadd(c, d, cd); // Compute the error in computing c*d
441 const auto dop = msub(a, b, cd);
442 return dop + err;
443}

448{
