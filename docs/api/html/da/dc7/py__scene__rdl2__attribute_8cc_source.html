214 template <typename T>
215 void
216 registerAttributeKeyPyBinding(const std::string& rdl2TypeName)
217 {
218 const std::string rdl2AttrKeyDocstring =
219 "An AttributeKey of type '" + rdl2TypeName + "' is a lightweight object for "
220 "retrieving the value of an attribute of this specific type ('" + rdl2TypeName + "') from a SceneObject."
221 "\n"
222 "AttributeKeys are templated on a C++ type corresponding to their attribute type. This allows "
223 "us to do static typechecking wherever possible, and most importantly, do fast, typesafe "
224 "gets and sets on attribute values."
225 "\n"
226 "AttributeKeys are lightweight (16 bytes), and can be compared for equality. However, "
227 "comparing AttributeKeys from different SceneClasses is invalid, and the result of "
228 "such a comparison is undefined."
229 "\n"
230 " AttributeKeys that are default constructed (not assigned from a valid\n"
231 " AttributeKey or constructed from an Attribute) are invalid until a valid\n"
232 " AttributeKey is assigned into them.\n"
233 "\n"
234 "Thread Safety:\n"
235 " - All data members are baked in at construction time. Since AttributeKey "
236 "objects are immutable after construction, reading their members from "
237 "multiple threads without synchronization is safe.";
238
239 using AttrKeyT_t = typename rdl2::AttributeKey<T>;
240 using PyAttributeKeyTClass_t = bp::class_<AttrKeyT_t, std::shared_ptr<AttrKeyT_t>>;
241
242 const std::string pyClassName = "AttributeKey" + rdl2TypeName;
243
244 PyAttributeKeyTClass_t(pyClassName.c_str(), rdl2AttrKeyDocstring.c_str(),
245 bp::init<const rdl2::Attribute&>(bp::arg("attribute")))
246
247 .def("__eq__", &AttrKeyT_t::operator==)
248 .def("__ne__", &AttrKeyT_t::operator!=)
249
250 .def("isValid",
251 &AttrKeyT_t::isValid,
252 "Returns true if the attribute key is valid. Default constructed "
253 "AttributeKeys are not valid.")
254
255 .def("isBindable",
256 &AttrKeyT_t::isBindable,
257 "Returns true if the underlying attribute is bindable.")
258
259 .def("isBlurrable",
260 &AttrKeyT_t::isBlurrable,
261 "Returns true if the underlying attribute is blurrable.")
262
263 .def("isEnumerable",
264 &AttrKeyT_t::isEnumerable,
265 "Returns true if the underlying attribute is an enumeration.")
266
267 .def("isFilename",
268 &AttrKeyT_t::isFilename,
269 "Returns true if the underlying attribute represents a filename.")
270 ;
271 }
