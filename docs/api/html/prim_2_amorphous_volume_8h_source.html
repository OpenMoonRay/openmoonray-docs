68 struct AmorphousSampler
69 {
70 using ScatterParams = amorphous::SScatterParams;
71 using ScatterSampler = amorphous::SScatterSampler;
72
73 const ScatterParams& scatterParams() const { return mScatterParams; }
74
75 // Initialize from scattering parameters.
76 bool initialize(const ScatterParams& scatterParams,
77 const std::vector<int>& volumeIds,
78 const scene_rdl2::rdl2::Geometry* rdlGeometry = nullptr)
79 {
80 if (!scatterParams.extinctionGrid() ||
81 scatterParams.extinctionGrid()->empty())
82 {
83 if (rdlGeometry) {
84 rdlGeometry->error("An extinction"
85 " (AKA opacity or shadowing) volume is required.");
86 }
87 return false;
88 }
89
90 mScatterParams = scatterParams;
91 mMasterSampler.reset(new ScatterSampler(mScatterParams));
92 mVolumeIdCount = volumeIds.size();
93 for (unsigned samplerId = 0; samplerId < mVolumeIdCount; ++samplerId) {
94 mVolumeIdToSamplerId[volumeIds[samplerId]] = samplerId;
95 }
96 unsigned samplerCount = mVolumeIdCount * mcrt_common::getNumTBBThreads();
97 for (unsigned i = 0; i < samplerCount; ++i) {
98 mThreadSamplers.push_back(*mMasterSampler);
99 }
100 return true;
101 }
102
103 // Initialize from a .vdb file.
104 bool initialize(const std::string& vdbFilePath,
105 const std::string& densityGridName,
106 const scene_rdl2::rdl2::Geometry& rdlGeometry,
107 const float emissionSampleRate,
108 const std::vector<int>& volumeIds)
109 {
110 amorphous::File vdbFile(vdbFilePath);
111 std::stringstream msg;
112 if (!vdbFile.read(msg, densityGridName)) {
113 rdlGeometry.error(msg.str());
114 return false;
115 }
116
117 amorphous::SScatterParams& scatterParams = vdbFile.scatterParams();
118 if (!scatterParams.extinctionGrid() ||
119 scatterParams.extinctionGrid()->empty())
120 {
121 rdlGeometry.error("VDB", vdbFilePath,
122 " contains no extinction metadata. "
123 "Extinction (AKA opacity or shadowing) is required.");
124 return false;
125 }
126
127 // downsample emission grid
128 if (scatterParams.incandescenceGrid() != nullptr &&
129 !scatterParams.incandescenceGrid()->empty() &&
130 emissionSampleRate > 0.0f && emissionSampleRate < 1.0f) {
131 scatterParams.downsampleIncand(emissionSampleRate);
132 }
133
134 return initialize(scatterParams, volumeIds);
135 }

137 scene_rdl2::math::Color extinct(mcrt_common::ThreadLocalState* tls,
138 uint32_t volumeId,
139 const openvdb::Vec3d& p,
140 Interpolation mode) const
141 {
142 tls->mGeomTls->mStatistics.incCounter(STATS_DENSITY_GRID_SAMPLES);
143 uint32_t threadIdx = tls->mThreadIdx;
144 unsigned samplerIdx = threadIdx * mVolumeIdCount + (mVolumeIdToSamplerId.at(volumeId));
145 return scene_rdl2::math::Color(mThreadSamplers[samplerIdx].extinc(
146 p, (amorphous::InterpolationMethod)mode));
147 }

149 scene_rdl2::math::Color albedo(mcrt_common::ThreadLocalState* tls,

154 tls->mGeomTls->mStatistics.incCounter(STATS_COLOR_GRID_SAMPLES);

158 p, (amorphous::InterpolationMethod)mode);

200 size_t getMemory() const
201 {
202 size_t mem = 0;
203 // The ExtincGrid is shared with the mTopology grid
204 // We only need to count the color grid and incand grid,
205 // because they are not stored anywhere else in VdbVolume.
206 if (mScatterParams.hasColor()) {
207 mem += mScatterParams.colorGrid()->memUsage();
208 }
209 if (mScatterParams.hasIncandescence()) {
210 mem += mScatterParams.incandescenceGrid()->memUsage();
211 }
212 if (mMasterSampler) {
213 mem += sizeof(amorphous::SScatterSampler);
214 }
215
216 mem += scene_rdl2::util::getVectorElementsMemory(mThreadSamplers);
217
218 return mem;
219 }
