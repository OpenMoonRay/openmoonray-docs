62Options
63parseCommandLine(int argc, char* argv[])
64{
65 Options options;
66 int index = 1;
67
68 options.simple = false;
69 while (index < argc) {
70 if (strcmp(argv[index], "-s") == 0 ||
71 strcmp(argv[index], "--simple") == 0) {
72 options.simple = true;
73 }
74 if (strcmp(argv[index], "-non_alphabetize") == 0) {
75 options.non_alphabetize = true;
76 } else if (options.rdlFile.empty() &&
77 (access(argv[index], R_OK) == 0)) {
78 options.rdlFile = argv[index];
79 } else if (options.rdlFile.empty()) {
80 options.sceneClass = argv[index];
81 } else if (!options.rdlFile.empty()) {
82 options.sceneObject = argv[index];
83 }
84 index++;
85 }
86 if (argc > 1) {
87 std::string dsoPath = rdl2::DsoFinder::parseDsoPath(argc, argv);
88 if (!dsoPath.empty()) {
89 options.dsoPath = dsoPath;
90 }
91 }
92
93 return options;
94}

96// Return the index of the highest significant bit set
97// Returns -1 of no bit is set
98int
99getHighestBit(const rdl2::SceneObjectInterface soi)
100{
101 int index = 0;
102 int mask = soi;
103 while (mask != 0) {
104 mask = mask >> 1;
105 index++;
106 }
107
108 return index -1;
109}

111bool
112compareSceneClasses(const rdl2::SceneClass *cls1, const rdl2::SceneClass *cls2)
113{
114 // Sort by interface type first, then by class name
115 rdl2::SceneObjectInterface interface1 = cls1->getDeclaredInterface();
116 rdl2::SceneObjectInterface interface2 = cls2->getDeclaredInterface();
117 int highestBit1 = getHighestBit(interface1);
118 int highestBit2 = getHighestBit(interface2);
119 if (highestBit1 == highestBit2) {
120 // Compare class names if equal
121 return (cls1->getName().compare(cls2->getName()) < 0);
122 } else {
123 return (highestBit1 < highestBit2);
124 }
125}

127void
128printAllSceneClasses(const rdl2::SceneContext& ctx, const bool simple, const bool alphabetize)
129{
130 std::vector<rdl2::SceneClass*> array;
131 for (auto iter = ctx.beginSceneClass(); iter != ctx.endSceneClass(); ++iter) {
132 array.push_back((iter->second));
133 }
134
135 if (array.empty()) {
136 return;
137 }
138
139 if (alphabetize) {
140 std::sort(array.begin(), array.end(),
141 [](const rdl2::SceneClass* sc1, const rdl2::SceneClass* sc2)
142 { return (sc1->getName() < sc2->getName());});
143 } else {
144 auto partIT = std::stable_partition(array.begin(), array.end(),
145 [](const rdl2::SceneClass* sc)
146 { return sc->getName() == "SceneVariables"; });
147 std::sort(partIT, array.end(), compareSceneClasses);
148 }
149
150 for (auto iter = array.cbegin(); iter != array.cend(); ++iter) {
151 std::cout << getSceneInfoStr(**(iter), simple, alphabetize);
152 std::cout << '\n';
153 }
154}

173 std::cout << getSceneInfoStr(*array.front(), simple, alphabetize);
174 for (auto iter = std::next(array.begin()); iter != array.end(); ++iter) {
175 std::cout << '\n' << getSceneInfoStr(**(iter), simple, alphabetize);

181int main(int argc, char* argv[])
182{
183 if (argc > 1 && std::string(argv[1]) == "-h") {
184 std::cerr << getUsageMessage(argv[0]) << '\n';
185 std::exit(EXIT_FAILURE);
186 }
187
188 Options options = parseCommandLine(argc, argv);
189
190 rdl2::SceneContext context;
191
192 // Use proxy mode. We only need the attribute declarations, and the proxy
193 // DSOs are much, much faster to open.
194 context.setProxyModeEnabled(true);
195
196 try {
197 if (!options.dsoPath.empty()) {
198 context.setDsoPath(options.dsoPath);
199 }
200 if (!options.rdlFile.empty()) {
201 rdl2::readSceneFromFile(options.rdlFile, context);
202 if (options.sceneObject.empty()) {
203 printAllSceneObjects(context, options.simple, !options.non_alphabetize);
204 } else {
205 std::cout << getSceneInfoStr(*(context.getSceneObject(options.sceneObject)), options.simple, !options.non_alphabetize);
206 }
207 } else if (!options.sceneClass.empty()) {
208 // Print just the DSO in question.
209 std::cout << getSceneInfoStr(*(context.createSceneClass(options.sceneClass)), options.simple, !options.non_alphabetize);
210 } else {
211 // Print all DSOs in the path.
212 context.loadAllSceneClasses();
213 printAllSceneClasses(context, options.simple, !options.non_alphabetize);
214 }
215 } catch (std::exception& e) {
216 std::cerr << "ERROR: " << e.what() << '\n';
217 std::exit(EXIT_FAILURE);
218 }
219
220 return EXIT_SUCCESS;
221}
