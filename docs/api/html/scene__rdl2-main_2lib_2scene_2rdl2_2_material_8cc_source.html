44Material::declare(SceneClass& sceneClass)
45{
46 auto interface = Parent::declare(sceneClass);
47
48 sExtraAovsKey = sceneClass.declareAttribute<SceneObject *>
49 ("extra_aovs", FLAGS_NONE, INTERFACE_MAP);
50 sceneClass.setMetadata(sExtraAovsKey, SceneClass::sComment, "Bind this attribute to a 'ListMap' that "
51 "contains references to ExtraAovMaps that specify additional outputs that can be "
52 "assigned to a RenderOutput \"light aov\" result");
53
54 sLabel = sceneClass.declareAttribute<String>("label", "");
55 sceneClass.setMetadata(sLabel, "comment", "label used in material and light aovs");
56
57 sPriority = sceneClass.declareAttribute<Int>(
58 "priority", 0, {"priority"});
59
60 sceneClass.setMetadata(sPriority, "comment",
61 "The material's place in an order of precedence for "
62 "overlapping dielectrics. A value of 0 means the priority should be ignored. "
63 "Materials with lower numbers (higher priority) \"override\" materials "
64 "with higher numbers (lower priority). To enable automatic removal of "
65 "self-overlapping geometry, a non-zero priority must be set on the "
66 "geometry's material.");
67
68 return interface | INTERFACE_MATERIAL;
69}
