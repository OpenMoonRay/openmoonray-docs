21template <typename M, typename V>
22__attribute__((noinline)) void
23generateRotation(std::vector<M>& data)
24{
25 M mr;
26 float offset = (float)pi * 0.001f;
27 for (int k = 0; k < 1000; ++k) {
28 float angle = 0;
29 for (int i = 0; i < 1000; ++i) {
30 angle += offset;
31 mr.setToRotation(V(1.1,1.2,1.3), angle);
32 data.push_back(mr);
33 }
34 }
35}

37template <typename M>
38__attribute__((noinline)) void
39generateScale(std::vector<M>& data)
40{
41 M ms(1.1,0.0,0.0,
42 0.0,1.1,0.0,
43 0.0,0.0,1.1);
44 data.resize(1000000, ms);
45}

47template <typename V>
48__attribute__((noinline)) void
49generatePoints(std::vector<V>& data)
50{
51 V offset(0.001, 0.001, 0.001);
52 for (int k = 0; k < 1000; ++k) {
53 V p(0.1, 0.2, 0.3);
54 for (int i = 0; i < 1000; ++i) {
55 p += offset;
56 data.push_back(p);
57 }
58 }
59}

61template <typename M, typename V>
62__attribute__((noinline)) void
63compute(const int iterations,
64 const std::vector<M>& rotation,
65 const std::vector<M>& scale,
66 const std::vector<V>& point,
67 V& v1,
68 V& v2)
69{
70 for (int i = 0; i < iterations; ++i) {
71 M mt = rotation[i] * rotation[i] * scale[i];
72 v1 += transformVector(mt, point[i]);
73 v2 += transformNormal(mt.inverse(), point[i]);
74 }
75}
76template <typename M, typename V>
77__attribute__((noinline)) void
78gmathCompute(const int iterations,
79 const std::vector<M>& rotation,
80 const std::vector<M>& scale,
81 const std::vector<V>& point,
82 V& v1,
83 V& v2)
84{
85 for (int i = 0; i < iterations; ++i) {
86 M mt = rotation[i] * rotation[i] * scale[i];
87 v1 += mt.transform(point[i]);
88 v2 += mt.inverse().pretransform(point[i]);
89 }
90}

501// Generate a uniformly disributed random rotation matrix.
502// For details see:
503// "Uniform Random Rotations" - Ken Shoemake. Graphics Gems III
504Mat3f randomRotationMatrix()
505{
506 // Uniform x0 in [0,1], x1 and x2 in [0,1)
507 float x0 = random<float>();
508 float x1 = random<float>();
509 float x2 = random<float>();
510 if (x1 == 1.0f) x1 = random<float>();
511 if (x2 == 1.0f) x1 = random<float>();
512
513 float r1 = math::sqrt(1.0f - x0);
514 float r2 = math::sqrt(x0);
515 float t1 = (2.0f * 3.14159265f) * x1;
516 float t2 = (2.0f * 3.14159265f) * x2;
517
518 float s1, c1, s2, c2;
519 sincos(t1, &s1, &c1);
520 sincos(t2, &s2, &c2);
521
522 Quaternion3f q(s1*r1, c1*r1, s2*r2, c2*r2);
523 return Mat3f(q);
524}

526// Generate a random scale matrix with eigenvalues uniformly distributed in [sMin, sMax]
527// and randomly oriented eigenvectors
528Mat3f randomScaleMatrix(float sMin, float sMax)
529{
530 float sx = sMin + random<float>() * (sMax - sMin);
531 float sy = sMin + random<float>() * (sMax - sMin);
532 float sz = sMin + random<float>() * (sMax - sMin);
533
534 Mat3f S;
535 S.setToScale(Vec3f(sx, sy, sz));
536 Mat3f R = randomRotationMatrix();
537 return R * S * R.transposed();
538}
