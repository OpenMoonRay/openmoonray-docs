18void
19validateXformAttrs(const moonray::geom::InstanceMethod instanceMethod,
20 const scene_rdl2::rdl2::Vec4fVector& orientations,
21 const scene_rdl2::rdl2::Vec3fVector& scales,
22 const scene_rdl2::rdl2::Geometry& rdlGeometry,
23 const size_t xformsCount,
24 bool& applyOrientation,
25 bool& applyScale)
26{
27 if (instanceMethod == moonray::geom::InstanceMethod::XFORM_ATTRIBUTES) {
28 // validate orientations
29 if (orientations.size() == xformsCount) {
30 applyOrientation = true;
31 } else if (orientations.empty()) {
32 applyOrientation = false;
33 } else {
34 rdlGeometry.warn("orientations count(", orientations.size(),
35 ") is not equal to positions count(", xformsCount,
36 " ). Skip applying orientations");
37 applyOrientation = false;
38 }
39
40 // validate scales
41 if (scales.size() == xformsCount) {
42 applyScale = true;
43 } else if (scales.empty()) {
44 applyScale = false;
45 } else {
46 rdlGeometry.warn("scales count(", scales.size(),
47 ") is not equal to positions count(", xformsCount,
48 "). Skip applying scales");
49 applyScale = false;
50 }
51 } else { // InstanceMethod::XFORMS
52 applyOrientation = false;
53 applyScale = false;
54 }
55}

57bool
58isValidXform(const moonray::shading::XformSamples& xforms)
59{
60 for (const auto& xform : xforms) {
61 if (!scene_rdl2::math::isFinite(xform.row0()) ||
62 !scene_rdl2::math::isFinite(xform.row1()) ||
63 !scene_rdl2::math::isFinite(xform.row2()) ||
64 !scene_rdl2::math::isFinite(xform.row3())) {
65 return false;
66 }
67 }
68 return true;
69}

77bool
78getReferenceData(const scene_rdl2::rdl2::Geometry& rdlGeometry,
79 const scene_rdl2::rdl2::SceneObjectVector& references,
80 std::vector<std::shared_ptr<SharedPrimitive>>& refPrimitiveGroups,
81 std::vector<moonray::shading::XformSamples>& refXforms,
82 std::vector<float>& refShadowRayEpsilons)
83{
84 if (references.empty()) {
85 rdlGeometry.error("Did not find any references geometry. "
86 "Please make sure the \"referencess\" field contains "
87 "at least one source references geometry");
88 return false;
89 }
90
91 for (size_t i = 0; i < references.size(); ++i) {
92 if (references[i]->isA<scene_rdl2::rdl2::Geometry>()) {
93 const scene_rdl2::rdl2::Geometry* g = references[i]->asA<scene_rdl2::rdl2::Geometry>();
94 refShadowRayEpsilons.push_back(g->getShadowRayEpsilon());
95 refPrimitiveGroups[i] = g->getProcedural()->getReference();
96
97 scene_rdl2::math::Mat4f l2r0 =
98 scene_rdl2::math::toFloat(g->get(scene_rdl2::rdl2::Node::sNodeXformKey,
99 scene_rdl2::rdl2::TIMESTEP_BEGIN));
100 refXforms[i].push_back(scene_rdl2::math::Xform3f(l2r0[0][0], l2r0[0][1], l2r0[0][2],
101 l2r0[1][0], l2r0[1][1], l2r0[1][2],
102 l2r0[2][0], l2r0[2][1], l2r0[2][2],
103 l2r0[3][0], l2r0[3][1], l2r0[3][2]));
104
105 scene_rdl2::math::Mat4f l2r1 =
106 scene_rdl2::math::toFloat(g->get(scene_rdl2::rdl2::Node::sNodeXformKey,
107 scene_rdl2::rdl2::TIMESTEP_END));
108 refXforms[i].push_back(scene_rdl2::math::Xform3f(l2r1[0][0], l2r1[0][1], l2r1[0][2],
109 l2r1[1][0], l2r1[1][1], l2r1[1][2],
110 l2r1[2][0], l2r1[2][1], l2r1[2][2],
111 l2r1[3][0], l2r1[3][1], l2r1[3][2]));
112
113 }
114 }
115
116 return true;
117}

128void
129InstanceProceduralLeaf::instanceWithXforms(const GenerateContext& generateContext,
130 const shading::XformSamples& parent2render,
131 const scene_rdl2::math::Mat4f& nodeXform,
132 const std::vector<std::shared_ptr<SharedPrimitive>>& ref,
133 const std::vector<moonray::shading::XformSamples>& refXforms,
134 const std::vector<float>& shadowRayEpsilons,
135 const InstanceMethod instanceMethod,
136 bool useRefXforms,
137 bool useRefAttrs,
138 const int instanceLevel,
139 const scene_rdl2::rdl2::Vec3fVector& velocities,
140 const float evaluationFrame,
141 const scene_rdl2::rdl2::IntVector& indices,
142 const scene_rdl2::rdl2::IntVector& disableIndices,
143 const scene_rdl2::rdl2::SceneObjectVector& attributes,
144 const scene_rdl2::rdl2::Vec3fVector& positions,
145 const scene_rdl2::rdl2::Vec4fVector& orientations,
146 const scene_rdl2::rdl2::Vec3fVector& scales,
147 const scene_rdl2::rdl2::Mat4dVector& xforms)
148 {
149 const scene_rdl2::rdl2::Geometry* rdlGeometry = generateContext.getRdlGeometry();
150
151 // how many instances are we going to create ?
152 size_t xformsCount;
153 if (instanceMethod == InstanceMethod::XFORM_ATTRIBUTES) {
154 xformsCount = positions.size();
155 } else { // InstanceMethod::XFORMS
156 xformsCount = xforms.size();
157 }
158
159 // motion blur
160 bool applyVelocity;
161 bool hasValidVelocity = velocities.size() == xformsCount;
162 if (generateContext.isMotionBlurOn() && !velocities.empty()) {
163 if (!hasValidVelocity) {
164 rdlGeometry->warn("velocities count(", velocities.size(),
165 ") is not equal to positions count(", xformsCount,
166 " ). Skip applying velocities");
167 applyVelocity = false;
168 } else {
169 applyVelocity = true;
170 }
171 } else {
172 applyVelocity = false;
173 }
174 float dt0 = 0.0f;
175 float dt1 = 0.0f;
176 if (applyVelocity) {
177 const scene_rdl2::rdl2::SceneVariables& vars =
178 rdlGeometry->getSceneClass().getSceneContext()->getSceneVariables();
179 float fps = std::max(vars.get(scene_rdl2::rdl2::SceneVariables::sFpsKey), 1.0f);
180 const auto& motionSteps = generateContext.getMotionSteps();
181 // TODO only support at most two time samples now for motion blur
182 MNRY_ASSERT(motionSteps.size() == 2);
183
184 // BAND-AID fix to evaluate from a different evaluation frame (relative)
185 // defaults to 0 (current frame)
186 dt0 = (motionSteps[0] - evaluationFrame) / fps;
187 dt1 = (motionSteps[1] - evaluationFrame) / fps;
188 }
189
190 // indexing
191 bool applyIndex;
192 bool groupOnly = false;
193 if (indices.size() == xformsCount) {
194 applyIndex = true;
195 } else if (indices.empty()) {
196 applyIndex = false;
197 } else if (xformsCount == 0) {
198 // If there are no xforms then treat as a group node
199 // with one instance per reference with no transforms
200 applyIndex = true;
201 groupOnly = true;
202 xformsCount = indices.size();
203 } else {
204 rdlGeometry->warn("refIndices count(", indices.size(),
205 ") is not equal to xforms/positions count(", xformsCount,
206 "). Skip applying refIndices");
207 applyIndex = false;
208 }
209
210 // disabled indices
211 std::unordered_set<int> disableIndicesSet;
212 for (auto &i : disableIndices) {
213 if (i < (int) xformsCount) disableIndicesSet.insert(i);
214 }
215
216 // primitive attributes
217 std::unordered_map<shading::AttributeKey, const void*, shading::AttributeKeyHash> attrMap;
218
219 bool addInstanceTransformAttribute = false;
220 shading::TypedAttributeKey<scene_rdl2::math::Mat4f> instanceLevelKey =
221 sInstanceLevelKeys[instanceLevel];
222
223 bool addInstanceObjectTransformAttribute = false;
224
225 for (const auto& key : generateContext.getRequestedAttributes()) {
226 if (key == instanceLevelKey) {
227 addInstanceTransformAttribute = true;
228 continue;
229 } else if (key == moonray::shading::StandardAttributes::sInstanceObjectTransform) {
230 addInstanceObjectTransformAttribute = true;
231 continue;
232 }
233
234 for (const auto& attr : attributes) {
235 if (!attr->isA<scene_rdl2::rdl2::UserData>()) {
236 continue;
237 }
238 if (attrMap.find(key) != attrMap.end()) {
239 continue;
240 }
241 const scene_rdl2::rdl2::UserData* data = attr->asA<scene_rdl2::rdl2::UserData>();
242 scene_rdl2::rdl2::AttributeType type = key.getType();
243 const std::string& name = key.getName();
244 if (type == scene_rdl2::rdl2::TYPE_BOOL && data->hasBoolData() &&
245 name == data->getBoolKey()) {
246 if (data->getBoolValues().size() == xformsCount) {
247 attrMap[key] = (const void*)(&data->getBoolValues());
248 }
249 } else if (type == scene_rdl2::rdl2::TYPE_INT && data->hasIntData() &&
250 name == data->getIntKey()) {
251 if (data->getIntValues().size() == xformsCount) {
252 attrMap[key] = (const void*)(&data->getIntValues());
253 }
254 } else if (type == scene_rdl2::rdl2::TYPE_FLOAT && data->hasFloatData() &&
255 name == data->getFloatKey()) {
256 if (data->getFloatValues().size() == xformsCount) {
257 attrMap[key] = (const void*)(&data->getFloatValues());
258 }
259 } else if (type == scene_rdl2::rdl2::TYPE_STRING && data->hasStringData() &&
260 name == data->getStringKey()) {
261 if (data->getStringValues().size() == xformsCount) {
262 attrMap[key] = (const void*)(&data->getStringValues());
263 }
264 } else if (type == scene_rdl2::rdl2::TYPE_RGB && data->hasColorData() &&
265 name == data->getColorKey()) {
266 if (data->getColorValues().size() == xformsCount) {
267 attrMap[key] = (const void*)(&data->getColorValues());
268 }
269 } else if (type == scene_rdl2::rdl2::TYPE_VEC2F && data->hasVec2fData() &&
270 name == data->getVec2fKey()) {
271 if (data->getVec2fValues().size() == xformsCount) {
272 attrMap[key] = (const void*)(&data->getVec2fValues());
273 }
274 } else if (type == scene_rdl2::rdl2::TYPE_VEC3F && data->hasVec3fData() &&
275 name == data->getVec3fKey()) {
276 if (data->getVec3fValues().size() == xformsCount) {
277 attrMap[key] = (const void*)(&data->getVec3fValues());
278 }
279 } else if (type == scene_rdl2::rdl2::TYPE_MAT4F && data->hasMat4fData() &&
280 name == data->getMat4fKey()) {
281 if (data->getMat4fValues().size() == xformsCount) {
282 attrMap[key] = (const void*)(&data->getMat4fValues());
283 }
284 }
285 }
286 attrMap.insert({key, nullptr}); // Will only succeed if key does not already exist.
287 }
288
289 bool addVelocityAttribute = false;
290 if (generateContext.requestAttribute(shading::StandardAttributes::sVelocity) &&
291 hasValidVelocity) {
292 addVelocityAttribute = true;
293 }
294
295 bool applyOrientation, applyScale;
296 if (!groupOnly) {
297 validateXformAttrs(instanceMethod,
298 orientations,
299 scales,
300 *rdlGeometry,
301 xformsCount,
302 applyOrientation,
303 applyScale);
304 }
305
306 size_t badXformCount = 0;
307 int maxIndex = ref.size() - 1;
308
309 // disableIndicesSet contains no duplicates or elements >= xformsCount
310 reservePrimitive(xformsCount - disableIndicesSet.size());
311 // combine above info to XformSamples
312 for (size_t i = 0; i < xformsCount; ++i) {
313 if (!disableIndicesSet.empty() && disableIndicesSet.find(i) != disableIndicesSet.end()) {
314 continue;
315 }
316
317 shading::XformSamples xform;
318
319 if (groupOnly) {
320 xform.emplace_back(scene_rdl2::math::Mat3f(scene_rdl2::math::one), scene_rdl2::math::Vec3f(scene_rdl2::math::zero));
321 } else {
322 if (instanceMethod == InstanceMethod::XFORM_ATTRIBUTES) {
323 Vec3f scale = applyScale ? scales[i] : Vec3f(scene_rdl2::math::one);
324 scene_rdl2::math::Quaternion3f orient(scene_rdl2::math::one);
325 if (applyOrientation && orientations[i].lengthSqr() > 0.0f) {
326 orient.i = orientations[i][0];
327 orient.j = orientations[i][1];
328 orient.k = orientations[i][2];
329 orient.r = orientations[i][3];
330 orient = normalize(orient);
331 }
332 scene_rdl2::math::Mat3f scaleRotate(
333 scene_rdl2::math::Mat3f::scale(scale) * scene_rdl2::math::Mat3f(orient));
334 if (applyVelocity) {
335 xform.emplace_back(
336 scaleRotate, positions[i] + velocities[i] * dt0);
337 xform.emplace_back(
338 scaleRotate, positions[i] + velocities[i] * dt1);
339 } else {
340 xform.emplace_back(scaleRotate, positions[i]);
341 }
342 } else if (instanceMethod == InstanceMethod::XFORMS) {
343 if (applyVelocity) {
344 const auto& mat4 = xforms[i];
345
346 scene_rdl2::math::Xform3f xf0 = scene_rdl2::math::xform<scene_rdl2::math::Xform3f>(mat4);
347 scene_rdl2::math::Xform3f xf1 = scene_rdl2::math::xform<scene_rdl2::math::Xform3f>(mat4);
348
349 xf0.p += velocities[i] * dt0;
350 xf1.p += velocities[i] * dt1;
351
352 xform.push_back(xf0);
353 xform.push_back(xf1);
354 } else {
355 const auto& mat4 = xforms[i];
356 xform.push_back(scene_rdl2::math::xform<scene_rdl2::math::Xform3f>(mat4));
357 }
358 } else {
359 rdlGeometry->error("Unknown instancing method.");
360 return;
361 }
362 }
363
364 int index = applyIndex ? indices[i] : 0;
365 if (index > maxIndex) {
366 index = 0;
367 }
368 if (!ref[index]) {
369 continue;
370 }
371
372 shading::PrimitiveAttributeTable primitiveAttributeTable;
373
374 // handle a special case that user requests velocity,
375 // which is stored in its own separate location "velocities"
376 // instead of "primitive attributes" field
377 if (addVelocityAttribute) {
378 primitiveAttributeTable.addAttribute(
379 shading::StandardAttributes::sVelocity, shading::RATE_CONSTANT,
380 {velocities[i]});
381 }
382 if (addInstanceTransformAttribute) {
383 scene_rdl2::math::Mat4f instanceTransform(xform[0]);
384 if (useRefXforms) {
385 instanceTransform = static_cast<scene_rdl2::math::Mat4f>(refXforms[index][0]) * instanceTransform;
386 }
387
388 primitiveAttributeTable.addAttribute(
389 instanceLevelKey,
390 shading::RATE_CONSTANT,
391 { instanceTransform });
392 }
393 if (addInstanceObjectTransformAttribute) {
394 primitiveAttributeTable.addAttribute(
395 moonray::shading::StandardAttributes::sInstanceObjectTransform,
396 shading::RATE_CONSTANT,
397 { nodeXform });
398 }
399 if (useRefAttrs) {
400 // Only adding shadow_ray_epsilon for now
401 // MOONRAY-4313 - Propagate common geometry attributes to instanced primitives
402 std::vector<float> shadowRayEpsilon = { shadowRayEpsilons[index] };
403 primitiveAttributeTable.addAttribute(
404 moonray::shading::StandardAttributes::sShadowRayEpsilon,
405 shading::RATE_CONSTANT,
406 std::move(shadowRayEpsilon));
407 }
408 for (const auto& kv : attrMap) {
409 const auto& key = kv.first;
410 const void* values = kv.second;
411 if (values == nullptr) {
412 continue;
413 }
414 switch (key.getType()) {
415 case scene_rdl2::rdl2::TYPE_BOOL:
416 primitiveAttributeTable.addAttribute(
417 shading::TypedAttributeKey<bool>(key), shading::RATE_CONSTANT,
418 std::vector<bool>({ (*(const scene_rdl2::rdl2::BoolVector*)values)[i]}) );
419 break;
420 case scene_rdl2::rdl2::TYPE_INT:
421 primitiveAttributeTable.addAttribute(
422 shading::TypedAttributeKey<int>(key), shading::RATE_CONSTANT,
423 std::vector<int>({ (*(const scene_rdl2::rdl2::IntVector*)values)[i]}) );
424 break;
425 case scene_rdl2::rdl2::TYPE_FLOAT:
426 primitiveAttributeTable.addAttribute(
427 shading::TypedAttributeKey<float>(key), shading::RATE_CONSTANT,
428 std::vector<float>({ (*(const scene_rdl2::rdl2::FloatVector*)values)[i]}) );
429 break;
430 case scene_rdl2::rdl2::TYPE_STRING:
431 primitiveAttributeTable.addAttribute(
432 shading::TypedAttributeKey<std::string>(key), shading::RATE_CONSTANT,
433 {(*(const scene_rdl2::rdl2::StringVector*)values)[i]});
434 break;
435 case scene_rdl2::rdl2::TYPE_RGB:
436 primitiveAttributeTable.addAttribute(
437 shading::TypedAttributeKey<scene_rdl2::math::Color>(key), shading::RATE_CONSTANT,
438 {(*(const scene_rdl2::rdl2::RgbVector*)values)[i]});
439 break;
440 case scene_rdl2::rdl2::TYPE_VEC2F:
441 primitiveAttributeTable.addAttribute(
442 shading::TypedAttributeKey<Vec2f>(key), shading::RATE_CONSTANT,
443 {(*(const scene_rdl2::rdl2::Vec2fVector*)values)[i]});
444 break;
445 case scene_rdl2::rdl2::TYPE_VEC3F:
446 primitiveAttributeTable.addAttribute(
447 shading::TypedAttributeKey<Vec3f>(key), shading::RATE_CONSTANT,
448 {(*(const scene_rdl2::rdl2::Vec3fVector*)values)[i]});
449 break;
450 case scene_rdl2::rdl2::TYPE_MAT4F:
451 primitiveAttributeTable.addAttribute(
452 shading::TypedAttributeKey<scene_rdl2::math::Mat4f>(key), shading::RATE_CONSTANT,
453 {(*(const scene_rdl2::rdl2::Mat4fVector*)values)[i]});
454 break;
455 default:
456 break;
457 }
458 }
459 if (useRefXforms) {
460 for (size_t j = 0; j < xform.size(); ++j) {
461 xform[j] = refXforms[index][j] * xform[j];
462 }
463 }
464 if (isValidXform(xform)) {
465 auto instance = createInstance(xform,
466 ref[index],
467 std::move(primitiveAttributeTable));
468 addPrimitive(std::move(instance),
469 generateContext.getMotionBlurParams(),
470 parent2render);
471 } else {
472 badXformCount++;
473 }
474 }
475 if (badXformCount > 0) {
476 rdlGeometry->warn("Skipped ", badXformCount,
477 " instances which contained invalid transform matrices");
478 }
479 }
