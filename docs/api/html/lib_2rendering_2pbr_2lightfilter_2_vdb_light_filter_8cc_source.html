98void
99VdbLightFilter::update(const LightFilterMap& /*lightFilters*/,
100 const Mat4d& world2Render) {
101 if (!mRdlLightFilter) {
102 return;
103 }
104
105 Mat4d render2World = world2Render.inverse();
106 Mat4d vdb2World0 = mRdlLightFilter->get(sVdbXformKey, 0.f);
107 Mat4d vdb2World1 = mRdlLightFilter->get(sVdbXformKey, 1.f);
108
109 if (!isValidXform(vdb2World0) || !isValidXform(vdb2World1)) {
110 mRdlLightFilter->error("node_xform is invalid");
111 }
112
113 Mat4d world2Vdb0 = vdb2World0.inverse();
114 Mat4d world2Vdb1 = vdb2World1.inverse();
115
116 mVdbR2V[0] = Mat4f(render2World * world2Vdb0);
117 mVdbR2V[1] = Mat4f(render2World * world2Vdb1);
118
119 Mat4d vdb2Render0 = vdb2World0 * world2Render;
120 Mat4d vdb2Render1 = vdb2World1 * world2Render;
121 mVdbPos[0] = vdb2Render0.row3().toVec3();
122 mVdbPos[1] = vdb2Render1.row3().toVec3();
123
124 // update vdb
125 if (mDensitySampler) {
126 delete mDensitySampler;
127 }
128 std::string vdbFilename = mRdlLightFilter->get(sVdbMapKey);
129 std::string densityGridName = mRdlLightFilter->get(sDensityGridNameKey);
130 if(densityGridName.empty()) {
131 densityGridName = "density";
132 }
133
134 if (vdbFilename.empty()) {
135 mRdlLightFilter->error("You must specify a valid .vdb filename in 'vdb_map'");
136 } else {
137 // Construct an uninitialized OpenVdbSampler, then try to initialize it.
138 // If initialization fails for any reason, destroy this useless object
139 // immediately. Two-phase construct-and-init design pattern
140 mDensitySampler = new moonray::shading::OpenVdbSampler();
141
142 std::string errorMsg;
143 if (mDensitySampler->initialize(vdbFilename,
144 densityGridName,
145 nullptr,
146 nullptr,
147 errorMsg)) {
148 } else {
149 delete mDensitySampler;
150 mDensitySampler = nullptr;
151 //this is an info and not warning as it's not necessary to supply a color grid
152 mRdlLightFilter->warn("Unable to load grid ", densityGridName," from file ", vdbFilename, ". Using default Density of 1");
153 }
154 }
155
156 std::vector<float> inDensityRampVec = mRdlLightFilter->get<rdl2::FloatVector>(sDensityRemapInputsKey);
157 std::vector<float> outDensityRampVec = mRdlLightFilter->get<rdl2::FloatVector>(sDensityRemapOutputsKey);
158 std::vector<int> interpolationTypesVec = mRdlLightFilter->get<rdl2::IntVector>(sDensityRemapInterpTypesKey);
159
160 if (inDensityRampVec.size() != outDensityRampVec.size() || inDensityRampVec.size() != interpolationTypesVec.size()) {
161 mRdlLightFilter->error(
162 "Vdb light filter density inputs, remaps and interpolation types are different sizes, using defaults");
163 inDensityRampVec = {0.f, 1.f};
164 outDensityRampVec = {0.f, 1.f};
165 interpolationTypesVec = {ispc::RAMP_INTERPOLATOR_MODE_LINEAR, ispc::RAMP_INTERPOLATOR_MODE_LINEAR};
166 }
167
168 bool rescaleDensity = mRdlLightFilter->get<rdl2::Bool>(sDensityRemapRescaleEnableKey);
169 if(rescaleDensity) {
170 float densityRemapInputMin = mRdlLightFilter->get<rdl2::Float>(sDensityRemapInputMinKey);
171 float densityRemapInputMax = mRdlLightFilter->get<rdl2::Float>(sDensityRemapInputMaxKey);
172 float densityRemapOutputMin = mRdlLightFilter->get<rdl2::Float>(sDensityRemapOutputMinKey);
173 float densityRemapOutputMax = mRdlLightFilter->get<rdl2::Float>(sDensityRemapOutputMaxKey);
174
175 if (densityRemapInputMin >= densityRemapInputMax) {
176 mRdlLightFilter->error(
177 "Vdb light filter density_remap_input_range_min is >= density_remap_input_range_max, using defaults");
178 densityRemapInputMin = 0.f;
179 densityRemapInputMax = 1.f;
180 }
181
182 if (densityRemapOutputMin >= densityRemapOutputMax) {
183 mRdlLightFilter->error(
184 "Vdb light filter density_remap_output_range_min is >= density_remap_output_range_max, using defaults");
185 densityRemapOutputMin = 0.f;
186 densityRemapOutputMax = 1.f;
187 }
188
189 //first find min/max of the density remap inputs
190 float detectedInputMin = inDensityRampVec[0];
191 float detectedInputMax = inDensityRampVec[0];
192 for(int i = 1; i < inDensityRampVec.size(); i++) {
193 detectedInputMin = min(detectedInputMin, inDensityRampVec[i]);
194 detectedInputMax = max(detectedInputMax, inDensityRampVec[i]);
195 }
196
197 //first find min/max of the density remap outputs
198 float detectedRemapMin = outDensityRampVec[0];
199 float detectedRemapMax = outDensityRampVec[0];
200 for(int i = 1; i < outDensityRampVec.size(); i++) {
201 detectedRemapMin = min(detectedRemapMin, outDensityRampVec[i]);
202 detectedRemapMax = max(detectedRemapMax, outDensityRampVec[i]);
203 }
204
205 if(detectedInputMin < 0.f || detectedInputMin > 1.f || detectedInputMax < 0.f || detectedInputMax > 1.f) {
206 mRdlLightFilter->warn(
207 "Vdb light filter: Ramp has inputs outside of [0 1] range. the rescale for those inputs will go "
208 "outside the range of density_remap_input_range_min and density_remap_input_range_max");
209 }
210
211 if(detectedRemapMin < 0.f || detectedRemapMin > 1.f || detectedRemapMax < 0.f || detectedRemapMax > 1.f) {
212 mRdlLightFilter->warn(
213 "Vdb light filter: Ramp has outputs outside of [0 1] range. the rescale for those inputs will go "
214 "outside the range of density_remap_output_range_min and density_remap_output_range_max");
215 }
216
217 auto remapHelper = [](std::vector<float>& values, float min, float max) {
218 //remap them to user provided min/max, any values outside [0-1] will exceed the min/max range
219 for(int i = 0; i < values.size(); i++) {
220 values[i] = values[i] * (max - min) + min;
221 }
222 };
223 remapHelper(inDensityRampVec, densityRemapInputMin, densityRemapInputMax);
224 remapHelper(outDensityRampVec, densityRemapOutputMin, densityRemapOutputMax);
225 }
226
227 mDensityRamp.init(
228 inDensityRampVec.size(),
229 inDensityRampVec.data(),
230 outDensityRampVec.data(),
231 reinterpret_cast<const ispc::RampInterpolatorMode*>(interpolationTypesVec.data()));
232
233 mBlurValue = mRdlLightFilter->get<rdl2::Float>(sBlurValueKey);
234 mBlurType = mRdlLightFilter->get<rdl2::Int>(sBlurType);
235
236 mInvertDensity = mRdlLightFilter->get<rdl2::Bool>(sInvertDensityKey);
237 mVdbInterpolation = mRdlLightFilter->get(sVdbInterpolation);
238 mColorTint = mRdlLightFilter->get<rdl2::Rgb>(sColorTintNameKey);
239}

253Color
254VdbLightFilter::eval(const EvalData& data) const {
255 // We need the vdb space position of the render space point.
256 Vec3f vdbSpaceP0 = transformPoint(mVdbR2V[0], data.shadingPointPosition);
257 Vec3f vdbSpaceP1 = transformPoint(mVdbR2V[1], data.shadingPointPosition);
258 Vec3f vdbSpaceP = lerp(vdbSpaceP0, vdbSpaceP1, data.time);
259
260 Vec3f offset = Vec3f(0.f, 0.f, 0.f);
261 if (mBlurValue > 0.f) {
262 switch (mBlurType) {
263 case VDBLIGHTFILTERBLURTYPE_GAUSSIAN:
264 // Quadratic bspline "gaussian" filter, which is the same filter the regular camera uses.
265 offset.x = quadraticBSplineWarp(data.randVar.r3.x) * mBlurValue;
266 offset.y = quadraticBSplineWarp(data.randVar.r3.y) * mBlurValue;
267 offset.z = quadraticBSplineWarp(data.randVar.r3.z) * mBlurValue;
268 break;
269 case VDBLIGHTFILTERBLURTYPE_SPHERICAL:
270 //get random point on sphere
271 offset = shading::sampleSphereUniform(data.randVar.r3.x, data.randVar.r3.y);
272 //scale it by cuberoot(random) * radius for uniform distribution within sphere
273 offset *= mBlurValue * scene_rdl2::math::pow(data.randVar.r3.z, 1.f/3.f);
274 break;
275 default:
276 MNRY_ASSERT(0 && "unhandled blur Type");
277 }
278 }
279
280 const Vec3f P(vdbSpaceP.x + offset.x, vdbSpaceP.y + offset.y, vdbSpaceP.z + offset.z);
281
282 float densitySample;
283 //sample color and density from vdb
284 VdbLightFilter::sampleVdb(reinterpret_cast<const rdl2::LightFilter*>(this),
285 data.tls->mShadingTls.get(),
286 P.x, P.y, P.z,
287 &densitySample);
288 // Invert the density sample before the remap
289 if (mInvertDensity) {
290 densitySample = 1.f - densitySample;
291 }
292 //remap the density
293 densitySample = mDensityRamp.eval1D(densitySample);
294
295 //clamp density/color to 0 separately in case both are negative, multiply below would make it positive
296 densitySample = scene_rdl2::math::max(densitySample, 0.f);
297
298 // density sample controls how much light should be unaffected. less density means
299 // we shift towards the tinted color
300 Color result = lerp(mColorTint, Color(1.0f), densitySample);
301 return result;
302}

304void
305VdbLightFilter::sampleVdb(const rdl2::LightFilter* lightFilter,
306 moonray::shading::TLState *tls,
307 const float px,
308 const float py,
309 const float pz,
310 float *outDensity) {
311 const VdbLightFilter* me = reinterpret_cast<const VdbLightFilter*>(lightFilter);
312 const Vec3f P(px, py, pz);
313 if (me->mDensitySampler) {
314 *outDensity = me->mDensitySampler->sample(tls, P,
315 static_cast<moonray::shading::OpenVdbSampler::Interpolation>(me->mVdbInterpolation)).r;
316 } else {
317 *outDensity = 1.f;
318 }
319}
