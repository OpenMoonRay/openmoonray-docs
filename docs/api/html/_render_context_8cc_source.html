90static unsigned
91getInitialSeed(std::uint32_t frame,
92 StereoView stereoView,
93 bool lockFrameNoise)
94{
95 if (lockFrameNoise) {
96 return 0u;
97 }
98
99 auto stereoSeed = static_cast<uint32_t>(stereoView);
100 // We seed the sequenceID based on the frame number and stereo view, so that
101 // we propagate different random seeds throughout the render. We can't use
102 // the frame number directly, because this value is incremented throughout
103 // the pipeline. This means that frame zero would use sequence [0, 1, 2, ...
104 // ], and frame 1 would use [1, 2, 3, ... ]. We would still be correlated,
105 // just maybe off a bounce.
106 //
107 // We multiply by a relatively large number that is co-prime to the number
108 // of total possibilities (2^32). It's okay if we overflow: being co-prime
109 // to the implicit modulus value means that we would eventually hit every
110 // value we can store in the type without repeats (until we've hit every
111 // value).
112
113 // If we're running at 30 fps, and we don't want to be near previous seeds
114 // for two minutes, that's 120 * 30 frames. So, we need a number near
115 // 2^32/(120 * 30) =~ 1,193,046. It doesn't matter much, though, since this
116 // value ends up getting hashed anyway.
117 constexpr std::uint32_t prime = 1193047u;
118
119 // We have to add at least one (the stereoSeed ultimately comes from an enum
120 // with 0 as the start), but I want to make this general enough that we can
121 // add further values in the future (e.g. a seed value). They should each
122 // add enough that they don't interfere with one another.
123 stereoSeed += 6037u;
124
125 // Production shots are not usually frame 0, but some of our test shots are.
126 frame += 8429u;
127
128 // It's not so important that these magic values be prime. What is important
129 // is that this ultimate value is relatively prime to the number of possible
130 // values, which should be a power of two. Being a power of two, the prime
131 // decomposition is 2^n, so any odd number is relatively prime to the number
132 // of possible values.
133 const uint32_t multValue = 2u * (prime + stereoSeed) + 1u;
134 return frame * multValue;
135}

140std::unique_ptr<pbr::PixelFilter>
141getPixelFilter(scene_rdl2::rdl2::PixelFilterType type, float width)
142{
143 std::unique_ptr<pbr::PixelFilter> pixelFilter;
144
145 switch (type) {
146 case scene_rdl2::rdl2::PixelFilterType::box:
147 pixelFilter.reset(new pbr::BoxPixelFilter(width));
148 break;
149 case scene_rdl2::rdl2::PixelFilterType::cubicBSpline:
150 pixelFilter.reset(new pbr::CubicBSplinePixelFilter(width));
151 break;
152 case scene_rdl2::rdl2::PixelFilterType::quadraticBSpline:
153 pixelFilter.reset(new pbr::QuadraticBSplinePixelFilter(width));
154 break;
155 default:
156 Logger::warn("Unknown pixel filter type specified");
157 pixelFilter.reset(new pbr::BoxPixelFilter(width));
158 break;
159 }
160
161 return pixelFilter;
162}

294void
295RenderContext::createMeshLightLayer() {
296 // Dummy LightSet
297 scene_rdl2::rdl2::LightSet* dummyLightSet = mSceneContext->
298 createSceneObject("LightSet", "MeshLightLightSet")->asA<scene_rdl2::rdl2::LightSet>();
299
300 mMeshLightLayer->beginUpdate();
301
302 // get all light sets from main layer
303 scene_rdl2::rdl2::Layer::LightSetSet lightSets;
304 mLayer->getAllLightSets(lightSets);
305
306 for (const scene_rdl2::rdl2::LightSet* lightSet : lightSets) {
307 const auto& lights = lightSet->getLights();
308 for (scene_rdl2::rdl2::SceneObject* so : lights) {
309 // get mesh light
310 if (so->getSceneClass().getName() != "MeshLight") {
311 continue;
312 }
313 const scene_rdl2::rdl2::Light* rdlLight = so->asA<scene_rdl2::rdl2::Light>();
314
315 // get geometry
316 scene_rdl2::rdl2::SceneObject* geomSo = rdlLight->get<scene_rdl2::rdl2::SceneObject*>("geometry");
317 if (!geomSo) {
318 continue;
319 }
320 scene_rdl2::rdl2::Geometry* geom = geomSo->asA<scene_rdl2::rdl2::Geometry>();
321
322 // We cannot load in a geometry that already exists in the main scene layer
323 // The reasons include:
324 // 1) A risk that there is a circular dependency between the mesh light and
325 // the reference geometry using the shadow link
326 // 2) Loading and tessellating the same primitive, but a primitive whose
327 // face set assignment ids are different. This would create a conflict
328 // over which face sets to tessellate.
329 // 3) We would need a better way to specify which primitives go into the
330 // EmbreeAccelerator and which do not when they come from the same procedural
331 if (mLayer->contains(geom)) {
332 rdlLight->warn("\"" + geom->getSceneClass().getName() + "(" +
333 geom->getName() + ")\" cannot be referenced in a MeshLight when "
334 "it is in \"" + mLayer->getSceneClass().getName() + "(" +
335 mLayer->getName() + ")\". Please use a different geometry.");
336 continue;
337 }
338
339 if (geom && geom->updateRequired()) {
340 // Material contains "map shader" so that it can grab requested primitive attributes
341 scene_rdl2::rdl2::Material *material = mSceneContext->
342 createSceneObject("DwaBaseMaterial",
343 geom->getName() + "_MeshLightMaterial")->asA<scene_rdl2::rdl2::Material>();
344 scene_rdl2::rdl2::SceneObject* mapShader= rdlLight->get<scene_rdl2::rdl2::SceneObject*>("map_shader");
345 material->beginUpdate();
346 material->setBinding("albedo", mapShader);
347 material->endUpdate();
348
349 // get part names
350 const std::vector<std::string>& partList = rdlLight->get<scene_rdl2::rdl2::StringVector>("parts");
351
352 // assign to layer
353 scene_rdl2::rdl2::LayerAssignment layerAssignment;
354 layerAssignment.mMaterial = material;
355 layerAssignment.mLightSet = dummyLightSet;
356 for (auto part : partList) {
357 mMeshLightLayer->assign(geom, part, layerAssignment);
358 }
359 if (partList.empty()) {
360 mMeshLightLayer->assign(geom, "", layerAssignment);
361 }
362 }
363 }
364 }
365
366 mMeshLightLayer->endUpdate();
367}

369void
370RenderContext::initialize(std::stringstream &initMessages, LoggingConfiguration loggingConfig)
371{
372 // Load the scene into the SceneContext.
373 loadScene(initMessages);
374
375 scene_rdl2::rdl2::SceneVariables& sceneVars = mSceneContext->getSceneVariables();
376
377 const bool infoLogging = sceneVars.get(scene_rdl2::rdl2::SceneVariables::sInfoKey);
378 const bool debugLogging = sceneVars.get(scene_rdl2::rdl2::SceneVariables::sDebugKey);
379 if (debugLogging) {
380 // debug logging includes info logging
381 scene_rdl2::logging::Logger::setDebugLevel();
382 } else if (infoLogging) {
383 // This is subtle: scene_rdl2::logging::Logger::init is called at load time
384 // before we ever hit main(). It captures the arguments passed to main.
385 // Intentionally or not, we use "-info" on the command line for
386 // moonray/moonray_gui to turn on human-readable verbose output. This
387 // is the same flag for which scene_rdl2::logging looks to turn on logging for
388 // the information log level, which is what we use for our
389 // human-readable verbose output. So, by passing "-info" as a
390 // command-line argument, we implicitly turn on the informational log
391 // level. If, however, we turn on the human-readable verbose output
392 // through other means (e.g. the rdl2 file), we have to switch the
393 // logging level ourselves.
394 scene_rdl2::logging::Logger::setInfoLevel();
395 }
396
397 mRenderStats->openInfoStream(infoLogging || debugLogging);
398 mRenderStats->openStatsStream(sceneVars.get(scene_rdl2::rdl2::SceneVariables::sStatsFile));
399
400#ifndef DEBUG
401 if (loggingConfig == LoggingConfiguration::ATHENA_ENABLED) {
402 const auto athenaDebug = sceneVars.get(scene_rdl2::rdl2::SceneVariables::sAthenaDebug);
403 mRenderStats->openAthenaStream(mOptions.getGUID(), athenaDebug);
404 }
405#endif
406
407 mRenderStats->logInfoPrependStringHeader();
408 mRenderStats->logHardwareConfiguration(mOptions, sceneVars);
409 mRenderStats->logInitializationConfiguration(initMessages);
410
411 mRenderStats->logInfoEmptyLine();
412 mRenderStats->logRenderOptions(mOptions);
413
414 // Get the camera and layer objects.
415 try {
416 std::vector<const scene_rdl2::rdl2::Camera*> cameras = mSceneContext->getActiveCameras();
417 initActiveCameras(cameras);
418 } catch (scene_rdl2::except::KeyError& e) {
419 std::stringstream errMsg;
420 errMsg << e.what();
421 throw scene_rdl2::except::RuntimeError(errMsg.str());
422 }
423 try {
424 scene_rdl2::rdl2::SceneObject *layer = sceneVars.getLayer();
425 if (!layer) {
426 throw scene_rdl2::except::RuntimeError("SceneVariables contains no layer.");
427 }
428 mLayer = layer->asA<scene_rdl2::rdl2::Layer>();
429 if (!mLayer) {
430 throw scene_rdl2::except::RuntimeError("Scene object is not a layer.");
431 }
432 sceneVars.beginUpdate();
433 sceneVars.set(scene_rdl2::rdl2::SceneVariables::sLayer, layer);
434 sceneVars.endUpdate();
435
436 // create mesh light layer, which is a dummy layer to load geometries
437 // that belong to mesh lights
438 mMeshLightLayer = mSceneContext->
439 createSceneObject("Layer", "MeshLightLayer")->asA<scene_rdl2::rdl2::Layer>();
440
441 } catch (scene_rdl2::except::KeyError& e) {
442 std::stringstream errMsg;
443 errMsg << e.what();
444 throw scene_rdl2::except::RuntimeError(errMsg.str());
445 }
446
447 mRenderStats->logSceneVariables(sceneVars);
448
449 // Update texture system limits.
450 texture::TextureSampler *sampler = MNRY_VERIFY(texture::getTextureSampler());
451 sampler->setOpenFileLimit(sceneVars.get(scene_rdl2::rdl2::SceneVariables::sTextureFileHandleCount));
452
453 // configure GeometryManager options
454 mGeometryManagerOptions->accelOptions.maxThreads = getNumTBBThreads();
455 mGeometryManagerOptions->accelOptions.verbose = false;
456
457 mGeometryManagerOptions->stats.logString =
458 [stats = mRenderStats.get()](const std::string& str)
459 {
460 stats->logString(str);
461 };
462 mGeometryManagerOptions->stats.logDebugString =
463 [stats = mRenderStats.get()](const std::string& str)
464 {
465 stats->logDebugString(str);
466 };
467 mGeometryManagerOptions->stats.logGeneratingProcedurals =
468 [stats = mRenderStats.get()](const std::string& sceneClass, const std::string& name)
469 {
470 stats->logGeneratingProcedurals(sceneClass, name);
471 };
472 // Initialize GeometryManager with current SceneContext and Layer
473 mGeometryManager.reset(new rt::GeometryManager(mSceneContext.get(),
474 *mGeometryManagerOptions));
475
476 createPbrScene();
477 mHasBeenInit = true;
478}

525void
526RenderContext::updateGeometry(const std::vector<moonray::ObjectData>& updateData)
527{
528 // this will time the function until it goes out of scope.
529 RenderTimer funcTimer(mRenderStats->mRebuildGeometryTime);
530
531 MNRY_ASSERT(mPbrScene, "Must render the scene at least once before updating geometry.");
532 MNRY_ASSERT_REQUIRE(!mRendering, "Cannot update geometry while rendering is in progress.");
533
534 incrementCurfield();
535
536 int currentFrame = getCurrentFrame();
537 // Get the motion blur related parameters
538 geom::MotionBlurParams motionBlurParams = getMotionBlurParams();
539
540 // Get the transform world --> render.
541 const Mat4d& world2render = mPbrScene->getWorld2Render();
542
543 tbb::blocked_range<size_t> range(0, updateData.size());
544 tbb::parallel_for(range, [&](const tbb::blocked_range<size_t> &r) {
545 // Handle each object in the update data.
546 for (size_t i = r.begin(); i < r.end(); ++i ) {
547 const moonray::ObjectData& objectData = updateData[i];
548
549 // Get the geometry object and procedural associated with this update.
550 scene_rdl2::rdl2::Geometry* geometry = nullptr;
551 std::string geometryName;
552 if (objectData.mNodeName.empty()) {
553 geometryName = "/" + objectData.mAssetName + "/geometry/" +
554 objectData.mSubAssetName;
555 } else {
556 geometryName = objectData.mNodeName;
557 }
558
559 try {
560 geometry = mSceneContext->getSceneObject(geometryName)->asA<scene_rdl2::rdl2::Geometry>();
561 // TODO: no way to pass -info to hostd so LOG_INFO doesn't print :-(
562 } catch (const std::exception& e) {
563 Logger::error("Cannot apply geometry update for geometry \""
564 , geometryName , "\" (" , e.what() , ")");
565 continue;
566 }
567
568 // Aggregate all the mesh names and vertex buffers for this object.
569 std::vector<std::string> meshNames;
570 std::vector<const std::vector<float>* > vertexBuffers;
571 std::for_each(objectData.mObjectMeshes.begin(), objectData.mObjectMeshes.end(),
572 [&meshNames, &vertexBuffers](const moonray::ObjectMesh& objectMesh) {
573 meshNames.push_back(objectMesh.mMeshName);
574 vertexBuffers.push_back(&objectMesh.mMeshPositions);
575 });
576
577 // change the status to update if it was none.
578 mGeometryManager->compareAndSwapFlag(rt::ChangeFlag::ALL, rt::ChangeFlag::NONE);
579 mGeometryManager->updateGeometryData(mLayer, geometry, meshNames,
580 vertexBuffers, world2render, currentFrame, motionBlurParams,
581 getNumTBBThreads());
582 mGeometryManager->updateGeometryData(mMeshLightLayer, geometry, meshNames,
583 vertexBuffers, world2render, currentFrame, motionBlurParams,
584 getNumTBBThreads());
585 }
586 });
587
588 mSceneUpdated = true;
589}

591bool
592RenderContext::bakeGeometry(std::vector<std::unique_ptr<geom::BakedMesh>>& bakedMeshes,
593 std::vector<std::unique_ptr<geom::BakedCurves>>& bakedCurves)
594{
595 // The RenderOutputDriver is needed to build the primitive attribute table
596 // because some of the attrs depend on whether there are certain render outputs.
597 try {
598 mRenderOutputDriver.reset(new RenderOutputDriver(this));
599 }
600 catch(const std::exception &e){
601 Logger::error(e.what());
602 return false;
603 }
604
605 resetShaderStatsAndLogs(); // initialize EventLog to allocate memory for messages
606
607 mSceneContext->applyUpdates(mLayer);
608 mSceneContext->applyUpdates(mMeshLightLayer);
609 // We need to create a layer after applyUpdates is called
610 createMeshLightLayer();
611 {
612 scene_rdl2::rdl2::Layer::RootShaderSet rootShaders;
613 mLayer->getAllRootShaders(rootShaders);
614 buildPrimitiveAttributeTables(rootShaders);
615
616 scene_rdl2::rdl2::Layer::RootShaderSet meshLightRootShaders;
617 mMeshLightLayer->getAllRootShaders(meshLightRootShaders);
618 buildPrimitiveAttributeTables(meshLightRootShaders);
619 }
620
621 MNRY_ASSERT(mPbrScene, "Must have loaded PBR library before loading procedurals.");
622
623 int currentFrame = getCurrentFrame();
624 // Get the motion blur related parameters
625 geom::MotionBlurParams motionBlurParams = getMotionBlurParams(true); // bake=true
626
627 // Get the transform world --> render
628 const Mat4d& world2render = mPbrScene->getWorld2Render();
629
630 // Cameras may require per geometry primitive attributes (e.g. the BakeCamera)
631 shading::PerGeometryAttributeKeySet perGeometryAttributes;
632 for (size_t i = 0; i < mPbrScene->getCameraCount(); i++) {
633 mPbrScene->getCamera(i)->getRequiredPrimAttributes(perGeometryAttributes);
634 }
635
636 mGeometryManager->loadGeometries(mLayer, rt::ChangeFlag::ALL, world2render, currentFrame, motionBlurParams,
637 getNumTBBThreads(), perGeometryAttributes);
638 mGeometryManager->loadGeometries(mMeshLightLayer, rt::ChangeFlag::ALL, world2render, currentFrame, motionBlurParams,
639 getNumTBBThreads(), perGeometryAttributes);
640
641 // Get the camera frustums and render to camera matrices for times points 0 and 1
642 // TODO: generalize for multi-segment motion blur. Currently we only have 2
643 // motion samples and assume the ray time points are 0 and 1. In multi-segment
644 // motion blur there will be multiple motion samples with a ray time range
645 // of [0,1].
646 std::vector<mcrt_common::Frustum> frustums;
647 for (size_t i = 0; i < mPbrScene->getCameraCount(); i++) {
648 const pbr::Camera *camera = mPbrScene->getCamera(i);
649 if (camera->hasFrustum()) {
650 frustums.push_back(mcrt_common::Frustum());
651 camera->computeFrustum(&frustums.back(), 0, true); // frustum at shutter open
652 frustums.push_back(mcrt_common::Frustum());
653 camera->computeFrustum(&frustums.back(), 1, true); // frustum at shutter close
654 }
655 }
656
657 mGeometryManager->bakeGeometry(mLayer,
658 motionBlurParams,
659 frustums,
660 world2render,
661 bakedMeshes,
662 bakedCurves);
663 return true;
664}

711RenderContext::RP_RESULT
712RenderContext::startFrame()
713{
714 mRenderPrepRun = true;
715
716 CheckpointSigIntHandler::disable(); // We don't need checkpointSigIntHandler for renderPrep stage
717
718 scene_rdl2::rec_time::RecTime recTimeWhole;
719 recTimeWhole.start();
720
721 mResumeHistoryMetaData->setFrameStartTime(); // record frame start timing for resume history
722 mRenderPrepExecTracker.init();
723
724 {
725 texture::TextureSampler* sampler = texture::getTextureSampler();
726
727 // Reset texture system stats
728 sampler->resetStats();
729
730 // Update texture system limits.
731 const scene_rdl2::rdl2::SceneVariables& sceneVars = mSceneContext->getSceneVariables();
732 int textureCacheSizeMb = sceneVars.get(scene_rdl2::rdl2::SceneVariables::sTextureCacheSizeMb);
733 if (mOptions.getTextureCacheSizeMb() != 0) {
734 textureCacheSizeMb = mOptions.getTextureCacheSizeMb();
735 }
736 if (static_cast<int>(sampler->getMemoryUsage()) != textureCacheSizeMb) {
737 sampler->setMemoryUsage(static_cast<float>(textureCacheSizeMb));
738 }
739 }
740
741 MNRY_ASSERT_REQUIRE(!mRendering, "Must stop rendering before starting it again.");
742
743 // The frame officially starts now! This time includes the update portion
744 // of the frame also.
745 double frameStartTime = scene_rdl2::util::getSeconds();
746
747 const scene_rdl2::rdl2::SceneVariables &vars = mSceneContext->getSceneVariables();
748
749 { // renderContext console setup for debug purpose
750 int debugConsolePort = vars.get(scene_rdl2::rdl2::SceneVariables::sDebugConsole);
751 if (debugConsolePort >= 0) {
752 if (!RenderContextConsoleDriver::get()) {
753 RenderContextConsoleDriver::init(debugConsolePort);
754 }
755 }
756 if (RenderContextConsoleDriver::get()) {
757 // Very conservative way of updating RenderContext pointer into renderContextConsoleDriver.
758 RenderContextConsoleDriver::get()->setRenderContext(this);
759 }
760 }
761
762 // Determine if to render this frame in scalar, vectorized, or xpu mode.
763 ExecutionMode executionMode = ExecutionMode::VECTORIZED;
764 ExecutionMode desiredExecutionMode = mOptions.getDesiredExecutionMode();
765 std::string executionModeString;
766 std::string missingVecFeatures;
767 const bool vecSupport = canRunVectorized(missingVecFeatures);
768
769 switch (desiredExecutionMode) {
770
771 case ExecutionMode::SCALAR:
772 executionModeString = "Executing a scalar render since execution mode was set to scalar.";
773 executionMode = ExecutionMode::SCALAR;
774 break;
775
776 case ExecutionMode::VECTORIZED:
777 executionModeString = "Executing a vectorized render since execution mode was set to vector.";
778 if (!vecSupport) {
779 executionModeString += " The following features will be missing: ";
780 executionModeString += missingVecFeatures;
781 executionModeString += ".";
782 }
783 executionMode = ExecutionMode::VECTORIZED;
784 break;
785
786 case ExecutionMode::XPU:
787 // XPU mode inherits the same limitations of vector mode
788 if (!vecSupport) {
789 executionModeString = "Executing a scalar render since execution mode was set to xpu.";
790 executionModeString += " The following features are missing xpu/vector mode support: ";
791 executionModeString += missingVecFeatures;
792 executionModeString += ".";
793 executionMode = ExecutionMode::SCALAR;
794 } else {
795 executionModeString = "Executing an xpu render since execution mode was set to xpu.";
796 executionMode = ExecutionMode::XPU;
797 // XPU may still fall back to vector mode later on if it fails to start up, but this is fine
798 // because we know that it's safe to run in vector mode.
799 }
800 break;
801
802 case ExecutionMode::AUTO:
803 default:
804 if (!vecSupport) {
805 executionModeString = "Executing a scalar render since execution mode was set to auto.";
806 executionModeString += " The following features are missing vector mode support: ";
807 executionModeString += missingVecFeatures;
808 executionModeString += ".";
809 executionMode = ExecutionMode::SCALAR;
810 } else {
811 executionModeString = "Executing a vectorized render since execution mode was set to auto.";
812 executionMode = ExecutionMode::VECTORIZED;
813 }
814 }
815
816 // Log information as to whether we're executing in scalar, vectorized, or xpu mode
817 // and the reason why.
818 mRenderStats->logExecModeConfiguration(executionMode);
819 Logger::info(executionModeString);
820
821 // Make sure everything is ready to render.
822 scene_rdl2::rec_time::RecTime recTime;
823 recTime.start();
824 RP_RESULT execResult = renderPrep(executionMode); // may throw
825 mDriver->pushRenderPrepTime(recTime.end()); // statistical info update for debug
826
827#if defined(USE_PARTITIONED_PIXEL) || defined(USE_PARTITIONED_LENS) || defined(USE_PARTITIONED_TIME)
828 // Ensure frame to frame noise is locked for the real-time case.
829 if (getRenderMode() != RenderMode::REALTIME) {
830 const auto temporalKey = scene_rdl2::rdl2::SceneVariables::sLockFrameNoise;
831 const auto frameNum = getCurrentFrame();
832
833#if defined(USE_PARTITIONED_PIXEL)
834 if (!mPbrScene->getRdlSceneContext()->getSceneVariables().get(temporalKey)) {
835 if (mOptions.getApplicationMode() != ApplicationMode::MOTIONCAPTURE) {
836 pbr::kPixelPartition.rotate(frameNum);
837 }
838 }
839#endif
840
841#if defined(USE_PARTITIONED_LENS)
842 if (!mPbrScene->getRdlSceneContext()->getSceneVariables().get(temporalKey)) {
843 if (mOptions.getApplicationMode() != ApplicationMode::MOTIONCAPTURE) {
844 pbr::kLensPartition.rotate(frameNum);
845 }
846 }
847#endif
848
849#if defined(USE_PARTITIONED_TIME)
850 if (!mPbrScene->getRdlSceneContext()->getSceneVariables().get(temporalKey)) {
851 if (mOptions.getApplicationMode() != ApplicationMode::MOTIONCAPTURE) {
852 pbr::kTimePartition.rotate(frameNum);
853 }
854 }
855#endif
856 }
857#endif
858
859 // Update pixel filter if needed.
860 scene_rdl2::rdl2::PixelFilterType pixelFilterType = (scene_rdl2::rdl2::PixelFilterType)vars.get(scene_rdl2::rdl2::SceneVariables::sPixelFilterType);
861 float pixelFilterWidth = vars.get(scene_rdl2::rdl2::SceneVariables::sPixelFilterWidth);
862 if (mCachedPixelFilterType != pixelFilterType ||
863 mCachedPixelFilterWidth != pixelFilterWidth) {
864 mPixelFilter = getPixelFilter(pixelFilterType, pixelFilterWidth);
865 mCachedPixelFilterType = pixelFilterType;
866 mCachedPixelFilterWidth = pixelFilterWidth;
867 }
868
869 // Loop through the cameras to collect the
870 // pixel sample maps and get the max pixel sample values.
871 for (size_t i = 0; i < mCameras.size(); ++i) { // loop through cameras
872 const std::string pixelSampleMapName = mCameras[i]->get(scene_rdl2::rdl2::Camera::sPixelSampleMap);
873 scene_rdl2::fb_util::PixelBuffer<float> *pixelSampleMap = mPixelSampleMaps->at(i);
874 std::string &cachedPixelSampleMapName = mCachedPixelSampleMapNames[i];
875
876 // update pixel sample map
877 if (pixelSampleMapName.empty()) {
878 if (!pixelSampleMap) {
879 continue;
880 }
881
882 // delete pixel sample map
883 pixelSampleMap->cleanUp();
884 delete pixelSampleMap;
885 mPixelSampleMaps->at(i) = nullptr;
886 cachedPixelSampleMapName.clear();
887 mMaxPixelSampleValues[i] = 1.0;
888 } else {
889 // create pixel buffer
890 if (!pixelSampleMap) {
891 pixelSampleMap = new scene_rdl2::fb_util::PixelBuffer<float>();
892 mPixelSampleMaps->at(i) = pixelSampleMap;
893 }
894
895 // change pixel sample map
896 unsigned width = vars.getRezedWidth();
897 unsigned height = vars.getRezedHeight();
898 if (pixelSampleMapName != cachedPixelSampleMapName ||
899 width != pixelSampleMap->getWidth() ||
900 height != pixelSampleMap->getHeight()) {
901 pixelSampleMap->init(width, height);
902 rndr::readPixelBuffer(*pixelSampleMap, pixelSampleMapName,
903 width, height);
904 cachedPixelSampleMapName = pixelSampleMapName;
905
906 // find the max value in map
907 unsigned numPixels = pixelSampleMap->getArea();
908 auto data = pixelSampleMap->getData();
909 mMaxPixelSampleValues[i] = *std::max_element(data, data + numPixels);
910 }
911 }
912 }
913
914 // get deep id channel names
915 mDeepIDChannelNames.reset(new std::vector<std::string>());
916 *mDeepIDChannelNames = vars.get(scene_rdl2::rdl2::SceneVariables::sDeepIDAttributeNames);
917
918 // Check if the user requested to record debug rays for this frame.
919 if (!vars.get(scene_rdl2::rdl2::SceneVariables::sDebugRaysFile).empty()) {
920 if (mDriver->getDebugRayState() == RenderDriver::READY) {
921 const Mat4d& render2world = mPbrScene->getRender2World();
922 Viewport vp = scene_rdl2::math::convertToClosedViewport(vars.getRezedRegionWindow());
923 BBox2i bboxVp(vp.min(), vp.max());
924 // Will accept a double to float precision loss for debug rays
925 pbr::DebugRayRecorder::enableRecording(bboxVp, toFloat(render2world));
926 mDriver->switchDebugRayState(RenderDriver::READY, RenderDriver::REQUEST_RECORD);
927 }
928 }
929
930 // Don't spam the logs if in real-time mode.
931 // mTotalRenderPrepTime was set inside of the prep function call
932 if (getRenderMode() != RenderMode::REALTIME &&
933 mOptions.getApplicationMode() != ApplicationMode::MOTIONCAPTURE) {
934 if (mLogTime) { // only print this if previous pass printing timing
935 mRenderStats->logInfoEmptyLine();
936 mRenderStats->updateAndLogRenderPrepStats();
937 }
938
939 // Print dso usage
940 mRenderStats->logDsoUsage(mSceneContext->getDsoCounts());
941
942 mRenderStats->startRenderStats();
943 }
944
945 // Condition scene variables and other state for this frame.
946 FrameState frameState;
947 buildFrameState(&frameState, frameStartTime, executionMode);
948
949 // Record some info for resume history from frameState
950 mResumeHistoryMetaData->setNumOfThreads(frameState.mNumRenderThreads);
951 if (frameState.mSamplingMode == SamplingMode::UNIFORM) {
952 mResumeHistoryMetaData->setUniformSample(frameState.mMinSamplesPerPixel,
953 frameState.mMaxSamplesPerPixel);
954 } else if (frameState.mSamplingMode == SamplingMode::ADAPTIVE) {
955 mResumeHistoryMetaData->setAdaptiveSample(frameState.mMinSamplesPerPixel,
956 frameState.mMaxSamplesPerPixel,
957 frameState.mTargetAdaptiveError);
958 }
959
960 // Construct pixelId scramble table setup.
961 Film::constructPixelFillOrderTable(frameState.mRenderNodeIdx, frameState.mNumRenderNodes);
962
963 // save sampling mode and parameters
964 mCachedSamplingMode = frameState.mSamplingMode;
965 mCachedMinSamplesPerPixel = frameState.mMinSamplesPerPixel;
966 mCachedMaxSamplesPerPixel = frameState.mMaxSamplesPerPixel;
967 mCachedTargetAdaptiveError = frameState.mTargetAdaptiveError;
968
969 // We store maxSamplesPerPixel info into renderOutputDriver for write action before start render
970 mRenderOutputDriver->setFinalMaxSamplesPerPix(frameState.mMaxSamplesPerPixel);
971
972 // add info to ImageWriteDriver
973 {
974 const scene_rdl2::rdl2::SceneVariables &vars = mSceneContext->getSceneVariables();
975 ImageWriteDriver::get()->setTwoStageOutput(frameState.mTwoStageOutput);
976 ImageWriteDriver::get()->setTmpDirectory(vars.getTmpDir());
977 ImageWriteDriver::get()->setMaxBgCache(vars.get(scene_rdl2::rdl2::SceneVariables::sCheckpointMaxBgCache));
978 ImageWriteDriver::get()->setRenderContext(this);
979 }
980
981 // Save some info for resume render related info
982 {
983 const scene_rdl2::rdl2::SceneVariables &vars = getSceneContext().getSceneVariables();
984 mOnResumeScript = vars.get(scene_rdl2::rdl2::SceneVariables::sOnResumeScript);
985 }
986 getResumeHistoryMetaData()->setBgCheckpointWriteMode(frameState.mCheckpointBgWrite);
987
988 // Update frame bounds for ray sorting.
989 pbr::IncoherentRayQueue::setBounds(frameState.mEmbreeAccel->getBounds());
990
991 // Update integrator state.
992 updatePbrState(frameState);
993
994 // In bundled vectorized and xpu mode we lazily allocate queues for any new
995 // Primitives and Materials introduced into the scene since the previous frame.
996 // In scalar mode we only set the id for each material.
997 if (frameState.mExecutionMode == mcrt_common::ExecutionMode::VECTORIZED ||
998 frameState.mExecutionMode == mcrt_common::ExecutionMode::XPU) {
999 unsigned shadeQueueSize = mcrt_common::getTLSInitParams().mShadeQueueSize;
1000 MNRY_ASSERT(shadeQueueSize);
1001 shading::Material::allocShadeQueues(shadeQueueSize, pbr::shadeBundleHandler);
1002 } else {
1003 shading::Material::initMaterialIds();
1004 }
1005
1006 // Setup RenderContext pointer into frame state
1007 frameState.mRenderContext = this;
1008
1009 // Setup the XPU queues in the RenderDriver if we are XPU accelerated.
1010 // isGPUEnabled() may return false even in XPU mode if we failed to create it,
1011 // e.g. unsupported geometry, out of VRAM, etc. In this case we don't have
1012 // any XPU queues.
1013 if (mGeometryManager->isGPUEnabled()) {
1014 mDriver->createXPUQueue();
1015 } else {
1016 mDriver->freeXPUQueue();
1017 }
1018
1019 if (execResult == RP_RESULT::CANCELED) {
1020 mRenderPrepRun = false;
1021 return RP_RESULT::CANCELED;
1022 }
1023
1024 // Invoke the render driver.
1025 mRendering = true;
1026 mDriver->startFrame(frameState);
1027
1028 mRenderPrepRun = false;
1029 mRenderPrepTimingStats->setWholeStartFrame(recTimeWhole.end());
1030
1031 return RP_RESULT::FINISHED;
1032}

1054RenderContext::stopFrame()
1055{
1056 mRenderPrepTimingStats->recTimeStart();
1057
1058 MNRY_ASSERT_REQUIRE(mRendering, "Must start rendering before it can be stopped.");
1059
1060 // Halt the render driver.
1061 mDriver->stopFrame();
1062
1063 mRenderPrepTimingStats->recTime(RenderPrepTimingStats::StopFrameTag::MDRIVER_STOPFRAME);
1064
1065 mPbrScene->postFrame();
1066
1067 mRenderPrepTimingStats->recTime(RenderPrepTimingStats::StopFrameTag::MPBRSCENE_POSTFRAME);
1068
1069 // Accumulate pbr stats data.
1070 mPbrStatistics->mMcrtTime = mDriver->getLastFrameMcrtDuration();
1071 mPbrStatistics->mMcrtUtilization = mDriver->getLastFrameMcrtUtilization();
1072 pbr::forEachTLS([this](pbr::TLState const *tls){ (*mPbrStatistics) += tls->mStatistics; });
1073
1074 // Accumulate geom stats data.
1075 geom::internal::forEachTLS([this](geom::internal::TLState const *tls) {
1076 (*mGeomStatistics) += tls->mStatistics;
1077 });
1078
1079 mRenderPrepTimingStats->recTime(RenderPrepTimingStats::StopFrameTag::ACCUMULATE_PBR_STATS_DATA);
1080
1081 // Collect all per-shader stats into mRenderStats
1082 collectShaderStats();
1083
1084 mRenderPrepTimingStats->recTime(RenderPrepTimingStats::StopFrameTag::COLLECTSHADERSTATS);
1085
1086 if (mRenderStats->getLogInfo() ||
1087 mRenderStats->getLogCsv() ||
1088 mRenderStats->getLogAthena()) {
1089
1090 reportGeometryStatistics();
1091 }
1092
1093 // Any errors that occurred during shading will be reported at this time
1094 reportShadingLogs();
1095
1096 mRenderPrepTimingStats->recTime(RenderPrepTimingStats::StopFrameTag::REPORTSHADINGLOGS);
1097
1098 // Don't spam the logs if in real-time mode, or if interactive changes interrupt
1099 // rendering too quickly (less than 10 seconds). I tried much shorter timeouts
1100 // but it has to be that long to avoid spamming that makes it impossible to see
1101 // useful messages with typical usage (at least by me) of moving the camera,
1102 // looking at the result, and then deciding to move the camera again.
1103 // The decision for this pass is remembered in mLogTime for next pass to stop
1104 // it from printing renderprep timing.
1105 if (getRenderMode() != RenderMode::REALTIME &&
1106 mOptions.getApplicationMode() != ApplicationMode::MOTIONCAPTURE) {
1107 if (isFrameComplete() || mPbrStatistics->mMcrtTime >= 10.0) {
1108 mLogTime = true;
1109 mRenderStats->logInfoEmptyLine();
1110 mRenderStats->logSamplingStats(*mPbrStatistics, *mGeomStatistics);
1111
1112 mRenderStats->logInfoEmptyLine();
1113 mRenderStats->logTexturingStats(*texture::getTextureSampler());
1114
1115 mRenderStats->logRenderingStats(*mPbrStatistics,
1116 static_cast<mcrt_common::ExecutionMode>(mDriver->getFrameState().mExecutionMode),
1117 mSceneContext->getSceneVariables());
1118
1119 mRenderStats->logRenderOutputs(mSceneContext->getAllRenderOutputs());
1120 } else {
1121 mLogTime = false;
1122 }
1123 }
1124
1125 mRenderStats->flush();
1126
1127 // Do debug ray database processing if we've just been recording rays.
1128 if (mDriver->getDebugRayState() == RenderDriver::RECORDING_COMPLETE) {
1129
1130 buildAndSaveRayDatabase();
1131
1132 pbr::forEachTLS([](pbr::TLState *tls) {
1133 tls->mRayRecorder->cleanUp();
1134 });
1135
1136 pbr::DebugRayRecorder::disableRecording();
1137
1138 // Only save the file out for one frame.
1139 scene_rdl2::rdl2::SceneVariables& sceneVars = mSceneContext->getSceneVariables();
1140 scene_rdl2::rdl2::SceneVariables::UpdateGuard guard(&sceneVars);
1141 sceneVars.set(scene_rdl2::rdl2::SceneVariables::sDebugRaysFile, std::string(""));
1142
1143 mDriver->switchDebugRayState(RenderDriver::RECORDING_COMPLETE, RenderDriver::READY);
1144 }
1145
1146 mRenderPrepTimingStats->recTime(RenderPrepTimingStats::StopFrameTag::LOG_FLUSH);
1147
1148 mRendering = false;
1149 mSceneUpdated = false;
1150 // TODO: Why is resetUpdates() not called at the end of
1151 // RenderContext::startFrame ? (at least after the call to mPbrScene->preFrame())
1152 mSceneContext->resetUpdates(mLayer);
1153 mSceneContext->resetUpdates(mMeshLightLayer);
1154 mRenderStats->reset();
1155
1156 mcrt_common::resetAllAccumulators();
1157
1158 mRenderPrepTimingStats->recTime(RenderPrepTimingStats::StopFrameTag::RESET);
1159 mRenderPrepTimingStats->recTimeEnd(RenderPrepTimingStats::StopFrameTag::WHOLE);
1160}

1254void
1255RenderContext::snapshotDelta(scene_rdl2::fb_util::RenderBuffer *renderBuffer,
1256 scene_rdl2::fb_util::FloatBuffer *weightBuffer,
1257 scene_rdl2::fb_util::ActivePixels &activePixels,
1258 bool parallel) const
1259//
1260// Snapshots the contents of the renderBuffer/weightBuffer w/ ActivePixels information
1261// for ProgressiveFrame message related logic. So renderBuffer is not normalized by weight yet.
1262// no resize, no extrapolation and no untiling logic is handled internally.
1263// Just create snapshot data with properly constructed activePixels based on
1264// difference between current and previous renderBuffer and weightBuffer.
1265// renderBuffer/weightBuffer is tiled format and renderBuffer is not normalized by weight.

1563void
1564RenderContext::snapshotRenderOutput(scene_rdl2::fb_util::VariablePixelBuffer *buffer, int roIndx,
1565 const scene_rdl2::fb_util::RenderBuffer *renderBuffer,
1566 const scene_rdl2::fb_util::HeatMapBuffer *heatMapBuffer,
1567 const scene_rdl2::fb_util::FloatBuffer *weightBuffer,
1568 const scene_rdl2::fb_util::RenderBuffer *renderBufferOdd,
1569 bool untile, bool parallel) const
1570{
1571 // First the odd balls
1572 // assume that renderBuffer, and heatMapBuffer, if
1573 // required have already been snapshot (with the proper untile)
1574 // for the appropriate film.
1575 if (mRenderOutputDriver->requiresRenderBuffer(roIndx) ||
1576 mRenderOutputDriver->requiresHeatMap(roIndx) ||
1577 mRenderOutputDriver->requiresWeightBuffer(roIndx) ||
1578 mRenderOutputDriver->requiresRenderBufferOdd(roIndx)) {
1579 mRenderOutputDriver->finishSnapshot(buffer, roIndx,
1580 renderBuffer, heatMapBuffer, weightBuffer, renderBufferOdd,
1581 parallel);
1582 return;
1583 }
1584
1585 if (mRenderOutputDriver->requiresDisplayFilter(roIndx)) {
1586 const int dfIdx = mRenderOutputDriver->getDisplayFilterIndex(roIndx);
1587 snapshotDisplayFilterBuffer(buffer, dfIdx, untile, parallel);
1588 return;
1589 }
1590
1591 const pbr::AovSchema& schema = mRenderOutputDriver->getAovSchema();
1592 const int aovIdx = mRenderOutputDriver->getAovBuffer(roIndx);
1593 if (aovIdx < 0) {
1594 return;
1595 }
1596
1597 if (schema.isVarianceEntry(aovIdx)) {
1598 const int varianceSource = schema.getVarianceToSourceAOV(aovIdx);
1599 MNRY_ASSERT(varianceSource >= 0);
1600 if (mRenderOutputDriver->isVisibilityAov(varianceSource)) {
1601 snapshotVisibilityVarianceBuffer(buffer, varianceSource, untile, parallel);
1602 return;
1603 }
1604 // else, fall through to snapshotAovBuffer
1605 }
1606
1607 if (mRenderOutputDriver->isVisibilityAov(aovIdx)) {
1608 snapshotVisibilityBuffer(buffer, aovIdx, untile, parallel);
1609 return;
1610 }
1611
1612 // All the rest come from the aov buffers
1613 snapshotAovBuffer(buffer, aovIdx, untile, parallel);

1815RenderContext::RP_RESULT
1816RenderContext::renderPrep(ExecutionMode executionMode)
1817{
1818 if (mRenderPrepExecTracker.startRenderPrep() == RenderPrepExecTracker::RESULT::CANCELED) {
1819 return RP_RESULT::CANCELED;
1820 }
1821
1822 mRenderPrepTimingStats->recTimeStart();
1823
1824 startUpdatePhaseOfFrame();
1825
1826 mRenderPrepTimingStats->recTime(RenderPrepTimingStats::RenderPrepTag::START_UPDATE_PHASE_OF_FRAME);
1827
1828 mRenderStats->logInfoEmptyLine();
1829 mRenderStats->startRenderPrep();
1830
1831 mRenderPrepTimingStats->recTime(RenderPrepTimingStats::RenderPrepTag::START_RENDERPREP);
1832
1833 mGeometryManager->resetStatistics();
1834
1835 mRenderPrepTimingStats->recTime(RenderPrepTimingStats::RenderPrepTag::RESET_STATISTICS);
1836
1837 // Setup the render output driver
1838 mRenderOutputDriver.reset(new RenderOutputDriver(this));
1839
1840 rt::ChangeFlag geomChangeFlag =
1841 rt::ChangeFlag::NONE;
1842
1843 mRenderPrepTimingStats->recTime(RenderPrepTimingStats::RenderPrepTag::RESET_RENDER_OUTPUT_DRIVER);
1844
1845 bool loadAllGeometries = false;
1846
1847 if (mFirstFrame) {
1848 mRenderStats->logString(
1849 "---------- Render Prep -----------------------------------");
1850 // Clear stats and logs on each frame
1851 if (mRenderPrepExecTracker.startApplyUpdate() == RenderPrepExecTracker::RESULT::CANCELED) {
1852 return RP_RESULT::CANCELED;
1853 }
1854 resetShaderStatsAndLogs(); // initialize EventLog to allocate memory for messages
1855 mSceneContext->applyUpdates(mLayer);
1856 mSceneContext->applyUpdates(mMeshLightLayer);
1857 // We need to create a layer after applyUpdates is called
1858 createMeshLightLayer();
1859 {
1860 scene_rdl2::rdl2::Layer::RootShaderSet rootShaders;
1861 mLayer->getAllRootShaders(rootShaders);
1862 buildPrimitiveAttributeTables(rootShaders);
1863
1864 scene_rdl2::rdl2::Layer::RootShaderSet meshLightRootShaders;
1865 mMeshLightLayer->getAllRootShaders(meshLightRootShaders);
1866 buildPrimitiveAttributeTables(meshLightRootShaders);
1867 }
1868 geomChangeFlag = rt::ChangeFlag::ALL;
1869 if (mRenderPrepExecTracker.endApplyUpdate() == RenderPrepExecTracker::RESULT::CANCELED) {
1870 return RP_RESULT::CANCELED;
1871 }
1872 mFirstFrame = false;
1873 } else if (mSceneUpdated) {
1874 mSceneContext->applyUpdates(mLayer);
1875 buildPrimitiveAttributeTables(mLayer->getChangedRootShaders());
1876
1877 mSceneContext->applyUpdates(mMeshLightLayer);
1878 buildPrimitiveAttributeTables(mMeshLightLayer->getChangedRootShaders());
1879
1880 const scene_rdl2::rdl2::SceneVariables & sv = mSceneContext->getSceneVariables();
1881
1882 // Check for the need to reload all procedurals for changes in frame
1883 // or motion blur. If the applicationMode is set to MOTIONCAPTURE
1884 // the program will not load geometry from disk each time
1885 // the frame ID changes
1886 bool frameChanged = (sv.hasChanged(scene_rdl2::rdl2::SceneVariables::sFrameKey) &&
1887 mOptions.getApplicationMode() != ApplicationMode::MOTIONCAPTURE);
1888
1889 // Our cameras could have changed.
1890 // A primary camera change requires a geometry rebuild, our definition of
1891 // render space depends on this. So changing the primary camera doesn't seem
1892 // like a particularly good work flow.
1893 const std::vector<const scene_rdl2::rdl2::Camera *> activeCameras = mSceneContext->getActiveCameras();
1894 const bool primaryCameraChanged = sv.hasChanged(scene_rdl2::rdl2::SceneVariables::sCamera);
1895 if (primaryCameraChanged || !(activeCameras == mCameras)) {

2001RenderContext::loadScene(std::stringstream &initMessages)
2002{
2003 RenderTimer timer(mRenderStats->mLoadSceneTime);
2004
2005 // Create an ASCII and binary reader which will both read into the same
2006 // SceneContext (though only one will be actively reading at a time). We
2007 // want to reuse these readers so that they maintain any state between
2008 // reading subsequent files.
2009 scene_rdl2::rdl2::AsciiReader asciiReader(*mSceneContext);
2010 scene_rdl2::rdl2::BinaryReader binaryReader(*mSceneContext);
2011
2012 // Apply Lua globals before loading the scene.
2013 const auto& rdlaGlobals = mOptions.getRdlaGlobals();
2014 for (const auto& global : rdlaGlobals) {
2015 asciiReader.fromString(
2016 scene_rdl2::util::buildString(global.mVar, " = ", global.mExpression),
2017 "@rdla_set");
2018 }
2019
2020 // Load each scene file in order.
2021 const auto& sceneFiles = mOptions.getSceneFiles();
2022 for (const auto& sceneFile : sceneFiles) {
2023 if (!sceneFile.empty()) {
2024 mRenderStats->logLoadingScene(initMessages, sceneFile);
2025 // Grab the file extension and convert it to lower case.
2026 auto ext = scene_rdl2::util::lowerCaseExtension(sceneFile);
2027 if (ext.empty()) {
2028 throw scene_rdl2::except::RuntimeError(scene_rdl2::util::buildString(
2029 "File '", sceneFile, "' has no extension."
2030 " Cannot determine file type."));
2031 }
2032
2033 if (ext == "rdla") {
2034 asciiReader.fromFile(sceneFile);
2035 } else if (ext == "rdlb") {
2036 binaryReader.fromFile(sceneFile);
2037 } else {
2038 throw scene_rdl2::except::RuntimeError(scene_rdl2::util::buildString(
2039 "File '", sceneFile, "' has an unknown extension."
2040 " Cannot determine file type."));
2041 }
2042 }
2043 }
2044
2045 // Now we've loaded our scene. Apply any queued updates
2046 mSceneLoaded = true;
2047
2048 // Apply any queued updates
2049 for (const auto& updateSceneData: mUpdateQueue) {
2050 // Two update paths depending on what was queued
2051 if (updateSceneData.first == FILE_TOKEN) {
2052 updateScene(updateSceneData.second);
2053 } else {
2054 updateScene(updateSceneData.first, updateSceneData.second);
2055 }
2056 }
2057 mUpdateQueue.clear();
2058
2059 // Set up the SceneVariables as a fallback for RenderOptions.
2060 mOptions.setSceneContext(mSceneContext.get());
2061
2062 // Apply any attribute overrides or scene variable overrides specified in
2063 // the RenderOptions.
2064 applyAttributeOverrides(*mSceneContext, *mRenderStats, mOptions, initMessages);
2065
2066 mRenderStats->logLoadingSceneReadDiskIO(initMessages);

2251 // create a bool array, one entry per entry in the aov

2410 const pbr::Scene * scene = getScene();
2411
2412 // Cameras may require per geometry primitive attributes (e.g. the BakeCamera)
2413 shading::PerGeometryAttributeKeySet perGeometryAttributes;
2414 for (size_t i = 0; i < scene->getCameraCount(); i++) {
2415 scene->getCamera(i)->getRequiredPrimAttributes(perGeometryAttributes);
2416 }
2417
2418 mGeometryManager->
2419 setStageIdAndCallBackLoadGeometries(0,
2420 mRenderPrepExecTracker.getRenderPrepStatsCallBack(),
2421 mRenderPrepExecTracker.getRenderPrepCancelCallBack());
2422 if (mRenderPrepExecTracker.startLoadGeom0() == RenderPrepExecTracker::RESULT::CANCELED) {
2423 return RP_RESULT::CANCELED;
2424 }
2425 if (mGeometryManager->loadGeometries(mLayer, flag, world2render, currentFrame, motionBlurParams,
2426 getNumTBBThreads(), perGeometryAttributes) ==
2427 rt::GeometryManager::GM_RESULT::CANCELED) {
2428 return RP_RESULT::CANCELED;
2429 }
2430 if (mRenderPrepExecTracker.endLoadGeom0() == RenderPrepExecTracker::RESULT::CANCELED) {
2431 return RP_RESULT::CANCELED;
2432 }
2433 mGeometryManager->
2434 setStageIdAndCallBackLoadGeometries(1,
2435 mRenderPrepExecTracker.getRenderPrepStatsCallBack(),
2436 mRenderPrepExecTracker.getRenderPrepCancelCallBack());
2437 if (mRenderPrepExecTracker.startLoadGeom1() == RenderPrepExecTracker::RESULT::CANCELED) {
2438 return RP_RESULT::CANCELED;
2439 }
2440 if (mGeometryManager->loadGeometries(mMeshLightLayer, flag, world2render, currentFrame, motionBlurParams,
2441 getNumTBBThreads(), perGeometryAttributes) ==
2442 rt::GeometryManager::GM_RESULT::CANCELED) {
2443 return RP_RESULT::CANCELED;
2444 }
2445 if (mRenderPrepExecTracker.endLoadGeom1() == RenderPrepExecTracker::RESULT::CANCELED) {
2446 return RP_RESULT::CANCELED;
2447 }
2448 mGeometryManager->setChangeFlag(flag);
2449
2450 timer.stop();
2451
2452 // return unused memory from malloc() arena to OS so process memory usage
2453 // stats are accurate
2454 malloc_trim(0);
2455
2456 mRenderStats->logEndGeneratingProcedurals();
2457
2458 // Get the camera frustums and render to camera matrices for times points 0 and 1
2459 // TODO: generalize for multi-segment motion blur. Currently we only have 2
2460 // motion samples and assume the ray time points are 0 and 1. In multi-segment
2461 // motion blur there will be multiple motion samples with a ray time range
2462 // of [0,1].
2463 std::vector<mcrt_common::Frustum> frustums;
2464 for (size_t i = 0; i < scene->getCameraCount(); i++) {
2465 const pbr::Camera *camera = getScene()->getCamera(i);
2466 if (camera->hasFrustum()) {
2467 frustums.push_back(mcrt_common::Frustum());
2468 camera->computeFrustum(&frustums.back(), 0, true); // frustum at shutter open
2469 frustums.push_back(mcrt_common::Frustum());
2470 camera->computeFrustum(&frustums.back(), 1, true); // frustum at shutter close
2471 }
2472 }
2473
2474 // configure the way to construct spatial accelerator
2475 constexpr rt::OptimizationTarget accelMode = rt::OptimizationTarget::HIGH_QUALITY_BVH_BUILD;
2476
2477 // this will trigger BVH build if needed.
2478
2479 mGeometryManager->
2480 setStageIdAndCallBackFinalizeChange(0,
2481 mRenderPrepExecTracker.getRenderPrepStatsCallBack(),
2482 mRenderPrepExecTracker.getRenderPrepCancelCallBack());
2483 if (mRenderPrepExecTracker.startFinalizeChange0() == RenderPrepExecTracker::RESULT::CANCELED) {
2484 return RP_RESULT::CANCELED;
2485 }
2486 if (mGeometryManager->finalizeChanges(mLayer, motionBlurParams, frustums,
2487 world2render, accelMode, true /*update scene bvh*/) ==
2488 rt::GeometryManager::GM_RESULT::CANCELED) {
2489 return RP_RESULT::CANCELED;
2490 }
2491 if (mRenderPrepExecTracker.endFinalizeChange0() == RenderPrepExecTracker::RESULT::CANCELED) {
2492 return RP_RESULT::CANCELED;
2493 }
2494 mGeometryManager->
2495 setStageIdAndCallBackFinalizeChange(1,
2496 mRenderPrepExecTracker.getRenderPrepStatsCallBack(),
2497 mRenderPrepExecTracker.getRenderPrepCancelCallBack());
2498 if (mRenderPrepExecTracker.startFinalizeChange1() == RenderPrepExecTracker::RESULT::CANCELED) {
2499 return RP_RESULT::CANCELED;
2500 }
2501 if (mGeometryManager->finalizeChanges(mMeshLightLayer, motionBlurParams, frustums,
2502 world2render, accelMode, false /*don't update scene bvh*/) ==
2503 rt::GeometryManager::GM_RESULT::CANCELED) {
2504 return RP_RESULT::CANCELED;
2505 }
2506 if (mRenderPrepExecTracker.endFinalizeChange1() == RenderPrepExecTracker::RESULT::CANCELED) {
2507 return RP_RESULT::CANCELED;
2508 }
2509 mGeometryManager->setChangeFlag(rt::ChangeFlag::NONE);
2510
2511 // collect statistics
2512 mRenderStats->mTessellationTime =
2513 mGeometryManager->getStatistics().mTessellationTime;
2514 mRenderStats->mPerPrimitiveTessellationTime =
2515 mGeometryManager->getStatistics().mPerPrimitiveTessellationTime;
2516 mRenderStats->mBuildAcceleratorTime =
2517 mGeometryManager->getStatistics().mBuildAcceleratorTime;
2518 mRenderStats->mBuildProceduralTime =
2519 mGeometryManager->getStatistics().mBuildProceduralTime;
2520 mRenderStats->mRtcCommitTime =
2521 mGeometryManager->getStatistics().mRtcCommitTime;
2522
2523 return RP_RESULT::FINISHED;
2524}

2745void
2746RenderContext::buildFrameState(FrameState *fs, double frameStartTime, ExecutionMode executionMode) const
2747{
2748 // cppcheck-suppress memsetClassFloat // floating point memset to 0 is fine
2749 memset(fs, 0, sizeof(FrameState));
2750
2751 const scene_rdl2::rdl2::SceneVariables &vars = mSceneContext->getSceneVariables();
2752 unsigned pixelSamplesSqrt = (unsigned)std::max(1, vars.get(scene_rdl2::rdl2::SceneVariables::sPixelSamplesSqrt));
2753
2754 unsigned numSamplesPerPixel = pixelSamplesSqrt * pixelSamplesSqrt;
2755
2756 SamplingMode samplingMode = (SamplingMode)vars.get(scene_rdl2::rdl2::SceneVariables::sSamplingMode);
2757 validateSamplingMode(samplingMode);
2758
2759 // Adaptive sampling mode takes priority over pixel samples maps if both are active.
2760 // This means to use the pixel sample maps functionality, adaptive sampling must be
2761 // set to off.
2762 // Adaptive sampling isn't currently supported in vector or xpu mode.
2763 fs->mSamplingMode = samplingMode;
2764 fs->mOriginalSamplesPerPixel = numSamplesPerPixel;
2765
2766 if (samplingMode == SamplingMode::UNIFORM) {
2767
2768 // The maximum number of samples per pixel over all the pixel sample maps from each camera
2769 fs->mMaxSamplesPerPixel = unsigned(numSamplesPerPixel *
2770 *std::max_element(mMaxPixelSampleValues.begin(), mMaxPixelSampleValues.end()));
2771 fs->mMinSamplesPerPixel = fs->mMaxSamplesPerPixel;
2772 fs->mTargetAdaptiveError = 0.f;
2773 fs->mPixelSampleMaps = MNRY_VERIFY(mPixelSampleMaps.get());
2774
2775 } else {
2776
2777 // Adaptive sampling is on.
2778 fs->mMinSamplesPerPixel = (unsigned)std::max(2, vars.get(scene_rdl2::rdl2::SceneVariables::sMinAdaptiveSamples));
2779 fs->mMaxSamplesPerPixel = (unsigned)std::max((int)fs->mMinSamplesPerPixel, vars.get(scene_rdl2::rdl2::SceneVariables::sMaxAdaptiveSamples));
2780
2781 // The paper on which our adaptive sampling is based uses an error value of 0.0002 for "convergence". We divide
2782 // by 10,000 in order to provide more user-friendly values (e.g. 2.0).
2783 const float targetAdaptiveError = vars.get(scene_rdl2::rdl2::SceneVariables::sTargetAdaptiveError) / 10000.0f;
2784 fs->mTargetAdaptiveError = std::max(0.000001f, targetAdaptiveError);
2785 fs->mPixelSampleMaps = nullptr;
2786 }
2787
2788 fs->mMinAdaptiveSamplesPerPixel = (unsigned)std::max(2, vars.get(scene_rdl2::rdl2::SceneVariables::sMinAdaptiveSamples));
2789 fs->mDeepFormat = vars.get(scene_rdl2::rdl2::SceneVariables::sDeepFormat);
2790 fs->mDeepCurvatureTolerance = vars.get(scene_rdl2::rdl2::SceneVariables::sDeepCurvatureTolerance);
2791 fs->mDeepZTolerance = vars.get(scene_rdl2::rdl2::SceneVariables::sDeepZTolerance);
2792 fs->mDeepVolCompressionRes = vars.get(scene_rdl2::rdl2::SceneVariables::sDeepVolCompressionRes);
2793
2794 fs->mDeepIDChannelNames = MNRY_VERIFY(mDeepIDChannelNames.get());
2795 if (fs->mDeepIDChannelNames->size() > 6) {
2796 // 6 is needed to maintain the 64-byte size of DeepData
2797 throw scene_rdl2::except::RuntimeError(scene_rdl2::util::buildString(
2798 "More than 6 deep ID channels specified."));
2799 }
2800
2801 fs->mDeepMaxLayers = vars.get(scene_rdl2::rdl2::SceneVariables::sDeepMaxLayers);
2802 fs->mDeepLayerBias = vars.get(scene_rdl2::rdl2::SceneVariables::sDeepLayerBias);
2803
2804 //
2805 // From pbr::FrameState (all the data which the pbr library requires should
2806 // be set here).
2807 //
2808 fs->mExecutionMode = executionMode;
2809 fs->mNumActiveFilms = 1;
2810 fs->mEmbreeAccel = MNRY_VERIFY(mGeometryManager->getEmbreeAccelerator());
2811 fs->mGPUAccel = mGeometryManager->getGPUAccelerator(); // may be nullptr if not in xpu mode
2812 fs->mLayer = MNRY_VERIFY(mLayer);
2813 fs->mTextureBlur = vars.get(scene_rdl2::rdl2::SceneVariables::sTextureBlur);
2814 fs->mFatalColor = vars.get(scene_rdl2::rdl2::SceneVariables::sFatalColor);
2815 fs->mPropagateVisibilityBounceType = vars.get(scene_rdl2::rdl2::SceneVariables::sPropagateVisibilityBounceType);
2816 fs->mIntegrator = MNRY_VERIFY(mIntegrator.get());
2817 fs->mScene = MNRY_VERIFY(mPbrScene.get());
2818 fs->mAovSchema = &mRenderOutputDriver->getAovSchema();
2819 fs->mMaterialAovs = &mRenderOutputDriver->getMaterialAovs();
2820 fs->mLightAovs = &mRenderOutputDriver->getLightAovs();
2821 fs->mRequiresHeatMap = mRenderOutputDriver->requiresHeatMap();
2822 fs->mShadingWorkloadChunkSize = mOptions.getShadingWorkloadChunkSize();
2823 fs->mRequiresCryptomatteBuffer = mRenderOutputDriver->requiresCryptomatteBuffer();
2824
2825 //
2826 // From rndr::FrameState.
2827 //
2828 fs->mNumRenderThreads = MNRY_VERIFY(getNumTBBThreads());
2829
2830 int machineId = vars.get(scene_rdl2::rdl2::SceneVariables::sMachineId);
2831 int numMachines = vars.get(scene_rdl2::rdl2::SceneVariables::sNumMachines);
2832 fs->mNumRenderNodes = std::max(numMachines, 1);
2833 fs->mRenderNodeIdx = clamp(machineId, 0, (int)(fs->mNumRenderNodes - 1));
2834 fs->mTaskDistributionType = (unsigned)vars.get(scene_rdl2::rdl2::SceneVariables::sTaskDistributionType);
2835
2836 fs->mLockFrameNoise = vars.get(scene_rdl2::rdl2::SceneVariables::sLockFrameNoise);
2837
2838 switch (getRenderMode())
2839 {
2840 case RenderMode::BATCH:
2841 fs->mTileSchedulerType = TileScheduler::MORTON;
2842 break;
2843 case RenderMode::PROGRESSIVE:
2844 case RenderMode::PROGRESSIVE_FAST:
2845 fs->mTileSchedulerType = (unsigned)vars.get(scene_rdl2::rdl2::SceneVariables::sProgressiveTileOrder);
2846 break;
2847 case RenderMode::REALTIME:
2848 // @@@@ ML: Temp, use spiral square for realtime mode until such a time
2849 // we can configure the merge node with MORTON.
2850#if 0
2851 fs->mTileSchedulerType = TileScheduler::MORTON;
2852#else
2853 fs->mTileSchedulerType = TileScheduler::SPIRAL_SQUARE;
2854#endif
2855 break;
2856 case RenderMode::PROGRESS_CHECKPOINT:
2857 fs->mTileSchedulerType = (unsigned)vars.get(scene_rdl2::rdl2::SceneVariables::sCheckpointTileOrder);
2858 break;
2859 default:
2860 MNRY_ASSERT(0);
2861 }
2862
2863 if (getRenderMode() == RenderMode::BATCH) {
2864 if (mSceneContext->getCheckpointActive() || mSceneContext->getResumeRender()) {
2865 // We can use checkpoint mode for regular BATCH rendering.
2866 // Also use checkpoint mode if resumeRender mode is activated (Currently checkpoint render is
2867 // only supported to re-construct resume sampling schedule. So we have to use checkpoint
2868 // render for resumeRender execution).
2869 fs->mRenderMode = RenderMode::PROGRESS_CHECKPOINT;
2870 } else if (mSceneContext->getSceneVariables().get(scene_rdl2::rdl2::SceneVariables::sPathGuideEnable)) {
2871 // we'll use progressive mode to enable pass resets
2872 fs->mRenderMode = RenderMode::PROGRESSIVE;
2873 } else if (fs->mSamplingMode == SamplingMode::ADAPTIVE) {
2874 fs->mRenderMode = RenderMode::PROGRESSIVE;
2875 }
2876 } else {
2877 fs->mRenderMode = getRenderMode();
2878
2879 /* useful debug code : progressCheckPoint mode test for moonray_gui
2880 if (getRenderMode() == RenderMode::PROGRESSIVE) {
2881 if (mSceneContext->getCheckpointActive() || mSceneContext->getResumeRender()) {
2882 fs->mRenderMode = RenderMode::PROGRESS_CHECKPOINT;
2883 }
2884 }
2885 */
2886 }
2887 fs->mFastMode = getFastRenderMode();
2888 fs->mRequiresDeepBuffer = mRenderOutputDriver->requiresDeepBuffer();
2889
2890 // Find a better way to drive pixel info...
2891 fs->mGeneratePixelInfo = mOptions.getGeneratePixelInfo();
2892
2893 fs->mWidth = vars.getRezedWidth();
2894 fs->mHeight = vars.getRezedHeight();
2895
2896 // Clamp viewport to rezed dimensions to ensure it's inside of our
2897 // allocated buffers.
2898 fs->mViewport = scene_rdl2::math::convertToClosedViewport(vars.getRezedSubViewport());
2899
2900 MNRY_ASSERT(fs->mViewport.mMinX >= 0 && fs->mViewport.mMinY >= 0);
2901
2902 fs->mDofEnabled = false;
2903 for (size_t i = 0; i < mPbrScene->getCameraCount(); i++) {
2904 if (mPbrScene->getCamera(i)->getIsDofEnabled()) {
2905 fs->mDofEnabled = true;
2906 break;
2907 }
2908 }
2909
2910 fs->mPixelFilter = MNRY_VERIFY(mPixelFilter.get());
2911
2912 fs->mFrameNumber = getCurrentFrame();
2913 fs->mFps = std::max(vars.get(scene_rdl2::rdl2::SceneVariables::sFpsKey), 1.f);
2914 if (mOptions.getFps() > 0.0f) {
2915 fs->mFps = mOptions.getFps(); // overwrite fps value by option
2916 }
2917
2918 fs->mFrameStartTime = frameStartTime;
2919
2920 for (size_t i = 0; i < mPbrScene->getCameraCount(); i++) {
2921 const pbr::Camera* camera = getScene()->getCamera(i);
2922 StereoView stereoView = camera->getStereoView();
2923 fs->mInitialSeed[i] = getInitialSeed(fs->mFrameNumber, stereoView, fs->mLockFrameNoise);
2924 }
2925
2926 // the presence shadow handler in vectorized mode need to access this value
2927 fs->mMaxPresenceDepth = vars.get(scene_rdl2::rdl2::SceneVariables::sMaxPresenceDepth);
2928 fs->mPresenceThreshold = vars.get(scene_rdl2::rdl2::SceneVariables::sPresenceThreshold);
2929
2930 fs->mShadowTerminatorFix = static_cast<shading::ShadowTerminatorFix>
2931 (vars.get(scene_rdl2::rdl2::SceneVariables::sShadowTerminatorFix));
2932
2933 // checkpoint related
2934 // Setup checkpoint interval (minute) : 0.5sec or bigger
