31PolygonMesh::PolygonMesh(FaceVertexCount&& faceVertexCount,
32 IndexBuffer&& indices,
33 VertexBuffer&& vertices,
34 LayerAssignmentId&& layerAssignmentId,
35 shading::PrimitiveAttributeTable&& primitiveAttributeTable)
36{
37 // n-gon to triangle
38 // nTriangle = n - 2
39 // e.g. pentagon(5) -> 3 triangles; quad(4) -> 2 triangles
40 size_t outputTriFaceCount = std::accumulate(
41 faceVertexCount.begin(), faceVertexCount.end(), 0,
42 [](const size_t& faceCount, size_t curFaceVerts) {
43 return faceCount + (curFaceVerts - 2);
44 });
45 // n-gon to quad
46 // nQuad = (n - 1) / 2
47 // e.g. triangle(3) -> 1 quad
48 // pentagon(5) -> 1 quad + 1 triangle -> 2 quads
49 // hexagon (6) -> 2 quads
50 // heptagon(7) -> 2 quads+ 1 triangle -> 3 quads
51 size_t outputQuadFaceCount = std::accumulate(
52 faceVertexCount.begin(), faceVertexCount.end(), 0,
53 [](const size_t& faceCount, size_t curFaceVerts) {
54 return faceCount + ((curFaceVerts - 1) >> 1);
55 });
56 // compare the total storage of indices of tessellated mesh
57 if (outputQuadFaceCount * 4 < outputTriFaceCount * 3) {
58 mImpl = fauxstd::make_unique<Impl>(new internal::QuadMesh(outputQuadFaceCount,
59 std::move(faceVertexCount), std::move(indices), std::move(vertices),
60 std::move(layerAssignmentId), std::move(primitiveAttributeTable)));
61 } else {
62 mImpl = fauxstd::make_unique<Impl>(new internal::TriMesh(outputTriFaceCount,
63 std::move(faceVertexCount), std::move(indices), std::move(vertices),
64 std::move(layerAssignmentId), std::move(primitiveAttributeTable)));
65 }
66}

183void
184PolygonMesh::transformPrimitive(
185 const MotionBlurParams& motionBlurParams,
186 const shading::XformSamples& prim2render)
187{
188 size_t motionSamplesCount = getMotionSamplesCount();
189 shading::XformSamples p2r = prim2render;
190 if (motionSamplesCount > 1 && prim2render.size() == 1) {
191 p2r.resize(motionSamplesCount, prim2render[0]);
192 }
193 const shading::PrimitiveAttributeTable* primAttrTab = mImpl->mPolyMesh->getPrimitiveAttributeTable();
194 transformVertexBuffer(mImpl->mPolyMesh->getVertexBuffer(), p2r, motionBlurParams,
195 mImpl->mPolyMesh->getMotionBlurType(), mImpl->mPolyMesh->getCurvedMotionBlurSampleCount(),
196 primAttrTab);
197 float shutterOpenDelta, shutterCloseDelta;
198 motionBlurParams.getMotionBlurDelta(shutterOpenDelta, shutterCloseDelta);
199 mImpl->mPolyMesh->setTransform(p2r, shutterOpenDelta, shutterCloseDelta);
200}
