24int
25LightSet::intersect(const Vec3f &P, const Vec3f *N, const Vec3f &wi, float time,
26 float maxDistance, bool includeRayTerminationLights, IntegratorSample1D &samples, int depth,
27 int visibilityMask, LightIntersection &chosenIsect, int &numHits) const
28{
29 chosenIsect.distance = maxDistance;
30
31 numHits = 0;
32
33 // Use acceleration if it would be quicker
34 const LightAccelerator* acc = mAccelerator;
35 if (acc && acc->useAcceleration()) {
36 int idx = acc->intersect(P, N, wi, time, maxDistance, includeRayTerminationLights, visibilityMask,
37 samples, depth, chosenIsect, numHits, mAcceleratorLightIdMap);
38 return idx >= 0 ? mAcceleratorLightIdMap[idx] : idx;
39 }
40
41 int chosenLightIdx = -1;
42
43 // Test lights for intersection
44 for (int l = 0; l < mLightCount; l++) {
45
46 LightIntersection currentIsect;
47 const Light* light = mLights[l];
48
49 if (!(visibilityMask & light->getVisibilityMask())) {
50 // skip light if it is masked
51 continue;
52 }
53
54 if (!includeRayTerminationLights && light->getIsRayTerminator()) {
55 // Skip any ray termination lights if we were told not to include them
56 continue;
57 }
58
59 if (light->intersect(P, N, wi, time, maxDistance, currentIsect)) {
60 numHits++;
61 if (chooseThisLight(samples, depth, numHits)) {
62 chosenLightIdx = l;
63 chosenIsect = currentIsect;
64 }
65 }
66 }
67
68 MNRY_ASSERT(numHits==0 || chosenLightIdx >= 0);
69
70 return chosenLightIdx;
71}

73void
74LightSet::intersectAndEval(mcrt_common::ThreadLocalState *tls, const Vec3f &P, const Vec3f *N,
75 const Vec3f &wi, const LightFilterRandomValues& filterR, float time, bool fromCamera, bool includeRayTerminationLights,
76 IntegratorSample1D &samples, int depth, int visibilityMask, LightContribution &lCo,
77 float rayDirFootprint) const
78{
79 // Initialize contribution
80 lCo.isInvalid = true;
81 lCo.light = nullptr;
82 lCo.distance = sMaxValue;
83 lCo.Li = sBlack;
84 lCo.pdf = 0.0f;
85
86 // Compute intersection with a random light
87 LightIntersection isect;
88 int numLightsHit = 0;
89 int lightIdx = intersect(P, N, wi, time, sMaxValue, includeRayTerminationLights,
90 samples, depth, visibilityMask, isect, numLightsHit);
91
92 const Light* light = nullptr;
93 const LightFilterList* lightFilterList = nullptr;
94 if (lightIdx >= 0) {
95 light = getLight(lightIdx);
96 lightFilterList = getLightFilterList(lightIdx);
97 }
98
99 // Early return if we didn't hit a light
100 if (numLightsHit == 0) {
101 return;
102 }
103
104 lCo.light = light;
105 lCo.distance = isect.distance;
106
107 // Evaluate the intersected light Li and pdf
108 lCo.Li = light->eval(tls, wi, P, filterR, time, isect, fromCamera, lightFilterList, rayDirFootprint, &lCo.pdf);
109 lCo.isInvalid = isSampleInvalid(lCo.Li, lCo.pdf);
110
111 // Radiance has to be scaled by number of lights hit because we're only sampling one
112 lCo.Li *= (float)numLightsHit;
113}
