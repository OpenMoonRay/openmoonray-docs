22void
23TestRenderOutput::setUp()
24{
25 mContext.reset(new SceneContext);
26 CPPUNIT_ASSERT(mContext);
27 RenderOutput *ro =
28 mContext->createSceneObject("RenderOutput",
29 "/renderOutput")->asA<RenderOutput>();
30 // check defaults
31 CPPUNIT_ASSERT(ro->getActive());
32 CPPUNIT_ASSERT(ro->getResult() == RenderOutput::RESULT_BEAUTY);
33 CPPUNIT_ASSERT(ro->getStateVariable() == RenderOutput::STATE_VARIABLE_N);
34 CPPUNIT_ASSERT(ro->getPrimitiveAttribute() == "");
35 CPPUNIT_ASSERT(ro->getPrimitiveAttributeType() == RenderOutput::PRIMITIVE_ATTRIBUTE_TYPE_FLOAT);
36 CPPUNIT_ASSERT(ro->getMaterialAov() == "");
37 CPPUNIT_ASSERT(ro->getLpe() == "");
38 CPPUNIT_ASSERT(ro->getFileName() == "scene.exr");
39 CPPUNIT_ASSERT(ro->getFilePart() == "");
40 CPPUNIT_ASSERT(ro->getCompression() == RenderOutput::COMPRESSION_ZIP);
41 CPPUNIT_ASSERT(ro->getChannelName() == "");
42 CPPUNIT_ASSERT(ro->getChannelFormat() == RenderOutput::CHANNEL_FORMAT_HALF);
43 CPPUNIT_ASSERT(ro->getCheckpointFileName() == "checkpoint.exr");
44 CPPUNIT_ASSERT(ro->getResumeFileName() == "");
45
46 // sets
47 ro->beginUpdate();
48 ro->setActive(false);
49 ro->setResult(RenderOutput::RESULT_DEPTH);
50 ro->setStateVariable(RenderOutput::STATE_VARIABLE_P);
51 ro->setPrimitiveAttribute("surface_st");
52 ro->setPrimitiveAttributeType(RenderOutput::PRIMITIVE_ATTRIBUTE_TYPE_VEC2F);
53 ro->setMaterialAov("diffuse");
54 ro->setLpe("CD*L");
55 ro->setFileName("foo.exr");
56 ro->setFilePart("bar_part");
57 ro->setCompression(RenderOutput::COMPRESSION_DWAA);
58 ro->setChannelName("baz_channel");
59 ro->setChannelFormat(RenderOutput::CHANNEL_FORMAT_FLOAT);
60 ro->setCheckpointFileName("qux.exr");
61 ro->setResumeFileName("quux.exr");
62 ro->endUpdate();
63}
