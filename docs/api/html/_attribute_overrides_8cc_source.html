33template <typename T>
34void
35setOverrideHelper(
36 scene_rdl2::rdl2::SceneContext& context,
37 scene_rdl2::rdl2::SceneObject* object,
38 const RenderOptions::AttributeOverride& attrOverride)
39{
40 scene_rdl2::rdl2::AttributeKey<T> key =
41 object->getSceneClass().getAttributeKey<T>(attrOverride.mAttribute);
42
43 scene_rdl2::rdl2::SceneObject::UpdateGuard guard(object);
44 if (!attrOverride.mValue.empty()) {
45 object->set(key, scene_rdl2::rdl2::convertFromString<T>(attrOverride.mValue));
46 }
47 if (!attrOverride.mBinding.empty() && key.isBindable()) {
48 object->setBinding(key, context.getSceneObject(attrOverride.mBinding));
49 }
50}

201void
202applyAttributeOverrides(scene_rdl2::rdl2::SceneContext& context,
203 const RenderStats& renderStats,
204 const RenderOptions& options,
205 std::stringstream& initMessages)
206{
207 std::vector<RenderOptions::AttributeOverride> overrides =
208 options.getAttributeOverrides();
209 std::string prepend = renderStats.getPrependString();
210
211 for (auto iter = overrides.begin(); iter != overrides.end(); ++iter) {
212 try {
213 setOverride(context, *iter);
214 if (!iter->mValue.empty()) {
215 initMessages << "Overriding '" << iter->mAttribute <<
216 "' value with '" << iter->mValue << "'." << '\n';
217 }
218 if (!iter->mBinding.empty()) {
219 initMessages << "Overriding '" << iter->mAttribute <<
220 "' binding with '" << iter->mBinding << "'." << '\n';
221 }
222 } catch (std::exception& e) {
223 scene_rdl2::logging::Logger::warn("Skipping override: " , e.what());
224 }
225 }
226}
