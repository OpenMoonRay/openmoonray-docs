53 template<typename T> void
54 computeVertexAttributeDerivatives(shading::TypedAttributeKey<T> key,
55 float invDs, int vertexOffset,
56 float time, shading::Intersection& intersection) const
57 {
58 int vid1 = vertexOffset;
59 int vid2 = vertexOffset + 1;
60 T dfds;
61 if (mAttributes->getTimeSampleCount(key) > 1) {
62 T f1 = mAttributes->getMotionBlurVertex(key, vid1, time);
63 T f2 = mAttributes->getMotionBlurVertex(key, vid2, time);
64 dfds = (f2 - f1) * invDs;
65 } else {
66 const T& f1 = mAttributes->getVertex(key, vid1);
67 const T& f2 = mAttributes->getVertex(key, vid2);
68 dfds = (f2 - f1) * invDs;
69 }
70 intersection.setdAttributeds(key, dfds);
71 }
