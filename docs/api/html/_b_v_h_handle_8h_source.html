19// BVHHandle is the embree side representation of a Primitive, it provides
20// functionality to remove/update a Primitive in embree BVH. Each Primitive
21// contains a BVHHandle after BVH is constructed and the life cycle of
22// the embree representation is bound to its corresponding Primitive
23class BVHHandle {
24public:
25
26 BVHHandle(RTCScene& parentScene, uint32_t geomID):
27 mParentScene(parentScene), mGeomID(geomID) {}
28
29 ~BVHHandle() {
30 if (mParentScene != nullptr && mGeomID != RTC_INVALID_GEOMETRY_ID) {
31 rtcDetachGeometry(mParentScene, mGeomID);
32 rtcReleaseGeometry(rtcGetGeometry(mParentScene, mGeomID));
33 }
34 mParentScene = nullptr;
35 mGeomID = RTC_INVALID_GEOMETRY_ID;
36 }
37
38 void update() {
39 if (mParentScene != nullptr && mGeomID != RTC_INVALID_GEOMETRY_ID) {
40 rtcCommitGeometry(rtcGetGeometry(mParentScene, mGeomID));
41 }
42 }
43
44 uint32_t getGeomID() const { return mGeomID; }
45
46private:
47 RTCScene mParentScene;
48 uint32_t mGeomID;
49};
50
