48// given an array of sorted shader stats, extract a common prefix from
49// the first maxEntry items.
50std::string
51getCommonPrefix(std::vector<RenderStats::ShaderStat>::const_iterator first,
52 std::vector<RenderStats::ShaderStat>::const_iterator last)
53{
54 std::string result = "";
55 if (first == last) {
56 return result;
57 }
58 MNRY_ASSERT(first < last);
59 const std::string firstName = first->first->getName();
60 int rootPos = firstName.length();
61 for (auto it = first; it != last; ++it) {
62 auto miss = std::mismatch(firstName.begin(), firstName.end(), it->first->getName().begin());
63 if (miss.first - firstName.begin() < rootPos) {
64 rootPos = miss.first - firstName.begin();
65 }
66 }
67 if (rootPos > 0) {
68 rootPos = firstName.rfind('/', rootPos);
69 if (rootPos != std::string::npos) { // found it
70 result = std::string(firstName.begin(), firstName.begin() + rootPos);
71 }
72 }
73
74 return result;
75}

97void
98addLaneUtilization(moonray_stats::StatsTable<2> &table, const pbr::Statistics &pbrStats, int counter, const char *name)
99{
100 const size_t laneActivity = pbrStats.getCounter(counter);
101 const size_t laneMax = pbrStats.getCounter(counter + 1);
102
103 MNRY_ASSERT(laneActivity <= laneMax);
104
105 const double utilization = (laneMax == 0) ? 1.0 :
106 static_cast<double>(laneActivity) / static_cast<double>(laneMax);
107
108 table.emplace_back(name, moonray_stats::percentage(utilization));
109}

208double
209RenderStats::ticksToSec(const uint64 value, const int numThreads)
210{
211 MNRY_ASSERT(numThreads);
212
213 // If this asserts then updateToMostRecentTicksPerSecond() needs to be called
214 // before calling ticksToSec.
215 MNRY_ASSERT(mInvTicksPerSecond > 0.0);
216
217 // we divide by numThreads in an attempt to make the
218 // per-thread shader times more visually relatable
219 // to the total render time.
220 return (value / numThreads) * mInvTicksPerSecond;
221}

276std::string
277RenderStats::getPrependString() const
278{
279 std::ostringstream out;
280
281 // our estimated process time from the construction of the renderStats
282 // class
283 double processTime = time::getTime() - mFrameStartTime;
284
285 if (mLogTime) {
286 out << timeIntervalFormat(processTime, 0);
287 out.put(' ');
288 }
289
290 if (mLogMemory) {
291 int64 memory = mProcessStats.getProcessMemory();
292 out << std::setw(9) << memFormat(memory);
293 }
294
295 if (mLogReadIO) {
296 out << memFormat(mProcessStats.getBytesRead());
297 out.put(' ');
298 }
299
300 if (mLogUtilization) {
301 util::ProcessUtilization us;
302 us = mProcessStats.getProcessUtilization();
303
304 out << std::fixed << std::right << std::setw(5);
305 out << (us.systemTime + us.userTime) / processTime << '%';
306 }
307 out << " | ";
308 return out.str();
309}

473void
474RenderStats::logRenderOptions(const RenderOptions& options, std::ostream& outs, OutputFormat format)
475{
476 const bool csvStream = format == OutputFormat::athenaCSV || format == OutputFormat::fileCSV;
477
478 char curPath[MAXPATHLEN];
479 getcwd(curPath, MAXPATHLEN);
480
481 const std::string executablePath = [] {
482 try {
483 return getExecutablePath();
484 } catch (const std::exception& e) {
485 Logger::warn("Unable to resolve executable path: ", e.what());
486 return std::string("Unable to resolve");
487 }
488 }();
489
490#if defined(__AVX512F__)
491 const constexpr char* const sSimdName = "avx512";
492#elif defined(__AVX__)
493 const constexpr char* const sSimdName = "avx";
494#else
495#error Unknown SIMD type
496#endif
497
498 const char *desiredExecutionMode = nullptr;
499 switch (options.getDesiredExecutionMode()) {
500 case mcrt_common::ExecutionMode::AUTO: desiredExecutionMode = "AUTO"; break;
501 case mcrt_common::ExecutionMode::VECTORIZED: desiredExecutionMode = "VECTORIZED"; break;
502 case mcrt_common::ExecutionMode::SCALAR: desiredExecutionMode = "SCALAR"; break;
503 case mcrt_common::ExecutionMode::XPU: desiredExecutionMode = "XPU"; break;
504 default:
505 MNRY_ASSERT(0);
506 }
507
508 const std::string& athenaTags = options.getAthenaTags();
509 StatsTable<2> table("Rendering Options");
510 table.emplace_back("Command line", options.getCommandLine());
511 table.emplace_back("Executable path", executablePath);
512 table.emplace_back("Moonray version", getMoonrayVersion(executablePath));
513 table.emplace_back("DSO path override", options.getDsoPath());
514 table.emplace_back("Desired vectorization mode", desiredExecutionMode); // Note: the Athena data team requested it be output under this name
515 table.emplace_back("SIMD build support", sSimdName);
516 table.emplace_back("Athena Tags", athenaTags );
517
518 const auto& sceneFiles = options.getSceneFiles();
519 for (const auto& sceneFile : sceneFiles) {
520 table.emplace_back("Scene file", getFullPath(sceneFile));
521 }
522
523 for (const auto& deltasFile : options.getDeltasFiles()) {
524 table.emplace_back("Deltas file", getFullPath(deltasFile));
525 }
526
527 using AttributesTable = StatsTable<4>;
528 AttributesTable attrTable("Attribute Overrides",
529 "Object", "Attribute", "Value", "Binding");
530 const auto& overrides = options.getAttributeOverrides();
531 if (csvStream || !overrides.empty()) {
532 for (const auto& override : overrides) {
533 if (!override.mIsSceneVar) {
534 attrTable.emplace_back(override.mObject,
535 override.mAttribute,
536 override.mValue,
537 override.mBinding);
538 }
539 }
540 }
541
542 StatsTable<2> luaTable("Lua Globals");
543 const auto& luaGlobals = options.getRdlaGlobals();
544 if (csvStream || !luaGlobals.empty()) {
545 for (const auto& luaglobal : luaGlobals) {
546 luaTable.emplace_back(luaglobal.mVar, luaglobal.mExpression);
547 }
548 }
549
550 if (csvStream) {
551 outs.precision(2);
552 writeEqualityCSVTable(outs, table, format == OutputFormat::athenaCSV);
553 writeCSVTable(outs, attrTable, format == OutputFormat::athenaCSV);
554 writeEqualityCSVTable(outs, luaTable, format == OutputFormat::athenaCSV);
555 } else {
556 outs.precision(2);
557 writeEqualityInfoTable(outs, getPrependString(), table);
558 if (!attrTable.empty()) {
559 writeInfoTable(outs, getPrependString(), attrTable);
560 }
561 if (!luaTable.empty()) {
562 writeEqualityInfoTable(outs, getPrependString(), luaTable);
563 }
564 }
565}

712void
713RenderStats::logSceneVariables(const scene_rdl2::rdl2::SceneVariables &vars, std::ostream& outs, OutputFormat format)
714{
715 const bool csvStream = format == OutputFormat::athenaCSV || format == OutputFormat::fileCSV;
716
717 // TODO: do we want to add anymore scene vars to this list?
718 HalfOpenViewport aperture = vars.getRezedApertureWindow();
719 HalfOpenViewport region = vars.getRezedRegionWindow();
720
721 StatsTable<2> sceneVarTable("Scene Variables");
722 sceneVarTable.emplace_back("Width", vars.get(scene_rdl2::rdl2::SceneVariables::sImageWidth));
723 sceneVarTable.emplace_back("Height", vars.get(scene_rdl2::rdl2::SceneVariables::sImageHeight));
724 sceneVarTable.emplace_back("Resolution", vars.get(scene_rdl2::rdl2::SceneVariables::sResKey));
725 sceneVarTable.emplace_back("Final width", vars.getRezedWidth());
726 sceneVarTable.emplace_back("Final height", vars.getRezedHeight());
727 sceneVarTable.emplace_back("Final aperture window min x", aperture.mMinX);
728 sceneVarTable.emplace_back("Final aperture window min y", aperture.mMinY);
729 sceneVarTable.emplace_back("Final aperture window max x", aperture.mMaxX);
730 sceneVarTable.emplace_back("Final aperture window max y", aperture.mMaxY);
731 sceneVarTable.emplace_back("Final region window min x", region.mMinX);
732 sceneVarTable.emplace_back("Final region window min y", region.mMinY);
733 sceneVarTable.emplace_back("Final region window max x", region.mMaxX);
734 sceneVarTable.emplace_back("Final region window max y", region.mMaxY);
735
736 const scene_rdl2::rdl2::PixelFilterType pfilter = static_cast<scene_rdl2::rdl2::PixelFilterType>(
737 vars.get(scene_rdl2::rdl2::SceneVariables::sPixelFilterType));
738
739 switch (pfilter) {
740 case scene_rdl2::rdl2::PixelFilterType::box:
741 sceneVarTable.emplace_back("Pixel filter type", "box");

1264template <typename Iterator, typename PartitionFunction, typename CompFunction>
1265std::pair<Iterator, Iterator>
1266getRelevantStats(Iterator first,
1267 Iterator last,
1268 PartitionFunction partFunction,
1269 CompFunction compFunction,
1270 std::size_t maxEntry)
1271{
1272
1273 MNRY_ASSERT(first <= last);
1274
1275 // Trim off the parts with 0 value.
1276 last = std::partition(first, last, partFunction);
1277 MNRY_ASSERT(first <= last);
1278
1279 // Count the number of relevant entries.
1280 const auto dist = std::distance(first, last);
1281 MNRY_ASSERT(dist >= 0);
1282 const std::size_t num = std::min(maxEntry, static_cast<std::size_t>(dist));
1283 MNRY_ASSERT(first <= last);
1284 MNRY_ASSERT(first + num <= last);
1285
1286 // Sort "num" values from the relevant list
1287 std::partial_sort(first, first + num, last, compFunction);
1288
1289 MNRY_ASSERT(first <= last);
1290 MNRY_ASSERT(first + num <= last);
1291 return std::make_pair(first, std::next(first, num));
1292}
