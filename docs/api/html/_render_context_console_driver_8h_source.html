17class RenderContextConsoleDriver : public scene_rdl2::grid_util::DebugConsoleDriver
18//
19// This class provides server side functionality of interactive moonray_gui control feature by
20// telnet connection. This functionality was designed to simulate arras environment action
21// (i.e. progmcrt computation) without arras as a simple procedure. This is a very powerful
22// solution in order to isolate issues that are related to arras or purely moonray trouble.
23// We can send commands to moonray_gui via telnet connection while moonray_gui is actively rendering.
24// It is pretty easy to add a new command to this class if you need to.
25//
26// This object boots thread inside the DebugConsoleDriver::initialize() and this is a main thread
27// to handle telnet connection. This thread is shut down automatically when this object is destructed.
28// Only support single incoming telnet connection at this moment.
29//
30{
31public:
32 using RenderContextConsoleDriverShPtr = std::shared_ptr<RenderContextConsoleDriver>;
33
34 // Negative port number disables debug console features and never constructs
35 // RenderContextConsoleDriver internally (In this case, get() return nullptr).
36 // Port = 0 is special, in this case, the kernel finds the available port for you.
37 // Usually you specify the port number which you want to use.
38 static void init(int port);
39 static RenderContextConsoleDriverShPtr get();
40
41 RenderContextConsoleDriver() :
42 DebugConsoleDriver(),
