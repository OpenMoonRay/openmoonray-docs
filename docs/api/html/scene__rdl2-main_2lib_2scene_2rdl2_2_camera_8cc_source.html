54
55 sPixelSampleMap = sceneClass.declareAttribute<String>("pixel_sample_map", "", { "pixel sample map" });
56 sceneClass.setMetadata(sPixelSampleMap, "label", "pixel sample map");
57
58 // Grouping the attributes - the order of the attributes should be the same as how they are defined.
59 sceneClass.setGroup("Frustum", sNearKey);
60 sceneClass.setGroup("Frustum", sFarKey);
61
62 sceneClass.setGroup("Motion Blur", sMbShutterOpenKey);
63 sceneClass.setGroup("Motion Blur", sMbShutterCloseKey);
64 sceneClass.setGroup("Motion Blur", sMbShutterBiasKey);
65
66 sceneClass.setGroup("Render Masks", sPixelSampleMap);
