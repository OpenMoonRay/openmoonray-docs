135 template<typename T> __forceinline QuaternionT<T> slerp(const QuaternionT<T>& a, const QuaternionT<T>& b, const T t) {
136 T angle = T(zero);
137 T sineAngle = T(zero);
138 T cosAngle = dot(a,b);
139 if (abs(cosAngle) < T(one)) { // if angle between a & b is neither 0 or 180 degree.
140 angle = acos(cosAngle);
141 sineAngle = sin(angle);
142 }
143 if (abs(sineAngle) < T(0.00001f)) {
144 return (T(one)-t)*a + static_cast<T>(t)*b;
145 }
146 T rcpSine = T(one)/sineAngle;
147 T rA = sin((T(one)-t)*angle)*rcpSine;
148 T rB = sin(t*angle)*rcpSine;
149 return a*rA + b*rB;
150 }

173 template<typename T> QuaternionT<T>::QuaternionT( const Vec3<T>& vx, const Vec3<T>& vy, const Vec3<T>& vz )
174 {
175 if ( vx.x + vy.y + vz.z >= T(zero) )
176 {
177 const T t = T(one) + (vx.x + vy.y + vz.z);
178 const T s = rsqrt(t)*T(0.5f);
179 r = t*s;
180 i = (vy.z - vz.y)*s;
181 j = (vz.x - vx.z)*s;
182 k = (vx.y - vy.x)*s;
183 }
184 else if ( vx.x >= max(vy.y, vz.z) )
185 {
186 const T t = (T(one) + vx.x) - (vy.y + vz.z);
187 const T s = rsqrt(t)*T(0.5f);
188 r = (vy.z - vz.y)*s;
189 i = t*s;
190 j = (vx.y + vy.x)*s;
191 k = (vz.x + vx.z)*s;
192 }
193 else if ( vy.y >= vz.z ) // if ( vy.y >= max(vz.z, vx.x) )
194 {
195 const T t = (T(one) + vy.y) - (vz.z + vx.x);
196 const T s = rsqrt(t)*T(0.5f);
197 r = (vz.x - vx.z)*s;
198 i = (vx.y + vy.x)*s;
199 j = t*s;
200 k = (vy.z + vz.y)*s;
201 }
202 else //if ( vz.z >= max(vy.y, vx.x) )
203 {
204 const T t = (T(one) + vz.z) - (vx.x + vy.y);
205 const T s = rsqrt(t)*T(0.5f);
206 r = (vx.y - vy.x)*s;
207 i = (vz.x + vx.z)*s;
208 j = (vy.z + vz.y)*s;
209 k = t*s;
210 }
211 }

213 template<typename T> QuaternionT<T>::QuaternionT( const T& yaw, const T& pitch, const T& roll )
214 {
215 const T cya = cos(yaw *T(0.5f));
216 const T cpi = cos(pitch*T(0.5f));
217 const T cro = cos(roll *T(0.5f));
218 const T sya = sin(yaw *T(0.5f));
219 const T spi = sin(pitch*T(0.5f));
220 const T sro = sin(roll *T(0.5f));
221 r = cro*cya*cpi + sro*sya*spi;
222 i = cro*cya*spi + sro*sya*cpi;
223 j = cro*sya*cpi - sro*cya*spi;
224 k = sro*cya*cpi - cro*sya*spi;
225 }
