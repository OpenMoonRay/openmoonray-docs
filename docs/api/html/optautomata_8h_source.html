48/// Optimized compact version of DfAutomata
49///
50/// Apparently hash maps suck in speed for our transition tables. This
51/// is a fast compact equivalent of the DfAutomata designed for read
52/// only operations.
53///
54class DfOptimizedAutomata
55{
56 public:
57
58 void compileFrom(const DfAutomata &dfautomata);
59
60 int getTransition(int state, ustring symbol)const
61 {
62 const State &mystate = m_states[state];
63 const Transition *begin = &m_trans[mystate.begin_trans];
64 const Transition *end = begin + mystate.ntrans;
65 while (begin < end) { // binary search
66 const Transition *middle = begin + ((end - begin)>>1);
67 if (symbol.data() < middle->symbol.data())
68 end = middle;
69 else if (middle->symbol.data() < symbol.data())
70 begin = middle + 1;
71 else // match
72 return middle->state;
73 }
74 return mystate.wildcard_trans;
75 }
76
77 void * const * getRules(int state, int &count)const
78 {
79 count = m_states[state].nrules;
80 return &m_rules[m_states[state].begin_rules];
81 }
82
83 protected:

86 unsigned int begin_trans;
87 unsigned int ntrans;
88 unsigned int begin_rules;
89 unsigned int nrules;
90 int wildcard_trans;
91 };

98 };
99 std::vector<Transition> m_trans;
100 std::vector<void *> m_rules;
101 std::vector<State> m_states;
102};
