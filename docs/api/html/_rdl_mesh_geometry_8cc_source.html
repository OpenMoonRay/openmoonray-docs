42moonray::geom::LayerAssignmentId
43createPerFaceAssignmentId(const RdlMeshGeometry* rdlMeshGeometry,
44 const scene_rdl2::rdl2::Layer* rdlLayer,
45 const size_t faceCount) {
46 // per face assignment id
47 const int meshAssignmentId = rdlLayer->getAssignmentId(rdlMeshGeometry,
48 sDefaultPartName);
49
50 auto& partList = rdlMeshGeometry->get(attrPartList);
51 if (!partList.empty()) {
52 auto& partFaceCountList = rdlMeshGeometry->get(
53 attrPartFaceCountList);
54 auto& partFaceIndices = rdlMeshGeometry->get(attrPartFaceIndices);
55
56 if (partList.size() == partFaceCountList.size()) {
57 // validate size of face indices
58 const int expectedFaces = std::accumulate(
59 partFaceCountList.begin(), partFaceCountList.end(), 0);
60 if (expectedFaces == partFaceIndices.size()) {
61 std::vector<int> faceAssignmentIds(faceCount,
62 meshAssignmentId);
63
64 size_t begin = 0;
65 for (size_t i = 0; i < partList.size(); ++i) {
66 const scene_rdl2::rdl2::String& partName = partList[i];
67 const int partFaceCount = partFaceCountList[i];
68 const int partAssignmentId = rdlLayer->getAssignmentId(
69 rdlMeshGeometry, partName);
70 for (size_t pF = begin; pF < begin + partFaceCount;
71 ++pF) {
72 const int faceIndex = partFaceIndices[pF];
73 faceAssignmentIds[faceIndex] = partAssignmentId;
74 }
75 begin += partFaceCount;
76 }
77
78 return moonray::geom::LayerAssignmentId(std::move(faceAssignmentIds));
79 } else {
80 rdlMeshGeometry->warn(
81 "total part face count does not match "
82 "part face indices count, skipping");
83 }
84 } else {
85 rdlMeshGeometry->warn("part list is incorrect size for "
86 "partface count list, skipping");
87 }
88 }
89
90 return moonray::geom::LayerAssignmentId(meshAssignmentId);
91}

138void
139RdlMeshProcedural::generate(
140 const GenerateContext &generateContext,
141 const shading::XformSamples &parent2render)
142{
143 const scene_rdl2::rdl2::Geometry *rdlGeometry = generateContext.getRdlGeometry();
144
145 const RdlMeshGeometry *rdlMeshGeometry =
146 static_cast<const RdlMeshGeometry*>(rdlGeometry);
147
148 const bool isSubd = rdlMeshGeometry->get(attrIsSubd);
149
150 const scene_rdl2::rdl2::Layer *rdlLayer = generateContext.getRdlLayer();
151
152 std::unique_ptr<Primitive> primitive;
153 if (!isSubd) {
154 primitive = createPolyMesh(rdlGeometry, rdlLayer);
155 } else {
156 primitive = createSubdMesh(rdlGeometry, rdlLayer);
157 }
158
159 if (primitive) {
160 // may need to convert the primitive to instance to handle
161 // rotation motion blur
162 std::unique_ptr<Primitive> p = convertForMotionBlur(
163 generateContext, std::move(primitive),
164 (rdlMeshGeometry->get(attrUseRotationMotionBlur) &&
165 parent2render.size() > 1));
166 addPrimitive(std::move(p),
167 generateContext.getMotionBlurParams(), parent2render);
168
169 }
170}

173void
174RdlMeshProcedural::update(
175 const UpdateContext &updateContext,
176 const shading::XformSamples &parent2render)
177{
178 const std::vector<const std::vector<float>*> &vertexDatas =
179 updateContext.getMeshVertexDatas();
180
181 shading::XformSamples prim2render = computePrim2Render(getState(), parent2render);
182 if (mSubdMesh != nullptr) {
183 mSubdMesh->updateVertexData(
184 *vertexDatas[0], prim2render);
185 } else {
186 mPolygonMesh->updateVertexData(
187 *vertexDatas[0], prim2render);
188 }
189
190 mDeformed = true;
191}

194VertexBuffer<Vec3fa, InterleavedTraits>
195RdlMeshProcedural::getVertexData(
196 const scene_rdl2::rdl2::Geometry* rdlGeometry,
197 shading::PrimitiveAttributeTable &primitiveAttributeTable,
198 const moonray::geom::RateCounts& rates)
199{
200 const RdlMeshGeometry* rdlMeshGeometry =
201 static_cast<const RdlMeshGeometry*>(rdlGeometry);
202
203 auto& procPosList0 = rdlMeshGeometry->get(attrPos0);
204 auto& procPosList1 = rdlMeshGeometry->get(attrPos1);
205 auto& procVelList0 = rdlMeshGeometry->get(attrVel0);
206 auto& procVelList1 = rdlMeshGeometry->get(attrVel1);
207 auto& procAccList0 = rdlMeshGeometry->get(attrAcc);
208
209 const size_t vertCount = procPosList0.size();
210
211 bool pos1Valid = sizeCheck(rdlMeshGeometry, getName(rdlMeshGeometry, attrPos1), procPosList1.size(), vertCount);
212 bool vel0Valid = sizeCheck(rdlMeshGeometry, getName(rdlMeshGeometry, attrVel0), procVelList0.size(), vertCount);
213 bool vel1Valid = sizeCheck(rdlMeshGeometry, getName(rdlMeshGeometry, attrVel1), procVelList1.size(), vertCount);
214 bool acc0Valid = sizeCheck(rdlMeshGeometry, getName(rdlMeshGeometry, attrAcc), procAccList0.size(), vertCount);
215
216 // Fall back on static case if we don't have sufficient data for requested mb type
217 int numPosSamples = 1;
218 int numVelSamples = 0;
219 int numAccSamples = 0;
220
221 bool err = false;
222 scene_rdl2::rdl2::MotionBlurType motionBlurType = static_cast<scene_rdl2::rdl2::MotionBlurType>(rdlMeshGeometry->get(attrMotionBlurType));
223 switch (motionBlurType) {
224
225 case scene_rdl2::rdl2::MotionBlurType::STATIC:
226 {
227 break;
228 }
229
230 case scene_rdl2::rdl2::MotionBlurType::VELOCITY:
231 {
232 if (vel0Valid) {
233 numVelSamples = 1;
234 } else {
235 err = true;
236 }
237 break;
238 }
239
240 case scene_rdl2::rdl2::MotionBlurType::FRAME_DELTA:
241 {
242 if (pos1Valid) {
243 numPosSamples = 2;
244 } else {
245 err = true;
246 }
247 break;
248 }
249
250 case scene_rdl2::rdl2::MotionBlurType::ACCELERATION:
251 {
252 if (vel0Valid && acc0Valid) {
253 numVelSamples = 1;
254 numAccSamples = 1;
255 } else {
256 err = true;
257 }
258 break;
259 }
260
261 case scene_rdl2::rdl2::MotionBlurType::HERMITE:
262 {
263 if (pos1Valid && vel0Valid && vel1Valid) {
264 numPosSamples = 2;
265 numVelSamples = 2;
266 } else {
267 err = true;
268 }
269 break;
270 }
271
272 case scene_rdl2::rdl2::MotionBlurType::BEST:
273 {
274 if (pos1Valid && vel0Valid && vel1Valid) {
275 // use Hermite mb type
276 numPosSamples = 2;
277 numVelSamples = 2;
278 } else if (vel0Valid && acc0Valid) {
279 // use acceleration mb type
280 numVelSamples = 1;
281 numAccSamples = 1;
282 } else if (pos1Valid) {
283 // use frame delta mb type
284 numPosSamples = 2;
285 } else if (vel0Valid) {
286 // use velocity mb type
287 numVelSamples =1;
288 }
289 // else just keep static mb type
290 break;
291 }
292
293 default:
294 {
295 err = true;
296 break;
297 }
298
299 } // end of switch statement
300
301 if (err) {
302 rdlMeshGeometry->warn("Insufficient data for requested motion blur type. "
303 "Falling back to static case.");
304 }
305
306 // Copy vertices
307 VertexBuffer<Vec3fa, InterleavedTraits> vertices(vertCount, numPosSamples);
308 for (size_t i = 0; i < vertCount; i++) {
309 const auto& p = procPosList0[i];
310 vertices(i, 0) = Vec3fa(p[0], p[1], p[2], 0.f);
311 }
312 if (numPosSamples == 2) {
313 for (size_t i = 0; i < vertCount; i++) {
314 const auto& p = procPosList1[i];
315 vertices(i, 1) = Vec3fa(p[0], p[1], p[2], 0.f);
316 }
317 }
318
319 // Add velocity data
320 if (numVelSamples > 0) {
321 const float velocityScale = rdlMeshGeometry->get(attrVelocityScale);
322 std::vector<std::vector<Vec3f>> velocities;
323 velocities.push_back(std::vector<Vec3f>(procVelList0.begin(), procVelList0.end()));
324 for (size_t i = 0; i < procVelList0.size(); i++) {
325 velocities.back()[i] *= velocityScale;
326 }
327 if (numVelSamples == 2) {
328 velocities.push_back(std::vector<Vec3f>(procVelList1.begin(), procVelList1.end()));
329 for (size_t i = 0; i < procVelList1.size(); i++) {
330 velocities.back()[i] *= velocityScale;
331 }
332 }
333 primitiveAttributeTable.addAttribute(shading::StandardAttributes::sVelocity,
334 shading::AttributeRate::RATE_VERTEX, std::move(velocities));
335 }
336
337 // Add acceleration data
338 if (numAccSamples > 0) {
339 std::vector<Vec3f> accelerations(procAccList0.begin(), procAccList0.end());
340 primitiveAttributeTable.addAttribute(shading::StandardAttributes::sAcceleration,
341 shading::AttributeRate::RATE_VERTEX, std::move(accelerations));
342 }
343
344 // Add UVs
345 auto& procUVList = rdlMeshGeometry->get(attrUVs);
346 if (!procUVList.empty()) {
347 shading::AttributeRate attrRate(
348 pickRate(rdlMeshGeometry, getName(rdlMeshGeometry, attrUVs), procUVList.size(), rates));
349 std::vector<Vec2f> textureUV(procUVList.begin(), procUVList.end());
350 primitiveAttributeTable.addAttribute(shading::StandardAttributes::sSurfaceST,
351 attrRate, std::move(textureUV));
352 }
353
354 // Add normals
355 auto& procNormalList = rdlMeshGeometry->get(attrNormals);
356 if (!procNormalList.empty()) {
357 shading::AttributeRate attrRate(
358 pickRate(rdlMeshGeometry, getName(rdlMeshGeometry, attrNormals), procNormalList.size(), rates));
359 std::vector<Vec3f> normals(procNormalList.begin(), procNormalList.end());
360 primitiveAttributeTable.addAttribute(shading::StandardAttributes::sNormal,
361 attrRate, std::move(normals));
362 }
363
364 return vertices;
365}

368std::unique_ptr<SubdivisionMesh>
369RdlMeshProcedural::createSubdMesh(
370 const scene_rdl2::rdl2::Geometry* rdlGeometry,
371 const scene_rdl2::rdl2::Layer* rdlLayer)
372{
373 const RdlMeshGeometry* rdlMeshGeometry =
374 static_cast<const RdlMeshGeometry*>(rdlGeometry);
375
376 shading::PrimitiveAttributeTable primitiveAttributeTable;
377
378 // Set the vert per face count
379 auto& procFaceVertexCount = rdlMeshGeometry->get(attrFaceVertexCount);
380 SubdivisionMesh::FaceVertexCount faceVertexCount(
381 procFaceVertexCount.begin(), procFaceVertexCount.end());
382
383 // Store the vert indices in order of face list to build the mesh
384 auto& procIndices = rdlMeshGeometry->get(attrVertexIndex);
385 SubdivisionMesh::IndexBuffer indices(procIndices.begin(), procIndices.end());
386 if (indices.empty()) {
387 return nullptr;
388 }
389
390 auto& procVertList = rdlMeshGeometry->get(attrPos0);
391 const size_t vertCount = procVertList.size();
392 const size_t faceCount = faceVertexCount.size();
393 const size_t faceVaryingCount = procIndices.size();
394
395 // Fill in table of faces->parts
396 auto& partFaceCountList = rdlMeshGeometry->get(attrPartFaceCountList);
397 const size_t partCount = partFaceCountList.size();
398 auto& partFaceIndices = rdlMeshGeometry->get(attrPartFaceIndices);
399 SubdivisionMesh::FaceToPartBuffer faceToPart(faceCount, 0);
400 for (size_t p = 0, j = 0; p < partCount; p++) {
401 for (int i = 0; i < partFaceCountList[p]; i++, j++) {
402 faceToPart[partFaceIndices[j]] = p;
403 }
404 }
405 if (faceVertexCount.empty()) {
406 return nullptr;
407 }
408 size_t vc = (faceCount == vertCount) ? 0 : vertCount; // make it prefer faces over vertices if they match
409 const moonray::geom::RateCounts rates{partCount, faceCount, vc, vc, faceVaryingCount};
410
411 // Get the vertices, velocities, uvs, normals etc.
412 SubdivisionMesh::VertexBuffer vertices = getVertexData(rdlGeometry, primitiveAttributeTable, rates);
413 if (vertices.empty()) {
414 return nullptr;
415 }
416
417 // per face assignment id
418 LayerAssignmentId layerAssignmentId =
419 createPerFaceAssignmentId(rdlMeshGeometry, rdlLayer, faceCount);
420
421 // get sidedness
422 const int sideType = rdlMeshGeometry->getSideType();
423 const bool singleSided = sideType == scene_rdl2::rdl2::Geometry::SINGLE_SIDED;
424
425 // get resolution
426 const int meshResolution = rdlMeshGeometry->get(attrMeshResolution);
427
428 // get adaptive error (only used when adaptive tessellation got enabled)
429 float adaptiveError = rdlMeshGeometry->get(attrAdaptiveError);
430 // TODO rotation motion blur involves instancing logic that would
431 // break adaptive tessellation right now.
432 // Remove this switching logic once we have instancing supports
433 // adaptive tessellation
434 if (rdlMeshGeometry->get(attrUseRotationMotionBlur)) {
435 adaptiveError = 0.0f;
436 }
437
438 // get subd scheme
439 const int procScheme = rdlMeshGeometry->get(attrSubdScheme);
440 SubdivisionMesh::Scheme scheme = SubdivisionMesh::Scheme::CATMULL_CLARK;
441 if (SubdivisionMesh::isValidScheme(procScheme)) {
442 // Values in the Rdl attribute align with those of the enum, so cast when valid:
443 scheme = static_cast<SubdivisionMesh::Scheme>(procScheme);
444 }
445
446 // get arbitrary data
447 scene_rdl2::rdl2::PrimitiveAttributeFrame primitiveAttributeFrame = static_cast<scene_rdl2::rdl2::PrimitiveAttributeFrame>(
448 rdlMeshGeometry->get(attrPrimitiveAttributeFrame));
449 bool useFirstFrame = (primitiveAttributeFrame != scene_rdl2::rdl2::PrimitiveAttributeFrame::SECOND_MOTION_STEP);
450 bool useSecondFrame = (primitiveAttributeFrame != scene_rdl2::rdl2::PrimitiveAttributeFrame::FIRST_MOTION_STEP);
451 moonray::geom::processArbitraryData(rdlMeshGeometry, attrPrimitiveAttributes, primitiveAttributeTable,
452 rates, useFirstFrame, useSecondFrame);
453
454 // build the primitive
455 std::unique_ptr<SubdivisionMesh> primitive = createSubdivisionMesh(
456 scheme, std::move(faceVertexCount), std::move(indices),
457 std::move(vertices), std::move(layerAssignmentId),
458 std::move(primitiveAttributeTable));
459
460 // set optional subdivision interpolation properties
461 SubdivisionMesh::BoundaryInterpolation subdBoundary =
462 SubdivisionMesh::BoundaryInterpolation::EDGE_AND_CORNER;
463 SubdivisionMesh::FVarLinearInterpolation subdFVarLinear =
464 SubdivisionMesh::FVarLinearInterpolation::CORNERS_ONLY;
465
466 const bool useStudioDefaultSubdInterpolationOptions = false;
467 if (!useStudioDefaultSubdInterpolationOptions) {
468 const int procBoundary = rdlMeshGeometry->get(attrSubdBoundary);
469 if (SubdivisionMesh::isValidBoundaryInterpolation(procBoundary)) {
470 // Values in the Rdl attribute align with those of the enum, so cast when valid:
471 subdBoundary = static_cast<SubdivisionMesh::BoundaryInterpolation>(procBoundary);
472 } else {
473 rdlGeometry->warn("Unknown subd boundary value, defaulting to 'edge and corner'.");
474 }
475
476 const int procFVarLinear = rdlMeshGeometry->get(attrSubdFVarLinear);
477 if (SubdivisionMesh::isValidFVarLinearInterpolation(procFVarLinear)) {
478 // Values in the Rdl attribute align with those of the enum, so cast when valid:
479 subdFVarLinear = static_cast<SubdivisionMesh::FVarLinearInterpolation>(procFVarLinear);
480 } else {
481 rdlGeometry->warn("Unknown subd fvar linear value, defaulting to 'corners only'.");
482 }
483 }
484 primitive->setSubdBoundaryInterpolation(subdBoundary);
485 primitive->setSubdFVarLinearInterpolation(subdFVarLinear);
486
487 // set optional subdivision creases, corners and holes:
488 auto& procCreaseIndices = rdlMeshGeometry->get(attrSubdCreaseIndices);
489 SubdivisionMesh::IndexBuffer creaseIndices(procCreaseIndices.begin(), procCreaseIndices.end());
490
491 auto& procCreaseSharpnesses = rdlMeshGeometry->get(attrSubdCreaseSharpnesses);
492 SubdivisionMesh::SharpnessBuffer creaseSharpnesses(procCreaseSharpnesses.begin(), procCreaseSharpnesses.end());
493
494 auto& procCornerIndices = rdlMeshGeometry->get(attrSubdCornerIndices);
495 SubdivisionMesh::IndexBuffer cornerIndices(procCornerIndices.begin(), procCornerIndices.end());
496
497 auto& procCornerSharpnesses = rdlMeshGeometry->get(attrSubdCornerSharpnesses);
498 SubdivisionMesh::SharpnessBuffer cornerSharpnesses(procCornerSharpnesses.begin(), procCornerSharpnesses.end());
499
500 if (!creaseIndices.empty() && !creaseSharpnesses.empty()) {
501 primitive->setSubdCreases(std::move(creaseIndices), std::move(creaseSharpnesses));
502 }
503 if (!cornerIndices.empty() && !cornerSharpnesses.empty()) {
504 primitive->setSubdCorners(std::move(cornerIndices), std::move(cornerSharpnesses));
505 }
506
507 // set additional primitive attributes
508 primitive->setMeshResolution(meshResolution);
509 primitive->setAdaptiveError(adaptiveError);
510 primitive->setName(sPrimitiveName);
511 primitive->setIsSingleSided(singleSided);
512 primitive->setIsNormalReversed(rdlMeshGeometry->getReverseNormals());
513 primitive->setIsOrientationReversed(rdlMeshGeometry->get(attrOrientation) == ORIENTATION_LEFT_HANDED);
514 primitive->setModifiability(Primitive::Modifiability::DEFORMABLE);
515 primitive->setParts(partCount, std::move(faceToPart));
516 primitive->setCurvedMotionBlurSampleCount(rdlMeshGeometry->get(attrCurvedMotionBlurSampleCount));
517
518 mSubdMesh = primitive.get();
519 return primitive;
520}

523std::unique_ptr<PolygonMesh>
524RdlMeshProcedural::createPolyMesh(
525 const scene_rdl2::rdl2::Geometry* rdlGeometry,
526 const scene_rdl2::rdl2::Layer* rdlLayer)
527{
528 const RdlMeshGeometry* rdlMeshGeometry =
529 static_cast<const RdlMeshGeometry*>(rdlGeometry);
530
531 shading::PrimitiveAttributeTable primitiveAttributeTable;
532
533 // Set the vert per face count
534 auto& procFaceVertexCount = rdlMeshGeometry->get(attrFaceVertexCount);
535 PolygonMesh::FaceVertexCount faceVertexCount(
536 procFaceVertexCount.begin(), procFaceVertexCount.end());
537
538 // Store the vert indices in order of face list to build the mesh
539 auto& procIndices = rdlMeshGeometry->get(attrVertexIndex);
540 PolygonMesh::IndexBuffer indices(procIndices.begin(), procIndices.end());
541 if (indices.empty()) {
542 return nullptr;
543 }
544
545 auto& procVertList = rdlMeshGeometry->get(attrPos0);
546 const size_t vertCount = procVertList.size();
547 const size_t faceCount = faceVertexCount.size();
548 const size_t faceVaryingCount = procIndices.size();
549
550 // Fill in table of faces->parts
551 auto& partFaceCountList = rdlMeshGeometry->get(attrPartFaceCountList);
552 const size_t partCount = partFaceCountList.size();
553 auto& partFaceIndices = rdlMeshGeometry->get(attrPartFaceIndices);
554 PolygonMesh::FaceToPartBuffer faceToPart(faceCount, 0);
555 for (size_t p = 0, j = 0; p < partCount; p++) {
556 for (int i = 0; i < partFaceCountList[p]; i++, j++) {
557 faceToPart[partFaceIndices[j]] = p;
558 }
559 }
560 if (faceVertexCount.empty()) {
561 return nullptr;
562 }
563 const moonray::geom::RateCounts rates{partCount, faceCount, 0, vertCount, faceVaryingCount};
564
565 // Get the vertices, velocities, uvs, normals etc.
566 PolygonMesh::VertexBuffer vertices = getVertexData(rdlGeometry, primitiveAttributeTable, rates);
567 if (vertices.empty()) {
568 return nullptr;
569 }
570
571 // per face assignment id
572 LayerAssignmentId layerAssignmentId =
573 createPerFaceAssignmentId(rdlMeshGeometry, rdlLayer, faceVertexCount.size());
574
575 // get sidedness
576 const int sideType = rdlMeshGeometry->getSideType();
577 const bool singleSided = sideType == scene_rdl2::rdl2::Geometry::SINGLE_SIDED;
578
579 // get resolution
580 const int meshResolution = rdlMeshGeometry->get(attrMeshResolution);
581
582 // get adaptive error (only used when adaptive tessellation got enabled)
583 float adaptiveError = rdlMeshGeometry->get(attrAdaptiveError);
584
585 // get arbitrary data
586 scene_rdl2::rdl2::PrimitiveAttributeFrame primitiveAttributeFrame = static_cast<scene_rdl2::rdl2::PrimitiveAttributeFrame>(
587 rdlMeshGeometry->get(attrPrimitiveAttributeFrame));
588 bool useFirstFrame = (primitiveAttributeFrame != scene_rdl2::rdl2::PrimitiveAttributeFrame::SECOND_MOTION_STEP);
589 bool useSecondFrame = (primitiveAttributeFrame != scene_rdl2::rdl2::PrimitiveAttributeFrame::FIRST_MOTION_STEP);
590 moonray::geom::processArbitraryData(rdlMeshGeometry, attrPrimitiveAttributes, primitiveAttributeTable,
591 rates, useFirstFrame, useSecondFrame);
592
593 removeUnassignedFaces(rdlLayer, layerAssignmentId, faceToPart,
594 faceVertexCount, indices, &primitiveAttributeTable);
595
596 // check if mesh is still valid
597 if (faceVertexCount.empty() || indices.empty()) {
598 // the mesh doesn't have any assigned materials,
599 // skip generating the primitive
600 return nullptr;
601 }
602
603 // build the primitive
604 std::unique_ptr<PolygonMesh> primitive = createPolygonMesh(
605 std::move(faceVertexCount), std::move(indices),
606 std::move(vertices), std::move(layerAssignmentId),
607 std::move(primitiveAttributeTable));
608
609 // set additional primitive attributes
610 primitive->setMeshResolution(meshResolution);
611 primitive->setAdaptiveError(adaptiveError);
612 primitive->setName(sPrimitiveName);
613 primitive->setIsSingleSided(singleSided);
614 primitive->setIsNormalReversed(rdlMeshGeometry->getReverseNormals());
615 primitive->setIsOrientationReversed(rdlMeshGeometry->get(attrOrientation) == ORIENTATION_LEFT_HANDED);
616 primitive->setParts(partCount, std::move(faceToPart));
617 primitive->setSmoothNormal(rdlMeshGeometry->get(attrSmoothNormal));
618 primitive->setCurvedMotionBlurSampleCount(rdlMeshGeometry->get(attrCurvedMotionBlurSampleCount));
619 mPolygonMesh = primitive.get();
620 return primitive;
621}
