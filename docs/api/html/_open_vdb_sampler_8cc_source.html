25openvdb::GridBase::Ptr
26readGrid(const std::string &fileName, const std::string &gridName,
27 std::string &foundGrid, std::string &errorMsg)
28{
29 openvdb::GridBase::Ptr grid;
30
31 if (fileName.empty()) {
32 errorMsg = "no .vdb filename specified";
33 return grid;
34 }
35
36 openvdb::initialize();
37 openvdb::io::File file(fileName);
38 try {
39 file.open();
40 } catch (const openvdb::IoError &e) {
41 errorMsg = e.what();
42 return grid;
43 }
44
45 std::string whichGrid = gridName;
46
47 if (gridName.empty()) {
48 // use first grid in the file
49 openvdb::io::File::NameIterator firstName = file.beginName();
50 if (firstName != file.endName()) {
51 whichGrid = firstName.gridName();
52 }
53 } else if (!file.hasGrid(gridName)) {
54 std::ostringstream os;
55 os << "'" << fileName << "' does not contain a grid named '" << gridName << "'";
56 file.close();
57 errorMsg = os.str();
58 return grid;
59 }
60
61 grid = file.readGrid(whichGrid);
62 file.close();
63
64 foundGrid = whichGrid;
65
66 return grid;
67}

82bool
83OpenVdbSampler::initialize(const std::string &fileName,
84 const std::string &gridName,
85 const scene_rdl2::math::Mat4d *render2world,
86 const scene_rdl2::math::Mat4d *grid2world,
87 std::string &errorMsg)
88{
89 mIsInitialized = false;
90
91 // try/catch to catch exceptions that could be thrown from
92 // the openvdb library itself, to prevent users of this class
93 // from needing to try/catch to prevent moonray from crashing.
94 try {
95 std::string foundGrid;
96 mGrid = readGrid(fileName, gridName, foundGrid, errorMsg);
97
98 if (!mGrid) {
99 // errorMsg is already set
100 return false;
101 }
102
103 mValueType = mGrid->valueType();
104
105 // Append any additional transform provided by the user
106 if (grid2world) {
107 const scene_rdl2::math::Vec4d &r0 = grid2world->row0();
108 const scene_rdl2::math::Vec4d &r1 = grid2world->row1();
109 const scene_rdl2::math::Vec4d &r2 = grid2world->row2();
110 const scene_rdl2::math::Vec4d &r3 = grid2world->row3();
111 const openvdb::Mat4R xform(
112 r0[0], r0[1], r0[2], r0[3],
113 r1[0], r1[1], r1[2], r1[3],
114 r2[0], r2[1], r2[2], r2[3],
115 r3[0], r3[1], r3[2], r3[3]
116 );
117 mGrid->transform().postMult(xform);
118 }
119
120 // create the appropriate TypedVDBSampler for this grid type
121 if (mGrid->isType<openvdb::FloatGrid>()) {
122 openvdb::FloatGrid::ConstPtr g = openvdb::gridConstPtrCast<openvdb::FloatGrid>(mGrid);
123 mSampler = fauxstd::make_unique<TypedVDBSampler<openvdb::FloatGrid>>(*g);
124 } else if (mGrid->isType<openvdb::VectorGrid>()) {
125 openvdb::VectorGrid::ConstPtr g = openvdb::gridConstPtrCast<openvdb::VectorGrid>(mGrid);
126 mSampler = fauxstd::make_unique<TypedVDBSampler<openvdb::VectorGrid>>(*g);
127 } else {
128 // TODO: provide more information to user
129 // eg. filename, gridname, which gridtype is it that is unsupported...
130 errorMsg = "unsupported grid type";
131 return false;
132 }
133
134 mFileName = fileName;
135 mGridName = foundGrid;
136 mR2W = render2world;
137 mIsInitialized = true;
138 } catch (const std::exception &e) {
139 errorMsg = e.what();
140 }
141
142 return mIsInitialized;
143}

145bool
146OpenVdbSampler::getIsActive(shading::TLState* tls,
147 const scene_rdl2::math::Vec3f &pos) const
148{
149 MNRY_ASSERT_REQUIRE(mIsInitialized);
150
151 openvdb::Vec3d p;
152 if (mR2W) {
153 scene_rdl2::math::Vec3f pw = transformPoint(*mR2W, pos);
154 p[0] = pw[0];
155 p[1] = pw[1];
156 p[2] = pw[2];
157 } else {
158 p[0] = pos[0];
159 p[1] = pos[1];
160 p[2] = pos[2];
161 }
162
163 return mSampler->getIsActive(tls->mThreadIdx, p);
164}

166scene_rdl2::math::Color
167OpenVdbSampler::sample(shading::TLState* tls,
168 const scene_rdl2::math::Vec3f &pos,
169 const Interpolation interp) const
170{
171 MNRY_ASSERT_REQUIRE(mIsInitialized);
172
173 scene_rdl2::math::Color result(0.f, 0.f, 0.f);
174
175 // convert to VDBSampler::Interpolation
176 const VDBSampler::Interpolation interpolation =
177 static_cast<VDBSampler::Interpolation>(interp);
178
179 openvdb::Vec3d p;
180 if (mR2W) {
181 scene_rdl2::math::Vec3f pw = transformPoint(*mR2W, pos);
182 p[0] = pw[0];
183 p[1] = pw[1];
184 p[2] = pw[2];
185 } else {
186 p[0] = pos[0];
187 p[1] = pos[1];
188 p[2] = pos[2];
189 }
190
191 // cast to appropriate type and sample
192 if (mGrid->isType<openvdb::FloatGrid>()) {
193 TypedVDBSampler<openvdb::FloatGrid>* sampler =
194 static_cast<TypedVDBSampler<openvdb::FloatGrid>*>(mSampler.get());
195 const float val = sampler->sample(tls->mThreadIdx, p, interpolation);
196 result = scene_rdl2::math::Color(val, val, val);
197 } else if (mGrid->isType<openvdb::VectorGrid>()) {
198 TypedVDBSampler<openvdb::VectorGrid>* sampler =
199 static_cast<TypedVDBSampler<openvdb::VectorGrid>*>(mSampler.get());
200 const openvdb::Vec3f val = sampler->sample(tls->mThreadIdx, p, interpolation);
201 result = scene_rdl2::math::Color(val[0], val[1], val[2]);
202 } else {
203 // Programming error. This object should have thrown an
204 // exception in the ctor if the grid type is unsupported
205 MNRY_ASSERT_REQUIRE(false, "only FloatGrid and VectorGrid types currently supported");
206 }
207
208 return result;
209}
