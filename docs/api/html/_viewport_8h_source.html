12/**
13 * Viewports represent a rectangular region in pixel space. They may contain
14 * positive or negative pixel coordinates, but the min X/Y will always be <=
15 * the max X/Y. There are 2 varieties defined here: ClosedViewport uses closed
16 * intervals to represent each axis, and HalfOpenViewport uses half-open
17 * intervals to represent each axis.
18 */
19
20namespace scene_rdl2 {
21namespace math {

23struct BaseViewport
24{
25 /**
26 * Default construct a viewport, whose region is 1x1 at the origin.
27 */
28 finline BaseViewport() :
29 mMinX(0),
30 mMinY(0),
31 mMaxX(0),
32 mMaxY(0) {}
33
34 /**
35 * Construct a viewport from individual min/max X/Y coordinates. The
36 * viewport guarantees that the min is actually the min and the max is
37 * actually the max.
38 *
39 * @param minX The minimum X coordinate.
40 * @param minY The minimum Y coordinate.
41 * @param maxX The maximum X coordinate.
42 * @param maxY The maximum Y coordinate.
43 */
44 finline BaseViewport(int minX, int minY, int maxX, int maxY) :
45 mMinX(math::min(minX, maxX)),
46 mMinY(math::min(minY, maxY)),
47 mMaxX(math::max(minX, maxX)),
48 mMaxY(math::max(minY, maxY)) {}
49
50 /**
51 * Construct a viewport from min and max vectors. The viewport guarantees
52 * that the min is actually the min and the max is actually the max.
53 *
54 * @param min The minimum 2D point.
55 * @param max The maximum 2D point.
56 */
57 finline BaseViewport(math::Vec2i min, math::Vec2i max) :
58 mMinX(math::min(min.x, max.x)),
59 mMinY(math::min(min.y, max.y)),
60 mMaxX(math::max(min.x, max.x)),
61 mMaxY(math::max(min.y, max.y)) {}
62
63 /**
64 * Construct a viewport from an array of 4 integers, in minX, minY, maxX,
65 * maxY order. The viewport guarantees that the min is actually the min
66 * and the max is actually the max.
67 *
68 * @param region An array of 4 integers in minX, minY, maxX, maxY order.
69 */
70 finline explicit BaseViewport(int region[4]) :
71 mMinX(math::min(region[0], region[2])),
72 mMinY(math::min(region[1], region[3])),
73 mMaxX(math::max(region[0], region[2])),
74 mMaxY(math::max(region[1], region[3])) {}
75
76 /// Returns the viewport min as a 2D point.
77 finline math::Vec2i min() const
78 {
79 return math::Vec2i(mMinX, mMinY);
80 }
81
82 /// Returns the viewport max as a 2D point.
83 finline math::Vec2i max() const
84 {
85 return math::Vec2i(mMaxX, mMaxY);
86 }
87
88 int mMinX;
89 int mMinY;
90 int mMaxX;
91 int mMaxY;
92
93protected:
94 finline ~BaseViewport() {}
95};

97finline bool operator==(const BaseViewport& a, const BaseViewport& b)
98{
99 return a.mMinX == b.mMinX &&
100 a.mMinY == b.mMinY &&
101 a.mMaxX == b.mMaxX &&
102 a.mMaxY == b.mMaxY;

110/*
111 * The viewport min and max are both inclusive. In other words, a viewport
112 * with a min X of 0 and a max X of 9 has a width of 10 pixels. While half-open
113 * intervals are usually more convenient, this is to maintain consistency
114 * with how the studio has dealt with viewports in the past. Just make sure to
115 * use <= instead of < when iterating between the min and max.
116 *
117 * For code which assumes half open invervals, see ViewportHalfOpen
118 */
119struct Viewport : public BaseViewport
120{
121 finline Viewport() : BaseViewport() {}
122
123 finline Viewport(int minX, int minY, int maxX, int maxY) :
124 BaseViewport(minX, minY, maxX, maxY) {}
125
126 finline Viewport(math::Vec2i min, math::Vec2i max) :
127 BaseViewport(min, max) {}
128
129 finline Viewport(int region[4]) :
130 BaseViewport(region) {}
131
132 /// Returns the width of the viewport, in pixels.
133 finline unsigned width() const
134 {
135 // Guaranteed to be positive since we verified that max > min during
136 // construction.
137 return mMaxX - mMinX + 1;
138 }
139
140 /// Returns the height of the viewport, in pixels.
141 finline unsigned height() const
142 {
143 // Guaranteed to be positive since we verified that max > min during
144 // construction.
145 return mMaxY - mMinY + 1;
146 }
147
148 /// Returns true if the given coordinate (x, y) is included within the
149 /// bounds of the viewport.
150 finline bool contains(int x, int y) const
151 {
152 return (x >= mMinX) && (x <= mMaxX) && (y >= mMinY) && (y <= mMaxY);
153 }
154};

157/**
158 * Half open scheme.
159 *
160 * A viewport with a width of 10 starting at 0 would have a minX of 0, and a
161 * maxX of 10 under this scheme.
162 */
163struct HalfOpenViewport : public BaseViewport
164{
165 finline HalfOpenViewport() : BaseViewport() {}
166
167 finline HalfOpenViewport(int minX, int minY, int maxX, int maxY) :
168 BaseViewport(minX, minY, maxX, maxY) {}
169
170 finline HalfOpenViewport(math::Vec2i min, math::Vec2i max) :
171 BaseViewport(min, max) {}
172
173 finline HalfOpenViewport(int region[4]) :
174 BaseViewport(region) {}
175
176 finline HalfOpenViewport(const std::vector<int>& window, float invRes)
177 {
178 int width = int(float(window[2] - window[0]) * invRes);
179 int height = int(float(window[3] - window[1]) * invRes);
180 mMinX = static_cast<int>(static_cast<float>(window[0]) * invRes);
181 mMinY = static_cast<int>(static_cast<float>(window[1]) * invRes);
182 mMaxX = mMinX + width;
183 mMaxY = mMinY + height;
184 }
185
186 /// Returns the width of the viewport, in pixels.
187 finline unsigned width() const
188 {
189 // Guaranteed to be positive since we verified that max > min during
190 // construction.
191 return mMaxX - mMinX;
192 }
193
194 /// Returns the height of the viewport, in pixels.
195 finline unsigned height() const
196 {
197 // Guaranteed to be positive since we verified that max > min during
198 // construction.
199 return mMaxY - mMinY;
200 }
201
202 /// Returns true if the given coordinate (x, y) is included within the
203 /// bounds of the viewport.
204 finline bool contains(int x, int y) const
205 {
206 return (x >= mMinX) && (x < mMaxX) && (y >= mMinY) && (y < mMaxY);
207 }
208};
