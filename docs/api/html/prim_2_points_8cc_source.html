102template <typename AttributeType> bool
103getAttribute(const Attributes* attributes,
104 const TypedAttributeKey<AttributeType> key,
105 const int primID,
106 AttributeType& value)
107{
108 if (!attributes->isSupported(key)) {
109 return false;
110 }
111
112 // Data for points is typically stored as constant
113 // (one value for all points) or varying (one value
114 // per point).
115 AttributeRate rate = attributes->getRate(key);
116 switch (rate) {
117 case RATE_CONSTANT:
118 value = attributes->getConstant(key);
119 break;
120 case RATE_VARYING:
121 value = attributes->getVarying(key, primID);
122 break;
123 default:
124 return false;
125 }
126
127 return true;
128}

130void
131Points::postIntersect(mcrt_common::ThreadLocalState& tls,
132 const scene_rdl2::rdl2::Layer* pRdlLayer, const mcrt_common::Ray& ray,
133 Intersection& intersection) const
134{
135 int assignmentId =
136 mLayerAssignmentId.getType() == LayerAssignmentId::Type::CONSTANT ?
137 mLayerAssignmentId.getConstId() :
138 mLayerAssignmentId.getVaryingId()[ray.primID];
139 intersection.setLayerAssignments(assignmentId, pRdlLayer);
140
141 const scene_rdl2::rdl2::Material* material = intersection.getMaterial();
142 const AttributeTable *table =
143 material->get<shading::RootShader>().getPrimitiveAttributeTable();
144 intersection.setTable(&tls.mArena, table);
145 intersection.setIds(ray.primID, 0, 0);
146 overrideInstanceAttrs(ray, intersection);
147 const Attributes* attributes = getAttributes();
148
149 // For each of the attributes below we first check the
150 // intersection since it may have been set from the
151 // instancer points which should override any local
152 // attributes.
153 Vec3f uv;
154 Vec2f st(1.0f, 1.0f);
155 if (intersection.isProvided(StandardAttributes::sUv)) {
156 uv = intersection.getAttribute<Vec3f>(StandardAttributes::sUv);
157 st.x = uv.x;
158 st.y = uv.y;
159 } else if (getAttribute<Vec3f>(attributes,
160 StandardAttributes::sUv,
161 ray.primID,
162 uv)) {
163 st.x = uv.x;
164 st.y = uv.y;
165 }
166
167 Vec3f Ng = normalize(ray.Ng);
168 bool hasNormal;
169 if (intersection.isProvided(StandardAttributes::sNormal)) {
170 Ng = intersection.getAttribute<Vec3f>(StandardAttributes::sNormal);
171 Ng = normalize(Ng);
172 hasNormal = true;
173 } else {
174 hasNormal = getAttribute<Vec3f>(attributes,
175 StandardAttributes::sNormal,
176 ray.primID,
177 Ng);
178 if (hasNormal) {
179 Ng = normalize(Ng);
180 }
181 }
182
183 Vec3f dPds = scene_rdl2::math::zero;
184 bool hasDPds;
185 if (intersection.isProvided(StandardAttributes::sdPds)) {
186 dPds = intersection.getAttribute<Vec3f>(StandardAttributes::sdPds);
187 hasDPds = true;
188 } else {
189 hasDPds = getAttribute<Vec3f>(attributes,
190 StandardAttributes::sdPds,
191 ray.primID,
192 dPds);
193 }
194
195 Vec3f dPdt = scene_rdl2::math::zero;
196 bool hasDPdt;
197 if (intersection.isProvided(StandardAttributes::sdPdt)) {
198 dPdt = intersection.getAttribute<Vec3f>(StandardAttributes::sdPdt);
199 hasDPdt = true;
200 } else {
201 hasDPdt = getAttribute<Vec3f>(attributes,
202 StandardAttributes::sdPdt,
203 ray.primID,
204 dPdt);
205 }
206
207 const bool hasDerivatives = hasDPds && hasDPdt;
208 const bool isFlatPoint = hasNormal && hasDerivatives;
209
210 // The geometric and shading normal use the value of the StandardAttribute
211 // if available and that means the point is not being treated as a real sphere.
212 // The "P" is on the surface of the sphere, but the "Ng" is the normal of a
213 // disk, so it's a weird hybrid geometry. This is ok because points are supposed
214 // to be small in the scene, so any inconsistency should not be noticeable.
215 intersection.setDifferentialGeometry(Ng, // geometric normal
216 Ng, // shading normal
217 st,
218 dPds,
219 dPdt,
220 hasDerivatives,
221 isFlatPoint);
222
223 PointsInterpolator interpolator(getAttributes(), ray.time, ray.primID);
224 intersection.setRequiredAttributes(interpolator);
225
226 const scene_rdl2::rdl2::Geometry* geometry = intersection.getGeometryObject();
227 MNRY_ASSERT(geometry != nullptr);
228 intersection.setEpsilonHint( geometry->getRayEpsilon() );
229
230 // wireframe AOV is blank
231 if (table->requests(StandardAttributes::sNumPolyVertices)) {
232 intersection.setAttribute(StandardAttributes::sNumPolyVertices, 0);
233 intersection.setAttribute(StandardAttributes::sPolyVertexType,
234 static_cast<int>(StandardAttributes::POLYVERTEX_TYPE_POLYGON));
235
236 }
237
238 // motion vectors
239 if (table->requests(StandardAttributes::sMotion)) {
240 Vec3f pos0, pos1;
241 const Vec3f *pos1Ptr = nullptr;
242 if (isMotionBlurOn()) {
243 pos0 = lerp(mPosition(ray.primID, 0), mPosition(ray.primID, 1), ray.getTime() - sHalfDt);
244 pos1 = lerp(mPosition(ray.primID, 0), mPosition(ray.primID, 1), ray.getTime() + sHalfDt);
245 pos1Ptr = &pos1;
246 } else {
247 pos0 = mPosition(ray.primID, 0);
248 pos1Ptr = nullptr;
249 }
250
251 // Motion vectors only support a single instance level, hence we only care
252 // about ray.instance0.
253 const Instance *instance = (ray.isInstanceHit())?
254 static_cast<const Instance *>(ray.ext.instance0OrLight) : nullptr;
255 const Vec3f motion = computePrimitiveMotion(pos0, pos1Ptr, ray.getTime(), instance);
256 intersection.setAttribute(StandardAttributes::sMotion, motion);
257 }
258}

260BBox3f
261Points::computeAABB() const
262{
263 if (mPosition.empty()) {
264 return BBox3f(scene_rdl2::math::zero);
265 }
266 BBox3f result(mPosition(0));
267 float maxRadius = 0.0f;
268 size_t motionSampleCount = getMotionSamplesCount();
269 for (size_t v = 0; v < mPosition.size(); ++v) {
270 maxRadius = std::max(maxRadius, mRadius[v]);
271 for (size_t t = 0; t < motionSampleCount; ++t) {
272 result.extend(mPosition(v, t));
273 }
274 }
275 result.lower -= Vec3f(maxRadius);
276 result.upper += Vec3f(maxRadius);
277 return result;
278}

280BBox3f
281Points::computeAABBAtTimeStep(int timeStep) const
282{
283 if (mPosition.empty()) {
284 return BBox3f(scene_rdl2::math::zero);

289 for (size_t v = 0; v < mPosition.size(); ++v) {
290 maxRadius = std::max(maxRadius, mRadius[v]);
291 result.extend(mPosition(v, timeStep));
292 }
293 result.lower -= Vec3f(maxRadius);
294 result.upper += Vec3f(maxRadius);
295 return result;
296}

298static void
299boundsFunc(const RTCBoundsFunctionArguments* args)
300{
301 const BVHUserData* userData = (const BVHUserData*)args->geometryUserPtr;
302 const Points* points = (const Points*)userData->mPrimitive;
303 RTCBounds* output = args->bounds_o;
304 size_t item = args->primID;
305 Vec3f pMin = points->getVertexBuffer()(item);
306 Vec3f pMax = pMin;
307 float r = points->getRadiusBuffer()[item];
308 if (points->isMotionBlurOn()) {
309 for (size_t i = 1; i < points->getVertexBuffer().get_time_steps(); i++) {
310 const Vec3f p = points->getVertexBuffer()(item, i);
311 pMin = min(pMin, p);
312 pMax = max(pMax, p);
313 }
314 }
315 output->lower_x = pMin.x - r;
316 output->lower_y = pMin.y - r;
317 output->lower_z = pMin.z - r;
318 output->upper_x = pMax.x + r;
319 output->upper_y = pMax.y + r;
320 output->upper_z = pMax.z + r;
321}

329static finline Vec3fa
330getPosition(const Points* points, size_t item, float rayTime)
331{
332 Vec3f p;
333 if (points->isMotionBlurOn()) {
334 const int numSegments = points->getVertexBuffer().get_time_steps() - 1;
335 const float clampedRayTime = scene_rdl2::math::clamp(rayTime, 0.0f, 1.0f);
336 const float idxPlusT = clampedRayTime * (float)numSegments;
337 const int idx0 = (int)floor(idxPlusT);
338 const int idx1 = idx0 + 1;
339 const float t = idxPlusT - (float)idx0;
340 p = lerp(points->getVertexBuffer()(item, idx0),
341 points->getVertexBuffer()(item, idx1),
342 t);
343 } else {
344 p = points->getVertexBuffer()(item);
345 }
346 return Vec3fa(p, 0.f);
347}

349static void
350intersectFunc(const RTCIntersectFunctionNArguments* args)
351{
352 // A common use case for us is very tiny spheres relative to
353 // ray length. This makes using the "standard" ray/sphere
354 // intersection problematic as it is highly suspectible to
355 // catstrophic cancellation. This implemtnation avoids this
356 // problem nicely, and was explained by Mike Day in an email
357 // thread. I'm pasting in some of the more salient points
358 // from that disucssion.
359 //
360 // Here is the somewhate paraphrased "standard" function
361 //
362 // // o = ray origin
363 // // d = ray direction
364 // // p = sphere centre
365 // // r = sphere radius
366 //
367 // const Vec3fa v = o - p;
368 //
369 // const float A = dot(d, d);
370 // const float B = 2.0f * dot(v, d);
371 // const float C = dot(v, v) - r * r;
372 // const float D = B * B - 4.0f * A * C;
373 // if (D < 0.0f) {
374 // return;
375 // }
376 //
377 // const float Q = scene_rdl2::math::sqrt(D);
378 // const float rcpA = scene_rdl2::math::rcp(A);
379 // const float t0 = 0.5f * rcpA * (-B - Q);
380 // const float t1 = 0.5f * rcpA * (-B + Q);
381 //
382 // if ((ray.tnear < t0) && (t0 < ray.tfar)) {
383 // ... // set other ray members
384 // ray.Ng = o + t0 * d - p;
385 // }
386 // if ((ray.tnear < t1) && (t1 < ray.tfar)) {
387 // ... // set other ray members
388 // ray.Ng = o + t1 * d - p;
389 // }
390 //
391 // This is all quite standard. Use the input values to set up
392 // the coefficients A, B, C of a quadratic equation,
393 // take the discriminant D, quit if D < 0
394 // otherwise go on to solve the quadratic equation and compute the other
395 // necessary intersection results.
396 //
397 // Let's see where the precision losses are occurring.
398 // The first source of precision loss is in computing the discriminant,
399 //
400 // const float D = B * B - 4.0f * A * C;
401 //
402 // Plugging in the expressions for A, B and C, this is computing
403 // 4 ( (v.d)^2 - (d.d)(v.v - r^2) ). So it's taking the difference of
404 // (v.d)^2 and (d.d)(v.v - r^2). When v and d are in very nearly the same
405 // direction, as they will be when the ray points at a distant tiny
406 // sphere, their dot product v.d is almost equal to the product of the 2
407 // lengths, |v| * |d|, which is to say (v.d)^2 is close to |v|^2 * |d|^2.
408 // And with r very small, the term (v.v - r^2) is close to v.v, which means
409 // (d.d)(v.v - r^2) is also close to |v|^2 * |d|^2. So we've got 2 numbers
410 // almost equal to each other, and we're taking their difference.
411 // Alarm bells should go off at this point!
412 // Catastrophic cancellation ensues!
413 //
414 // The result is that D will retain very few significant bits, i.e.
415 // instead of having a mantissa which can vary over any of the possible
416 // 2^23 values, it may only vary over a small handful of values.
417 // In extreme cases, as with very small spheres indeed,
418 // the cancellation is absolutely catastrophic
419 // and we'll end up with D = 0, so all information about the structure
420 // of the sphere has effectively been lost. It has become a black hole.
421 //
422 // The point to note here is that in floating point the dot product has
423 // very poor resolution for vectors which are nearly parallel. It varies
424 // with the cosine of the angle between the vectors - the cosine is 1
425 // when the angle is zero (actually -1 in the code, because v and d point
426 // in opposite directions), and because the gradient flattens to zero here,
427 // very little change occurs as you vary the vector directions by small
428 // amounts in this vicinity, so you can expect very little change in the
429 // mantissa and hence a very 'steppy' value - lots of quantisation
430 // instead of a smoothly changing value.
431 //
432 // We can save the day using the cross product instead of the dot product.
433 // The cross product varies with the sine of the angle between the vectors,
434 // and sine has maximum gradient when the angle is zero, so we can expect it
435 // to have good resolution in floating point arithmetic in situations where
436 // the vectors are nearly parallel. Here's how to change the expression for
437 // D to make this possible:
438 //
439 // D = B^2 - 4AC
440 // = 4 ( (v.d)^2 - (d.d)(v.v - r^2) )
441 // = 4 ( (d.d)r^2 - ((d.d)(v.v) - (v.d)^2) )
442 // = 4 ( (d.d)r^2 - (d x v)^2 )
443 //
444 // The last step here is just a consequence of the familiar
445 // identity sin^2(theta) + cos^2(theta) = 1.
446 //
447 // In this revised version for D, we've replaced the difference of 2
448 // nearly equal terms (v.d)^2 and (d.d)(v.v) by a single, more accurately
449 // computed term. We can expect the result to have correspondingly
450 // improved resolution in floating point.
451 //
452 // Even with this improvement, things still go badly wrong even for
453 // spheres with a modicum of smallness. In this case
454 // the culprit is the line
455 //
456 // ray.Ng = o + t0 * d - p;
457 //
458 // This involves more catastrophic cancellation for small radii.
459 // It's computing the intersection point, o + t0*d,
460 // and then subtracting off it a very nearby point, p.
461 // The closeness of these 2 points is the cause of the trouble.
462 // And changing the order of summation won't help, since in general we
463 // can expect all 3 of the vectors o, t0*d, and p to be large compared to
464 // the result. To fix things, let's examine the expression for the normal
465 // more closely:
466 //
467 // o + t0 * d - p = v + t0 * d
468 // = v + d * (-B - Q) / (2A)
469 // = v + d * (-2v.d - Q) / (2d.d)
470 // = ( (d.d)v - (v.d)d - (Q/2)d ) / (d.d)
471 // = ( d x (v x d) - (Q/2)d ) / (d.d)
472 //
473 // The last step uses a standard identity from vector algebra.
474 // So once again, we've taken a dot-product-type expression involving
475 // the difference of (vector) quantities which will be very similar
476 // for nearly parallel vectors, (d.d)v - (v.d)d, and replaced it with a
477 // cross-product-type expression, d x (d x v), which will retain
478 // much greater accuracy in this situation.
479
480 unsigned int N = args->N;
481 unsigned int primID = args->primID;
482 const BVHUserData* userData = (const BVHUserData*)args->geometryUserPtr;
483 const Points* points = (const Points*)userData->mPrimitive;
484
485 float r = points->getRadiusBuffer()[primID];
486 float r2 = r * r;
487
488 if (N == 1) {
489 RTCRayHit& rayhit = *((RTCRayHit*)args->rayhit);
490 RTCRay& ray = rayhit.ray;
491 RTCHit& hit = rayhit.hit;
492
493 Vec3fa p = getPosition(points, primID, ray.time);
494 const Vec3fa o(ray.org_x, ray.org_y, ray.org_z, 0.f);
495 const Vec3fa d(ray.dir_x, ray.dir_y, ray.dir_z, 0.f);
496 const Vec3fa u = p - o;
497 if (dot(d, u) < 0.0f) {
498 // ray is travelling away from sphere centre
499 return;
500 }
501 if (dot(u, u) < r2) {
502 // ray origin is inside sphere
503 return;
504 }
505 const Vec3fa v = cross(d, u);
506 float v2 = dot(v, v);
507 float d2 = dot(d, d);
508 float r2d2 = r2 * d2;
509 float D = r2d2 - v2;
510 if (D < 0.0f) {
511 // no intersections
512 return;
513 }
514 float s = scene_rdl2::math::sqrt(D);
515 const Vec3fa w = cross(d, v);
516 const Vec3fa rn = scene_rdl2::math::rcp(d2) * (w - s * d);
517 float t = length(u + rn) * scene_rdl2::math::rsqrt(d2);
518 if (ray.tnear < t && t < ray.tfar) {
519 ray.tfar = t;
520 hit.instID[0] = args->context->instID[0];
521 hit.geomID = points->getGeomID();
522 hit.primID = primID;
523 hit.u = 0.0f;
524 hit.v = 0.0f;
525 Vec3fa Ng = scene_rdl2::math::rcp(r) * rn;
526 hit.Ng_x = Ng.x;
527 hit.Ng_y = Ng.y;
528 hit.Ng_z = Ng.z;
529 }
530 } else {
531 int* valid = (int*)args->valid;
532 RTCRayHitN* rayhit = (RTCRayHitN*)args->rayhit;
533 RTCRayN* rays = RTCRayHitN_RayN(rayhit, N);
534 RTCHitN* hits = RTCRayHitN_HitN(rayhit, N);
535 for (unsigned int index = 0; index < N; ++index) {
536 if (valid[index] == 0) {
537 continue;
538 }
539 Vec3fa p = getPosition(points, primID,
540 RTCRayN_time(rays, N, index));
541 const Vec3fa o(
542 RTCRayN_org_x(rays, N, index),
543 RTCRayN_org_y(rays, N, index),
544 RTCRayN_org_z(rays, N, index), 0.f);
545 const Vec3fa d(
546 RTCRayN_dir_x(rays, N, index),
547 RTCRayN_dir_y(rays, N, index),
548 RTCRayN_dir_z(rays, N, index), 0.f);
549 float& rayTnear = RTCRayN_tnear(rays, N, index);
550 float& rayTfar = RTCRayN_tfar(rays, N, index);
551 const Vec3fa u = p - o;
552 if (dot(d, u) < 0.0f) {
553 // ray is travelling away from sphere centre
554 continue;
555 }
556 if (dot(u, u) < r2) {
557 // ray origin is inside sphere
558 continue;
559 }
560 const Vec3fa v = cross(d, u);
561 float v2 = dot(v, v);
562 float d2 = dot(d, d);
563 float r2d2 = r2 * d2;
564 float D = r2d2 - v2;
565 if (D < 0.0f) {
566 // no intersections
567 continue;
568 }
569 float s = scene_rdl2::math::sqrt(D);
570 const Vec3fa w = cross(d, v);
571 const Vec3fa rn = scene_rdl2::math::rcp(d2) * (w - s * d);
572 float t = length(u + rn) * scene_rdl2::math::rsqrt(d2);
573 if (rayTnear < t && t < rayTfar) {
574 rayTfar = t;
575 RTCHitN_instID(hits, N, index, 0) = args->context->instID[0];
576 RTCHitN_geomID(hits, N, index) = points->getGeomID();
577 RTCHitN_primID(hits, N, index) = primID;
578 RTCHitN_u(hits, N, index) = 0.0f;
579 RTCHitN_v(hits, N, index) = 0.0f;
580 Vec3fa Ng = scene_rdl2::math::rcp(r) * rn;
581 RTCHitN_Ng_x(hits, N, index) = Ng.x;
582 RTCHitN_Ng_y(hits, N, index) = Ng.y;
583 RTCHitN_Ng_z(hits, N, index) = Ng.z;
584 }
585 }

595static void
596occludedFunc(const RTCOccludedFunctionNArguments* args)
597{
598 unsigned int N = args->N;
599 unsigned int primID = args->primID;
600 const BVHUserData* userData = (const BVHUserData*)args->geometryUserPtr;
601 const Points* points = (const Points*)userData->mPrimitive;
602
603 float r = points->getRadiusBuffer()[primID];
604 float r2 = r * r;
605
606 if (N == 1) {
607 RTCRay& ray = *((RTCRay*)args->ray);
608 Vec3fa p = getPosition(points, primID, ray.time);
609 const Vec3fa o(ray.org_x, ray.org_y, ray.org_z, 0.f);
610 const Vec3fa d(ray.dir_x, ray.dir_y, ray.dir_z, 0.f);
611 const Vec3fa u = p - o;
612 if (dot(d, u) < 0.0f) {
613 // ray is travelling away from sphere centre
614 return;
615 }
616 if (dot(u, u) < r2) {
617 // ray origin is inside sphere
618 return;
619 }
620 const Vec3fa v = cross(d, u);
621 float v2 = dot(v, v);
622 float d2 = dot(d, d);
623 float r2d2 = r2 * d2;
624 float D = r2d2 - v2;
625 if (D < 0.0f) {
626 // no intersections
627 return;
628 }
629 float s = scene_rdl2::math::sqrt(D);
630 const Vec3fa w = cross(d, v);
631 const Vec3fa rn = scene_rdl2::math::rcp(d2) * (w - s * d);
632 float t = length(u + rn) * scene_rdl2::math::rsqrt(d2);
633 // TODO call occlude filter functions
634 // mark the tfar negative is the official signal
635 // for embree that the ray is occluded
636 if (ray.tnear < t && t < ray.tfar) {
637 ray.tfar = -FLT_MAX;
638 }
639 } else {
640 int* valid = (int*)args->valid;
641 RTCRayN* rays = args->ray;
642 for (unsigned int index = 0; index < N; ++index) {
643 if (valid[index] == 0) {
644 continue;
645 }
646 Vec3fa p = getPosition(points, primID,
647 RTCRayN_time(rays, N, index));
648 const Vec3fa o(
649 RTCRayN_org_x(rays, N, index),
650 RTCRayN_org_y(rays, N, index),
651 RTCRayN_org_z(rays, N, index), 0.f);
652 const Vec3fa d(
653 RTCRayN_dir_x(rays, N, index),
654 RTCRayN_dir_y(rays, N, index),
655 RTCRayN_dir_z(rays, N, index), 0.f);
656 float& rayTnear = RTCRayN_tnear(rays, N, index);
657 float& rayTfar = RTCRayN_tfar(rays, N, index);
658 const Vec3fa u = p - o;
659 if (dot(d, u) < 0.0f) {
660 // ray is travelling away from sphere centre
661 continue;
662 }
663 if (dot(u, u) < r2) {
664 // ray origin is inside sphere
665 continue;
666 }
667 const Vec3fa v = cross(d, u);
668 float v2 = dot(v, v);
669 float d2 = dot(d, d);
670 float r2d2 = r2 * d2;
671 float D = r2d2 - v2;
672 if (D < 0.0f) {
673 // no intersections
674 continue;
675 }
676 float s = scene_rdl2::math::sqrt(D);
677 const Vec3fa w = cross(d, v);
678 const Vec3fa rn = scene_rdl2::math::rcp(d2) * (w - s * d);
679 float t = length(u + rn) * scene_rdl2::math::rsqrt(d2);
680 // TODO call occlude filter functions
681 // mark the tfar negative is the official signal
682 // for embree that the ray is occluded
683 if (rayTnear < t && t < rayTfar) {
684 rayTfar = -FLT_MAX;
685 }
686 }
687 }
688}
