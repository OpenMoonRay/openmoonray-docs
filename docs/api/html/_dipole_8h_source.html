25///
26/// @class DipoleBssrdf Dipole.h <shading/bssrdf/Dipole.h>
27/// @brief Derived class to implement the dipole Bssrdf with importance sampling.
28///
29/// This Bssrdf implements the dipole approximation as in (multiple scattering
30/// only, single scattering is ignored):
31/// [1] "A Practical Model for Subsurface Light Transport",
32/// Jensen et al., Siggraph 2001
33///
34/// It implements an exact dipole importance sampling scheme for local multiple
35/// scattering is described in:
36/// [2] "Efficient Rendering of Local Subsurface Scattering", Mertens et al.,
37/// Pacific Conference on Computer Graphics and Applications, 2003
38///
39/// It uses optionally the Bssrdf re-parameterization described in:
40/// [3] "A Rapid Hierarchical Rendering Technique for Translucent Materials",
41/// Jensen & Buhler, Siggraph 2002
42///
43class DipoleBssrdf : public Bssrdf
44{
45public:
46 /// Initialize with standard physical parameters (cross sections in mm-1)
47 /// The cross sections are converted to world-units-1 (in render space)
48 /// according to the world scale:
49 /// world space unit * worldScale = meter
50 DipoleBssrdf(const scene_rdl2::math::Vec3f &N, float eta, const scene_rdl2::math::Color &sigmaA,
51 const scene_rdl2::math::Color &sigmaSP, float sceneScale /* = 0.01f */,
52 const scene_rdl2::rdl2::Material* material,
53 const scene_rdl2::rdl2::EvalNormalFunc evalNormalFn);
54
55 /// Initialize with artist-friendly parameters (see [3] for conversion),
56 /// where the color-dependent radius is specified in render space. Note that
57 /// the translucentColor cannot be grater than 1 due to the applied conversion.
58 DipoleBssrdf(const scene_rdl2::math::Vec3f &N, float eta, const scene_rdl2::math::Color &translucentColor,
59 const scene_rdl2::math::Color &radius, const scene_rdl2::rdl2::Material* material,
60 const scene_rdl2::rdl2::EvalNormalFunc evalNormalFn);
61
62 /// Initialize using the corresponding vectorized version of this class.
63 DipoleBssrdf(scene_rdl2::alloc::Arena *arena, const DipoleBssrdfv &bssrdfv, int lane);
