34std::string
35ActiveBitTable::showBlock(const std::string &hd,
36 const unsigned blockId) const
37{
38 unsigned len = std::to_string(mTotalItems).size();
39 unsigned len2 = std::to_string(calcBlockTotal(mTotalItems)).size();
40 unsigned startId = blockId * 64;
41 unsigned endId = std::min(startId + 64 - 1, mTotalItems - 1);
42
43 std::ostringstream ostr;
44 ostr << hd
45 << std::setw(len2) << std::setfill('0') << blockId
46 << "(" << std::setw(len) << std::setfill('0') << startId << '~'
47 << std::setw(len) << std::setfill('0') << endId << ')'
48 << " 0x" << std::setw(16) << std::setfill('0') << std::hex << getBlock(blockId) << std::dec
49 << ' ';
50 for (int bitId = 63; bitId >= 0; --bitId) {
51 if (blockId * 64 + bitId >= mTotalItems) {
52 ostr << ' ';
53 if (bitId > 0) {
54 if (!(bitId % 4)) ostr << ' ';
55 }
56 } else {
57 ostr << ((getBlock(blockId) >> bitId) & 0x1);
58 if (bitId > 0) {
59 if (!(bitId % 8)) ostr << '-';
60 else if (!(bitId % 4)) ostr << '/';
61 }
62 }
63 }
64 return ostr.str();
65}

70ActiveBitTables::DumpMode
71ActiveBitTables::finalize()
72//
73// You need to call before use enqTblDump().
74// Update hierarchical multi-level table condition if needed internally and return
75// proper DumpMode for encoding (= minimize data size).
76//
77{
78 size_t fullDeltaSize = calcSerializedTileAddrInfoSizeFullDeltaDump();
79 size_t tblDumpSize = calcSerializedTileAddrInfoSizeTblDump();
80
81 // We know full delta dump is always better than full dump. So only test with full delta dump
82 if (fullDeltaSize <= tblDumpSize) {
83 mDataSize = fullDeltaSize;
84 return DumpMode::FULL_DELTA_DUMP;
85 }
86
87 // We should pick table dump mode.
88 mDataSize = tblDumpSize;
89 if (mFullActiveTable) {
90 return DumpMode::LEAF_TABLE_DUMP;
91 }
92 return DumpMode::TABLE_DUMP;
93}

113void
114ActiveBitTables::enqFullDeltaDump(VContainerEnq &vContainerEnq)
115{
116 unsigned prevItemId = std::numeric_limits<unsigned>::max();
117 crawlActiveTblItem([&](unsigned tileId) {
118 unsigned deltaId;
119 if (prevItemId >= tileId) {
120 deltaId = tileId; // very 1st tile
121 } else {
122 deltaId = tileId - prevItemId;
123 }
124 vContainerEnq.enqVLUInt(deltaId);
125 prevItemId = tileId;
126 });
127}

129void
130ActiveBitTables::deqFullDeltaDump(VContainerDeq &vContainerDeq, const unsigned activeTileTotal)
131{
132 unsigned prevId = 0;
133 for (unsigned i = 0; i < activeTileTotal; ++i) {
134 unsigned deltaId, tileId;
135 vContainerDeq.deqVLUInt(deltaId);
136
137 if (i == 0) {
138 tileId = deltaId;
139 } else {
140 tileId = prevId + deltaId;
141 }
142 setOn(tileId);
143
144 prevId = tileId;
145 }

148void
149ActiveBitTables::enqTblDump(VContainerEnq &vContainerEnq)
150//
151// This API should call after execute calcSerializedTileAddrInfoSizeTblDump()
152//
153{
154 if (mFullActiveTable) {
155 //
156 // full active leaf table : all blocks are active. We simply dump only leaf table
157 //
158 for (unsigned blockId = 0; blockId < mTables[0]->getTotalBlock(); ++blockId) {
159 vContainerEnq.enqMask64(mTables[0]->getBlock(blockId));
160 }
161
162 } else {
163 //
164 // We have to dump all tables
165 //
166 size_t tblId = mTables.size() - 1;
167 uint64_t currMask = mTables[tblId]->getBlock(0); // always 1 item
168 vContainerEnq.enqMask64(currMask);
169
170 if (mTables.size() > 1) {
171 do {
172 crawlActiveTblBlockUseNextLevel
173 (--tblId,
174 [&](unsigned blockId) {
175 vContainerEnq.enqMask64(mTables[tblId]->getBlock(blockId));
176 });
177 } while (tblId > 0);
178 }
179 }
180}

182void
183ActiveBitTables::deqTblDump(VContainerDeq &vContainerDeq, bool fullActiveTable)
184{
185 mFullActiveTable = fullActiveTable;
186
187 if (mFullActiveTable) {
188 for (unsigned blockId = 0; blockId < mTables[0]->getTotalBlock(); ++blockId) {
189 uint64_t currMask;
190 vContainerDeq.deqMask64(currMask);
191 mTables[0]->setBlock(blockId, currMask);
192 }
193
194 } else {
195 uint64_t currMask;
196 vContainerDeq.deqMask64(currMask);
197
198 size_t tblId = mTables.size() - 1;
199 mTables[tblId]->setBlock(0, currMask); // always 1 item
200
201 if (mTables.size() > 1) {
202 do {
203 crawlActiveTblBlockUseNextLevel(--tblId,
204 [&](unsigned blockId) {
205 vContainerDeq.deqMask64(currMask);
206 mTables[tblId]->setBlock(blockId, currMask);
207 });
208 } while (tblId > 0);
209 }
210 }

213size_t
214ActiveBitTables::debugGetSizeInfo(size_t &fullDump,
215 size_t &fullDeltaDump,
216 size_t &tblDump) const
217//
218// Available to use after finalize() call,
219// Get different dump mode data size and return encoded data size in order to compare
220// all possible choices by performance.
221//
222{
223 fullDump = calcSerializedTileAddrInfoSizeFullDump();
224 fullDeltaDump = calcSerializedTileAddrInfoSizeFullDeltaDump();
225
226 if (mFullActiveTable) {
227 tblDump = calcSerializedTileAddrInfoSizeLeafTblDump();
228 } else {
229 tblDump = calcSerializedTileAddrInfoSizeAllTblDump();
230 }
231
232 return mDataSize;

344size_t
345ActiveBitTables::calcSerializedTileAddrInfoSizeTblDump()
346//
347// return serialized data size by byte when choose DumpMode as TABLE_DUMP or LEAF_TABLE_DUMP.
348// TABLE_DUMP or LEAF_TABLE_DUMP is automatically selected based on the data condition.
349// You can understand which type is selected by mFullActiveTable flag.
350// If mFullActiveTable is true, DumpMode should be LEAF_TABLE_DUMP.
351// If not, DumpMode should be TABLE_DUMP.
352//
353{
354 // update all tables
355 if (finalizeTables()) {
356 // full active table : all blocks are active
357 return calcSerializedTileAddrInfoSizeLeafTblDump();
358 }
359
360 return calcSerializedTileAddrInfoSizeAllTblDump();
361}
