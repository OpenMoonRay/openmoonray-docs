49inline void
50atomicLoadFloat4(float* __restrict dst, const float* __restrict src)
51{
52 // We don't allow our array of four floats to overlap since we're using restrict.
53 MNRY_ASSERT(dst + 4 <= src || src + 4 <= dst);
54
55 // assumes 1) val and newVal point to aligned float[4] types
56 // 2) val[3] and newVal[3] store the depth
57 // 3) These types should produce lock-free atomics
58 struct alignas(util::kAtomicAlignment) Float4Aligned
59 {
60 float x, y, z, w;
61 };
62 Float4Aligned* const dstStruct = reinterpret_cast<Float4Aligned*>(dst);
63 const Float4Aligned* const srcStruct = reinterpret_cast<const Float4Aligned*>(src);
64
65 MNRY_ASSERT(reinterpret_cast<uintptr_t>(dst) % util::kAtomicAlignment == 0);
66 MNRY_ASSERT(reinterpret_cast<uintptr_t>(src) % util::kAtomicAlignment == 0);
67 MNRY_ASSERT(__atomic_is_lock_free(sizeof(Float4Aligned), srcStruct));
68 //static_assert(__atomic_always_lock_free(sizeof(Float4Aligned), nullptr), "");
69
70 __atomic_load(srcStruct, dstStruct, __ATOMIC_RELAXED);
71}

73//
74// This particular heuristic is a variation on the error metric proposed in the paper
75// "A Hierarchical Automatic Stopping Condition for Monte Carlo Global Illumination",
76// by Dammertz et al.
77//
78inline float
79estimatePixelErrorInternal(unsigned px,
80 unsigned py,
81 const scene_rdl2::fb_util::Tiler& tiler,
82 const scene_rdl2::fb_util::RenderBuffer& renderBuf,
83 const scene_rdl2::fb_util::FloatBuffer& numSamplesBuf,
84 const scene_rdl2::fb_util::RenderBuffer& renderBufOdd)
85{
86 tiler.linearToTiledCoords(px, py, &px, &py);
87
88 const float* const sampleCountPointer = &(numSamplesBuf.getPixel(px, py));
89 const float totalSamples = util::atomicLoad(sampleCountPointer, std::memory_order_relaxed);
90 const float numOddSamples = scene_rdl2::math::floor(totalSamples * 0.5f);
91
92 const float numEvenSamples = totalSamples - numOddSamples;
93
94 if (numEvenSamples <= 0.0f || numOddSamples <= 0.0f) {
95 return std::numeric_limits<float>::infinity();
96 }
97
98 const float* const totalColorPointer = &(renderBuf.getPixel(px, py)[0]);
99 const float* const oddColorPointer = &(renderBufOdd.getPixel(px, py)[0]);
100 alignas(util::kAtomicAlignment) scene_rdl2::math::Vec4f totalColor;
101 atomicLoadFloat4(&totalColor[0], totalColorPointer);
102 alignas(util::kAtomicAlignment) scene_rdl2::math::Vec4f oddColor;
103 atomicLoadFloat4(&oddColor[0], oddColorPointer);
104
105 if (isnan(totalColor) || isnan(oddColor)) {
106 return 0.0f;
107 }
108
109 //
110 // We want to compute:
111 //
112 // even_color oddColor
113 // diff = --------------- - -------------
114 // numEvenSamples numOddSamples
115 //
116 // where even_color = totalColor - oddColor
117 //
118 // We can refactor this expression remove one of the divides.
119 //
120 const scene_rdl2::math::Vec4f diff = (totalColor * numOddSamples - oddColor * totalSamples) /
121 (numOddSamples * numEvenSamples);
122 const scene_rdl2::math::Vec4f absdiff = abs(diff);
123 const float lumDiff = luma(absdiff);
124 const float lumAvg = luma(totalColor / totalSamples);
125
126 const float alphaScore = absdiff[3];
127
128 // Radiance is pure black. Return the difference in the alpha to see if there's still work to do.
129 if (lumAvg <= 0.0f) {
130 return alphaScore;
131 }
132
133 // Normalize result and use sqrt to approximate eye's response to linear light
134 // (i.e. fake a pseudo gamma curve).
135 //
136 // Alpha and color are both considered for error evaluation. Alpha should
137 // be in [0, 1], whereas the color luma is in non-negative real
138 // space. Even though they are in different scales, this seems to work
139 // fairly well to make sure we resolve edges.
140 return (lumDiff * scene_rdl2::math::rsqrt(lumAvg)) + alphaScore;
141}

143inline float
144estimatePixelErrorInternalDammertz(unsigned px,
145 unsigned py,
146 const scene_rdl2::fb_util::Tiler& tiler,
147 const scene_rdl2::fb_util::RenderBuffer& renderBuf,
148 const scene_rdl2::fb_util::FloatBuffer& numSamplesBuf,
149 const scene_rdl2::fb_util::RenderBuffer& renderBufOdd)
150{
151 tiler.linearToTiledCoords(px, py, &px, &py);
152
153 const float* const sampleCountPointer = &(numSamplesBuf.getPixel(px, py));
154 const float totalSamples = util::atomicLoad(sampleCountPointer, std::memory_order_relaxed);
155 const float numOddSamples = scene_rdl2::math::floor(totalSamples * 0.5f);
156
157 if (totalSamples <= 0.0f || numOddSamples <= 0.0f) {
158 return std::numeric_limits<float>::infinity();
159 }
160
161 const float* const totalColorPointer = &(renderBuf.getPixel(px, py)[0]);
162 const float* const oddColorPointer = &(renderBufOdd.getPixel(px, py)[0]);
163 alignas(util::kAtomicAlignment) scene_rdl2::math::Vec4f totalColor;
164 atomicLoadFloat4(&totalColor[0], totalColorPointer);
165 alignas(util::kAtomicAlignment) scene_rdl2::math::Vec4f oddColor;
166 atomicLoadFloat4(&oddColor[0], oddColorPointer);
167
168 if (isnan(totalColor) || isnan(oddColor)) {
169 return 0.0f;
170 }
171
172 const float numerator = scene_rdl2::math::abs(totalColor[0] - oddColor[0]) +
173 scene_rdl2::math::abs(totalColor[1] - oddColor[1]) +
174 scene_rdl2::math::abs(totalColor[2] - oddColor[2]) +
175 scene_rdl2::math::abs(totalColor[3] - oddColor[3]);
176 const float sumImage = reduce_add(totalColor);
177
178 if (sumImage > 0.0f) {
179 return numerator * scene_rdl2::math::rsqrt(sumImage);
180 } else {
181 return numerator;
182 }
183}

185/// @fuction orientedAccumulatedPixelError
186/// The partial sum of errors in a direction is used for finding a split location when subdividing a tree node.
187/// @return This returns a partial sum of sums along an axis in _region_
188//
189// If our data looks like this:
190//
191// 4 2 5 2 6
192// 2 6 2 2 7
193// 2 6 3 3 2
194// 4 6 4 3 1
195//
196// The vertical marginal sum looks like this:
197//
198// 12 20 14 10 16
199//
200// And the partial sum of that looks like this:
201// 12 32 46 56 72
202// This is used to find our split position.
203template <int axis>
204std::vector<float, AdaptiveNS::PoolAllocator<float>>
205orientedAccumulatedPixelError(const scene_rdl2::math::BBox2i& baseRegion,
206 const scene_rdl2::math::BBox2i& region,
207 const scene_rdl2::util::Array2D<float>& pixelErrors,
208 AdaptiveNS::PoolAllocator<float>& allocator)
209{
210 const int length = extents(region, axis);
211
212 // We're using an allocator that has a fixed size: we don't want to let the std::vector manage its memory by growing
213 // past its capacity: we may run out of space as it has to move all of its elements later in the array.
214 std::vector<float, AdaptiveNS::PoolAllocator<float>> marginalErrors(length, 0.0f, allocator);
215
216 // This doesn't work on ICC 15 for some reason. It works on other compilers. :-/
217 //for (auto p : region) {
218
219 for (auto pit = begin(region); pit != end(region); ++pit) {
220 const scene_rdl2::math::Vec2i& p = *pit;
221 const scene_rdl2::math::Vec2i pBaseRelative = p - baseRegion.lower;
222 const int idx = p[axis] - region.lower[axis];
223 const float pixelError = pixelErrors(pBaseRelative[0], pBaseRelative[1]);
224 marginalErrors[idx] += pixelError;
225 }
226
227 std::partial_sum(marginalErrors.begin(), marginalErrors.end(), marginalErrors.begin());
228 return marginalErrors;
229}

310 /// @return A fraction along the continuous space approximated by accumulatedError that splits the error in half.
311 template <typename Allocator>
312 static float findMidpointSplitLocation(const std::vector<float, Allocator>& accumulatedError)
313 {
314 MNRY_ASSERT(!accumulatedError.empty());
315
316 // Precondition: accumulated error is a partial sum (implying it's weakly increasing).
317
318 // The last value will be the total sum, because the accumulated error is a partial sum.
319 const float half = accumulatedError.back() * 0.5f;
320
321 // Binary search for the mid-point value.
322 const auto lbiter = std::upper_bound(accumulatedError.cbegin(), accumulatedError.cend(), half);
323
324 int i;
325 if (accumulatedError.back() == 0.f)
326 i = accumulatedError.size() / 2;
327 else
328 i = std::distance(accumulatedError.cbegin(), lbiter);
329
330 const float lower = accumulatedError[i - 1];
331 const float higher = accumulatedError[i];
332
333 // The rough offset gives us the index to a value that's close to the middle.
334 const float roughOffset = static_cast<float>(i);
335
336 // Further refine the rough offset between the lower and higher values.
337 const float delta = higher - lower;
338 const float deltaOffset = (half - lower) / delta;
339 const float fraction = (roughOffset + deltaOffset) / accumulatedError.size();
340 return fraction;
341 }

408 static float calculateAreaError(float error, float nodeArea)
409 {
410 // We used to pass in image area based on the actual...get this, image area. However, this makes the results
411 // inconsistent when doing sub-regions, so now we just hard-code an image area.
412
413 // These values come from Trolls2 a_vp_xmin, a_vp_xmax, a_vp_ymin, a_vp_ymax.
414 // We divide by four, because at the time of this change, we had for regions for adaptive sampling, and we want
415 // this to be relatively consistent with renders before that change.
416 constexpr float xmin = -24.0f;
417 constexpr float xmax = 1943.0f;
418 constexpr float ymin = 0.0f;
419 constexpr float ymax = 815.0f;
420 constexpr float imageArea = ((ymax - ymin) * (xmax - xmin)) / 4.0f;
421
422 // This comes out of the Dammertz paper, but we re-write the math a bit.
423 //const float r = std::sqrt(nodeArea/static_cast<float>(imageArea));
424 //const float e = r/nodeArea * error;
425
426 // The above math can be re-written as this, assuming the values are positive.
427 const float e = error * scene_rdl2::math::rsqrt(nodeArea*imageArea);
428 return e;
429 }

458 static void splitHorizontal(AdaptiveNS::MemoryPool<Node>& pool, Node& node, float offset)
459 {
460 node.mChildren = new (pool) Node[2];
461
462 // P2 upper
463 // +---------+---+
464 // | | |
465 // | | |
466 // | | |
467 // +---------+---+
468 // lower P1
469
470 const float deltaX = extents(node.mBounds, 0);
471 const scene_rdl2::math::Vec2f P1(node.mBounds.lower[0] + offset * deltaX, node.mBounds.lower[1]);
472 const scene_rdl2::math::Vec2f P2(node.mBounds.lower[0] + offset * deltaX, node.mBounds.upper[1]);
473
474 node.getLeftChild().mBounds = scene_rdl2::math::BBox2f(node.mBounds.lower, P2);
475 node.getRightChild().mBounds = scene_rdl2::math::BBox2f(P1, node.mBounds.upper);
476 }
