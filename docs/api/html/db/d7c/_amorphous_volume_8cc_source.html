91AmorphousVolume::transformPrimitive(

94{
95 shading::XformSamples p2r = primToRender;
96 if (p2r.size() == 1) {

100 if (motionBlurParams.isMotionBlurOn()) {
101 motionBlurParams.getMotionBlurDelta(shutterOpenDelta, shutterCloseDelta);
102 }
