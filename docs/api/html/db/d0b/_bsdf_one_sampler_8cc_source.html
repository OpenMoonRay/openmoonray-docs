29BsdfOneSampler::BsdfOneSampler(const shading::Bsdf &bsdf, const shading::BsdfSlice &slice) :
30 mBsdf(bsdf),
31 mSlice(slice)
32{
33 // Allocate and initialize lobe cdf and index arrays
34 mLobeCount = bsdf.getLobeCount();
35 memset(mLobeCdf, 0, sizeof(mLobeCdf[0]) * mLobeCount);
36 for (int l=0; l < mLobeCount; l++) {
37 mLobeIndex[l] = l;
38 }
39
40 // Bubble-up the lobes that match the flags at the beginning of the array,
41 // and compute their approximated albedo.
42 // Note: since we use the approximated albedo, our sampling decisions might
43 // be slightly off in cases where a lobe crosses the surface (at grazing
44 // angles). In these cases, we may spend slightly more samples than we
45 // should.
46 // TODO: We should find a way to devote only a single sample to mirror ?
47 int last = mLobeCount - 1;
48 int current = 0;
49 float normalize = 0.0f;
50 shading::BsdfLobe::Type flags = mSlice.getFlags();
51 while (current <= last) {
52 const shading::BsdfLobe *lobe = bsdf.getLobe(mLobeIndex[current]);
53 if (lobe->matchesFlags(flags)) {
54 // Use the mirror reflection direction assumption to compute the
55 // albedo approximaion, which drives our lobe sampling decisions.
56 // A negative albedo shouldn't happen, the abs() is for paranoia.
57 mLobeCdf[current] = scene_rdl2::math::abs(luminance(lobe->albedo(mSlice)));
58 mLobeCdf[current] = max(mLobeCdf[current], sEpsilon);
59 normalize += mLobeCdf[current];
60 current++;
61 } else {
62 std::swap(mLobeIndex[current], mLobeIndex[last]);
63 mLobeCdf[current] = mLobeCdf[last];
64 mLobeCdf[last] = 0;
65 last--;
66 }
67 }
68 mLobeCount = last + 1;
69
70 // Handle if no lobes match
71 if (mLobeCount == 0) {
72 return;
73 }
74
75 // Normalize the albedos and integrate into a cdf, all in one pass
76 normalize = 1.0f / normalize;
77 float cdf = 0.0f;
78 for (int l=0; l < mLobeCount; l++) {
79 mLobeCdf[l] *= normalize;
80 mLobeCdf[l] += cdf;
81 cdf = mLobeCdf[l];
82 }
83 MNRY_ASSERT(isOne(cdf));
84}

94Color
95BsdfOneSampler::eval(const scene_rdl2::math::Vec3f &wi, float &pdf,
96 LobesContribution* lobesContribution) const
97{
98 shading::BsdfLobe::Type flags = mSlice.getSurfaceFlags(mBsdf, wi);
99
100 if (lobesContribution != nullptr) {
101 lobesContribution->mMatchedLobeCount = 0;
102 }
103 // Add up all the matching lobe's contributions and pdfs
104 Color f(zero);
105 pdf = 0.0f;
106 float prevCdf = 0.0f;
107 for (int l=0; l < mLobeCount; l++) {
108 const shading::BsdfLobe * const lobe = mBsdf.getLobe(mLobeIndex[l]);
109
110 // We need to account for lobe pdf, even if the surface flag doesn't match
111 float tmpPdf = 0.0f;
112 Color color = lobe->eval(mSlice, wi, &tmpPdf);
113 if (lobe->matchesFlags(flags)) {
114 f += color;
115 if (lobesContribution != nullptr) {
116 int& index = lobesContribution->mMatchedLobeCount;
117 lobesContribution->mFs[index] = color;
118 lobesContribution->mLobes[index] = lobe;
119 index++;
120 }
121 }
122 pdf += tmpPdf * (mLobeCdf[l] - prevCdf);
123 prevCdf = mLobeCdf[l];
124 }
125
126 return f;
127}

129int
130BsdfOneSampler::sampleCdf(float r, float &pdf) const
131{
132 MNRY_ASSERT(mLobeCount > 0);
133
134 const float *ptr = std::upper_bound(mLobeCdf, mLobeCdf + mLobeCount - 1, r);
135 const int cdfIndex = ptr - mLobeCdf;
136 MNRY_ASSERT(cdfIndex >= 0);
137 MNRY_ASSERT(cdfIndex < mLobeCount);
138
139 pdf = (cdfIndex == 0) ?
140 mLobeCdf[cdfIndex] :
141 mLobeCdf[cdfIndex] - mLobeCdf[cdfIndex - 1];
142
143 return cdfIndex;
144}

147const shading::BsdfLobe *
148BsdfOneSampler::sampleLobe(float r, float &pdf) const
149{
150 if (mLobeCount == 0) {
151 pdf = 0.0f;
152 return nullptr;
153 }
154
155 const int cdfIndex = sampleCdf(r, pdf);
156 const shading::BsdfLobe *lobe = mBsdf.getLobe(mLobeIndex[cdfIndex]);
157
158 return lobe;

161Color
162BsdfOneSampler::sample(float r0, float r1, float r2, scene_rdl2::math::Vec3f &wi,
163 float &pdf, LobesContribution* lobesContribution) const
164{
165 if (lobesContribution != nullptr) {
166 lobesContribution->mMatchedLobeCount = 0;
167 }
168 // If there is no matching lobes, just return an invalid sample
169 if (mLobeCount == 0) {
170 pdf = 0.0f;
171 return sBlack;
172 }
173
174 float lobePdf;
175 const int cdfIndex = sampleCdf(r0, lobePdf);
176 const shading::BsdfLobe *lobe = mBsdf.getLobe(mLobeIndex[cdfIndex]);
177
178 // Sample that lobe
179 Color f = lobe->sample(mSlice, r1, r2, wi, pdf);
180
181 // Treat invalid samples
182 if (pdf == 0.0f) {
183 return sBlack;
184 }
185
186 // Include the discrete probability of picking this lobe
187 pdf *= lobePdf;
188 if (lobesContribution != nullptr) {
189 lobesContribution->mFs[lobesContribution->mMatchedLobeCount] = f;
190 lobesContribution->mLobes[lobesContribution->mMatchedLobeCount] = lobe;
191 lobesContribution->mMatchedLobeCount++;
192 }
193
194 // Note: This code and the whole mirror sampling rules assume the lobe
195 // types are mutually exclusive.
196 // TODO: Do we change this by sampling mirror lobes separately (see other
197 // comment related to that in BsdfOneSampler::init()).
198 bool lobeIsNotMirror = !lobe->matchesFlag(shading::BsdfLobe::MIRROR);
199 if (lobeIsNotMirror) {
200
201 // Ignore btdfs or brdfs based on geometric surface normal
202 shading::BsdfLobe::Type flags = mSlice.getSurfaceFlags(mBsdf, wi);
203
204 // Compute the overall bsdf for the sampled direction
205
206 // First make sure to re-test the sampled lobe with new flags
207 // TODO: bug, like in PBRT, we're not testing this for mirror lobes
208 if (!lobe->matchesFlags(flags)) {
209 f = sBlack;
210 }
211
212 // Then add up all the other lobe's contributions
213 float prevCdf = 0.0f;
214 for (int cI = 0; cI < mLobeCount; ++cI) {
215 if (cI != cdfIndex) {
216 lobe = mBsdf.getLobe(mLobeIndex[cI]);
217
218 // We need to account for lobe pdf, even if the surface flag
219 // doesn't match
220 float tmpPdf = 0.0f;
221 Color color = lobe->eval(mSlice, wi, &tmpPdf);
222 if (lobe->matchesFlags(flags)) {
223 f += color;
224 if (lobesContribution != nullptr) {
225 lobesContribution->mFs[lobesContribution->mMatchedLobeCount] = color;
226 lobesContribution->mLobes[lobesContribution->mMatchedLobeCount] = lobe;
227 lobesContribution->mMatchedLobeCount++;
228 }
229 }
230 pdf += tmpPdf * (mLobeCdf[cI] - prevCdf);
231 }
232 prevCdf = mLobeCdf[cI];
233 }
234 }
235
236 return f;
237}
