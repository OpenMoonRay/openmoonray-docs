18/// This test mimics what happens in AdaptiveRegions, where a tile informs the region that it is done rendering.
19/// AdaptiveRegions keeps a count of the number of tiles in each region, and when all of the tiles in the region have
20/// been updated, the corresponding tree is locked and updated. This code mimics counting the number of tiles that pass
21/// through a region.
22template <int sRegionOverlap, int sTileWidth, int sTileHeight, int sDesiredRegionsPerDimension>
23void
24count(const OverlappingRegions<sRegionOverlap, sTileWidth, sTileHeight, sDesiredRegionsPerDimension>& regions, int line)
25{
26 using Regions = OverlappingRegions<sRegionOverlap, sTileWidth, sTileHeight, sDesiredRegionsPerDimension>;
27 using IndexArray = typename Regions::IndexArray;
28
29 std::cout << "Count test line: " << line << '\n';
30
31 const int numRegions = regions.getNumRegions();
32 std::vector<int> tileCount(numRegions, 0);
33
34 for (int i = 0; i < numRegions; ++i) {
35 const scene_rdl2::math::BBox2i regionBounds = regions.getOverlappingRegionBounds(i);
36 const int tilesHorizontal = roundUpDivision(extents(regionBounds, 0), sTileWidth);
37 const int tilesVertical = roundUpDivision(extents(regionBounds, 1), sTileHeight);
38
39 tileCount[i] = tilesHorizontal * tilesVertical;
40 }
41
42 const scene_rdl2::math::BBox2i& bounds = regions.getOverallBounds();
43
44 for (int y = bounds.lower[1]; y < bounds.upper[1]; y += sTileHeight) {
45 for (int x = bounds.lower[0]; x < bounds.upper[0]; x += sTileWidth) {
46 const scene_rdl2::math::BBox2i tile{scene_rdl2::math::Vec2i{x, y}, scene_rdl2::math::Vec2i{x + sTileWidth, y + sTileHeight}};
47 IndexArray indexArray;
48 std::vector<bool> visited(numRegions, false);
49
50 regions.getOverlappedRegions(tile, indexArray);
51
52 for (std::size_t i = 0; i < indexArray.size() && indexArray[i] >= 0; ++i) {
53 const int idx = indexArray[i];
54 if (visited[idx]) {
55 continue;
56 }
57 visited[idx] = true;
58 --tileCount[idx];
59 }
60 }
61 }
62 for (int i = 0; i < numRegions; ++i) {
63 CPPUNIT_ASSERT(tileCount[i] == 0);
64 }
65}
66} // anonymous namespace
