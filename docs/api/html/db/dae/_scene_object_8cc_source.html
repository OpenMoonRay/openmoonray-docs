27// Interpolation functions for types that we know how to interpolate.
28template <typename T>
29T
30interpolate(const T& begin, const T& end, float t)
31{
32 // This form of linear interpolation (written as a blend) is better for
33 // floating point precision because we avoid scaling (end - begin) by t.
34 return (begin * (1.0f - t)) + (end * t);
35}
36
37template <>
38Vec2d
39interpolate(const Vec2d& begin, const Vec2d& end, float t)
40{
41 return (begin * double(1.0f - t)) + (end * double(t));
42}
43
44template <>
45Vec3d
46interpolate(const Vec3d& begin, const Vec3d& end, float t)
47{
48 return (begin * double(1.0f - t)) + (end * double(t));
49}
50
51template <>
52Vec4d
53interpolate(const Vec4d& begin, const Vec4d& end, float t)
54{
55 return (begin * double(1.0f - t)) + (end * double(t));
56}
57
58template <>
59Mat4f
60interpolate(const Mat4f& begin, const Mat4f& end, float t)
61{
62 return math::slerp(begin, end, t);
63}
64
65template <>
66Mat4d
67interpolate(const Mat4d& begin, const Mat4d& end, float t)
68{
69 return math::slerp(begin, end, double(t));
70}

74SceneObject::SceneObject(const SceneClass& sceneClass, const std::string& name) :
75 mAttributeStorage(nullptr),
76 mBindings(nullptr),
77 mSceneClass(sceneClass),
78 mName(name),
79 mType(INTERFACE_GENERIC),
80 mAttributeSetMask(sceneClass.mAttributes.size()),
81 mBindingSetMask(sceneClass.mAttributes.size()),
82 mAttributeUpdateMask(sceneClass.mAttributes.size()),
83 mBindingUpdateMask(sceneClass.mAttributes.size()),
84 mUpdateActive(false),
85 mDirty(true),
86 mUpdatePrepApplied(false),
87 mAttributeTreeChanged(false),
88 mBindingTreeChanged(false),
89 mUpdateRequested(false)
90{
91 mAttributeStorage = mSceneClass.createStorage();
92 mAttributeUpdateMask.set(); // all attributes just got set to defaults
93
94 mBindings = new SceneObject*[sceneClass.mAttributes.size()];
95 // Yes, even though we have an "attribute is set mask", we must initialize
96 // these pointers to null. The Binary writer will check all bindings slots
97 // for zero when doing the 'delta' mode
98 for (std::size_t i = 0; i < sceneClass.mAttributes.size(); i++) {
99 mBindings[i] = nullptr;
100 }
101}

106 delete[] mBindings;

120template <typename T, typename SET>
121void
122getBindingTransitiveClosureImpl(T * parentObj, SET & result)
123{
124 result.insert(parentObj);
125
126 uint i = 0;
127 const SceneClass& sc = parentObj->getSceneClass();
128 for (auto it = sc.beginAttributes(); it != sc.endAttributes(); ++it, ++i) {
129
130 T *childObj = parentObj->mBindings[i];
131
132 if (childObj == nullptr && (*it)->getType() == TYPE_SCENE_OBJECT) {
133 childObj = parentObj->get(AttributeKey<SceneObject *>(**it));
134 }
135
136 if (childObj) {
137 getBindingTransitiveClosureImpl(childObj, result);
138 }
139 }
140}

155template <typename T>
156T
157SceneObject::get(AttributeKey<T> key, float t) const
158{
159 // If the attribute isn't blurrable, it's constant at all timesteps.
160 if (!key.isBlurrable()) {
161 return SceneClass::getValue(mAttributeStorage, key, TIMESTEP_BEGIN);
162 }
163
164 // Rescale time according to the fast time rescaling coefficients. See
165 // Types.h for more info.
166 TimeRescalingCoeffs coeffs = mSceneClass.mContext->mTimeRescalingCoeffs;
167 float tScaled = coeffs.mScale * t + coeffs.mOffset;
168
169 return interpolate(
170 SceneClass::getValue(mAttributeStorage, key, TIMESTEP_BEGIN),
171 SceneClass::getValue(mAttributeStorage, key, TIMESTEP_END),
172 tScaled);
173}

175template <typename T>
176void
177SceneObject::set(AttributeKey<T> key, const T& value)
178{
179 if (!mUpdateActive) {
180 std::stringstream errMsg;
181 errMsg << "Attribute '" << mSceneClass.getAttribute(key)->getName() <<
182 "' of SceneObject '" << mName << "' can only be set between"
183 " beginUpdate() and endUpdate() calls.";
184 throw except::RuntimeError(errMsg.str());
185 }
186
187 int timestep = TIMESTEP_BEGIN;
188 bool changed = false;
189 do {
190 changed |= SceneClass::setValue(mAttributeStorage, key, static_cast<AttributeTimestep>(timestep), value);
191 ++timestep;
192 } while (key.isBlurrable() && timestep < NUM_TIMESTEPS);
193
194 if (changed) {
195 mAttributeSetMask.set(key.mIndex, true);
196 mAttributeUpdateMask.set(key.mIndex, true);
197 mDirty = true;
198 }
199}

204{
205 if (!mUpdateActive) {

208 "' of SceneObject '" << mName << "' can only be set between"
209 " beginUpdate() and endUpdate() calls.";
210 throw except::RuntimeError(errMsg.str());
211 }
212
213 // Type check each value in the vector against the attribute's object type.
214 for (typename Container::const_iterator iter = value.begin();

219 "' only allows values of type '" << interfaceTypeName(key.mObjectType) <<
220 "', but an element in the vector, SceneObject '" << (*iter)->getName() <<
221 "' is type '" << interfaceTypeName((*iter)->getType()) << "'.";
222 throw except::TypeError(errMsg.str());
223 }
224 }
225
226 int timestep = TIMESTEP_BEGIN;
227 bool changed = false;
228 do {
229 changed |= SceneClass::setValue(mAttributeStorage, key, static_cast<AttributeTimestep>(timestep), value);
230 ++timestep;
231 } while (key.isBlurrable() && timestep < NUM_TIMESTEPS);
232
233 if (changed) {
234 mAttributeSetMask.set(key.mIndex, true);
235 mAttributeUpdateMask.set(key.mIndex, true);
236 mDirty = true;
237 }

491template <typename T>
492bool
493SceneObject::isDefault(AttributeKey<T> key) const
494{
495 const Attribute* attr = mSceneClass.getAttribute(key);
496 const T& d = attr->getDefaultValue<T>();
497 if (d != get(key, TIMESTEP_BEGIN)) return false;
498 if (attr->isBlurrable()) return d == get(key, TIMESTEP_END);
499 return true;
500}

554template <typename F>
555void
556SceneObject::setBinding(uint32_t index, bool bindable,
557 SceneObjectInterface objectType, SceneObject* sceneObject,
558 F attributeNameFetcher)
559{
560 // Must be during an update.
561 if (!mUpdateActive) {
562 throw except::RuntimeError(util::buildString("Attribute '",
563 attributeNameFetcher(), "' of SceneObject '", mName,
564 "' can only be bound between beginUpdate() and endUpdate() calls."));
565 }
566
567 // Attribute must be bindable.
568 if (!bindable) {
569 throw except::RuntimeError(util::buildString("Cannot set binding for"
570 " Attribute '", attributeNameFetcher(), "' on SceneObject '",
571 mName, "' because it is not bindable."));
572 }
573
574 // Type check the bound object.
575 if (sceneObject && !sceneObject->isA(objectType)) {
576 throw except::TypeError(util::buildString("Cannot bind SceneObject '",
577 sceneObject->getName(), "' (of type '",
578 interfaceTypeName(sceneObject->getType()), "') to Attribute '",
579 attributeNameFetcher(), "' on SceneObject '", mName,
580 "' because it expects bound objects of type '",
581 interfaceTypeName(objectType), "'."));
582 }
583
584 mBindings[index] = sceneObject;
585 mBindingSetMask.set(index, true);
586 mBindingUpdateMask.set(index, true);
587 mDirty = true;
588}
