35std::string
36classNameFromFileName(const std::string& baseName,
37 const std::string& expectedExtension)
38{
39 // Base name must be at least expectedExtension + 1 chars ("a<extension>").
40 if (baseName.size() < expectedExtension.size() + 1) {
41 return "";
42 }
43
44 // Final characters must match expectedExtension.
45 std::string extension =
46 baseName.substr(baseName.size() - expectedExtension.size(), std::string::npos);
47 std::transform(extension.begin(), extension.end(), extension.begin(), ::tolower);
48 if (extension != expectedExtension) {
49 return "";
50 }
51
52 // Class name is the file name without the extension.
53 return baseName.substr(0, baseName.size() - expectedExtension.size());
54}

59std::string
60Dso::classNameFromFileName(const std::string& filePath)
61{
62 char* dirStr = strdup(filePath.c_str());
63 std::string directory(dirname(dirStr));
64 free(dirStr);
65 char* baseStr = strdup(filePath.c_str());
66 std::string baseName(basename(baseStr));
67 free(baseStr);
68
69 // Bail early if we can't determine the class name.
70

72 if (className.empty()) {

79Dso::Dso(const std::string& className, const std::string& searchPath, bool proxyModeEnabled) :
80 mHandle(nullptr),
81 mDeclareFunc(nullptr),
82 mCreateFunc(nullptr),
83 mDestroyFunc(nullptr)
84{
85 MNRY_ASSERT(!className.empty(), "Dso must be constructed with a non-empty SceneClass name.");
86 mFilePath = className + ".so";
87
88 if (proxyModeEnabled) {
89 mFilePath += ".proxy";
90 }
91
92 // If they explicitly specified a search path, attempt to find the DSO.
93 if (!searchPath.empty()) {
94 mFilePath = util::findFile(mFilePath, searchPath);
95 }
96
97 // If the file path is empty, we couldn't find it.
98 if (mFilePath.empty()) {
99 throw except::IoError(util::buildString("Couldn't find DSO for '",
100 className, "' in search path '", searchPath, "'."));
101 }
102
103 // Attempt to open the DSO.
104 mHandle = dlopen(mFilePath.c_str(), RTLD_LAZY);
105 if (!mHandle) {
106 std::stringstream errMsg;
107 errMsg << "Found RDL2 DSO '" << mFilePath << "', but failed to"
108 " dlopen() it";
109 char* error = dlerror();
110 if (error) {
111 errMsg << ": " << error;
112 } else {
113 errMsg << ".";
114 }
115 throw except::RuntimeError(errMsg.str());
116 }
117}

119Dso::~Dso()
120{
121 if (mHandle) {
122 dlclose(mHandle);
123 }
124}

126ClassDeclareFunc
127Dso::getDeclare()
128{
129 // Return cached function pointer if we already looked it up.
130 if (mDeclareFunc) {
131 return mDeclareFunc;
132 }
133
134 // Clear errors.
135 dlerror();
136
137 // Attempt to load the rdl_declare symbol.
138 MNRY_ASSERT(mHandle, "Tried to load symbol from bad DSO handle.");
139 void* declareSymbol = dlsym(mHandle, "rdl2_declare");
140 if (!declareSymbol) {
141 // Technically null symbols are valid, but a null function pointer
142 // is useless to us, so that's really a failure case, too.
143 std::stringstream errMsg;
144 errMsg << "Failed to load symbol 'rdl2_declare' from RDL2 DSO '" <<
145 mFilePath << "'";
146 char* error = dlerror();
147 if (error) {
148 errMsg << ": " << error;
149 } else {
150 errMsg << ".";
151 }
152 throw except::RuntimeError(errMsg.str());
153 }
154
155 // Cache the function pointer.
156 mDeclareFunc = (ClassDeclareFunc)declareSymbol;
157
158 return mDeclareFunc;
159}

231bool
232Dso::isValidDso(const std::string& filePath, bool proxyModeEnabled)
233{
234 // Break the path into directory and basename components. Painfully, both
235 // dirname() and basename() may do just about anything with your pointers,
236 // so its safest to make a copy first.
237 char* dirStr = strdup(filePath.c_str());
238 std::string directory(dirname(dirStr));
239 free(dirStr);
240 char* baseStr = strdup(filePath.c_str());
241 std::string baseName(basename(baseStr));
242 free(baseStr);
243
244 // Bail early if we can't determine the class name.
245 const char* extension = (proxyModeEnabled) ? ".so.proxy" : ".so";
246 std::string className = internal::classNameFromFileName(baseName, extension);
247 if (className.empty()) {
248 return false;
249 }
250
251 // Attempt to load it as an RDL Dso object and get the expected function
252 // pointers.
253 try {
254 std::unique_ptr<Dso> dso(new Dso(className, directory, proxyModeEnabled));
255 dso->getDeclare();
256 if (!proxyModeEnabled) {
257 dso->getCreate();
258 dso->getDestroy();
259 }
260 } catch (...) {
261 return false;
262 }
263
264 // It's a valid DSO.
265 return true;
266}
