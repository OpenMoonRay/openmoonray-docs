33 // This class is written to allow for private inheritance, the preferred way
34 // to use this class.
35 //
36 // Use CRTP (curiously recurring template pattern) to so that we don't
37 // have to use virtual functions.
38 //
39 // Many concepts for this class taken from Peter Weinert's Dr. Dobb's
40 // article, "A Base Class for Intrusively Reference-Counted Objects in C++."
41 template <typename Derived, typename Deleter = std::default_delete<Derived>>
42 class RefCount
43 {
44 public:
45 friend void refInc(const Derived* d)
46 {
47 // C-style cast to allow private inheritance.
48 // We can't static_cast to an inaccessible base class.
49 // reinterpret_cast will fail in the face of multiple inheritance.
50 ++((const RefCount*) d)->refCounter;
51 }
52 friend void refDec(const Derived* d)
53 {
54 // C-style cast to allow private inheritance.
55 // We can't static_cast to an inaccessible base class.
56 // reinterpret_cast will fail in the face of multiple inheritance.
57 if (--((const RefCount*) d)->refCounter == 0u) {
58 Deleter()(const_cast<Derived *>(d));
59 }
60 }
61
62 protected:
63 explicit RefCount(unsigned val = 0u) : refCounter(val) {}
64 RefCount(const RefCount&) : refCounter(0) {}
65 RefCount& operator=(const RefCount&) { return *this; }
66 ~RefCount() = default;
67 void swap(RefCount&) noexcept {};
68 unsigned refCount() const { return refCounter; }
69
70 private:
71 // Mutable so that we can accept const versions of the objects.
72 mutable std::atomic_uint refCounter;
73 };
