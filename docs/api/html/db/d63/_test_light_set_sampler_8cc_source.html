46//
47// Integrate reflected radiance at point p with normal n. We compute this value
48// both with our C++ and ISPC codepaths and make sure the results are the same
49//
50void
51TestLightSetSampler::testLightSetRadiance(const LightSet &lightSet,
52 const scene_rdl2::math::Vec3f &p,
53 const scene_rdl2::math::Vec3f &n)
54{
55 const scene_rdl2::math::Color result = scene_rdl2::math::Color(0.268678, 0.340643, 0.539275);
56
57 //
58 // Compute radiance for (p, n) with C++ codepath
59 //
60
61 double timeCpp = 0.0;
62 time::TimerDouble timerCpp(timeCpp);
63 timerCpp.start();
64
65 scene_rdl2::math::Color testRadiance = doReductionOverUnitSquare<scene_rdl2::math::Color>(scene_rdl2::math::Color(zero),
66 NUM_SAMPLES_PER_AXIS, GRAINSIZE_PER_AXIS,
67 [&](const tbb::blocked_range2d<unsigned> &range, const scene_rdl2::math::Color &current,
68 float scl, float ofs) -> scene_rdl2::math::Color {
69
70 // Access thread-local arena allocator
71 mcrt_common::ThreadLocalState *tls = mcrt_common::getFrameUpdateTLS();
72 scene_rdl2::alloc::Arena *arena = &tls->mArena;
73 SCOPED_MEM(arena);
74
75 // Create the LightSetSampler
76 unsigned sampleCount = range.rows().size() * range.cols().size();
77 LightSetSampler lSampler(arena, lightSet, shading::Bsdf(), p, sampleCount);
78
79 double red = 0.0;
80 double green = 0.0;
81 double blue = 0.0;
82
83 // Loop on lights
84 for (int l=0; l < lSampler.getLightCount(); ++l) {
85
86 const Light* light = lSampler.getLight(l);
87
88 for (unsigned y = range.rows().begin(); y != range.rows().end(); ++y) {
89 float r0 = float(y) * scl + ofs;
90 for (unsigned x = range.cols().begin(); x != range.cols().end(); ++x) {
91 float r1 = float(x) * scl + ofs;
92 const scene_rdl2::math::Vec3f r(r0, r1, 0.f);
93 LightSample lsmp;
94 LightFilterRandomValues filterR = {scene_rdl2::math::Vec2f(0.f, 0.f),
95 scene_rdl2::math::Vec3f(0.f, 0.f, 0.f)};
96 lSampler.sampleIntersectAndEval(tls, light, nullptr /*LighFilterList*/, p, &n, filterR, 0.f,
97 r, lsmp, 0.0f);
98 if (lsmp.isValid()) {
99
100 // Compute reflected radiance from white lambertian brdf,
101 // accounting for the cosine term in the rendering eqn
102 scene_rdl2::math::Color radiance = lsmp.Li * rcp(lsmp.pdf) * sOneOverPi
103 * saturate(dot(lsmp.wi, n));
104
105 // Accumulate
106 red += double(radiance.r);
107 green += double(radiance.g);
108 blue += double(radiance.b);
109 }
110 }
111 }
112 }
113 return current + scene_rdl2::math::Color(float(red * INV_NUM_SAMPLES),
114 float(green * INV_NUM_SAMPLES),
115 float(blue * INV_NUM_SAMPLES));
116 });
117
118 timerCpp.stop();
119
120
121 //
122 // Compute radiance for (p, n) with ISPC codepath
123 //
124
125 double timeIspc = 0.0;
126 time::TimerDouble timerIspc(timeIspc);
127 timerIspc.start();
128
129 scene_rdl2::math::Color testIspcRadiance = doReductionOverUnitSquare<scene_rdl2::math::Color>(scene_rdl2::math::Color(zero),
130 NUM_SAMPLES_PER_AXIS, GRAINSIZE_PER_AXIS,
131 [&](const tbb::blocked_range2d<unsigned> &range, const scene_rdl2::math::Color &current,
132 float scl, float ofs) -> scene_rdl2::math::Color {
133
134 // Access thread-local arena allocator
135 mcrt_common::ThreadLocalState *tls = mcrt_common::getFrameUpdateTLS();
136 scene_rdl2::alloc::Arena *arena = &tls->mArena;
137 SCOPED_MEM(arena);
138
139 ispc::Range2d r;
140 initIspcRange(r, range);
141
142 ispc::Color color;
143 ispc::testLightSetRadiance(reinterpret_cast<ispc::Arena *>(arena),
144 lightSet.asIspc(), asIspc(p), asIspc(n),
145 r, scl, ofs, true, color);
146
147 return current + asCpp(color) * INV_NUM_SAMPLES;
148 });
149
150 timerIspc.stop();
151
152
153 //
154 // Print results and run test asserts
155 //
156
157 const float radianceEquality = 0.001f;
158
159 printInfo(" time radiance = %f", timeCpp);
160 printInfo(" time radiance ispc = %f", timeIspc);
161
162 printInfo(" test radiance = [%f, %f, %f]",
163 testRadiance[0], testRadiance[1], testRadiance[2]);
164 printInfo(" test radiance ispc = [%f, %f, %f]",
165 testIspcRadiance[0], testIspcRadiance[1], testIspcRadiance[2]);
166
167 testAssert(isEqual(testRadiance, result, radianceEquality),
168 "radiance mismatch: test = [%f, %f, %f], result = [%f, %f, %f]",
169 testRadiance[0], testRadiance[1], testRadiance[2],
170 result[0], result[1], result[2]);
171 testAssert(isEqual(testRadiance, testIspcRadiance, radianceEquality),
172 "radiance mismatch: test = [%f, %f, %f], testIspc = [%f, %f, %f]",
173 testRadiance[0], testRadiance[1], testRadiance[2],
174 testIspcRadiance[0], testIspcRadiance[1], testIspcRadiance[2]);
175}
