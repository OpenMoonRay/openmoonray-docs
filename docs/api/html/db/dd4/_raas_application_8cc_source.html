60void
61RaasApplication::parseOptions(bool guiMode)
62{
63 // Check for no flags or help flag.
64 if (mArgc == 1 || std::string(mArgv[1]) == "-h" || std::string(mArgv[1]) == "-help" || std::string(mArgv[1]) == "--help") {
65 std::cerr << rndr::RenderOptions::getUsageMessage(mArgv[0], guiMode) << std::endl;
66 const int status = (mArgc == 1 ? EXIT_FAILURE : EXIT_SUCCESS);
67 exit(status);
68 }
69
70 // Initialize our logging
71 scene_rdl2::logging::Logger::init();
72
73 // Parse the command line.
74 mOptions.parseFromCommandLine(mArgc, mArgv);
75
76 // Check for scene files and "-in" flag because
77 // calls to main()::run() without any scene files will cause
78 // Moonray to exit with 'Error: scene context contains no cameras'
79 if (mOptions.getSceneFiles().empty()) {
80 std::cerr << "No scene file specified. Did you forget the `-in` flag?. " << std::endl;
81 exit(EXIT_FAILURE);
82 }
83}

96void
97stackTraceHandler(int sig)
98{
99 std::string crashReason;
100 switch (sig) {
101 break;
102 case SIGSEGV:
103 mcrt_common::debugPrintCallstack("\nSIGSEGV(segfault) callstack");
104 crashReason = "SIGSEGV (Segmentation fault)";
105 break;
106 case SIGILL:
107 mcrt_common::debugPrintCallstack("\nSIGILL(illegal instruction) callstack");
108 crashReason = "SIGILL (Illegal instruction)";
109 break;
110 case SIGFPE:
111 mcrt_common::debugPrintCallstack("\nSIGFPE(float point exception) callstack");
112 crashReason = "SIGFPE (Floating point exception)";
113 break;
114 default:
115 mcrt_common::debugPrintCallstack(nullptr);
116 break;
117 }
118
119 signal(SIGABRT, SIG_DFL);
120 abort();
121}

142void
143RaasApplication::printStatusLine(rndr::RenderContext& renderContext, double startTime, bool done)
144{
145 bool estimationActive =
146 renderContext.getSceneContext().getCheckpointActive() || renderContext.getSceneContext().getResumeRender();
147
148 rndr::RenderProgressEstimation *progEst = renderContext.getFrameProgressEstimation();
149
150 float progressFraction = 0.0f;
151 {
152 std::size_t total = 0;
153 progressFraction = renderContext.getFrameProgressFraction(nullptr, &total);
154 if (total == 0) {
155 return;
156 }
157 progressFraction = scene_rdl2::math::clamp(progressFraction, 0.0f, 1.0f);
158 }
159
160 //------------------------------
161
162 const float pctComplete = progressFraction * 100.f;
163
164 const double currentTime = scene_rdl2::util::getSeconds();
165 const double elapsedMcrtTime = currentTime - startTime;
166
167 // Should we log progress?
168 bool logProgress = false;
169 if (elapsedMcrtTime >= mNextLogProgressTime || pctComplete >= mNextLogProgressPercentage) {
170 mNextLogProgressTime = elapsedMcrtTime + LOG_PROGRESS_TIME_INTERVAL;
171 mNextLogProgressPercentage = pctComplete + 1.0;
172 logProgress = true;
173 }
174
175 // In a live terminal, print the progress bar
176 if (isatty(STDOUT_FILENO)) {
177
178 // Get the terminal width.
179 winsize win;
180 ioctl(STDOUT_FILENO, TIOCGWINSZ, &win);
181
182 if (win.ws_col == 0) {
183 win.ws_col = 80; // situation like emacs shell
184 }
185
186 // Compute progress bar layout.
187 int barWidth = win.ws_col
188 - 6 // " [/] " (spinner)
189 - 9 // "Rendering"
190 - 4 // " [" + "] " (bar borders and spacing)
191 - 7 // "100.0% "
192 - 14 // "xx:xx:xx ETA " Estimated Time Of Accomplishment
193 ;
194
195 // Have we underflowed?
196 if (barWidth < 0) {
197 barWidth = 0;
198 }
199
200 int fullBars = progressFraction * barWidth + 0.5f;
201 static int spinnerIdx = 0;
202 static char spinnerAnim[] = "-\\|/";
203
204 char spinner = (done) ? '+' : spinnerAnim[spinnerIdx];
205
206 // Print the status line.
207 if (estimationActive) {
208 std::printf(" [%c] %s [", spinner, progEst->getModeBanner().c_str());
209 } else {
210 std::printf(" [%c] Rendering [", spinner);
211 }
212 for (int i = 0; i < fullBars; ++i) {
213 std::printf("=");
214 }
215 for (int i = 0; i < barWidth - fullBars; ++i) {
216 std::printf(" ");
217 }
218 if (done) {
219 std::printf("] 100.0%% \n");
220 } else {
221 if (estimationActive) {
222 if (progEst->isAdaptiveSampling()) {
223 std::printf("] %5.1f%%\r", pctComplete);
224 } else {
225 std::printf("] %5.1f%% %s ETA\r", pctComplete,
226 secStr(progEst->getEstimatedSecOfCompletionFromNow()).c_str());
227 }
228 } else {
229 std::printf("] %5.1f%% \r", pctComplete);
230 }
231 std::fflush(stdout);
232 }
233
234 MOONRAY_THREADSAFE_STATIC_WRITE(spinnerIdx = ++spinnerIdx % std::strlen(spinnerAnim));
235
236 } else if (logProgress) {
237 std::stringstream ss;
238 ss << "Rendering [" << std::fixed << std::setw(3) << std::setprecision(0) << pctComplete << "%]";
239 if (estimationActive) {
240 ss << " " << secStr(progEst->getEstimatedSecOfCompletionFromNow()) << " ETA";
241 }
242 renderContext.getSceneRenderStats().logString(ss.str());
243 }
244
245 // Update Athena.
246 if (logProgress) {
247 renderContext.getSceneRenderStats().logMcrtProgress(float(elapsedMcrtTime), pctComplete);
248 }
249}

251std::string
252RaasApplication::secStr(double sec) const
253{
254 std::ostringstream ostr;
255
256 if (sec <= 0.0) {
257 ostr << "- : -";
258 } else {
259 unsigned secInt = static_cast<unsigned>(sec);
260
261 if (secInt >= 3600) { // hour
262 ostr << std::setw(2) << std::setfill('0') << (secInt / 3600) << ':';
263 secInt %= 3600;
264 }
265
266 if (secInt >= 60) { // minute
267 ostr << std::setw(2) << std::setfill('0') << (secInt / 60) << ':';
268 secInt %= 60;
269 } else {
270 ostr << "00:";
271 }
272
273 ostr << std::setw(2) << std::setfill('0') << secInt; // secound
274 }
275
276 return ostr.str();
277}

303int
304writeRenderOutputsWithMessages(const rndr::RenderOutputDriver *rod,
305 const pbr::DeepBuffer *deepBuffer,
306 pbr::CryptomatteBuffer *cryptomatteBuffer,
307 const scene_rdl2::fb_util::HeatMapBuffer *heatMap,
308 const scene_rdl2::fb_util::FloatBuffer *weightBuffer,
309 const scene_rdl2::fb_util::RenderBuffer *renderBufferOdd,
310 const std::vector<scene_rdl2::fb_util::VariablePixelBuffer> &aovBuffers,
311 const std::vector<scene_rdl2::fb_util::VariablePixelBuffer> &displayFilterBuffers)
312{
313 int err = 0;
314
315 const rndr::ImageWriteCache *lastImageWriteCache =
316 rndr::ImageWriteDriver::get()->getLastImageWriteCache();
317 if (lastImageWriteCache && lastImageWriteCache->getTwoStageOutput()) {
318 //
319 // Checkpoint render with two stage output mode
320 //
321 // We already have lastImageWriteCache which was created by checkpoint output with two stage
322 // output mode (i.e. checkpoint was created as tmpfile first then copied to final location).
323 // This last checkpoint output is exactly same as final output.
324 // We only need copy instead of generate file here.
325 if (!lastImageWriteCache->allFinalizeFinalFile()) err = 1;
326
327 } else {
328 //
329 // Non checkpoint render or checkpoint render w/ two stage output off
330 // This case, we don't have any previously written temporary file data and we
331 // need to output final data by standard way.
332 //
333 rndr::ImageWriteDriver::ImageWriteCacheUqPtr cache;
334 if (!rod->requiresDeepBuffer() && rndr::ImageWriteDriver::get()->getTwoStageOutput()) {
335 // Two stage output mode for final output w/ STD mode
336 cache = rndr::ImageWriteDriver::get()->newImageWriteCache(rod);
337 cache->setTwoStageOutput(true);
338 } else {
339 // standard final output (STD mode) with out two stage output
340 MNRY_ASSERT(!cache);
341 }
342
343 rod->write(deepBuffer,
344 cryptomatteBuffer,
345 heatMap,
346 weightBuffer,
347 renderBufferOdd,
348 aovBuffers,
349 displayFilterBuffers,
350 cache.get());
351 err = rod->loggingErrorAndInfo(cache.get())? 0: 1;
352
353 if (cache && cache->getTwoStageOutput()) {
354 if (!cache->allFinalizeFinalFile()) err = 1;
355 }
356 }
357
358 return err;
359}

363watchShaderDsos(ChangeWatcher& watcher, const rndr::RenderContext& renderContext)
364{
365 const scene_rdl2::rdl2::SceneContext& sceneContext = renderContext.getSceneContext();
366
367 // For each active SceneClass, install a file watcher for its DSO.
368 for (auto iter = sceneContext.beginSceneClass(); iter != sceneContext.endSceneClass(); ++iter) {
369 const scene_rdl2::rdl2::SceneClass* sceneClass = iter->second;
370 std::string sourcePath(sceneClass->getSourcePath());
371 if (!sourcePath.empty()) {
372 watcher.watchFile(sourcePath);
373 }
374 }
