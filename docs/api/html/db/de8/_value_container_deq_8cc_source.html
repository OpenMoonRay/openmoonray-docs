49void
50ValueContainerDeq::dataSizeCheck(const void *addr, const size_t dataSize)
51{
52 if (dataSize < sizeof(size_t)) {
53 throw except::RuntimeError("Could not get ValueContainerDeq header size info.");
54 }
55
56 size_t savedDataSize;
57 loadSizeT(getDeqDataAddrUpdate(sizeof(size_t)), savedDataSize);
58 if (dataSize != savedDataSize) {
59 std::ostringstream ostr;
60 ostr << "Encode data length is not match with ValueContainerDeq header. dataSize:"
61 << dataSize << " != header:" << savedDataSize;
62 throw except::RuntimeError(ostr.str());
63 }
64 mDataSize = savedDataSize;
65}
