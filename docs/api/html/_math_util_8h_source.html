17/// Given A and b in the linear system Ax = b, solve for x, i.e. find (A^-1 * b)
18/// A is an array of 4 floats laid out row by row, x and b are both 2
19/// component columns of an arbitrary type. Returns false if a solution could
20/// not be found (matrix was non-invertible).
21template<typename T>
22inline bool
23solve2x2LinearSystem(float const *A, T *x, T const *b)
24{
25 const auto det_a = differenceOfProducts(A[0], A[3], A[1], A[2]);
26 if (!std::isnormal(det_a)) {
27 return false;
28 }
29
30 const auto inv_det_a = 1.0f / det_a;
31 x[0] = differenceOfProducts(b[0], A[3], b[1], A[1]) * inv_det_a;
32 x[1] = differenceOfProducts(b[1], A[0], b[0], A[2]) * inv_det_a;
33
34 return true;
35}

37///
38/// Compute the first order partial derivatives for some arbitrary quantity
39/// over the surface of a triangle with respect to the triangle's uv coordinates.
40///
41/// For a value x over the surface of a triangle, we want to compute the
42/// quantities dxdu and dxdv such that they satisfy the exression:
43///
44/// x' = x + (d_u * dxdu) + (d_v * dxdv)
45///
46/// Given the inputs, we can create 3 equations based on the above:
47///
48/// 1. x0 = x + (u0-u0) * dxdu + (v0-v0) * dxdv
49/// 2. x1 = x + (u1-u0) * dxdu + (v1-v0) * dxdv
50/// 3. x2 = x + (u2-u0) * dxdu + (v2-v0) * dxdv
51///
52/// Eqn 1. just says x = x0. Eqns 2 & 3 form a simple linear system
53/// of 2 equations and 2 unknowns. By rearranging these eqns, we get:
54///
55/// x1 - x0 = (u1-u0) * dxdu + (v1-v0) * dxdv
56/// x2 - x0 = (u2-u0) * dxdu + (v2-v0) * dxdv
57///
58/// which, when put into matrix form (Ax=b), equals:
59///
60/// | u1-u0 v1-v0 | | dxdu | | x1 - x0 |
61/// | u2-u0 v2-v0 | X | dxdv | = | x2 - x0 |
62///
63/// Now all that remains is to solve for dxdu and dxdv (x=(A^-1)*b):
64///
65/// | dxdu | | u1-u0 v1-v0 |-1 | x1 - x0 |
66/// | dxdv | = | u2-u0 v2-v0 | X | x2 - x0 |
67///
68template<typename T>
69inline bool
70computeTrianglePartialDerivatives(T const &x0, T const &x1, T const &x2,
71 Vec2f const &uv0, Vec2f const &uv1, Vec2f const &uv2,
72 T *results)
73{
74 float A[4] =
75 {
76 uv1.x - uv0.x, uv1.y - uv0.y,
77 uv2.x - uv0.x, uv2.y - uv0.y,
78 };
79
80 T b[2] =
81 {
82 x1 - x0,
83 x2 - x0,
84 };
85
86 return solve2x2LinearSystem(A, results, b) && isFinite(results[0]) && isFinite(results[1]);
87}

115// This code is an ugly implementation of the following algorithm, fitting the
116// constraints of C++11's constexpr.
117// function is_prime(n)
118// if n <= 1
119// return false
120// else if n <= 3
121// return true
122// else if n mod 2 = 0 or n mod 3 = 0
123// return false
124// let i = 5
125// while i * i <= n
126// if n mod i = 0 or n mod (i + 2) = 0
127// return false
128// i = i + 6
129// return true
130//
131// When last checked, this function handled compile-time values a little over
132// 9,300,000 in GCC and 35,500,000 in ICC. Anything much larger blew up the
133// compile-time stack space (which can be changed with the compiler flag
134// -fconstexpr-depth).
135constexpr bool isPrime(int n)
136{
137 return n > 1 && (n == 2 || n == 3 || (n % 2 != 0 && n % 3 != 0 && primeDetail::isPrimeHelper(n, 5)));
138}

148template <typename IntType>
149IntType log2i(IntType v)
150{
151 static_assert(std::is_integral<IntType>::value, "Requires an integer type.");
152 IntType r = 0;
153 while (v >>= 1) {
154 ++r;
155 }
156
157 return r;
158}

163inline float
164bias(float value, float bias)
165{
166 if (bias == 0.5f || value <= 0.0f || value >= 1.0f)
167 return value;
168
169 if (bias <= 0.0f)
170 return 0.0f;
171
172 static const float sFactor = 1.0f / math::log(0.5f);
173 return math::pow(value, math::log(bias) * sFactor);
174}

177inline float
178gain(float value, float gain)
179{
180 if (gain == 0.5f) {
181 return value;
182 } else {
183 return value < 0.5f ?
184 bias(2.0f * value, 1.0f - gain) * 0.5f :
185 1.0f - bias(2.0f - 2.0f * value, 1.0f - gain) * 0.5f;
186 }
187}

240//
241// Compute a scale and offset such that when an input in the range (start, end)
242// has this scale and offset applied, we get a result within (0, 1). That is:
243//
244// f(x) = x * scale + offset
245// f(start) = 0
246// f(end) = 1
247//
248// Derivation:
249//
250// s * scale + offset = 0
251// e * scale + offset = 1
252// (s * scale) - (e * scale) = -1
253// (s - e) * scale = -1
254// scale = -1 / (s - e)
255// scale = 1 / (e - s)
256//
257// offset = -(s * scale)
258// offset = -s * scale
259//
260template <typename T>
261finline void
262getScaleOffset(T start, T end, T *scale, T *offset)
263{
264 *scale = T(1) / (end - start);
265 *offset = -start * (*scale);
266}
267
268//
269// Compute a scale and offset such that when an input in the range (start, end)
270// has this scale and offset applied, we get a result within
271// (startMapTo, endMapTo). That is:
272//
273// f(x) = x * scale + offset
274// f(start) = startMapTo
275// f(end) = endMapTo
276//
277template <typename T>
278finline void
279getScaleOffset(T start, T end, T startMapTo, T endMapTo, T *scale, T *offset)
280{
281 *scale = (endMapTo - startMapTo) / (end - start);
282 *offset = startMapTo - (start * (*scale));
283}
