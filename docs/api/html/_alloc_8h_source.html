66/// A C++-like interface into using the MemoryArena (or something compatible).
67/// Supply the type, it will be allocated with the correct number of bytes, and
68/// any parameters passed in will be used in its constructor.
69template <typename T, typename Allocator, typename... Params>
70__forceinline ArenaPointer<T> arenaAlloc(Allocator& allocator, Params&&... params)
71{
72 void* const p = allocator.alloc(sizeof(T), kMemoryAlignment);

86///
87/// @class MemoryArenaAllocator Alloc.h <scene_rdl2/render/util/Alloc.h>
88/// @brief The MemoryArenaAllocator meets the requiresments of a C++ std
89/// allocator (it can be used in std containers, etc). It does its memory
90/// allocation through the supplied Arena.
91///
92template <typename T, std::size_t alignment = kMemoryAlignment>
93class ArenaAllocator
94{
95 Arena* mArena;
96
97public:
98 typedef T value_type;
99 typedef T* pointer;
100 typedef const T* const_pointer;
101 typedef T& reference;
102 typedef const T& const_reference;
103 typedef std::size_t size_type;
104 typedef std::ptrdiff_t difference_type;
105 typedef std::true_type propagate_on_container_move_assignment;
106
107 template <typename U>
108 struct rebind { typedef ArenaAllocator<U, alignment> other; };
109
110 explicit ArenaAllocator(Arena& ma) : mArena(&ma) {}
111 ArenaAllocator(const ArenaAllocator&) = default;
112 ArenaAllocator& operator=(const ArenaAllocator&) = default;
113
114 template <typename U>
115 ArenaAllocator(const ArenaAllocator<U, alignment>& other) :
116 mArena(std::addressof(other.getArena()))
117 {
118 }
119
120 pointer address(reference x) const { return std::addressof(x); }
121 const_pointer address(const_reference x) const { return std::addressof(x); }
122
123 pointer allocate(size_type n, void* const /*hint*/ = nullptr)
124 {
125 MNRY_ASSERT(mArena);
126 MNRY_ASSERT(n * sizeof(T) <= max_size());
127 return reinterpret_cast<pointer>(mArena->alloc(n * sizeof(T), alignment));
128 }
129
130 void deallocate(pointer /*p*/, size_type /*n*/)
131 {
132 // Do nothing. Somebody has to clear out the memory arena.
133 }
134
135 size_type max_size() const noexcept
136 {
137 MNRY_ASSERT(mArena);
138 return mArena->getBlockSize();
139 }
140
141 template <typename U, typename... Args>
142 void construct(U* p, Args&&... args)
143 {
144 assert(p);
145 ::new(reinterpret_cast<void*>(p)) U(std::forward<Args>(args)...);
146 }
147
148 template <typename U>
149 void destroy(U* p)
150 {
151 if (p) {
152 p->~U();
153 }
154 }
155
156 Arena& getArena() const
157 {
158 MNRY_ASSERT(mArena);
159 return *mArena;
160 }
161};
162
163template <typename T1, typename T2, std::size_t alignment>
164bool operator==(const ArenaAllocator<T1, alignment>& lhs, const ArenaAllocator<T2, alignment>& rhs)
165{
166 return std::addressof(lhs.getArena()) == std::addressof(rhs.getArena());
167}
168
169template <typename T1, typename T2, std::size_t alignment>
170bool operator!=(const ArenaAllocator<T1, alignment>& lhs, const ArenaAllocator<T2, alignment>& rhs)
171{
172 return !(lhs == rhs);
173}
