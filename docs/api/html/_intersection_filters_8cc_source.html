18bool
19originVolumeTest(uint32_t threadIdx,
20 int volumeId, const RTCRay &ray, geom::internal::VdbVolume *vdbVolume)
21{
22 Vec3f rayOrg(ray.org_x, ray.org_y, ray.org_z);
23 Vec3f rayDir(ray.dir_x, ray.dir_y, ray.dir_z);
24 Vec3f pStart = rayOrg + rayDir * ray.tnear;
25 if (vdbVolume->isInActiveField(threadIdx, pStart, ray.time)) {
26 return true; // ray origin is inside the active volume. We consider this volume as originVolume.
27 }
28
29 //
30 // queryIntersections() returns all possible intersection (= volume interval) with active volumes.
31 // However, in some cases, volume is not active at the front-most volume interval position.
32 // Even if the front-most volume interval distance from occlusion ray origin is almost 0.0f,
33 // isInActiveField() might return false and it is pretty difficult to robustly make this point is
34 // inside volume or not when only using isInActiveField()
35 // Following logic try to decide whether occlusion ray origin is reasonably close to active volume.
36 // If ray origin is reasonably close to active volume, we consider this volumeId as originVolumeId.
37 //
38 geom::internal::VolumeRayState tmpVolState;
39 tmpVolState.resetState(ray.tfar, true);
40 bool intersectVoxel =
41 vdbVolume->queryIntersections(rayOrg, rayDir, ray.tnear, ray.time, threadIdx, volumeId, tmpVolState,
42 true);
43 if (!intersectVoxel) {
44 return false; // no shadow occlusion -> no originVolume.
45 }
46
47 geom::internal::VolumeTransition* volTransition = tmpVolState.getVolumeIntervals();
48 if (volTransition[0].mVolumeId != volumeId) {
49 return false; // probably never happen.
50 }
51
52 float t0 = volTransition[0].mTRenderSpace[0];
53 float t1 = volTransition[0].mTRenderSpace[1];
54 if (t0 < 0.0 || t1 < 0.0) {
55 return false; // probably never happen
56 }
57
58 //
59 // Using the first interval's t0, t1 (both are renderSpace distance), we can auto-calibrate the
60 // decision distance to the closest volume as its size.
61 // Currently, 10% (heuristically decided value : offsetScale) of the closest volume size
62 // (= distance of t0 and t1) is used and so far so good (based on the production scene test).
63 // We might need to control by the user in the future if needed.
64 //
65 constexpr float offsetScale = 0.1;
66 float t = t0 + (t1 - t0) * offsetScale;
67 pStart = rayOrg + rayDir * t; // New detection point for origin volume judgment.
68 // This point is offsetScale farther from the ray origin of its interval.
69 if (vdbVolume->isInActiveField(threadIdx, pStart, ray.time)) {
70 // The ray origin is reasonably close to active volume or inside volume.
71 // We can consider this volume as originVolume.
72 return true;
73 }
74 return false; // This volume is not a originVolume
75}

77void
78vdbVolumeIntervalFilter(const RTCFilterFunctionNArguments* args)
79{
80 if (args->N == 1) {
81 RTCRay* ray = (RTCRay*)args->ray;
82 RTCHit* hit = (RTCHit*)args->hit;
83 // Immediately exit if this is not a volume intersection
84 mcrt_common::IntersectContext* context =
85 (mcrt_common::IntersectContext*)args->context;
86 if (context->mRayExtension->materialID >= 0) {
87 return;
88 }
89 auto tls = (geom::internal::TLState*)context->mRayExtension->geomTls;
90 const int volumeMask = scene_rdl2::rdl2::ALL_VISIBLE <<
91 scene_rdl2::rdl2::sNumVisibilityTypes;
92 if ((tls == nullptr) || (ray->mask & volumeMask) == 0) {
93 return;
94 }
95 const geom::internal::BVHUserData* userData =
96 (const geom::internal::BVHUserData*)args->geometryUserPtr;
97 auto vdbVolume = const_cast<geom::internal::VdbVolume*>(
98 (const geom::internal::VdbVolume*)userData->mPrimitive);
99 int assignmentId = vdbVolume->getIntersectionAssignmentId(hit->primID);
100 if (assignmentId < 0) {
101 return;
102 }
103 // invalidate the intersection so this ray continues to collect other
104 // volume intersections
105 args->valid[0] = 0;
106 auto& volumeRayState = tls->mVolumeRayState;
107 const int volumeId = volumeRayState.getVolumeId(assignmentId, context->mRayExtension->volumeInstanceState);
108 bool eval = !volumeRayState.isVisited(volumeId);
109 if (ray->mask & (scene_rdl2::rdl2::SHADOW << scene_rdl2::rdl2::sNumVisibilityTypes)) {
110 // volume shadow ray
111 const geom::internal::ShadowLinking* shadowLinking = vdbVolume->getShadowLinking(assignmentId);
112 if (shadowLinking && shadowLinking->getLights().size()) {
113 //
114 // We are adjusting volume interval construction depending on the condition of this call is
115 // for light transmittance computation of volume or not (i.e. return value of
116 // volumeRayState.getEstimateInScatter()).
117 // If we are not estimateInScatter condition, we simply construct every volume interval
118 // without any special case.
119 //
120 // If we are estimateInScatter condition, we want to skip construction of volume interval
121 // if this volume does not cast shadow controlled by ShadowSet attribute.
122 // In order to judge this decision (i.e. construct volume interval or not), we need volumeId
123 // at ray origin position (= origin volumeId).
124 // If origin volumeId is equal of current volume, we have to construct volume interval in
125 // order to compute self-shadowing because ray origin volume is myself (we need self-shadowing
126 // for this volume even ShadowSet attribute says this volume does not cast).
127 // If origin volumeId is empty or not the same as current volumeId, this is a case of this
128 // volume cast shadow to ray origin volume/surface.
129 // In this case, we can safely skip this volume interval for transmittance computation if
130 // ShadowSet attribute has this light (i.e. this volume does not cast a shadow on the ray
131 // origin).
132 //
133 // Current logic sets first found origin volume as the final origin volume.
134 // This means logic potentially has a problem if 2 or more volumes overlapped at the ray
135 // origin point. In this case, we need a clever way to detect which volume should be picked
136 // as the origin volume or what should we do.
137 //
138 if (volumeRayState.getOriginVolumeId() == geom::internal::VolumeRayState::ORIGIN_VOLUME_INIT) {
139 if (volumeRayState.getEstimateInScatter()) {
140 if (originVolumeTest(tls->mThreadIdx, volumeId, *ray, vdbVolume)) {
141 volumeRayState.setOriginVolumeId(volumeId);
142 } else {
143 // not set origin volume for the next vdbVolumeIntervalFilter() call.
144 }
145 } else {
146 volumeRayState.setOriginVolumeId(geom::internal::VolumeRayState::ORIGIN_VOLUME_EMPTY);
147 }
148 }
149
150 if (volumeId != volumeRayState.getOriginVolumeId()) {
151 int rayId = RTCRayN_id((RTCRayN *)args->ray, 1, 0);
152 mcrt_common::RayExtension& rayExtension = context->mRayExtension[rayId];
153 if (!shadowLinking->
154 canCastShadow((const scene_rdl2::rdl2::Light*)rayExtension.instance0OrLight)) {
155 eval = false;
156 }
157 }
158 }
159 }
160 if (eval) {
161 Vec3f rayOrg(ray->org_x, ray->org_y, ray->org_z);
162 Vec3f rayDir(ray->dir_x, ray->dir_y, ray->dir_z);
163 bool intersectVoxel =
164 vdbVolume->queryIntersections(rayOrg, rayDir, ray->tnear, ray->time, tls->mThreadIdx,
165 volumeId, volumeRayState, false);
166 // whether the starting point of ray segment is inside volume
167 // there seems to be rare edge cases that the ray starting point
168 // is inside active voxel but the vdb VolumeRayIntersector doesn't
169 // intersect any voxel. In this case we treat the ray starting point
170 // outside of volume
171 Vec3f pStart = rayOrg + rayDir * ray->tnear;
172 if (vdbVolume->isInActiveField(tls->mThreadIdx, pStart, ray->time) &&
173 intersectVoxel) {
174 volumeRayState.turnOn(volumeId);
175 }
176 vdbVolume->initVolumeSampleInfo(
177 &volumeRayState.getVolumeSampleInfo(volumeId),
178 rayOrg, rayDir, ray->time,
179 volumeRayState.getVolumeAssignmentTable()->lookupWithAssignmentId(
180 assignmentId), volumeId);
181 volumeRayState.setVisited(volumeId);
182 }
183 } else {
184 MNRY_ASSERT_REQUIRE(false, "vectorized volume is not available yet");
185 }
186}

188void
189manifoldVolumeIntervalFilter(const RTCFilterFunctionNArguments* args)
190{
191 if (args->N == 1) {
192 RTCRay* ray = (RTCRay*)args->ray;
193 RTCHit* hit = (RTCHit*)args->hit;
194 // Immediately exit if this is not a volume intersection
195 mcrt_common::IntersectContext* context =
196 (mcrt_common::IntersectContext*)args->context;
197 if (context->mRayExtension->materialID >= 0) {
198 return;
199 }
200 auto tls = (geom::internal::TLState*)context->mRayExtension->geomTls;
201 const int volumeMask = scene_rdl2::rdl2::ALL_VISIBLE <<
202 scene_rdl2::rdl2::sNumVisibilityTypes;
203 if ((tls == nullptr) || (ray->mask & volumeMask) == 0) {
204 return;
205 }
206 const geom::internal::BVHUserData* userData =
207 (const geom::internal::BVHUserData*)args->geometryUserPtr;
208 auto primitive = (const geom::internal::NamedPrimitive*)userData->mPrimitive;
209 int assignmentId = primitive->getIntersectionAssignmentId(hit->primID);
210 if (assignmentId < 0) {
211 return;
212 }
213 // invalidate the intersection so this ray continues to collect other
214 // volume intersections
215 args->valid[0] = 0;
216 auto& volumeRayState = tls->mVolumeRayState;
217 const int volumeId = volumeRayState.getVolumeId(assignmentId, context->mRayExtension->volumeInstanceState);
218 // ray casting method to determine whether ray origin is inside
219 // this volume. If the ray hits this primitive even times,
220 // it's out of this primitive, otherwise it's in this primitive.
221 // This algorithm has the assumption that primitive with
222 // volume assignment is a closed manifold.
223 Vec3f rayDir(ray->dir_x, ray->dir_y, ray->dir_z);
224 float rayTfar = ray->tfar;
225 float dotProduct =
226 rayDir.x * hit->Ng_x +
227 rayDir.y * hit->Ng_y +
228 rayDir.z * hit->Ng_z;
229 bool isEntry = (dotProduct < 0.0f);
230
231 // Eliminate repeated entry or exit points with identical t-values, which Embree generates in abundance.
232 // Similar elimination can be achieved by switching on RTC_SCENE_FLAG_ROBUST, but we don't want to pay the
233 // cost for that flag unless we must.
234 int count = volumeRayState.getIntervalCount();
235 geom::internal::VolumeTransition* intervals = volumeRayState.getVolumeIntervals();
236 if (count > 0) {
237 if (intervals[count-1].mT == rayTfar &&
238 intervals[count-1].mVolumeId == volumeId &&
239 intervals[count-1].mIsEntry == isEntry) {
240 return;
241 }
242 }
243
244 volumeRayState.addInterval(rayTfar, volumeId, isEntry);
245 volumeRayState.toggle(volumeId);
246 Vec3f rayOrg(ray->org_x, ray->org_y, ray->org_z);
247
248 if (!volumeRayState.isVisited(volumeId)) {
249 primitive->initVolumeSampleInfo(
250 &volumeRayState.getVolumeSampleInfo(volumeId), rayOrg, rayDir, ray->time,
251 volumeRayState.getVolumeAssignmentTable()->lookupWithAssignmentId(
252 assignmentId), volumeId);
253 volumeRayState.setVisited(volumeId);
254 }
255 } else {
256 MNRY_ASSERT_REQUIRE(false, "vectorized volume is not available yet");
257 }

260void
261bssrdfTraceSetIntersectionFilter(const RTCFilterFunctionNArguments* args)
262{
263 // TODO can we make this filter only get called while firing bssrdf
264 // projection ray instead of register it to every intersection call?
265 // (it seems embree3 can do some kind of per ray filter function call)
266 if (args->N == 1) {
267 // If this is not a subsurface intersection,
268 // immediately exit filter function.
269 mcrt_common::IntersectContext* context =
270 (mcrt_common::IntersectContext*)args->context;
271 mcrt_common::RayExtension* rayExtension = context->mRayExtension;
272 if (rayExtension->materialID == -1) {
273 return;
274 }
275 auto geomTls = (geom::internal::TLState*)rayExtension->geomTls;
276 RTCRay* ray = (RTCRay*)args->ray;
277 RTCHit* hit = (RTCHit*)args->hit;
278 const geom::internal::BVHUserData* userData =
279 (const geom::internal::BVHUserData*)args->geometryUserPtr;
280 auto primitive = (const geom::internal::NamedPrimitive*)userData->mPrimitive;
281 int assignmentId = primitive->getIntersectionAssignmentId(hit->primID);
282 if (assignmentId < 0) {
283 return;
284 }
285 if (geomTls->mSubsurfaceTraceSet != nullptr) {
286 // there is a user specified trace set
287 const scene_rdl2::rdl2::TraceSet* traceSet =
288 (const scene_rdl2::rdl2::TraceSet *)geomTls->mSubsurfaceTraceSet;
289 const auto& gp = userData->mLayer->lookupGeomAndPart(assignmentId);
290 // getAssignmentId(Geometry, part) is a slow function.
291 // If there is ever a complaint that using a trace set
292 // slows down the render, this is the place to optimize.
293 if (traceSet->getAssignmentId(gp.first, gp.second) < 0) {
294 args->valid[0] = 0;
295 }
296 } else {
297 // default case: we compare material ids of subsurface ray
298 // and intersection
299 const scene_rdl2::rdl2::Material* rdl2Material =
300 userData->mLayer->lookupMaterial(assignmentId);
301 const shading::RootShader* material =
302 &rdl2Material->get<const shading::RootShader>();
303 if (rayExtension->materialID != material->getMaterialId()) {
304 args->valid[0] = 0;
305 }
306 }
307 }

310void
311backFaceCullingFilter(const RTCFilterFunctionNArguments* args)
312{
313 int* valid = args->valid;
314 unsigned int N = args->N;
315 RTCRayN* rays = args->ray;
316 RTCHitN* hits = args->hit;
317 for (unsigned int index = 0; index < N; ++index) {
318 if (valid[index] == 0) {
319 continue;
320 }
321 float dotProduct =
322 RTCRayN_dir_x(rays, N, index) * RTCHitN_Ng_x(hits, N, index) +
323 RTCRayN_dir_y(rays, N, index) * RTCHitN_Ng_y(hits, N, index) +
324 RTCRayN_dir_z(rays, N, index) * RTCHitN_Ng_z(hits, N, index);
325 // reject back facing intersection
326 if (dotProduct >= 0.0f) {
327 valid[index] = 0;
328 }
329 }

332void
333skipOcclusionFilter(const RTCFilterFunctionNArguments* args)
334{
335
336 // Although "args" points to a packet of N rays, in practice we always set N=1 (and there are parts of the code
337 // base which already rely on this assumption). So here we hardcode for that case - thus we only need to inspect
338 // valid[0] here, which we can write as *valid.
339 int* valid = args->valid;
340 if (*valid == 0) {
341 return;
342 }
343
344 // We continue to assume N=1 here and cast the various data types to their single-ray counterparts.
345 const geom::internal::BVHUserData* userData = static_cast<const geom::internal::BVHUserData*>(args->geometryUserPtr);
346 const geom::internal::NamedPrimitive* prim = static_cast<const geom::internal::NamedPrimitive*>(userData->mPrimitive);
347 RTCRay* ray = (RTCRay*)args->ray;
348 RTCHit* hit = (RTCHit*)args->hit;
349 const mcrt_common::IntersectContext* context = reinterpret_cast<const mcrt_common::IntersectContext*>(args->context);
350 mcrt_common::RayExtension& rayExtension = context->mRayExtension[ray->id];
351 int casterId = prim->getIntersectionAssignmentId(hit->primID);
352
353 // Skip self-occlusion for assigned volumes.
354 // Scattering from volumes is the only case that passes an assigmnent id != -1 to isRayOccluded(),
355 // and this will have been stored in the ray extension before calling Embree's occlusion test.
356 // By comparing to the hit id, we can skip occlusion if they match. (See MOONRAY-4130.)
357 int receiverId = rayExtension.shadowReceiverId;
358 if (receiverId == casterId) {
359 *valid = 0;
360 return;
361 }
362
363 // Shadow linking
364 // We currently don't support per instance shadow linking.
365 MNRY_ASSERT(userData->mPrimitive->getType() != geom::internal::Primitive::INSTANCE);
366 const geom::internal::ShadowLinking* shadowLinking = prim->getShadowLinking(casterId);
367 if (shadowLinking != nullptr) {
368 // Suppress shadows if this light is marked as not casting shadows from the caster geometry
369 if (!shadowLinking->canCastShadow((const scene_rdl2::rdl2::Light*)rayExtension.instance0OrLight)) {
370 *valid = 0;
371 return;
372 }
373
374 // Suppress shadows if this receiver is marked as not receiving shadows from the caster geometry
375 if (!shadowLinking->canReceiveShadow(receiverId)) {
376 *valid = 0;
377 return;
378 }
379 }
