28/**
29 * @class Xform
30 * @brief Xform, a utility library for shaders - for space transforms.
31 *
32 * Xform can be used to transform a point, normal or vector from one space
33 * to another (refer to each transform function for the list of supported spaces).
34 * It encapsulates the transformation matrices and utilizes them during shading.
35 * The intended utilization of this class is to create an instance in the shader's
36 * update() function. Doing so will set up the right transform data once.
37 *
38 * In order to use this library at sample time in ISPC, you should a pointer to the data needs
39 * to be held onto, in the ISPC shared structure of the shader. The pointer is obtained
40 * by calling getIspcXform().
41 *
42 */
43
44class Xform
45{
46public:
47 /**
48 * @brief Construct a Xform object with a custom object, camera or window
49 * @param shader Pointer to the current shader's rdl2 dso class
50 * @param customObject Override the geometry to use for object space transforms
51 * or pass nullptr to use the object space of the shading point's object
52 * @param customCamera Override the camera to use for camera space transforms
53 * or pass nullptr to use the scene's current active camera
54 * @param customWindow Override the window to use for screen space transforms
55 * or pass nullptr to use the scene's current aspect ratio to form a window
56 *
57 */
58 Xform(const scene_rdl2::rdl2::SceneObject *shader,
59 const scene_rdl2::rdl2::Node *customObject,
60 const scene_rdl2::rdl2::Camera *customCamera,
61 const std::array<float, 4> *customWindow);
62
63 /**
64 * @brief Construct a Xform object with an matrix representing
65 * an object or projector's transform
66 * @param shader Pointer to the current shader's rdl2 dso class
67 * @param objectMatrix 4x4 matrix representing and object
68 * or projection's transform
69 */
70 Xform(const scene_rdl2::rdl2::SceneObject *shader,
71 const scene_rdl2::math::Mat4d& objectMatrix);
72
73 /**
74 * @brief Construct a Xform object with default transforms
75 *
76 * This constructor uses object space of the shading point's object,
77 * scene's current active camera and scene's current aspect ratio to form a window
78 *
79 * @param shader Pointer to the current shader's rdl2 dso class
80 */
81 explicit Xform(const scene_rdl2::rdl2::SceneObject *shader);

83 DISALLOW_COPY_OR_ASSIGNMENT(Xform);
