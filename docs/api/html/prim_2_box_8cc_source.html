30Box::Box(float length,
31 float width,
32 float height,
33 LayerAssignmentId&& layerAssignmentId,
34 PrimitiveAttributeTable&& primitiveAttributeTable)
35: NamedPrimitive(std::move(layerAssignmentId))
36, mL2P(scene_rdl2::math::one)
37, mP2L(scene_rdl2::math::one)
38, mLength(scene_rdl2::math::abs(length))
39, mWidth(scene_rdl2::math::abs(width))
40, mHeight(scene_rdl2::math::abs(height))
41, mIsSingleSided(false)
42, mIsNormalReversed(false)
43{
44 MNRY_ASSERT_REQUIRE(
45 mLayerAssignmentId.getType() == LayerAssignmentId::Type::CONSTANT);
46 // facevarying/varying/vertex are the same for Box:
47 // bilinear interpolation on the intersection uv across 4 specified values
48 for (auto& kv : primitiveAttributeTable) {
49 for (auto& attribute : kv.second) {
50 if (attribute->getRate() == RATE_VARYING ||
51 attribute->getRate() == RATE_FACE_VARYING) {
52 attribute->setRate(RATE_VERTEX);
53 }
54 }
55 }
56 setAttributes(Attributes::interleave(primitiveAttributeTable,
57 0, 1, 4, std::vector<size_t>(), 4));
58}

72void
73Box::postIntersect(mcrt_common::ThreadLocalState& tls,
74 const scene_rdl2::rdl2::Layer* pRdlLayer, const mcrt_common::Ray& ray,
75 Intersection& intersection) const
76{
77 intersection.setLayerAssignments(mLayerAssignmentId.getConstId(), pRdlLayer);
78
79 const scene_rdl2::rdl2::Material* material = intersection.getMaterial();
80 const AttributeTable *table =
81 material->get<shading::RootShader>().getPrimitiveAttributeTable();
82 intersection.setTable(&tls.mArena, table);
83 intersection.setIds(ray.primID, 0, 0);
84 overrideInstanceAttrs(ray, intersection);
85
86 const Vec3f& nLocal = ray.Ng;
87 Vec3f Ng = normalize(scene_rdl2::math::transformNormal(mP2L, nLocal));
88
89 // calculate st, dpds, dpdt
90 Vec2f st;
91 Vec3f dpds;
92 Vec3f dpdt;
93
94 if (nLocal.x != 0.0) { // left or right face
95 dpds = scene_rdl2::math::transformVector(mL2P, Vec3f(0.0, mHeight, 0.0));
96 dpdt = scene_rdl2::math::transformVector(mL2P, Vec3f(0.0, 0.0, mWidth));
97 } else if (nLocal.y != 0.0) { // top or bottom face
98 dpds = scene_rdl2::math::transformVector(mL2P, Vec3f(mLength, 0.0, 0.0));
99 dpdt = scene_rdl2::math::transformVector(mL2P, Vec3f(0.0, 0.0, mWidth));
100 } else { // back or front face
101 dpds = scene_rdl2::math::transformVector(mL2P, Vec3f(mLength, 0.0, 0.0));
102 dpdt = scene_rdl2::math::transformVector(mL2P, Vec3f(0.0, mHeight, 0.0));
103 }
104 st = Vec2f(ray.u, ray.v);
105
106 if (mIsNormalReversed) {
107 Ng = -Ng;
108 }
109 intersection.setDifferentialGeometry(Ng, Ng, st, dpds, dpdt, true);
110
111 // interpolate primitive attributes
112 QuadricInterpolator interpolator(getAttributes(), ray.time, ray.u, ray.v);
113 intersection.setRequiredAttributes(interpolator);
114
115 const scene_rdl2::rdl2::Geometry* geometry = intersection.getGeometryObject();
116 MNRY_ASSERT(geometry != nullptr);
117 intersection.setEpsilonHint( geometry->getRayEpsilon() );
118
119 // wireframe AOV
120 if (table->requests(StandardAttributes::sNumPolyVertices)) {
121 intersection.setAttribute(StandardAttributes::sNumPolyVertices, 4);
122 }
123 if (table->requests(StandardAttributes::sPolyVertexType)) {
124 intersection.setAttribute(StandardAttributes::sPolyVertexType,
125 static_cast<int>(StandardAttributes::POLYVERTEX_TYPE_POLYGON));
126 }
127 // When the box is not an instancing object,
128 // mL2P stands for local space to render space, which
129 // transforms box center from origin to world space, then
130 // from world space to render space.
131 // When the box is an instancing object,
132 // mL2P stands for local space to parent space,
133 // where parent space is then instanced around and ray.ext.l2r transforms
134 // instance to render space.
135 const Mat43 transform = ray.isInstanceHit() ? mL2P * ray.ext.l2r : mL2P;
136 const Vec3f lower = getMinCorner();
137 const Vec3f upper = getMaxCorner();
138 const Vec3f verts[8] = {
139 Vec3f(lower.x, lower.y, lower.z), // 2---------3
140 Vec3f(upper.x, lower.y, lower.z), // /| /|
141 Vec3f(lower.x, upper.y, lower.z), // 6---------7 |
142 Vec3f(upper.x, upper.y, lower.z), // | | | |
143 Vec3f(lower.x, lower.y, upper.z), // | | | |
144 Vec3f(upper.x, lower.y, upper.z), // | 0-------|-1
145 Vec3f(lower.x, upper.y, upper.z), // |/ |/
146 Vec3f(upper.x, upper.y, upper.z), // 4---------5
147 };
148 const int indices[7][4] = {
149 {3, 7, 5, 1}, // right face
150 {6, 2, 0, 4}, // left face
151 {3, 2, 6, 7}, // top face
152 {4, 0, 1, 5}, // bottom face
153 {7, 6, 4, 5}, // front face
154 {3, 1, 0, 2}, // back face
155 {0, 0, 0, 0} // degenerate face used in case nLocal has a nonsensical value
156 };
157 int iFace = scene_rdl2::math::isEqual(nLocal.x, 1.0f) ? 0 :
158 scene_rdl2::math::isEqual(nLocal.x, -1.0f) ? 1 :
159 scene_rdl2::math::isEqual(nLocal.y, 1.0f) ? 2 :
160 scene_rdl2::math::isEqual(nLocal.y, -1.0f) ? 3 :
161 scene_rdl2::math::isEqual(nLocal.z, 1.0f) ? 4 :
162 scene_rdl2::math::isEqual(nLocal.z, -1.0f) ? 5 : 6;
163 for (int iVert = 0; iVert < 4; iVert++) {
164 if (table->requests(StandardAttributes::sPolyVertices[iVert])) {
165 int index = indices[iFace][iVert];
166 Vec3f v = scene_rdl2::math::transformPoint(transform, verts[index]);
167 intersection.setAttribute(StandardAttributes::sPolyVertices[iVert], v);
168 }
169 }
170
171 // motion vectors
172 if (table->requests(StandardAttributes::sMotion)) {
173 // box has no motion slices, primitive motion can only come from instancing
174 Vec3f motion(0.0f);
175 if (ray.isInstanceHit()) {
176 // Motion vectors only support a single instance level, hence we only care
177 // about ray.instance0.
178 motion = computePrimitiveMotion(intersection.getP(), nullptr, ray.getTime(),
179 static_cast<const Instance *>(ray.ext.instance0OrLight));
180 }
181 intersection.setAttribute(StandardAttributes::sMotion, motion);
182 }
183}

213static bool
214bboxTest(const Vec3f& org, const Vec3f& dir, float rayTnear, float rayTfar,
215 const Vec3f& pMin, const Vec3f& pMax,
216 bool isSingleSided, bool isNormalReversed, float& tHit)
217{
218 float t0 = scene_rdl2::math::neg_inf;
219 float t1 = scene_rdl2::math::pos_inf;
220 bool hitBBox = true;
221 // update t interval for each plane of box
222 // not that we don't need to verify whether dir[i] == 0.
223 // if it is 0, the invDir will hold an infinite value, either
224 // -inf or inf, and the rest of the algorithm still works correctly
225 // (this assumes that the architecture being used supports
226 // IEEE floating-point arithmetic)
227 for (size_t i = 0; i < 3; i++) {
228 float invDir = 1.0 / dir[i];
229 float tnear = (pMin[i] - org[i]) * invDir;
230 float tfar = (pMax[i] - org[i]) * invDir;
231 if (tnear > tfar) {
232 std::swap(tnear, tfar);
233 }
234 if (t0 > tfar || t1 < tnear) {
235 hitBBox = false;
236 break;
237 }
238 t0 = tnear > t0 ? tnear : t0;
239 t1 = tfar < t1 ? tfar : t1;
240 }
241 // compute intersection distance along ray
242 if (!hitBBox || t0 > rayTfar || t1 < rayTnear) {
243 return false;
244 }
245 tHit = t0;
246 if (t0 < rayTnear || (isSingleSided && isNormalReversed)) {
247 tHit = t1;
248 if (tHit > rayTfar || (isSingleSided && !isNormalReversed)) {
249 return false;
250 }
251 }
252 return true;
253}

255static void
256intersectFunc(const RTCIntersectFunctionNArguments* args)
257{
258 int* valid = (int*)args->valid;
259 unsigned int N = args->N;
260 RTCRayHitN* rayhit = (RTCRayHitN*)args->rayhit;
261 RTCRayN* rays = RTCRayHitN_RayN(rayhit, N);
262 RTCHitN* hits = RTCRayHitN_HitN(rayhit, N);
263 const BVHUserData* userData = (const BVHUserData*)args->geometryUserPtr;
264 const Box* box = (const Box*)userData->mPrimitive;
265 // get sidedness and reverse normals
266 bool isSingleSided = box->getIsSingleSided();
267 bool isNormalReversed = box->getIsNormalReversed();
268 Vec3f pMin = box->getMinCorner();
269 Vec3f pMax = box->getMaxCorner();
270 const Mat43& P2L = box->getP2L();
271 for (unsigned int index = 0; index < N; ++index) {
272 if (valid[index] == 0) {
273 continue;
274 }
275 // transform ray to object space
276 Vec3f org = scene_rdl2::math::transformPoint(P2L, Vec3f(
277 RTCRayN_org_x(rays, N, index),
278 RTCRayN_org_y(rays, N, index),
279 RTCRayN_org_z(rays, N, index)));
280 Vec3f dir = scene_rdl2::math::transformVector(P2L, Vec3f(
281 RTCRayN_dir_x(rays, N, index),
282 RTCRayN_dir_y(rays, N, index),
283 RTCRayN_dir_z(rays, N, index)));
284 float& rayTnear = RTCRayN_tnear(rays, N, index);
285 float& rayTfar = RTCRayN_tfar(rays, N, index);
286 float tHit;
287 if (!bboxTest(org, dir, rayTnear, rayTfar, pMin, pMax,
288 isSingleSided, isNormalReversed, tHit)) {
289 continue;
290 }
291 // compute hit position
292 Vec3f pHit = org + tHit * dir;
293 // local space normal will be transformed
294 // to parent space at postIntersect
295 if (pHit.x < pMin.x * (1 - sEpsilonEdge) ||
296 pHit.x > pMax.x * (1 - sEpsilonEdge)) { // left or right face
297 RTCHitN_Ng_x(hits, N, index) = scene_rdl2::math::sign(pHit.x);
298 RTCHitN_Ng_y(hits, N, index) = 0.0f;
299 RTCHitN_Ng_z(hits, N, index) = 0.0f;
300 RTCHitN_u(hits, N, index) = (pHit.y - pMin.y) / (pMax.y - pMin.y);
301 RTCHitN_v(hits, N, index) = (pHit.z - pMin.z) / (pMax.z - pMin.z);
302 } else if (pHit.y < pMin.y * (1 - sEpsilonEdge) ||
303 pHit.y > pMax.y * (1 - sEpsilonEdge)) { // top or bottom face
304 RTCHitN_Ng_x(hits, N, index) = 0.0f;
305 RTCHitN_Ng_y(hits, N, index) = scene_rdl2::math::sign(pHit.y);
306 RTCHitN_Ng_z(hits, N, index) = 0.0f;
307 RTCHitN_u(hits, N, index) = (pHit.x - pMin.x) / (pMax.x - pMin.x);
308 RTCHitN_v(hits, N, index) = (pHit.z - pMin.z) / (pMax.z - pMin.z);
309 } else { // front or back face
310 RTCHitN_Ng_x(hits, N, index) = 0.0f;
311 RTCHitN_Ng_y(hits, N, index) = 0.0f;
312 RTCHitN_Ng_z(hits, N, index) = scene_rdl2::math::sign(pHit.z);
313 RTCHitN_u(hits, N, index) = (pHit.x - pMin.x) / (pMax.x - pMin.x);
314 RTCHitN_v(hits, N, index) = (pHit.y - pMin.y) / (pMax.y - pMin.y);
315 }
316 // TODO call intersect filter functions
317 rayTfar = tHit;
318 RTCHitN_instID(hits, N, index, 0) = args->context->instID[0];
319 RTCHitN_geomID(hits, N, index) = box->getGeomID();
320 RTCHitN_primID(hits, N, index) = 0;
321 }
322}

330static void
331occludedFunc(const RTCOccludedFunctionNArguments* args)
332{
333 int* valid = (int*)args->valid;
334 unsigned int N = args->N;
335 RTCRayN* rays = args->ray;
336 const BVHUserData* userData = (const BVHUserData*)args->geometryUserPtr;
337 const Box* box = (const Box*)userData->mPrimitive;
338 // get sidedness and reverse normals
339 bool isSingleSided = box->getIsSingleSided();
340 bool isNormalReversed = box->getIsNormalReversed();
341 Vec3f pMin = box->getMinCorner();
342 Vec3f pMax = box->getMaxCorner();
343 const Mat43& P2L = box->getP2L();
344 for (unsigned int index = 0; index < N; ++index) {
345 if (valid[index] == 0) {
346 continue;
347 }
348 // transform ray to object space
349 Vec3f org = scene_rdl2::math::transformPoint(P2L, Vec3f(
350 RTCRayN_org_x(rays, N, index),
351 RTCRayN_org_y(rays, N, index),
352 RTCRayN_org_z(rays, N, index)));
353 Vec3f dir = scene_rdl2::math::transformVector(P2L, Vec3f(
354 RTCRayN_dir_x(rays, N, index),
355 RTCRayN_dir_y(rays, N, index),
356 RTCRayN_dir_z(rays, N, index)));
357 float& rayTnear = RTCRayN_tnear(rays, N, index);
358 float& rayTfar = RTCRayN_tfar(rays, N, index);
359 float tHit;
360 if (!bboxTest(org, dir, rayTnear, rayTfar, pMin, pMax,
361 isSingleSided, isNormalReversed, tHit)) {
362 continue;
363 }
364 // TODO call occlude filter functions
365 // mark the tfar negative is the official signal
366 // for embree that the ray is occluded
367 rayTfar = -FLT_MAX;
368 }
