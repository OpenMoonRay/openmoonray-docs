23HairBsdfLobe::HairBsdfLobe(Type type,
24 const Vec3f& hairDir,
25 const Vec2f& hairUV,
26 const float mediumIOR,
27 const float ior,
28 ispc::HairFresnelType fresnelType,
29 const float cuticleLayerThickness,
30 const float longShift,
31 const float longRoughness,
32 const float azimRoughness,
33 const Color& hairColor,
34 const Color& hairSigmaA,
35 const Color& tint,
36 const float hairRotation,
37 const Vec3f& hairNormal):
38 BsdfLobe(type,
39 DifferentialFlags(IGNORES_INCOMING_DIFFERENTIALS),
40 true,
41 PROPERTY_ROUGHNESS | PROPERTY_COLOR |
42 PROPERTY_NORMAL | PROPERTY_PBR_VALIDITY),
43 mHairDir(hairDir),
44 mHairUV(hairUV),
45 mMediumIOR(mediumIOR),
46 mIOR(ior),
47 mFresnelType(fresnelType),
48 mLongitudinalRoughness(longRoughness),
49 mAzimuthalRoughness(azimRoughness),
50 mHairColor(hairColor),
51 mSigmaA(hairSigmaA),
52 mLongitudinalShift(longShift),
53 mTint(tint),
54 mHairRotation(hairRotation),
55 mHairNormal(hairNormal)
56{
57 setIsHair(true);
58 scene_rdl2::math::sincos(longShift, &mSinAlpha, &mCosAlpha);
59
60 // Derive the longitudinal & azimuthal variance using Disney's approximation
61 mLongitudinalVariance = HairUtil::longitudinalVar(mLongitudinalRoughness);
62 mAzimuthalVariance = HairUtil::azimuthalVar(mAzimuthalRoughness);
63
64 // For Moonray, Texture 't' varies across the width of hair curve
65 // and goes from [0,1]. We remap this to [-1,1].
66 // Note - this is not true of all curve types
67 // Please take a look at MOONRAY-3116 and make sure that is fixed.
68 mH = mHairUV[1] * 2.0f - 1.0f;
69
70 // Cuticle-Layer has been parameterized to be between 0 and 1
71 // Layers vary in between [0.5, 1.5]
72 // See Table 2
74 mCuticleLayerThickness = scene_rdl2::math::lerp(0.5f, 1.5f,
75 scene_rdl2::math::clamp(
76 cuticleLayerThickness));
77
78 // Derive a directional differential scale that varies according to the
79 // smallest of longitudinal and azimuthal width
80 float minWidth = scene_rdl2::math::min(shading::roughness2Width(longRoughness),
81 shading::roughness2Width(azimRoughness));
82
83 // Derive a directional differential scale that varies according to width
84 mdDFactor = sdDFactorMin + minWidth * sdDFactorSlope;
85}

88Color
89HairBsdfLobe::eval(const BsdfSlice &slice,
90 const Vec3f &wi,
91 float *pdf) const
92{
93 if (!HairUtil::clampTest(mHairDir,
94 wi)) {
95 if (pdf) *pdf = 0.0f;
96 return sBlack;
97 }
98
99 // calculate all the relevant hair bsdf parameters
100 HairState hairState(slice.getWo(),
101 wi,
102 mHairDir,
103 mH,
104 mIOR,
105 mSigmaA,
106 mHairRotation * mHairUV.x, // use hair s coord to vary rotation from base to tip
107 mHairNormal);
108
109 Color bsdf = getScale() * mTint * evalBsdf(hairState, slice.getIncludeCosineTerm());
110
111 if (pdf) *pdf = evalPdf(hairState);
112
113 return bsdf;
114}

117// BSDF = Fresnel * Absorption * M * N * CosineTerm
118Color
119HairBsdfLobe::evalBsdf(const HairState& hairState,
120 bool includeCosineTerm) const
121{
122 // The Cosine terms cancel out since the Longitudinal Function includes
123 // a oneOverCosThetaI
124 return fresnel(hairState, hairState.cosThetaD()) *
125 evalMTerm(hairState) *
126 evalNTermWithAbsorption(hairState);
127}

129//----------------------------------------------------------------------------
130// Default Longitudinal Function
131float
132HairBsdfLobe::evalMTerm(const HairState& hairState) const
133{
134 float sinThetaICone = hairState.sinThetaI() * mCosAlpha + hairState.cosThetaI() * mSinAlpha;
135 float cosThetaICone = hairState.cosThetaI() * mCosAlpha - hairState.sinThetaI() * mSinAlpha;
136
137 // Handle out-of-range thetaI values because of the shift
138 cosThetaICone = scene_rdl2::math::abs(cosThetaICone);
139
140 // Perfect Longitudinal Function From [1]
141 return HairUtil::deonLongitudinalM(mLongitudinalVariance,
142 sinThetaICone,
143 cosThetaICone,
144 hairState.sinThetaO(),
145 hairState.cosThetaO());
146}

148// Default Azimuthal Function
149float
150HairBsdfLobe::evalNTerm(const HairState& hairState) const
151{
152 // Normalized Azimuthal Lobe - 1/4 * cos(phiD/2)
153 const float cosPhiDOver2 = max(0.0f, 0.25f * hairState.cosPhiDOverTwo());
154 return cosPhiDOver2;
155}

157Color
158HairBsdfLobe::evalNTermWithAbsorption(const HairState& hairState) const
159{
160 return evalNTerm(hairState) * absorption(hairState);

165Color
166HairBsdfLobe::sample(const BsdfSlice &slice,
167 float r1, float r2,
168 Vec3f &wi,
169 float &pdf) const
170{
171 // calculate all the relevant hair bsdf parameters based on omegaO
172 HairState hairState(slice.getWo(),
173 mHairDir,
174 mH,
175 mIOR,
176 mSigmaA,
177 mHairRotation * mHairUV.x, // use hair s coord to vary rotation from base to tip
178 mHairNormal);
179
180#ifdef PBR_HAIR_USE_UNIFORM_SAMPLING
181 wi = hairState.localToGlobal(
182 sampleSphereUniform(r1, r2));
183 return eval(slice, wi, &pdf);
184
185#else
186
187 // Sample Theta and Phi
188 float thetaI;
189 const float thetaPdf = sampleTheta(r1,
190 hairState.thetaO(),
191 thetaI);
192
193 float phiI = 0.0f;
194 const float phiPdf = samplePhi(r2,
195 hairState.phiO(),
196 phiI); // phiI is in (-pi, pi]
197 MNRY_ASSERT(phiI >= -sPi && phiI <= sPi);
198
199
200 float sinPhi, cosPhi;
201 sincos(phiI, &sinPhi, &cosPhi);
202
203 float sinTheta, cosTheta;
204 sincos(thetaI, &sinTheta, &cosTheta);
205
206 if (!HairUtil::clampTest(cosTheta)) {
207 pdf = 0.0f;
208 return sBlack;
209 }
210
211 pdf = max(0.0f, phiPdf * thetaPdf);
212
213 // Compute the light direction vector for shading.
214 // TODO: use ReferenceFrame local2global
215 const float uWgt = sinTheta;
216 const float vWgt = cosTheta * cosPhi;
217 const float wWgt = cosTheta * sinPhi;
218 wi = normalize(hairState.localToGlobal(uWgt, vWgt, wWgt));
219
220 // Update HairState based on omegaI
221 hairState.updateAngles(wi, phiI, thetaI);
222
223 // Return the shaded color.
224 return getScale() * mTint * evalBsdf(hairState, slice.getIncludeCosineTerm());
225#endif
226}

228float
229HairBsdfLobe::evalPdf(const HairState& hairState) const
230{
231#ifdef PBR_HAIR_USE_UNIFORM_SAMPLING
232 return 1.0f / sFourPi;
233#else
234
235 // This much is the same for all lobes --
236 // we need to have a parent Bsdf object, and we generally
237 // return zero probability when the light direction is pretty
238 // just about parallel to the hair direction.
239 // clamptest done at beginning of eval, done before pdf() is called
240
241 const float phiPdf = evalPhiPdf(hairState);
242 const float thetaPdf = evalThetaPdf(hairState);
243
244 const float pdf = max(0.0f, phiPdf * thetaPdf);
245 return pdf;
246#endif
247}

249float
250HairBsdfLobe::sampleTheta(float r,
251 float thetaO,
252 float& thetaI) const
253{
254 // Create Two Unique Random Numbers using a Morton Curve
255 // https://fgiesen.wordpress.com/2009/12/13/decoding-morton-codes/
256 // PBRT Section 4.3.3
257 Vec2f eps = HairUtil::demuxFloat(r);
258
259 float sinThetaO, cosThetaO;
260 sincos(thetaO,
261 &sinThetaO,
262 &cosThetaO);
263
264 eps[0] = max(eps[0], 1e-5f);
265
266 // Eugene's Derivation - [3] Section 3.2
267 // float cosTheta = mLongitudinalVariance *
268 // log( exp(rcp(mLongitudinalVariance)) - 2 * eps[0] * sinh(rcp(mLongitudinalVariance)) );
269
270 // PBRT Derivation (More Stable at low variance values)
271 float cosTheta =
272 1.0f + mLongitudinalVariance * scene_rdl2::math::log(
273 eps[0] + (1.0f - eps[0]) * scene_rdl2::math::exp(-2.0f / mLongitudinalVariance));
274
275 float sinTheta = HairUtil::safeSqrt(1.0f - sqr(cosTheta));
276 float cosPhi = scene_rdl2::math::cos(sTwoPi * eps[1]);
277 float sinThetaI = -cosTheta * sinThetaO + sinTheta * cosPhi * cosThetaO;
278 float cosThetaI = HairUtil::safeSqrt(1.0f - sqr(sinThetaI));
279
280 // Update sampled $\sin \thetai$ and $\cos \thetai$ to account for scales
281 // Note - Shift by -negative Alpha, because in eval() you shift by positive
282 float sinThetaICone = sinThetaI * mCosAlpha - cosThetaI * mSinAlpha;
283 float cosThetaICone = cosThetaI * mCosAlpha + sinThetaI * mSinAlpha;
284 // Handle out-of-range thetaI values because of the shift
285 cosThetaICone = scene_rdl2::math::abs(cosThetaICone);
286
287 // This is our sample direction. We clamp sinThetaI to [-1,1] to remove
288 // small floating point math errors, because asin() is extremely sensitive
289 // and will throw NaNs if the sine value is even slightly outside this range.
290 sinThetaI = clamp(sinThetaICone, -1.0f, 1.0f);
291 cosThetaI = cosThetaICone;
292
293 thetaI = scene_rdl2::math::asin(sinThetaI);
294
295 // Shift by +alpha to calculate the PDF, same as will happen in eval()
296 sinThetaICone = sinThetaI * mCosAlpha + cosThetaI * mSinAlpha;
297 cosThetaICone = cosThetaI * mCosAlpha - sinThetaI * mSinAlpha;
298
299 // Handle out-of-range thetaI values because of the shift
300 cosThetaICone = scene_rdl2::math::abs(cosThetaICone);
301
302 const float thetaPDF = HairUtil::deonLongitudinalM(mLongitudinalVariance,
303 sinThetaICone, cosThetaICone,
304 sinThetaO, cosThetaO);
305 return thetaPDF;
306}

308// Default Azimuthal Lobe Sampling
309float
310HairBsdfLobe::samplePhi(float randomV,
311 float phiO,
312 float& phiI) const
313{
314 float sinPhi_2 = 2.0f * randomV - 1.0f;
315 sinPhi_2 = clamp(sinPhi_2, -1.0f, 1.0f);
316 const float phi = 2.0f * scene_rdl2::math::asin(sinPhi_2);
317
318 phiI = rangeAngle(phiO + phi);
319
320 float phiPdf = scene_rdl2::math::cos(phi * 0.5f) * 0.25f;
321 return phiPdf;
322}

331Color
332HairBsdfLobe::evalHairFresnel(const HairState& hairState,
333 float cosTheta) const
334{
335 Color fresnel = scene_rdl2::math::sWhite;
336 switch(mFresnelType) {
337 case ispc::HAIR_FRESNEL_SIMPLE_LONGITUDINAL:
338 default:
339 // Simple longitudinal cosine term
340 fresnel = computeFresnel(cosTheta);
341 break;
342 case ispc::HAIR_FRESNEL_DIELECTRIC_CYLINDER:
343 // From "An Energy-Conserving Hair Reflectance Model"
344 // This curve is very similar to the Marschner Fresnel equations
345 // and slightly cheaper to evaluate without etaP & etaPP in Marschner:
346 // https://www.desmos.com/calculator/fmxsatvxi3
347 fresnel = computeFresnel(cosTheta*hairState.cosGammaO());
348 break;
349 case ispc::HAIR_FRESNEL_LAYERED_CUTICLES:
350 // From "Physically-Accurate Fur Reflectance: Modeling, Measurement and Rendering"
351 // https://www.desmos.com/calculator/fmxsatvxi3
352 fresnel = LayeredDielectricFresnel::evalFresnel(mMediumIOR,
353 hairState.eta(),
354 hairState.etaP(),
355 hairState.etaPP(),
356 hairState.cosGammaO(),
357 mCuticleLayerThickness);
358 break;
359 }
360 return fresnel;
361}

388//----------------------------------------------------------------------------
389bool
390HairBsdfLobe::getProperty(Property property,
391 float *dest) const
392{
393 bool handled = true;
394
395 switch (property)
396 {
397 case PROPERTY_ROUGHNESS:
398 {
399 *dest = mLongitudinalRoughness;
400 *(dest + 1) = mAzimuthalRoughness;
401 }
402 break;
403 case PROPERTY_COLOR:
404 {
405 const Color c = getScale() * mHairColor;
406 *dest = c.r;
407 *(dest + 1) = c.g;
408 *(dest + 2) = c.b;
409 }
410 break;
411 case PROPERTY_NORMAL:
412 {
413 *dest = mHairDir.x;
414 *(dest + 1) = mHairDir.y;
415 *(dest + 2) = mHairDir.z;
416 }
417 break;
418 case PROPERTY_PBR_VALIDITY:
419 {
420 const Fresnel* fresnel = getFresnel();
421 // Get the property from the fresnel object
422 Color res = scene_rdl2::math::sBlack;
423 if (fresnel) {
424 res = fresnel->computePbrValidity();
425 }
426 *dest = res.r;
427 *(dest + 1) = res.g;
428 *(dest + 2) = res.b;
429 }
430 break;
431 default:
432 handled = BsdfLobe::getProperty(property, dest);
433 break;
434 }
435 return handled;
436}
