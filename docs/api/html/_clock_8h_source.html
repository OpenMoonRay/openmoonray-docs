69 struct timespec end;
70 clock_gettime(mClkId, &end);
71 const int64_t sdiff = end.tv_sec - mStart.tv_sec;
72 const int64_t nsdiff = end.tv_nsec - mStart.tv_nsec;
73 *mStat += sdiff * NSPERSEC + nsdiff;
74 mStopped = true;
75 }
