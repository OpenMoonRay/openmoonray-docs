32std::string
33SrgbF2CLUT::tblGen15bit()
34{
35 auto linearToSrgb = [](float f) -> float {
36 // https://en.wikipedia.org/wiki/SRGB
37 return (f <= 0.0031308f)? (f * 12.92f): (1.055f * powf(f, 1.0f / 2.4f) - 0.055f);
38 };
39
40 std::ostringstream ostr;
41
42 int max = 32768;
43
44 ostr << "unsigned char f2c255[" << max << "] = {\n"; // You may change array name if you want
45
46 SrgbF2CLUT gLUT;
47 bool topST = true;
48 for (int tblId = 0; tblId < max; ++tblId) {
49 if (topST) {
50 ostr << " /* tblId:" << std::setw(5) << tblId << " */ ";
51 }
52
53 int exponent = (tblId >> 7) & 0xff;
54 int mantissa = (tblId & 0x7f) << 16;
55 gLUT.set(0, exponent, mantissa); // re-construct float value based on exponent and mantissa
56
57 int uc;
58 if (tblId > 32640) {
59 // tblId > 32640 : nan
60 uc = 0; // we can not compute value -> just set 0
61 } else if (tblId == 32640) {
62 // tblId = 32640 : inf
63 uc = 255;
64 } else if (tblId >= 16256) {
65 // getF() is more than 1.0
66 uc = 255;
67 } else {
68 uc = static_cast<int>(linearToSrgb(gLUT.getF()) * 255);
69 if (uc > 255) uc = 255; // just in case
70 }
71 ostr << std::setw(3) << uc;
72
73 if (tblId != (max - 1)) ostr << ',';
74
75 topST = false;
76 if (!((tblId + 1) % 16)) {
77 ostr << '\n';
78 topST = true;
79 }
80 }
81 ostr << "};";
82 return ostr.str();
83}
